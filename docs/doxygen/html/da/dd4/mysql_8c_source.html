48static bool mysql_parse_kv_field(const char *conninfo, const char *key, char *out, size_t out_cap,
49 bool *present) {
50 size_t key_len = strlen(key);
51 out[0] = '\0';
52 *present = false;
53 const char *p = conninfo;
54 while (*p) {
55 while (*p == ' ' || *p == ';' || *p == '&') p++;
56 const char *key_start = p;
57 while (*p && *p != ' ' && *p != '=' && *p != ';' && *p != '&') p++;
58 const char *key_end = p;
59 while (*p == ' ') p++;
60 if (*p != '=') {
61 while (*p && *p != ' ' && *p != ';' && *p != '&') p++;
62 continue;
63 }
64 p++;
65 while (*p == ' ') p++;
66 char quote = (*p == '\'' || *p == '"') ? *p++ : 0;
67 const char *value_start = p;
68 while (*p && (quote ? *p != quote : (*p != ' ' && *p != ';' && *p != '&'))) p++;
69 size_t value_len = (size_t)(p - value_start);
70 if (quote) {
71 if (*p != quote) return false;
72 p++;
73 }
74 while (*p == ' ') p++;
75 if (*p && *p != ';' && *p != '&') return false;
76 if ((size_t)(key_end - key_start) == key_len && strncmp(key_start, key, key_len) == 0) {
77 if (value_len >= out_cap) return false;
78 memcpy(out, value_start, value_len);
79 out[value_len] = '\0';
80 *present = true;
81 return true;
82 }
83 if (quote && *p == quote) p++;
84 while (*p && *p != ' ' && *p != ';' && *p != '&') p++;
85 }
86 return true;

89static int mysql_hex_digit(unsigned char c) {
90 if (c >= '0' && c <= '9') return c - '0';
91 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
92 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
93 return -1;

96static bool mysql_uri_component(char *out, size_t out_cap, const char *start, size_t len) {
97 size_t out_len = 0;
98 for (size_t i = 0; i < len; i++) {
99 unsigned char c = (unsigned char)start[i];
100 if (c == '%') {
101 if (i + 2 >= len) return false;
102 int hi = mysql_hex_digit((unsigned char)start[i + 1]);
103 int lo = mysql_hex_digit((unsigned char)start[i + 2]);
104 if (hi < 0 || lo < 0 || (hi == 0 && lo == 0)) return false;
105 c = (unsigned char)((hi << 4) | lo);
106 i += 2;
107 }
108 if (out_len + 1 >= out_cap) return false;
109 out[out_len++] = (char)c;
110 }
111 out[out_len] = '\0';
112 return true;
113}

115static bool mysql_parse_uint(const char *start, size_t len, unsigned int max, unsigned int *out) {
116 if (len == 0) return false;
117 unsigned int value = 0;
118 for (size_t i = 0; i < len; i++) {
119 unsigned char c = (unsigned char)start[i];
120 if (c < '0' || c > '9') return false;
121 unsigned int digit = c - '0';
122 if (value > (max - digit) / 10) return false;
123 value = value * 10 + digit;
124 }
125 if (value == 0 || value > max) return false;
126 *out = value;
127 return true;
128}
