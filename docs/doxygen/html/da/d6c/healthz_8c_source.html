130void cwist_healthz_run(cwist_health_probe_t *out_probes, size_t max_probes, size_t *out_count,
131 cwist_health_status_t *out_overall) {
132 size_t count = 0;
133 cwist_health_status_t overall = CWIST_HEALTH_OK;
134
135 for (int i = 0; i < g_entry_count; ++i) {
136 if (!g_entries[i].active) continue;
137 cwist_health_probe_t r = g_entries[i].fn(g_entries[i].ctx);
138 if (out_probes && count < max_probes) {
139 out_probes[count] = r;
140 }
141 if (!out_probes || count < max_probes) {
142 count++;
143 }
144 if (r.status == CWIST_HEALTH_FAIL)
145 overall = CWIST_HEALTH_FAIL;
146 else if (r.status == CWIST_HEALTH_DEGRADED && overall == CWIST_HEALTH_OK)
147 overall = CWIST_HEALTH_DEGRADED;
148 }
149
150 if (out_count) *out_count = count;
151 if (out_overall) *out_overall = overall;
152}
