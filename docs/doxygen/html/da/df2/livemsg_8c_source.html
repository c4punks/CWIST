47static void peer_destroy(cwist_livemsg_peer *peer) {
48 if (!peer) return;
49 cwist_livemsg_ctx *ctx = peer->ctx;
50 cwist_livemsg_peer **link = &ctx->peers;
51 while (*link && *link != peer) link = &(*link)->next;
52 if (*link) *link = peer->next;
53 if (ctx->config.on_disconnect) ctx->config.on_disconnect(peer, ctx->config.user_data);
54#ifdef CWIST_WEBRTC

71static int peer_send(cwist_livemsg_peer *peer, const void *data, size_t len, bool text) {
72 (void)text;
73 if (!peer || (len && !data)) return -EINVAL;
74 if (len > CWIST_LIVEMSG_MAX_MESSAGE_BYTES) return -EMSGSIZE;
75#ifdef CWIST_WS
76 if (peer->ws) {
77 size_t queued = cwist_websocket_async_buffered_amount(peer->ws);
78 size_t framed = len + LIVEMSG_WS_FRAME_OVERHEAD;
79 if (queued > CWIST_LIVEMSG_MAX_BUFFERED_BYTES ||
80 framed > CWIST_LIVEMSG_MAX_BUFFERED_BYTES - queued)
81 return -EAGAIN;
82 int rc = cwist_websocket_async_send(
83 peer->ws, text ? CWIST_WS_FRAME_TEXT : CWIST_WS_FRAME_BINARY, data, len);
84 if (rc == CWIST_WEBSOCKET_ASYNC_SEND_BACKPRESSURE) return -EAGAIN;
85 return rc == 0 ? 0 : -EIO;
86 }
87#endif
88#ifdef CWIST_WEBRTC
89 if (peer->rtc) {
90 size_t queued = cwist_webrtc_conn_buffered_amount(peer->rtc);
91 if (queued > CWIST_LIVEMSG_MAX_BUFFERED_BYTES ||
92 len > CWIST_LIVEMSG_MAX_BUFFERED_BYTES - queued)
93 return -EAGAIN;
94 int rc = cwist_webrtc_conn_send(peer->rtc, peer->channel, data, len,
95 text ? CWIST_WEBRTC_DATA_STRING : CWIST_WEBRTC_DATA_BINARY);
96 return rc == 0 ? 0 : -EIO;
97 }
98#endif
99 return -ENOTSUP;
100}

130int cwist_livemsg_reply_write(cwist_livemsg_reply *reply, const void *data, size_t len) {
131 if (!reply || (len && !data)) return -EINVAL;
132 if (len > CWIST_LIVEMSG_MAX_MESSAGE_BYTES - reply->len) return -EMSGSIZE;
133 if (!len) return 0;
134 size_t needed = reply->len + len;
135 if (needed > reply->cap) {
136 size_t cap = reply->cap ? reply->cap : 256;
137 while (cap < needed) cap *= 2;
138 uint8_t *next = cwist_realloc(reply->data, cap);
139 if (!next) return -ENOMEM;
140 reply->data = next;
141 reply->cap = cap;
142 }
143 memcpy(reply->data + reply->len, data, len);
144 reply->len = needed;
145 return 0;
146}

158static void dispatch(cwist_livemsg_peer *peer, const uint8_t *data, size_t len) {
159 cwist_livemsg_ctx *ctx = peer->ctx;
160 if (len > CWIST_LIVEMSG_MAX_MESSAGE_BYTES) {
161#ifdef CWIST_WS
162 if (peer->ws) cwist_websocket_async_close_code(peer->ws, 1009);
163#endif
164 return;
165 }
166 cwist_arena_t *arena = cwist_arena_create(LIVEMSG_ARENA_BYTES);
167 if (!arena) return;
168 uint8_t *input = cwist_arena_alloc(arena, len + 1);
169 if (!input) {
170 cwist_arena_destroy(arena);
171 return;
172 }
173 if (len) memcpy(input, data, len);
174 input[len] = 0;
175 peer->message_arena = arena;
176
177 cJSON *json = NULL;
178 const void *msg = input;
179 if (ctx->config.codec == CWIST_LIVEMSG_CODEC_JSON) {
180 json = cJSON_ParseWithLengthOpts((const char *)input, len + 1, NULL, true);
181 if (!json) {
182#ifdef CWIST_WS
183 if (peer->ws) cwist_websocket_async_close_code(peer->ws, 1007);
184#endif
185 goto done;
186 }
187 msg = json;
188 }
189
190 void *result = NULL;
191 if (ctx->config.process(peer, msg, len, &result, ctx->config.user_data) != CWIST_LIVEMSG_OK)
192 goto done;
193 cwist_livemsg_reply reply = {0};
194 int rc = ctx->config.respond(peer, result, &reply, ctx->config.user_data);
195 if (rc == CWIST_LIVEMSG_OK && reply.len)
196 (void)peer_send(peer, reply.data, reply.len, ctx->config.codec == CWIST_LIVEMSG_CODEC_JSON);
197 cwist_free(reply.data);
198
199done:
200 peer->message_arena = NULL;
201 if (json) cJSON_Delete(json);
202 cwist_arena_destroy(arena);
203}
204#endif

290static bool config_copy(cwist_livemsg_ctx *ctx, const cwist_livemsg_config *config) {
291 if (!config || !config->process || !config->respond) {
292 errno = EINVAL;
293 return false;
294 }
295 if ((config->transport != CWIST_LIVEMSG_TRANSPORT_WS &&
296 config->transport != CWIST_LIVEMSG_TRANSPORT_WEBRTC) ||
297 (config->codec != CWIST_LIVEMSG_CODEC_RAW && config->codec != CWIST_LIVEMSG_CODEC_JSON)) {
298 errno = ENOTSUP;
299 return false;
300 }
301 ctx->config = *config;
302 ctx->config.ws_path = cwist_strdup(config->ws_path ? config->ws_path : "/");
303 ctx->config.signaling_path =
304 config->signaling_path ? cwist_strdup(config->signaling_path) : NULL;
305 if (!ctx->config.ws_path || (config->signaling_path && !ctx->config.signaling_path)) {
306 config_free(&ctx->config);
307 return false;
308 }
309 return true;

336cwist_livemsg_ctx *cwist_livemsg_create(const cwist_livemsg_config *config) {
337 cwist_livemsg_ctx *ctx = cwist_alloc(sizeof(*ctx));
338 if (!ctx) return NULL;
339 if (!config_copy(ctx, config)) {
340 int error = errno;
341 cwist_free(ctx);
342 errno = error;
343 return NULL;
344 }
345 ctx->app = cwist_app_create();
346 if (!ctx->app) {
347 config_free(&ctx->config);
348 cwist_free(ctx);
349 return NULL;
350 }
351 bool ok = false;
352 if (config->transport == CWIST_LIVEMSG_TRANSPORT_WS) {
353#ifdef CWIST_WS
354 ok = cwist_app_internal_ws_async(ctx->app, ctx->config.ws_path, ws_on_message, ws_on_open,
355 ws_on_close, ctx, CWIST_LIVEMSG_MAX_MESSAGE_BYTES);
356#else
357 errno = ENOTSUP;
358#endif
359 } else {
360#ifdef CWIST_WEBRTC
361 ctx->rtc = cwist_webrtc_ctx_new(config->udp_port);
362 if (ctx->rtc) {
363 cwist_webrtc_ctx_set_message_handler(ctx->rtc, rtc_on_message, ctx);
364 cwist_webrtc_ctx_set_channel_handler(ctx->rtc, rtc_on_channel, ctx);
365 cwist_webrtc_ctx_set_close_handler(ctx->rtc, rtc_on_close, ctx);
366 ok = true;
367 if (ctx->config.signaling_path) {
368 cwist_error_t err = cwist_app_post_ex(ctx->app, ctx->config.signaling_path,
369 signaling_route_handler, ctx, NULL);
370 ok = err.error.err_i16 == 0;
371 }
372 }
373#else
374 errno = ENOTSUP;
375#endif
376 }
377 if (!ok) {
378 int error = errno;
379 cwist_livemsg_destroy(ctx);
380 errno = error;
381 return NULL;
382 }
383 return ctx;
384}
