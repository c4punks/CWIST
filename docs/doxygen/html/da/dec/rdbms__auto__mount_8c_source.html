38static bool probe_postgresql(int sock) {
39 char startup[64];
40 memset(startup, 0, sizeof(startup));
41
42 /* "user\0postgres\0\0" payload after version */
43 const char *user_key = "user";
44 const char *user_val = "postgres";
45 size_t payload = strlen(user_key) + 1 + strlen(user_val) + 1 + 1; /* +1 terminator */
46 uint32_t len = htonl((uint32_t)(4 + 4 + payload));
47 uint32_t version = htonl(196608); /* 3.0 */
48
49 memcpy(startup, &len, 4);
50 memcpy(startup + 4, &version, 4);
51 strcpy(startup + 8, user_key);
52 strcpy(startup + 8 + strlen(user_key) + 1, user_val);
53 /* trailing null already present from memset */
54
55 size_t total = 8 + payload;
56 if (send(sock, startup, total, MSG_NOSIGNAL) != (ssize_t)total) return false;
57
58 char resp = 0;
59 ssize_t n = recv(sock, &resp, 1, 0);
60 return (n == 1 && resp == 'R');
61}

70static cwist_rdbms_provider_t classify_mysql(const char *buf, size_t len) {
71 if (len < 5) return CWIST_RDBMS_NONE;
72
73 unsigned char protocol = (unsigned char)buf[4];
74 if (protocol != 0x0a) return CWIST_RDBMS_NONE;
75
76 /* Look for "MariaDB" in the server version string that follows */
77 for (size_t i = 5; i + 6 < len; ++i) {
78 if (memcmp(&buf[i], "MariaDB", 7) == 0) return CWIST_RDBMS_MARIADB;
79 }
80 return CWIST_RDBMS_MYSQL;
81}

96cwist_rdbms_provider_t cwist_rdbms_probe_port(int port) {
97 int sock = socket(AF_INET, SOCK_STREAM, 0);
98 if (sock < 0) {
99 CWIST_LOG_ERROR("RDBMS probe: socket creation failed: %s", strerror(errno));
100 return CWIST_RDBMS_NONE;
101 }
102
103 rdbms_no_sigpipe(sock);
104 struct timeval tv;
105 tv.tv_sec = 2;
106 tv.tv_usec = 0;
107 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
108 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
109
110 struct sockaddr_in addr;
111 memset(&addr, 0, sizeof(addr));
112 addr.sin_family = AF_INET;
113 addr.sin_port = htons((uint16_t)port);
114 inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);
115
116 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
117 CWIST_LOG_ERROR("RDBMS probe: connect to 127.0.0.1:%d failed: %s", port, strerror(errno));
118 close(sock);
119 return CWIST_RDBMS_NONE;
120 }
121
122 char buf[512];
123 ssize_t n = recv(sock, buf, sizeof(buf), 0);
124 if (n > 0) {
125 cwist_rdbms_provider_t p = classify_mysql(buf, (size_t)n);
126 if (p != CWIST_RDBMS_NONE) {
127 close(sock);
128 return p;
129 }
130 /* Data was received but did not look like MySQL; could be noise.
131 * Close and re-probe as PostgreSQL below. */
132 close(sock);
133 sock = socket(AF_INET, SOCK_STREAM, 0);
134 if (sock < 0) return CWIST_RDBMS_NONE;
135 rdbms_no_sigpipe(sock);
136 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
137 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
138 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
139 close(sock);
140 return CWIST_RDBMS_NONE;
141 }
142 } else if (n < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
143 CWIST_LOG_ERROR("RDBMS probe: recv error on port %d: %s", port, strerror(errno));
144 close(sock);
145 return CWIST_RDBMS_NONE;
146 }
147
148 /* PostgreSQL speaks only after the client sends a StartupMessage */
149 if (probe_postgresql(sock)) {
150 close(sock);
151 return CWIST_RDBMS_POSTGRES;
152 }
153
154 close(sock);
155 return CWIST_RDBMS_NONE;
156}
