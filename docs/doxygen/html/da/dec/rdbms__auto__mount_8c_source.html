97static bool probe_postgresql(int sock) {
98 char startup[64];
99 memset(startup, 0, sizeof(startup));
100
101 /* "user\0postgres\0\0" payload after version */
102 const char *user_key = "user";
103 const char *user_val = "postgres";
104 size_t payload = strlen(user_key) + 1 + strlen(user_val) + 1 + 1; /* +1 terminator */
105 uint32_t len = htonl((uint32_t)(4 + 4 + payload));
106 uint32_t version = htonl(196608); /* 3.0 */
107
108 memcpy(startup, &len, 4);
109 memcpy(startup + 4, &version, 4);
110 strcpy(startup + 8, user_key);
111 strcpy(startup + 8 + strlen(user_key) + 1, user_val);
112 /* trailing null already present from memset */
113
114 size_t total = 8 + payload;
115 if (rdbms_send_retry(sock, startup, total, MSG_NOSIGNAL) != (ssize_t)total) return false;
116
117 char resp = 0;
118 ssize_t n = rdbms_recv_retry(sock, &resp, 1, 0);
119 return (n == 1 && resp == 'R');
120}

129static cwist_rdbms_provider_t classify_mysql(const char *buf, size_t len) {
130 if (len < 5) return CWIST_RDBMS_NONE;
131
132 unsigned char protocol = (unsigned char)buf[4];
133 if (protocol != 0x0a) return CWIST_RDBMS_NONE;
134
135 /* Look for "MariaDB" in the server version string that follows */
136 for (size_t i = 5; i + 6 < len; ++i) {
137 if (memcmp(&buf[i], "MariaDB", 7) == 0) return CWIST_RDBMS_MARIADB;
138 }
139 return CWIST_RDBMS_MYSQL;
140}

155cwist_rdbms_provider_t cwist_rdbms_probe_port(int port) {
156 if (port < 1 || port > 65535) return CWIST_RDBMS_NONE;
157 int sock = socket(AF_INET, SOCK_STREAM, 0);
158 if (sock < 0) {
159 CWIST_LOG_ERROR("RDBMS probe: socket creation failed: %s", strerror(errno));
160 return CWIST_RDBMS_NONE;
161 }
162
163 rdbms_no_sigpipe(sock);
164 struct timeval tv;
165 tv.tv_sec = 2;
166 tv.tv_usec = 0;
167 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
168 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
169
170 struct sockaddr_in addr;
171 memset(&addr, 0, sizeof(addr));
172 addr.sin_family = AF_INET;
173 addr.sin_port = htons((uint16_t)port);
174 inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);
175
176 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
177 CWIST_LOG_ERROR("RDBMS probe: connect to 127.0.0.1:%d failed: %s", port, strerror(errno));
178 close(sock);
179 return CWIST_RDBMS_NONE;
180 }
181
182 char buf[512];
183 ssize_t n = rdbms_recv_retry(sock, buf, sizeof(buf), 0);
184 if (n > 0) {
185 cwist_rdbms_provider_t p = classify_mysql(buf, (size_t)n);
186 if (p != CWIST_RDBMS_NONE) {
187 close(sock);
188 return p;
189 }
190 /* Data was received but did not look like MySQL; could be noise.
191 * Close and re-probe as PostgreSQL below. */
192 close(sock);
193 sock = socket(AF_INET, SOCK_STREAM, 0);
194 if (sock < 0) return CWIST_RDBMS_NONE;
195 rdbms_no_sigpipe(sock);
196 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
197 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
198 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
199 close(sock);
200 return CWIST_RDBMS_NONE;
201 }
202 } else if (n < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
203 CWIST_LOG_ERROR("RDBMS probe: recv error on port %d: %s", port, strerror(errno));
204 close(sock);
205 return CWIST_RDBMS_NONE;
206 }
207
208 /* Try provider-specific probes */
209 if (port == 1433) {
210 if (probe_mssql(sock)) {
211 close(sock);
212 return CWIST_RDBMS_MSSQL;
213 }
214 } else if (port == 1521) {
215 if (probe_oracle(sock)) {
216 close(sock);
217 return CWIST_RDBMS_ORACLE;
218 }
219 }
220
221 if (probe_postgresql(sock)) {
222 close(sock);
223 return CWIST_RDBMS_POSTGRES;
224 }
225
226 /* Check MS-SQL or Oracle if port was generic */
227 if (port != 1433 && port != 1521) {
228 close(sock);
229 sock = socket(AF_INET, SOCK_STREAM, 0);
230 if (sock >= 0) {
231 rdbms_no_sigpipe(sock);
232 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
233 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
234 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) == 0) {
235 if (probe_mssql(sock)) {
236 close(sock);
237 return CWIST_RDBMS_MSSQL;
238 }
239 }
240 close(sock);
241 }
242 sock = socket(AF_INET, SOCK_STREAM, 0);
243 if (sock >= 0) {
244 rdbms_no_sigpipe(sock);
245 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
246 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
247 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) == 0) {
248 if (probe_oracle(sock)) {
249 close(sock);
250 return CWIST_RDBMS_ORACLE;
251 }
252 }
253 close(sock);
254 }
255 return CWIST_RDBMS_NONE;
256 }
257
258 close(sock);
259 return CWIST_RDBMS_NONE;
260}

286bool cwist_rdbms_mount_runtime(cwist_app *app, cwist_rdbms_provider_t provider, int port) {
287 if (!app || provider <= CWIST_RDBMS_NONE || provider > CWIST_RDBMS_MSSQL || port < 1 ||
288 port > 65535)
289 return false;
290
291 if (app->rdbms) {
292 CWIST_LOG_INFO("RDBMS runtime already mounted");
293 return true;
294 }
295
296 struct cwist_rdbms_runtime *rt =
297 (struct cwist_rdbms_runtime *)cwist_alloc_array(1, sizeof(*rt));
298 if (!rt) {
299 CWIST_LOG_ERROR("RDBMS mount: out of memory");
300 return false;
301 }
302 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(rt);
303
304 rt->provider = provider;
305 rt->port = port;
306 rt->ready = true;
307
308 app->rdbms = rt;
309 CWIST_LOG_INFO("RDBMS runtime mounted: provider=%d port=%d", (int)provider, port);
310 return true;
311}
