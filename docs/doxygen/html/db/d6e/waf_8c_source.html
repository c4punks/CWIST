18/* Signature scanning runs a single Aho-Corasick pass over the input instead
19 * of re-probing every rule at every byte: the automaton is built once from
20 * the pattern set (dense 128-entry transitions cover the lowered ASCII
21 * alphabet; failure edges are folded in at build time, so the hot loop is
22 * one table index per byte) and matching cost stays O(input length)
23 * regardless of how many signatures are registered. */
24typedef struct {
25 const char *text;
26 size_t length;
27} waf_signature;

60static void waf_build(void) {
61 uint16_t queue[WAF_MAX_TEXT];
62 waf_states = 1; /* state 0: root */
63
64 for (size_t rule = 0; rule < sizeof(waf_signatures) / sizeof(waf_signatures[0]); ++rule) {
65 uint16_t s = 0;
66 for (size_t i = 0; i < waf_signatures[rule].length; ++i) {
67 unsigned char c = (unsigned char)waf_signatures[rule].text[i];
68 if (waf_goto[s][c] == 0) {
69 if (waf_states > 255) return; /* uint8_t state space exhausted: fail open */
70 waf_goto[s][c] = (uint8_t)waf_states++;
71 }
72 s = waf_goto[s][c];
73 }
74 waf_out[s] = true;
75 }
76
77 /* BFS failure links, then fold them into the goto table so the scan
78 * loop never walks the fail chain per byte. */
79 size_t qh = 0, qt = 0;
80 for (unsigned c = 0; c < WAF_ALPHABET; ++c) {
81 if (waf_goto[0][c]) queue[qt++] = waf_goto[0][c];
82 }
83 while (qh < qt) {
84 uint16_t r = queue[qh++];
85 if (waf_out[waf_fail[r]]) waf_out[r] = true;
86 for (unsigned c = 0; c < WAF_ALPHABET; ++c) {
87 uint8_t s = waf_goto[r][c];
88 if (s) {
89 queue[qt++] = s;
90 uint16_t f = waf_fail[r];
91 while (f && waf_goto[f][c] == 0) f = waf_fail[f];
92 waf_fail[s] = waf_goto[f][c];
93 } else {
94 waf_goto[r][c] = waf_goto[waf_fail[r]][c];
95 }
96 }
97 }
98}

111bool cwist_waf_is_safe(const char *input, size_t length) {
112 if (!input) return true;
113 pthread_once(&waf_once, waf_build);
114 uint16_t s = 0;
115 for (size_t i = 0; i < length; ++i) {
116 unsigned char c = (unsigned char)input[i];
117 if (c == 0 || (c < 0x20U && c != '\t' && c != '\n' && c != '\r')) return false;
118 /* Bytes outside ASCII cannot be part of any signature; resetting
119 * keeps matching identical to the old per-position compare. */
120 if (c >= WAF_ALPHABET) {
121 s = 0;
122 continue;
123 }
124 s = waf_goto[s][ascii_lower(c)];
125 if (waf_out[s]) return false;
126 }
127 return true;
128}

139char *cwist_sanitize_html(const char *input) {
140 if (!input) return NULL;
141 size_t length = strlen(input), extra = 0;
142 for (size_t i = 0; i < length; ++i) {
143 switch (input[i]) {
144 case '&': extra += 4; break;
145 case '<':
146 case '>': extra += 3; break;
147 case '"': extra += 5; break;
148 case '\'': extra += 4; break;
149 default: break;
150 }
151 }
152 if (length > SIZE_MAX - extra - 1) return NULL;
153 char *output = cwist_alloc(length + extra + 1);
154 if (!output) return NULL;
155 size_t out = 0;
156 for (size_t i = 0; i < length; ++i) {
157 const char *replacement = NULL;
158 switch (input[i]) {
159 case '&': replacement = "&amp;"; break;
160 case '<': replacement = "&lt;"; break;
161 case '>': replacement = "&gt;"; break;
162 case '"': replacement = "&quot;"; break;
163 case '\'': replacement = "&#39;"; break;
164 default: break;
165 }
166 if (replacement) {
167 size_t n = strlen(replacement);
168 memcpy(output + out, replacement, n);
169 out += n;
170 } else
171 output[out++] = input[i];
172 }
173 output[out] = '\0';
174 return output;

177static int waf_hex(unsigned char c) {
178 if (c >= '0' && c <= '9') return c - '0';
179 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
180 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
181 return -1;

184static bool waf_url_safe(const cwist_sstring *input, bool plus_is_space) {
185 if (!input || !input->data) return true;
186 const char *data = input->data;
187 size_t len = input->size;
188 if (!memchr(data, '%', len) && !(plus_is_space && memchr(data, '+', len)))
189 return cwist_waf_is_safe(data, len);
190
191 char *decoded = (char *)cwist_alloc(len + 1);
192 if (!decoded) return false;
193 size_t out = 0;
194 for (size_t i = 0; i < len;) {
195 if (data[i] == '%') {
196 if (i + 2 >= len) goto invalid;
197 int hi = waf_hex((unsigned char)data[i + 1]);
198 int lo = waf_hex((unsigned char)data[i + 2]);
199 if (hi < 0 || lo < 0) goto invalid;
200 decoded[out++] = (char)((hi << 4) | lo);
201 i += 3;
202 } else {
203 decoded[out++] = plus_is_space && data[i] == '+' ? ' ' : data[i];
204 i++;
205 }
206 }
207 bool safe = cwist_waf_is_safe(decoded, out);
208 cwist_free(decoded);
209 return safe;
210
211invalid:
212 cwist_free(decoded);
213 return false;
214}

233static void waf_handler(cwist_http_request *req, cwist_http_response *res,
234 cwist_handler_func next) {
235 const char *content_type = req ? cwist_http_header_get(req->headers, "Content-Type") : NULL;
236 bool form_body = content_type &&
237 strncasecmp(content_type, "application/x-www-form-urlencoded", 33) == 0;
238 bool safe = req && waf_url_safe(req->path, false) && waf_url_safe(req->query, true) &&
239 (form_body ? waf_url_safe(req->body, true)
240 : (!req->body || cwist_waf_is_safe(req->body->data, req->body->size))) &&
241 waf_headers_safe(req->headers);
242 if (!safe) {
243 res->status_code = CWIST_HTTP_BAD_REQUEST;
244 cwist_sstring_assign(res->body, "Request rejected by WAF");
245 return;
246 }
247 next(req, res);
248}
