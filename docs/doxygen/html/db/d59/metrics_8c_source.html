112cwist_metrics_registry_t *cwist_metrics_registry(void) {
113 static cwist_metrics_registry_t *g_reg = NULL;
114 if (__builtin_expect(g_reg != NULL, 1)) return g_reg;
115
116 cwist_metrics_registry_t *alloc = cwist_alloc(sizeof(*alloc));
117 if (!alloc) return NULL;
118 memcpy(alloc->metrics, metric_defaults, sizeof(metric_defaults));
119 for (int i = 0; i < CWIST_METRIC_COUNT; ++i) {
120 atomic_init(&alloc->metrics[i].value.raw, 0);
121 atomic_init(&alloc->metrics[i].count, 0);
122 atomic_init(&alloc->metrics[i].sum, 0);
123 }
124 g_reg = alloc;
125 return g_reg;
126}

134 if (!reg) return;

138 atomic_store_explicit(&reg->metrics[i].sum, 0, memory_order_relaxed);

146/* Mirror the TLS counters owned by src/net/http/https.c into the registry,
147 * same synchronization contract as the shed counter below. */
148static void cwist_metrics_sync_tls(cwist_metrics_registry_t *reg) {
149 static const struct {
150 cwist_metric_id_t id;
151 long (*read)(void);
152 } tls_counters[] = {
153 {CWIST_METRIC_TLS_HANDSHAKES_TOTAL, cwist_https_tls_handshakes_total},
154 {CWIST_METRIC_TLS_HANDSHAKES_RESUMED, cwist_https_tls_handshakes_resumed_total},
155 {CWIST_METRIC_TLS_CONNECTIONS_ACTIVE, cwist_https_tls_connections_active},
156 {CWIST_METRIC_TLS_HANDSHAKES_TLS12, cwist_https_tls_handshakes_tls12_total},
157 {CWIST_METRIC_TLS_HANDSHAKES_TLS13, cwist_https_tls_handshakes_tls13_total},
158 {CWIST_METRIC_TLS_CIPHERS_AES128_GCM, cwist_https_tls_ciphers_aes128_gcm_total},
159 {CWIST_METRIC_TLS_CIPHERS_AES256_GCM, cwist_https_tls_ciphers_aes256_gcm_total},
160 {CWIST_METRIC_TLS_CIPHERS_CHACHA20, cwist_https_tls_ciphers_chacha20_total},
161 {CWIST_METRIC_TLS_CIPHERS_OTHER, cwist_https_tls_ciphers_other_total},
162 };
163 for (size_t i = 0; i < sizeof(tls_counters) / sizeof(tls_counters[0]); ++i) {
164 uintmax_t v = (uintmax_t)tls_counters[i].read();
165 atomic_store_explicit(&reg->metrics[tls_counters[i].id].value.raw, v * 1000,
166 memory_order_relaxed);
167 atomic_store_explicit(&reg->metrics[tls_counters[i].id].count, v, memory_order_relaxed);
168 }
169}

247uintmax_t cwist_metric_load(const cwist_metrics_registry_t *reg, cwist_metric_id_t id) {
248 if (!reg || id < 0 || id >= CWIST_METRIC_COUNT) return 0;
249 cwist_metrics_sync_tls((cwist_metrics_registry_t *)reg);
250 if (id == CWIST_METRIC_HTTP_CONTINUATION_SHED) {
251 return (uintmax_t)cwist_http_continuation_shed_count();
252 }
253 return atomic_load_explicit(&reg->metrics[id].value.raw, memory_order_acquire);
254}

286char *cwist_metrics_render_prometheus(const cwist_metrics_registry_t *reg) {
287 if (!reg) return NULL;
288
289 /* Synchronize read-only shed counter from net/http */
290 long shed = cwist_http_continuation_shed_count();
291 atomic_store_explicit(
292 &((cwist_metrics_registry_t *)reg)->metrics[CWIST_METRIC_HTTP_CONTINUATION_SHED].value.raw,
293 (uintmax_t)shed * 1000, memory_order_relaxed);
294 atomic_store_explicit(
295 &((cwist_metrics_registry_t *)reg)->metrics[CWIST_METRIC_HTTP_CONTINUATION_SHED].count,
296 (uintmax_t)shed, memory_order_relaxed);
297
298 /* Synchronize TLS counters from the HTTPS layer */
299 cwist_metrics_sync_tls((cwist_metrics_registry_t *)reg);
300
301 size_t cap = 4096;
302 char *buf = malloc(cap);
303 if (!buf) return NULL;
304 size_t len = 0;
305
306 for (int i = 0; i < CWIST_METRIC_COUNT; ++i) {
307 const cwist_metric_t *m = &reg->metrics[i];
308 uintmax_t raw = atomic_load_explicit(&m->value.raw, memory_order_acquire);
309 uintmax_t cnt = atomic_load_explicit(&m->count, memory_order_acquire);
310
311 int need = snprintf(NULL, 0,
312 "# HELP %s %s\n"
313 "# TYPE %s %s\n"
314 "%s %lu.%03lu\n"
315 "%s_count %lu\n\n",
316 m->name, m->help, m->name, type_str(m->type), m->name,
317 (unsigned long)(raw / 1000), (unsigned long)(raw % 1000), m->name,
318 (unsigned long)cnt);
319 if (need < 0) {
320 free(buf);
321 return NULL;
322 }
323
324 size_t required = (size_t)need + 1;
325 if (required > cap - len) {
326 size_t new_cap = cap;
327 while (required > new_cap - len) new_cap *= 2;
328 char *nb = realloc(buf, new_cap);
329 if (!nb) {
330 free(buf);
331 return NULL;
332 }
333 buf = nb;
334 cap = new_cap;
335 }
336
337 int written = snprintf(buf + len, cap - len,
338 "# HELP %s %s\n"
339 "# TYPE %s %s\n"
340 "%s %lu.%03lu\n"
341 "%s_count %lu\n\n",
342 m->name, m->help, m->name, type_str(m->type), m->name,
343 (unsigned long)(raw / 1000), (unsigned long)(raw % 1000), m->name,
344 (unsigned long)cnt);
345 if (written < 0 || (size_t)written >= cap - len) {
346 free(buf);
347 return NULL;
348 }
349 len += (size_t)written;
350 }
351
352 buf[len] = '\0';
353 return buf;
354}

366void cwist_metrics_serve_http(cwist_http_response *res) {
367 if (!res) return;
368 cwist_metrics_registry_t *reg = cwist_metrics_registry();
369 char *text = cwist_metrics_render_prometheus(reg);
370 if (!text) {
371 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
372 cwist_sstring_assign(res->body, "metrics render failed");
373 return;
374 }
375 res->status_code = CWIST_HTTP_OK;
376 cwist_sstring_assign(res->body, text);
377 cwist_http_header_add(&res->headers, "Content-Type",
378 "text/plain; version=0.0.4; charset=utf-8");
379 free(text);
