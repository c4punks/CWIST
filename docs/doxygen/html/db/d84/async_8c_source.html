41struct cwist_async {
42 _Atomic size_t refs; /* Completion plus retained producers/timers. */
43 _Atomic int state;
44 _Atomic bool ack; /* Dispatch path observed the handoff. */
45 pthread_t dispatch_thread;
46 /* Serializes a producer's mutation of the exchange with the dispatch
47 * ack: before the ack, dispatch still reads req/res (GC disown,
48 * keep-alive), so a producer that completes first must not touch them
49 * concurrently, and the "who sends" decision below must be atomic.
50 * Held only for short, non-blocking sections; nobody waits for the
51 * other side while holding it. */
52 pthread_mutex_t mu;
53 bool finish_on_ack; /* A completion arrived before the ack; ack sends it. Guarded by mu. */
54 cwist_http_request *req;
55 cwist_http_response *res; /* Handler's response (request arena). */
56 cwist_http_response *final_res;
57 bool final_res_owned; /* final_res came from respond_with. */
58 int client_fd;
59 bool keep_alive;
60 struct cwist_app *app;
61 cwist_reactor_t *reactor; /* NULL on the classic pool path. */
62 uint64_t reactor_gen; /* Pool generation @c reactor belongs to. */
63 bool orphaned; /* @c reactor was destroyed before completion. */
64 cwist_http_async_conn_t *conn;
65 void *https_conn; /* cwist_https_connection * on TLS path. */
66 cwist_h2_async_queue *h2_queue; /* Per-connection queue on the H2 path. */
67 uint32_t h2_stream_id;
68 cwist_reactor_post_t post;
69};

103cwist_async *cwist_async_defer(cwist_http_request *req, cwist_http_response *res) {
104 if (!req || !res || res->deferred) return NULL;
105 cwist_async *a = cwist_alloc(sizeof(*a));
106 if (!a) return NULL;
107 memset(a, 0, sizeof(*a));
108 atomic_init(&a->refs, 1);
109 atomic_init(&a->state, CWIST_ASYNC_ST_PENDING);
110 atomic_init(&a->ack, false);
111 if (pthread_mutex_init(&a->mu, NULL) != 0) {
112 cwist_free(a);
113 return NULL;
114 }
115 a->dispatch_thread = pthread_self();
116 a->req = req;
117 a->res = res;
118 a->final_res = res;
119 a->client_fd = req->client_fd;
120 a->keep_alive = req->keep_alive && res->keep_alive;
121 a->app = req->app;
122 cwist_http_async_conn_t *conn = (cwist_http_async_conn_t *)req->async_conn;
123 if (conn) {
124 a->reactor = conn->reactor;
125 a->reactor_gen = cwist_http_reactor_generation();
126 a->conn = conn;
127 }
128 if (req->https_conn) {
129 a->https_conn = req->https_conn;
130 ((cwist_https_connection *)req->https_conn)->deferred = true;
131 /* Only an HTTP/1.1 completion takes the conn itself; an h2 stream's
132 * completion goes through h2_queue and the h2 loop keeps the conn. */
133 if (!req->h2_queue) t_https_deferred = true;
134 }
135 if (req->h2_queue) {
136 /* Keep the queue alive until the completion has been enqueued, even
137 * if the connection tears down first (teardown closes the queue and
138 * releases only its own reference). */
139 a->h2_queue = cwist_h2_async_queue_acquire((cwist_h2_async_queue *)req->h2_queue);
140 a->h2_stream_id = req->stream_id;
141 }
142 a->post.cb = cwist_async_reactor_complete;
143 a->post.ctx = a;
144 /* Explicit async refs, not the creator's TLS GC sweep, own this handle. */
145 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(a);
146 cwist_http_async_disown_request(req);
147 cwist_http_async_disown_response(res);
148 res->async = a;
149 res->deferred = true;
150 cwist_async_record_pending(a, res);
151 return a;
152}

167void cwist_async_dispatch_ack(cwist_async *a) {
168 if (!a) return;
169 cwist_async_forget_handle(a);
170 /* Middleware may allocate after next() returns. No producer may mutate
171 * or destroy this graph until the acknowledgement; a producer that
172 * already completed waits on mu and left its result to send here. */
173 pthread_mutex_lock(&a->mu);
174 cwist_http_async_disown_request(a->req);
175 cwist_http_async_disown_response(a->res);
176 a->keep_alive = a->keep_alive && a->req->keep_alive && a->res->keep_alive;
177 bool finish = a->finish_on_ack;
178 atomic_store_explicit(&a->ack, true, memory_order_release);
179 pthread_mutex_unlock(&a->mu);
180 /* Without a recorded completion a foreign producer may free a from
181 * here on; with one, this thread owns sending it. */
182 if (finish) cwist_async_deliver(a);
183}

198/* Send the final response, re-arm or close the connection, then release the
199 * request/response pair and the handle itself. */
200static void cwist_async_complete(cwist_async *a) {
201 cwist_http_response *res = a->final_res;
202 bool keep = a->keep_alive && atomic_load(&g_cwist_running);
203 res->keep_alive = keep;
204
205 if (a->h2_queue) {
206 /* Plan B handoff: never write frames from a worker thread. Enqueue
207 * the finished exchange and poke the connection's wake fd; the
208 * connection thread drains the queue, HPACK-encodes, and sends.
209 * Wait for the dispatch ack first: enqueueing transfers req/res
210 * ownership, which is only legal once dispatch observed the defer. */
211 while (!atomic_load_explicit(&a->ack, memory_order_acquire)) sched_yield();
212 cwist_h2_async_queue_enqueue(a->h2_queue, a->h2_stream_id, a->req, a->final_res, a->res,
213 a->final_res_owned);
214 cwist_h2_async_queue_release(a->h2_queue);
215 cwist_async_release(a);
216 return;
217 }
218
219 if (a->https_conn) {
220 cwist_https_connection *conn = (cwist_https_connection *)a->https_conn;
221 cwist_error_t err = (a->req && a->req->method == CWIST_HTTP_HEAD)
222 ? cwist_https_send_response_head(conn, res)
223 : cwist_https_send_response(conn, res);
224 bool ok = cwist_error_is_ok(&err);
225
226 if (keep && ok && a->app && a->app->ssl_ctx && a->app->https_request_handler) {
227 conn->deferred = false;
228 https_pool_submit_conn(conn, a->app->ssl_ctx, a->app->https_request_handler, a->app);
229 } else {
230 cwist_https_close_connection(conn);
231 }
232 } else if (a->orphaned) {
233 /* The server stopped and destroyed the worker reactor this
234 * connection was parked on: nothing can send on it or re-arm it any
235 * more, so close it. */
236 cwist_http_async_close_orphan(a->client_fd, a->conn);
237 } else if (a->reactor) {
238 /* Resumable write: on a partial send the remainder is parked on a
239 * POLLOUT slot and the reactor thread is freed immediately; the
240 * parked callback performs the rearm/close when the drain ends. */
241 cwist_http_async_send_response(a->client_fd, res, a->reactor, a->conn, keep,
242 a->req && a->req->method == CWIST_HTTP_HEAD);
243 } else {
244 cwist_error_t err = (a->req && a->req->method == CWIST_HTTP_HEAD)
245 ? cwist_http_send_response_head(a->client_fd, res)
246 : cwist_http_send_response(a->client_fd, res);
247 bool ok = err.error.err_i16 == 0;
248
249 if (keep && ok && a->app) {
250 cwist_http_pool_rearm_current(a->client_fd, cwist_app_http_handler, a->app);
251 } else {
252 close(a->client_fd);
253 }
254 }
255
256 /* See the file header: never free before dispatch observed the defer. */
257 while (!atomic_load_explicit(&a->ack, memory_order_acquire)) sched_yield();
258
259 if (a->final_res_owned) cwist_http_response_destroy(res);
260 cwist_http_response_destroy(a->res);
261 cwist_http_request_destroy(a->req);
262 cwist_async_release(a);
263}

269/* Producer side, called with mu held after the response is final: detach
270 * what the producer allocated from its own GC scope, then decide who sends.
271 * Returns true if the caller must deliver (after unlocking); false if the
272 * ack has not happened yet and will deliver. */
273static bool cwist_async_publish_locked(cwist_async *a) {
274 /* Runs on the producer, before a reactor/H2 consumer can see or free
275 * newly allocated body/header/string state (also timeout and abort). */
276 cwist_http_async_disown_response(a->final_res);
277 if (!atomic_load_explicit(&a->ack, memory_order_acquire)) {
278 /* Dispatch has not unwound yet (or this is the creator completing
279 * inline): record the completion; the ack sends it. Needs no
280 * allocation, so an abort still works when scheduling failed. */
281 a->finish_on_ack = true;
282 return false;
283 }
284 return true;
285}

287/* Send the completed exchange: post it to the owning reactor, or finish
288 * inline on the classic/TLS/orphaned paths. Runs exactly once, on the
289 * producer or on the dispatch ack, never under mu. */
290static void cwist_async_deliver(cwist_async *a) {
291 if (a->reactor) {
292 if (cwist_http_reactor_post_live(a->reactor, a->reactor_gen, &a->post)) return;
293 /* The pool that owned the reactor has been destroyed (the server
294 * stopped while this exchange was pending): the reactor is freed, so
295 * close out inline without it. */
296 a->orphaned = true;
297 }
298 cwist_async_complete(a);
299}

310static void cwist_async_timeout_job(void *arg) {
311 cwist_async *a = (cwist_async *)arg;
312 if (!cwist_async_claim(a)) {
313 cwist_async_release(a); /* A real response beat the timeout. */
314 return;
315 }
316 pthread_mutex_lock(&a->mu);
317 cwist_http_response *res = a->res;
318 res->status_code = CWIST_HTTP_GATEWAY_TIMEOUT;
319 cwist_sstring_assign(res->status_text, (char *)"Gateway Timeout");
320 cwist_sstring_assign(res->body, (char *)"Gateway Timeout");
321 cwist_http_header_add(&res->headers, "Content-Type", "text/plain");
322 bool deliver = cwist_async_publish_locked(a);
323 pthread_mutex_unlock(&a->mu);
324 if (deliver) cwist_async_deliver(a);
325 cwist_async_release(a);
326}

328void cwist_async_set_timeout(cwist_async *a, uint64_t ms) {
329 if (!a || ms == 0) return;
330 pthread_once(&g_timeout_once, cwist_async_timeout_sched_init);
331 if (!g_timeout_sched) return;
332 cwist_async_retain(a);
333 if (!cwist_scheduler_schedule(g_timeout_sched, cwist_async_timeout_job, a, ms)) {
334 cwist_async_release(a);
335 }
