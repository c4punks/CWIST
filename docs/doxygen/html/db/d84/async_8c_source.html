84cwist_async *cwist_async_defer(cwist_http_request *req, cwist_http_response *res) {
85 if (!req || !res || res->deferred) return NULL;
86 cwist_async *a = cwist_alloc(sizeof(*a));
87 if (!a) return NULL;
88 memset(a, 0, sizeof(*a));
89 atomic_init(&a->refs, 1);
90 atomic_init(&a->state, CWIST_ASYNC_ST_PENDING);
91 atomic_init(&a->ack, false);
92 a->dispatch_thread = pthread_self();
93 a->req = req;
94 a->res = res;
95 a->final_res = res;
96 a->client_fd = req->client_fd;
97 a->keep_alive = req->keep_alive && res->keep_alive;
98 a->app = req->app;
99 cwist_http_async_conn_t *conn = (cwist_http_async_conn_t *)req->async_conn;
100 if (conn) {
101 a->reactor = conn->reactor;
102 a->reactor_gen = cwist_http_reactor_generation();
103 a->conn = conn;
104 }
105 if (req->https_conn) {
106 a->https_conn = req->https_conn;
107 ((cwist_https_connection *)req->https_conn)->deferred = true;
108 }
109 if (req->h2_queue) {
110 /* Keep the queue alive until the completion has been enqueued, even
111 * if the connection tears down first (teardown closes the queue and
112 * releases only its own reference). */
113 a->h2_queue = cwist_h2_async_queue_acquire((cwist_h2_async_queue *)req->h2_queue);
114 a->h2_stream_id = req->stream_id;
115 }
116 a->post.cb = cwist_async_reactor_complete;
117 a->post.ctx = a;
118 /* Explicit async refs, not the creator's TLS GC sweep, own this handle. */
119 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(a);
120 cwist_http_async_disown_request(req);
121 cwist_http_async_disown_response(res);
122 res->async = a;
123 res->deferred = true;
124 return a;
125}

138void cwist_async_dispatch_ack(cwist_async *a) {
139 if (!a) return;
140 /* Middleware may allocate after next() returns. No producer may mutate
141 * or destroy this graph until the release/acquire acknowledgement. */
142 cwist_http_async_disown_request(a->req);
143 cwist_http_async_disown_response(a->res);
144 a->keep_alive = a->keep_alive && a->req->keep_alive && a->res->keep_alive;
145 bool finish = a->finish_on_ack;
146 atomic_store_explicit(&a->ack, true, memory_order_release);
147 /* Foreign completion may free a after ack. Only a creator-claimed
148 * pending completion permits this access after publication. */
149 if (finish) cwist_async_finish(a);
150}

152static bool cwist_async_claim(cwist_async *a) {
153 int expected = CWIST_ASYNC_ST_PENDING;
154 if (!atomic_compare_exchange_strong_explicit(&a->state, &expected, CWIST_ASYNC_ST_CLAIMED,
155 memory_order_acq_rel, memory_order_acquire))
156 return false;
157 while (!atomic_load_explicit(&a->ack, memory_order_acquire)) {
158 if (pthread_equal(pthread_self(), a->dispatch_thread)) return true;
159 sched_yield();
160 }
161 return true;
162}

166/* Send the final response, re-arm or close the connection, then release the
167 * request/response pair and the handle itself. */
168static void cwist_async_complete(cwist_async *a) {
169 cwist_http_response *res = a->final_res;
170 bool keep = a->keep_alive && atomic_load(&g_cwist_running);
171 res->keep_alive = keep;
172
173 if (a->h2_queue) {
174 /* Plan B handoff: never write frames from a worker thread. Enqueue
175 * the finished exchange and poke the connection's wake fd; the
176 * connection thread drains the queue, HPACK-encodes, and sends.
177 * Wait for the dispatch ack first: enqueueing transfers req/res
178 * ownership, which is only legal once dispatch observed the defer. */
179 while (!atomic_load_explicit(&a->ack, memory_order_acquire)) sched_yield();
180 cwist_h2_async_queue_enqueue(a->h2_queue, a->h2_stream_id, a->req, a->final_res, a->res,
181 a->final_res_owned);
182 cwist_h2_async_queue_release(a->h2_queue);
183 cwist_async_release(a);
184 return;
185 }
186
187 if (a->https_conn) {
188 cwist_https_connection *conn = (cwist_https_connection *)a->https_conn;
189 cwist_error_t err = (a->req && a->req->method == CWIST_HTTP_HEAD)
190 ? cwist_https_send_response_head(conn, res)
191 : cwist_https_send_response(conn, res);
192 bool ok = cwist_error_is_ok(&err);
193
194 if (keep && ok && a->app && a->app->ssl_ctx && a->app->https_request_handler) {
195 conn->deferred = false;
196 https_pool_submit_conn(conn, a->app->ssl_ctx, a->app->https_request_handler, a->app);
197 } else {
198 cwist_https_close_connection(conn);
199 }
200 } else if (a->orphaned) {
201 /* The server stopped and destroyed the worker reactor this
202 * connection was parked on: nothing can send on it or re-arm it any
203 * more, so close it. */
204 cwist_http_async_close_orphan(a->client_fd, a->conn);
205 } else if (a->reactor) {
206 /* Resumable write: on a partial send the remainder is parked on a
207 * POLLOUT slot and the reactor thread is freed immediately; the
208 * parked callback performs the rearm/close when the drain ends. */
209 cwist_http_async_send_response(a->client_fd, res, a->reactor, a->conn, keep,
210 a->req && a->req->method == CWIST_HTTP_HEAD);
211 } else {
212 cwist_error_t err = (a->req && a->req->method == CWIST_HTTP_HEAD)
213 ? cwist_http_send_response_head(a->client_fd, res)
214 : cwist_http_send_response(a->client_fd, res);
215 bool ok = err.error.err_i16 == 0;
216
217 if (keep && ok && a->app) {
218 cwist_http_pool_rearm_current(a->client_fd, cwist_app_http_handler, a->app);
219 } else {
220 close(a->client_fd);
221 }
222 }
223
224 /* See the file header: never free before dispatch observed the defer. */
225 while (!atomic_load_explicit(&a->ack, memory_order_acquire)) sched_yield();
226
227 if (a->final_res_owned) cwist_http_response_destroy(res);
228 cwist_http_response_destroy(a->res);
229 cwist_http_request_destroy(a->req);
230 cwist_async_release(a);
231}

237static void cwist_async_finish(cwist_async *a) {
238 /* Runs on the producer, before a reactor/H2 consumer can see or free
239 * newly allocated body/header/string state (also timeout and abort). */
240 cwist_http_async_disown_response(a->final_res);
241 if (!atomic_load_explicit(&a->ack, memory_order_acquire)) {
242 /* Creator-side abort must work even if scheduling failed. Dispatch
243 * owns this completion until unwind; no allocation is required. */
244 a->finish_on_ack = true;
245 return;
246 }
247 if (a->reactor) {
248 if (cwist_http_reactor_post_live(a->reactor, a->reactor_gen, &a->post)) return;
249 /* The pool that owned the reactor has been destroyed (the server
250 * stopped while this exchange was pending): the reactor is freed, so
251 * close out inline without it. */
252 a->orphaned = true;
253 }
254 cwist_async_complete(a);
255}

266static void cwist_async_timeout_job(void *arg) {
267 cwist_async *a = (cwist_async *)arg;
268 if (!cwist_async_claim(a)) {
269 cwist_async_release(a); /* A real response beat the timeout. */
270 return;
271 }
272 cwist_http_response *res = a->res;
273 res->status_code = CWIST_HTTP_GATEWAY_TIMEOUT;
274 cwist_sstring_assign(res->status_text, (char *)"Gateway Timeout");
275 cwist_sstring_assign(res->body, (char *)"Gateway Timeout");
276 cwist_http_header_add(&res->headers, "Content-Type", "text/plain");
277 cwist_async_finish(a);
278 cwist_async_release(a);
279}
