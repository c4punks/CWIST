95/* Seed (or, after fork, re-seed) the global DRBG. Caller holds the lock. */
96static int cwist_mtls_rng_init_locked(void) {
97 pid_t pid = getpid();
98 if (cwist_mtls_rng_ready && cwist_mtls_rng_pid == pid) return 0;
99
100 if (cwist_mtls_rng_ready) {
101 /* Forked worker: the inherited DRBG state would produce streams
102 * identical to the parent's, so reseed before generating anything. */
103 int ret = mbedtls_ctr_drbg_reseed(&cwist_mtls_drbg, NULL, 0);
104 if (ret != 0) {
105 cwist_mtls_set_err(ret);
106 return -1;
107 }
108 cwist_mtls_rng_pid = pid;
109 return 0;
110 }
111
112 mbedtls_entropy_init(&cwist_mtls_entropy);
113 mbedtls_ctr_drbg_init(&cwist_mtls_drbg);
114 int ret =
115 mbedtls_ctr_drbg_seed(&cwist_mtls_drbg, mbedtls_entropy_func, &cwist_mtls_entropy, NULL, 0);
116 if (ret != 0) {
117 cwist_mtls_set_err(ret);
118 mbedtls_ctr_drbg_free(&cwist_mtls_drbg);
119 mbedtls_entropy_free(&cwist_mtls_entropy);
120 return -1;
121 }
122 cwist_mtls_rng_ready = 1;
123 cwist_mtls_rng_pid = pid;
124 return 0;
125}

320int cwist_mtls_SSL_select_next_proto(unsigned char **out, unsigned char *outlen,
321 const unsigned char *server, unsigned int server_len,
322 const unsigned char *client, unsigned int client_len) {
323 /* Server-preference scan, same contract as OpenSSL. */
324 for (unsigned int i = 0; i < server_len;) {
325 unsigned int slen = server[i++];
326 if (i + slen > server_len) break;
327 for (unsigned int j = 0; j < client_len;) {
328 unsigned int clen = client[j++];
329 if (j + clen > client_len) return 0;
330 if (clen == slen && memcmp(server + i, client + j, slen) == 0) {
331 *out = (unsigned char *)(server + i);
332 *outlen = (unsigned char)slen;
333 return OPENSSL_NPN_NEGOTIATED;
334 }
335 j += clen;
336 }
337 i += slen;
338 }
339 return 0;
340}

394SSL_CTX *cwist_mtls_SSL_CTX_new(const SSL_METHOD *method) {
395 if (!method) return NULL;
396 SSL_CTX *ctx = (SSL_CTX *)cwist_malloc(sizeof(*ctx));
397 if (ctx) memset(ctx, 0, sizeof(*ctx));
398 if (!ctx) return NULL;
399
400 ctx->is_server = method->is_server;
401 mbedtls_ssl_config_init(&ctx->conf);
402 mbedtls_x509_crt_init(&ctx->cert);
403 mbedtls_pk_init(&ctx->pkey);
404 mbedtls_x509_crt_init(&ctx->ca_chain);
405 mbedtls_ssl_cache_init(&ctx->cache);
406
407 if (mbedtls_ssl_config_defaults(
408 &ctx->conf, ctx->is_server ? MBEDTLS_SSL_IS_SERVER : MBEDTLS_SSL_IS_CLIENT,
409 MBEDTLS_SSL_TRANSPORT_STREAM, MBEDTLS_SSL_PRESET_DEFAULT) != 0) {
410 mbedtls_ssl_cache_free(&ctx->cache);
411 mbedtls_x509_crt_free(&ctx->ca_chain);
412 mbedtls_pk_free(&ctx->pkey);
413 mbedtls_x509_crt_free(&ctx->cert);
414 mbedtls_ssl_config_free(&ctx->conf);
415 cwist_free(ctx);
416 return NULL;
417 }
418
419 mbedtls_ssl_conf_rng(&ctx->conf, cwist_mtls_rng_cb, NULL);
420 /* CWIST does not do client-certificate authentication. */
421 mbedtls_ssl_conf_authmode(&ctx->conf, MBEDTLS_SSL_VERIFY_NONE);
422 if (ctx->is_server) {
423 mbedtls_ssl_conf_session_cache(&ctx->conf, &ctx->cache, mbedtls_ssl_cache_get,
424 mbedtls_ssl_cache_set);
425#if defined(MBEDTLS_SSL_PROTO_TLS1_3)
426 /* mbedTLS writes the TLS 1.3 NewSessionTicket after handshake
427 * wrap-up (a trailing write BoringSSL folds into its second flight).
428 * A client that closes right after its Finished - routine for the
429 * short-lived probes in the CWIST test suite - resets the socket and
430 * that trailing write then fails the whole handshake with
431 * ECONNRESET. Suppress server-side NST: TLS 1.3 ticket resumption
432 * is off, TLS 1.2 session-ID resumption via the server cache above
433 * keeps working. */
434 mbedtls_ssl_conf_new_session_tickets(&ctx->conf, 0);
435#endif
436 }
437 atomic_init(&ctx->refs, 1);
438 return ctx;
439}

497/* OpenSSL cipher name -> mbedTLS ciphersuite IANA id, for the fixed profile
498 * https.c applies. Unknown names are skipped by the caller. */
499static int cwist_mtls_cipher_id(const char *name, size_t len, int *is_tls13) {
500 struct entry {
501 const char *name;
502 int id;
503 int tls13;
504 };
505 static const struct entry table[] = {
506 {"ECDHE-ECDSA-AES128-GCM-SHA256", MBEDTLS_TLS_ECDHE_ECDSA_WITH_AES_128_GCM_SHA256, 0},
507 {"ECDHE-RSA-AES128-GCM-SHA256", MBEDTLS_TLS_ECDHE_RSA_WITH_AES_128_GCM_SHA256, 0},
508 {"ECDHE-ECDSA-AES256-GCM-SHA384", MBEDTLS_TLS_ECDHE_ECDSA_WITH_AES_256_GCM_SHA384, 0},
509 {"ECDHE-RSA-AES256-GCM-SHA384", MBEDTLS_TLS_ECDHE_RSA_WITH_AES_256_GCM_SHA384, 0},
510 {"ECDHE-ECDSA-CHACHA20-POLY1305", MBEDTLS_TLS_ECDHE_ECDSA_WITH_CHACHA20_POLY1305_SHA256, 0},
511 {"ECDHE-RSA-CHACHA20-POLY1305", MBEDTLS_TLS_ECDHE_RSA_WITH_CHACHA20_POLY1305_SHA256, 0},
512 {"TLS_AES_128_GCM_SHA256", MBEDTLS_TLS1_3_AES_128_GCM_SHA256, 1},
513 {"TLS_AES_256_GCM_SHA384", MBEDTLS_TLS1_3_AES_256_GCM_SHA384, 1},
514 {"TLS_CHACHA20_POLY1305_SHA256", MBEDTLS_TLS1_3_CHACHA20_POLY1305_SHA256, 1},
515 };
516 for (size_t i = 0; i < sizeof(table) / sizeof(table[0]); i++) {
517 if (strlen(table[i].name) == len && memcmp(table[i].name, name, len) == 0) {
518 *is_tls13 = table[i].tls13;
519 return table[i].id;
520 }
521 }
522 return -1;
523}

526 * TLS 1.2 picks, matching the server-preference order https.c requests). */
527static int cwist_mtls_apply_ciphersuites(SSL_CTX *ctx) {
528 size_t n = 0;
529 for (size_t i = 0; i < ctx->cs13_count && n < 30; i++) ctx->cs_applied[n++] = ctx->cs13[i];
530 for (size_t i = 0; i < ctx->cs12_count && n < 30; i++) ctx->cs_applied[n++] = ctx->cs12[i];

532 ctx->cs_applied[n] = 0;
533 mbedtls_ssl_conf_ciphersuites(&ctx->conf, ctx->cs_applied);

537static int cwist_mtls_set_cipher_list(SSL_CTX *ctx, const char *list, int want_tls13) {
538 if (!ctx || !list) return 0;
539 size_t recognized = 0;
540 const char *p = list;
541 while (*p) {
542 const char *colon = strchr(p, ':');
543 size_t len = colon ? (size_t)(colon - p) : strlen(p);
544 int is_tls13 = 0;
545 int id = cwist_mtls_cipher_id(p, len, &is_tls13);
546 if (id >= 0) {
547 if (is_tls13 && ctx->cs13_count < 8) {
548 ctx->cs13[ctx->cs13_count++] = id;
549 recognized++;
550 } else if (!is_tls13 && ctx->cs12_count < 16) {
551 ctx->cs12[ctx->cs12_count++] = id;
552 recognized++;
553 }
554 }
555 /* Unmapped names are ignored: OpenSSL would accept any name it
556 * knows, we only carry the ones CWIST's profiles use. */
557 if (!colon) break;
558 p = colon + 1;
559 }
560 if (recognized == 0) return 0; /* mirror OpenSSL's all-unknown failure */
561 return want_tls13 ? 1 : cwist_mtls_apply_ciphersuites(ctx);
562}

653void cwist_mtls_SSL_CTX_set_alpn_select_cb(SSL_CTX *ctx,
654 int (*cb)(SSL *ssl, const unsigned char **out,
655 unsigned char *outlen, const unsigned char *in,
656 unsigned int inlen, void *arg),
657 void *arg) {
658 if (!ctx || !cb) return;
659
660 /* mbedTLS has no server ALPN callback; it picks the first conf-list
661 * protocol the client offered. CWIST's callback (https.c) is a pure
662 * function of the context's HTTP/2 flag, which is final by the time the
663 * callback is installed (init options / reload), so evaluate it once
664 * here against a synthetic offer of every TCP-TLS protocol CWIST
665 * supports and install the implied server-preference list statically.
666 * ssl is NULL: CWIST's callback never dereferences it. */
667 static const unsigned char offer[] = "\x02h2\x08http/1.1";
668 static char proto_h2[] = "h2";
669 static char proto_http11[] = "http/1.1";
670 const unsigned char *sel = NULL;
671 unsigned char sel_len = 0;
672 if (cb(NULL, &sel, &sel_len, offer, (unsigned int)sizeof(offer) - 1, arg) !=
673 SSL_TLSEXT_ERR_OK ||
674 !sel || sel_len == 0) {
675 return; /* NOACK: no ALPN configured, exactly the OpenSSL outcome */
676 }
677
678 int n = 0;
679 if (sel_len == 2 && memcmp(sel, "h2", 2) == 0) {
680 ctx->alpn_list[n++] = proto_h2;
681 ctx->alpn_list[n++] = proto_http11;
682 } else if (sel_len == 8 && memcmp(sel, "http/1.1", 8) == 0) {
683 ctx->alpn_list[n++] = proto_http11;
684 } else {
685 return; /* unknown protocol: leave ALPN off rather than guess */
686 }
687 ctx->alpn_list[n] = NULL;
688 ctx->alpn_count = n;
689 mbedtls_ssl_conf_alpn_protocols(&ctx->conf, ctx->alpn_list);
690}

692int cwist_mtls_SSL_CTX_set_alpn_protos(SSL_CTX *ctx, const unsigned char *protos,
693 unsigned int protos_len) {
694 /* Client side: parse the wire-format list into a NUL-terminated string
695 * array stored on the ctx (mbedTLS does not copy the list). */
696 if (!ctx || (!protos && protos_len > 0)) return -1;
697 ctx->alpn_count = 0;
698 ctx->alpn_list[0] = NULL;
699 if (protos_len == 0) {
700 mbedtls_ssl_conf_alpn_protocols(&ctx->conf, NULL);
701 return 0;
702 }
703
704 unsigned int i = 0;
705 int n = 0;
706 while (i < protos_len) {
707 unsigned int len = protos[i++];
708 if (i + len > protos_len || len == 0 || len >= sizeof(ctx->alpn_names[0]) || n >= 2) {
709 return -1;
710 }
711 memcpy(ctx->alpn_names[n], protos + i, len);
712 ctx->alpn_names[n][len] = '\0';
713 ctx->alpn_list[n] = ctx->alpn_names[n];
714 n++;
715 i += len;
716 }
717 ctx->alpn_list[n] = NULL;
718 ctx->alpn_count = n;
719 if (mbedtls_ssl_conf_alpn_protocols(&ctx->conf, ctx->alpn_list) != 0) {
720 return -1;
721 }
722 return 0;
723}

725/* Mirrors cwist_tls_ticket_key in src/net/http/https.c: the shared session
726 * ticket key https.c generates pre-fork and stores as SSL_CTX ex_data. */
727struct cwist_mtls_ticket_key {
728 unsigned char name[16];
729 unsigned char aes_key[32];
730 unsigned char hmac_key[32];
731};
732
733void cwist_mtls_SSL_CTX_set_tlsext_ticket_key_cb(SSL_CTX *ctx,
734 int (*cb)(SSL *ssl, unsigned char *key_name,
735 unsigned char *iv, EVP_CIPHER_CTX *ectx,
736 HMAC_CTX *hctx, int enc)) {
737 (void)cb; /* the mbedTLS ticket module replaces the callback mechanism */
738 if (!ctx || !ctx->is_server) return;
739 if (ctx->options & SSL_OP_NO_TICKET) {
740 mbedtls_ssl_conf_session_tickets(&ctx->conf, MBEDTLS_SSL_SESSION_TICKETS_DISABLED);
741 return;
742 }
743
744 /* The shared key is always installed as ex_data before this call (both
745 * the init and the reload path in https.c). Using it - rather than the
746 * random key mbedtls_ssl_ticket_setup() would mint - keeps tickets
747 * resumable across certificate hot reloads; all forked workers inherit
748 * the same material because the parent builds the context pre-fork. */
749 const struct cwist_mtls_ticket_key *key = NULL;
750 for (int i = 0; i < CWIST_MTLS_EX_DATA_MAX; i++) {
751 if (ctx->ex_data[i]) {
752 key = (const struct cwist_mtls_ticket_key *)ctx->ex_data[i];
753 break;
754 }
755 }
756 if (!key) return; /* no shared key: leave tickets unconfigured */
757
758 mbedtls_ssl_ticket_context *t = (mbedtls_ssl_ticket_context *)cwist_malloc(sizeof(*t));
759 if (t) memset(t, 0, sizeof(*t));
760 if (!t) return;
761 mbedtls_ssl_ticket_init(t);
762 uint32_t lifetime = ctx->timeout_sec > 0 ? (uint32_t)ctx->timeout_sec : 300;
763 int ret =
764 mbedtls_ssl_ticket_setup(t, cwist_mtls_rng_cb, NULL, MBEDTLS_CIPHER_AES_256_GCM, lifetime);
765 if (ret != 0) {
766 cwist_mtls_set_err(ret);
767 mbedtls_ssl_ticket_free(t);
768 cwist_free(t);
769 return;
770 }
771
772 /* Re-key slot 0 deterministically: same key name and AES-256-GCM key on
773 * every worker and every reload, so a ticket issued anywhere decrypts
774 * everywhere (mirrors the BoringSSL STEK callback). */
775 mbedtls_cipher_free(&t->MBEDTLS_PRIVATE(keys)[0].MBEDTLS_PRIVATE(ctx));
776 mbedtls_cipher_init(&t->MBEDTLS_PRIVATE(keys)[0].MBEDTLS_PRIVATE(ctx));
777 ret = mbedtls_cipher_setup(&t->MBEDTLS_PRIVATE(keys)[0].MBEDTLS_PRIVATE(ctx),
778 mbedtls_cipher_info_from_type(MBEDTLS_CIPHER_AES_256_GCM));
779 if (ret == 0) {
780 ret = mbedtls_cipher_setkey(&t->MBEDTLS_PRIVATE(keys)[0].MBEDTLS_PRIVATE(ctx), key->aes_key,
781 256, MBEDTLS_ENCRYPT);
782 }
783 if (ret != 0) {
784 cwist_mtls_set_err(ret);
785 mbedtls_ssl_ticket_free(t);
786 cwist_free(t);
787 return;
788 }
789 memcpy(t->MBEDTLS_PRIVATE(keys)[0].MBEDTLS_PRIVATE(name), key->name,
790 MBEDTLS_SSL_TICKET_KEY_NAME_BYTES);
791 t->MBEDTLS_PRIVATE(keys)[0].MBEDTLS_PRIVATE(lifetime) = lifetime;
792 t->MBEDTLS_PRIVATE(active) = 0;
793
794 mbedtls_ssl_conf_session_tickets_cb(&ctx->conf, mbedtls_ssl_ticket_write,
795 mbedtls_ssl_ticket_parse, t);
796 if (ctx->ticket) {
797 mbedtls_ssl_ticket_free(ctx->ticket);
798 cwist_free(ctx->ticket);
799 }
800 ctx->ticket = t;
801}

913/* Map one mbedtls_ssl_* return code to the OpenSSL (ret, SSL_get_error)
914 * convention: 1 on success, 0 on clean close, -1 otherwise. */
915static int cwist_mtls_map_result(SSL *ssl, int ret, int ok_value) {
916 if (ret == ok_value) {
917 ssl->last_err = SSL_ERROR_NONE;
918 return 1;
919 }
920 switch (ret) {
921 case MBEDTLS_ERR_SSL_WANT_READ: ssl->last_err = SSL_ERROR_WANT_READ; return -1;
922 case MBEDTLS_ERR_SSL_WANT_WRITE: ssl->last_err = SSL_ERROR_WANT_WRITE; return -1;
923 case MBEDTLS_ERR_SSL_PEER_CLOSE_NOTIFY: ssl->last_err = SSL_ERROR_ZERO_RETURN; return 0;
924 default:
925 cwist_mtls_set_err(ret);
926 ssl->last_err = SSL_ERROR_SSL;
927 return -1;
928 }
929}
