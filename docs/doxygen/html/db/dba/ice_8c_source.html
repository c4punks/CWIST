41static void crc32_init(void) {
42 for (uint32_t i = 0; i < 256; i++) {
43 uint32_t c = i;
44 for (int k = 0; k < 8; k++) c = (c & 1) ? 0xEDB88320u ^ (c >> 1) : c >> 1;
45 crc32_table[i] = c;
46 }

56static uint32_t crc32_update(uint32_t c, const uint8_t *data, size_t len) {
57 pthread_once(&crc32_once, crc32_init);
58 for (size_t i = 0; i < len; i++) c = crc32_table[(c ^ data[i]) & 0xFF] ^ (c >> 8);
59 return c;
60}

135static bool walk_attrs(const uint8_t *msg, size_t len, attr_cb cb, void *arg) {
136 uint16_t msg_len = rd16(msg + 2);
137 if ((size_t)msg_len + STUN_HDR_LEN > len || msg_len % 4 != 0) return false;
138 size_t off = STUN_HDR_LEN;
139 size_t end = STUN_HDR_LEN + msg_len;
140 while (off + 4 <= end) {
141 uint16_t type = rd16(msg + off);
142 uint16_t alen = rd16(msg + off + 2);
143 off += 4;
144 if (off + alen > end) return false;
145 if (!cb(type, msg + off, alen, arg)) return false;
146 off += (alen + 3) & ~3u;
147 }
148 return off == end;
149}

226static int stun_hmac_sha1(const uint8_t *hdr, const uint8_t *body, size_t body_len, const char *pwd,
227 uint8_t out[20]) {
228 HMAC_CTX *ctx = HMAC_CTX_new();
229 if (!ctx) return -1;
230 unsigned int out_len = 0;
231 int ok = HMAC_Init_ex(ctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL) == 1 &&
232 HMAC_Update(ctx, hdr, STUN_HDR_LEN) == 1 && HMAC_Update(ctx, body, body_len) == 1 &&
233 HMAC_Final(ctx, out, &out_len) == 1 && out_len == 20;
234 HMAC_CTX_free(ctx);
235 return ok ? 0 : -1;
236}

248int cwist_ice_stun_validate_request(const uint8_t *buf, size_t len, const char *pwd) {
249 uint8_t hmac[EVP_MAX_MD_SIZE];
250 size_t mi_off;
251 if (!cwist_ice_stun_is_message(buf, len)) return 0;
252 if (!find_mi(buf, len, &mi_off)) return 0;
253 /* MI covers the message up to and including the MI attribute header, with
254 * the message length field set to end at the MI attribute. */
255 uint8_t hdr[STUN_HDR_LEN];
256 memcpy(hdr, buf, STUN_HDR_LEN);
257 wr16(hdr + 2, (uint16_t)(mi_off - STUN_HDR_LEN + 4 + 20));
258 return stun_hmac_sha1(hdr, buf + STUN_HDR_LEN, mi_off - STUN_HDR_LEN, pwd, hmac) == 0 &&
259 memcmp(hmac, buf + mi_off + 4, 20) == 0;
260}

277int cwist_ice_stun_build_response(uint8_t *out, size_t cap, const uint8_t *req, size_t req_len,
278 const struct sockaddr_in *mapped, const char *pwd) {
279 if (!req || req_len < STUN_HDR_LEN) return -1;
280
281 uint8_t body[64];
282 size_t blen = 0;
283
284 /* XOR-MAPPED-ADDRESS, IPv4 */
285 wr16(body + blen, ATTR_XOR_MAPPED_ADDRESS);
286 wr16(body + blen + 2, 8);
287 blen += 4;
288 body[blen++] = 0x00;
289 body[blen++] = 0x01; /* family IPv4 */
290 wr16(body + blen, (uint16_t)(ntohs(mapped->sin_port) ^ 0xFFFF));
291 blen += 2;
292 uint32_t addr = ntohl(mapped->sin_addr.s_addr) ^ STUN_MAGIC;
293 wr32(body + blen, addr);
294 blen += 4;
295 while (blen % 4) body[blen++] = 0;
296
297 /* MESSAGE-INTEGRITY over header+body+MI attr header with adjusted length */
298 uint16_t len_for_mi = (uint16_t)(blen + 4 + 20);
299 uint8_t hdr[STUN_HDR_LEN];
300 memset(hdr, 0, STUN_HDR_LEN);
301 wr16(hdr, CWIST_STUN_BINDING_RESPONSE);
302 wr16(hdr + 2, len_for_mi);
303 wr32(hdr + 4, STUN_MAGIC);
304 memcpy(hdr + 8, req + 8, 12);
305 uint8_t mi[20];
306 if (stun_hmac_sha1(hdr, body, blen, pwd, mi) != 0) return -1;
307
308 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
309 wr16(body + blen + 2, 20);
310 blen += 4;
311 memcpy(body + blen, mi, 20);
312 blen += 20;
313
314 /* FINGERPRINT */
315 uint16_t len_for_fp = (uint16_t)(blen + 8);
316 uint8_t hdr2[STUN_HDR_LEN];
317 memcpy(hdr2, hdr, STUN_HDR_LEN);
318 wr16(hdr2 + 2, len_for_fp);
319 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
320 c = crc32_update(c, body, blen);
321 c ^= 0xFFFFFFFFu;
322 wr16(body + blen, ATTR_FINGERPRINT);
323 wr16(body + blen + 2, 4);
324 wr32(body + blen + 4, c ^ 0x5354554Eu);
325 blen += 8;
326
327 if (cap < STUN_HDR_LEN + blen) return -1;
328 memcpy(out, hdr2, STUN_HDR_LEN);
329 memcpy(out + STUN_HDR_LEN, body, blen);
330 return (int)(STUN_HDR_LEN + blen);
331}

348int cwist_ice_stun_build_request(uint8_t *out, size_t cap, const uint8_t txid[12],
349 const char *username) {
350 size_t ulen = strlen(username);
351 uint8_t body[544];
352 if (ulen > UINT16_MAX) return -1;
353 size_t username_attr_len = ulen ? 4 + ((ulen + 3) & ~3u) : 0;
354 const size_t fixed_attrs_len = 12 + 4 + 8;
355 if (username_attr_len > sizeof(body) - fixed_attrs_len ||
356 cap < STUN_HDR_LEN + username_attr_len + fixed_attrs_len)
357 return -1;
358
359 memset(out, 0, STUN_HDR_LEN);
360 wr16(out, CWIST_STUN_BINDING_REQUEST);
361 wr32(out + 4, STUN_MAGIC);
362 memcpy(out + 8, txid, 12);
363
364 size_t blen = 0;
365 /* SOFTWARE-ish placeholder skipped; USERNAME */
366 if (ulen > 0) {
367 wr16(body + blen, ATTR_USERNAME);
368 wr16(body + blen + 2, (uint16_t)ulen);
369 blen += 4;
370 memcpy(body + blen, username, ulen);
371 blen += ulen;
372 while (blen % 4) body[blen++] = 0;
373 }
374 /* ICE-CONTROLLING (8 bytes) */
375 wr16(body + blen, ATTR_ICE_CONTROLLING);
376 wr16(body + blen + 2, 8);
377 memset(body + blen + 4, 0, 8);
378 blen += 12;
379 /* USE-CANDIDATE */
380 wr16(body + blen, ATTR_USE_CANDIDATE);
381 wr16(body + blen + 2, 0);
382 blen += 4;
383 /* PRIORITY */
384 wr16(body + blen, ATTR_PRIORITY);
385 wr16(body + blen + 2, 4);
386 wr32(body + blen + 4, 0x7EFFFFFFu);
387 blen += 8;
388
389 /* MESSAGE-INTEGRITY + FINGERPRINT added by the caller via _finalize; here we
390 * keep the request unsigned because the MI key (peer pwd) is supplied at
391 * send time. cwist_ice_stun_sign_request() handles it. */
392 wr16(out + 2, (uint16_t)blen);
393 if (cap < STUN_HDR_LEN + blen) return -1;
394 memcpy(out + STUN_HDR_LEN, body, blen);
395 return (int)(STUN_HDR_LEN + blen);
396}

502int cwist_ice_stun_parse_response(const uint8_t *buf, size_t len, const uint8_t txid[12],
503 const char *pwd, struct sockaddr_in *mapped) {
504 if (!cwist_ice_stun_is_message(buf, len) || rd16(buf) != CWIST_STUN_BINDING_RESPONSE) return 0;
505 if (memcmp(buf + 8, txid, 12) != 0) return 0;
506 if (!cwist_ice_stun_validate_request(buf, len, pwd)) return 0;
507 parse_resp_arg a = {0};
508 if (!walk_attrs(buf, len, parse_resp_cb, &a)) return 0;
509 if (!a.mi_val) return 0;
510 if (mapped) {
511 if (!a.has_mapped) return 0;
512 *mapped = a.mapped;
513 }
514 return 1;

535int cwist_ice_stun_get_remote_ufrag(const uint8_t *buf, size_t len, char *out, size_t cap) {
536 if (!out || cap == 0 || !cwist_ice_stun_is_message(buf, len)) return -1;
537 uint16_t msg_len = rd16(buf + 2);
538 if (msg_len % 4 != 0 || msg_len > len - STUN_HDR_LEN) return -1;
539 size_t end = STUN_HDR_LEN + msg_len;
540 size_t off = STUN_HDR_LEN;
541 while (off + 4 <= end) {
542 uint16_t type = rd16(buf + off);
543 uint16_t alen = rd16(buf + off + 2);
544 if (type == ATTR_USERNAME) {
545 if (off + 4 + alen > end) return -1;
546 const uint8_t *name = buf + off + 4;
547 size_t colon = 0;
548 while (colon < alen && name[colon] != ':') colon++;
549 if (colon == alen) return -1; /* no ':' separator */
550 size_t i = 0;
551 for (size_t j = colon + 1; j < alen && i + 1 < cap; j++) out[i++] = (char)name[j];
552 out[i] = '\0';
553 return 0;
554 }
555 off += 4 + ((alen + 3) & ~3u);
556 }
557 return -1;
558}

571void cwist_ice_random_creds(char *ufrag, size_t ufrag_cap, char *pwd, size_t pwd_cap) {
572 uint8_t raw[24];
573 if (ufrag && ufrag_cap > 0) ufrag[0] = '\0';
574 if (pwd && pwd_cap > 0) pwd[0] = '\0';
575 if (!ufrag || !pwd || ufrag_cap == 0 || pwd_cap == 0 || RAND_bytes(raw, sizeof(raw)) != 1)
576 return;
577
578 size_t n = ufrag_cap - 1 < 8 ? ufrag_cap - 1 : 8;
579 for (size_t i = 0; i < n; i++) ufrag[i] = b64url_chars[raw[i] & 63];
580 ufrag[n] = '\0';
581 size_t m = pwd_cap - 1 < 32 ? pwd_cap - 1 : 32;
582 for (size_t i = 0; i < m; i++) pwd[i] = b64url_chars[raw[8 + (i % 16)] & 63];
583 pwd[m] = '\0';
