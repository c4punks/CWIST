42static void crc32_init(void) {
43 if (crc32_ready)
44 return;
45 for (uint32_t i = 0; i < 256; i++) {
46 uint32_t c = i;
47 for (int k = 0; k < 8; k++)
48 c = (c & 1) ? 0xEDB88320u ^ (c >> 1) : c >> 1;
49 crc32_table[i] = c;
50 }
51 crc32_ready = true;
52}

61static uint32_t crc32_update(uint32_t c, const uint8_t *data, size_t len) {
62 crc32_init();
63 for (size_t i = 0; i < len; i++)
64 c = crc32_table[(c ^ data[i]) & 0xFF] ^ (c >> 8);
65 return c;

141static bool walk_attrs(const uint8_t *msg, size_t len, attr_cb cb, void *arg) {
142 uint16_t msg_len = rd16(msg + 2);
143 if ((size_t)msg_len + STUN_HDR_LEN > len || msg_len % 4 != 0)
144 return false;
145 size_t off = STUN_HDR_LEN;
146 size_t end = STUN_HDR_LEN + msg_len;
147 while (off + 4 <= end) {
148 uint16_t type = rd16(msg + off);
149 uint16_t alen = rd16(msg + off + 2);
150 off += 4;
151 if (off + alen > end)
152 return false;
153 if (!cb(type, msg + off, alen, arg))
154 return false;
155 off += (alen + 3) & ~3u;
156 }
157 return off == end;
158}

241int cwist_ice_stun_validate_request(const uint8_t *buf, size_t len, const char *pwd) {
242 size_t pwd_len = strlen(pwd);
243 uint8_t hmac[EVP_MAX_MD_SIZE];
244 unsigned int hmac_len = 0;
245 size_t mi_off;
246 if (!cwist_ice_stun_is_message(buf, len))
247 return 0;
248 if (!find_mi(buf, len, &mi_off))
249 return 1; /* no MI: accept (RFC 8445 requires it, but stay permissive at MVP) */
250 /* MI covers the message up to and including the MI attribute header, with
251 * the message length field set to end at the MI attribute. */
252 uint8_t hdr[STUN_HDR_LEN];
253 memcpy(hdr, buf, STUN_HDR_LEN);
254 wr16(hdr + 2, (uint16_t)(mi_off - STUN_HDR_LEN + 4 + 20));
255 HMAC_CTX *ctx = HMAC_CTX_new();
256 if (!ctx)
257 return 0;
258 HMAC_Init_ex(ctx, pwd, (int)pwd_len, EVP_sha1(), NULL);
259 HMAC_Update(ctx, hdr, STUN_HDR_LEN);
260 HMAC_Update(ctx, buf + STUN_HDR_LEN, mi_off - STUN_HDR_LEN);
261 HMAC_Final(ctx, hmac, &hmac_len);
262 HMAC_CTX_free(ctx);
263 return hmac_len == 20 && memcmp(hmac, buf + mi_off + 4, 20) == 0;
264}

281int cwist_ice_stun_build_response(uint8_t *out, size_t cap, const uint8_t *req, size_t req_len,
282 const struct sockaddr_in *mapped, const char *pwd) {
283 (void)req_len;
284 uint8_t body[64];
285 size_t blen = 0;
286
287 /* XOR-MAPPED-ADDRESS, IPv4 */
288 wr16(body + blen, ATTR_XOR_MAPPED_ADDRESS);
289 wr16(body + blen + 2, 8);
290 blen += 4;
291 body[blen++] = 0x00;
292 body[blen++] = 0x01; /* family IPv4 */
293 wr16(body + blen, (uint16_t)(ntohs(mapped->sin_port) ^ 0xFFFF));
294 blen += 2;
295 uint32_t addr = ntohl(mapped->sin_addr.s_addr) ^ STUN_MAGIC;
296 wr32(body + blen, addr);
297 blen += 4;
298 while (blen % 4)
299 body[blen++] = 0;
300
301 /* MESSAGE-INTEGRITY over header+body+MI attr header with adjusted length */
302 uint16_t len_for_mi = (uint16_t)(blen + 4 + 20);
303 uint8_t hdr[STUN_HDR_LEN];
304 memset(hdr, 0, STUN_HDR_LEN);
305 wr16(hdr, CWIST_STUN_BINDING_RESPONSE);
306 wr16(hdr + 2, len_for_mi);
307 wr32(hdr + 4, STUN_MAGIC);
308 memcpy(hdr + 8, req + 8, 12);
309 uint8_t mi[20];
310 unsigned int mi_len = 0;
311 HMAC_CTX *hctx = HMAC_CTX_new();
312 if (!hctx)
313 return -1;
314 HMAC_Init_ex(hctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL);
315 HMAC_Update(hctx, hdr, STUN_HDR_LEN);
316 HMAC_Update(hctx, body, blen);
317 HMAC_Final(hctx, mi, &mi_len);
318 HMAC_CTX_free(hctx);
319
320 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
321 wr16(body + blen + 2, 20);
322 blen += 4;
323 memcpy(body + blen, mi, 20);
324 blen += 20;
325
326 /* FINGERPRINT */
327 uint16_t len_for_fp = (uint16_t)(blen + 8);
328 uint8_t hdr2[STUN_HDR_LEN];
329 memcpy(hdr2, hdr, STUN_HDR_LEN);
330 wr16(hdr2 + 2, len_for_fp);
331 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
332 c = crc32_update(c, body, blen);
333 c ^= 0xFFFFFFFFu;
334 wr16(body + blen, ATTR_FINGERPRINT);
335 wr16(body + blen + 2, 4);
336 wr32(body + blen + 4, c ^ 0x5354554Eu);
337 blen += 8;
338
339 if (cap < STUN_HDR_LEN + blen)
340 return -1;
341 memcpy(out, hdr2, STUN_HDR_LEN);
342 memcpy(out + STUN_HDR_LEN, body, blen);
343 return (int)(STUN_HDR_LEN + blen);
344}

361int cwist_ice_stun_build_request(uint8_t *out, size_t cap, const uint8_t txid[12],
362 const char *username) {
363 size_t ulen = strlen(username);
364 if (cap < STUN_HDR_LEN)
365 return -1;
366 memset(out, 0, STUN_HDR_LEN);
367 wr16(out, CWIST_STUN_BINDING_REQUEST);
368 wr32(out + 4, STUN_MAGIC);
369 memcpy(out + 8, txid, 12);
370
371 size_t blen = 0;
372 uint8_t body[256];
373 /* SOFTWARE-ish placeholder skipped; USERNAME */
374 if (ulen > 0) {
375 wr16(body + blen, ATTR_USERNAME);
376 wr16(body + blen + 2, (uint16_t)ulen);
377 blen += 4;
378 memcpy(body + blen, username, ulen);
379 blen += ulen;
380 while (blen % 4)
381 body[blen++] = 0;
382 }
383 /* ICE-CONTROLLING (8 bytes) */
384 wr16(body + blen, ATTR_ICE_CONTROLLING);
385 wr16(body + blen + 2, 8);
386 memset(body + blen + 4, 0, 8);
387 blen += 12;
388 /* USE-CANDIDATE */
389 wr16(body + blen, ATTR_USE_CANDIDATE);
390 wr16(body + blen + 2, 0);
391 blen += 4;
392 /* PRIORITY */
393 wr16(body + blen, ATTR_PRIORITY);
394 wr16(body + blen + 2, 4);
395 wr32(body + blen + 4, 0x7EFFFFFFu);
396 blen += 8;
397
398 /* MESSAGE-INTEGRITY + FINGERPRINT added by the caller via _finalize; here we
399 * keep the request unsigned because the MI key (peer pwd) is supplied at
400 * send time. cwist_ice_stun_sign_request() handles it. */
401 wr16(out + 2, (uint16_t)blen);
402 if (cap < STUN_HDR_LEN + blen)
403 return -1;
404 memcpy(out + STUN_HDR_LEN, body, blen);
405 return (int)(STUN_HDR_LEN + blen);
406}

518int cwist_ice_stun_parse_response(const uint8_t *buf, size_t len, const uint8_t txid[12],
519 const char *pwd, struct sockaddr_in *mapped) {
520 if (!cwist_ice_stun_is_message(buf, len) || rd16(buf) != CWIST_STUN_BINDING_RESPONSE)
521 return 0;
522 if (memcmp(buf + 8, txid, 12) != 0)
523 return 0;
524 if (!cwist_ice_stun_validate_request(buf, len, pwd))
525 return 0;
526 parse_resp_arg a = { 0 };
527 if (!walk_attrs(buf, len, parse_resp_cb, &a))
528 return 0;
529 if (!a.mi_val)
530 return 0;
531 if (mapped) {
532 if (!a.has_mapped)
533 return 0;
534 *mapped = a.mapped;
535 }
536 return 1;
537}

556int cwist_ice_stun_get_remote_ufrag(const uint8_t *buf, size_t len, char *out, size_t cap) {
557 if (!cwist_ice_stun_is_message(buf, len))
558 return -1;
559 uint16_t msg_len = rd16(buf + 2);
560 size_t end = STUN_HDR_LEN + msg_len;

562 while (off + 4 <= end) {
563 uint16_t type = rd16(buf + off);
564 uint16_t alen = rd16(buf + off + 2);
565 if (type == ATTR_USERNAME) {
566 if (off + 4 + alen > end) return -1;
567 const uint8_t *name = buf + off + 4;
568 size_t colon = 0;
569 while (colon < alen && name[colon] != ':') colon++;
570 if (colon == alen) return -1; /* no ':' separator */
571 size_t i = 0;

573 out[i] = '\0';
574 return 0;
575 }
576 off += 4 + ((alen + 3) & ~3u);
577 }
578 return -1;
