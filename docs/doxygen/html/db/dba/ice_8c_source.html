39static void crc32_init(void) {
40 for (uint32_t i = 0; i < 256; i++) {
41 uint32_t c = i;
42 for (int k = 0; k < 8; k++)
43 c = (c & 1) ? 0xEDB88320u ^ (c >> 1) : c >> 1;
44 crc32_table[i] = c;
45 }
46}

55static uint32_t crc32_update(uint32_t c, const uint8_t *data, size_t len) {
56 pthread_once(&crc32_once, crc32_init);
57 for (size_t i = 0; i < len; i++)
58 c = crc32_table[(c ^ data[i]) & 0xFF] ^ (c >> 8);
59 return c;
60}

141static bool walk_attrs(const uint8_t *msg, size_t len, attr_cb cb, void *arg) {
142 uint16_t msg_len = rd16(msg + 2);
143 if ((size_t)msg_len + STUN_HDR_LEN > len || msg_len % 4 != 0)
144 return false;
145 size_t off = STUN_HDR_LEN;
146 size_t end = STUN_HDR_LEN + msg_len;
147 while (off + 4 <= end) {
148 uint16_t type = rd16(msg + off);
149 uint16_t alen = rd16(msg + off + 2);
150 off += 4;
151 if (off + alen > end)
152 return false;
153 if (!cb(type, msg + off, alen, arg))
154 return false;
155 off += (alen + 3) & ~3u;
156 }
157 return off == end;
158}

251int cwist_ice_stun_validate_request(const uint8_t *buf, size_t len, const char *pwd) {
252 size_t pwd_len = strlen(pwd);
253 uint8_t hmac[EVP_MAX_MD_SIZE];
254 unsigned int hmac_len = 0;
255 size_t mi_off;
256 if (!cwist_ice_stun_is_message(buf, len))
257 return 0;
258 if (!find_mi(buf, len, &mi_off))
259 return 1; /* no MI: accept (RFC 8445 requires it, but stay permissive at MVP) */
260 /* MI covers the message up to and including the MI attribute header, with
261 * the message length field set to end at the MI attribute. */
262 uint8_t hdr[STUN_HDR_LEN];
263 memcpy(hdr, buf, STUN_HDR_LEN);
264 wr16(hdr + 2, (uint16_t)(mi_off - STUN_HDR_LEN + 4 + 20));
265 HMAC_CTX *ctx = HMAC_CTX_new();
266 if (!ctx)
267 return 0;
268 HMAC_Init_ex(ctx, pwd, (int)pwd_len, EVP_sha1(), NULL);
269 HMAC_Update(ctx, hdr, STUN_HDR_LEN);
270 HMAC_Update(ctx, buf + STUN_HDR_LEN, mi_off - STUN_HDR_LEN);
271 HMAC_Final(ctx, hmac, &hmac_len);
272 HMAC_CTX_free(ctx);
273 return hmac_len == 20 && memcmp(hmac, buf + mi_off + 4, 20) == 0;
274}

291int cwist_ice_stun_build_response(uint8_t *out, size_t cap, const uint8_t *req, size_t req_len,
292 const struct sockaddr_in *mapped, const char *pwd) {
293 (void)req_len;
294 uint8_t body[64];
295 size_t blen = 0;
296
297 /* XOR-MAPPED-ADDRESS, IPv4 */
298 wr16(body + blen, ATTR_XOR_MAPPED_ADDRESS);
299 wr16(body + blen + 2, 8);
300 blen += 4;
301 body[blen++] = 0x00;
302 body[blen++] = 0x01; /* family IPv4 */
303 wr16(body + blen, (uint16_t)(ntohs(mapped->sin_port) ^ 0xFFFF));
304 blen += 2;
305 uint32_t addr = ntohl(mapped->sin_addr.s_addr) ^ STUN_MAGIC;
306 wr32(body + blen, addr);
307 blen += 4;
308 while (blen % 4)
309 body[blen++] = 0;
310
311 /* MESSAGE-INTEGRITY over header+body+MI attr header with adjusted length */
312 uint16_t len_for_mi = (uint16_t)(blen + 4 + 20);
313 uint8_t hdr[STUN_HDR_LEN];
314 memset(hdr, 0, STUN_HDR_LEN);
315 wr16(hdr, CWIST_STUN_BINDING_RESPONSE);
316 wr16(hdr + 2, len_for_mi);
317 wr32(hdr + 4, STUN_MAGIC);
318 memcpy(hdr + 8, req + 8, 12);
319 uint8_t mi[20];
320 unsigned int mi_len = 0;
321 HMAC_CTX *hctx = HMAC_CTX_new();
322 if (!hctx)
323 return -1;
324 HMAC_Init_ex(hctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL);
325 HMAC_Update(hctx, hdr, STUN_HDR_LEN);
326 HMAC_Update(hctx, body, blen);
327 HMAC_Final(hctx, mi, &mi_len);
328 HMAC_CTX_free(hctx);
329
330 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
331 wr16(body + blen + 2, 20);
332 blen += 4;
333 memcpy(body + blen, mi, 20);
334 blen += 20;
335
336 /* FINGERPRINT */
337 uint16_t len_for_fp = (uint16_t)(blen + 8);
338 uint8_t hdr2[STUN_HDR_LEN];
339 memcpy(hdr2, hdr, STUN_HDR_LEN);
340 wr16(hdr2 + 2, len_for_fp);
341 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
342 c = crc32_update(c, body, blen);
343 c ^= 0xFFFFFFFFu;
344 wr16(body + blen, ATTR_FINGERPRINT);
345 wr16(body + blen + 2, 4);
346 wr32(body + blen + 4, c ^ 0x5354554Eu);
347 blen += 8;
348
349 if (cap < STUN_HDR_LEN + blen)
350 return -1;
351 memcpy(out, hdr2, STUN_HDR_LEN);
352 memcpy(out + STUN_HDR_LEN, body, blen);
353 return (int)(STUN_HDR_LEN + blen);
354}

371int cwist_ice_stun_build_request(uint8_t *out, size_t cap, const uint8_t txid[12],
372 const char *username) {
373 size_t ulen = strlen(username);
374 uint8_t body[256];
375 if (ulen > UINT16_MAX) return -1;
376 size_t username_attr_len = ulen ? 4 + ((ulen + 3) & ~3u) : 0;
377 const size_t fixed_attrs_len = 12 + 4 + 8;
378 if (username_attr_len > sizeof(body) - fixed_attrs_len ||
379 cap < STUN_HDR_LEN + username_attr_len + fixed_attrs_len)
380 return -1;
381
382 memset(out, 0, STUN_HDR_LEN);
383 wr16(out, CWIST_STUN_BINDING_REQUEST);
384 wr32(out + 4, STUN_MAGIC);
385 memcpy(out + 8, txid, 12);
386
387 size_t blen = 0;
388 /* SOFTWARE-ish placeholder skipped; USERNAME */
389 if (ulen > 0) {
390 wr16(body + blen, ATTR_USERNAME);
391 wr16(body + blen + 2, (uint16_t)ulen);
392 blen += 4;
393 memcpy(body + blen, username, ulen);
394 blen += ulen;
395 while (blen % 4)
396 body[blen++] = 0;
397 }
398 /* ICE-CONTROLLING (8 bytes) */
399 wr16(body + blen, ATTR_ICE_CONTROLLING);
400 wr16(body + blen + 2, 8);
401 memset(body + blen + 4, 0, 8);
402 blen += 12;
403 /* USE-CANDIDATE */
404 wr16(body + blen, ATTR_USE_CANDIDATE);
405 wr16(body + blen + 2, 0);
406 blen += 4;
407 /* PRIORITY */
408 wr16(body + blen, ATTR_PRIORITY);
409 wr16(body + blen + 2, 4);
410 wr32(body + blen + 4, 0x7EFFFFFFu);
411 blen += 8;
412
413 /* MESSAGE-INTEGRITY + FINGERPRINT added by the caller via _finalize; here we
414 * keep the request unsigned because the MI key (peer pwd) is supplied at
415 * send time. cwist_ice_stun_sign_request() handles it. */
416 wr16(out + 2, (uint16_t)blen);
417 if (cap < STUN_HDR_LEN + blen)
418 return -1;
419 memcpy(out + STUN_HDR_LEN, body, blen);
420 return (int)(STUN_HDR_LEN + blen);
421}

438static int stun_sign(uint8_t *msg, size_t cap, size_t msg_len, const char *pwd) {
439 uint8_t body[64];
440 size_t blen = 0;
441 uint16_t len_for_mi = (uint16_t)(msg_len - STUN_HDR_LEN + 4 + 20);
442 uint8_t hdr[STUN_HDR_LEN];
443 memcpy(hdr, msg, STUN_HDR_LEN);
444 wr16(hdr + 2, len_for_mi);
445 uint8_t mi[20];
446 unsigned int mi_len = 0;
447 HMAC_CTX *hctx = HMAC_CTX_new();
448 if (!hctx)
449 return -1;
450 HMAC_Init_ex(hctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL);
451 HMAC_Update(hctx, hdr, STUN_HDR_LEN);
452 HMAC_Update(hctx, msg + STUN_HDR_LEN, msg_len - STUN_HDR_LEN);
453 HMAC_Final(hctx, mi, &mi_len);
454 HMAC_CTX_free(hctx);
455
456 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
457 wr16(body + blen + 2, 20);
458 blen += 4;
459 memcpy(body + blen, mi, 20);
460 blen += 20;
461
462 uint16_t len_for_fp = (uint16_t)(msg_len - STUN_HDR_LEN + blen + 8);
463 uint8_t hdr2[STUN_HDR_LEN];
464 memcpy(hdr2, hdr, STUN_HDR_LEN);
465 wr16(hdr2 + 2, len_for_fp);
466 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
467 c = crc32_update(c, msg + STUN_HDR_LEN, msg_len - STUN_HDR_LEN);
468 c = crc32_update(c, body, blen);
469 c ^= 0xFFFFFFFFu;
470 wr16(body + blen, ATTR_FINGERPRINT);
471 wr16(body + blen + 2, 4);
472 wr32(body + blen + 4, c ^ 0x5354554Eu);
473 blen += 8;
474
475 if (cap < msg_len + blen)
476 return -1;
477 memcpy(msg + msg_len, body, blen);
478 memcpy(msg, hdr2, STUN_HDR_LEN);
479 return (int)(msg_len + blen);
480}

533int cwist_ice_stun_parse_response(const uint8_t *buf, size_t len, const uint8_t txid[12],
534 const char *pwd, struct sockaddr_in *mapped) {
535 if (!cwist_ice_stun_is_message(buf, len) || rd16(buf) != CWIST_STUN_BINDING_RESPONSE)
536 return 0;
537 if (memcmp(buf + 8, txid, 12) != 0)
538 return 0;
539 if (!cwist_ice_stun_validate_request(buf, len, pwd))
540 return 0;
541 parse_resp_arg a = { 0 };
542 if (!walk_attrs(buf, len, parse_resp_cb, &a))
543 return 0;
544 if (!a.mi_val)
545 return 0;
546 if (mapped) {
547 if (!a.has_mapped)
548 return 0;
549 *mapped = a.mapped;
550 }
551 return 1;
552}

571int cwist_ice_stun_get_remote_ufrag(const uint8_t *buf, size_t len, char *out, size_t cap) {
572 if (!out || cap == 0 || !cwist_ice_stun_is_message(buf, len)) return -1;
573 uint16_t msg_len = rd16(buf + 2);
574 if (msg_len % 4 != 0 || msg_len > len - STUN_HDR_LEN) return -1;
575 size_t end = STUN_HDR_LEN + msg_len;
576 size_t off = STUN_HDR_LEN;
577 while (off + 4 <= end) {
578 uint16_t type = rd16(buf + off);
579 uint16_t alen = rd16(buf + off + 2);
580 if (type == ATTR_USERNAME) {
581 if (off + 4 + alen > end) return -1;
582 const uint8_t *name = buf + off + 4;
583 size_t colon = 0;
584 while (colon < alen && name[colon] != ':') colon++;
585 if (colon == alen) return -1; /* no ':' separator */
586 size_t i = 0;
587 for (size_t j = colon + 1; j < alen && i + 1 < cap; j++) out[i++] = (char)name[j];
588 out[i] = '\0';
589 return 0;
590 }
591 off += 4 + ((alen + 3) & ~3u);
592 }
593 return -1;
594}
