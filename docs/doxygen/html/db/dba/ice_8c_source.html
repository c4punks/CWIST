39static void crc32_init(void) {
40 for (uint32_t i = 0; i < 256; i++) {
41 uint32_t c = i;
42 for (int k = 0; k < 8; k++)
43 c = (c & 1) ? 0xEDB88320u ^ (c >> 1) : c >> 1;
44 crc32_table[i] = c;
45 }
46}

55static uint32_t crc32_update(uint32_t c, const uint8_t *data, size_t len) {
56 pthread_once(&crc32_once, crc32_init);
57 for (size_t i = 0; i < len; i++)
58 c = crc32_table[(c ^ data[i]) & 0xFF] ^ (c >> 8);
59 return c;
60}

135static bool walk_attrs(const uint8_t *msg, size_t len, attr_cb cb, void *arg) {
136 uint16_t msg_len = rd16(msg + 2);
137 if ((size_t)msg_len + STUN_HDR_LEN > len || msg_len % 4 != 0)
138 return false;
139 size_t off = STUN_HDR_LEN;
140 size_t end = STUN_HDR_LEN + msg_len;
141 while (off + 4 <= end) {
142 uint16_t type = rd16(msg + off);
143 uint16_t alen = rd16(msg + off + 2);
144 off += 4;
145 if (off + alen > end)
146 return false;
147 if (!cb(type, msg + off, alen, arg))
148 return false;
149 off += (alen + 3) & ~3u;
150 }
151 return off == end;
152}

232static int stun_hmac_sha1(const uint8_t *hdr, const uint8_t *body, size_t body_len,
233 const char *pwd, uint8_t out[20]) {
234 HMAC_CTX *ctx = HMAC_CTX_new();
235 if (!ctx) return -1;
236 unsigned int out_len = 0;
237 int ok = HMAC_Init_ex(ctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL) == 1 &&
238 HMAC_Update(ctx, hdr, STUN_HDR_LEN) == 1 &&
239 HMAC_Update(ctx, body, body_len) == 1 && HMAC_Final(ctx, out, &out_len) == 1 &&
240 out_len == 20;
241 HMAC_CTX_free(ctx);
242 return ok ? 0 : -1;
243}

255int cwist_ice_stun_validate_request(const uint8_t *buf, size_t len, const char *pwd) {
256 uint8_t hmac[EVP_MAX_MD_SIZE];
257 size_t mi_off;
258 if (!cwist_ice_stun_is_message(buf, len)) return 0;
259 if (!find_mi(buf, len, &mi_off)) return 0;
260 /* MI covers the message up to and including the MI attribute header, with
261 * the message length field set to end at the MI attribute. */
262 uint8_t hdr[STUN_HDR_LEN];
263 memcpy(hdr, buf, STUN_HDR_LEN);
264 wr16(hdr + 2, (uint16_t)(mi_off - STUN_HDR_LEN + 4 + 20));
265 return stun_hmac_sha1(hdr, buf + STUN_HDR_LEN, mi_off - STUN_HDR_LEN, pwd, hmac) == 0 &&
266 memcmp(hmac, buf + mi_off + 4, 20) == 0;
267}

284int cwist_ice_stun_build_response(uint8_t *out, size_t cap, const uint8_t *req, size_t req_len,
285 const struct sockaddr_in *mapped, const char *pwd) {
286 (void)req_len;
287 uint8_t body[64];
288 size_t blen = 0;
289
290 /* XOR-MAPPED-ADDRESS, IPv4 */
291 wr16(body + blen, ATTR_XOR_MAPPED_ADDRESS);
292 wr16(body + blen + 2, 8);
293 blen += 4;
294 body[blen++] = 0x00;
295 body[blen++] = 0x01; /* family IPv4 */
296 wr16(body + blen, (uint16_t)(ntohs(mapped->sin_port) ^ 0xFFFF));
297 blen += 2;
298 uint32_t addr = ntohl(mapped->sin_addr.s_addr) ^ STUN_MAGIC;
299 wr32(body + blen, addr);
300 blen += 4;
301 while (blen % 4)
302 body[blen++] = 0;
303
304 /* MESSAGE-INTEGRITY over header+body+MI attr header with adjusted length */
305 uint16_t len_for_mi = (uint16_t)(blen + 4 + 20);
306 uint8_t hdr[STUN_HDR_LEN];
307 memset(hdr, 0, STUN_HDR_LEN);
308 wr16(hdr, CWIST_STUN_BINDING_RESPONSE);
309 wr16(hdr + 2, len_for_mi);
310 wr32(hdr + 4, STUN_MAGIC);
311 memcpy(hdr + 8, req + 8, 12);
312 uint8_t mi[20];
313 if (stun_hmac_sha1(hdr, body, blen, pwd, mi) != 0) return -1;
314
315 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
316 wr16(body + blen + 2, 20);
317 blen += 4;
318 memcpy(body + blen, mi, 20);
319 blen += 20;
320
321 /* FINGERPRINT */
322 uint16_t len_for_fp = (uint16_t)(blen + 8);
323 uint8_t hdr2[STUN_HDR_LEN];
324 memcpy(hdr2, hdr, STUN_HDR_LEN);
325 wr16(hdr2 + 2, len_for_fp);
326 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
327 c = crc32_update(c, body, blen);
328 c ^= 0xFFFFFFFFu;
329 wr16(body + blen, ATTR_FINGERPRINT);
330 wr16(body + blen + 2, 4);
331 wr32(body + blen + 4, c ^ 0x5354554Eu);
332 blen += 8;
333
334 if (cap < STUN_HDR_LEN + blen)
335 return -1;
336 memcpy(out, hdr2, STUN_HDR_LEN);
337 memcpy(out + STUN_HDR_LEN, body, blen);
338 return (int)(STUN_HDR_LEN + blen);
339}

356int cwist_ice_stun_build_request(uint8_t *out, size_t cap, const uint8_t txid[12],
357 const char *username) {
358 size_t ulen = strlen(username);
359 uint8_t body[256];
360 if (ulen > UINT16_MAX) return -1;
361 size_t username_attr_len = ulen ? 4 + ((ulen + 3) & ~3u) : 0;
362 const size_t fixed_attrs_len = 12 + 4 + 8;
363 if (username_attr_len > sizeof(body) - fixed_attrs_len ||
364 cap < STUN_HDR_LEN + username_attr_len + fixed_attrs_len)
365 return -1;
366
367 memset(out, 0, STUN_HDR_LEN);
368 wr16(out, CWIST_STUN_BINDING_REQUEST);
369 wr32(out + 4, STUN_MAGIC);
370 memcpy(out + 8, txid, 12);
371
372 size_t blen = 0;
373 /* SOFTWARE-ish placeholder skipped; USERNAME */
374 if (ulen > 0) {
375 wr16(body + blen, ATTR_USERNAME);
376 wr16(body + blen + 2, (uint16_t)ulen);
377 blen += 4;
378 memcpy(body + blen, username, ulen);
379 blen += ulen;
380 while (blen % 4)
381 body[blen++] = 0;
382 }
383 /* ICE-CONTROLLING (8 bytes) */
384 wr16(body + blen, ATTR_ICE_CONTROLLING);
385 wr16(body + blen + 2, 8);
386 memset(body + blen + 4, 0, 8);
387 blen += 12;
388 /* USE-CANDIDATE */
389 wr16(body + blen, ATTR_USE_CANDIDATE);
390 wr16(body + blen + 2, 0);
391 blen += 4;
392 /* PRIORITY */
393 wr16(body + blen, ATTR_PRIORITY);
394 wr16(body + blen + 2, 4);
395 wr32(body + blen + 4, 0x7EFFFFFFu);
396 blen += 8;
397
398 /* MESSAGE-INTEGRITY + FINGERPRINT added by the caller via _finalize; here we
399 * keep the request unsigned because the MI key (peer pwd) is supplied at
400 * send time. cwist_ice_stun_sign_request() handles it. */
401 wr16(out + 2, (uint16_t)blen);
402 if (cap < STUN_HDR_LEN + blen)
403 return -1;
404 memcpy(out + STUN_HDR_LEN, body, blen);
405 return (int)(STUN_HDR_LEN + blen);
406}

512int cwist_ice_stun_parse_response(const uint8_t *buf, size_t len, const uint8_t txid[12],
513 const char *pwd, struct sockaddr_in *mapped) {
514 if (!cwist_ice_stun_is_message(buf, len) || rd16(buf) != CWIST_STUN_BINDING_RESPONSE)
515 return 0;
516 if (memcmp(buf + 8, txid, 12) != 0)
517 return 0;
518 if (!cwist_ice_stun_validate_request(buf, len, pwd))
519 return 0;
520 parse_resp_arg a = { 0 };
521 if (!walk_attrs(buf, len, parse_resp_cb, &a))
522 return 0;
523 if (!a.mi_val)
524 return 0;
525 if (mapped) {
526 if (!a.has_mapped)
527 return 0;
528 *mapped = a.mapped;
529 }
530 return 1;
531}

550int cwist_ice_stun_get_remote_ufrag(const uint8_t *buf, size_t len, char *out, size_t cap) {
551 if (!out || cap == 0 || !cwist_ice_stun_is_message(buf, len)) return -1;
552 uint16_t msg_len = rd16(buf + 2);
553 if (msg_len % 4 != 0 || msg_len > len - STUN_HDR_LEN) return -1;
554 size_t end = STUN_HDR_LEN + msg_len;
555 size_t off = STUN_HDR_LEN;
556 while (off + 4 <= end) {
557 uint16_t type = rd16(buf + off);
558 uint16_t alen = rd16(buf + off + 2);
559 if (type == ATTR_USERNAME) {
560 if (off + 4 + alen > end) return -1;
561 const uint8_t *name = buf + off + 4;
562 size_t colon = 0;
563 while (colon < alen && name[colon] != ':') colon++;
564 if (colon == alen) return -1; /* no ':' separator */
565 size_t i = 0;
566 for (size_t j = colon + 1; j < alen && i + 1 < cap; j++) out[i++] = (char)name[j];
567 out[i] = '\0';
568 return 0;
569 }
570 off += 4 + ((alen + 3) & ~3u);
571 }
572 return -1;
573}

586void cwist_ice_random_creds(char *ufrag, size_t ufrag_cap, char *pwd, size_t pwd_cap) {
587 uint8_t raw[24];
588 if (ufrag && ufrag_cap > 0) ufrag[0] = '\0';
589 if (pwd && pwd_cap > 0) pwd[0] = '\0';
590 if (!ufrag || !pwd || ufrag_cap == 0 || pwd_cap == 0 ||
591 RAND_bytes(raw, sizeof(raw)) != 1)
592 return;
593
594 size_t n = ufrag_cap - 1 < 8 ? ufrag_cap - 1 : 8;
595 for (size_t i = 0; i < n; i++)
596 ufrag[i] = b64url_chars[raw[i] & 63];
597 ufrag[n] = '\0';
598 size_t m = pwd_cap - 1 < 32 ? pwd_cap - 1 : 32;
599 for (size_t i = 0; i < m; i++)
600 pwd[i] = b64url_chars[raw[8 + (i % 16)] & 63];
601 pwd[m] = '\0';
602}
