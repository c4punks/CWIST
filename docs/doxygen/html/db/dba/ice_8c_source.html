42static void crc32_init(void) {
43 if (crc32_ready)
44 return;
45 for (uint32_t i = 0; i < 256; i++) {
46 uint32_t c = i;
47 for (int k = 0; k < 8; k++)
48 c = (c & 1) ? 0xEDB88320u ^ (c >> 1) : c >> 1;
49 crc32_table[i] = c;
50 }
51 crc32_ready = true;
52}

61static uint32_t crc32_update(uint32_t c, const uint8_t *data, size_t len) {
62 crc32_init();
63 for (size_t i = 0; i < len; i++)
64 c = crc32_table[(c ^ data[i]) & 0xFF] ^ (c >> 8);
65 return c;
66}

141static bool walk_attrs(const uint8_t *msg, size_t len, attr_cb cb, void *arg) {
142 uint16_t msg_len = rd16(msg + 2);
143 if ((size_t)msg_len + STUN_HDR_LEN > len || msg_len % 4 != 0)
144 return false;
145 size_t off = STUN_HDR_LEN;
146 size_t end = STUN_HDR_LEN + msg_len;
147 while (off + 4 <= end) {
148 uint16_t type = rd16(msg + off);
149 uint16_t alen = rd16(msg + off + 2);
150 off += 4;
151 if (off + alen > end)
152 return false;
153 if (!cb(type, msg + off, alen, arg))
154 return false;
155 off += (alen + 3) & ~3u;
156 }
157 return off == end;
158}

248int cwist_ice_stun_validate_request(const uint8_t *buf, size_t len, const char *pwd) {
249 size_t pwd_len = strlen(pwd);
250 uint8_t hmac[EVP_MAX_MD_SIZE];
251 unsigned int hmac_len = 0;
252 size_t mi_off;
253 if (!cwist_ice_stun_is_message(buf, len)) return 0;
254 if (!find_mi(buf, len, &mi_off)) return 0;
255 /* MI covers the message up to and including the MI attribute header, with
256 * the message length field set to end at the MI attribute. */
257 uint8_t hdr[STUN_HDR_LEN];
258 memcpy(hdr, buf, STUN_HDR_LEN);
259 wr16(hdr + 2, (uint16_t)(mi_off - STUN_HDR_LEN + 4 + 20));
260 HMAC_CTX *ctx = HMAC_CTX_new();
261 if (!ctx)
262 return 0;
263 HMAC_Init_ex(ctx, pwd, (int)pwd_len, EVP_sha1(), NULL);
264 HMAC_Update(ctx, hdr, STUN_HDR_LEN);
265 HMAC_Update(ctx, buf + STUN_HDR_LEN, mi_off - STUN_HDR_LEN);
266 HMAC_Final(ctx, hmac, &hmac_len);
267 HMAC_CTX_free(ctx);
268 return hmac_len == 20 && memcmp(hmac, buf + mi_off + 4, 20) == 0;
269}

286int cwist_ice_stun_build_response(uint8_t *out, size_t cap, const uint8_t *req, size_t req_len,
287 const struct sockaddr_in *mapped, const char *pwd) {
288 (void)req_len;
289 uint8_t body[64];
290 size_t blen = 0;
291
292 /* XOR-MAPPED-ADDRESS, IPv4 */
293 wr16(body + blen, ATTR_XOR_MAPPED_ADDRESS);
294 wr16(body + blen + 2, 8);
295 blen += 4;
296 body[blen++] = 0x00;
297 body[blen++] = 0x01; /* family IPv4 */
298 wr16(body + blen, (uint16_t)(ntohs(mapped->sin_port) ^ 0xFFFF));
299 blen += 2;
300 uint32_t addr = ntohl(mapped->sin_addr.s_addr) ^ STUN_MAGIC;
301 wr32(body + blen, addr);
302 blen += 4;
303 while (blen % 4)
304 body[blen++] = 0;
305
306 /* MESSAGE-INTEGRITY over header+body+MI attr header with adjusted length */
307 uint16_t len_for_mi = (uint16_t)(blen + 4 + 20);
308 uint8_t hdr[STUN_HDR_LEN];
309 memset(hdr, 0, STUN_HDR_LEN);
310 wr16(hdr, CWIST_STUN_BINDING_RESPONSE);
311 wr16(hdr + 2, len_for_mi);
312 wr32(hdr + 4, STUN_MAGIC);
313 memcpy(hdr + 8, req + 8, 12);
314 uint8_t mi[20];
315 unsigned int mi_len = 0;
316 HMAC_CTX *hctx = HMAC_CTX_new();
317 if (!hctx)
318 return -1;
319 HMAC_Init_ex(hctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL);
320 HMAC_Update(hctx, hdr, STUN_HDR_LEN);
321 HMAC_Update(hctx, body, blen);
322 HMAC_Final(hctx, mi, &mi_len);
323 HMAC_CTX_free(hctx);
324
325 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
326 wr16(body + blen + 2, 20);
327 blen += 4;
328 memcpy(body + blen, mi, 20);
329 blen += 20;
330
331 /* FINGERPRINT */
332 uint16_t len_for_fp = (uint16_t)(blen + 8);
333 uint8_t hdr2[STUN_HDR_LEN];
334 memcpy(hdr2, hdr, STUN_HDR_LEN);
335 wr16(hdr2 + 2, len_for_fp);
336 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
337 c = crc32_update(c, body, blen);
338 c ^= 0xFFFFFFFFu;
339 wr16(body + blen, ATTR_FINGERPRINT);
340 wr16(body + blen + 2, 4);
341 wr32(body + blen + 4, c ^ 0x5354554Eu);
342 blen += 8;
343
344 if (cap < STUN_HDR_LEN + blen)
345 return -1;
346 memcpy(out, hdr2, STUN_HDR_LEN);
347 memcpy(out + STUN_HDR_LEN, body, blen);
348 return (int)(STUN_HDR_LEN + blen);
349}

366int cwist_ice_stun_build_request(uint8_t *out, size_t cap, const uint8_t txid[12],
367 const char *username) {
368 size_t ulen = strlen(username);
369 uint8_t body[256];
370 if (ulen > UINT16_MAX) return -1;
371 size_t username_attr_len = ulen ? 4 + ((ulen + 3) & ~3u) : 0;
372 const size_t fixed_attrs_len = 12 + 4 + 8;
373 if (username_attr_len > sizeof(body) - fixed_attrs_len ||
374 cap < STUN_HDR_LEN + username_attr_len + fixed_attrs_len)
375 return -1;
376
377 memset(out, 0, STUN_HDR_LEN);
378 wr16(out, CWIST_STUN_BINDING_REQUEST);
379 wr32(out + 4, STUN_MAGIC);
380 memcpy(out + 8, txid, 12);
381
382 size_t blen = 0;
383 /* SOFTWARE-ish placeholder skipped; USERNAME */
384 if (ulen > 0) {
385 wr16(body + blen, ATTR_USERNAME);
386 wr16(body + blen + 2, (uint16_t)ulen);
387 blen += 4;
388 memcpy(body + blen, username, ulen);
389 blen += ulen;
390 while (blen % 4)
391 body[blen++] = 0;
392 }
393 /* ICE-CONTROLLING (8 bytes) */
394 wr16(body + blen, ATTR_ICE_CONTROLLING);
395 wr16(body + blen + 2, 8);
396 memset(body + blen + 4, 0, 8);
397 blen += 12;
398 /* USE-CANDIDATE */
399 wr16(body + blen, ATTR_USE_CANDIDATE);
400 wr16(body + blen + 2, 0);
401 blen += 4;
402 /* PRIORITY */
403 wr16(body + blen, ATTR_PRIORITY);
404 wr16(body + blen + 2, 4);
405 wr32(body + blen + 4, 0x7EFFFFFFu);
406 blen += 8;
407
408 /* MESSAGE-INTEGRITY + FINGERPRINT added by the caller via _finalize; here we
409 * keep the request unsigned because the MI key (peer pwd) is supplied at
410 * send time. cwist_ice_stun_sign_request() handles it. */
411 wr16(out + 2, (uint16_t)blen);
412 if (cap < STUN_HDR_LEN + blen)
413 return -1;
414 memcpy(out + STUN_HDR_LEN, body, blen);
415 return (int)(STUN_HDR_LEN + blen);
416}

433static int stun_sign(uint8_t *msg, size_t cap, size_t msg_len, const char *pwd) {
434 uint8_t body[64];
435 size_t blen = 0;
436 uint16_t len_for_mi = (uint16_t)(msg_len - STUN_HDR_LEN + 4 + 20);
437 uint8_t hdr[STUN_HDR_LEN];
438 memcpy(hdr, msg, STUN_HDR_LEN);
439 wr16(hdr + 2, len_for_mi);
440 uint8_t mi[20];
441 unsigned int mi_len = 0;
442 HMAC_CTX *hctx = HMAC_CTX_new();
443 if (!hctx)
444 return -1;
445 HMAC_Init_ex(hctx, pwd, (int)strlen(pwd), EVP_sha1(), NULL);
446 HMAC_Update(hctx, hdr, STUN_HDR_LEN);
447 HMAC_Update(hctx, msg + STUN_HDR_LEN, msg_len - STUN_HDR_LEN);
448 HMAC_Final(hctx, mi, &mi_len);
449 HMAC_CTX_free(hctx);
450
451 wr16(body + blen, ATTR_MESSAGE_INTEGRITY);
452 wr16(body + blen + 2, 20);
453 blen += 4;
454 memcpy(body + blen, mi, 20);
455 blen += 20;
456
457 uint16_t len_for_fp = (uint16_t)(msg_len - STUN_HDR_LEN + blen + 8);
458 uint8_t hdr2[STUN_HDR_LEN];
459 memcpy(hdr2, hdr, STUN_HDR_LEN);
460 wr16(hdr2 + 2, len_for_fp);
461 uint32_t c = crc32_update(0xFFFFFFFFu, hdr2, STUN_HDR_LEN);
462 c = crc32_update(c, msg + STUN_HDR_LEN, msg_len - STUN_HDR_LEN);
463 c = crc32_update(c, body, blen);
464 c ^= 0xFFFFFFFFu;
465 wr16(body + blen, ATTR_FINGERPRINT);
466 wr16(body + blen + 2, 4);
467 wr32(body + blen + 4, c ^ 0x5354554Eu);
468 blen += 8;
469
470 if (cap < msg_len + blen)
471 return -1;
472 memcpy(msg + msg_len, body, blen);
473 memcpy(msg, hdr2, STUN_HDR_LEN);
474 return (int)(msg_len + blen);
475}

528int cwist_ice_stun_parse_response(const uint8_t *buf, size_t len, const uint8_t txid[12],
529 const char *pwd, struct sockaddr_in *mapped) {
530 if (!cwist_ice_stun_is_message(buf, len) || rd16(buf) != CWIST_STUN_BINDING_RESPONSE)
531 return 0;
532 if (memcmp(buf + 8, txid, 12) != 0)
533 return 0;
534 if (!cwist_ice_stun_validate_request(buf, len, pwd))
535 return 0;
536 parse_resp_arg a = { 0 };
537 if (!walk_attrs(buf, len, parse_resp_cb, &a))
538 return 0;
539 if (!a.mi_val)
540 return 0;
541 if (mapped) {
542 if (!a.has_mapped)
543 return 0;
544 *mapped = a.mapped;
545 }
546 return 1;
547}

566int cwist_ice_stun_get_remote_ufrag(const uint8_t *buf, size_t len, char *out, size_t cap) {
567 if (!out || cap == 0 || !cwist_ice_stun_is_message(buf, len)) return -1;
568 uint16_t msg_len = rd16(buf + 2);
569 if (msg_len % 4 != 0 || msg_len > len - STUN_HDR_LEN) return -1;
570 size_t end = STUN_HDR_LEN + msg_len;
571 size_t off = STUN_HDR_LEN;
572 while (off + 4 <= end) {
573 uint16_t type = rd16(buf + off);
574 uint16_t alen = rd16(buf + off + 2);
575 if (type == ATTR_USERNAME) {
576 if (off + 4 + alen > end) return -1;
577 const uint8_t *name = buf + off + 4;
578 size_t colon = 0;
579 while (colon < alen && name[colon] != ':') colon++;
580 if (colon == alen) return -1; /* no ':' separator */
581 size_t i = 0;
582 for (size_t j = colon + 1; j < alen && i + 1 < cap; j++) out[i++] = (char)name[j];
583 out[i] = '\0';
584 return 0;
585 }
586 off += 4 + ((alen + 3) & ~3u);
587 }
588 return -1;
589}
