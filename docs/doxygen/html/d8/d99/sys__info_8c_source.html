57static bool sys_cgroup_headroom(uint64_t *out) {
58 uint64_t limit = 0, usage = 0;
59 /* cgroup v2: the process's own group from /proc/self/cgroup ("0::/path"). */
60 FILE *fp = fopen("/proc/self/cgroup", "r");
61 if (fp) {
62 char line[512];
63 while (fgets(line, sizeof(line), fp)) {
64 if (strncmp(line, "0::", 3) != 0) continue;
65 char *path = line + 3;
66 path[strcspn(path, "\r\n")] = '\0';
67 char file[640];
68 snprintf(file, sizeof(file), "/sys/fs/cgroup%s/memory.max",
69 strcmp(path, "/") == 0 ? "" : path);
70 bool have_limit = sys_read_u64(file, &limit);
71 snprintf(file, sizeof(file), "/sys/fs/cgroup%s/memory.current",
72 strcmp(path, "/") == 0 ? "" : path);
73 if (have_limit && sys_read_u64(file, &usage)) {
74 fclose(fp);
75 *out = usage < limit ? limit - usage : 1; /* 0 would mean "unknown" */
76 return true;
77 }
78 break;
79 }
80 fclose(fp);
81 }
82 /* cgroup v1 memory controller; "no limit" reads as a huge number. */
83 if (sys_read_u64("/sys/fs/cgroup/memory/memory.limit_in_bytes", &limit) &&
84 limit < ((uint64_t)1 << 60) &&
85 sys_read_u64("/sys/fs/cgroup/memory/memory.usage_in_bytes", &usage)) {
86 *out = usage < limit ? limit - usage : 1; /* 0 would mean "unknown" */
87 return true;
88 }
89 return false;
90}

103uint64_t cwist_get_available_ram(void) {
104 uint64_t avail = 0;
105 bool have = sys_mem_available(&avail);
106 if (!have) {
107 struct sysinfo si;
108 if (sysinfo(&si) == 0) {
109 avail = (uint64_t)si.freeram * (uint64_t)si.mem_unit;
110 have = true;
111 }
112 }
113 uint64_t headroom = 0;
114 if (sys_cgroup_headroom(&headroom) && (!have || headroom < avail)) {
115 avail = headroom;
116 have = true;
117 }
118 return have ? avail : 0;
119}
