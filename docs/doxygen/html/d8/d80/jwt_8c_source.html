31static void b64url_encode(const unsigned char *src, size_t src_len, char *dst) {
32 static const char tbl[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
33 size_t i = 0, j = 0;
34 while (i + 2 < src_len) {
35 uint32_t v = ((uint32_t)src[i] << 16) | ((uint32_t)src[i + 1] << 8) | src[i + 2];
36 dst[j++] = tbl[(v >> 18) & 0x3F];
37 dst[j++] = tbl[(v >> 12) & 0x3F];
38 dst[j++] = tbl[(v >> 6) & 0x3F];
39 dst[j++] = tbl[v & 0x3F];
40 i += 3;
41 }
42 if (i + 1 == src_len) {
43 uint32_t v = (uint32_t)src[i] << 4;
44 dst[j++] = tbl[(v >> 6) & 0x3F];
45 dst[j++] = tbl[v & 0x3F];
46 } else if (i + 2 == src_len) {
47 uint32_t v = ((uint32_t)src[i] << 10) | ((uint32_t)src[i + 1] << 2);
48 dst[j++] = tbl[(v >> 12) & 0x3F];
49 dst[j++] = tbl[(v >> 6) & 0x3F];
50 dst[j++] = tbl[v & 0x3F];
51 }
52 dst[j] = '\0';
53}

84 /* Calculate how many padding chars to expect */
85 size_t padded_len = src_len;

88 }

103 }
104 out[j++] = (unsigned char)((a << 2) | (b >> 4));

106 out[j++] = (unsigned char)((c << 6) | d);
107 i += 4;
108 }
109
110 size_t rem = padded_len - i;
111 if (rem == 2) {
112 signed char a = lut[(unsigned char)src[i]];
113 signed char b = lut[(unsigned char)src[i + 1]];
114 if (a < 0 || b < 0) {
115 cwist_free(out);
116 return NULL;
117 }
118 out[j++] = (unsigned char)((a << 2) | (b >> 4));
119 } else if (rem == 3) {
120 signed char a = lut[(unsigned char)src[i]];
121 signed char b = lut[(unsigned char)src[i + 1]];
122 signed char c = lut[(unsigned char)src[i + 2]];
123 if (a < 0 || b < 0 || c < 0) {

125 return NULL;
126 }
127 out[j++] = (unsigned char)((a << 2) | (b >> 4));
128 out[j++] = (unsigned char)((b << 4) | (c >> 2));
129 } else if (rem == 1) {
130 /* RFC 4648 section 4: 1 base64 character (6 bits) cannot encode a whole byte.
131 * A remainder of 1 is malformed and must be rejected. */
132 cwist_free(out);
133 return NULL;
134 }
135
136 out[j] = '\0';
137 *out_len = j;
138 return out;

146static bool hmac_sha256(const char *key, size_t key_len, const char *msg, size_t msg_len,
147 unsigned char out[32]) {
148 unsigned int out_len = 0;
149 unsigned char *r =
150 HMAC(EVP_sha256(), key, (int)key_len, (const unsigned char *)msg, msg_len, out, &out_len);

261cwist_jwt_claims *cwist_jwt_verify(const char *token, const char *secret) {
262 if (!token || !secret) return NULL;
263
264 /* --- Split the token into three parts --------------------------------- */
265 /* We need mutable copies to split on '.' */
266 char *tok_copy = cwist_strdup(token);
267 if (!tok_copy) return NULL;
268
269 char *dot1 = strchr(tok_copy, '.');
270 if (!dot1) {
271 cwist_free(tok_copy);
272 return NULL;
273 }
274 *dot1 = '\0';
275
276 char *dot2 = strchr(dot1 + 1, '.');
277 if (!dot2) {
278 cwist_free(tok_copy);
279 return NULL;
280 }
281 *dot2 = '\0';
282
283 /* --- Validate the header: must declare alg=HS256 (RFC 8725 section 3.1) ----- */
284 size_t hdr_json_len = 0;
285 unsigned char *hdr_json = b64url_decode(tok_copy, strlen(tok_copy), &hdr_json_len);
286 if (!hdr_json) {
287 cwist_free(tok_copy);
288 return NULL;
289 }
290 cJSON *hdr = cJSON_ParseWithLength((const char *)hdr_json, hdr_json_len);
291 cwist_free(hdr_json);
292 if (!hdr) {
293 cwist_free(tok_copy);
294 return NULL;
295 }
296 cJSON *alg_item = cJSON_GetObjectItemCaseSensitive(hdr, "alg");
297 bool alg_ok =
298 alg_item && cJSON_IsString(alg_item) && strcmp(alg_item->valuestring, "HS256") == 0;
299 cJSON_Delete(hdr);
300 if (!alg_ok) {
301 cwist_free(tok_copy);
302 return NULL;
303 }
304
305 const char *pay_enc = dot1 + 1;
306 const char *sig_enc = dot2 + 1;
307
308 /* --- Re-build the signing input from the original token --------------- */
309 /* signing_input = original "hdr_enc.pay_enc" (up to the second dot) */
310 size_t first_two_len = (size_t)(dot2 - tok_copy);
311 /* dot2 points inside tok_copy which is already modified; use original */
312 cwist_sstring *signing_input = cwist_sstring_create();
313 if (!signing_input) {
314 cwist_free(tok_copy);
315 return NULL;
316 }
317 cwist_sstring_assign_len(signing_input, token, first_two_len);
318
319 /* --- Recompute expected signature ------------------------------------- */
320 unsigned char expected_sig[32];
321 if (!hmac_sha256(secret, strlen(secret), signing_input->data, signing_input->size,
322 expected_sig)) {
323 cwist_sstring_destroy(signing_input);
324 cwist_free(tok_copy);
325 return NULL;
326 }
327 cwist_sstring_destroy(signing_input);
328
329 /* --- Decode the provided signature ------------------------------------ */
330 size_t provided_sig_len = 0;
331 unsigned char *provided_sig = b64url_decode(sig_enc, strlen(sig_enc), &provided_sig_len);
332 if (!provided_sig || provided_sig_len != 32) {
333 cwist_free(tok_copy);
334 cwist_free(provided_sig);
335 return NULL;
336 }
337
338 /* --- Constant-time comparison ---------------------------------------- */
339 int cmp = 0;
340 for (int i = 0; i < 32; i++) {
341 cmp |= expected_sig[i] ^ provided_sig[i];
342 }
343 cwist_free(provided_sig);
344
345 if (cmp != 0) {
346 cwist_free(tok_copy);
347 return NULL; /* signature mismatch */
348 }
349
350 /* --- Decode and parse the payload ------------------------------------- */
351 size_t pay_json_len = 0;
352 unsigned char *pay_json = b64url_decode(pay_enc, strlen(pay_enc), &pay_json_len);
353 cwist_free(tok_copy);
354
355 if (!pay_json) return NULL;
356
357 cJSON *json = cJSON_ParseWithLength((const char *)pay_json, pay_json_len);
358 cwist_free(pay_json);
359
360 if (!json) return NULL;
361
362 /* --- Validate "exp" and "nbf" claims with leeway for clock skew/high-RTT --- */
363#define CWIST_JWT_TIME_LEEWAY 300
364 time_t now = time(NULL);
365 cJSON *exp_item = cJSON_GetObjectItemCaseSensitive(json, "exp");
366 if (exp_item && cJSON_IsNumber(exp_item)) {
367 time_t exp_time = (time_t)exp_item->valuedouble;
368 if (now > CWIST_JWT_TIME_LEEWAY && exp_time < now - CWIST_JWT_TIME_LEEWAY) {
369 cJSON_Delete(json);
370 return NULL; /* token expired */
371 }
372 }
373 cJSON *nbf_item = cJSON_GetObjectItemCaseSensitive(json, "nbf");
374 if (nbf_item && cJSON_IsNumber(nbf_item)) {
375 time_t nbf_time = (time_t)nbf_item->valuedouble;
376 if (nbf_time > now && (nbf_time - now) > CWIST_JWT_TIME_LEEWAY) {
377 cJSON_Delete(json);
378 return NULL; /* token not yet valid */
379 }
380 }
381
382 /* --- Build and return claims object ----------------------------------- */
383 cwist_jwt_claims *claims = (cwist_jwt_claims *)cwist_alloc(sizeof(cwist_jwt_claims));
384 if (!claims) {
385 cJSON_Delete(json);
386 return NULL;
387 }
388 claims->json = json;
389 return claims;
390}
