247void cwist_full_gc(bool enable) {
248 if (!g_full_gc_guard) return; /* guard page unavailable; fail safe, not silently unhardened */
249
250 /* Fast path: once locked, this plain load is the only thing every
251 * later caller ever does -- safe indefinitely, even after the page
252 * below becomes read-only. */
253 if (atomic_load_explicit(&g_full_gc_guard->locked, memory_order_acquire)) return;
254
255 pthread_mutex_lock(&g_full_gc_claim_mu);
256 if (atomic_load_explicit(&g_full_gc_guard->locked, memory_order_acquire)) {
257 /* Someone else claimed it while we were waiting for the mutex. */
258 pthread_mutex_unlock(&g_full_gc_claim_mu);
259 return;
260 }
261
262 /* We hold the mutex and locked is still false: nobody has mprotect()'d
263 * the page yet, so it is still writable and we are its sole writer. */
264 atomic_store_explicit(&g_full_gc_guard->enabled, enable, memory_order_relaxed);
265 atomic_store_explicit(&g_full_gc_guard->locked, true, memory_order_release);
266 long page_size = sysconf(_SC_PAGESIZE);
267 mprotect(g_full_gc_guard, (size_t)(page_size > 0 ? page_size : 4096), PROT_READ);
268 pthread_mutex_unlock(&g_full_gc_claim_mu);
269
270 cwist_gc_auto_rotate(cwist_full_gc_instance(), enable);
271}

332static inline size_t cwist_gc_pending_slot(const void *ptr, size_t mask) {
333 uint64_t h = (uint64_t)((uintptr_t)ptr >> 4) * UINT64_C(0x9E3779B97F4A7C15);

338static bool cwist_gc_pending_grow(cwist_gc_pending_t *pending) {
339 size_t new_cap = pending->cap ? pending->cap * 2 : CWIST_GC_PENDING_MIN_CAP;
340 void **slots = (void **)calloc(new_cap, sizeof(void *));
341 if (!slots) return false;
342 size_t mask = new_cap - 1;
343 for (size_t i = 0; i < pending->cap; i++) {
344 void *key = pending->slots[i];
345 if (!key) continue;
346 size_t j = cwist_gc_pending_slot(key, mask);
347 while (slots[j]) j = (j + 1) & mask;
348 slots[j] = key;
349 }
350 free(pending->slots);
351 pending->slots = slots;
352 pending->cap = new_cap;
353 return true;
354}

366static void cwist_gc_pending_flush(cwist_gc_pending_t *pending) {
367 if (!pending) return;
368 void **slots = pending->slots;
369 size_t cap = pending->cap;
370 size_t count = pending->count;
371 if (count == 0 && cap <= CWIST_GC_PENDING_KEEP_CAP) return;
372
373 pending->slots = NULL;
374 pending->cap = 0;
375 pending->count = 0;
376 for (size_t i = 0; i < cap && count > 0; i++) {
377 if (!slots[i]) continue;
378 cwist_ebr_free(slots[i]);
379 count--;
380 }
381 if (cap <= CWIST_GC_PENDING_KEEP_CAP && !pending->slots) {
382 memset(slots, 0, cap * sizeof(void *));
383 pending->slots = slots;
384 pending->cap = cap;
385 } else {
386 free(slots);
387 }
388}

454void cwist_gc_scope_track(void *ptr) {
455 if (!ptr) return;
456 cwist_gc_pending_t *pending = cwist_gc_pending_get();
457 if (!pending) return;
458 /* Best-effort, as before: if the table cannot grow, leave ptr untracked
459 * rather than fail the allocation. */
460 if ((pending->count + 1) * 2 > pending->cap && !cwist_gc_pending_grow(pending)) return;
461 size_t mask = pending->cap - 1;
462 size_t i = cwist_gc_pending_slot(ptr, mask);
463 while (pending->slots[i]) {
464 if (pending->slots[i] == ptr) return; /* already tracked: keep one entry */
465 i = (i + 1) & mask;
466 }
467 pending->slots[i] = ptr;
468 pending->count++;
469}

481bool cwist_gc_scope_untrack(void *ptr) {
482 if (!ptr) return false;
483 cwist_gc_pending_t *pending = cwist_gc_pending_get();
484 if (!pending || pending->count == 0) return false;
485 size_t mask = pending->cap - 1;
486 size_t i = cwist_gc_pending_slot(ptr, mask);
487 while (pending->slots[i] != ptr) {
488 if (!pending->slots[i]) return false;
489 i = (i + 1) & mask;
490 }
491
492 /* Backward-shift deletion: pull each following entry of the probe run
493 * back into the hole unless the hole lies before that entry's home
494 * slot, so every remaining key stays reachable without tombstones. */
495 size_t hole = i;
496 for (size_t j = (i + 1) & mask; pending->slots[j]; j = (j + 1) & mask) {
497 size_t home = cwist_gc_pending_slot(pending->slots[j], mask);
498 if (((j - home) & mask) >= ((j - hole) & mask)) {
499 pending->slots[hole] = pending->slots[j];
500 hole = j;
501 }
502 }
503 pending->slots[hole] = NULL;
504 pending->count--;
505 return true;
506}

600static void cwist_conn_pending_destroy(void *arg) {

602 if (!pending) return;

641 /* Register the process-exit sweep exactly once, the first time any
642 * thread ever tracks a connection (not eagerly at load time: most
643 * processes never enable full-GC, and atexit() handlers are a
644 * process-wide resource other code may also be competing for). */

646 atexit(cwist_conn_registry_atexit);
647 }

650 pending->global_next = g_conn_registry_head;
651 g_conn_registry_head = pending;
652 pthread_mutex_unlock(&g_conn_registry_lock);

655}

666void cwist_conn_registry_track(void *handle, cwist_conn_close_fn close_fn) {
667 if (!handle || !close_fn) return;
668 if (!cwist_full_gc_enabled()) return;
669 cwist_conn_pending_t *pending = cwist_conn_pending_get();
670 if (!pending) return;
671
672 pthread_mutex_lock(&pending->lock);
673 if (pending->count == pending->cap) {
674 size_t new_cap = pending->cap ? pending->cap * 2 : 8;
675 cwist_conn_entry_t *grown =
676 (cwist_conn_entry_t *)realloc(pending->items, new_cap * sizeof(*grown));
677 if (grown) {
678 pending->items = grown;
679 pending->cap = new_cap;
680 }
681 }
682 if (pending->count < pending->cap) {
683 pending->items[pending->count++] =
684 (cwist_conn_entry_t){.handle = handle, .close_fn = close_fn};
685 }
