18/* --- cwist_reg_ptr registry: CWIST-side bookkeeping for pointers handed to
19 * a GC context via cwist_reg_ptr()/cwist_reg_ptr_sized().
20 *
21 * libttak's epoch GC releases whatever is still in its mem_tree at destroy
22 * time with ttak_mem_free(), which validates the ttak block header and
23 * aborts on any pointer that does not carry one (lib/libttak/src/mem/mem.c's
24 * V_HEADER). cwist_alloc() blocks have no such header when the owner guard
25 * is off (the default), so the wrapper must detach every CWIST-registered
26 * block from the tree before destruction and release it with cwist_free()
27 * itself. This table records exactly which pointers went through the
28 * wrapper (and which context owns them) so cwist_gc_shutdown(), an explicit
29 * cwist_free(), and cross-context re-registration can do that detachment.
30 *
31 * Pointers registered directly with ttak via cwist_gc_raw() are
32 * intentionally absent here: they keep libttak's native teardown semantics
33 * (freed by ttak_mem_free() at destroy), e.g. example/rps-showcase's
34 * ttak_mem_alloc_safe() payloads. Never mix the two registration paths for
35 * the same pointer.
36 *
37 * The table, like the other registries in this file, uses the raw
38 * allocator: it is the GC's own bookkeeping and must never sit on a
39 * full-GC pending list. */
40
41typedef struct {
42 void *ptr; /* NULL marks an empty slot */
43 cwist_gc_t *owner; /* context responsible for releasing the block */
44} cwist_reg_entry_t;

57static cwist_reg_entry_t *cwist_reg_find_locked(const void *ptr) {
58 if (g_reg_count == 0) return NULL;
59 size_t mask = g_reg_cap - 1;
60 for (size_t i = cwist_reg_slot(ptr, mask);; i = (i + 1) & mask) {
61 if (!g_reg_slots[i].ptr) return NULL;
62 if (g_reg_slots[i].ptr == ptr) return &g_reg_slots[i];
63 }

67static void cwist_reg_remove_locked(cwist_reg_entry_t *e) {
68 size_t mask = g_reg_cap - 1;
69 size_t hole = (size_t)(e - g_reg_slots);
70 for (size_t j = (hole + 1) & mask; g_reg_slots[j].ptr; j = (j + 1) & mask) {
71 size_t home = cwist_reg_slot(g_reg_slots[j].ptr, mask);
72 if (((j - home) & mask) >= ((j - hole) & mask)) {
73 g_reg_slots[hole] = g_reg_slots[j];
74 hole = j;
75 }
76 }
77 g_reg_slots[hole] = (cwist_reg_entry_t){0};
78 g_reg_count--;
79}

82static bool cwist_reg_reserve_locked(void) {
83 if ((g_reg_count + 1) * 2 <= g_reg_cap) return true;
84 size_t new_cap = g_reg_cap ? g_reg_cap * 2 : 64;
85 cwist_reg_entry_t *slots = (cwist_reg_entry_t *)calloc(new_cap, sizeof(*slots));
86 if (!slots) return false;
87 size_t mask = new_cap - 1;
88 for (size_t i = 0; i < g_reg_cap; i++) {
89 if (!g_reg_slots[i].ptr) continue;
90 size_t j = cwist_reg_slot(g_reg_slots[i].ptr, mask);
91 while (slots[j].ptr) j = (j + 1) & mask;
92 slots[j] = g_reg_slots[i];
93 }
94 free(g_reg_slots);
95 g_reg_slots = slots;
96 g_reg_cap = new_cap;
97 return true;
98}

260void cwist_reg_ptr_sized(cwist_gc_t *gc, void *ptr, size_t size) {
261 if (!gc || !gc->initialized || !ptr) return;
262
263 pthread_mutex_lock(&g_reg_lock);
264 cwist_reg_entry_t *e = cwist_reg_find_locked(ptr);
265 if (e) {
266 if (e->owner == gc) {
267 /* Already tracked by this context: skip a duplicate ttak tree
268 * node, which destroy would hand to ttak_mem_free() twice. */
269 pthread_mutex_unlock(&g_reg_lock);
270 return;
271 }
272 /* Registration moves to a new context: drop the pointer from the
273 * previous owner first, so that context's shutdown neither frees
274 * the block nor trips over it. */
275 cwist_gc_t *prev = e->owner;
276 cwist_reg_remove_locked(e);
277 pthread_mutex_unlock(&g_reg_lock);
278 if (prev && prev->initialized) {
279 ttak_mem_node_t *node = ttak_mem_tree_find_node(&prev->impl.tree, ptr);
280 if (node) ttak_mem_tree_remove(&prev->impl.tree, node);
281 }
282 pthread_mutex_lock(&g_reg_lock);
283 if (cwist_reg_find_locked(ptr)) {
284 /* Another thread registered it in the meantime; that
285 * registration stands. */
286 pthread_mutex_unlock(&g_reg_lock);
287 return;
288 }
289 }
290 bool tracked = cwist_reg_reserve_locked();
291 if (tracked) {
292 size_t mask = g_reg_cap - 1;
293 size_t i = cwist_reg_slot(ptr, mask);
294 while (g_reg_slots[i].ptr) i = (i + 1) & mask;
295 g_reg_slots[i] = (cwist_reg_entry_t){.ptr = ptr, .owner = gc};
296 g_reg_count++;
297 }
298 pthread_mutex_unlock(&g_reg_lock);
299
300 /* Best-effort, matching cwist_gc_scope_track(): when the registry
301 * itself cannot record the pointer, leave it untracked rather than
302 * register it with ttak, whose destroy-time free would abort on a
303 * headerless cwist_alloc block. */
304 if (!tracked) return;
305 ttak_epoch_gc_register(&gc->impl, ptr, size);
306}

470static pthread_mutex_t g_full_gc_claim_mu = PTHREAD_MUTEX_INITIALIZER;

482void cwist_full_gc(bool enable) {
483 if (!g_full_gc_guard) return; /* guard page unavailable; fail safe, not silently unhardened */
484
485 /* Fast path: once locked, this plain load is the only thing every
486 * later caller ever does -- safe indefinitely, even after the page
487 * below becomes read-only. */
488 if (atomic_load_explicit(&g_full_gc_guard->locked, memory_order_acquire)) return;
489
490 pthread_mutex_lock(&g_full_gc_claim_mu);
491 if (atomic_load_explicit(&g_full_gc_guard->locked, memory_order_acquire)) {
492 /* Someone else claimed it while we were waiting for the mutex. */
493 pthread_mutex_unlock(&g_full_gc_claim_mu);
494 return;
495 }
496
497 /* We hold the mutex and locked is still false: nobody has mprotect()'d
498 * the page yet, so it is still writable and we are its sole writer. */
499 atomic_store_explicit(&g_full_gc_guard->enabled, enable, memory_order_relaxed);
500 atomic_store_explicit(&g_full_gc_guard->locked, true, memory_order_release);
501#if !defined(__EMSCRIPTEN__) && !defined(__wasi__)
502 long page_size = sysconf(_SC_PAGESIZE);
503 mprotect(g_full_gc_guard, (size_t)(page_size > 0 ? page_size : 4096), PROT_READ);
504#endif
505 pthread_mutex_unlock(&g_full_gc_claim_mu);
506
507 cwist_gc_auto_rotate(cwist_full_gc_instance(), enable);
508}

575static bool cwist_gc_pending_grow(cwist_gc_pending_t *pending) {
576 size_t new_cap = pending->cap ? pending->cap * 2 : CWIST_GC_PENDING_MIN_CAP;
577 void **slots = (void **)calloc(new_cap, sizeof(void *));
578 if (!slots) return false;
579 size_t mask = new_cap - 1;
580 for (size_t i = 0; i < pending->cap; i++) {
581 void *key = pending->slots[i];
582 if (!key) continue;
583 size_t j = cwist_gc_pending_slot(key, mask);
584 while (slots[j]) j = (j + 1) & mask;
585 slots[j] = key;
586 }
587 free(pending->slots);
588 pending->slots = slots;
589 pending->cap = new_cap;
590 return true;
591}

603static void cwist_gc_pending_flush(cwist_gc_pending_t *pending) {
604 if (!pending) return;
605 void **slots = pending->slots;
606 size_t cap = pending->cap;
607 size_t count = pending->count;
608 if (count == 0 && cap <= CWIST_GC_PENDING_KEEP_CAP) return;
609
610 pending->slots = NULL;
611 pending->cap = 0;
612 pending->count = 0;
613 size_t retired = 0;
614 for (size_t i = 0; i < cap && count > 0; i++) {
615 if (!slots[i]) continue;
616 cwist_ebr_free(slots[i]);
617 count--;
618 retired++;
619 }
620 if (cap <= CWIST_GC_PENDING_KEEP_CAP && !pending->slots) {
621 memset(slots, 0, cap * sizeof(void *));
622 pending->slots = slots;
623 pending->cap = cap;
624 } else {
625 free(slots);
626 }
627 /* Drive reclaim here: the full-GC instance rotates manually, so nothing
628 * else frees what was just retired unless the app happens to run an
629 * io_queue (#347). Every retired block was tracked by this thread and,
630 * by the cwist_gc_scope_flush() contract, is no longer in use, so
631 * freeing it once the grace period has passed is safe.
632 * Skipped when nothing was retired, so an io_queue job that freed
633 * everything itself pays nothing. */
634 if (retired > 0) {
635 /* libttak keeps a retired block for TTAK_EPOCH_GRACE epochs, and
636 * each reclaim advances at most one: with no reader inside an epoch
637 * this many calls free what was just retired. */
638 for (unsigned i = 0; i < TTAK_EPOCH_GRACE; i++) ttak_epoch_reclaim();
639 }
640}

706void cwist_gc_scope_track(void *ptr) {
707 if (!ptr) return;
708 cwist_gc_pending_t *pending = cwist_gc_pending_get();
709 if (!pending) return;
710 /* Best-effort, as before: if the table cannot grow, leave ptr untracked
711 * rather than fail the allocation. */
712 if ((pending->count + 1) * 2 > pending->cap && !cwist_gc_pending_grow(pending)) return;
713 size_t mask = pending->cap - 1;
714 size_t i = cwist_gc_pending_slot(ptr, mask);
715 while (pending->slots[i]) {
716 if (pending->slots[i] == ptr) return; /* already tracked: keep one entry */
717 i = (i + 1) & mask;
718 }
719 pending->slots[i] = ptr;
720 pending->count++;
721}

733bool cwist_gc_scope_untrack(void *ptr) {
734 if (!ptr) return false;
735 cwist_gc_pending_t *pending = cwist_gc_pending_get();
736 if (!pending || pending->count == 0) return false;
737 size_t mask = pending->cap - 1;
738 size_t i = cwist_gc_pending_slot(ptr, mask);
739 while (pending->slots[i] != ptr) {
740 if (!pending->slots[i]) return false;
741 i = (i + 1) & mask;
742 }
743
744 /* Backward-shift deletion: pull each following entry of the probe run
745 * back into the hole unless the hole lies before that entry's home
746 * slot, so every remaining key stays reachable without tombstones. */
747 size_t hole = i;
748 for (size_t j = (i + 1) & mask; pending->slots[j]; j = (j + 1) & mask) {
749 size_t home = cwist_gc_pending_slot(pending->slots[j], mask);
750 if (((j - home) & mask) >= ((j - hole) & mask)) {
751 pending->slots[hole] = pending->slots[j];
752 hole = j;
753 }
754 }
755 pending->slots[hole] = NULL;
756 pending->count--;
757 return true;
758}

788/* --- Connection registry: thread/process-exit sweep for non-memory
789 * resources (sockets, TLS sessions, protocol state). See gc.h's
790 * cwist_conn_registry_* doc comments for the contract.
791 *
792 * One process-wide table maps each tracked handle to its close_fn and its
793 * owner, the thread whose exit sweep closes it (#345). The owner starts as
794 * the tracking thread and moves with cwist_conn_registry_adopt() when
795 * another thread takes over serving the connection, so a thread that exits
796 * no longer closes connections other threads are still using. Untrack is a
797 * hash delete instead of a walk over every thread's list, and close_fn
798 * always runs with no registry lock held, so a close_fn may track or
799 * untrack other handles.
800 *
801 * The table, batches and owner tokens use the raw allocator: they are the
802 * GC's own bookkeeping and must never sit on a full-GC pending list. */
803
805typedef struct cwist_conn_owner {
806 char unused;
807} cwist_conn_owner_t;
808
809typedef struct {
810 void *handle; /* NULL marks an empty slot */
811 cwist_conn_close_fn close_fn;
812 cwist_conn_owner_t *owner;
813} cwist_conn_entry_t;
814

817static pthread_mutex_t g_conn_registry_lock = PTHREAD_MUTEX_INITIALIZER;
818static cwist_conn_entry_t *g_conn_slots = NULL; /* open addressing, power-of-two cap */
819static size_t g_conn_cap = 0;
820static size_t g_conn_count = 0;
821static _Atomic bool g_conn_hooks_registered = false;

823static size_t cwist_conn_slot(const void *handle, size_t mask) {
824 uintptr_t x = (uintptr_t)handle;
825 x ^= x >> 33;
826 x *= (uintptr_t)0xff51afd7ed558ccdULL;
827 x ^= x >> 33;
828 return (size_t)x & mask;
829}

842static void cwist_conn_remove_at_locked(size_t i) {
843 size_t mask = g_conn_cap - 1;
844 size_t hole = i;
845 for (size_t j = (i + 1) & mask; g_conn_slots[j].handle; j = (j + 1) & mask) {
846 size_t home = cwist_conn_slot(g_conn_slots[j].handle, mask);
847 if (((j - home) & mask) >= ((j - hole) & mask)) {
848 g_conn_slots[hole] = g_conn_slots[j];
849 hole = j;
850 }
851 }
852 g_conn_slots[hole] = (cwist_conn_entry_t){0};
853 g_conn_count--;
854}

857static bool cwist_conn_reserve_locked(void) {
858 if ((g_conn_count + 1) * 2 <= g_conn_cap) return true;
859 size_t new_cap = g_conn_cap ? g_conn_cap * 2 : 64;
860 cwist_conn_entry_t *slots = (cwist_conn_entry_t *)calloc(new_cap, sizeof(*slots));
861 if (!slots) return false;
862 size_t mask = new_cap - 1;
863 for (size_t i = 0; i < g_conn_cap; i++) {
864 if (!g_conn_slots[i].handle) continue;
865 size_t j = cwist_conn_slot(g_conn_slots[i].handle, mask);
866 while (slots[j].handle) j = (j + 1) & mask;
867 slots[j] = g_conn_slots[i];
868 }
869 free(g_conn_slots);
870 g_conn_slots = slots;
871 g_conn_cap = new_cap;
872 return true;
873}

883static void cwist_conn_close_owned(cwist_conn_owner_t *owner) {
884 for (;;) {
885 pthread_mutex_lock(&g_conn_registry_lock);
886 size_t want = 0;
887 for (size_t i = 0; i < g_conn_cap; i++) {
888 if (g_conn_slots[i].handle && (!owner || g_conn_slots[i].owner == owner)) want++;
889 }
890 if (want == 0) {
891 pthread_mutex_unlock(&g_conn_registry_lock);
892 return;
893 }
894 cwist_conn_entry_t one;
895 cwist_conn_entry_t *heap = (cwist_conn_entry_t *)calloc(want, sizeof(*heap));
896 cwist_conn_entry_t *batch = heap ? heap : &one;
897 size_t room = heap ? want : 1;
898 size_t n = 0;
899 for (size_t i = 0; i < g_conn_cap && n < room;) {
900 if (g_conn_slots[i].handle && (!owner || g_conn_slots[i].owner == owner)) {
901 batch[n++] = g_conn_slots[i];
902 cwist_conn_remove_at_locked(i); /* may shift a later entry into i */
903 } else {
904 i++;
905 }
906 }
907 pthread_mutex_unlock(&g_conn_registry_lock);
908 for (size_t k = 0; k < n; k++) batch[k].close_fn(batch[k].handle);
909 free(heap);
910 }
911}

948static cwist_conn_owner_t *cwist_conn_owner_get(bool create) {
949 pthread_once(&g_conn_key_once, cwist_conn_key_init);
950 cwist_conn_owner_t *owner = (cwist_conn_owner_t *)pthread_getspecific(g_conn_key);
951 if (owner || !create) return owner;
952
953 owner = (cwist_conn_owner_t *)calloc(1, sizeof(*owner));
954 if (!owner) return NULL;
955 if (pthread_setspecific(g_conn_key, owner) != 0) {
956 free(owner);
957 return NULL;
958 }
959
960 /* Register the process-exit sweep and the fork handler exactly once,
961 * the first time any thread tracks a connection (not eagerly at load
962 * time: most processes never enable full-GC). */
963 if (!atomic_exchange_explicit(&g_conn_hooks_registered, true, memory_order_acq_rel)) {
964 atexit(cwist_conn_registry_atexit);
965 pthread_atfork(cwist_conn_atfork_prepare, cwist_conn_atfork_parent, cwist_conn_atfork_child);
966 }
967 return owner;
968}

981void cwist_conn_registry_track(void *handle, cwist_conn_close_fn close_fn) {
982 if (!handle || !close_fn) return;
983 if (!cwist_full_gc_enabled()) return;
984 cwist_conn_owner_t *owner = cwist_conn_owner_get(true);
985 if (!owner) return;
986
987 pthread_mutex_lock(&g_conn_registry_lock);
988 size_t i = cwist_conn_find_locked(handle);
989 if (i != SIZE_MAX) {
990 g_conn_slots[i].close_fn = close_fn;
991 g_conn_slots[i].owner = owner;
992 } else if (cwist_conn_reserve_locked()) {
993 size_t mask = g_conn_cap - 1;
994 size_t j = cwist_conn_slot(handle, mask);
995 while (g_conn_slots[j].handle) j = (j + 1) & mask;
996 g_conn_slots[j] = (cwist_conn_entry_t){.handle = handle, .close_fn = close_fn, .owner = owner};
997 g_conn_count++;
998 }
999 pthread_mutex_unlock(&g_conn_registry_lock);
1000}
