195__attribute__((constructor)) static void cwist_full_gc_guard_init(void) {
196#if defined(__wasi__)
197 /* WASI has no mmap: leave the guard NULL so cwist_full_gc_enabled()
198 * reports full-GC as permanently unavailable (the same fail-safe the
199 * MAP_FAILED path reaches natively). */
200 (void)g_full_gc_guard;
201 return;
202#else
203 long page_size = sysconf(_SC_PAGESIZE);
204 if (page_size <= 0) page_size = 4096;
205 void *page =
206 mmap(NULL, (size_t)page_size, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
207 if (page == MAP_FAILED) return;
208 cwist_full_gc_guard_t *guard = (cwist_full_gc_guard_t *)page;
209 atomic_init(&guard->enabled, false);
210 atomic_init(&guard->locked, false);
211 g_full_gc_guard = guard;
212#endif
213}

257void cwist_full_gc(bool enable) {
258 if (!g_full_gc_guard) return; /* guard page unavailable; fail safe, not silently unhardened */
259
260 /* Fast path: once locked, this plain load is the only thing every
261 * later caller ever does -- safe indefinitely, even after the page
262 * below becomes read-only. */
263 if (atomic_load_explicit(&g_full_gc_guard->locked, memory_order_acquire)) return;
264
265 pthread_mutex_lock(&g_full_gc_claim_mu);
266 if (atomic_load_explicit(&g_full_gc_guard->locked, memory_order_acquire)) {
267 /* Someone else claimed it while we were waiting for the mutex. */
268 pthread_mutex_unlock(&g_full_gc_claim_mu);
269 return;
270 }
271
272 /* We hold the mutex and locked is still false: nobody has mprotect()'d
273 * the page yet, so it is still writable and we are its sole writer. */
274 atomic_store_explicit(&g_full_gc_guard->enabled, enable, memory_order_relaxed);
275 atomic_store_explicit(&g_full_gc_guard->locked, true, memory_order_release);
276#if !defined(__wasi__)
277 /* Unreachable under WASI (the guard is never mapped there), where
278 * mprotect does not exist. */
279 long page_size = sysconf(_SC_PAGESIZE);
280 mprotect(g_full_gc_guard, (size_t)(page_size > 0 ? page_size : 4096), PROT_READ);
281#endif
282 pthread_mutex_unlock(&g_full_gc_claim_mu);
283
284 cwist_gc_auto_rotate(cwist_full_gc_instance(), enable);
285}

352static bool cwist_gc_pending_grow(cwist_gc_pending_t *pending) {
353 size_t new_cap = pending->cap ? pending->cap * 2 : CWIST_GC_PENDING_MIN_CAP;
354 void **slots = (void **)calloc(new_cap, sizeof(void *));
355 if (!slots) return false;
356 size_t mask = new_cap - 1;
357 for (size_t i = 0; i < pending->cap; i++) {
358 void *key = pending->slots[i];
359 if (!key) continue;
360 size_t j = cwist_gc_pending_slot(key, mask);
361 while (slots[j]) j = (j + 1) & mask;
362 slots[j] = key;
363 }
364 free(pending->slots);
365 pending->slots = slots;
366 pending->cap = new_cap;
367 return true;
368}

380static void cwist_gc_pending_flush(cwist_gc_pending_t *pending) {
381 if (!pending) return;
382 void **slots = pending->slots;
383 size_t cap = pending->cap;
384 size_t count = pending->count;
385 if (count == 0 && cap <= CWIST_GC_PENDING_KEEP_CAP) return;
386
387 pending->slots = NULL;
388 pending->cap = 0;
389 pending->count = 0;
390 for (size_t i = 0; i < cap && count > 0; i++) {
391 if (!slots[i]) continue;
392 cwist_ebr_free(slots[i]);
393 count--;
394 }
395 if (cap <= CWIST_GC_PENDING_KEEP_CAP && !pending->slots) {
396 memset(slots, 0, cap * sizeof(void *));
397 pending->slots = slots;
398 pending->cap = cap;
399 } else {
400 free(slots);
401 }
402}

468void cwist_gc_scope_track(void *ptr) {
469 if (!ptr) return;
470 cwist_gc_pending_t *pending = cwist_gc_pending_get();
471 if (!pending) return;
472 /* Best-effort, as before: if the table cannot grow, leave ptr untracked
473 * rather than fail the allocation. */
474 if ((pending->count + 1) * 2 > pending->cap && !cwist_gc_pending_grow(pending)) return;
475 size_t mask = pending->cap - 1;
476 size_t i = cwist_gc_pending_slot(ptr, mask);
477 while (pending->slots[i]) {
478 if (pending->slots[i] == ptr) return; /* already tracked: keep one entry */
479 i = (i + 1) & mask;
480 }
481 pending->slots[i] = ptr;
482 pending->count++;
483}

495bool cwist_gc_scope_untrack(void *ptr) {
496 if (!ptr) return false;
497 cwist_gc_pending_t *pending = cwist_gc_pending_get();
498 if (!pending || pending->count == 0) return false;
499 size_t mask = pending->cap - 1;
500 size_t i = cwist_gc_pending_slot(ptr, mask);
501 while (pending->slots[i] != ptr) {
502 if (!pending->slots[i]) return false;
503 i = (i + 1) & mask;
504 }
505
506 /* Backward-shift deletion: pull each following entry of the probe run
507 * back into the hole unless the hole lies before that entry's home
508 * slot, so every remaining key stays reachable without tombstones. */
509 size_t hole = i;
510 for (size_t j = (i + 1) & mask; pending->slots[j]; j = (j + 1) & mask) {
511 size_t home = cwist_gc_pending_slot(pending->slots[j], mask);
512 if (((j - home) & mask) >= ((j - hole) & mask)) {
513 pending->slots[hole] = pending->slots[j];
514 hole = j;
515 }
516 }
517 pending->slots[hole] = NULL;
518 pending->count--;
519 return true;
520}

614static void cwist_conn_pending_destroy(void *arg) {
615 cwist_conn_pending_t *pending = (cwist_conn_pending_t *)arg;
616 if (!pending) return;
617
618 pthread_mutex_lock(&pending->lock);
619 cwist_conn_pending_sweep_locked(pending);

641 if (pending) return pending;

680void cwist_conn_registry_track(void *handle, cwist_conn_close_fn close_fn) {
681 if (!handle || !close_fn) return;
682 if (!cwist_full_gc_enabled()) return;
683 cwist_conn_pending_t *pending = cwist_conn_pending_get();
684 if (!pending) return;
685
686 pthread_mutex_lock(&pending->lock);
687 if (pending->count == pending->cap) {
688 size_t new_cap = pending->cap ? pending->cap * 2 : 8;
689 cwist_conn_entry_t *grown =
690 (cwist_conn_entry_t *)realloc(pending->items, new_cap * sizeof(*grown));
691 if (grown) {
692 pending->items = grown;
693 pending->cap = new_cap;
694 }
695 }
696 if (pending->count < pending->cap) {
697 pending->items[pending->count++] =
698 (cwist_conn_entry_t){.handle = handle, .close_fn = close_fn};
699 }
700 pthread_mutex_unlock(&pending->lock);
701}
