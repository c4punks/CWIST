50static int orm_recv_all(int fd, void *buf, size_t n) {
51 size_t total = 0;
52 char *p = (char *)buf;
53 while (total < n) {
54 ssize_t r = recv(fd, p + total, n - total, MSG_WAITALL);
55 if (r <= 0) return -1;
56 total += (size_t)r;
57 }
58 return 0;
59}
60
65static int orm_send_all(int fd, const void *buf, size_t n) {
66 size_t total = 0;
67 const char *p = (const char *)buf;
68 while (total < n) {
69 ssize_t w = send(fd, p + total, n - total, 0);
70 if (w <= 0) return -1;
71 total += (size_t)w;
72 }
73 return 0;
74}

88static int orm_exchange(cwist_orm_t *orm, const char *sql, char **out_payload) {
89 *out_payload = NULL;
90 if (!orm || orm->sock_fd < 0 || !sql) return -1;
91
92 size_t sql_len = strlen(sql);
93 if (sql_len > UINT32_MAX - 1) return -1;
94
95 uint32_t net_len = htonl((uint32_t)sql_len);
96 if (orm_send_all(orm->sock_fd, &net_len, sizeof(net_len)) != 0) return -1;
97 if (orm_send_all(orm->sock_fd, sql, sql_len) != 0) return -1;
98
99 uint32_t net_payload = 0;
100 if (orm_recv_all(orm->sock_fd, &net_payload, sizeof(net_payload)) != 0) return -1;
101 uint32_t payload_len = ntohl(net_payload);
102 if (payload_len > 64 * 1024 * 1024) return -1; /* sanity ceiling */
103
104 char *payload = (char *)malloc(payload_len + 1);
105 if (!payload) return -1;
106 if (payload_len > 0 && orm_recv_all(orm->sock_fd, payload, payload_len) != 0) {
107 free(payload);
108 return -1;
109 }
110 payload[payload_len] = '\0';
111 *out_payload = payload;
112 return 0;
113}

163void cwist_orm_close_socket(cwist_orm_t *orm) {
164 if (!orm) return;
165 if (orm->sock_fd >= 0) {
166 /* Best-effort rollback of any dangling transaction. */
167 char *dummy = NULL;
168 (void)orm_exchange(orm, "ROLLBACK;", &dummy);
169 free(dummy);

171 }

241cwist_error_t cwist_orm_exec(cwist_orm_t *orm, const char *sql) {
242 cwist_error_t err = make_error(CWIST_ERR_INT16);
243 if (!orm || !sql) {
244 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
245 return err;
246 }
247
248 char *payload = NULL;
249 if (orm_exchange(orm, sql, &payload) != 0) {
250 err.error.err_i16 = CWIST_ERROR_IO;
251 return err;
252 }
253
254 int status = orm_parse_status(payload);
255 free(payload);
256
257 if (status != 0) {
258 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
259 return err;
260 }
261
262 /* Auto-commit write helpers if requested. */
263 if (g_immediate_commit) {
264 const char *upper = sql;
265 while (*upper && *upper <= ' ') upper++;
266 if ((strncasecmp(upper, "INSERT", 6) == 0) || (strncasecmp(upper, "UPDATE", 6) == 0) ||
267 (strncasecmp(upper, "DELETE", 6) == 0)) {
268 char *commit_payload = NULL;
269 (void)orm_exchange(orm, "COMMIT;", &commit_payload);
270 free(commit_payload);
271 }
272 }
273
274 err.error.err_i16 = CWIST_SUCCESS;
275 return err;
276}

343static char *cwist_orm_escape_sqlite(const char *src) {
344 if (!src) return NULL;
345 size_t len = strlen(src);
346 size_t extra = 0;
347 for (size_t i = 0; i < len; i++) {
348 if (src[i] == '\'') extra++;
349 }
350 char *dst = (char *)malloc(len + extra + 1);
351 if (!dst) return NULL;
352 size_t j = 0;
353 for (size_t i = 0; i < len; i++) {
354 if (src[i] == '\'') dst[j++] = '\'';
355 dst[j++] = src[i];
356 }
357 dst[j] = '\0';
358 return dst;
359}

370static char *cwist_orm_quote_identifier(const cwist_orm_t *orm, const char *id) {
371 if (!orm || !id) return NULL;
372 char quote_ch = '\0';
373 char escape_ch = '\0';
374 switch (orm->dialect) {
375 case CWIST_ORM_POSTGRES:
376 quote_ch = '"';
377 escape_ch = '"';
378 break;
379 case CWIST_ORM_MYSQL:
380 case CWIST_ORM_MARIADB:
381 case CWIST_ORM_SQLITE:
382 quote_ch = '`';
383 escape_ch = '`';
384 break;
385 default: return strdup(id);
386 }
387
388 size_t len = strlen(id);
389 size_t extra = 0;
390 for (size_t i = 0; i < len; i++) {
391 if (id[i] == quote_ch) extra++;
392 }
393 char *out = (char *)malloc(len + extra + 3);
394 if (!out) return NULL;
395 size_t j = 0;
396 out[j++] = quote_ch;
397 for (size_t i = 0; i < len; i++) {
398 if (id[i] == quote_ch) out[j++] = escape_ch;
399 out[j++] = id[i];
400 }
401 out[j++] = quote_ch;
402 out[j] = '\0';
403 return out;
404}

418static char *cwist_orm_json_to_sql_literal(const cwist_orm_t *orm, const cJSON *node) {
419 if (!node) return strdup("NULL");
420
421 if (cJSON_IsNull(node)) {
422 return strdup("NULL");
423 }
424 if (cJSON_IsBool(node)) {
425 if (orm && orm->dialect == CWIST_ORM_POSTGRES) {
426 return strdup(cJSON_IsTrue(node) ? "TRUE" : "FALSE");
427 }
428 return strdup(cJSON_IsTrue(node) ? "1" : "0");
429 }
430 if (cJSON_IsNumber(node)) {
431 char buf[64];
432 snprintf(buf, sizeof(buf), "%g", node->valuedouble);
433 return strdup(buf);
434 }
435 if (cJSON_IsString(node)) {
436 if (!node->valuestring) return strdup("''");
437 char *escaped = cwist_orm_escape_sqlite(node->valuestring);
438 if (!escaped) return NULL;
439 size_t elen = strlen(escaped);
440 char *quoted = (char *)malloc(elen + 3);
441 if (!quoted) {
442 free(escaped);
443 return NULL;
444 }
445 quoted[0] = '\'';
446 memcpy(quoted + 1, escaped, elen);
447 quoted[elen + 1] = '\'';
448 quoted[elen + 2] = '\0';
449 free(escaped);
450 return quoted;
451 }
452
453 /* Fallback: stringify arrays/objects as JSON text. */
454 char *raw = cJSON_PrintUnformatted((cJSON *)node);
455 if (!raw) return strdup("NULL");
456 char *escaped = cwist_orm_escape_sqlite(raw);
457 free(raw);
458 if (!escaped) return NULL;
459 size_t elen = strlen(escaped);
460 char *quoted = (char *)malloc(elen + 3);
461 if (!quoted) {
462 free(escaped);
463 return NULL;
464 }
465 quoted[0] = '\'';
466 memcpy(quoted + 1, escaped, elen);
467 quoted[elen + 1] = '\'';
468 quoted[elen + 2] = '\0';
469 free(escaped);
470 return quoted;
471}

485static char *orm_build_insert_sql(const cwist_orm_t *orm, const char *table, const cJSON *data) {
486 if (!table || !data || !cJSON_IsObject(data)) return NULL;
487
488 int col_count = 0;
489 cJSON *item = NULL;
490 cJSON_ArrayForEach(item, (cJSON *)data) col_count++;
491 if (col_count == 0) return NULL;
492
493 char *columns = (char *)malloc(1);
494 char *values = (char *)malloc(1);
495 if (!columns || !values) {
496 free(columns);
497 free(values);
498 return NULL;
499 }
500 columns[0] = '\0';
501 values[0] = '\0';
502
503 char *qtable = cwist_orm_quote_identifier(orm, table);
504 if (!qtable) {
505 free(columns);
506 free(values);
507 return NULL;
508 }
509
510 int idx = 0;
511 cJSON_ArrayForEach(item, (cJSON *)data) {
512 char *literal = cwist_orm_json_to_sql_literal(orm, item);
513 char *qid = cwist_orm_quote_identifier(orm, item->string);
514 if (!literal || !qid) {
515 free(literal);
516 free(qid);
517 free(qtable);
518 free(columns);
519 free(values);
520 return NULL;
521 }
522
523 size_t need_col = strlen(columns) + strlen(qid) + 2;
524 char *new_col = (char *)realloc(columns, need_col);
525 if (!new_col) {
526 free(literal);
527 free(qid);
528 free(qtable);
529 free(columns);
530 free(values);
531 return NULL;
532 }
533 columns = new_col;
534
535 size_t need_val = strlen(values) + strlen(literal) + 2;
536 char *new_val = (char *)realloc(values, need_val);
537 if (!new_val) {
538 free(literal);
539 free(qid);
540 free(qtable);
541 free(columns);
542 free(values);
543 return NULL;
544 }
545 values = new_val;
546
547 if (idx > 0) {
548 strcat(columns, ",");
549 strcat(values, ",");
550 }
551 strcat(columns, qid);
552 strcat(values, literal);
553 free(literal);
554 free(qid);
555 idx++;
556 }
557
558 size_t sql_len = strlen(qtable) + strlen(columns) + strlen(values) + 32;
559 char *sql = (char *)malloc(sql_len);
560 if (!sql) {
561 free(qtable);
562 free(columns);
563 free(values);
564 return NULL;
565 }
566 snprintf(sql, sql_len, "INSERT INTO %s (%s) VALUES (%s)", qtable, columns, values);
567 free(qtable);
568 free(columns);
569 free(values);
570 return sql;
571}

818static char *orm_build_returning_sql(const cwist_orm_t *orm, const char *table, const cJSON *data,
819 const char *returning_col) {
820 char *base = orm_build_insert_sql(orm, table, data);
821 if (!base) return NULL;
822
823 char *qret = cwist_orm_quote_identifier(orm, returning_col);
824 if (!qret) {
825 free(base);
826 return NULL;
827 }
828
829 const char *fmt;
830 switch (orm->dialect) {
831 case CWIST_ORM_POSTGRES:
832 case CWIST_ORM_MARIADB: fmt = "%s RETURNING %s;"; break;
833 case CWIST_ORM_MYSQL: fmt = "%s; SELECT LAST_INSERT_ID() AS %s;"; break;
834 case CWIST_ORM_SQLITE:
835 default: fmt = "%s; SELECT last_insert_rowid() AS %s;"; break;
836 }
837 /* Size from the format itself: a fixed +16 truncated the SQLite and
838 * MySQL suffixes, so the INSERT ran but the SELECT never did. Raw
839 * malloc like every SQL buffer in this file (the caller frees it). */
840 int need = snprintf(NULL, 0, fmt, base, qret);
841 char *sql = need >= 0 ? (char *)malloc((size_t)need + 1) : NULL;
842 if (!sql) {
843 free(base);
844 free(qret);
845 return NULL;
846 }
847 snprintf(sql, (size_t)need + 1, fmt, base, qret);
848 free(base);
849 free(qret);
850 return sql;
851}

1233/* ------------------------------------------------------------------ */
1234/* Exact 64-bit integers (#354) */
1235/* ------------------------------------------------------------------ */
1236
1237/* The typed helpers above read values through cJSON numbers, which are
1238 * doubles. The worker sends canonical integers as raw JSON digits, so the
1239 * exact path takes them from the response text instead: cJSON validates the
1240 * response, then a small scanner finds the same cell in the raw text. */
1241
1242static const char *orm_json_ws(const char *p) {
1243 while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\r') p++;
1244 return p;
1245}
1246
1248static const char *orm_json_skip_string(const char *p) {
1249 for (p++; *p && *p != '"'; p++) {
1250 if (*p == '\\' && !*++p) return NULL;
1251 }
1252 return *p == '"' ? p + 1 : NULL;
1253}
1254
1256static const char *orm_json_skip_value(const char *p) {
1257 p = orm_json_ws(p);
1258 if (*p == '"') return orm_json_skip_string(p);
1259 if (*p == '{' || *p == '[') {
1260 int depth = 0;
1261 while (*p) {
1262 if (*p == '"') {
1263 p = orm_json_skip_string(p);
1264 if (!p) return NULL;
1265 continue;
1266 }
1267 if (*p == '{' || *p == '[') {
1268 depth++;
1269 } else if (*p == '}' || *p == ']') {
1270 if (--depth == 0) return p + 1;
1271 }
1272 p++;
1273 }
1274 return NULL;
1275 }
1276 const char *start = p;
1277 while (*p && *p != ',' && *p != '}' && *p != ']' && *p != ' ' && *p != '\t' && *p != '\n' &&
1278 *p != '\r')
1279 p++;
1280 return p > start ? p : NULL;
1281}
1282
1291static const char *orm_json_object_get(const char *p, const char *key) {
1292 size_t key_len = strlen(key);
1293 p = orm_json_ws(p);
1294 if (*p != '{') return NULL;
1295 p = orm_json_ws(p + 1);
1296 while (*p == '"') {
1297 const char *ks = p;
1298 const char *ke = orm_json_skip_string(p);
1299 if (!ke) return NULL;
1300 p = orm_json_ws(ke);
1301 if (*p != ':') return NULL;
1302 p = orm_json_ws(p + 1);
1303 size_t n = (size_t)(ke - ks - 2);
1304 if (n == key_len && !memchr(ks + 1, '\\', n) && memcmp(ks + 1, key, n) == 0) return p;
1305 p = orm_json_skip_value(p);
1306 if (!p) return NULL;
1307 p = orm_json_ws(p);
1308 if (*p != ',') return NULL;
1309 p = orm_json_ws(p + 1);
1310 }
1311 return NULL;
1312}

1320static bool orm_parse_exact_i64(const char *p, int64_t *out) {
1321 const char *s = p;
1322 const char *e;
1323 if (*p == '"') {
1324 e = orm_json_skip_string(p);
1325 if (!e) return false;
1326 s = p + 1;
1327 e--; /* closing quote */
1328 } else {
1329 e = orm_json_skip_value(p);
1330 if (!e) return false;
1331 }
1332 size_t n = (size_t)(e - s);
1333 size_t i = (n > 0 && s[0] == '-') ? 1 : 0;
1334 if (i >= n || n > 20) return false;
1335 if (s[i] == '0' && (n - i > 1 || i == 1)) return false;
1336 for (size_t k = i; k < n; k++) {
1337 if (s[k] < '0' || s[k] > '9') return false;
1338 }
1339 char buf[24];
1340 memcpy(buf, s, n);
1341 buf[n] = '\0';
1342 errno = 0;
1343 long long v = strtoll(buf, NULL, 10);
1344 if (errno == ERANGE) return false;
1345 *out = (int64_t)v;
1346 return true;
1347}

1429static int orm_b64_value(unsigned char c) {
1430 if (c >= 'A' && c <= 'Z') return c - 'A';
1431 if (c >= 'a' && c <= 'z') return c - 'a' + 26;
1432 if (c >= '0' && c <= '9') return c - '0' + 52;
1433 if (c == '+') return 62;
1434 if (c == '/') return 63;
1435 return -1;
1436}
1437
1443static unsigned char *orm_b64_decode(const char *in, size_t *out_len) {
1444 size_t n = strlen(in);
1445 if (n % 4 != 0) {
1446 *out_len = SIZE_MAX;
1447 return NULL;
1448 }
1449 size_t pad = 0;
1450 if (n > 0 && in[n - 1] == '=') pad++;
1451 if (n > 1 && in[n - 2] == '=') pad++;
1452 size_t cap = n / 4 * 3 - pad;
1453 unsigned char *out = (unsigned char *)malloc(cap ? cap : 1);
1454 if (!out) {
1455 *out_len = SIZE_MAX;
1456 return NULL;
1457 }
1458 size_t o = 0;
1459 for (size_t i = 0; i < n; i += 4) {
1460 int v[4];
1461 int chars = 3;
1462 for (int k = 0; k < 4; k++) {
1463 char c = in[i + k];
1464 if (c == '=') {
1465 if (i + 4 != n || k < 2 || (k == 2 && in[i + 3] != '=')) {
1466 free(out);
1467 *out_len = SIZE_MAX;
1468 return NULL;
1469 }
1470 v[k] = 0;
1471 chars--;
1472 } else {
1473 v[k] = orm_b64_value((unsigned char)c);
1474 if (v[k] < 0) {
1475 free(out);
1476 *out_len = SIZE_MAX;
1477 return NULL;
1478 }
1479 }
1480 }
1481 unsigned triple = ((unsigned)v[0] << 18) | ((unsigned)v[1] << 12) |
1482 ((unsigned)v[2] << 6) | (unsigned)v[3];
1483 if (o < cap) out[o++] = (unsigned char)(triple >> 16);
1484 if (o < cap) out[o++] = (unsigned char)(triple >> 8);
1485 if (o < cap) out[o++] = (unsigned char)triple;
1486 }
1487 *out_len = cap;
1488 return out;
1489}

1499static int orm_fill_typed_cell(cwist_orm_value_t *v, const cJSON *cell) {
1500 const cJSON *tag = cJSON_GetObjectItemCaseSensitive(cell, "t");
1501 if (!cJSON_IsString(tag) || !tag->valuestring) return CWIST_ERROR_PROTOCOL;
1502 const char *t = tag->valuestring;
1503 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(cell, "v");
1504
1505 if (strcmp(t, "null") == 0) {
1506 v->type = CWIST_ORM_TYPE_NULL;
1507 return CWIST_SUCCESS;
1508 }
1509 if (strcmp(t, "int") == 0) {
1510 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1511 const char *s = payload->valuestring;
1512 if (!*s) return CWIST_ERROR_PROTOCOL;
1513 errno = 0;
1514 char *endptr = NULL;
1515 long long val = strtoll(s, &endptr, 10);
1516 if (errno == ERANGE || !endptr || *endptr != '\0') return CWIST_ERROR_PROTOCOL;
1517 v->type = CWIST_ORM_TYPE_INTEGER;
1518 v->i64 = (int64_t)val;
1519 return CWIST_SUCCESS;
1520 }
1521 if (strcmp(t, "real") == 0) {
1522 if (!cJSON_IsNumber(payload)) return CWIST_ERROR_PROTOCOL;
1523 v->type = CWIST_ORM_TYPE_REAL;
1524 v->real = payload->valuedouble;
1525 return CWIST_SUCCESS;
1526 }
1527 if (strcmp(t, "text") == 0) {
1528 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1529 v->type = CWIST_ORM_TYPE_TEXT;
1530 v->text = strdup(payload->valuestring);
1531 return v->text ? CWIST_SUCCESS : CWIST_ERROR_NOMEM;
1532 }
1533 if (strcmp(t, "blob") == 0) {
1534 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1535 size_t len = 0;
1536 unsigned char *bytes = orm_b64_decode(payload->valuestring, &len);
1537 if (len == SIZE_MAX) return CWIST_ERROR_PROTOCOL;
1538 v->type = CWIST_ORM_TYPE_BLOB;
1539 v->blob = len ? bytes : NULL;
1540 v->blob_len = len;
1541 if (!len) free(bytes);
1542 return CWIST_SUCCESS;
1543 }
1544 return CWIST_ERROR_PROTOCOL;
1545}

1564cwist_error_t cwist_orm_query_typed(cwist_orm_t *orm, const char *sql,
1565 cwist_orm_result_t **result) {
1566 cwist_error_t err = make_error(CWIST_ERR_INT16);
1567 if (result) *result = NULL;
1568 if (!orm || !sql || !result) {
1569 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
1570 return err;
1571 }
1572
1573 char *payload = NULL;
1574 if (orm_exchange(orm, sql, &payload) != 0) {
1575 err.error.err_i16 = CWIST_ERROR_IO;
1576 return err;
1577 }
1578
1579 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
1580 cJSON *root = cJSON_Parse(payload);
1581 free(payload);
1582 if (!root) return err;
1583
1584 cwist_orm_result_t *res = NULL;
1585
1586 cJSON *status = cJSON_GetObjectItemCaseSensitive(root, "status");
1587 if (!cJSON_IsNumber(status) || (int)status->valuedouble != 0) goto done;
1588
1589 cJSON *typed = cJSON_GetObjectItemCaseSensitive(root, "typed");
1590 cJSON *columns = cJSON_GetObjectItemCaseSensitive(typed, "columns");
1591 cJSON *trows = cJSON_GetObjectItemCaseSensitive(typed, "rows");
1592 if (!cJSON_IsArray(columns) || !cJSON_IsArray(trows)) goto done;
1593
1594 res = (cwist_orm_result_t *)calloc(1, sizeof(*res));
1595 if (!res) {
1596 err.error.err_i16 = CWIST_ERROR_NOMEM;
1597 goto done;
1598 }
1599
1600 res->column_count = (size_t)cJSON_GetArraySize(columns);
1601 res->row_count = (size_t)cJSON_GetArraySize(trows);
1602
1603 res->columns = (char **)calloc(res->column_count ? res->column_count : 1, sizeof(char *));
1604 if (!res->columns) {
1605 err.error.err_i16 = CWIST_ERROR_NOMEM;
1606 goto fail;
1607 }
1608 for (size_t i = 0; i < res->column_count; i++) {
1609 cJSON *name = cJSON_GetArrayItem(columns, (int)i);
1610 if (!cJSON_IsString(name) || !name->valuestring) goto fail;
1611 res->columns[i] = strdup(name->valuestring);
1612 if (!res->columns[i]) {
1613 err.error.err_i16 = CWIST_ERROR_NOMEM;
1614 goto fail;
1615 }
1616 }
1617
1618 if (res->row_count > 0) {
1619 if (res->column_count == 0 ||
1620 res->row_count > SIZE_MAX / res->column_count / sizeof(cwist_orm_value_t))
1621 goto fail;
1622 res->cells = (cwist_orm_value_t *)calloc(
1623 res->row_count * res->column_count, sizeof(cwist_orm_value_t));
1624 if (!res->cells) {
1625 err.error.err_i16 = CWIST_ERROR_NOMEM;
1626 goto fail;
1627 }
1628 }
1629
1630 size_t idx = 0;
1631 cJSON *trow = NULL;
1632 cJSON_ArrayForEach(trow, trows) {
1633 if (!cJSON_IsArray(trow) || (size_t)cJSON_GetArraySize(trow) != res->column_count)
1634 goto fail;
1635 for (size_t c = 0; c < res->column_count; c++, idx++) {
1636 int rc = orm_fill_typed_cell(&res->cells[idx], cJSON_GetArrayItem(trow, (int)c));
1637 if (rc != CWIST_SUCCESS) {
1638 err.error.err_i16 = rc;
1639 goto fail;
1640 }
1641 }
1642 }
1643
1644 *result = res;
1645 err.error.err_i16 = CWIST_SUCCESS;
1646 goto done;
1647
1648fail:
1649 cwist_orm_result_free(res);
1650done:
1651 cJSON_Delete(root);
1652 return err;
1653}

1663void cwist_orm_result_free(cwist_orm_result_t *result) {
1664 if (!result) return;
1665 if (result->cells) {
1666 size_t total = result->row_count * result->column_count;
1667 for (size_t i = 0; i < total; i++) {
1668 free(result->cells[i].text);
1669 free(result->cells[i].blob);

1672 if (result->columns) {
1673 for (size_t i = 0; i < result->column_count; i++) free(result->columns[i]);
1674 }
1675 free(result->columns);

1680size_t cwist_orm_result_column_index(const cwist_orm_result_t *result, const char *column) {
