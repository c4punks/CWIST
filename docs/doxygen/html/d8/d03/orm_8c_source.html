56static int orm_recv_all(int fd, void *buf, size_t n) {
57 size_t total = 0;
58 char *p = (char *)buf;
59 while (total < n) {
60 ssize_t r = recv(fd, p + total, n - total, MSG_WAITALL);
61 if (r <= 0) return -1;
62 total += (size_t)r;
63 }
64 return 0;
65}
66
71static int orm_send_all(int fd, const void *buf, size_t n) {
72 size_t total = 0;
73 const char *p = (const char *)buf;
74 while (total < n) {
75 ssize_t w = send(fd, p + total, n - total, MSG_NOSIGNAL);
76 if (w < 0 && errno == EINTR) continue;
77 if (w <= 0) return -1;
78 total += (size_t)w;
79 }
80 return 0;
81}

95static int orm_exchange(cwist_orm_t *orm, const char *sql, char **out_payload) {
96 *out_payload = NULL;
97 if (!orm || orm->sock_fd < 0 || !sql) return -1;
98
99 size_t sql_len = strlen(sql);
100 if (sql_len > UINT32_MAX - 1) return -1;
101
102 uint32_t net_len = htonl((uint32_t)sql_len);
103 if (orm_send_all(orm->sock_fd, &net_len, sizeof(net_len)) != 0) return -1;
104 if (orm_send_all(orm->sock_fd, sql, sql_len) != 0) return -1;
105
106 uint32_t net_payload = 0;
107 if (orm_recv_all(orm->sock_fd, &net_payload, sizeof(net_payload)) != 0) return -1;
108 uint32_t payload_len = ntohl(net_payload);
109 if (payload_len > 64 * 1024 * 1024) return -1; /* sanity ceiling */
110
111 char *payload = (char *)malloc(payload_len + 1);
112 if (!payload) return -1;
113 if (payload_len > 0 && orm_recv_all(orm->sock_fd, payload, payload_len) != 0) {
114 free(payload);
115 return -1;
116 }
117 payload[payload_len] = '\0';
118 *out_payload = payload;
119 return 0;
120}

152cwist_orm_t *cwist_orm_open_socket(int socket_fd) {
153 if (socket_fd < 0) return NULL;
154#ifdef SO_NOSIGPIPE
155 int one = 1;
156 (void)setsockopt(socket_fd, SOL_SOCKET, SO_NOSIGPIPE, &one, sizeof(one));
157#endif
158 cwist_orm_t *orm = (cwist_orm_t *)calloc(1, sizeof(*orm));
159 if (!orm) return NULL;
160 orm->sock_fd = socket_fd;
161 orm->auto_commit = g_immediate_commit;
162 orm->dialect = CWIST_ORM_SQLITE;
163 return orm;
164}

174void cwist_orm_close_socket(cwist_orm_t *orm) {
175 if (!orm) return;
176 if (orm->sock_fd >= 0) {
177 /* Best-effort rollback of any dangling transaction. */
178 char *dummy = NULL;
179 (void)orm_exchange(orm, "ROLLBACK;", &dummy);
180 free(dummy);
181 close(orm->sock_fd);
182 }
183 free(orm);
184}

252cwist_error_t cwist_orm_exec(cwist_orm_t *orm, const char *sql) {
253 cwist_error_t err = make_error(CWIST_ERR_INT16);
254 if (!orm || !sql) {
255 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
256 return err;
257 }
258
259 char *payload = NULL;
260 if (orm_exchange(orm, sql, &payload) != 0) {
261 err.error.err_i16 = CWIST_ERROR_IO;
262 return err;
263 }
264
265 int status = orm_parse_status(payload);
266 free(payload);
267
268 if (status != 0) {
269 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
270 return err;
271 }
272
273 /* Auto-commit write helpers if requested. */
274 if (g_immediate_commit) {
275 const char *upper = sql;
276 while (*upper && *upper <= ' ') upper++;
277 if ((strncasecmp(upper, "INSERT", 6) == 0) || (strncasecmp(upper, "UPDATE", 6) == 0) ||
278 (strncasecmp(upper, "DELETE", 6) == 0)) {
279 char *commit_payload = NULL;
280 (void)orm_exchange(orm, "COMMIT;", &commit_payload);
281 free(commit_payload);
282 }
283 }
284
285 err.error.err_i16 = CWIST_SUCCESS;
286 return err;
287}

306 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
307 return err;

312 err.error.err_i16 = CWIST_ERROR_IO;

324 if (!cJSON_IsNumber(status) || (int)status->valuedouble != 0) {
325 cJSON_Delete(root);
326 err.error.err_i16 = CWIST_ERROR_PROTOCOL;

329

334 rows = cJSON_CreateArray();

336

354static char *cwist_orm_escape_sqlite(const char *src) {
355 if (!src) return NULL;
356 size_t len = strlen(src);
357 size_t extra = 0;
358 for (size_t i = 0; i < len; i++) {
359 if (src[i] == '\'') extra++;
360 }
361 char *dst = (char *)malloc(len + extra + 1);
362 if (!dst) return NULL;
363 size_t j = 0;
364 for (size_t i = 0; i < len; i++) {
365 if (src[i] == '\'') dst[j++] = '\'';
366 dst[j++] = src[i];
367 }
368 dst[j] = '\0';
369 return dst;
370}

372static char *cwist_orm_literal(const cwist_orm_t *orm, const char *src) {
373 if (!src) return NULL;
374 bool mysql = orm && (orm->dialect == CWIST_ORM_MYSQL || orm->dialect == CWIST_ORM_MARIADB);
375 if (mysql) {
376 static const char hex[] = "0123456789ABCDEF";
377 size_t len = strlen(src);
378 if (len > (SIZE_MAX - 4) / 2) return NULL;
379 char *out = (char *)malloc(len * 2 + 4);
380 if (!out) return NULL;
381 out[0] = 'X';
382 out[1] = '\'';
383 for (size_t i = 0; i < len; i++) {
384 unsigned char c = (unsigned char)src[i];
385 out[2 + i * 2] = hex[c >> 4];
386 out[3 + i * 2] = hex[c & 0x0f];
387 }
388 out[2 + len * 2] = '\'';
389 out[3 + len * 2] = '\0';
390 return out;
391 }
392
393 char *escaped = cwist_orm_escape_sqlite(src);
394 if (!escaped) return NULL;
395 size_t len = strlen(escaped);
396 char *out = (char *)malloc(len + 3);
397 if (!out) {
398 free(escaped);
399 return NULL;
400 }
401 out[0] = '\'';
402 memcpy(out + 1, escaped, len);
403 out[len + 1] = '\'';
404 out[len + 2] = '\0';
405 free(escaped);
406 return out;
407}

418static char *cwist_orm_quote_identifier(const cwist_orm_t *orm, const char *id) {
419 if (!orm || !id) return NULL;
420 char open_ch = '\0';
421 char close_ch = '\0';
422 char escape_ch = '\0';
423 switch (orm->dialect) {
424 case CWIST_ORM_POSTGRES:
425 case CWIST_ORM_ORACLE:
426 open_ch = '"';
427 close_ch = '"';
428 escape_ch = '"';
429 break;
430 case CWIST_ORM_MYSQL:
431 case CWIST_ORM_MARIADB:
432 case CWIST_ORM_SQLITE:
433 open_ch = '`';
434 close_ch = '`';
435 escape_ch = '`';
436 break;
437 case CWIST_ORM_MSSQL:
438 open_ch = '[';
439 close_ch = ']';
440 escape_ch = ']';
441 break;
442 default: return strdup(id);
443 }
444
445 size_t len = strlen(id);
446 size_t extra = 0;
447 for (size_t i = 0; i < len; i++) {
448 if (id[i] == escape_ch) extra++;
449 }
450 char *out = (char *)malloc(len + extra + 3);
451 if (!out) return NULL;
452 size_t j = 0;
453 out[j++] = open_ch;
454 for (size_t i = 0; i < len; i++) {
455 if (id[i] == escape_ch) out[j++] = escape_ch;
456 out[j++] = id[i];
457 }
458 out[j++] = close_ch;
459 out[j] = '\0';
460 return out;
461}

475static char *cwist_orm_json_to_sql_literal(const cwist_orm_t *orm, const cJSON *node) {
476 if (!node) return strdup("NULL");
477
478 if (cJSON_IsNull(node)) {
479 return strdup("NULL");
480 }
481 if (cJSON_IsBool(node)) {
482 if (orm && orm->dialect == CWIST_ORM_POSTGRES) {
483 return strdup(cJSON_IsTrue(node) ? "TRUE" : "FALSE");
484 }
485 return strdup(cJSON_IsTrue(node) ? "1" : "0");
486 }
487 if (cJSON_IsNumber(node)) {
488 double value = node->valuedouble;
489 if (!isfinite(value)) return NULL;
490 if (trunc(value) == value && (value > 9007199254740991.0 || value < -9007199254740991.0))
491 return NULL;
492 char buf[64];
493 if (value >= -9007199254740991.0 && value <= 9007199254740991.0 && trunc(value) == value) {
494 snprintf(buf, sizeof(buf), "%.0f", value);
495 } else {
496 snprintf(buf, sizeof(buf), "%.17g", value);
497 }
498 return strdup(buf);
499 }
500 if (cJSON_IsString(node)) {
501 if (!node->valuestring) return strdup("''");
502 return cwist_orm_literal(orm, node->valuestring);
503 }
504
505 /* Fallback: stringify arrays/objects as JSON text. */
506 char *raw = cJSON_PrintUnformatted((cJSON *)node);
507 if (!raw) return strdup("NULL");
508 char *literal = cwist_orm_literal(orm, raw);
509 cJSON_free(raw);
510 return literal;
511}

525static char *orm_build_insert_sql(const cwist_orm_t *orm, const char *table, const cJSON *data) {
526 if (!table || !data || !cJSON_IsObject(data)) return NULL;
527
528 int col_count = 0;
529 cJSON *item = NULL;
530 cJSON_ArrayForEach(item, (cJSON *)data) col_count++;
531 if (col_count == 0) return NULL;
532
533 char *columns = (char *)malloc(1);
534 char *values = (char *)malloc(1);
535 if (!columns || !values) {
536 free(columns);
537 free(values);
538 return NULL;
539 }
540 columns[0] = '\0';
541 values[0] = '\0';
542
543 char *qtable = cwist_orm_quote_identifier(orm, table);
544 if (!qtable) {
545 free(columns);
546 free(values);
547 return NULL;
548 }
549
550 int idx = 0;
551 cJSON_ArrayForEach(item, (cJSON *)data) {
552 char *literal = cwist_orm_json_to_sql_literal(orm, item);
553 char *qid = cwist_orm_quote_identifier(orm, item->string);
554 if (!literal || !qid) {
555 free(literal);
556 free(qid);
557 free(qtable);
558 free(columns);
559 free(values);
560 return NULL;
561 }
562
563 size_t need_col = strlen(columns) + strlen(qid) + 2;
564 char *new_col = (char *)realloc(columns, need_col);
565 if (!new_col) {
566 free(literal);
567 free(qid);
568 free(qtable);
569 free(columns);
570 free(values);
571 return NULL;
572 }
573 columns = new_col;
574
575 size_t need_val = strlen(values) + strlen(literal) + 2;
576 char *new_val = (char *)realloc(values, need_val);
577 if (!new_val) {
578 free(literal);
579 free(qid);
580 free(qtable);
581 free(columns);
582 free(values);
583 return NULL;
584 }
585 values = new_val;
586
587 if (idx > 0) {
588 strcat(columns, ",");
589 strcat(values, ",");
590 }
591 strcat(columns, qid);
592 strcat(values, literal);
593 free(literal);
594 free(qid);
595 idx++;
596 }
597
598 size_t sql_len = strlen(qtable) + strlen(columns) + strlen(values) + 32;
599 char *sql = (char *)malloc(sql_len);
600 if (!sql) {
601 free(qtable);
602 free(columns);
603 free(values);
604 return NULL;
605 }
606 snprintf(sql, sql_len, "INSERT INTO %s (%s) VALUES (%s)", qtable, columns, values);
607 free(qtable);
608 free(columns);
609 free(values);
610 return sql;
611}

858static char *orm_build_returning_sql(const cwist_orm_t *orm, const char *table, const cJSON *data,
859 const char *returning_col) {
860 char *base = orm_build_insert_sql(orm, table, data);
861 if (!base) return NULL;
862
863 char *qret = cwist_orm_quote_identifier(orm, returning_col);
864 if (!qret) {
865 free(base);
866 return NULL;
867 }
868
869 const char *fmt;
870 switch (orm->dialect) {
871 case CWIST_ORM_POSTGRES:
872 case CWIST_ORM_MARIADB:
873 case CWIST_ORM_ORACLE: fmt = "%s RETURNING %s;"; break;
874 case CWIST_ORM_MYSQL: fmt = "%s; SELECT LAST_INSERT_ID() AS %s;"; break;
875 case CWIST_ORM_MSSQL: fmt = "%s; SELECT SCOPE_IDENTITY() AS %s;"; break;
876 case CWIST_ORM_SQLITE:
877 default: fmt = "%s; SELECT last_insert_rowid() AS %s;"; break;
878 }
879 /* Size from the format itself: a fixed +16 truncated the SQLite and
880 * MySQL suffixes, so the INSERT ran but the SELECT never did. Raw
881 * malloc like every SQL buffer in this file (the caller frees it). */
882 int need = snprintf(NULL, 0, fmt, base, qret);
883 char *sql = need >= 0 ? (char *)malloc((size_t)need + 1) : NULL;
884 if (!sql) {
885 free(base);
886 free(qret);
887 return NULL;
888 }
889 snprintf(sql, (size_t)need + 1, fmt, base, qret);
890 free(base);
891 free(qret);
892 return sql;
893}

1296/* ------------------------------------------------------------------ */
1297
1298/* The typed helpers above read values through cJSON numbers, which are
1299 * doubles. The worker sends canonical integers as raw JSON digits, so the
1300 * exact path takes them from the response text instead: cJSON validates the
1301 * response, then a small scanner finds the same cell in the raw text. */
1302
1303static const char *orm_json_ws(const char *p) {
1304 while (*p == ' ' || *p == '\t' || *p == '\n' || *p == '\r') p++;
1305 return p;
1306}
1307
1309static const char *orm_json_skip_string(const char *p) {
1310 for (p++; *p && *p != '"'; p++) {
1311 if (*p == '\\' && !*++p) return NULL;
1312 }
1313 return *p == '"' ? p + 1 : NULL;
1314}
1315
1317static const char *orm_json_skip_value(const char *p) {
1318 p = orm_json_ws(p);
1319 if (*p == '"') return orm_json_skip_string(p);
1320 if (*p == '{' || *p == '[') {
1321 int depth = 0;
1322 while (*p) {
1323 if (*p == '"') {
1324 p = orm_json_skip_string(p);
1325 if (!p) return NULL;
1326 continue;
1327 }
1328 if (*p == '{' || *p == '[') {
1329 depth++;
1330 } else if (*p == '}' || *p == ']') {
1331 if (--depth == 0) return p + 1;
1332 }
1333 p++;
1334 }
1335 return NULL;
1336 }
1337 const char *start = p;
1338 while (*p && *p != ',' && *p != '}' && *p != ']' && *p != ' ' && *p != '\t' && *p != '\n' &&
1339 *p != '\r')
1340 p++;
1341 return p > start ? p : NULL;
1342}
1343
1352static const char *orm_json_object_get(const char *p, const char *key) {
1353 size_t key_len = strlen(key);
1354 p = orm_json_ws(p);
1355 if (*p != '{') return NULL;
1356 p = orm_json_ws(p + 1);
1357 while (*p == '"') {
1358 const char *ks = p;
1359 const char *ke = orm_json_skip_string(p);
1360 if (!ke) return NULL;
1361 p = orm_json_ws(ke);
1362 if (*p != ':') return NULL;
1363 p = orm_json_ws(p + 1);
1364 size_t n = (size_t)(ke - ks - 2);
1365 if (n == key_len && !memchr(ks + 1, '\\', n) && memcmp(ks + 1, key, n) == 0) return p;
1366 p = orm_json_skip_value(p);
1367 if (!p) return NULL;
1368 p = orm_json_ws(p);
1369 if (*p != ',') return NULL;
1370 p = orm_json_ws(p + 1);
1371 }
1372 return NULL;
1373}

1381static bool orm_parse_exact_i64(const char *p, int64_t *out) {
1382 const char *s = p;
1383 const char *e;
1384 if (*p == '"') {
1385 e = orm_json_skip_string(p);
1386 if (!e) return false;
1387 s = p + 1;
1388 e--; /* closing quote */
1389 } else {
1390 e = orm_json_skip_value(p);
1391 if (!e) return false;
1392 }
1393 size_t n = (size_t)(e - s);
1394 size_t i = (n > 0 && s[0] == '-') ? 1 : 0;
1395 if (i >= n || n > 20) return false;
1396 if (s[i] == '0' && (n - i > 1 || i == 1)) return false;
1397 for (size_t k = i; k < n; k++) {
1398 if (s[k] < '0' || s[k] > '9') return false;
1399 }
1400 char buf[24];
1401 memcpy(buf, s, n);
1402 buf[n] = '\0';
1403 errno = 0;
1404 long long v = strtoll(buf, NULL, 10);
1405 if (errno == ERANGE) return false;
1406 *out = (int64_t)v;
1407 return true;
1408}

1489static int orm_b64_value(unsigned char c) {
1490 if (c >= 'A' && c <= 'Z') return c - 'A';
1491 if (c >= 'a' && c <= 'z') return c - 'a' + 26;
1492 if (c >= '0' && c <= '9') return c - '0' + 52;
1493 if (c == '+') return 62;
1494 if (c == '/') return 63;
1495 return -1;
1496}

1503static unsigned char *orm_b64_decode(const char *in, size_t *out_len) {
1504 size_t n = strlen(in);
1505 if (n % 4 != 0) {
1506 *out_len = SIZE_MAX;
1507 return NULL;
1508 }
1509 size_t pad = 0;
1510 if (n > 0 && in[n - 1] == '=') pad++;
1511 if (n > 1 && in[n - 2] == '=') pad++;
1512 size_t cap = n / 4 * 3 - pad;
1513 unsigned char *out = (unsigned char *)malloc(cap ? cap : 1);
1514 if (!out) {
1515 *out_len = SIZE_MAX;
1516 return NULL;
1517 }
1518 size_t o = 0;
1519 for (size_t i = 0; i < n; i += 4) {
1520 int v[4];
1521 int chars = 3;
1522 for (int k = 0; k < 4; k++) {
1523 char c = in[i + k];
1524 if (c == '=') {
1525 if (i + 4 != n || k < 2 || (k == 2 && in[i + 3] != '=')) {
1526 free(out);
1527 *out_len = SIZE_MAX;
1528 return NULL;
1529 }
1530 v[k] = 0;
1531 chars--;
1532 } else {
1533 v[k] = orm_b64_value((unsigned char)c);
1534 if (v[k] < 0) {
1535 free(out);
1536 *out_len = SIZE_MAX;
1537 return NULL;
1538 }
1539 }
1540 }
1541 unsigned triple = ((unsigned)v[0] << 18) | ((unsigned)v[1] << 12) | ((unsigned)v[2] << 6) |
1542 (unsigned)v[3];
1543 if (o < cap) out[o++] = (unsigned char)(triple >> 16);
1544 if (o < cap) out[o++] = (unsigned char)(triple >> 8);
1545 if (o < cap) out[o++] = (unsigned char)triple;
1546 }
1547 *out_len = cap;
1548 return out;
1549}

1559static int orm_fill_typed_cell(cwist_orm_value_t *v, const cJSON *cell) {
1560 const cJSON *tag = cJSON_GetObjectItemCaseSensitive(cell, "t");
1561 if (!cJSON_IsString(tag) || !tag->valuestring) return CWIST_ERROR_PROTOCOL;
1562 const char *t = tag->valuestring;
1563 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(cell, "v");
1564
1565 if (strcmp(t, "null") == 0) {
1566 v->type = CWIST_ORM_TYPE_NULL;
1567 return CWIST_SUCCESS;
1568 }
1569 if (strcmp(t, "int") == 0) {
1570 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1571 const char *s = payload->valuestring;
1572 if (!*s) return CWIST_ERROR_PROTOCOL;
1573 const char *digits = s[0] == '-' ? s + 1 : s;
1574 if (!*digits || (digits[0] == '0' && digits[1] != '\0')) return CWIST_ERROR_PROTOCOL;
1575 for (const char *p = digits; *p; p++)
1576 if (*p < '0' || *p > '9') return CWIST_ERROR_PROTOCOL;
1577 errno = 0;
1578 char *endptr = NULL;
1579 long long val = strtoll(s, &endptr, 10);
1580 if (errno == ERANGE || !endptr || *endptr != '\0') return CWIST_ERROR_PROTOCOL;
1581 v->type = CWIST_ORM_TYPE_INTEGER;
1582 v->i64 = (int64_t)val;
1583 return CWIST_SUCCESS;
1584 }
1585 if (strcmp(t, "real") == 0) {
1586 if (!cJSON_IsNumber(payload)) return CWIST_ERROR_PROTOCOL;
1587 v->type = CWIST_ORM_TYPE_REAL;
1588 v->real = payload->valuedouble;
1589 return CWIST_SUCCESS;
1590 }
1591 if (strcmp(t, "text") == 0) {
1592 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1593 v->type = CWIST_ORM_TYPE_TEXT;
1594 v->text = strdup(payload->valuestring);
1595 return v->text ? CWIST_SUCCESS : CWIST_ERROR_NOMEM;
1596 }
1597 if (strcmp(t, "blob") == 0) {
1598 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1599 size_t len = 0;
1600 unsigned char *bytes = orm_b64_decode(payload->valuestring, &len);
1601 if (len == SIZE_MAX) return CWIST_ERROR_PROTOCOL;
1602 v->type = CWIST_ORM_TYPE_BLOB;
1603 v->blob = len ? bytes : NULL;
1604 v->blob_len = len;
1605 if (!len) free(bytes);
1606 return CWIST_SUCCESS;
1607 }
1608 return CWIST_ERROR_PROTOCOL;
1609}

1628cwist_error_t cwist_orm_query_typed(cwist_orm_t *orm, const char *sql,
1629 cwist_orm_result_t **result) {
1630 cwist_error_t err = make_error(CWIST_ERR_INT16);
1631 if (result) *result = NULL;
1632 if (!orm || !sql || !result) {
1633 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
1634 return err;
1635 }
1636
1637 char *payload = NULL;
1638 if (orm_exchange(orm, sql, &payload) != 0) {
1639 err.error.err_i16 = CWIST_ERROR_IO;
1640 return err;
1641 }
1642
1643 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
1644 cJSON *root = cJSON_Parse(payload);
1645 free(payload);
1646 if (!root) return err;
1647
1648 cwist_orm_result_t *res = NULL;
1649
1650 cJSON *status = cJSON_GetObjectItemCaseSensitive(root, "status");
1651 if (!cJSON_IsNumber(status) || (int)status->valuedouble != 0) goto done;
1652
1653 cJSON *typed = cJSON_GetObjectItemCaseSensitive(root, "typed");
1654 cJSON *columns = cJSON_GetObjectItemCaseSensitive(typed, "columns");
1655 cJSON *trows = cJSON_GetObjectItemCaseSensitive(typed, "rows");
1656 if (!cJSON_IsArray(columns) || !cJSON_IsArray(trows)) goto done;
1657
1658 res = (cwist_orm_result_t *)calloc(1, sizeof(*res));
1659 if (!res) {
1660 err.error.err_i16 = CWIST_ERROR_NOMEM;
1661 goto done;
1662 }
