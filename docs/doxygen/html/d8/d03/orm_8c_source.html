58static int orm_recv_all(int fd, void *buf, size_t n) {
59 size_t total = 0;
60 char *p = (char *)buf;
61 while (total < n) {
62 ssize_t r = recv(fd, p + total, n - total, MSG_WAITALL);
63 if (r < 0 && errno == EINTR) continue;
64 if (r <= 0) return -1;
65 total += (size_t)r;
66 }
67 return 0;
68}

74static int orm_send_all(int fd, const void *buf, size_t n) {
75 size_t total = 0;
76 const char *p = (const char *)buf;
77 while (total < n) {
78 ssize_t w = send(fd, p + total, n - total, MSG_NOSIGNAL);
79 if (w < 0 && errno == EINTR) continue;
80 if (w <= 0) return -1;
81 total += (size_t)w;
82 }
83 return 0;

98static int orm_exchange(cwist_orm_t *orm, const char *sql, char **out_payload) {
99 *out_payload = NULL;
100 if (!orm || orm->sock_fd < 0 || !sql) return -1;
101
102 size_t sql_len = strlen(sql);
103 if (sql_len > UINT32_MAX - 1) return -1;
104
105 uint32_t net_len = htonl((uint32_t)sql_len);
106 if (orm_send_all(orm->sock_fd, &net_len, sizeof(net_len)) != 0) return -1;
107 if (orm_send_all(orm->sock_fd, sql, sql_len) != 0) return -1;
108
109 uint32_t net_payload = 0;
110 if (orm_recv_all(orm->sock_fd, &net_payload, sizeof(net_payload)) != 0) return -1;
111 uint32_t payload_len = ntohl(net_payload);
112 if (payload_len > 64 * 1024 * 1024) return -1; /* sanity ceiling */
113
114 char *payload = (char *)malloc(payload_len + 1);
115 if (!payload) return -1;
116 if (payload_len > 0 && orm_recv_all(orm->sock_fd, payload, payload_len) != 0) {
117 free(payload);
118 return -1;
119 }
120 payload[payload_len] = '\0';
121 *out_payload = payload;
122 return 0;
123}

264cwist_error_t cwist_orm_exec(cwist_orm_t *orm, const char *sql) {
265 cwist_error_t err = make_error(CWIST_ERR_INT16);
266 if (!orm || !sql) {
267 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
268 return err;
269 }
270
271 char *payload = NULL;
272 if (orm_exchange(orm, sql, &payload) != 0) {
273 err.error.err_i16 = CWIST_ERROR_IO;
274 return err;
275 }
276
277 int status = orm_parse_status(payload);
278 free(payload);
279
280 if (status != 0) {
281 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
282 return err;
283 }
284
285 const char *upper = sql;
286 while (*upper && *upper <= ' ') upper++;
287 if (strncasecmp(upper, "BEGIN", 5) == 0) {
288 orm->in_transaction = true;
289 } else if (strncasecmp(upper, "COMMIT", 6) == 0 || strncasecmp(upper, "ROLLBACK", 8) == 0) {
290 orm->in_transaction = false;
291 }
292
293 /* An explicit transaction needs a final COMMIT; standalone writes are
294 * already committed by SQLite and must not receive a spurious one. */
295 if (orm->auto_commit && orm->in_transaction &&
296 (strncasecmp(upper, "INSERT", 6) == 0 || strncasecmp(upper, "UPDATE", 6) == 0 ||
297 strncasecmp(upper, "DELETE", 6) == 0)) {
298 char *commit_payload = NULL;
299 if (orm_exchange(orm, "COMMIT;", &commit_payload) != 0) {
300 err.error.err_i16 = CWIST_ERROR_IO;
301 return err;
302 }
303 int commit_status = orm_parse_status(commit_payload);
304 free(commit_payload);
305 if (commit_status != 0) {
306 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
307 return err;
308 }
309 orm->in_transaction = false;
310 }
311
312 err.error.err_i16 = CWIST_SUCCESS;
313 return err;
314}

403static char *cwist_orm_literal(const cwist_orm_t *orm, const char *src) {
404 if (!src) return NULL;
405 bool mysql = orm && (orm->dialect == CWIST_ORM_MYSQL || orm->dialect == CWIST_ORM_MARIADB);
406 if (mysql) {
407 static const char hex[] = "0123456789ABCDEF";
408 size_t len = strlen(src);
409 if (len > (SIZE_MAX - 4) / 2) return NULL;
410 char *out = (char *)malloc(len * 2 + 4);
411 if (!out) return NULL;
412 out[0] = 'X';
413 out[1] = '\'';
414 for (size_t i = 0; i < len; i++) {
415 unsigned char c = (unsigned char)src[i];
416 out[2 + i * 2] = hex[c >> 4];
417 out[3 + i * 2] = hex[c & 0x0f];
418 }
419 out[2 + len * 2] = '\'';
420 out[3 + len * 2] = '\0';
421 return out;
422 }
423
424 char *escaped = cwist_orm_escape_sqlite(src);
425 if (!escaped) return NULL;
426 size_t len = strlen(escaped);
427 char *out = (char *)malloc(len + 3);
428 if (!out) {
429 free(escaped);
430 return NULL;
431 }
432 out[0] = '\'';
433 memcpy(out + 1, escaped, len);
434 out[len + 1] = '\'';
435 out[len + 2] = '\0';
436 free(escaped);
437 return out;
438}

449static char *cwist_orm_quote_identifier(const cwist_orm_t *orm, const char *id) {
450 if (!orm || !id) return NULL;
451 char open_ch = '\0';
452 char close_ch = '\0';
453 char escape_ch = '\0';
454 switch (orm->dialect) {
455 case CWIST_ORM_POSTGRES:
456 case CWIST_ORM_ORACLE:
457 open_ch = '"';
458 close_ch = '"';
459 escape_ch = '"';
460 break;
461 case CWIST_ORM_MYSQL:
462 case CWIST_ORM_MARIADB:
463 case CWIST_ORM_SQLITE:
464 open_ch = '`';
465 close_ch = '`';
466 escape_ch = '`';
467 break;
468 case CWIST_ORM_MSSQL:
469 open_ch = '[';
470 close_ch = ']';
471 escape_ch = ']';
472 break;
473 default: return strdup(id);
474 }
475
476 size_t len = strlen(id);
477 size_t extra = 0;
478 for (size_t i = 0; i < len; i++) {
479 if (id[i] == escape_ch) extra++;
480 }
481 char *out = (char *)malloc(len + extra + 3);
482 if (!out) return NULL;
483 size_t j = 0;
484 out[j++] = open_ch;
485 for (size_t i = 0; i < len; i++) {
486 if (id[i] == escape_ch) out[j++] = escape_ch;
487 out[j++] = id[i];
488 }
489 out[j++] = close_ch;
490 out[j] = '\0';
491 return out;

506static char *cwist_orm_json_to_sql_literal(const cwist_orm_t *orm, const cJSON *node) {
507 if (!node) return strdup("NULL");
508
509 if (cJSON_IsNull(node)) {
510 return strdup("NULL");
511 }
512 if (cJSON_IsBool(node)) {
513 if (orm && orm->dialect == CWIST_ORM_POSTGRES) {
514 return strdup(cJSON_IsTrue(node) ? "TRUE" : "FALSE");
515 }
516 return strdup(cJSON_IsTrue(node) ? "1" : "0");
517 }
518 if (cJSON_IsNumber(node)) {
519 double value = node->valuedouble;
520 if (!isfinite(value)) return NULL;
521 if (trunc(value) == value && (value > 9007199254740991.0 || value < -9007199254740991.0))
522 return NULL;
523 char buf[64];
524 if (value >= -9007199254740991.0 && value <= 9007199254740991.0 && trunc(value) == value) {
525 snprintf(buf, sizeof(buf), "%.0f", value);
526 } else {
527 snprintf(buf, sizeof(buf), "%.17g", value);
528 }
529 return strdup(buf);
530 }
531 if (cJSON_IsString(node)) {
532 if (!node->valuestring) return strdup("''");
533 return cwist_orm_literal(orm, node->valuestring);
534 }
535
536 /* Fallback: stringify arrays/objects as JSON text. */
537 char *raw = cJSON_PrintUnformatted((cJSON *)node);
538 if (!raw) return strdup("NULL");
539 char *literal = cwist_orm_literal(orm, raw);
540 cJSON_free(raw);
541 return literal;
542}

639 free(columns);

1355static const char *orm_json_ws(const char *p) {

1357 return p;

1361static const char *orm_json_skip_string(const char *p) {
1362 for (p++; *p && *p != '"'; p++) {
1363 if (*p == '\\' && !*++p) return NULL;
1364 }
1365 return *p == '"' ? p + 1 : NULL;
1366}

1369static const char *orm_json_skip_value(const char *p) {
1370 p = orm_json_ws(p);
1371 if (*p == '"') return orm_json_skip_string(p);
1372 if (*p == '{' || *p == '[') {
1373 int depth = 0;
1374 while (*p) {
1375 if (*p == '"') {
1376 p = orm_json_skip_string(p);
1377 if (!p) return NULL;
1378 continue;
1379 }
1380 if (*p == '{' || *p == '[') {
1381 depth++;
1382 } else if (*p == '}' || *p == ']') {
1383 if (--depth == 0) return p + 1;
1384 }
1385 p++;
1386 }
1387 return NULL;
1388 }
1389 const char *start = p;
1390 while (*p && *p != ',' && *p != '}' && *p != ']' && *p != ' ' && *p != '\t' && *p != '\n' &&
1391 *p != '\r')
1392 p++;
1393 return p > start ? p : NULL;
1394}

1404static const char *orm_json_object_get(const char *p, const char *key) {
1405 size_t key_len = strlen(key);
1406 p = orm_json_ws(p);
1407 if (*p != '{') return NULL;
1408 p = orm_json_ws(p + 1);
1409 while (*p == '"') {
1410 const char *ks = p;
1411 const char *ke = orm_json_skip_string(p);
1412 if (!ke) return NULL;
1413 p = orm_json_ws(ke);
1414 if (*p != ':') return NULL;
1415 p = orm_json_ws(p + 1);
1416 size_t n = (size_t)(ke - ks - 2);
1417 if (n == key_len && !memchr(ks + 1, '\\', n) && memcmp(ks + 1, key, n) == 0) return p;
1418 p = orm_json_skip_value(p);
1419 if (!p) return NULL;
1420 p = orm_json_ws(p);
1421 if (*p != ',') return NULL;
1422 p = orm_json_ws(p + 1);
1423 }
1424 return NULL;
1425}

1433static bool orm_parse_exact_i64(const char *p, int64_t *out) {
1434 const char *s = p;
1435 const char *e;
1436 if (*p == '"') {
1437 e = orm_json_skip_string(p);
1438 if (!e) return false;
1439 s = p + 1;
1440 e--; /* closing quote */
1441 } else {
1442 e = orm_json_skip_value(p);
1443 if (!e) return false;
1444 }
1445 size_t n = (size_t)(e - s);
1446 size_t i = (n > 0 && s[0] == '-') ? 1 : 0;
1447 if (i >= n || n > 20) return false;
1448 if (s[i] == '0' && (n - i > 1 || i == 1)) return false;
1449 for (size_t k = i; k < n; k++) {
1450 if (s[k] < '0' || s[k] > '9') return false;
1451 }
1452 char buf[24];
1453 memcpy(buf, s, n);
1454 buf[n] = '\0';
1455 errno = 0;
1456 long long v = strtoll(buf, NULL, 10);
1457 if (errno == ERANGE) return false;
1458 *out = (int64_t)v;
1459 return true;
1460}

1541static int orm_b64_value(unsigned char c) {
1542 if (c >= 'A' && c <= 'Z') return c - 'A';
1543 if (c >= 'a' && c <= 'z') return c - 'a' + 26;
1544 if (c >= '0' && c <= '9') return c - '0' + 52;
1545 if (c == '+') return 62;
1546 if (c == '/') return 63;
1547 return -1;
1548}

1555static unsigned char *orm_b64_decode(const char *in, size_t *out_len) {
1556 size_t n = strlen(in);
1557 if (n % 4 != 0) {
1558 *out_len = SIZE_MAX;
1559 return NULL;
1560 }
1561 size_t pad = 0;
1562 if (n > 0 && in[n - 1] == '=') pad++;
1563 if (n > 1 && in[n - 2] == '=') pad++;
1564 size_t cap = n / 4 * 3 - pad;
1565 unsigned char *out = (unsigned char *)malloc(cap ? cap : 1);
1566 if (!out) {
1567 *out_len = SIZE_MAX;
1568 return NULL;
1569 }
1570 size_t o = 0;
1571 for (size_t i = 0; i < n; i += 4) {
1572 int v[4];
1573 int chars = 3;
1574 for (int k = 0; k < 4; k++) {
1575 char c = in[i + k];
1576 if (c == '=') {
1577 if (i + 4 != n || k < 2 || (k == 2 && in[i + 3] != '=')) {
1578 free(out);
1579 *out_len = SIZE_MAX;
1580 return NULL;
1581 }
1582 v[k] = 0;
1583 chars--;
1584 } else {
1585 v[k] = orm_b64_value((unsigned char)c);
1586 if (v[k] < 0) {
1587 free(out);
1588 *out_len = SIZE_MAX;
1589 return NULL;
1590 }
1591 }
1592 }
1593 if (i + 4 == n &&
1594 ((chars == 1 && (v[1] & 0x0f) != 0) || (chars == 2 && (v[2] & 0x03) != 0))) {
1595 free(out);
1596 *out_len = SIZE_MAX;
1597 return NULL;
1598 }
1599 unsigned triple = ((unsigned)v[0] << 18) | ((unsigned)v[1] << 12) | ((unsigned)v[2] << 6) |
1600 (unsigned)v[3];
1601 if (o < cap) out[o++] = (unsigned char)(triple >> 16);
1602 if (o < cap) out[o++] = (unsigned char)(triple >> 8);
1603 if (o < cap) out[o++] = (unsigned char)triple;
1604 }
1605 *out_len = cap;
1606 return out;
1607}

1617static int orm_fill_typed_cell(cwist_orm_value_t *v, const cJSON *cell) {
1618 const cJSON *tag = cJSON_GetObjectItemCaseSensitive(cell, "t");
1619 if (!cJSON_IsString(tag) || !tag->valuestring) return CWIST_ERROR_PROTOCOL;
1620 const char *t = tag->valuestring;
1621 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(cell, "v");
1622
1623 if (strcmp(t, "null") == 0) {
1624 v->type = CWIST_ORM_TYPE_NULL;
1625 return CWIST_SUCCESS;
1626 }
1627 if (strcmp(t, "int") == 0) {
1628 if (!cJSON_IsString(payload) || !payload->valuestring) return CWIST_ERROR_PROTOCOL;
1629 const char *s = payload->valuestring;
1630 if (!*s) return CWIST_ERROR_PROTOCOL;
1631 const char *digits = s[0] == '-' ? s + 1 : s;
1632 if (!*digits || (digits[0] == '0' && digits[1] != '\0')) return CWIST_ERROR_PROTOCOL;
1633 for (const char *p = digits; *p; p++)
1634 if (*p < '0' || *p > '9') return CWIST_ERROR_PROTOCOL;
1635 errno = 0;
1636 char *endptr = NULL;
1637 long long val = strtoll(s, &endptr, 10);
1638 if (errno == ERANGE || !endptr || *endptr != '\0') return CWIST_ERROR_PROTOCOL;
1639 v->type = CWIST_ORM_TYPE_INTEGER;
1640 v->i64 = (int64_t)val;
1641 return CWIST_SUCCESS;
1642 }

1710

1712 cJSON *columns = cJSON_GetObjectItemCaseSensitive(typed, "columns");
1713 cJSON *trows = cJSON_GetObjectItemCaseSensitive(typed, "rows");
1714 if (!cJSON_IsArray(columns) || !cJSON_IsArray(trows)) goto done;
1715
1716 res = (cwist_orm_result_t *)calloc(1, sizeof(*res));
1717 if (!res) {
1718 err.error.err_i16 = CWIST_ERROR_NOMEM;
1719 goto done;
1720 }
1721
1722 res->column_count = (size_t)cJSON_GetArraySize(columns);
1723 res->row_count = (size_t)cJSON_GetArraySize(trows);
1724
1725 res->columns = (char **)calloc(res->column_count ? res->column_count : 1, sizeof(char *));
1726 if (!res->columns) {
1727 err.error.err_i16 = CWIST_ERROR_NOMEM;
1728 goto fail;
1729 }

1731 cJSON *name = cJSON_GetArrayItem(columns, (int)i);
1732 if (!cJSON_IsString(name) || !name->valuestring) goto fail;
1733 res->columns[i] = strdup(name->valuestring);
1734 if (!res->columns[i]) {
1735 err.error.err_i16 = CWIST_ERROR_NOMEM;
1736 goto fail;
1737 }
1738 }
1739

1743 goto fail;
1744 res->cells = (cwist_orm_value_t *)calloc(res->row_count * res->column_count,

1747 err.error.err_i16 = CWIST_ERROR_NOMEM;
1748 goto fail;
1749 }
1750 }
1751

1753 cJSON *trow = NULL;
1754 cJSON_ArrayForEach(trow, trows) {

1756 goto fail;

1759 if (rc != CWIST_SUCCESS) {

1762 }
1763 }
1764 }
1765

1769
1770fail:

1772done:

1775}
