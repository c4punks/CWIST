58static int db_sync_connect(const struct addrinfo *addresses) {
59 for (const struct addrinfo *ai = addresses; ai; ai = ai->ai_next) {
60 int fd = socket(ai->ai_family, ai->ai_socktype, ai->ai_protocol);
61 if (fd < 0) continue;
62 int flags = fcntl(fd, F_GETFL, 0);
63 if (flags < 0 || fcntl(fd, F_SETFL, flags | O_NONBLOCK) != 0) {
64 close(fd);
65 continue;
66 }
67 int rc = connect(fd, ai->ai_addr, ai->ai_addrlen);
68 if (rc != 0 && (errno == EINPROGRESS || errno == EINTR)) {
69 struct pollfd pfd = {.fd = fd, .events = POLLOUT};
70 do {
71 rc = poll(&pfd, 1, CWIST_DB_SYNC_TIMEOUT_SEC * 1000);
72 } while (rc < 0 && errno == EINTR);
73 if (rc > 0) {
74 int error = 0;
75 socklen_t len = sizeof(error);
76 rc = getsockopt(fd, SOL_SOCKET, SO_ERROR, &error, &len) == 0 && error == 0 ? 0 : -1;
77 } else {
78 rc = -1;
79 }
80 }
81 if (rc != 0 || fcntl(fd, F_SETFL, flags) != 0 || db_sync_configure_socket(fd) != 0) {
82 close(fd);
83 continue;
84 }
85 return fd;
86 }
87 return -1;
88}

96static int db_sync_wait(int fd, short events, uint64_t deadline) {
97 for (;;) {
98 uint64_t now = db_sync_now_ms();
99 if (now >= deadline) return -1;
100 struct pollfd pfd = {.fd = fd, .events = events};
101 int rc = poll(&pfd, 1, (int)(deadline - now));
102 if (rc < 0 && errno == EINTR) continue;
103 if (rc <= 0 || (pfd.revents & (POLLERR | POLLNVAL))) return -1;
104 return 0;
105 }
106}

115static int db_sync_write_all(int fd, const void *buf, size_t len, uint64_t deadline) {
116 const unsigned char *p = (const unsigned char *)buf;
117 while (len > 0) {
118 if (db_sync_wait(fd, POLLOUT, deadline) != 0) return -1;
119 ssize_t n = send(fd, p, len, MSG_NOSIGNAL);
120 if (n < 0 && errno == EINTR) continue;
121 if (n <= 0) return -1;
122 p += (size_t)n;
123 len -= (size_t)n;
124 }

148static int db_sync_mac(const cwist_db_crypt_ctx_t *ctx, const char *domain,
149 const unsigned char client_nonce[DB_SYNC_NONCE_LEN],
150 const unsigned char server_nonce[DB_SYNC_NONCE_LEN],
151 const unsigned char *data, size_t data_len,
152 unsigned char out[DB_SYNC_MAC_LEN]) {
153 static const unsigned char zero_nonce[DB_SYNC_NONCE_LEN];
154 HMAC_CTX *mac = HMAC_CTX_new();
155 if (!mac) return -1;
156 unsigned int out_len = 0;
157 int ok = HMAC_Init_ex(mac, ctx->kek, sizeof(ctx->kek), EVP_sha256(), NULL) == 1 &&
158 HMAC_Update(mac, (const unsigned char *)domain, strlen(domain)) == 1 &&
159 HMAC_Update(mac, client_nonce, DB_SYNC_NONCE_LEN) == 1 &&
160 HMAC_Update(mac, server_nonce ? server_nonce : zero_nonce, DB_SYNC_NONCE_LEN) == 1 &&
161 (!data_len || HMAC_Update(mac, data, data_len) == 1) &&
162 HMAC_Final(mac, out, &out_len) == 1 && out_len == DB_SYNC_MAC_LEN;
163 HMAC_CTX_free(mac);
164 return ok ? 0 : -1;
165}

180 unsigned char *db_bytes = sqlite3_serialize(db, "main", &db_size, 0);

291int cwist_db_sync_pull(const char *host, int port, const cwist_db_crypt_ctx_t *ctx,
292 unsigned char **out_bytes, size_t *out_len) {
293 if (!host || !ctx || !out_bytes || !out_len) return CWIST_DB_SYNC_ERR_ARGS;
294 if (port <= 0) port = CWIST_DB_SYNC_DEFAULT_PORT;
295 uint64_t deadline = db_sync_now_ms() + CWIST_DB_SYNC_TIMEOUT_SEC * 1000;
296
297 /* 1. Resolve host. */
298 struct addrinfo hints, *res = NULL;
299 memset(&hints, 0, sizeof(hints));
300 hints.ai_family = AF_UNSPEC;
301 hints.ai_socktype = SOCK_STREAM;
302
303 char port_str[16];
304 snprintf(port_str, sizeof(port_str), "%d", port);
305
306 if (getaddrinfo(host, port_str, &hints, &res) != 0 || !res) {
307 fprintf(stderr, "[db_sync] getaddrinfo failed for %s\n", host);
308 return CWIST_DB_SYNC_ERR_NET;
309 }
310
311 /* 2. Connect. */
312 int fd = db_sync_connect(res);
313 freeaddrinfo(res);
314 if (fd < 0) {
315 fprintf(stderr, "[db_sync] connect to %s:%d failed\n", host, port);
316 return CWIST_DB_SYNC_ERR_NET;
317 }
318
319 unsigned char client_nonce[DB_SYNC_NONCE_LEN];
320 unsigned char client_auth[DB_SYNC_MAC_LEN];
321 if (RAND_bytes(client_nonce, sizeof(client_nonce)) != 1 ||
322 db_sync_mac(ctx, "CWSY client", client_nonce, NULL, NULL, 0, client_auth) != 0) {
323 close(fd);
324 return CWIST_DB_SYNC_ERR_CRYPTO;
325 }
326 if (db_sync_write_all(fd, SYNC_MAGIC, 4, deadline) < 0 ||
327 db_sync_write_all(fd, client_nonce, sizeof(client_nonce), deadline) < 0 ||
328 db_sync_write_all(fd, client_auth, sizeof(client_auth), deadline) < 0) {
329 close(fd);
330 return CWIST_DB_SYNC_ERR_NET;
331 }
332
333 unsigned char server_nonce[DB_SYNC_NONCE_LEN];
334 unsigned char server_auth[DB_SYNC_MAC_LEN];
335 unsigned char expected[DB_SYNC_MAC_LEN];
336 unsigned char len_buf[8];
337 if (read_all(fd, server_nonce, sizeof(server_nonce), deadline) < 0 ||
338 read_all(fd, server_auth, sizeof(server_auth), deadline) < 0 ||
339 db_sync_mac(ctx, "CWSY server", client_nonce, server_nonce, NULL, 0, expected) != 0 ||
340 CRYPTO_memcmp(expected, server_auth, DB_SYNC_MAC_LEN) != 0 ||
341 read_all(fd, len_buf, sizeof(len_buf), deadline) < 0) {
342 close(fd);
343 return CWIST_DB_SYNC_ERR_PROTO;
344 }
345 uint64_t blob_len_u64 = 0;
346 for (int i = 0; i < 8; i++) blob_len_u64 |= ((uint64_t)len_buf[i]) << (8 * i);
347 if (blob_len_u64 == 0) {
348 close(fd);
349 return CWIST_DB_SYNC_ERR_PROTO;
350 }
351
352 if (blob_len_u64 > CWIST_DB_SYNC_MAX_BLOB_SIZE || blob_len_u64 > (uint64_t)SIZE_MAX) {
353 fprintf(stderr, "[db_sync] blob_len %" PRIu64 " exceeds limit\n", blob_len_u64);
354 close(fd);
355 return CWIST_DB_SYNC_ERR_PROTO;
356 }
357 size_t blob_len = (size_t)blob_len_u64;
358 unsigned char *blob = (unsigned char *)malloc(blob_len);
359 if (!blob) {
360 close(fd);
361 return CWIST_DB_SYNC_ERR_MEM;
362 }
363
364 /* 5. Read the snapshot and its challenge-bound authentication tag. */
365 unsigned char blob_auth[DB_SYNC_MAC_LEN];
366 if (read_all(fd, blob, blob_len, deadline) < 0 ||
367 read_all(fd, blob_auth, sizeof(blob_auth), deadline) < 0) {
368 free(blob);
369 close(fd);
370 return CWIST_DB_SYNC_ERR_NET;
371 }
372 close(fd);
373 if (db_sync_mac(ctx, "CWSY blob", client_nonce, server_nonce, blob, blob_len, expected) != 0 ||
374 CRYPTO_memcmp(expected, blob_auth, DB_SYNC_MAC_LEN) != 0) {
375 free(blob);
376 return CWIST_DB_SYNC_ERR_CRYPTO;
377 }
378
379 /* 6. Decrypt. */
380 size_t pt_len = 0;
381 unsigned char *pt = cwist_db_crypt_open(ctx, blob, blob_len, &pt_len);
382 free(blob);
383
384 if (!pt) {
385 fprintf(stderr, "[db_sync] decrypt failed\n");
386 return CWIST_DB_SYNC_ERR_CRYPTO;
387 }
388
389 *out_bytes = pt;
390 *out_len = pt_len;
391 return CWIST_DB_SYNC_OK;
392}
