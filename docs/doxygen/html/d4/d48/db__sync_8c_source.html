51static int db_sync_connect(const struct addrinfo *addresses) {
52 for (const struct addrinfo *ai = addresses; ai; ai = ai->ai_next) {
53 int fd = socket(ai->ai_family, ai->ai_socktype, ai->ai_protocol);
54 if (fd < 0) continue;
55 int flags = fcntl(fd, F_GETFL, 0);
56 if (flags < 0 || fcntl(fd, F_SETFL, flags | O_NONBLOCK) != 0) {
57 close(fd);
58 continue;
59 }
60 int rc = connect(fd, ai->ai_addr, ai->ai_addrlen);
61 if (rc != 0 && (errno == EINPROGRESS || errno == EINTR)) {
62 struct pollfd pfd = {.fd = fd, .events = POLLOUT};
63 do {
64 rc = poll(&pfd, 1, CWIST_DB_SYNC_TIMEOUT_SEC * 1000);
65 } while (rc < 0 && errno == EINTR);
66 if (rc > 0) {
67 int error = 0;
68 socklen_t len = sizeof(error);
69 rc = getsockopt(fd, SOL_SOCKET, SO_ERROR, &error, &len) == 0 && error == 0 ? 0 : -1;
70 } else {
71 rc = -1;
72 }
73 }
74 if (rc != 0 || fcntl(fd, F_SETFL, flags) != 0 || db_sync_configure_socket(fd) != 0) {
75 close(fd);
76 continue;
77 }

79 }

183 close(srv_fd);

212 }

215 free(blob);
216 return CWIST_DB_SYNC_OK;

262 unsigned char len_buf[8];

268 for (int i = 0; i < 8; i++) blob_len_u64 |= ((uint64_t)len_buf[i]) << (8 * i);
269 if (blob_len_u64 == 0) {

271 return CWIST_DB_SYNC_ERR_PROTO;
272 }
273

276 close(fd);
277 return CWIST_DB_SYNC_ERR_PROTO;
278 }
279 size_t blob_len = (size_t)blob_len_u64;
280 unsigned char *blob = (unsigned char *)malloc(blob_len);
281 if (!blob) {

284 }

292 close(fd);
