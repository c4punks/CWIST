135static int redis_recv_line(cwist_redis_t *r, char *line, size_t line_len) {
136 size_t i = 0;
137 while (1) {
138 if (r->recv_len > 0) {
139 while (r->recv_len > 0 && i < line_len - 1) {
140 char c = r->recv_buf[0];
141 /* simple shift */
142 memmove(r->recv_buf, r->recv_buf + 1, r->recv_len - 1);
143 r->recv_len--;
144 if (c == '\n') {
145 if (i > 0 && line[i - 1] == '\r') i--;
146 line[i] = '\0';
147 return 0;
148 }
149 line[i++] = c;
150 }
151 if (i >= line_len - 1) return -1;
152 }
153 ssize_t n =
154 recv(r->fd, r->recv_buf + r->recv_len, CWIST_REDIS_BUF_SIZE - r->recv_len - 1, 0);
155 if (n < 0) {
156 if (errno == EINTR) continue;
157 return -1;
158 }
159 if (n == 0) return -1;
160 r->recv_len += (size_t)n;
161 }

168static int redis_recv_bytes(cwist_redis_t *r, char *out, size_t len) {
169 size_t got = 0;
170 while (got < len) {
171 if (r->recv_len > 0) {
172 size_t take = r->recv_len < (len - got) ? r->recv_len : (len - got);
173 memcpy(out + got, r->recv_buf, take);
174 memmove(r->recv_buf, r->recv_buf + take, r->recv_len - take);
175 r->recv_len -= take;
176 got += take;
177 continue;
178 }
179 ssize_t n =

181 if (n < 0) {

183 return -1;
184 }
185 if (n == 0) return -1;
186 r->recv_len += (size_t)n;
187 }
188 return 0;

202static int redis_remaining_ms(uint64_t deadline) {
203 if (deadline == UINT64_MAX) return INT_MAX;

236static int redis_send_deadline(int fd, const char *buf, size_t len, uint64_t deadline) {
237 size_t sent = 0;
238 while (sent < len) {
239 if (redis_now_ns() >= deadline) return -ETIMEDOUT;
240 ssize_t n = send(fd, buf + sent, len - sent, CWIST_REDIS_NOSIGNAL);
241 if (n > 0) {
242 sent += (size_t)n;
243 continue;
244 }
245 if (n < 0 && errno == EINTR) continue;
246 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK || errno == ENOBUFS)) {
247 int rc = redis_wait_fd(fd, POLLOUT, deadline);
248 if (rc != 0) return rc;
249 continue;
250 }
251 return -EIO;
252 }
253 return 0;

327static bool redis_parse_ll(const char *s, size_t n, long long *out) {
328 size_t i = 0;
329 bool neg = n > 0 && s[0] == '-';
330 if (neg) i = 1;
331 if (i >= n) return false;
332 unsigned long long limit =
333 neg ? (unsigned long long)LLONG_MAX + 1u : (unsigned long long)LLONG_MAX;
334 unsigned long long v = 0;
335 for (; i < n; i++) {
336 if (s[i] < '0' || s[i] > '9') return false;
337 unsigned d = (unsigned)(s[i] - '0');
338 if (v > (limit - d) / 10u) return false;
339 v = v * 10u + d;
340 }
341 if (!neg)
342 *out = (long long)v;
343 else if (v == (unsigned long long)LLONG_MAX + 1u)
344 *out = LLONG_MIN;
345 else
346 *out = -(long long)v;
347 return true;
348}

378static int redis_reconnect(const cwist_redis_pool_t *pool, cwist_redis_t *r, uint64_t deadline) {
379 if (pool->peer_len == 0) return -EIO;
380 int fd = socket(pool->peer.ss_family, SOCK_STREAM, 0);
381 if (fd < 0) return -EIO;
382 int rc = -EIO;
383 (void)fcntl(fd, F_SETFD, FD_CLOEXEC);
384#ifdef SO_NOSIGPIPE
385 int one = 1;
386 (void)setsockopt(fd, SOL_SOCKET, SO_NOSIGPIPE, &one, sizeof(one));
387#endif
388 int flags = fcntl(fd, F_GETFL, 0);
389 if (flags < 0 || fcntl(fd, F_SETFL, flags | O_NONBLOCK) != 0) goto out;
390 if (connect(fd, (const struct sockaddr *)&pool->peer, pool->peer_len) != 0) {
391 if (errno != EINPROGRESS && errno != EINTR) goto out;
392 rc = redis_wait_fd(fd, POLLOUT, deadline);
393 if (rc != 0) goto out;
394 rc = -EIO;
395 int soerr = 0;
396 socklen_t sl = sizeof(soerr);
397 if (getsockopt(fd, SOL_SOCKET, SO_ERROR, &soerr, &sl) != 0 || soerr != 0) goto out;
398 }
399 if (fcntl(fd, F_SETFL, flags) != 0) goto out;
400 r->fd = fd;
401 r->recv_len = 0;
402 return 0;
403out:
404 close(fd);
405 return rc;
406}

449static cwist_error_t redis_read_resp(cwist_redis_t *r, int depth, cwist_redis_reply_t **out) {
450 char line[CWIST_REDIS_LINE_MAX];
451 cwist_redis_reply_t *node = cwist_alloc(sizeof(*node));
452 if (!node) return make_error(CWIST_ERR_INT16);
453 *out = node;
454 if (redis_recv_line(r, line, sizeof(line)) != 0) goto fail;
455 node->type = line[0];
456
457 switch (line[0]) {
458 case '+': /* simple string */
459 case '-': /* error */
460 node->str = cwist_strdup(line + 1);
461 if (!node->str) goto fail;
462 node->len = strlen(line + 1);
463 /* Legacy behavior: a '-' reply is signalled through the reply
464 * tree (type '-') with a zeroed error value. */
465 return (line[0] == '-')
466 ? make_error(CWIST_ERR_INT16)
467 : (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
468
469 case ':': { /* integer */
470 char *end = NULL;
471 long long v = strtoll(line + 1, &end, 10);
472 if (end == line + 1 || *end != '\0') goto fail;
473 node->str = cwist_strdup(line + 1);
474 if (!node->str) goto fail;
475 node->len = strlen(line + 1);
476 node->integer = v;
477 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
478 }
479
480 case '$': { /* bulk string */
481 char *end = NULL;
482 long long blen = strtoll(line + 1, &end, 10);
483 if (end == line + 1 || *end != '\0') goto fail;
484 if (blen < 0) return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
485 /* Guard against a rogue/malicious server advertising a huge bulk
486 * string that would exhaust memory before we read a single byte. */
487 if (blen > CWIST_REDIS_MAX_BULK_BYTES) goto fail;
488 char *buf = cwist_alloc((size_t)blen + 1);
489 if (!buf) goto fail;
490 if (redis_recv_bytes(r, buf, (size_t)blen) != 0) {
491 cwist_free(buf);
492 goto fail;
493 }
494 buf[blen] = '\0';
495 /* consume trailing \r\n */
496 char crlf[2];
497 if (redis_recv_bytes(r, crlf, 2) != 0) {
498 cwist_free(buf);
499 goto fail;
500 }
501 node->str = buf;
502 node->len = (size_t)blen;
503 node->integer = blen;
504 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
505 }
506
507 case '*': { /* array */
508 if (depth >= CWIST_REDIS_MAX_REPLY_DEPTH) goto fail;
509 char *end = NULL;
510 long long count = strtoll(line + 1, &end, 10);
511 if (end == line + 1 || *end != '\0' || count < 0 ||
512 count > CWIST_REDIS_MAX_REPLY_ELEMENTS)
513 goto fail;
514 node->elements = (size_t)count;
515 if (count > 0) {
516 node->element = cwist_alloc_array((size_t)count, sizeof(*node->element));
517 if (!node->element) goto fail;
518 for (long long i = 0; i < count; i++) {
519 node->element[i] = NULL;
520 cwist_error_t e = redis_read_resp(r, depth + 1, &node->element[i]);
521 if (e.error.err_i16 != 0) goto fail;
522 }
523 }
524 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
525 }
526
527 default: goto fail;
528 }
529
530fail:
531 cwist_redis_reply_free(*out);
532 *out = NULL;
533 return make_error(CWIST_ERR_INT16);
534}

580static cwist_error_t redis_command_frame_tree(cwist_redis_t *r, const char *frame, size_t frame_len,
581 cwist_redis_reply_t **out_tree) {
582 if (!r || !frame) return make_error(CWIST_ERR_INT16);
583 if (out_tree) *out_tree = NULL;
584 /* A request/reply pair must be serialized as one critical section. Keeping
585 * send and receive separate allows two callers to consume each other's
586 * replies on the same connection. */
587 pthread_mutex_lock(&r->mtx);
588 cwist_error_t err = redis_send_all(r->fd, frame, frame_len) == 0
589 ? redis_read_resp(r, 0, out_tree)
590 : make_error(CWIST_ERR_INT16);
591 pthread_mutex_unlock(&r->mtx);
592 return err;
593}

663 cwist_error_t err = redis_command_frame_tree(r, frame->data, frame->size, reply);
664 cwist_sstring_destroy(frame);

666 * failure so callers can tell a server error from an empty array. */

668 cwist_redis_reply_free(*reply);
669 *reply = NULL;
670 return make_error(CWIST_ERR_INT16);
671 }
672 return err;
673fail:
674 cwist_sstring_destroy(frame);
675 return make_error(CWIST_ERR_INT16);
676}

975cwist_error_t cwist_redis_subscribe(cwist_redis_t *r, const char **channels, cwist_redis_msg_cb cb,
976 void *ctx) {
977 if (!r || !channels || !channels[0] || !cb) return make_error(CWIST_ERR_INT16);
978
979 cwist_sstring *cmd = cwist_sstring_create();
980 if (!cmd) return make_error(CWIST_ERR_INT16);
981 size_t count = 0;
982 while (channels[count]) count++;
983 char prefix[32];
984 snprintf(prefix, sizeof(prefix), "*%zu\r\n", count + 1);
985 cwist_sstring_append(cmd, prefix);
986 write_bulk_string(cmd, "SUBSCRIBE");
987 for (size_t i = 0; i < count; i++) {
988 write_bulk_string(cmd, channels[i]);
989 }
990 cwist_error_t err = send_command(r, cmd->data);
991 cwist_sstring_destroy(cmd);
992 if (err.error.err_i16 != 0) return err;
993
994 /* Consume subscription confirmations. */
995 for (size_t i = 0; i < count; i++) {
996 char *out = NULL;
997 err = recv_reply(r, &out);
998 cwist_free(out);
999 if (err.error.err_i16 != 0) return err;
1000 }
1001
1002 /* Blocking read loop for messages. */
1003 while (1) {
1004 char line[CWIST_REDIS_LINE_MAX];
1005 if (redis_recv_line(r, line, sizeof(line)) != 0) break;
1006 if (line[0] != '*') continue;
1007 char *arr_end = NULL;
1008 long long arr_count = strtoll(line + 1, &arr_end, 10);
1009 if (arr_end == line + 1 || *arr_end != '\0') continue;
1010 if (arr_count < 3) continue;
1011
1012 char *type = NULL;
1013 if (read_reply(r, &type, NULL, NULL).error.err_i16 != 0) break;
1014
1015 char *channel = NULL;
1016 if (read_reply(r, &channel, NULL, NULL).error.err_i16 != 0) {
1017 cwist_free(type);
1018 break;
1019 }
1020
1021 char *message = NULL;
1022 if (read_reply(r, &message, NULL, NULL).error.err_i16 != 0) {
1023 cwist_free(type);
1024 cwist_free(channel);
1025 break;
1026 }
1027
1028 if (type && strcmp(type, "message") == 0) {
1029 cb(channel, message, ctx);
1030 }
1031 cwist_free(type);
1032 cwist_free(channel);
1033 cwist_free(message);
1034 }
1035 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
1036}

1046cwist_redis_pool_t *cwist_redis_pool_create(const char *host, int port, size_t max_conns) {
1047 if (!host || port <= 0 || max_conns == 0) return NULL;
1048
1049 cwist_redis_pool_t *pool = cwist_alloc(sizeof(*pool));
1050 if (!pool) return NULL;
1051 pool->wake_rd = -1;
1052 pool->wake_wr = -1;
1053 pool->host = cwist_strdup(host);
1054 pool->port = port;
1055 pool->max_conns = max_conns;
1056 pool->conns = cwist_alloc_array(max_conns, sizeof(cwist_redis_t *));
1057 pool->available = cwist_alloc_array(max_conns, sizeof(bool));
1058 if (!pool->host || !pool->conns || !pool->available) goto fail;
1059
1060 for (size_t i = 0; i < max_conns; i++) {
1061 pool->conns[i] = cwist_redis_connect(host, port);
1062 if (!pool->conns[i]) goto fail;
1063 pool->available[i] = true;
1064 }
1065
1066 /* Cache the resolved peer once so request-time reconnects need no DNS. */
1067 pool->peer_len = (socklen_t)sizeof(pool->peer);
1068 if (getpeername(pool->conns[0]->fd, (struct sockaddr *)&pool->peer, &pool->peer_len) != 0)
1069 pool->peer_len = 0;
1070 pool->owner_pid = getpid();
1071
1072 int wake[2];
1073 if (pipe(wake) == 0) {
1074 bool ok = true;
1075 for (int i = 0; i < 2; i++) {
1076 int fl = fcntl(wake[i], F_GETFL, 0);
1077 ok = ok && fl >= 0 && fcntl(wake[i], F_SETFL, fl | O_NONBLOCK) == 0 &&
1078 fcntl(wake[i], F_SETFD, FD_CLOEXEC) == 0;
1079 }
1080 if (ok) {
1081 pool->wake_rd = wake[0];
1082 pool->wake_wr = wake[1];
1083 } else {
1084 close(wake[0]);
1085 close(wake[1]);
1086 }
1087 }
1088
1089 pthread_mutex_init(&pool->mtx, NULL);
1090 pthread_cond_init(&pool->cond, NULL);
1091 return pool;
1092
1093fail:
1094 if (pool) {
1095 for (size_t i = 0; i < max_conns && pool->conns; i++) {
1096 if (pool->conns[i]) cwist_redis_close(pool->conns[i]);
1097 }
1098 if (pool->wake_rd >= 0) close(pool->wake_rd);
1099 if (pool->wake_wr >= 0) close(pool->wake_wr);
1100 cwist_free(pool->conns);
1101 cwist_free(pool->available);
1102 cwist_free(pool->host);
1103 cwist_free(pool);
1104 }
1105 return NULL;
1106}

1131static cwist_redis_t *redis_pool_acquire(cwist_redis_pool_t *pool) {
1132 if (!pool) return NULL;
1133 pthread_mutex_lock(&pool->mtx);
1134 while (1) {
1135 for (size_t i = 0; i < pool->max_conns; i++) {
1136 if (pool->available[i]) {
1137 pool->available[i] = false;
1138 pthread_mutex_unlock(&pool->mtx);
1139 cwist_redis_t *r = pool->conns[i];
1140 if (r->fd < 0) (void)redis_reconnect(pool, r, UINT64_MAX);
1141 return r;
1142 }
1143 }
1144 pthread_cond_wait(&pool->cond, &pool->mtx);
1145 }

1161 (void)n;
1162 }
1163 break;
1164 }
1165 }

1236
1247static void redis_pool_wait_until(cwist_redis_pool_t *pool, uint64_t deadline) {
1248 uint64_t now = redis_now_ns();
1249 if (now >= deadline) return;
1250 uint64_t rem = deadline - now;
1251 if (pool->wake_rd >= 0) {
1252 uint64_t ms = (rem + 999999u) / 1000000u;
1253 struct pollfd pfd = {.fd = pool->wake_rd, .events = POLLIN, .revents = 0};
1254 pool->timed_waiters++;
1255 pthread_mutex_unlock(&pool->mtx);
1256 (void)poll(&pfd, 1, ms > (uint64_t)INT_MAX ? INT_MAX : (int)ms);
1257 pthread_mutex_lock(&pool->mtx);
1258 pool->timed_waiters--;
1259 char sink[64];
1260 if (pool->timed_waiters == 0) {
1261 while (read(pool->wake_rd, sink, sizeof(sink)) > 0) {
1262 }
1263 } else {
1264 ssize_t n = read(pool->wake_rd, sink, 1);
1265 (void)n;
1266 }
1267 return;
1268 }
1269 /* Pipe allocation failed: use short monotonic rechecks, not a
1270 * realtime condition timer that can jump backwards or be coalesced. */
1271 int ms = redis_remaining_ms(deadline);
1272 pthread_mutex_unlock(&pool->mtx);
1273 if (ms > 0) (void)poll(NULL, 0, ms < 2 ? ms : 2);
1274 pthread_mutex_lock(&pool->mtx);
1275}

1283static int redis_pool_acquire_deadline(cwist_redis_pool_t *pool, uint64_t deadline,
1284 cwist_redis_t **out) {
1285 pthread_mutex_lock(&pool->mtx);
1286 for (;;) {
1287 size_t pick = pool->max_conns;
1288 for (size_t i = 0; i < pool->max_conns; i++) {
1289 if (!pool->available[i]) continue;
1290 if (pick == pool->max_conns) pick = i;
1291 if (pool->conns[i]->fd >= 0) {
1292 pick = i;
1293 break;
1294 }
1295 }
1296 if (pick < pool->max_conns) {
1297 pool->available[pick] = false;
1298 pthread_mutex_unlock(&pool->mtx);
1299 *out = pool->conns[pick];
1300 return 0;
1301 }
1302 if (redis_now_ns() >= deadline) {
1303 pthread_mutex_unlock(&pool->mtx);
1304 return -ETIMEDOUT;
1305 }
1306 redis_pool_wait_until(pool, deadline);
1307 }
1308}

1322static int redis_frame_size(size_t argc, const void *const *argv, const size_t *argv_lens,
1323 size_t *out) {
1324 if (argc > CWIST_REDIS_TIMED_MAX_ARGC) return -EINVAL;
1325 size_t total = 1 + redis_digits(argc) + 2;

1357static int redis_timed_read(redis_timed_parse *ps, int depth, cwist_redis_reply_t **out) {
1358 *out = NULL;
1359 if (++ps->nodes > CWIST_REDIS_TIMED_MAX_NODES) return -EPROTO;
1360 cwist_redis_t *r = ps->r;
1361 size_t n = 0;
1362 int rc = redis_peek_line(r, ps->deadline, &n);
1363 if (rc != 0) return rc;
1364 if (n == 0) return -EPROTO;
1365 const char *line = r->recv_buf;
1366 char type = line[0];
1367 long long v = 0;
1368 if ((type == ':' || type == '$' || type == '*') && !redis_parse_ll(line + 1, n - 1, &v))
1369 return -EPROTO;
1370 if (type != '+' && type != '-' && type != ':' && type != '$' && type != '*') return -EPROTO;
1371 if (n - 1 > CWIST_REDIS_TIMED_MAX_PAYLOAD - ps->payload) return -EPROTO;
1372
1373 cwist_redis_reply_t *node = cwist_alloc(sizeof(*node));
1374 if (!node) return -ENOMEM;
1375 node->type = type;
1376 rc = -EPROTO;
1377 switch (type) {
1378 case '+':
1379 case '-':
1380 case ':':
1381 ps->payload += n - 1;
1382 node->str = redis_copy_str(line + 1, n - 1);
1383 if (!node->str) {
1384 rc = -ENOMEM;
1385 goto fail;
1386 }
1387 node->len = n - 1;
1388 node->integer = type == ':' ? v : 0;
1389 redis_consume(r, n + 2);
1390 break;
1391 case '$': {
1392 redis_consume(r, n + 2);
1393 if (v == -1) {
1394 node->integer = -1;
1395 break;
1396 }
1397 if (v < 0 || (unsigned long long)v > CWIST_REDIS_TIMED_MAX_BULK ||
1398 (size_t)v > CWIST_REDIS_TIMED_MAX_PAYLOAD - ps->payload)
1399 goto fail;
1400 ps->payload += (size_t)v;
1401 node->str = cwist_alloc((size_t)v + 1);
1402 if (!node->str) {
1403 rc = -ENOMEM;
1404 goto fail;
1405 }
1406 rc = redis_read_exact(r, node->str, (size_t)v, ps->deadline);
1407 if (rc != 0) goto fail;
1408 node->str[v] = '\0';
1409 node->len = (size_t)v;
1410 node->integer = v;
1411 char crlf[2];
1412 rc = redis_read_exact(r, crlf, 2, ps->deadline);
1413 if (rc != 0) goto fail;
1414 if (crlf[0] != '\r' || crlf[1] != '\n') {
1415 rc = -EPROTO;
1416 goto fail;
1417 }
1418 break;
1419 }
1420 case '*':
1421 redis_consume(r, n + 2);
1422 if (v == -1) {
1423 node->integer = -1;
1424 break;
1425 }
1426 if (v < 0 || depth >= CWIST_REDIS_TIMED_MAX_DEPTH ||
1427 (unsigned long long)v > CWIST_REDIS_TIMED_MAX_NODES - ps->nodes)
1428 goto fail;
1429 node->integer = v;
1430 if (v > 0) {
1431 node->element = cwist_alloc_array((size_t)v, sizeof(*node->element));
1432 if (!node->element) {
1433 rc = -ENOMEM;
1434 goto fail;
1435 }
1436 node->elements = (size_t)v;
1437 for (size_t i = 0; i < (size_t)v; i++) {
1438 node->element[i] = NULL;
1439 rc = redis_timed_read(ps, depth + 1, &node->element[i]);
1440 if (rc != 0) goto fail;
1441 }
1442 }
1443 break;
1444 }
1445 *out = node;
1446 return 0;
1447fail:
1448 cwist_redis_reply_free(node);
1449 return rc;
1450}

1453cwist_error_t cwist_redis_pool_command_argv_reply_timeout(cwist_redis_pool_t *pool, size_t argc,
1454 const void *const *argv,
1455 const size_t *argv_lens,
1456 unsigned timeout_ms,
1457 cwist_redis_reply_t **reply) {
1458 if (reply) *reply = NULL;
1459 if (!pool || !argc || !argv || !argv_lens || !reply || timeout_ms == 0)
1460 return redis_code_error(-EINVAL);
1461 const uint64_t deadline = redis_now_ns() + (uint64_t)timeout_ms * 1000000u;
1462 /* A fork() child shares the parent's sockets and may hold a copy of a
1463 * locked mutex: reject before touching any pool state. */
1464 if (pool->owner_pid != getpid()) return redis_code_error(-EPERM);
1465
1466 size_t frame_len = 0;
1467 int rc = redis_frame_size(argc, argv, argv_lens, &frame_len);
1468 if (rc != 0) return redis_code_error(rc);
1469 char *frame = cwist_malloc(frame_len + 1);
1470 if (!frame) return redis_code_error(-ENOMEM);
1471 rc = redis_frame_fill(frame, frame_len + 1, argc, argv, argv_lens);
1472 if (rc != 0) {
1473 cwist_free(frame);
1474 return redis_code_error(rc);
1475 }
1476
1477 cwist_redis_t *r = NULL;
1478 rc = redis_pool_acquire_deadline(pool, deadline, &r);
1479 if (rc != 0) {
1480 cwist_free(frame);
1481 return redis_code_error(rc);
1482 }
1483
1484 cwist_redis_reply_t *tree = NULL;
1485 pthread_mutex_lock(&r->mtx);
1486 if (redis_now_ns() >= deadline) {
1487 rc = -ETIMEDOUT;
1488 } else {
1489 /* Nothing has been sent yet, so replacing a stale, hung-up, or
1490 * previously discarded connection is not a retry. */
1491 if (r->fd >= 0 && redis_conn_unclean(r)) redis_conn_discard(r);
1492 rc = r->fd >= 0 ? 0 : redis_reconnect(pool, r, deadline);
1493 }
1494 if (rc == 0) {
1495#if !defined(MSG_NOSIGNAL) && defined(SO_NOSIGPIPE)
1496 /* No per-send flag: mark the (possibly legacy-created) socket. */
1497 int nosig = 1;
1498 (void)setsockopt(r->fd, SOL_SOCKET, SO_NOSIGPIPE, &nosig, sizeof(nosig));
1499#endif
1500 /* Non-blocking only for this call; legacy APIs expect blocking fds. */
1501 int fl = fcntl(r->fd, F_GETFL, 0);
1502 if (fl < 0 || fcntl(r->fd, F_SETFL, fl | O_NONBLOCK) != 0) rc = -EIO;
1503 /* From the first written byte on, the command is ambiguous: any
1504 * failure discards the connection and is never retried. */
1505 if (rc == 0) rc = redis_send_deadline(r->fd, frame, frame_len, deadline);
1506 if (rc == 0) {
1507 redis_timed_parse ps = {.r = r, .deadline = deadline, .nodes = 0, .payload = 0};
1508 rc = redis_timed_read(&ps, 0, &tree);
1509 }
1510 if (rc == 0 && fcntl(r->fd, F_SETFL, fl) != 0) rc = -EIO;
1511 /* Bytes beyond the reply mean the peer is out of sync with us. */
1512 if (rc != 0 || r->recv_len != 0) redis_conn_discard(r);
1513 if (rc == 0 && tree->type == '-') {
1514 cwist_redis_reply_free(tree);
1515 tree = NULL;
1516 rc = -EBADMSG;
1517 }
1518 }
1519 pthread_mutex_unlock(&r->mtx);
1520 redis_pool_release(pool, r);
1521 cwist_free(frame);
1522 if (rc != 0) {
1523 cwist_redis_reply_free(tree);
1524 return redis_code_error(rc);
1525 }
1526 *reply = tree;
1527 return redis_code_error(0);
1528}
