57static int redis_open_socket(const char *host, int port) {
58 struct addrinfo hints = {0}, *res = NULL;
59 hints.ai_family = AF_INET;
60 hints.ai_socktype = SOCK_STREAM;
61 char port_str[16];
62 snprintf(port_str, sizeof(port_str), "%d", port);
63 if (getaddrinfo(host, port_str, &hints, &res) != 0 || !res) return -1;
64
65 int fd = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
66 if (fd < 0) {
67 freeaddrinfo(res);
68 return -1;
69 }
70 if (connect(fd, res->ai_addr, res->ai_addrlen) != 0) {
71 close(fd);
72 freeaddrinfo(res);
73 return -1;
74 }
75 freeaddrinfo(res);
76 return fd;
77}

82static int redis_send_all(int fd, const char *buf, size_t len) {
83 size_t sent = 0;
84 while (sent < len) {
85 /* RESP speaks plaintext TCP by design (same as redis-cli without
86 * --tls). Credentials must be protected at the deployment layer:
87 * loopback/Unix sockets, private network, or a TLS-terminating proxy
88 * such as stunnel/spiped in front of Redis. */
89 ssize_t n = send(fd, buf + sent, len - sent, 0); // codeql[cpp/cleartext-transmission]
90 if (n < 0) {
91 if (errno == EINTR) continue;
92 return -1;
93 }
94 if (n == 0) return -1;
95 sent += (size_t)n;
96 }
97 return 0;
98}

106static int redis_recv_line(cwist_redis_t *r, char *line, size_t line_len) {
107 size_t i = 0;
108 while (1) {
109 if (r->recv_len > 0) {
110 while (r->recv_len > 0 && i < line_len - 1) {
111 char c = r->recv_buf[0];
112 /* simple shift */
113 memmove(r->recv_buf, r->recv_buf + 1, r->recv_len - 1);
114 r->recv_len--;
115 if (c == '\n') {
116 if (i > 0 && line[i - 1] == '\r') i--;
117 line[i] = '\0';
118 return 0;
119 }
120 line[i++] = c;
121 }
122 if (i >= line_len - 1) return -1;
123 }
124 ssize_t n =
125 recv(r->fd, r->recv_buf + r->recv_len, CWIST_REDIS_BUF_SIZE - r->recv_len - 1, 0);
126 if (n < 0) {
127 if (errno == EINTR) continue;
128 return -1;
129 }
130 if (n == 0) return -1;
131 r->recv_len += (size_t)n;
132 }
133}

139static int redis_recv_bytes(cwist_redis_t *r, char *out, size_t len) {
140 size_t got = 0;
141 while (got < len) {
142 if (r->recv_len > 0) {
143 size_t take = r->recv_len < (len - got) ? r->recv_len : (len - got);
144 memcpy(out + got, r->recv_buf, take);
145 memmove(r->recv_buf, r->recv_buf + take, r->recv_len - take);
146 r->recv_len -= take;
147 got += take;
148 continue;
149 }
150 ssize_t n =
151 recv(r->fd, r->recv_buf + r->recv_len, CWIST_REDIS_BUF_SIZE - r->recv_len - 1, 0);
152 if (n < 0) {
153 if (errno == EINTR) continue;
154 return -1;
155 }
156 if (n == 0) return -1;
157 r->recv_len += (size_t)n;
158 }
159 return 0;
160}

180static void flatten_reply(const cwist_redis_reply_t *tree, char **out_value, size_t *out_len) {
181 if (out_value) *out_value = NULL;
182 if (out_len) *out_len = 0;
183 if (!tree) return;
184 if (out_value) {
185 if ((tree->type == '+' || tree->type == ':') && tree->str) {
186 *out_value = cwist_strdup(tree->str);
187 if (out_len) *out_len = strlen(tree->str);
188 } else if (tree->type == '$' && tree->str) {
189 char *buf = cwist_alloc(tree->len + 1);
190 if (!buf) return; /* legacy contract: allocation failure yields NULL */
191 memcpy(buf, tree->str, tree->len);
192 buf[tree->len] = '\0';
193 *out_value = buf;
194 if (out_len) *out_len = tree->len;
195 }
196 } else if (out_len && tree->str) {
197 *out_len = tree->len;
198 }
199}

203static cwist_error_t redis_read_resp(cwist_redis_t *r, int depth, cwist_redis_reply_t **out) {
204 char line[CWIST_REDIS_LINE_MAX];
205 cwist_redis_reply_t *node = cwist_alloc(sizeof(*node));
206 if (!node) return make_error(CWIST_ERR_INT16);
207 *out = node;
208 if (redis_recv_line(r, line, sizeof(line)) != 0) goto fail;
209 node->type = line[0];
210
211 switch (line[0]) {
212 case '+': /* simple string */
213 case '-': /* error */
214 node->str = cwist_strdup(line + 1);
215 if (!node->str) goto fail;
216 node->len = strlen(line + 1);
217 /* Legacy behavior: a '-' reply is signalled through the reply
218 * tree (type '-') with a zeroed error value. */
219 return (line[0] == '-')
220 ? make_error(CWIST_ERR_INT16)
221 : (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
222
223 case ':': { /* integer */
224 char *end = NULL;
225 long long v = strtoll(line + 1, &end, 10);
226 if (end == line + 1 || *end != '\0') goto fail;
227 node->str = cwist_strdup(line + 1);
228 if (!node->str) goto fail;
229 node->len = strlen(line + 1);
230 node->integer = v;
231 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
232 }
233
234 case '$': { /* bulk string */
235 char *end = NULL;
236 long long blen = strtoll(line + 1, &end, 10);
237 if (end == line + 1 || *end != '\0') goto fail;
238 if (blen < 0) return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
239 /* Guard against a rogue/malicious server advertising a huge bulk
240 * string that would exhaust memory before we read a single byte. */
241 if (blen > CWIST_REDIS_MAX_BULK_BYTES) goto fail;
242 char *buf = cwist_alloc((size_t)blen + 1);
243 if (!buf) goto fail;
244 if (redis_recv_bytes(r, buf, (size_t)blen) != 0) {
245 cwist_free(buf);
246 goto fail;
247 }
248 buf[blen] = '\0';
249 /* consume trailing \r\n */
250 char crlf[2];
251 if (redis_recv_bytes(r, crlf, 2) != 0) {
252 cwist_free(buf);
253 goto fail;
254 }
255 node->str = buf;
256 node->len = (size_t)blen;
257 node->integer = blen;
258 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
259 }
260
261 case '*': { /* array */
262 if (depth >= CWIST_REDIS_MAX_REPLY_DEPTH) goto fail;
263 char *end = NULL;
264 long long count = strtoll(line + 1, &end, 10);
265 if (end == line + 1 || *end != '\0' || count < 0 ||
266 count > CWIST_REDIS_MAX_REPLY_ELEMENTS)
267 goto fail;
268 node->elements = (size_t)count;
269 if (count > 0) {
270 node->element = cwist_alloc_array((size_t)count, sizeof(*node->element));
271 if (!node->element) goto fail;
272 for (long long i = 0; i < count; i++) {
273 node->element[i] = NULL;
274 cwist_error_t e = redis_read_resp(r, depth + 1, &node->element[i]);
275 if (e.error.err_i16 != 0) goto fail;
276 }
277 }
278 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
279 }
280
281 default: goto fail;
282 }
283
284fail:
285 cwist_redis_reply_free(*out);
286 *out = NULL;
287 return make_error(CWIST_ERR_INT16);
288}

293 char *out_type) {
294 cwist_redis_reply_t *tree = NULL;
295 cwist_error_t err = redis_read_resp(r, 0, &tree);
296 if (out_type) *out_type = tree ? (char)tree->type : '\0';

334static cwist_error_t redis_command_frame_tree(cwist_redis_t *r, const char *frame, size_t frame_len,
335 cwist_redis_reply_t **out_tree) {
336 if (!r || !frame) return make_error(CWIST_ERR_INT16);
337 if (out_tree) *out_tree = NULL;
338 /* A request/reply pair must be serialized as one critical section. Keeping
339 * send and receive separate allows two callers to consume each other's
340 * replies on the same connection. */
341 pthread_mutex_lock(&r->mtx);
342 cwist_error_t err = redis_send_all(r->fd, frame, frame_len) == 0
343 ? redis_read_resp(r, 0, out_tree)
344 : make_error(CWIST_ERR_INT16);
345 pthread_mutex_unlock(&r->mtx);
346 return err;
347}

438void cwist_redis_close(cwist_redis_t *r) {
439 if (!r) return;
440 /* shutdown wakes a subscriber blocked in recv before the descriptor is
441 * released. This is also the documented way to stop subscribe(). */
442 if (r->fd >= 0) {
443 shutdown(r->fd, SHUT_RDWR);
444 close(r->fd);
445 r->fd = -1;
446 }
447 cwist_free(r->host);
448 pthread_mutex_destroy(&r->mtx);
449 cwist_free(r);
450}

711cwist_error_t cwist_redis_subscribe(cwist_redis_t *r, const char **channels, cwist_redis_msg_cb cb,
712 void *ctx) {
713 if (!r || !channels || !channels[0] || !cb) return make_error(CWIST_ERR_INT16);
714
715 cwist_sstring *cmd = cwist_sstring_create();
716 if (!cmd) return make_error(CWIST_ERR_INT16);
717 size_t count = 0;
718 while (channels[count]) count++;
719 char prefix[32];
720 snprintf(prefix, sizeof(prefix), "*%zu\r\n", count + 1);
721 cwist_sstring_append(cmd, prefix);
722 write_bulk_string(cmd, "SUBSCRIBE");
723 for (size_t i = 0; i < count; i++) {
724 write_bulk_string(cmd, channels[i]);
725 }
726 cwist_error_t err = send_command(r, cmd->data);
727 cwist_sstring_destroy(cmd);
728 if (err.error.err_i16 != 0) return err;
729
730 /* Consume subscription confirmations. */
731 for (size_t i = 0; i < count; i++) {
732 char *out = NULL;
733 err = recv_reply(r, &out);
734 cwist_free(out);
735 if (err.error.err_i16 != 0) return err;
736 }
737
738 /* Blocking read loop for messages. */
739 while (1) {
740 char line[CWIST_REDIS_LINE_MAX];
741 if (redis_recv_line(r, line, sizeof(line)) != 0) break;
742 if (line[0] != '*') continue;
743 char *arr_end = NULL;
744 long long arr_count = strtoll(line + 1, &arr_end, 10);
745 if (arr_end == line + 1 || *arr_end != '\0') continue;
746 if (arr_count < 3) continue;
747
748 char *type = NULL;
749 if (read_reply(r, &type, NULL, NULL).error.err_i16 != 0) break;
750
751 char *channel = NULL;
752 if (read_reply(r, &channel, NULL, NULL).error.err_i16 != 0) {
753 cwist_free(type);
754 break;
755 }
756
757 char *message = NULL;
758 if (read_reply(r, &message, NULL, NULL).error.err_i16 != 0) {
759 cwist_free(type);
760 cwist_free(channel);
761 break;
762 }
763
764 if (type && strcmp(type, "message") == 0) {
765 cb(channel, message, ctx);
766 }
767 cwist_free(type);
768 cwist_free(channel);
769 cwist_free(message);
770 }
771 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
772}

782cwist_redis_pool_t *cwist_redis_pool_create(const char *host, int port, size_t max_conns) {
783 if (!host || port <= 0 || max_conns == 0) return NULL;
784
785 cwist_redis_pool_t *pool = cwist_alloc(sizeof(*pool));
786 if (!pool) return NULL;
787 pool->host = cwist_strdup(host);
788 pool->port = port;
789 pool->max_conns = max_conns;
790 pool->conns = cwist_alloc_array(max_conns, sizeof(cwist_redis_t *));
791 pool->available = cwist_alloc_array(max_conns, sizeof(bool));
792 if (!pool->host || !pool->conns || !pool->available) goto fail;
793
794 for (size_t i = 0; i < max_conns; i++) {
795 pool->conns[i] = cwist_redis_connect(host, port);
796 if (!pool->conns[i]) goto fail;
797 pool->available[i] = true;
798 }
799
800 pthread_mutex_init(&pool->mtx, NULL);
801 pthread_cond_init(&pool->cond, NULL);
802 return pool;
803
804fail:
805 if (pool) {
806 for (size_t i = 0; i < max_conns && pool->conns; i++) {
807 if (pool->conns[i]) cwist_redis_close(pool->conns[i]);
808 }
809 cwist_free(pool->conns);
810 cwist_free(pool->available);
811 cwist_free(pool->host);
812 cwist_free(pool);
813 }
814 return NULL;
815}

818void cwist_redis_pool_destroy(cwist_redis_pool_t *pool) {
819 if (!pool) return;
820 pthread_mutex_lock(&pool->mtx);
821 for (size_t i = 0; i < pool->max_conns; i++) {
822 if (pool->conns[i]) cwist_redis_close(pool->conns[i]);
823 }
824 pthread_mutex_unlock(&pool->mtx);
825 pthread_mutex_destroy(&pool->mtx);
826 pthread_cond_destroy(&pool->cond);
827 cwist_free(pool->conns);
828 cwist_free(pool->available);
829 cwist_free(pool->host);
830 cwist_free(pool);
831}

836static cwist_redis_t *redis_pool_acquire(cwist_redis_pool_t *pool) {
837 if (!pool) return NULL;
838 pthread_mutex_lock(&pool->mtx);
839 while (1) {
840 for (size_t i = 0; i < pool->max_conns; i++) {
841 if (pool->available[i]) {
842 pool->available[i] = false;
843 pthread_mutex_unlock(&pool->mtx);
844 return pool->conns[i];
845 }
846 }
847 pthread_cond_wait(&pool->cond, &pool->mtx);
848 }
849}

852static void redis_pool_release(cwist_redis_pool_t *pool, cwist_redis_t *r) {
853 if (!pool || !r) return;
854 pthread_mutex_lock(&pool->mtx);
855 for (size_t i = 0; i < pool->max_conns; i++) {
856 if (pool->conns[i] == r) {
857 pool->available[i] = true;
858 pthread_cond_signal(&pool->cond);
859 break;
860 }
861 }
862 pthread_mutex_unlock(&pool->mtx);
863}
