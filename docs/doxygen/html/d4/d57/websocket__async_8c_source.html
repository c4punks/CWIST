188static void ws_async_sweep_unlink(cwist_websocket_async *ws) {
189 ws_async_sweep_t *sweep = ws->sweep;
190 if (!sweep) return;
191 pthread_mutex_lock(&sweep->lock);
192 cwist_websocket_async **pp = &sweep->head;
193 while (*pp && *pp != ws) pp = &(*pp)->sweep_next;
194 if (*pp) *pp = ws->sweep_next;
195 bool empty = (sweep->head == NULL);
196 pthread_mutex_unlock(&sweep->lock);
197 ws->sweep = NULL;
198 if (empty) ws_async_sweep_mark_dead(sweep);
199}

220static void ws_async_terminate(cwist_websocket_async *ws) {
221 if (!ws) return;
222 ws_async_sweep_unlink(ws);
223 bool parked = ws->park_head != NULL;
224 if (ws->read_armed || parked) {
225 ws->closing = true;
226 /* With a read slot armed, the read callback performs the teardown;
227 * cancel the parked-write slot so it cannot dispatch as well. With
228 * only the write slot pending, shutdown() wakes it and its callback
229 * performs the teardown. */
230 if (ws->read_armed && parked) cwist_reactor_del(ws->reactor, ws->fd);
231 /* FIN now; the peer sees EOF immediately even though the final close
232 * waits for the woken slot. */
233 shutdown(ws->fd, SHUT_RDWR);
234 return;
235 }
236 ws_async_teardown(ws);
237}

320static bool ws_async_idle_reap(cwist_websocket_async *ws) {
321 uint32_t timeout_sec = ws_async_idle_timeout_sec();
322 if (timeout_sec == 0) return false;
323 uint32_t now = cwist_fast_monotonic_sec();

332/* Periodic idle sweep (one watchdog thread per reactor) */
333/* ------------------------------------------------------------------ */
334
335/* Lazy per-event reaping (the HTTP keep-alive precedent) cannot reap a
336 * totally silent connection: the one-shot slot model gives each fd at most
337 * one read + one POLLOUT slot, so an idle fd fires no event. Instead each
338 * reactor that owns WS async connections gets ONE watchdog thread (plain
339 * pthreads + nanosleep — portable across io_uring, epoll, and kqueue, with
340 * no Linux-only timerfd). The thread sleeps the sweep interval, then hands
341 * the sweep to the reactor thread via cwist_reactor_post(); the posted
342 * callback runs the same conn-list walk + terminate as the per-event reaper,
343 * so all list mutation stays on the reactor thread. The per-event reaper
344 * still runs at every read/write event, so the sweep cadence only affects
345 * how late a silent connection is reaped, not whether.
346 *
347 * Lifetime: the context is marked dead when its connection list empties
348 * (terminate) or the walk finds it empty. The watchdog notices the dead
349 * flag within one 100 ms sleep slice, stops posting, and exits; the context
350 * is freed by the watchdog thread alone after it exits its posting loop and
351 * observes posts == 0: the callback only ever decrements the posts counter
352 * (as its last touch of the context), so the watchdog can never reference
353 * freed memory. The posts counter covers the window where a post is in
354 * flight while the context is being torn down; cwist_reactor_destroy()
355 * drains pending posts, so the wait always terminates. */
356
366static uint32_t ws_async_sweep_interval(uint32_t timeout_sec) {
367 uint32_t interval = timeout_sec / 2;
368 if (interval < 1) interval = 1;
369 if (interval > 60) interval = 60;
370 return interval;
371}

385static void ws_async_sweep_watchdog_free(ws_async_sweep_t *sweep) {
386 const struct timespec slice = {.tv_sec = 0, .tv_nsec = 100000000L};
387 for (;;) {
388 pthread_mutex_lock(&sweep->lock);
389 bool drained = (sweep->posts == 0);
390 pthread_mutex_unlock(&sweep->lock);
391 if (drained) break;
392 nanosleep(&slice, NULL);
393 }
394
395 pthread_mutex_lock(&g_ws_sweeps_lock);
396 ws_async_sweep_t **pp = &g_ws_sweeps;
397 while (*pp && *pp != sweep) pp = &(*pp)->next;
398 if (*pp) *pp = sweep->next;
399 pthread_mutex_unlock(&g_ws_sweeps_lock);
400 pthread_mutex_destroy(&sweep->lock);
401 cwist_free(sweep);
402}

415static void ws_async_sweep_post_cb(void *ctx) {
416 ws_async_sweep_t *sweep = (ws_async_sweep_t *)ctx;
417
418 uint32_t timeout_sec = ws_async_idle_timeout_sec();
419 if (timeout_sec > 0) {
420 uint32_t now = cwist_fast_monotonic_sec();
421 for (;;) {
422 pthread_mutex_lock(&sweep->lock);
423 cwist_websocket_async *victim = NULL;
424 for (cwist_websocket_async *ws = sweep->head; ws; ws = ws->sweep_next) {
425 if (ws->last_active_sec > 0 && (now - ws->last_active_sec) > timeout_sec) {
426 victim = ws;
427 break;
428 }
429 }
430 pthread_mutex_unlock(&sweep->lock);
431 if (!victim) break;
432 ws_async_terminate(victim);
433 }
434 }
435
436 /* posts-- is the last touch of the context: the watchdog frees it only
437 * after observing posts == 0. */
438 pthread_mutex_lock(&sweep->lock);
439 bool empty = (sweep->head == NULL);
440 if (sweep->posts > 0) sweep->posts--;
441 pthread_mutex_unlock(&sweep->lock);
442 if (empty) ws_async_sweep_mark_dead(sweep);
443}

445/* Sleep the interval in 100 ms slices so dead-flag shutdown latency stays
446 * low without a condvar; nanosleep is EINTR-tolerant. */
447#define WS_ASYNC_SWEEP_SLICE_MS 100u
448
460static void *ws_async_sweep_watchdog(void *arg) {
461 ws_async_sweep_t *sweep = (ws_async_sweep_t *)arg;
462 const struct timespec slice = {.tv_sec = 0, .tv_nsec = 100000000L};
463 uint32_t slices = sweep->interval_sec * (1000u / WS_ASYNC_SWEEP_SLICE_MS);
464
465 for (;;) {
466 for (uint32_t i = 0; i < slices; i++) {
467 nanosleep(&slice, NULL);
468 pthread_mutex_lock(&sweep->lock);
469 bool dead = sweep->dead;
470 pthread_mutex_unlock(&sweep->lock);
471 if (dead) goto out;
472 }
473 pthread_mutex_lock(&sweep->lock);
474 bool dead = sweep->dead;
475 if (!dead) sweep->posts++;
476 pthread_mutex_unlock(&sweep->lock);
477 if (dead) goto out;
478 if (!cwist_reactor_post(sweep->reactor, &sweep->node)) {
479 pthread_mutex_lock(&sweep->lock);
480 sweep->posts--;
481 pthread_mutex_unlock(&sweep->lock);
482 }
483 }
484
485out:
486 /* Sole freer of the context: the watchdog is done with it, and any
487 * posted sweep still queued must drain before the memory is released. */
488 ws_async_sweep_watchdog_free(sweep);
489 return NULL;
490}

504static void ws_async_sweep_link(cwist_websocket_async *ws, cwist_reactor_t *reactor) {
505 uint32_t timeout_sec = ws_async_idle_timeout_sec();
506 if (timeout_sec == 0) return;
507
508 pthread_mutex_lock(&g_ws_sweeps_lock);
509 for (ws_async_sweep_t *s = g_ws_sweeps; s; s = s->next) {
510 /* A dead context awaiting teardown is unowned: a fresh one is
511 * created so the new connection is swept. dead is only written
512 * under g_ws_sweeps_lock (mark_dead), so this read is safe. */
513 if (s->reactor == reactor && !s->dead) {
514 pthread_mutex_lock(&s->lock);
515 ws->sweep = s;
516 ws->sweep_next = s->head;
517 s->head = ws;
518 pthread_mutex_unlock(&s->lock);
519 pthread_mutex_unlock(&g_ws_sweeps_lock);
520 return;
521 }
522 }
523 pthread_mutex_unlock(&g_ws_sweeps_lock);
524
525 ws_async_sweep_t *sweep = (ws_async_sweep_t *)cwist_alloc(sizeof(*sweep));
526 if (!sweep) return;
527 sweep->reactor = reactor;
528 sweep->interval_sec = ws_async_sweep_interval(timeout_sec);
529 sweep->head = ws;
530 sweep->dead = false;
531 sweep->posts = 0;
532 sweep->node.cb = ws_async_sweep_post_cb;
533 sweep->node.ctx = sweep;
534 sweep->next = NULL;
535 pthread_mutex_init(&sweep->lock, NULL);
536
537 ws->sweep = sweep;
538 ws->sweep_next = NULL;
539
540 pthread_t thread;
541 if (pthread_create(&thread, NULL, ws_async_sweep_watchdog, sweep) != 0) {
542 ws->sweep = NULL;
543 pthread_mutex_destroy(&sweep->lock);
544 cwist_free(sweep);
545 return;
546 }
547 pthread_detach(thread);
548
549 pthread_mutex_lock(&g_ws_sweeps_lock);
550 sweep->next = g_ws_sweeps;
551 g_ws_sweeps = sweep;
552 pthread_mutex_unlock(&g_ws_sweeps_lock);
553}

570static uint8_t *ws_async_serialize_frame(cwist_ws_opcode_t opcode, const uint8_t *data, size_t len,
571 size_t *out_total) {
572 uint8_t head[10];
573 size_t head_len = 2;
574
575 head[0] = 0x80 | (opcode & 0x0F);
576 if (len < 126) {
577 head[1] = (uint8_t)len;
578 } else if (len < 65536) {
579 head[1] = 126;
580 head[2] = (uint8_t)((len >> 8) & 0xFF);
581 head[3] = (uint8_t)(len & 0xFF);
582 head_len += 2;
583 } else {
584 head[1] = 127;
585 for (int i = 0; i < 8; i++) head[2 + i] = (uint8_t)(len >> ((7 - i) * 8));
586 head_len += 8;
587 }
588
589 uint8_t *buf = (uint8_t *)cwist_alloc(head_len + len);
590 if (!buf) return NULL;
591 memcpy(buf, head, head_len);
592 if (len > 0) memcpy(buf + head_len, data, len);
593 *out_total = head_len + len;
594 return buf;
595}

597static int ws_async_park_drain(cwist_websocket_async *ws, bool *made_progress) {
598 int flags = MSG_NOSIGNAL | MSG_DONTWAIT;
599 *made_progress = false;
600 while (ws->park_head) {
601 ws_async_park_node *node = ws->park_head;
602 ssize_t n = send(ws->fd, node->buf + node->off, node->len - node->off, flags);
603 if (n > 0) {
604 node->off += (size_t)n;
605 ws->parked_bytes -= (size_t)n;
606 *made_progress = true;
607 if (node->off == node->len) {
608 ws->park_head = node->next;
609 if (!ws->park_head) ws->park_tail = NULL;
610 cwist_free(node->buf);
611 cwist_free(node);
612 }
613 continue;
614 }
615 if (n < 0 && errno == EINTR) continue;
616 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) return 1;
617 return -1;
618 }
619 return 0;
620}

629static int ws_async_queue_frame(cwist_websocket_async *ws, cwist_ws_opcode_t opcode,
630 const uint8_t *data, size_t len) {
631 size_t frame_len = 0;
632 uint8_t *frame = ws_async_serialize_frame(opcode, data, len, &frame_len);
633 if (!frame) return -1;
634
635 size_t limit = ws_async_park_limit();
636 if (frame_len > limit || ws->parked_bytes > limit - frame_len) {
637 cwist_free(frame);
638 return 1;
639 }
640 ws_async_park_node *node = (ws_async_park_node *)cwist_alloc(sizeof(*node));
641 if (!node) {
642 cwist_free(frame);
643 return -1;
644 }
645 node->buf = frame;
646 node->off = 0;
647 node->len = frame_len;
648 node->next = NULL;
649 if (ws->park_tail)
650 ws->park_tail->next = node;
651 else
652 ws->park_head = node;
653 ws->park_tail = node;
654 ws->parked_bytes += frame_len;
655
656 bool made_progress = false;
657 int drain_rc = ws_async_park_drain(ws, &made_progress);
658 if (drain_rc < 0) return -1;
659 if (drain_rc == 0) {
660 ws_async_touch(ws);
661 return 0;
662 }
663
664 ws->park_deadline_sec = ws_async_park_deadline();
665 if (ws->write_armed) return 0;
666 ws_async_slot_t slot = {.ws = ws};
667 if (!cwist_reactor_add_out(ws->reactor, ws->fd, ws_async_write_ready, &slot, sizeof(slot)))
668 return -1;
669 ws->write_armed = true;
670 return 0;
671}

679static void ws_async_write_ready(int fd, void *ctx) {
680 ws_async_slot_t *slot = (ws_async_slot_t *)ctx;
681 cwist_websocket_async *ws = slot->ws;
682 (void)fd;
683 ws->write_armed = false;
684
685 if (ws->closing) {
686 ws_async_teardown(ws);
687 return;
688 }
689 if (ws_async_idle_reap(ws)) return;
690
691 bool made_progress = false;
692 int drain_rc = ws_async_park_drain(ws, &made_progress);
693 if (made_progress) {
694 ws->park_deadline_sec = ws_async_park_deadline();
695 ws_async_touch(ws);
696 }
697 if (drain_rc > 0 && cwist_fast_monotonic_sec() <= ws->park_deadline_sec &&
698 cwist_reactor_add_out(ws->reactor, fd, ws_async_write_ready, slot, sizeof(*slot))) {
699 ws->write_armed = true;
700 return;
701 }
702 if (drain_rc != 0) {
703 ws_async_terminate(ws);
704 return;
705 }
706
707 if (ws->terminate_pending) {
708 ws_async_terminate(ws);
709 return;
710 }
711 if (!ws->read_armed) {
712 ws_async_slot_t rslot = {.ws = ws};
713 if (cwist_reactor_add(ws->reactor, fd, ws_async_read_ready, &rslot, sizeof(rslot)))
714 ws->read_armed = true;
715 else
716 ws_async_terminate(ws);
717 }
718}

750void cwist_websocket_async_close(cwist_websocket_async *ws) {
751 if (!ws || ws->closed) return;
752 ws->closed = true;
753 /* Best-effort CLOSE frame; failure still tears the connection down. */
754 if (ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, NULL, 0) != 0) {
755 ws->terminate_pending = true;
756 return;
757 }
758 if (ws->parked_bytes == 0) {
759 /* Nothing parked: the CLOSE frame is on the wire; tear down now. The
760 * enclosing read turn (we run on the reactor thread) performs the
761 * actual free via terminate_pending. */
762 ws->terminate_pending = true;
763 }
764 /* Otherwise the parked-write completion terminates after draining. */
765}

779void cwist_websocket_async_close_code(cwist_websocket_async *ws, uint16_t code) {
780 if (!ws || ws->closed) return;
781 ws->closed = true;
782 uint8_t body[2] = {(uint8_t)(code >> 8), (uint8_t)(code & 0xFF)};
783 if (ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, body, sizeof(body)) != 0) {
784 ws->terminate_pending = true;
785 return;
786 }
787 if (ws->parked_bytes == 0) {
788 /* CLOSE frame drained: the enclosing reactor turn performs the free
789 * via terminate_pending, mirroring cwist_websocket_async_close(). */
790 ws->terminate_pending = true;
791 }
792}

807static bool ws_async_stash_append(cwist_websocket_async *ws, const uint8_t *src, size_t len) {
808 if (len == 0) return true;
809 size_t need = ws->stash_len + len;
810 if (need > ws->stash_cap) {
811 size_t new_cap = ws->stash_cap ? ws->stash_cap * 2 : 4096;
812 if (new_cap < need) new_cap = need;
813 uint8_t *nb = (uint8_t *)cwist_alloc(new_cap);
814 if (!nb) return false;
815 if (ws->stash_len) memcpy(nb, ws->stash, ws->stash_len);
816 cwist_free(ws->stash);
817 ws->stash = nb;
818 ws->stash_cap = new_cap;
819 }
820 memcpy(ws->stash + ws->stash_len, src, len);
821 ws->stash_len += len;
822 return true;
823}

836static bool ws_async_frag_append(cwist_websocket_async *ws, const uint8_t *src, size_t len) {
837 if (len > WS_ASYNC_MAX_MESSAGE_BYTES - ws->frag_len) return false;
838 size_t need = ws->frag_len + len + 1; /* +1 for null terminator */
839 if (need > ws->frag_cap) {
840 size_t new_cap = ws->frag_cap ? ws->frag_cap * 2 : 4096;
841 if (new_cap < need) new_cap = need;
842 uint8_t *nb = (uint8_t *)cwist_alloc(new_cap);
843 if (!nb) return false;
844 if (ws->frag_len) memcpy(nb, ws->frag_buf, ws->frag_len);
845 cwist_free(ws->frag_buf);
846 ws->frag_buf = nb;
847 ws->frag_cap = new_cap;
848 }
849 if (len > 0) memcpy(ws->frag_buf + ws->frag_len, src, len);
850 ws->frag_len += len;
851 return true;
852}

872static int ws_async_process_frame(cwist_websocket_async *ws, bool fin, cwist_ws_opcode_t opcode,
873 uint8_t *payload, size_t payload_len) {
874 bool is_control = (opcode >= 0x8);
875
876 /* --- Fragmented-message reassembly (RFC 6455 section 5.4) -------------- */
877 if (!is_control) {
878 if (opcode == CWIST_WS_FRAME_CONTINUATION) {
879 if (!ws->frag_active) {
880 cwist_free(payload);
881 return -1;
882 }
883 } else {
884 if (ws->frag_active) {
885 cwist_free(payload);
886 return -1;
887 }
888 if (!fin) {
889 ws->frag_active = true;
890 ws->frag_opcode = opcode;
891 }
892 }
893
894 if (ws->frag_active) {
895 if (!ws_async_frag_append(ws, payload, payload_len)) {
896 cwist_free(payload);
897 cwist_free(ws->frag_buf);
898 ws->frag_buf = NULL;
899 ws->frag_len = ws->frag_cap = 0;
900 ws->frag_active = false;
901 return -1;
902 }
903 cwist_free(payload);
904 if (!fin) return 0;
905
906 payload = ws->frag_buf;
907 payload_len = ws->frag_len;
908 payload[payload_len] = '\0';
909 opcode = ws->frag_opcode;
910
911 ws->frag_buf = NULL;
912 ws->frag_len = ws->frag_cap = 0;
913 ws->frag_active = false;
914 }
915 }
916
917 /* --- Deliver the complete frame ---------------------------------- */
918 if (opcode == CWIST_WS_FRAME_CLOSE) {
919 /* RFC 6455 section 5.5.1: a CLOSE body, if present, MUST be >= 2 bytes. */
920 if (payload_len == 1) {
921 cwist_free(payload);
922 return -1;
923 }
924 /* Echo the status code (first 2 bytes) back before closing. */
925 ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, payload, (payload_len >= 2) ? 2 : 0);
926 cwist_free(payload);
927 ws->closed = true;
928 ws->terminate_pending = true;
929 return 0;
930 }
931 if (opcode == CWIST_WS_FRAME_PING) {
932 /* RFC 6455 section 5.5.3: respond to every PING with a PONG carrying
933 * the same payload (up to 125 bytes per section 5.5). */
934 ws_async_queue_frame(ws, CWIST_WS_FRAME_PONG, payload, payload_len);
935 }
936
937 cwist_ws_frame *frame = (cwist_ws_frame *)cwist_alloc(sizeof(cwist_ws_frame));
938 if (!frame) {
939 cwist_free(payload);
940 return -1;
941 }
942 frame->fin = fin;
943 frame->opcode = opcode;
944 frame->payload = payload;
945 frame->payload_len = payload_len;
946 /* Ownership moves to the callback; it must destroy the frame. */
947 ws->on_message(ws, frame, ws->user_data);
948 return 0;
949}

962static int ws_async_feed(cwist_websocket_async *ws) {
963 size_t off = 0;
964
965 while (ws->stash_len - off >= 2) {
966 const uint8_t *base = ws->stash + off;
967 bool fin = (base[0] & 0x80) != 0;
968 cwist_ws_opcode_t opcode = base[0] & 0x0F;
969 if (opcode != CWIST_WS_FRAME_CONTINUATION && opcode != CWIST_WS_FRAME_TEXT &&
970 opcode != CWIST_WS_FRAME_BINARY && opcode != CWIST_WS_FRAME_CLOSE &&
971 opcode != CWIST_WS_FRAME_PING && opcode != CWIST_WS_FRAME_PONG)
972 return -1;
973
974 /* RFC 6455 section 5.2: RSV1/RSV2/RSV3 MUST be 0 unless an extension
975 * defines their meaning. CWIST has no such extension. */
976 if (base[0] & 0x70) return -1;
977
978 /* Client-to-server frames must be masked per RFC 6455 section 5.3. */
979 if (!(base[1] & 0x80)) return -1;
980
981 uint64_t payload_len = base[1] & 0x7F;
982 size_t hdr = 2;
983 if (payload_len == 126) {
984 if (ws->stash_len - off < 4) break;
985 payload_len = ((uint64_t)base[2] << 8) | (uint64_t)base[3];
986 hdr = 4;
987 } else if (payload_len == 127) {
988 if (ws->stash_len - off < 10) break;
989 const uint8_t *e = base + 2;
990 payload_len = ((uint64_t)e[0] << 56) | ((uint64_t)e[1] << 48) | ((uint64_t)e[2] << 40) |
991 ((uint64_t)e[3] << 32) | ((uint64_t)e[4] << 24) | ((uint64_t)e[5] << 16) |
992 ((uint64_t)e[6] << 8) | (uint64_t)e[7];
993 hdr = 10;
994 }
995
996 /* Reject frames that would force a huge allocation before any
997 * payload is read (OOM DoS vector). */
998 if (payload_len > WS_ASYNC_MAX_PAYLOAD_BYTES) return -1;
999
1000 /* RFC 6455 section 5.5: control frames MUST NOT carry more than 125
1001 * bytes and MUST NOT be fragmented. */
1002 bool is_control_early = (opcode >= 0x8);
1003 if (is_control_early && (payload_len > 125 || !fin)) return -1;
1004
1005 size_t total = hdr + 4 + (size_t)payload_len;
1006 if (ws->stash_len - off < total) break; /* Partial frame: wait for more. */
1007
1008 const uint8_t *mask = base + hdr;
1009 const uint8_t *src = base + hdr + 4;
1010 uint8_t *payload = NULL;
1011 if (payload_len > 0) {
1012 payload = (uint8_t *)cwist_alloc((size_t)payload_len + 1);
1013 if (!payload) return -1;
1014 for (uint64_t i = 0; i < payload_len; i++) payload[i] = src[i] ^ mask[i % 4];
1015 payload[payload_len] = '\0';
1016 }
1017
1018 off += total;
1019 if (ws_async_process_frame(ws, fin, opcode, payload, (size_t)payload_len) != 0) return -1;
1020 /* The CLOSE handler marks the connection for teardown; keep parsing
1021 * nothing further — the caller terminates. */
1022 if (ws->terminate_pending) break;
1023 }
1024
1025 if (off > 0) {
1026 memmove(ws->stash, ws->stash + off, ws->stash_len - off);
1027 ws->stash_len -= off;
1028 }
1029 return 0;
1030}

1049static void ws_async_read_ready(int fd, void *ctx) {
1050 ws_async_slot_t *slot = (ws_async_slot_t *)ctx;
1051 cwist_websocket_async *ws = slot->ws;
1052 uint8_t buf[WS_ASYNC_READ_CHUNK];
1053 bool eof = false, err = false, got_bytes = false;
1054
1055 ws->read_armed = false;
1056
1057 /* Deferred termination (sweep reap / async_close with slots pending):
1058 * shutdown() already signalled the peer; this woken slot is the only one
1059 * left, so it performs the real teardown. */
1060 if (ws->closing) {
1061 ws_async_teardown(ws);
1062 return;
1063 }
1064
1065 /* Lazy idle reaper (mirrors http_async_event_cb): checked before any
1066 * work so an event arriving after the deadline closes the connection
1067 * without touching the stash. */
1068 if (ws_async_idle_reap(ws)) return;
1069
1070 /* Drain everything the kernel has without blocking. */
1071 for (;;) {
1072 ssize_t n = recv(fd, buf, sizeof(buf), MSG_DONTWAIT);
1073 if (n > 0) {
1074 got_bytes = true;
1075 if (!ws_async_stash_append(ws, (const uint8_t *)buf, (size_t)n)) {
1076 err = true;
1077 break;
1078 }
1079 continue;
1080 }
1081 if (n == 0) {
1082 eof = true;
1083 break;
1084 }
1085 if (errno == EINTR) continue;
1086 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
1087 err = true;
1088 break;
1089 }
1090
1091 if (!err && ws->stash_len > 0 && ws_async_feed(ws) != 0) err = true;
1092
1093 /* Bytes on the wire (and therefore every complete frame, PING/PONG
1094 * included) count as activity for the idle reaper. */
1095 if (got_bytes && !err && !eof) ws_async_touch(ws);
1096
1097 /* CLOSE-from-peer, async_close() from the callback, EOF and errors all
1098 * converge here: tear the connection down exactly once. */
1099 if (err || eof || ws->terminate_pending) {
1100 ws_async_terminate(ws);
1101 return;
1102 }
1103
1104 /* One-shot slot consumed: re-arm for the next read event. */
1105 if (!cwist_reactor_add(ws->reactor, fd, ws_async_read_ready, slot, sizeof(*slot))) {
1106 ws_async_terminate(ws);
1107 return;
1108 }
1109 ws->read_armed = true;
1110}

1133cwist_websocket_async *cwist_websocket_async_attach_ex(int fd, cwist_reactor_t *reactor,
1134 cwist_ws_on_message_t on_message,
1135 cwist_ws_on_close_t on_close,
1136 void *user_data, const uint8_t *initial,
1137 size_t initial_len) {
1138 if (fd < 0 || !reactor || !on_message) return NULL;
1139
1140 cwist_websocket_async *ws = (cwist_websocket_async *)cwist_alloc(sizeof(*ws));
1141 if (!ws) {
1142 close(fd);
1143 return NULL;
1144 }
1145 ws->fd = fd;
1146 ws->reactor = reactor;
1147 ws->on_message = on_message;
1148 ws->on_close = on_close;
1149 ws->user_data = user_data;
1150 ws->stash = NULL;
1151 ws->stash_len = ws->stash_cap = 0;
1152 ws->frag_buf = NULL;
1153 ws->frag_len = ws->frag_cap = 0;
1154 ws->frag_opcode = CWIST_WS_FRAME_CONTINUATION;
1155 ws->frag_active = false;
1156 ws->park_head = ws->park_tail = NULL;
1157 ws->parked_bytes = 0;
1158 ws->park_deadline_sec = 0;
1159 ws->last_active_sec = cwist_fast_monotonic_sec();
1160 ws->sweep = NULL;
1161 ws->sweep_next = NULL;
1162 ws->closed = false;
1163 ws->closing = false;
1164 ws->terminate_pending = false;
1165 ws->read_armed = false;
1166 ws->write_armed = false;
1167
1168 /* The fd must never block the reactor thread. */
1169 int fl = fcntl(fd, F_GETFL, 0);
1170 if (fl >= 0) fcntl(fd, F_SETFL, fl | O_NONBLOCK);
1171
1172 /* Bytes already read past the upgrade request (conn->rbuf) go through the
1173 * parser before the first reactor wait. Delivery can happen inline here,
1174 * including a cwist_websocket_async_close() from the callback. */
1175 bool fail = false;
1176 if (initial_len > 0) {
1177 if (!ws_async_stash_append(ws, initial, initial_len)) fail = true;
1178 if (!fail && ws_async_feed(ws) != 0) fail = true;
1179 }
1180 if (fail || ws->terminate_pending) {
1181 /* No reactor slot was registered yet: nothing to cancel. on_close
1182 * does not fire here; attach failed, so the caller keeps ownership of
1183 * any state installed during inline initial-bytes delivery. */
1184 close(fd);
1185 cwist_free(ws->stash);
1186 cwist_free(ws->frag_buf);
1187 ws_async_park_clear(ws);
1188 cwist_free(ws);
1189 return NULL;
1190 }
1191
1192 /* Everything the reactor thread reads is set before the slot is added:
1193 * once cwist_reactor_add() returns, the read callback may already run
1194 * on the reactor thread (attach can be called from any thread), so this
1195 * thread must not write ws any more (#351).
1196 *
1197 * Join the reactor's idle sweep (no-op when reaping is disabled). */
1198 ws_async_sweep_link(ws, reactor);
1199 ws->read_armed = true;
1200
1201 ws_async_slot_t slot = {.ws = ws};
1202 if (!cwist_reactor_add(reactor, fd, ws_async_read_ready, &slot, sizeof(slot))) {
1203 ws->read_armed = false;
1204 ws_async_sweep_unlink(ws);
1205 close(fd);
1206 cwist_free(ws->stash);
1207 cwist_free(ws->frag_buf);
1208 ws_async_park_clear(ws);
1209 cwist_free(ws);
1210 return NULL;
1211 }
1212 return ws;
1213}
