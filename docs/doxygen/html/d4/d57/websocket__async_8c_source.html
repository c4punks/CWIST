189static void ws_async_sweep_unlink(cwist_websocket_async *ws) {
190 ws_async_sweep_t *sweep = ws->sweep;
191 if (!sweep) return;
192 pthread_mutex_lock(&sweep->lock);
193 cwist_websocket_async **pp = &sweep->head;
194 while (*pp && *pp != ws) pp = &(*pp)->sweep_next;
195 if (*pp) *pp = ws->sweep_next;
196 bool empty = (sweep->head == NULL);
197 pthread_mutex_unlock(&sweep->lock);
198 ws->sweep = NULL;
199 if (empty) ws_async_sweep_mark_dead(sweep);
200}

221static void ws_async_terminate(cwist_websocket_async *ws) {
222 if (!ws) return;
223 ws_async_sweep_unlink(ws);
224 bool parked = ws->park_head != NULL;
225 if (ws->read_armed || parked) {
226 ws->closing = true;
227 /* With a read slot armed, the read callback performs the teardown;
228 * cancel the parked-write slot so it cannot dispatch as well. With
229 * only the write slot pending, shutdown() wakes it and its callback
230 * performs the teardown. */
231 if (ws->read_armed && parked) cwist_reactor_del(ws->reactor, ws->fd);
232 /* FIN now; the peer sees EOF immediately even though the final close
233 * waits for the woken slot. */
234 shutdown(ws->fd, SHUT_RDWR);
235 return;
236 }
237 ws_async_teardown(ws);
238}

322 uint32_t timeout_sec = ws_async_idle_timeout_sec();
323 if (timeout_sec == 0) return false;
324 uint32_t now = cwist_fast_monotonic_sec();

336/* Lazy per-event reaping (the HTTP keep-alive precedent) cannot reap a
337 * totally silent connection: the one-shot slot model gives each fd at most
338 * one read + one POLLOUT slot, so an idle fd fires no event. Instead each
339 * reactor that owns WS async connections gets ONE watchdog thread (plain
340 * pthreads + nanosleep — portable across io_uring, epoll, and kqueue, with
341 * no Linux-only timerfd). The thread sleeps the sweep interval, then hands
342 * the sweep to the reactor thread via cwist_reactor_post(); the posted
343 * callback runs the same conn-list walk + terminate as the per-event reaper,
344 * so all list mutation stays on the reactor thread. The per-event reaper
345 * still runs at every read/write event, so the sweep cadence only affects
346 * how late a silent connection is reaped, not whether.
347 *
348 * Lifetime: the context is marked dead when its connection list empties
349 * (terminate) or the walk finds it empty. The watchdog notices the dead
350 * flag within one 100 ms sleep slice, stops posting, and exits; the context
351 * is freed by the watchdog thread alone after it exits its posting loop and
352 * observes posts == 0: the callback only ever decrements the posts counter
353 * (as its last touch of the context), so the watchdog can never reference
354 * freed memory. The posts counter covers the window where a post is in
355 * flight while the context is being torn down; cwist_reactor_destroy()
356 * drains pending posts, so the wait always terminates. */
357
367static uint32_t ws_async_sweep_interval(uint32_t timeout_sec) {
368 uint32_t interval = timeout_sec / 2;
369 if (interval < 1) interval = 1;
370 if (interval > 60) interval = 60;
371 return interval;
372}

386static void ws_async_sweep_watchdog_free(ws_async_sweep_t *sweep) {
387 const struct timespec slice = {.tv_sec = 0, .tv_nsec = 100000000L};
388 for (;;) {
389 pthread_mutex_lock(&sweep->lock);
390 bool drained = (sweep->posts == 0);
391 pthread_mutex_unlock(&sweep->lock);
392 if (drained) break;
393 nanosleep(&slice, NULL);
394 }
395
396 pthread_mutex_lock(&g_ws_sweeps_lock);
397 ws_async_sweep_t **pp = &g_ws_sweeps;
398 while (*pp && *pp != sweep) pp = &(*pp)->next;
399 if (*pp) *pp = sweep->next;
400 pthread_mutex_unlock(&g_ws_sweeps_lock);
401 pthread_mutex_destroy(&sweep->lock);
402 cwist_free(sweep);
403}

416static void ws_async_sweep_post_cb(void *ctx) {
417 ws_async_sweep_t *sweep = (ws_async_sweep_t *)ctx;
418
419 uint32_t timeout_sec = ws_async_idle_timeout_sec();
420 if (timeout_sec > 0) {
421 uint32_t now = cwist_fast_monotonic_sec();
422 for (;;) {
423 pthread_mutex_lock(&sweep->lock);
424 cwist_websocket_async *victim = NULL;
425 for (cwist_websocket_async *ws = sweep->head; ws; ws = ws->sweep_next) {
426 if (ws->last_active_sec > 0 && (now - ws->last_active_sec) > timeout_sec) {
427 victim = ws;
428 break;
429 }
430 }
431 pthread_mutex_unlock(&sweep->lock);
432 if (!victim) break;
433 ws_async_terminate(victim);
434 }
435 }
436
437 /* posts-- is the last touch of the context: the watchdog frees it only
438 * after observing posts == 0. */
439 pthread_mutex_lock(&sweep->lock);
440 bool empty = (sweep->head == NULL);
441 if (sweep->posts > 0) sweep->posts--;
442 pthread_mutex_unlock(&sweep->lock);
443 if (empty) ws_async_sweep_mark_dead(sweep);
444}

446/* Sleep the interval in 100 ms slices so dead-flag shutdown latency stays
447 * low without a condvar; nanosleep is EINTR-tolerant. */
448#define WS_ASYNC_SWEEP_SLICE_MS 100u
449
461static void *ws_async_sweep_watchdog(void *arg) {
462 ws_async_sweep_t *sweep = (ws_async_sweep_t *)arg;
463 const struct timespec slice = {.tv_sec = 0, .tv_nsec = 100000000L};
464 uint32_t slices = sweep->interval_sec * (1000u / WS_ASYNC_SWEEP_SLICE_MS);
465
466 for (;;) {
467 for (uint32_t i = 0; i < slices; i++) {
468 nanosleep(&slice, NULL);
469 pthread_mutex_lock(&sweep->lock);
470 bool dead = sweep->dead;
471 pthread_mutex_unlock(&sweep->lock);
472 if (dead) goto out;
473 }
474 pthread_mutex_lock(&sweep->lock);
475 bool dead = sweep->dead;
476 if (!dead) sweep->posts++;
477 pthread_mutex_unlock(&sweep->lock);
478 if (dead) goto out;
479 if (!cwist_reactor_post(sweep->reactor, &sweep->node)) {
480 pthread_mutex_lock(&sweep->lock);
481 sweep->posts--;
482 pthread_mutex_unlock(&sweep->lock);
483 }
484 }
485
486out:
487 /* Sole freer of the context: the watchdog is done with it, and any
488 * posted sweep still queued must drain before the memory is released. */
489 ws_async_sweep_watchdog_free(sweep);
490 return NULL;
491}

505static void ws_async_sweep_link(cwist_websocket_async *ws, cwist_reactor_t *reactor) {
506 uint32_t timeout_sec = ws_async_idle_timeout_sec();
507 if (timeout_sec == 0) return;
508
509 pthread_mutex_lock(&g_ws_sweeps_lock);
510 for (ws_async_sweep_t *s = g_ws_sweeps; s; s = s->next) {
511 /* A dead context awaiting teardown is unowned: a fresh one is
512 * created so the new connection is swept. dead is only written
513 * under g_ws_sweeps_lock (mark_dead), so this read is safe. */
514 if (s->reactor == reactor && !s->dead) {
515 pthread_mutex_lock(&s->lock);
516 ws->sweep = s;
517 ws->sweep_next = s->head;
518 s->head = ws;
519 pthread_mutex_unlock(&s->lock);
520 pthread_mutex_unlock(&g_ws_sweeps_lock);
521 return;
522 }
523 }
524 pthread_mutex_unlock(&g_ws_sweeps_lock);
525
526 ws_async_sweep_t *sweep = (ws_async_sweep_t *)cwist_alloc(sizeof(*sweep));
527 if (!sweep) return;
528 sweep->reactor = reactor;
529 sweep->interval_sec = ws_async_sweep_interval(timeout_sec);
530 sweep->head = ws;
531 sweep->dead = false;
532 sweep->posts = 0;
533 sweep->node.cb = ws_async_sweep_post_cb;
534 sweep->node.ctx = sweep;
535 sweep->next = NULL;
536 pthread_mutex_init(&sweep->lock, NULL);
537
538 ws->sweep = sweep;
539 ws->sweep_next = NULL;
540
541 pthread_t thread;
542 if (pthread_create(&thread, NULL, ws_async_sweep_watchdog, sweep) != 0) {
543 ws->sweep = NULL;
544 pthread_mutex_destroy(&sweep->lock);
545 cwist_free(sweep);
546 return;
547 }
548 pthread_detach(thread);
549
550 pthread_mutex_lock(&g_ws_sweeps_lock);
551 sweep->next = g_ws_sweeps;
552 g_ws_sweeps = sweep;
553 pthread_mutex_unlock(&g_ws_sweeps_lock);
554}

571static uint8_t *ws_async_serialize_frame(cwist_ws_opcode_t opcode, const uint8_t *data, size_t len,
572 size_t *out_total) {
573 uint8_t head[10];
574 size_t head_len = 2;
575
576 head[0] = 0x80 | (opcode & 0x0F);
577 if (len < 126) {
578 head[1] = (uint8_t)len;
579 } else if (len < 65536) {
580 head[1] = 126;
581 head[2] = (uint8_t)((len >> 8) & 0xFF);
582 head[3] = (uint8_t)(len & 0xFF);
583 head_len += 2;
584 } else {
585 head[1] = 127;
586 for (int i = 0; i < 8; i++) head[2 + i] = (uint8_t)(len >> ((7 - i) * 8));
587 head_len += 8;
588 }
589
590 uint8_t *buf = (uint8_t *)cwist_alloc(head_len + len);
591 if (!buf) return NULL;
592 memcpy(buf, head, head_len);
593 if (len > 0) memcpy(buf + head_len, data, len);
594 *out_total = head_len + len;
595 return buf;
596}

598static int ws_async_park_drain(cwist_websocket_async *ws, bool *made_progress) {
599 int flags = MSG_NOSIGNAL | MSG_DONTWAIT;
600 *made_progress = false;
601 while (ws->park_head) {
602 ws_async_park_node *node = ws->park_head;
603 ssize_t n = send(ws->fd, node->buf + node->off, node->len - node->off, flags);
604 if (n > 0) {
605 node->off += (size_t)n;
606 ws->parked_bytes -= (size_t)n;
607 *made_progress = true;
608 if (node->off == node->len) {
609 ws->park_head = node->next;
610 if (!ws->park_head) ws->park_tail = NULL;
611 cwist_free(node->buf);
612 cwist_free(node);
613 }
614 continue;
615 }
616 if (n < 0 && errno == EINTR) continue;
617 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) return 1;
618 return -1;
619 }
620 return 0;
621}

630static int ws_async_queue_frame(cwist_websocket_async *ws, cwist_ws_opcode_t opcode,
631 const uint8_t *data, size_t len) {
632 if (!cwist_ws_internal_send_args_valid(opcode, len)) return -1;
633
634 size_t frame_len = 0;
635 uint8_t *frame = ws_async_serialize_frame(opcode, data, len, &frame_len);
636 if (!frame) return -1;
637
638 size_t limit = ws_async_park_limit();
639 if (frame_len > limit || ws->parked_bytes > limit - frame_len) {
640 cwist_free(frame);
641 return 1;
642 }
643 ws_async_park_node *node = (ws_async_park_node *)cwist_alloc(sizeof(*node));
644 if (!node) {
645 cwist_free(frame);
646 return -1;
647 }
648 node->buf = frame;
649 node->off = 0;
650 node->len = frame_len;
651 node->next = NULL;
652 if (ws->park_tail)
653 ws->park_tail->next = node;
654 else
655 ws->park_head = node;
656 ws->park_tail = node;
657 ws->parked_bytes += frame_len;
658
659 bool made_progress = false;
660 int drain_rc = ws_async_park_drain(ws, &made_progress);
661 if (drain_rc < 0) return -1;
662 if (drain_rc == 0) {
663 ws_async_touch(ws);
664 return 0;
665 }
666
667 ws->park_deadline_sec = ws_async_park_deadline();
668 if (ws->write_armed) return 0;
669 ws_async_slot_t slot = {.ws = ws};
670 if (!cwist_reactor_add_out(ws->reactor, ws->fd, ws_async_write_ready, &slot, sizeof(slot)))
671 return -1;
672 ws->write_armed = true;
673 return 0;
674}

682static void ws_async_write_ready(int fd, void *ctx) {
683 ws_async_slot_t *slot = (ws_async_slot_t *)ctx;
684 cwist_websocket_async *ws = slot->ws;
685 (void)fd;
686 ws->write_armed = false;
687
688 if (ws->closing) {
689 ws_async_teardown(ws);
690 return;
691 }
692 if (ws_async_idle_reap(ws)) return;
693
694 bool made_progress = false;
695 int drain_rc = ws_async_park_drain(ws, &made_progress);
696 if (made_progress) {
697 ws->park_deadline_sec = ws_async_park_deadline();
698 ws_async_touch(ws);
699 }
700 if (drain_rc > 0 && cwist_fast_monotonic_sec() <= ws->park_deadline_sec &&
701 cwist_reactor_add_out(ws->reactor, fd, ws_async_write_ready, slot, sizeof(*slot))) {
702 ws->write_armed = true;
703 return;
704 }
705 if (drain_rc != 0) {
706 ws_async_terminate(ws);
707 return;
708 }
709
710 if (ws->terminate_pending) {
711 ws_async_terminate(ws);
712 return;
713 }
714 if (!ws->read_armed) {
715 ws_async_slot_t rslot = {.ws = ws};
716 if (cwist_reactor_add(ws->reactor, fd, ws_async_read_ready, &rslot, sizeof(rslot)))
717 ws->read_armed = true;
718 else
719 ws_async_terminate(ws);
720 }
721}

753void cwist_websocket_async_close(cwist_websocket_async *ws) {
754 if (!ws || ws->closed) return;
755 ws->closed = true;
756 /* Best-effort CLOSE frame; failure still tears the connection down. */
757 if (ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, NULL, 0) != 0) {
758 ws->terminate_pending = true;
759 return;
760 }
761 if (ws->parked_bytes == 0) {
762 /* Nothing parked: the CLOSE frame is on the wire; tear down now. The
763 * enclosing read turn (we run on the reactor thread) performs the
764 * actual free via terminate_pending. */
765 ws->terminate_pending = true;
766 }
767 /* Otherwise the parked-write completion terminates after draining. */
768}

782void cwist_websocket_async_close_code(cwist_websocket_async *ws, uint16_t code) {
783 if (!ws || ws->closed) return;
784 ws->closed = true;
785 uint8_t body[2] = {(uint8_t)(code >> 8), (uint8_t)(code & 0xFF)};
786 if (ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, body, sizeof(body)) != 0) {
787 ws->terminate_pending = true;
788 return;
789 }
790 if (ws->parked_bytes == 0) {
791 /* CLOSE frame drained: the enclosing reactor turn performs the free
792 * via terminate_pending, mirroring cwist_websocket_async_close(). */
793 ws->terminate_pending = true;
794 }
795}

810static bool ws_async_stash_append(cwist_websocket_async *ws, const uint8_t *src, size_t len) {
811 if (len == 0) return true;
812 size_t need = ws->stash_len + len;
813 if (need > ws->stash_cap) {
814 size_t new_cap = ws->stash_cap ? ws->stash_cap * 2 : 4096;
815 if (new_cap < need) new_cap = need;
816 uint8_t *nb = (uint8_t *)cwist_alloc(new_cap);
817 if (!nb) return false;
818 if (ws->stash_len) memcpy(nb, ws->stash, ws->stash_len);
819 cwist_free(ws->stash);
820 ws->stash = nb;
821 ws->stash_cap = new_cap;
822 }
823 memcpy(ws->stash + ws->stash_len, src, len);
824 ws->stash_len += len;
825 return true;
826}

839static bool ws_async_frag_append(cwist_websocket_async *ws, const uint8_t *src, size_t len) {
840 if (len > WS_ASYNC_MAX_MESSAGE_BYTES - ws->frag_len) return false;
841 size_t need = ws->frag_len + len + 1; /* +1 for null terminator */
842 if (need > ws->frag_cap) {
843 size_t new_cap = ws->frag_cap ? ws->frag_cap * 2 : 4096;
844 if (new_cap < need) new_cap = need;
845 uint8_t *nb = (uint8_t *)cwist_alloc(new_cap);
846 if (!nb) return false;
847 if (ws->frag_len) memcpy(nb, ws->frag_buf, ws->frag_len);
848 cwist_free(ws->frag_buf);
849 ws->frag_buf = nb;
850 ws->frag_cap = new_cap;
851 }
852 if (len > 0) memcpy(ws->frag_buf + ws->frag_len, src, len);
853 ws->frag_len += len;
854 return true;
855}

875static int ws_async_process_frame(cwist_websocket_async *ws, bool fin, cwist_ws_opcode_t opcode,
876 uint8_t *payload, size_t payload_len) {
877 bool is_control = (opcode >= 0x8);
878
879 /* --- Fragmented-message reassembly (RFC 6455 section 5.4) -------------- */
880 if (!is_control) {
881 if (opcode == CWIST_WS_FRAME_CONTINUATION) {
882 if (!ws->frag_active) {
883 cwist_free(payload);
884 return -1;
885 }
886 } else {
887 if (ws->frag_active) {
888 cwist_free(payload);
889 return -1;
890 }
891 if (!fin) {
892 ws->frag_active = true;
893 ws->frag_opcode = opcode;
894 }
895 }
896
897 if (ws->frag_active) {
898 if (!ws_async_frag_append(ws, payload, payload_len)) {
899 cwist_free(payload);
900 cwist_free(ws->frag_buf);
901 ws->frag_buf = NULL;
902 ws->frag_len = ws->frag_cap = 0;
903 ws->frag_active = false;
904 return -1;
905 }
906 cwist_free(payload);
907 if (!fin) return 0;
908
909 payload = ws->frag_buf;
910 payload_len = ws->frag_len;
911 payload[payload_len] = '\0';
912 opcode = ws->frag_opcode;
913
914 ws->frag_buf = NULL;
915 ws->frag_len = ws->frag_cap = 0;
916 ws->frag_active = false;
917 }
918 }
919
920 /* --- Deliver the complete frame ---------------------------------- */
921 if (opcode == CWIST_WS_FRAME_TEXT && !cwist_ws_internal_utf8_valid(payload, payload_len)) {
922 cwist_websocket_async_close_code(ws, 1007);
923 cwist_free(payload);
924 return -1;
925 }
926 if (opcode == CWIST_WS_FRAME_CLOSE) {
927 int close_error = cwist_ws_internal_close_error(payload, payload_len);
928 if (close_error) {
929 cwist_websocket_async_close_code(ws, (uint16_t)close_error);
930 cwist_free(payload);
931 return -1;
932 }
933 /* Echo the status code (first 2 bytes) back before closing. */
934 ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, payload, (payload_len >= 2) ? 2 : 0);
935 cwist_free(payload);
936 ws->closed = true;
937 ws->terminate_pending = true;
938 return 0;
939 }
940 if (opcode == CWIST_WS_FRAME_PING) {
941 /* RFC 6455 section 5.5.3: respond to every PING with a PONG carrying
942 * the same payload (up to 125 bytes per section 5.5). */
943 ws_async_queue_frame(ws, CWIST_WS_FRAME_PONG, payload, payload_len);
944 }
945
946 cwist_ws_frame *frame = (cwist_ws_frame *)cwist_alloc(sizeof(cwist_ws_frame));
947 if (!frame) {
948 cwist_free(payload);
949 return -1;
950 }
951 frame->fin = fin;
952 frame->opcode = opcode;
953 frame->payload = payload;
954 frame->payload_len = payload_len;
955 /* Ownership moves to the callback; it must destroy the frame. */
956 ws->on_message(ws, frame, ws->user_data);
957 return 0;
958}

971static int ws_async_feed(cwist_websocket_async *ws) {
972 size_t off = 0;
973
974 while (ws->stash_len - off >= 2) {
975 const uint8_t *base = ws->stash + off;
976 bool fin = (base[0] & 0x80) != 0;
977 cwist_ws_opcode_t opcode = base[0] & 0x0F;
978 if (opcode != CWIST_WS_FRAME_CONTINUATION && opcode != CWIST_WS_FRAME_TEXT &&
979 opcode != CWIST_WS_FRAME_BINARY && opcode != CWIST_WS_FRAME_CLOSE &&
980 opcode != CWIST_WS_FRAME_PING && opcode != CWIST_WS_FRAME_PONG)
981 return -1;
982
983 /* RFC 6455 section 5.2: RSV1/RSV2/RSV3 MUST be 0 unless an extension
984 * defines their meaning. CWIST has no such extension. */
985 if (base[0] & 0x70) return -1;
986
987 /* Client-to-server frames must be masked per RFC 6455 section 5.3. */
988 if (!(base[1] & 0x80)) return -1;
989
990 uint64_t payload_len = base[1] & 0x7F;
991 size_t hdr = 2;
992 if (payload_len == 126) {
993 if (ws->stash_len - off < 4) break;
994 payload_len = ((uint64_t)base[2] << 8) | (uint64_t)base[3];
995 if (payload_len < 126) {
996 cwist_websocket_async_close_code(ws, 1002);
997 return -1;
998 }
999 hdr = 4;
1000 } else if (payload_len == 127) {
1001 if (ws->stash_len - off < 10) break;
1002 const uint8_t *e = base + 2;
1003 payload_len = ((uint64_t)e[0] << 56) | ((uint64_t)e[1] << 48) | ((uint64_t)e[2] << 40) |
1004 ((uint64_t)e[3] << 32) | ((uint64_t)e[4] << 24) | ((uint64_t)e[5] << 16) |
1005 ((uint64_t)e[6] << 8) | (uint64_t)e[7];
1006 if (payload_len < 65536) {
1007 cwist_websocket_async_close_code(ws, 1002);
1008 return -1;
1009 }
1010 hdr = 10;
1011 }
1012
1013 /* Reject frames that would force a huge allocation before any
1014 * payload is read (OOM DoS vector). */
1015 if (payload_len > WS_ASYNC_MAX_PAYLOAD_BYTES) return -1;
1016
1017 /* RFC 6455 section 5.5: control frames MUST NOT carry more than 125
1018 * bytes and MUST NOT be fragmented. */
1019 bool is_control_early = (opcode >= 0x8);
1020 if (is_control_early && (payload_len > 125 || !fin)) return -1;
1021
1022 size_t total = hdr + 4 + (size_t)payload_len;
1023 if (ws->stash_len - off < total) break; /* Partial frame: wait for more. */
1024
1025 const uint8_t *mask = base + hdr;
1026 const uint8_t *src = base + hdr + 4;
1027 uint8_t *payload = NULL;
1028 if (payload_len > 0) {
1029 payload = (uint8_t *)cwist_alloc((size_t)payload_len + 1);
1030 if (!payload) return -1;
1031 for (uint64_t i = 0; i < payload_len; i++) payload[i] = src[i] ^ mask[i % 4];
1032 payload[payload_len] = '\0';
1033 }
1034
1035 off += total;
1036 if (ws_async_process_frame(ws, fin, opcode, payload, (size_t)payload_len) != 0) return -1;
1037 /* The CLOSE handler marks the connection for teardown; keep parsing
1038 * nothing further — the caller terminates. */
1039 if (ws->terminate_pending) break;
1040 }
1041
1042 if (off > 0) {
1043 memmove(ws->stash, ws->stash + off, ws->stash_len - off);
1044 ws->stash_len -= off;
1045 }
1046 return 0;
1047}

1066static void ws_async_read_ready(int fd, void *ctx) {
1067 ws_async_slot_t *slot = (ws_async_slot_t *)ctx;
1068 cwist_websocket_async *ws = slot->ws;
1069 uint8_t buf[WS_ASYNC_READ_CHUNK];
1070 bool eof = false, err = false, got_bytes = false;
1071
1072 ws->read_armed = false;
1073
1074 /* Deferred termination (sweep reap / async_close with slots pending):
1075 * shutdown() already signalled the peer; this woken slot is the only one
1076 * left, so it performs the real teardown. */
1077 if (ws->closing) {
1078 ws_async_teardown(ws);
1079 return;
1080 }
1081
1082 /* Lazy idle reaper (mirrors http_async_event_cb): checked before any
1083 * work so an event arriving after the deadline closes the connection
1084 * without touching the stash. */
1085 if (ws_async_idle_reap(ws)) return;
1086
1087 /* Drain everything the kernel has without blocking. */
1088 for (;;) {
1089 ssize_t n = recv(fd, buf, sizeof(buf), MSG_DONTWAIT);
1090 if (n > 0) {
1091 got_bytes = true;
1092 if (!ws_async_stash_append(ws, (const uint8_t *)buf, (size_t)n)) {
1093 err = true;
1094 break;
1095 }
1096 continue;
1097 }
1098 if (n == 0) {
1099 eof = true;
1100 break;
1101 }
1102 if (errno == EINTR) continue;
1103 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
1104 err = true;
1105 break;
1106 }
1107
1108 if (!err && ws->stash_len > 0 && ws_async_feed(ws) != 0) err = true;
1109
1110 /* Bytes on the wire (and therefore every complete frame, PING/PONG
1111 * included) count as activity for the idle reaper. */
1112 if (got_bytes && !err && !eof) ws_async_touch(ws);
1113
1114 /* CLOSE-from-peer, async_close() from the callback, EOF and errors all
1115 * converge here: tear the connection down exactly once. */
1116 if (err || eof || ws->terminate_pending) {
1117 ws_async_terminate(ws);
1118 return;
1119 }
1120
1121 /* One-shot slot consumed: re-arm for the next read event. */
1122 if (!cwist_reactor_add(ws->reactor, fd, ws_async_read_ready, slot, sizeof(*slot))) {
1123 ws_async_terminate(ws);
1124 return;
1125 }
1126 ws->read_armed = true;
1127}

1150cwist_websocket_async *cwist_websocket_async_attach_ex(int fd, cwist_reactor_t *reactor,
1151 cwist_ws_on_message_t on_message,
1152 cwist_ws_on_close_t on_close,
1153 void *user_data, const uint8_t *initial,
1154 size_t initial_len) {
1155 if (fd < 0 || !reactor || !on_message) return NULL;
1156
1157 cwist_websocket_async *ws = (cwist_websocket_async *)cwist_alloc(sizeof(*ws));
1158 if (!ws) {
1159 close(fd);
1160 return NULL;
1161 }
1162 ws->fd = fd;
1163 ws->reactor = reactor;
1164 ws->on_message = on_message;
1165 ws->on_close = on_close;
1166 ws->user_data = user_data;
1167 ws->stash = NULL;
1168 ws->stash_len = ws->stash_cap = 0;
1169 ws->frag_buf = NULL;
1170 ws->frag_len = ws->frag_cap = 0;
1171 ws->frag_opcode = CWIST_WS_FRAME_CONTINUATION;
1172 ws->frag_active = false;
1173 ws->park_head = ws->park_tail = NULL;
1174 ws->parked_bytes = 0;
1175 ws->park_deadline_sec = 0;
1176 ws->last_active_sec = cwist_fast_monotonic_sec();
1177 ws->sweep = NULL;
1178 ws->sweep_next = NULL;
1179 ws->closed = false;
1180 ws->closing = false;
1181 ws->terminate_pending = false;
1182 ws->read_armed = false;
1183 ws->write_armed = false;
1184
1185 /* The fd must never block the reactor thread. */
1186 int fl = fcntl(fd, F_GETFL, 0);
1187 if (fl >= 0) fcntl(fd, F_SETFL, fl | O_NONBLOCK);
1188
1189 /* Bytes already read past the upgrade request (conn->rbuf) go through the
1190 * parser before the first reactor wait. Delivery can happen inline here,
1191 * including a cwist_websocket_async_close() from the callback. */
1192 bool fail = false;
1193 if (initial_len > 0) {
1194 if (!ws_async_stash_append(ws, initial, initial_len)) fail = true;
1195 if (!fail && ws_async_feed(ws) != 0) fail = true;
1196 }
1197 if (fail || ws->terminate_pending) {
1198 /* No reactor slot was registered yet: nothing to cancel. on_close
1199 * does not fire here; attach failed, so the caller keeps ownership of
1200 * any state installed during inline initial-bytes delivery. */
1201 close(fd);
1202 cwist_free(ws->stash);
1203 cwist_free(ws->frag_buf);
1204 ws_async_park_clear(ws);
1205 cwist_free(ws);
1206 return NULL;
1207 }
1208
1209 /* Everything the reactor thread reads is set before the slot is added:
1210 * once cwist_reactor_add() returns, the read callback may already run
1211 * on the reactor thread (attach can be called from any thread), so this
1212 * thread must not write ws any more (#351).
1213 *
1214 * Join the reactor's idle sweep (no-op when reaping is disabled). */
1215 ws_async_sweep_link(ws, reactor);
1216 ws->read_armed = true;
1217
1218 ws_async_slot_t slot = {.ws = ws};
1219 if (!cwist_reactor_add(reactor, fd, ws_async_read_ready, &slot, sizeof(slot))) {
1220 ws->read_armed = false;
1221 ws_async_sweep_unlink(ws);
1222 close(fd);
1223 cwist_free(ws->stash);
1224 cwist_free(ws->frag_buf);
1225 ws_async_park_clear(ws);
1226 cwist_free(ws);
1227 return NULL;
1228 }
1229 return ws;
1230}
