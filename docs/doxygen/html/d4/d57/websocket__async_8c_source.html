181static void ws_async_terminate(cwist_websocket_async *ws) {
182 if (!ws) return;
183 if (ws->sweep) {
184 /* Unlink from the reactor's sweep list; when the last connection is
185 * gone the sweep context is marked dead and its watchdog frees it
186 * once the last posted sweep has drained. */
187 ws_async_sweep_t *sweep = ws->sweep;
188 pthread_mutex_lock(&sweep->lock);
189 cwist_websocket_async **pp = &sweep->head;
190 while (*pp && *pp != ws) pp = &(*pp)->sweep_next;
191 if (*pp) *pp = ws->sweep_next;
192 bool empty = (sweep->head == NULL);
193 pthread_mutex_unlock(&sweep->lock);
194 ws->sweep = NULL;
195 if (empty) ws_async_sweep_mark_dead(sweep);
196 }
197 bool parked = ws->park_len > ws->park_off;
198 if (ws->read_armed || parked) {
199 ws->closing = true;
200 /* With a read slot armed, the read callback performs the teardown;
201 * cancel the parked-write slot so it cannot dispatch as well. With
202 * only the write slot pending, shutdown() wakes it and its callback
203 * performs the teardown. */
204 if (ws->read_armed && parked) cwist_reactor_del(ws->reactor, ws->fd);
205 /* FIN now; the peer sees EOF immediately even though the final close
206 * waits for the woken slot. */
207 shutdown(ws->fd, SHUT_RDWR);
208 return;
209 }
210 ws_async_teardown(ws);
211}

234static uint32_t ws_async_idle_timeout_sec(void) {
235 static _Atomic int cached = -1;
236 int v = atomic_load_explicit(&cached, memory_order_relaxed);
237 if (v < 0) {
238 const char *env = getenv("CWIST_WS_ASYNC_IDLE_TIMEOUT_SEC");
239 v = 300; /* default: 5 minutes */
240 if (env && *env) {
241 char *end = NULL;
242 long parsed = strtol(env, &end, 10);
243 if (end != env && *end == '\0' && parsed >= 0 && parsed <= INT_MAX) v = (int)parsed;
244 }
245 atomic_store_explicit(&cached, v, memory_order_relaxed);
246 }
247 return (uint32_t)v;
248}

276static bool ws_async_idle_reap(cwist_websocket_async *ws) {
277 uint32_t timeout_sec = ws_async_idle_timeout_sec();
278 if (timeout_sec == 0) return false;
279 uint32_t now = cwist_fast_monotonic_sec();
280 if (ws->last_active_sec > 0 && (now - ws->last_active_sec) > timeout_sec) {

291/* Lazy per-event reaping (the HTTP keep-alive precedent) cannot reap a
292 * totally silent connection: the one-shot slot model gives each fd at most
293 * one read + one POLLOUT slot, so an idle fd fires no event. Instead each
294 * reactor that owns WS async connections gets ONE watchdog thread (plain
295 * pthreads + nanosleep — portable across io_uring, epoll, and kqueue, with
296 * no Linux-only timerfd). The thread sleeps the sweep interval, then hands
297 * the sweep to the reactor thread via cwist_reactor_post(); the posted
298 * callback runs the same conn-list walk + terminate as the per-event reaper,
299 * so all list mutation stays on the reactor thread. The per-event reaper
300 * still runs at every read/write event, so the sweep cadence only affects
301 * how late a silent connection is reaped, not whether.
302 *
303 * Lifetime: the context is marked dead when its connection list empties
304 * (terminate) or the walk finds it empty. The watchdog notices the dead
305 * flag within one 100 ms sleep slice, stops posting, and exits; the context
306 * is freed by the watchdog thread alone after it exits its posting loop and
307 * observes posts == 0: the callback only ever decrements the posts counter
308 * (as its last touch of the context), so the watchdog can never reference
309 * freed memory. The posts counter covers the window where a post is in
310 * flight while the context is being torn down; cwist_reactor_destroy()
311 * drains pending posts, so the wait always terminates. */
312
322static uint32_t ws_async_sweep_interval(uint32_t timeout_sec) {
323 uint32_t interval = timeout_sec / 2;
324 if (interval < 1) interval = 1;
325 if (interval > 60) interval = 60;
326 return interval;
327}

341static void ws_async_sweep_watchdog_free(ws_async_sweep_t *sweep) {
342 const struct timespec slice = {.tv_sec = 0, .tv_nsec = 100000000L};
343 for (;;) {
344 pthread_mutex_lock(&sweep->lock);
345 bool drained = (sweep->posts == 0);
346 pthread_mutex_unlock(&sweep->lock);
347 if (drained) break;
348 nanosleep(&slice, NULL);
349 }
350
351 pthread_mutex_lock(&g_ws_sweeps_lock);
352 ws_async_sweep_t **pp = &g_ws_sweeps;
353 while (*pp && *pp != sweep) pp = &(*pp)->next;
354 if (*pp) *pp = sweep->next;
355 pthread_mutex_unlock(&g_ws_sweeps_lock);
356 pthread_mutex_destroy(&sweep->lock);
357 cwist_free(sweep);
358}

371static void ws_async_sweep_post_cb(void *ctx) {
372 ws_async_sweep_t *sweep = (ws_async_sweep_t *)ctx;
373
374 uint32_t timeout_sec = ws_async_idle_timeout_sec();
375 if (timeout_sec > 0) {
376 uint32_t now = cwist_fast_monotonic_sec();
377 /* Reap at most one victim per tick: ws_async_terminate unlinks the
378 * connection (and marks the sweep dead when the list empties), so
379 * the next posted sweep restarts from the head. Sweeps are small
380 * and ticks are ~interval apart, so the pace is fine. */
381 pthread_mutex_lock(&sweep->lock);
382 cwist_websocket_async *victim = NULL;
383 for (cwist_websocket_async *ws = sweep->head; ws; ws = ws->sweep_next) {
384 if (ws->last_active_sec > 0 && (now - ws->last_active_sec) > timeout_sec) {
385 victim = ws;
386 break;
387 }
388 }
389 pthread_mutex_unlock(&sweep->lock);
390 if (victim) ws_async_terminate(victim);
391 }
392
393 /* posts-- is the last touch of the context: the watchdog frees it only
394 * after observing posts == 0. */
395 pthread_mutex_lock(&sweep->lock);
396 bool empty = (sweep->head == NULL);
397 if (sweep->posts > 0) sweep->posts--;
398 pthread_mutex_unlock(&sweep->lock);
399 if (empty) ws_async_sweep_mark_dead(sweep);
400}

402/* Sleep the interval in 100 ms slices so dead-flag shutdown latency stays
403 * low without a condvar; nanosleep is EINTR-tolerant. */
404#define WS_ASYNC_SWEEP_SLICE_MS 100u
405
417static void *ws_async_sweep_watchdog(void *arg) {
418 ws_async_sweep_t *sweep = (ws_async_sweep_t *)arg;
419 const struct timespec slice = {.tv_sec = 0, .tv_nsec = 100000000L};
420 uint32_t slices = sweep->interval_sec * (1000u / WS_ASYNC_SWEEP_SLICE_MS);
421
422 for (;;) {
423 for (uint32_t i = 0; i < slices; i++) {
424 nanosleep(&slice, NULL);
425 pthread_mutex_lock(&sweep->lock);
426 bool dead = sweep->dead;
427 pthread_mutex_unlock(&sweep->lock);
428 if (dead) goto out;
429 }
430 pthread_mutex_lock(&sweep->lock);
431 bool dead = sweep->dead;
432 if (!dead) sweep->posts++;
433 pthread_mutex_unlock(&sweep->lock);
434 if (dead) goto out;
435 if (!cwist_reactor_post(sweep->reactor, &sweep->node)) {
436 pthread_mutex_lock(&sweep->lock);
437 sweep->posts--;
438 pthread_mutex_unlock(&sweep->lock);
439 }
440 }
441
442out:
443 /* Sole freer of the context: the watchdog is done with it, and any
444 * posted sweep still queued must drain before the memory is released. */
445 ws_async_sweep_watchdog_free(sweep);
446 return NULL;
447}

459static ws_async_sweep_t *ws_async_sweep_get(cwist_reactor_t *reactor) {
460 uint32_t timeout_sec = ws_async_idle_timeout_sec();
461 if (timeout_sec == 0) return NULL;
462
463 pthread_mutex_lock(&g_ws_sweeps_lock);
464 for (ws_async_sweep_t *s = g_ws_sweeps; s; s = s->next) {
465 /* A dead context awaiting teardown is unowned: a fresh one is
466 * created so the new connection is swept. dead is only written
467 * under g_ws_sweeps_lock (mark_dead), so this read is safe. */
468 if (s->reactor == reactor && !s->dead) {
469 pthread_mutex_unlock(&g_ws_sweeps_lock);
470 return s;
471 }
472 }
473 pthread_mutex_unlock(&g_ws_sweeps_lock);
474
475 ws_async_sweep_t *sweep = (ws_async_sweep_t *)cwist_alloc(sizeof(*sweep));
476 if (!sweep) return NULL;
477 sweep->reactor = reactor;
478 sweep->interval_sec = ws_async_sweep_interval(timeout_sec);
479 sweep->head = NULL;
480 sweep->dead = false;
481 sweep->posts = 0;
482 sweep->node.cb = ws_async_sweep_post_cb;
483 sweep->node.ctx = sweep;
484 sweep->next = NULL;
485 pthread_mutex_init(&sweep->lock, NULL);
486
487 pthread_t thread;
488 if (pthread_create(&thread, NULL, ws_async_sweep_watchdog, sweep) != 0) {
489 pthread_mutex_destroy(&sweep->lock);
490 cwist_free(sweep);
491 return NULL;
492 }
493 pthread_detach(thread);
494
495 pthread_mutex_lock(&g_ws_sweeps_lock);
496 sweep->next = g_ws_sweeps;
497 g_ws_sweeps = sweep;
498 pthread_mutex_unlock(&g_ws_sweeps_lock);
499 return sweep;
500}

517static uint8_t *ws_async_serialize_frame(cwist_ws_opcode_t opcode, const uint8_t *data, size_t len,
518 size_t *out_total) {
519 uint8_t head[10];
520 size_t head_len = 2;
521
522 head[0] = 0x80 | (opcode & 0x0F);
523 if (len < 126) {
524 head[1] = (uint8_t)len;
525 } else if (len < 65536) {
526 head[1] = 126;
527 head[2] = (uint8_t)((len >> 8) & 0xFF);
528 head[3] = (uint8_t)(len & 0xFF);
529 head_len += 2;
530 } else {
531 head[1] = 127;
532 for (int i = 0; i < 8; i++) head[2 + i] = (uint8_t)(len >> ((7 - i) * 8));
533 head_len += 8;
534 }
535
536 uint8_t *buf = (uint8_t *)cwist_alloc(head_len + len);
537 if (!buf) return NULL;
538 memcpy(buf, head, head_len);
539 if (len > 0) memcpy(buf + head_len, data, len);
540 *out_total = head_len + len;
541 return buf;
542}

559static int ws_async_queue_frame(cwist_websocket_async *ws, cwist_ws_opcode_t opcode,
560 const uint8_t *data, size_t len) {
561 size_t frame_len = 0;
562 uint8_t *frame = ws_async_serialize_frame(opcode, data, len, &frame_len);
563 if (!frame) return -1;
564
565 size_t pending = ws->park_len - ws->park_off;
566 if (pending > 0) {
567 /* Append behind the unsent remainder, compacting to the front. */
568 uint8_t *nb = (uint8_t *)cwist_alloc(pending + frame_len);
569 if (!nb) {
570 cwist_free(frame);
571 return -1;
572 }
573 memcpy(nb, ws->park_buf + ws->park_off, pending);
574 memcpy(nb + pending, frame, frame_len);
575 cwist_free(ws->park_buf);
576 ws->park_buf = nb;
577 ws->park_off = 0;
578 ws->park_len = pending + frame_len;
579 } else {
580 cwist_free(ws->park_buf);
581 ws->park_buf = frame;
582 ws->park_off = 0;
583 ws->park_len = frame_len;
584 frame = NULL;
585 }
586 cwist_free(frame);
587
588 int flags = MSG_NOSIGNAL | MSG_DONTWAIT;
589 while (ws->park_off < ws->park_len) {
590 ssize_t n = send(ws->fd, ws->park_buf + ws->park_off, ws->park_len - ws->park_off, flags);
591 if (n > 0) {
592 ws->park_off += (size_t)n;
593 continue;
594 }
595 if (n < 0 && errno == EINTR) continue;
596 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) break;
597 return -1; /* Fatal send error. */
598 }
599
600 if (ws->park_off == ws->park_len) {
601 /* Fully drained. */
602 cwist_free(ws->park_buf);
603 ws->park_buf = NULL;
604 ws->park_off = ws->park_len = 0;
605 ws_async_touch(ws);
606 return 0;
607 }
608
609 ws->park_deadline_sec = ws_async_park_deadline();
610 ws_async_slot_t slot = {.ws = ws};
611 if (!cwist_reactor_add_out(ws->reactor, ws->fd, ws_async_write_ready, &slot, sizeof(slot)))
612 return -1;
613 return 0;
614}

628static void ws_async_write_ready(int fd, void *ctx) {
629 ws_async_slot_t *slot = (ws_async_slot_t *)ctx;
630 cwist_websocket_async *ws = slot->ws;
631 int flags = MSG_NOSIGNAL | MSG_DONTWAIT;
632
633 /* Deferred termination: shutdown() woke this slot so it can perform the
634 * real teardown. */
635 if (ws->closing) {
636 ws_async_teardown(ws);
637 return;
638 }
639
640 if (ws_async_idle_reap(ws)) return;
641
642 while (ws->park_off < ws->park_len) {
643 ssize_t n = send(fd, ws->park_buf + ws->park_off, ws->park_len - ws->park_off, flags);
644 if (n > 0) {
645 ws->park_off += (size_t)n;
646 /* Progress resets the budget so a slow-but-alive client can drain. */
647 ws->park_deadline_sec = ws_async_park_deadline();
648 ws_async_touch(ws);
649 continue;
650 }
651 if (n < 0 && errno == EINTR) continue;
652 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK) &&
653 cwist_fast_monotonic_sec() <= ws->park_deadline_sec &&
654 cwist_reactor_add_out(ws->reactor, fd, ws_async_write_ready, slot, sizeof(*slot))) {
655 /* Payload copied into the new slot; buffer ownership moves with it. */
656 return;
657 }
658 ws_async_terminate(ws);
659 return;
660 }
661
662 cwist_free(ws->park_buf);
663 ws->park_buf = NULL;
664 ws->park_off = ws->park_len = 0;
665
666 if (ws->terminate_pending) {
667 ws_async_terminate(ws);
668 return;
669 }
670 /* Re-arm the read side only when the read turn is not already armed. */
671 if (!ws->read_armed) {
672 ws_async_slot_t rslot = {.ws = ws};
673 if (cwist_reactor_add(ws->reactor, fd, ws_async_read_ready, &rslot, sizeof(rslot)))
674 ws->read_armed = true;
675 else
676 ws_async_terminate(ws);
677 }
678}

709void cwist_websocket_async_close(cwist_websocket_async *ws) {
710 if (!ws || ws->closed) return;
711 ws->closed = true;
712 /* Best-effort CLOSE frame; failure still tears the connection down. */
713 if (ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, NULL, 0) != 0) {
714 ws->terminate_pending = true;
715 return;
716 }
717 if (ws->park_off == ws->park_len) {
718 /* Nothing parked: the CLOSE frame is on the wire; tear down now. The
719 * enclosing read turn (we run on the reactor thread) performs the
720 * actual free via terminate_pending. */
721 ws->terminate_pending = true;
722 }
723 /* Otherwise the parked-write completion terminates after draining. */
724}

738void cwist_websocket_async_close_code(cwist_websocket_async *ws, uint16_t code) {
739 if (!ws || ws->closed) return;
740 ws->closed = true;
741 uint8_t body[2] = {(uint8_t)(code >> 8), (uint8_t)(code & 0xFF)};
742 if (ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, body, sizeof(body)) != 0) {
743 ws->terminate_pending = true;
744 return;
745 }
746 if (ws->park_off == ws->park_len) {
747 /* CLOSE frame drained: the enclosing reactor turn performs the free
748 * via terminate_pending, mirroring cwist_websocket_async_close(). */
749 ws->terminate_pending = true;
750 }

766static bool ws_async_stash_append(cwist_websocket_async *ws, const uint8_t *src, size_t len) {
767 if (len == 0) return true;
768 size_t need = ws->stash_len + len;
769 if (need > ws->stash_cap) {
770 size_t new_cap = ws->stash_cap ? ws->stash_cap * 2 : 4096;
771 if (new_cap < need) new_cap = need;
772 uint8_t *nb = (uint8_t *)cwist_alloc(new_cap);
773 if (!nb) return false;
774 if (ws->stash_len) memcpy(nb, ws->stash, ws->stash_len);
775 cwist_free(ws->stash);
776 ws->stash = nb;
777 ws->stash_cap = new_cap;
778 }
779 memcpy(ws->stash + ws->stash_len, src, len);
780 ws->stash_len += len;
781 return true;
782}

832static int ws_async_process_frame(cwist_websocket_async *ws, bool fin, cwist_ws_opcode_t opcode,
833 uint8_t *payload, size_t payload_len) {
834 bool is_control = (opcode >= 0x8);
835
836 /* --- Fragmented-message reassembly (RFC 6455 section 5.4) -------------- */
837 if (!is_control && !fin) {
838 /* Fragment with FIN=0: start or continue reassembly. */
839 if (opcode == CWIST_WS_FRAME_CONTINUATION) {
840 /* CONTINUATION is only valid when a fragmented message is
841 * already in progress. */
842 if (ws->frag_len == 0) {
843 cwist_free(payload);
844 return -1;
845 }
846 } else {
847 /* First fragment: save opcode (text vs binary). */
848 ws->frag_opcode = opcode;
849 }
850 if (!ws_async_frag_append(ws, payload, payload_len)) {
851 cwist_free(payload);
852 return -1;
853 }
854 cwist_free(payload);
855 return 0;
856 }
857
858 /* A CONTINUATION frame is only valid inside an active fragmented-message
859 * sequence; an orphan CONTINUATION with FIN=1 must not be delivered as a
860 * complete message (mirrors the blocking path, issue #158). */
861 if (!is_control && opcode == CWIST_WS_FRAME_CONTINUATION && ws->frag_len == 0) {
862 cwist_free(payload);
863 return -1;
864 }
865
866 /* FIN=1 data frame: may be the last fragment of a multi-frame message. */
867 if (!is_control && ws->frag_len > 0) {
868 if (!ws_async_frag_append(ws, payload, payload_len)) {
869 cwist_free(payload);
870 cwist_free(ws->frag_buf);
871 ws->frag_buf = NULL;
872 ws->frag_len = ws->frag_cap = 0;
873 return -1;
874 }
875 cwist_free(payload);
876
877 /* Hand ownership of the reassembly buffer to the delivered frame. */
878 payload = ws->frag_buf;
879 payload_len = ws->frag_len;
880 payload[payload_len] = '\0';
881 opcode = ws->frag_opcode;
882 fin = true;
883
884 ws->frag_buf = NULL;
885 ws->frag_len = ws->frag_cap = 0;
886 }
887
888 /* --- Deliver the complete frame ---------------------------------- */
889 if (opcode == CWIST_WS_FRAME_CLOSE) {
890 /* RFC 6455 section 5.5.1: a CLOSE body, if present, MUST be >= 2 bytes. */
891 if (payload_len == 1) {
892 cwist_free(payload);
893 return -1;
894 }
895 /* Echo the status code (first 2 bytes) back before closing. */
896 ws_async_queue_frame(ws, CWIST_WS_FRAME_CLOSE, payload, (payload_len >= 2) ? 2 : 0);
897 cwist_free(payload);
898 ws->closed = true;
899 ws->terminate_pending = true;
900 return 0;
901 }
902 if (opcode == CWIST_WS_FRAME_PING) {
903 /* RFC 6455 section 5.5.3: respond to every PING with a PONG carrying
904 * the same payload (up to 125 bytes per section 5.5). */
905 ws_async_queue_frame(ws, CWIST_WS_FRAME_PONG, payload, payload_len);
906 }
907
908 cwist_ws_frame *frame = (cwist_ws_frame *)cwist_alloc(sizeof(cwist_ws_frame));
909 if (!frame) {
910 cwist_free(payload);
911 return -1;
912 }
913 frame->fin = fin;
914 frame->opcode = opcode;
915 frame->payload = payload;
916 frame->payload_len = payload_len;
917 /* Ownership moves to the callback; it must destroy the frame. */
918 ws->on_message(ws, frame, ws->user_data);
919 return 0;
920}

933static int ws_async_feed(cwist_websocket_async *ws) {
934 size_t off = 0;
935
936 while (ws->stash_len - off >= 2) {
937 const uint8_t *base = ws->stash + off;
938 bool fin = (base[0] & 0x80) != 0;
939 cwist_ws_opcode_t opcode = base[0] & 0x0F;
940
941 /* RFC 6455 section 5.2: RSV1/RSV2/RSV3 MUST be 0 unless an extension
942 * defines their meaning. CWIST has no such extension. */
943 if (base[0] & 0x70) return -1;
944
945 /* Client-to-server frames must be masked per RFC 6455 section 5.3. */
946 if (!(base[1] & 0x80)) return -1;
947
948 uint64_t payload_len = base[1] & 0x7F;
949 size_t hdr = 2;
950 if (payload_len == 126) {
951 if (ws->stash_len - off < 4) break;
952 payload_len = ((uint64_t)base[2] << 8) | (uint64_t)base[3];
953 hdr = 4;
954 } else if (payload_len == 127) {
955 if (ws->stash_len - off < 10) break;
956 const uint8_t *e = base + 2;
957 payload_len = ((uint64_t)e[0] << 56) | ((uint64_t)e[1] << 48) | ((uint64_t)e[2] << 40) |
958 ((uint64_t)e[3] << 32) | ((uint64_t)e[4] << 24) | ((uint64_t)e[5] << 16) |
959 ((uint64_t)e[6] << 8) | (uint64_t)e[7];
960 hdr = 10;
961 }
962
963 /* Reject frames that would force a huge allocation before any
964 * payload is read (OOM DoS vector). */
965 if (payload_len > WS_ASYNC_MAX_PAYLOAD_BYTES) return -1;
966
967 /* RFC 6455 section 5.5: control frames MUST NOT carry more than 125
968 * bytes and MUST NOT be fragmented. */
969 bool is_control_early = (opcode >= 0x8);
970 if (is_control_early && (payload_len > 125 || !fin)) return -1;
971
972 size_t total = hdr + 4 + (size_t)payload_len;
973 if (ws->stash_len - off < total) break; /* Partial frame: wait for more. */
974
975 const uint8_t *mask = base + hdr;
976 const uint8_t *src = base + hdr + 4;
977 uint8_t *payload = NULL;
978 if (payload_len > 0) {
979 payload = (uint8_t *)cwist_alloc((size_t)payload_len + 1);
980 if (!payload) return -1;
981 for (uint64_t i = 0; i < payload_len; i++) payload[i] = src[i] ^ mask[i % 4];
982 payload[payload_len] = '\0';
983 }
984
985 off += total;
986 if (ws_async_process_frame(ws, fin, opcode, payload, (size_t)payload_len) != 0) return -1;
987 /* The CLOSE handler marks the connection for teardown; keep parsing
988 * nothing further — the caller terminates. */
989 if (ws->terminate_pending) break;
990 }
991
992 if (off > 0) {
993 memmove(ws->stash, ws->stash + off, ws->stash_len - off);
994 ws->stash_len -= off;
995 }
996 return 0;
997}

1016static void ws_async_read_ready(int fd, void *ctx) {
1017 ws_async_slot_t *slot = (ws_async_slot_t *)ctx;
1018 cwist_websocket_async *ws = slot->ws;
1019 uint8_t buf[WS_ASYNC_READ_CHUNK];
1020 bool eof = false, err = false, got_bytes = false;
1021
1022 ws->read_armed = false;
1023
1024 /* Deferred termination (sweep reap / async_close with slots pending):
1025 * shutdown() already signalled the peer; this woken slot is the only one
1026 * left, so it performs the real teardown. */
1027 if (ws->closing) {
1028 ws_async_teardown(ws);
1029 return;
1030 }
1031
1032 /* Lazy idle reaper (mirrors http_async_event_cb): checked before any
1033 * work so an event arriving after the deadline closes the connection
1034 * without touching the stash. */
1035 if (ws_async_idle_reap(ws)) return;
1036
1037 /* Drain everything the kernel has without blocking. */
1038 for (;;) {
1039 ssize_t n = recv(fd, buf, sizeof(buf), MSG_DONTWAIT);
1040 if (n > 0) {
1041 got_bytes = true;
1042 if (!ws_async_stash_append(ws, (const uint8_t *)buf, (size_t)n)) {
1043 err = true;
1044 break;
1045 }
1046 continue;
1047 }
1048 if (n == 0) {
1049 eof = true;
1050 break;
1051 }
1052 if (errno == EINTR) continue;
1053 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
1054 err = true;
1055 break;
1056 }
1057
1058 if (!err && ws->stash_len > 0 && ws_async_feed(ws) != 0) err = true;
1059
1060 /* Bytes on the wire (and therefore every complete frame, PING/PONG
1061 * included) count as activity for the idle reaper. */
1062 if (got_bytes && !err && !eof) ws_async_touch(ws);
1063
1064 /* CLOSE-from-peer, async_close() from the callback, EOF and errors all
1065 * converge here: tear the connection down exactly once. */
1066 if (err || eof || ws->terminate_pending) {
1067 ws_async_terminate(ws);
1068 return;
1069 }
1070
1071 /* One-shot slot consumed: re-arm for the next read event. */
1072 if (!cwist_reactor_add(ws->reactor, fd, ws_async_read_ready, slot, sizeof(*slot))) {
1073 ws_async_terminate(ws);
1074 return;
1075 }
1076 ws->read_armed = true;
1077}

1100cwist_websocket_async *cwist_websocket_async_attach_ex(int fd, cwist_reactor_t *reactor,
1101 cwist_ws_on_message_t on_message,
1102 cwist_ws_on_close_t on_close,
1103 void *user_data, const uint8_t *initial,
1104 size_t initial_len) {
1105 if (fd < 0 || !reactor || !on_message) return NULL;
1106
1107 cwist_websocket_async *ws = (cwist_websocket_async *)cwist_alloc(sizeof(*ws));
1108 if (!ws) {
1109 close(fd);
1110 return NULL;
1111 }
1112 ws->fd = fd;
1113 ws->reactor = reactor;
1114 ws->on_message = on_message;
1115 ws->on_close = on_close;
1116 ws->user_data = user_data;
1117 ws->stash = NULL;
1118 ws->stash_len = ws->stash_cap = 0;
1119 ws->frag_buf = NULL;
1120 ws->frag_len = ws->frag_cap = 0;
1121 ws->frag_opcode = CWIST_WS_FRAME_CONTINUATION;
1122 ws->park_buf = NULL;
1123 ws->park_off = ws->park_len = 0;
1124 ws->park_deadline_sec = 0;
1125 ws->last_active_sec = cwist_fast_monotonic_sec();
1126 ws->sweep = NULL;
1127 ws->sweep_next = NULL;
1128 ws->closed = false;
1129 ws->closing = false;
1130 ws->terminate_pending = false;
1131 ws->read_armed = false;
1132
1133 /* The fd must never block the reactor thread. */
1134 int fl = fcntl(fd, F_GETFL, 0);
1135 if (fl >= 0) fcntl(fd, F_SETFL, fl | O_NONBLOCK);
1136
1137 /* Bytes already read past the upgrade request (conn->rbuf) go through the
1138 * parser before the first reactor wait. Delivery can happen inline here,
1139 * including a cwist_websocket_async_close() from the callback. */
1140 bool fail = false;
1141 if (initial_len > 0) {
1142 if (!ws_async_stash_append(ws, initial, initial_len)) fail = true;
1143 if (!fail && ws_async_feed(ws) != 0) fail = true;
1144 }
1145 if (fail || ws->terminate_pending) {
1146 /* No reactor slot was registered yet: nothing to cancel. on_close
1147 * does not fire here; attach failed, so the caller keeps ownership of
1148 * any state installed during inline initial-bytes delivery. */
1149 close(fd);
1150 cwist_free(ws->stash);
1151 cwist_free(ws->frag_buf);
1152 cwist_free(ws->park_buf);
1153 cwist_free(ws);
1154 return NULL;
1155 }
1156
1157 ws_async_slot_t slot = {.ws = ws};
1158 if (!cwist_reactor_add(reactor, fd, ws_async_read_ready, &slot, sizeof(slot))) {
1159 close(fd);
1160 cwist_free(ws->stash);
1161 cwist_free(ws->frag_buf);
1162 cwist_free(ws->park_buf);
1163 cwist_free(ws);
1164 return NULL;
1165 }
1166 ws->read_armed = true;
1167
1168 /* Join the reactor's idle sweep (no-op when reaping is disabled). Done
1169 * after the read slot is armed so a failed attach never unlinks. The
1170 * g_ws_sweeps_lock -> sweep->lock order matches the sweeper's teardown
1171 * protocol, so a sweep marked dead is never pushed to. */
1172 ws_async_sweep_t *sweep = ws_async_sweep_get(reactor);
1173 if (sweep) {
1174 pthread_mutex_lock(&g_ws_sweeps_lock);
1175 pthread_mutex_lock(&sweep->lock);
1176 if (!sweep->dead) {
1177 ws->sweep = sweep;
1178 ws->sweep_next = sweep->head;
1179 sweep->head = ws;
1180 }
1181 pthread_mutex_unlock(&sweep->lock);
1182 pthread_mutex_unlock(&g_ws_sweeps_lock);
1183 }
1184 return ws;
1185}
