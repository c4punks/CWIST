48static bool pool_cond_init(pthread_cond_t *cond) {
49#if defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__) || defined(__APPLE__)
50 return pthread_cond_init(cond, NULL) == 0;
51#else
52 pthread_condattr_t attr;
53 if (pthread_condattr_init(&attr) != 0) return false;
54 bool ok = pthread_condattr_setclock(&attr, CLOCK_MONOTONIC) == 0 &&
55 pthread_cond_init(cond, &attr) == 0;
56 pthread_condattr_destroy(&attr);
57 return ok;
58#endif
59}

64static void pool_deadline_after_ms(struct timespec *deadline, int timeout_ms) {
65 clock_gettime(pool_cond_clock(), deadline);
66 deadline->tv_sec += timeout_ms / 1000;
67 deadline->tv_nsec += (long)(timeout_ms % 1000) * 1000000L;
68 if (deadline->tv_nsec >= 1000000000L) {
69 ++deadline->tv_sec;
70 deadline->tv_nsec -= 1000000000L;
71 }
72}

93cwist_db_pool_t *cwist_db_pool_create(const char *path, size_t max_conns) {
94 if (!path || max_conns == 0) return NULL;
95 cwist_db_pool_t *pool = cwist_alloc(sizeof(*pool));
96 if (!pool) return NULL;
97 memset(pool, 0, sizeof(*pool));
98 pool->path = cwist_strdup(path);
99 pool->open_path = pool_open_path(path);
100 pool->conns = cwist_alloc_array(max_conns, sizeof(*pool->conns));
101 pool->idle_slots = cwist_alloc_array(max_conns, sizeof(*pool->idle_slots));
102 pool->leased = cwist_alloc_array(max_conns, sizeof(*pool->leased));
103 bool mutex_ready = false;
104 bool cond_ready = false;
105 if (!pool->path || !pool->open_path || !pool->conns || !pool->idle_slots || !pool->leased)
106 goto fail;
107 /* cwist_alloc_array does not zero the slot arrays. */
108 memset(pool->conns, 0, max_conns * sizeof(*pool->conns));
109 memset(pool->idle_slots, 0, max_conns * sizeof(*pool->idle_slots));
110 memset(pool->leased, 0, max_conns * sizeof(*pool->leased));
111 if (pthread_mutex_init(&pool->mtx, NULL) != 0) goto fail;
112 mutex_ready = true;
113 if (!pool_cond_init(&pool->cond)) goto fail;
114 cond_ready = true;
115 pool->max_conns = max_conns;
116 for (size_t i = 0; i < max_conns; ++i) {
117 /* cwist_db_open reports failures on the JSON channel, not INT16. */
118 cwist_error_t open_err = cwist_db_open(&pool->conns[i], pool->open_path);
119 if (!cwist_error_is_ok(&open_err)) {
120 cwist_error_dispose(&open_err);
121 goto fail;
122 }
123 pool->conns[i]->pool_slot = i;
124 sqlite3_busy_timeout(pool->conns[i]->conn, 5000);
125 pool->idle_slots[pool->idle_count++] = i;
126 }
127 return pool;
128fail:
129 if (pool->conns)
130 for (size_t i = 0; i < max_conns; ++i) cwist_db_close(pool->conns[i]);
131 if (cond_ready) pthread_cond_destroy(&pool->cond);
132 if (mutex_ready) pthread_mutex_destroy(&pool->mtx);
133 cwist_free(pool->leased);
134 cwist_free(pool->idle_slots);
135 cwist_free(pool->conns);
136 cwist_free(pool->open_path);
137 cwist_free(pool->path);
138 cwist_free(pool);
139 return NULL;
140}

145static cwist_db *pool_acquire_locked(cwist_db_pool_t *pool) {
146 if (pool->closing || pool->idle_count == 0) return NULL;
147 size_t slot = pool->idle_slots[--pool->idle_count];
148 pool->leased[slot] = true;

157cwist_db *cwist_db_pool_acquire(cwist_db_pool_t *pool) {
158 if (!pool) return NULL;
159 pthread_mutex_lock(&pool->mtx);
160 while (!pool->closing && pool->idle_count == 0) pthread_cond_wait(&pool->cond, &pool->mtx);
161 cwist_db *db = pool_acquire_locked(pool);
162 pthread_mutex_unlock(&pool->mtx);

171cwist_db *cwist_db_pool_acquire_timeout(cwist_db_pool_t *pool, int timeout_ms) {
172 if (!pool || timeout_ms < 0) return NULL;
173 struct timespec deadline;
174 pool_deadline_after_ms(&deadline, timeout_ms);
175 pthread_mutex_lock(&pool->mtx);
176 while (!pool->closing && pool->idle_count == 0) {
177 if (pthread_cond_timedwait(&pool->cond, &pool->mtx, &deadline) != 0) break;
178 }
179 cwist_db *db = pool_acquire_locked(pool);
180 pthread_mutex_unlock(&pool->mtx);
181 return db;
182}

189void cwist_db_pool_release(cwist_db_pool_t *pool, cwist_db *conn) {
190 if (!pool || !conn) return;
191 pthread_mutex_lock(&pool->mtx);
192 size_t slot = conn->pool_slot;
193 if (slot < pool->max_conns && pool->conns[slot] == conn && pool->leased[slot] &&
194 pool->idle_count < pool->max_conns) {
195 pool->leased[slot] = false;
196 pool->idle_slots[pool->idle_count++] = slot;
197 --pool->in_use;
198 pthread_cond_signal(&pool->cond);
199 }
200 pthread_mutex_unlock(&pool->mtx);
201}

222bool cwist_db_pool_destroy_timeout(cwist_db_pool_t *pool, int timeout_ms) {
223 if (!pool || timeout_ms < -1) return false;
224 pthread_mutex_lock(&pool->mtx);
225 pool->closing = true;
226 pthread_cond_broadcast(&pool->cond);
227 if (timeout_ms < 0) {
228 while (pool->in_use != 0) pthread_cond_wait(&pool->cond, &pool->mtx);
229 } else {
230 struct timespec deadline;
231 pool_deadline_after_ms(&deadline, timeout_ms);
232 while (pool->in_use != 0) {
233 int rc = pthread_cond_timedwait(&pool->cond, &pool->mtx, &deadline);
234 if (rc != 0 && pool->in_use != 0) {
235 pthread_mutex_unlock(&pool->mtx);
236 return false;
237 }
238 }
239 }
240 pthread_mutex_unlock(&pool->mtx);
241 for (size_t i = 0; i < pool->max_conns; ++i) cwist_db_close(pool->conns[i]);
242 pthread_cond_destroy(&pool->cond);
243 pthread_mutex_destroy(&pool->mtx);
244 cwist_free(pool->leased);
245 cwist_free(pool->idle_slots);
246 cwist_free(pool->conns);
247 cwist_free(pool->open_path);
248 cwist_free(pool->path);
249 cwist_free(pool);
250 return true;
251}
