101const char *cwist_flash_get(cwist_http_request *req, const char *key) {
102 if (!req || !key) return NULL;
103 flash_session_load(req, true);
104 if (!req->flash || !req->flash->buckets || req->flash->size == 0) return NULL;
105 uint64_t hash = siphash24(key, strlen(key), req->flash->seed);
106 size_t idx = hash % req->flash->size;
107
108 cwist_query_bucket **prev = &req->flash->buckets[idx];
109 cwist_query_bucket *curr = *prev;
110 while (curr) {
111 if (strcmp(curr->key, key) == 0) {
112 *prev = curr->next;
113 if (req->flash->count > 0) req->flash->count--;
114 cwist_free(curr->key);
115 /* ownership of value transfers to the caller, who must free it */
116 char *ret_val = curr->value;
117 cwist_free(curr);
118 return ret_val;
119 }
120 prev = &curr->next;
121 curr = curr->next;
122 }
123 return NULL;
124}
