186static bool is_void_element(const char *tag) {
187 static const char *const void_tags[] = {"area", "base", "br", "col", "embed", "hr", "img",
188 "input", "link", "meta", "source", "track", "wbr"};
189 for (size_t i = 0; i < sizeof(void_tags) / sizeof(void_tags[0]); i++) {
190 const char *a = tag, *b = void_tags[i];
191 while (*a && (*a | 0x20) == *b) {
192 a++;
193 b++;
194 }
195 if (*a == '\0' && *b == '\0') return true;
196 }
197 return false;
198}

218static bool append_raw_text(cwist_sstring *out, const char *text, const char *tag) {
219 size_t tag_len = strlen(tag);
220 const char *start = text;
221 for (const char *p = text; *p; p++) {
222 if (p[0] == '<' && p[1] == '/' && strncasecmp(p + 2, tag, tag_len) == 0) {
223 unsigned char next = (unsigned char)p[2 + tag_len];
224 bool end_tag = next == '>' || next == '/' || next == ' ' || next == '\t' ||
225 next == '\n' || next == '\r' || next == '\f';
226 if (!end_tag) continue;
227 if (!html_append_ok(cwist_sstring_append_len(out, start, (size_t)(p - start))) ||
228 !html_append_ok(cwist_sstring_append(out, "<\\/")))
229 return false;
230 start = p + 2;
231 p++;
232 }
233 }
234 return html_append_ok(cwist_sstring_append(out, start));
235}

248static bool render_element(cwist_html_element_t *el, cwist_sstring *out) {
249 if (!el || !out) return false;
250
251 if (el->tag && el->tag->data && *el->tag->data && html_tag_name_is_valid(el->tag->data)) {
252 if (!html_append_ok(cwist_sstring_append(out, "<")) ||
253 !html_append_ok(cwist_sstring_append(out, el->tag->data)))
254 return false;
255 cJSON *attr = NULL;
256 cJSON_ArrayForEach(attr, el->attributes) {
257 if (!attr || !html_attr_name_is_valid(attr->string)) continue;
258 if (cJSON_IsString(attr) && attr->valuestring) {
259 if (!html_append_ok(cwist_sstring_append(out, " ")) ||
260 !html_append_ok(cwist_sstring_append(out, attr->string)) ||
261 !html_append_ok(cwist_sstring_append(out, "=\"")) ||
262 !html_append_ok(cwist_sstring_append_escaped(out, attr->valuestring)) ||
263 !html_append_ok(cwist_sstring_append(out, "\"")))
264 return false;
265 } else if (cJSON_IsTrue(attr)) {
266 if (!html_append_ok(cwist_sstring_append(out, " ")) ||
267 !html_append_ok(cwist_sstring_append(out, attr->string)))
268 return false;
269 } else if (cJSON_IsNumber(attr)) {
270 char num_buf[64];
271 cwist_number_format_double(num_buf, sizeof(num_buf), attr->valuedouble);
272 if (!html_append_ok(cwist_sstring_append(out, " ")) ||
273 !html_append_ok(cwist_sstring_append(out, attr->string)) ||
274 !html_append_ok(cwist_sstring_append(out, "=\"")) ||
275 !html_append_ok(cwist_sstring_append(out, num_buf)) ||
276 !html_append_ok(cwist_sstring_append(out, "\"")))
277 return false;
278 }
279 }
280 if (!html_append_ok(cwist_sstring_append(out, ">"))) return false;
281
282 /* A void element cannot have content; an end tag like </br> would be
283 * parsed by browsers as a second element. */
284 if (is_void_element(el->tag->data)) return true;
285
286 if (el->inner_text && el->inner_text->data) {
287 if (is_raw_text_element(el->tag->data)) {
288 if (!append_raw_text(out, el->inner_text->data, el->tag->data)) return false;
289 } else if (!html_append_ok(cwist_sstring_append_escaped(out, el->inner_text->data))) {
290 return false;
291 }
292 }
293
294 if (el->children) {
295 for (int i = 0; i < el->child_count; i++) {
296 if (!render_element(el->children[i], out)) return false;
297 }
298 }
299
300 if (!html_append_ok(cwist_sstring_append(out, "</")) ||
301 !html_append_ok(cwist_sstring_append(out, el->tag->data)) ||
302 !html_append_ok(cwist_sstring_append(out, ">")))
303 return false;
304 } else {
305 /* A tagless element is a fragment: its text, then its children
306 * (#964). */
307 if (el->inner_text && el->inner_text->data &&
308 !html_append_ok(cwist_sstring_append_escaped(out, el->inner_text->data)))
309 return false;
310 if (el->children) {
311 for (int i = 0; i < el->child_count; i++) {
312 if (!render_element(el->children[i], out)) return false;
313 }
314 }
315 }
316 return true;
317}
