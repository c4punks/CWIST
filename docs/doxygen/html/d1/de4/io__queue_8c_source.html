57static job_node_t *cwist_job_node_create(cwist_job_func func, void *arg) {
58 job_node_t *node = cwist_alloc(sizeof(*node));

107static void cwist_queue_reclaim(cwist_io_queue *q) {
108 ttak_epoch_reclaim();
109 unsigned int epoch = atomic_load_explicit(&g_epoch_mgr.global_epoch, memory_order_acquire);
110
111 pthread_mutex_lock(&q->retire_lock);
112 if (epoch != q->last_epoch) {
113 q->last_epoch = epoch;
114 job_node_t *old = q->retire_old;
115 q->retire_old = q->retire_new;
116 q->retire_new = NULL;
117 pthread_mutex_unlock(&q->retire_lock);
118
119 while (old) {
120 job_node_t *next = old->retire_next;
121 ttak_epoch_retire(old, cwist_job_node_free_cb);
122 old = next;
123 }
124 } else {
125 pthread_mutex_unlock(&q->retire_lock);
126 }

134static void cwist_queue_push(cwist_io_queue *q, job_node_t *node) {
135 atomic_store_explicit(&node->next, NULL, memory_order_relaxed);
136 ttak_epoch_enter();
137 while (1) {
138 job_node_t *tail = atomic_load_explicit(&q->tail, memory_order_acquire);
139 job_node_t *next = atomic_load_explicit(&tail->next, memory_order_acquire);
140 if (tail == atomic_load_explicit(&q->tail, memory_order_acquire)) {
141 if (!next) {
142 if (atomic_compare_exchange_weak_explicit(
143 &tail->next, &next, node, memory_order_release, memory_order_relaxed)) {
144 atomic_compare_exchange_strong_explicit(
145 &q->tail, &tail, node, memory_order_release, memory_order_relaxed);
146 break;
147 }
148 } else {
149 atomic_compare_exchange_weak_explicit(&q->tail, &tail, next, memory_order_release,
150 memory_order_relaxed);
151 }
152 }
153 }
154 ttak_epoch_exit();
155
156 size_t prev = atomic_fetch_add_explicit(&q->pending_jobs, 1, memory_order_release);
157 if (prev == 0) {
158 pthread_mutex_lock(&q->sleep_lock);
159 pthread_cond_signal(&q->sleep_cond);
160 pthread_mutex_unlock(&q->sleep_lock);
161 }

175static bool cwist_queue_pop(cwist_io_queue *q, cwist_job_func *func, void **arg) {
176 bool popped = false;
177 ttak_epoch_enter();
178 while (1) {
179 job_node_t *head = atomic_load_explicit(&q->head, memory_order_acquire);
180 job_node_t *tail = atomic_load_explicit(&q->tail, memory_order_acquire);
181 job_node_t *next = atomic_load_explicit(&head->next, memory_order_acquire);
182 if (!next) {
183 break;
184 }
185 if (head == tail) {
186 atomic_compare_exchange_weak_explicit(&q->tail, &tail, next, memory_order_release,
187 memory_order_relaxed);
188 continue;
189 }
190 if (atomic_compare_exchange_weak_explicit(&q->head, &head, next, memory_order_acq_rel,
191 memory_order_relaxed)) {
192 /* head is now unreachable from the queue, but a preempted
193 * producer may still hold it; defer the free across an epoch
194 * boundary instead of releasing it here. */
195 cwist_queue_defer_free(q, head);
196 *func = next->func;
197 *arg = next->arg;
198 popped = true;
199 break;
200 }
201 }
202 ttak_epoch_exit();
203 return popped;
204}

211static bool cwist_queue_wait(cwist_io_queue *q) {
212 pthread_mutex_lock(&q->sleep_lock);
213 while (atomic_load_explicit(&q->pending_jobs, memory_order_acquire) == 0 &&
214 atomic_load_explicit(&q->running, memory_order_acquire)) {
215 pthread_cond_wait(&q->sleep_cond, &q->sleep_lock);
216 }
217 bool should_continue = atomic_load_explicit(&q->running, memory_order_acquire) ||
218 atomic_load_explicit(&q->pending_jobs, memory_order_acquire) > 0;
219 pthread_mutex_unlock(&q->sleep_lock);

228cwist_io_queue *cwist_io_queue_create(size_t capacity) {
229 (void)capacity;
230 cwist_io_queue *q = cwist_alloc(sizeof(*q));
231 if (!q) return NULL;
232 /* A returned queue can outlive its creator; destroy owns its shell. */
233 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(q);
234
235 job_node_t *stub = cwist_job_node_create(NULL, NULL);
236 if (!stub) {
237 cwist_free(q);
238 return NULL;
239 }
240
241 atomic_store_explicit(&q->head, stub, memory_order_relaxed);
242 atomic_store_explicit(&q->tail, stub, memory_order_relaxed);
243 atomic_init(&q->pending_jobs, 0);
244 atomic_store_explicit(&q->running, true, memory_order_release);
245 q->retire_new = NULL;
246 q->retire_old = NULL;
247 q->last_epoch = atomic_load_explicit(&g_epoch_mgr.global_epoch, memory_order_acquire);
248
249 if (pthread_mutex_init(&q->retire_lock, NULL) != 0) {
250 cwist_job_node_destroy(stub);
251 cwist_free(q);
252 return NULL;
253 }
254 if (pthread_mutex_init(&q->sleep_lock, NULL) != 0) {
255 pthread_mutex_destroy(&q->retire_lock);
256 cwist_job_node_destroy(stub);
257 cwist_free(q);
258 return NULL;
259 }
260 if (pthread_cond_init(&q->sleep_cond, NULL) != 0) {
261 pthread_mutex_destroy(&q->sleep_lock);
262 pthread_mutex_destroy(&q->retire_lock);
263 cwist_job_node_destroy(stub);
264 cwist_free(q);
265 return NULL;
266 }
267 return q;

293void cwist_io_queue_run(cwist_io_queue *q) {
294 if (!q) return;
295 unsigned int since_reclaim = 0;
296 while (atomic_load_explicit(&q->running, memory_order_acquire) ||
297 atomic_load_explicit(&q->pending_jobs, memory_order_acquire) > 0) {
298 cwist_job_func func = NULL;
299 void *arg = NULL;
300 if (!cwist_queue_pop(q, &func, &arg)) {
301 cwist_queue_reclaim(q);
302 since_reclaim = 0;
303 if (!cwist_queue_wait(q)) {
304 break;
305 }
306 continue;
307 }
308
309 atomic_fetch_sub_explicit(&q->pending_jobs, 1, memory_order_acq_rel);
310 if (func) func(arg);
311
312 // Full-GC mode: catch whatever this job forgot to cwist_free()
313 // right away instead of letting it pile up for the lifetime of this
314 // (long-lived, pooled) worker thread. The existing reclaim cadence
315 // below actually frees it.
316 if (cwist_full_gc_enabled()) {
317 cwist_gc_scope_flush();
318 }
319
320 if (++since_reclaim >= 64) {
321 since_reclaim = 0;
322 cwist_queue_reclaim(q);
323 }
324 }

343void cwist_io_queue_destroy(cwist_io_queue *q) {
344 if (!q) return;
345 atomic_store_explicit(&q->running, false, memory_order_release);
346 pthread_mutex_lock(&q->sleep_lock);
347 pthread_cond_broadcast(&q->sleep_cond);
348 pthread_mutex_unlock(&q->sleep_lock);
349
350 job_node_t *node = atomic_load_explicit(&q->head, memory_order_acquire);
351 while (node) {
352 job_node_t *next = atomic_load_explicit(&node->next, memory_order_acquire);
353 cwist_job_node_destroy(node);
354 node = next;
355 }
356
357 /* All runners are gone by contract, so staged nodes can be freed
358 * directly without an epoch grace period. */
359 for (int stage = 0; stage < 2; stage++) {
360 job_node_t *list = stage == 0 ? q->retire_new : q->retire_old;
361 while (list) {
362 job_node_t *next = list->retire_next;
363 cwist_job_node_destroy(list);
364 list = next;
365 }
366 }
367
368 pthread_cond_destroy(&q->sleep_cond);
369 pthread_mutex_destroy(&q->sleep_lock);
370 pthread_mutex_destroy(&q->retire_lock);
371 cwist_free(q);
372}
