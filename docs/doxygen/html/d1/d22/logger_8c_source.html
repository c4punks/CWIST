78static void logger_vlog(cwist_logger *log, cwist_log_level_t level, const char *fmt, va_list args) {
79 if (!log || level < log->level || (unsigned)level > CWIST_LOG_ERROR) return;
80 FILE *out = level_file(level);
81 /* localtime_r() takes glibc's timezone lock on every call, which is the
82 * most expensive part of a log line once many threads log at once. The
83 * stamp only changes once a second, so each thread keeps its last one. */
84 static _Thread_local time_t cached_sec = (time_t)-1;
85 static _Thread_local char time_buf[32];
86 time_t now = time(NULL);
87 if (now != cached_sec) {
88 struct tm tm_info;
89 strftime(time_buf, sizeof(time_buf), "%Y-%m-%d %H:%M:%S", localtime_r(&now, &tm_info));
90 cached_sec = now;
91 }

93 /* Format the whole line first, so it is written with one call (or queued
94 * for the async writer) instead of a prefix, a message and a newline. */
95 char stack_buf[1024];
96 char *line = stack_buf;
97 const char *color = log->use_colors ? level_colors[level] : "";
98 const char *reset = log->use_colors ? color_reset : "";
99 /* With colors off, color and reset are "" and the line reads
100 * "[time] [LEVEL] name: ", as before. */
101 int prefix = snprintf(NULL, 0, "%s[%s]%s [%s] %s: ", color, time_buf, reset,
102 level_strings[level], log->name);
103 va_list measure;
104 va_copy(measure, args);
105 int body = vsnprintf(NULL, 0, fmt, measure);
106 va_end(measure);
107 if (prefix < 0 || body < 0) return;
108 size_t len = (size_t)prefix + (size_t)body + 1; /* + newline */
109 if (len + 1 > sizeof(stack_buf)) {
110 line = cwist_alloc(len + 1);
111 if (!line) return;
112 }
113 snprintf(line, len + 1, "%s[%s]%s [%s] %s: ", color, time_buf, reset, level_strings[level],
114 log->name);
115 vsnprintf(line + prefix, len + 1 - (size_t)prefix, fmt, args);
116 line[len - 1] = '\n';
117 line[len] = '\0';
118
119 if (!cwist_log_async_submit(fileno(out), line, len)) {
120 /* Sync fallback: no per-line fflush. stdio batches lines in its
121 * buffer (block-buffered on pipes and files, line-buffered on a TTY),
122 * so delivery is not immediate. Call cwist_log_async_start() for
123 * immediate per-line delivery. */
124 fwrite(line, 1, len, out);
125 }
126 if (line != stack_buf) cwist_free(line);
127}
