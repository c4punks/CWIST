50static void cwist_http2_flow_control_adjust(cwist_http2_flow_control *flow_control) {
51 uint64_t delay_us;
52 uint64_t multiplier;
53 uint64_t rate_floor;
54
55 if (!flow_control->rtt_initialized) {
56 return;
57 }
58
59 delay_us = flow_control->srtt_us + (4U * flow_control->rttvar_us);
60 if (flow_control->rttvar_us > flow_control->srtt_us / 2U) {
61 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_UNSTABLE;
62 } else if (delay_us >= 200000U) {
63 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_HIGH_RTT;
64 } else {
65 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_NORMAL;
66 }
67
68 multiplier = delay_us / 200000U;
69 if (multiplier < 1U) {
70 multiplier = 1U;
71 }
72 if (flow_control->network_quality != CWIST_HTTP2_NETWORK_QUALITY_NORMAL) {
73 ++multiplier;
74 }
75
76 flow_control->target_window =
77 cwist_http2_window_clamp((uint64_t)flow_control->min_window * multiplier,
78 flow_control->min_window, flow_control->max_window);
79 flow_control->pacing_rate_bytes_per_sec = ((uint64_t)flow_control->target_window * 1000000U) /
80 (flow_control->srtt_us ? flow_control->srtt_us : 1U);
81 /* Floor: a single spiked RTT sample must not throttle the connection to
82 * a crawl. Overrating is harmless — the peer's flow-control windows
83 * still gate in-flight bytes; pacing only smooths bursts. */
84 rate_floor = (uint64_t)flow_control->min_window * 10U;
85 if (flow_control->pacing_rate_bytes_per_sec < rate_floor) {
86 flow_control->pacing_rate_bytes_per_sec = rate_floor;
87 }
88}

100static int cwist_http2_write_window_update(uint8_t *buffer, size_t buffer_len, size_t *written,
101 uint32_t stream_id, uint32_t increment) {
102 if (written != NULL) {
103 *written = 0;
104 }
105 /* A WINDOW_UPDATE frame is 9 bytes of header + 4 bytes of payload;
106 * accepting buffer_len < 13 used to write 4 bytes past the caller's
107 * buffer. */
108 if (buffer == NULL || written == NULL || buffer_len < 13U || increment == 0U) {
109 return -1;
110 }
111
112 h2_put_u24(buffer, 4);
113 buffer[3] = 0x08; /* WINDOW_UPDATE */
114 buffer[4] = 0;
115 h2_put_u31(buffer + 5, stream_id);
116 h2_put_u32(buffer + 9, increment);
117 *written = 13U;
118 return 0;
119}

219void cwist_http2_flow_control_update_rtt(cwist_http2_flow_control *flow_control,
220 uint64_t rtt_sample_us) {
221 uint64_t difference;
222
223 if (flow_control == NULL || rtt_sample_us == 0U) {
224 return;
225 }
226 if (!flow_control->rtt_initialized) {
227 flow_control->srtt_us = rtt_sample_us;
228 flow_control->rttvar_us = rtt_sample_us / 2U;
229 flow_control->rtt_initialized = true;
230 } else {
231 difference = cwist_http2_abs_diff(flow_control->srtt_us, rtt_sample_us);
232 flow_control->rttvar_us = (3U * flow_control->rttvar_us + difference) / 4U;
233 flow_control->srtt_us = (7U * flow_control->srtt_us + rtt_sample_us) / 8U;
234 }
235 cwist_http2_flow_control_adjust(flow_control);
236}

324static uint32_t cwist_http2_flow_control_retune_target(uint32_t pending_update, uint64_t srtt_us,
325 uint64_t interval_us,
326 uint32_t current_target, uint32_t minimum,
327 uint32_t maximum) {
328 uint64_t bdp2;
329 uint64_t grow_cap;
330 uint64_t shrink_floor;
331 uint32_t target;
332
333 if (interval_us == 0) {
334 return current_target;
335 }
336 bdp2 = (2ULL * (uint64_t)pending_update * srtt_us) / interval_us;
337 target = cwist_http2_window_clamp(bdp2, minimum, maximum);
338 grow_cap = (uint64_t)current_target * 2U;
339 shrink_floor = current_target / 2U;
340 if ((uint64_t)target > grow_cap) {
341 target = cwist_http2_window_clamp(grow_cap, minimum, maximum);
342 } else if ((uint64_t)target < shrink_floor) {
343 target = cwist_http2_window_clamp(shrink_floor, minimum, maximum);
344 }
345 return target;
346}

385int cwist_http2_flow_control_maybe_window_update(cwist_http2_flow_control *flow_control,
386 uint8_t *buffer, size_t buffer_len,
387 size_t *written, uint64_t now_us, bool force) {
388 uint32_t increment;
389
390 if (written != NULL) {
391 *written = 0;
392 }
393 if (flow_control == NULL || flow_control->pending_update == 0U ||
394 (!force && flow_control->pending_update < flow_control->target_window / 2U)) {
395 return 0;
396 }
397
398 if (flow_control->rtt_initialized && flow_control->last_window_update_us != 0U &&

400 flow_control->target_window = cwist_http2_flow_control_retune_target(
401 flow_control->pending_update, flow_control->srtt_us,

409 return -1;
410 }
411 flow_control->receive_window = increment > CWIST_HTTP2_MAX_WINDOW - flow_control->receive_window
412 ? CWIST_HTTP2_MAX_WINDOW
413 : flow_control->receive_window + increment;
414 flow_control->pending_update = 0;
415 flow_control->last_window_update_us = now_us;

541size_t cwist_http2_flow_control_pacing_allowance(
542 cwist_http2_flow_control *connection_flow_control,
543 const cwist_http2_stream_flow_control *stream_flow_control, size_t requested, uint64_t now_us) {
544 uint64_t elapsed;
545 uint64_t added;
546 uint64_t fill_us;
547 size_t allowed;
548
549 if (connection_flow_control == NULL || stream_flow_control == NULL) {
550 return 0;
551 }
552 if (connection_flow_control->pacing_last_us == 0U) {
553 connection_flow_control->pacing_last_us = now_us;
554 } else if (now_us > connection_flow_control->pacing_last_us) {
555 elapsed = now_us - connection_flow_control->pacing_last_us;
556 /* The bucket tops out at target_window, so accruing beyond one
557 * window's worth of idle time buys nothing. Capping elapsed at the
558 * exact fill time keeps the multiplication inside uint64 no matter
559 * how long the connection idled or how large the rate is. */
560 fill_us = ((uint64_t)connection_flow_control->target_window * 1000000U) /
561 (connection_flow_control->pacing_rate_bytes_per_sec
562 ? connection_flow_control->pacing_rate_bytes_per_sec
563 : 1U) +
564 1U;
565 if (elapsed > fill_us) {
566 elapsed = fill_us;
567 }
568 added = (elapsed * connection_flow_control->pacing_rate_bytes_per_sec) / 1000000U;
569 connection_flow_control->pacing_tokens += added;
570 if (connection_flow_control->pacing_tokens > connection_flow_control->target_window) {
571 connection_flow_control->pacing_tokens = connection_flow_control->target_window;
572 }
573 connection_flow_control->pacing_last_us = now_us;
574 }
575
576 allowed = requested;
577 if (allowed > connection_flow_control->send_window) {
578 allowed = connection_flow_control->send_window;
579 }
580 if (allowed > stream_flow_control->send_window) {
581 allowed = stream_flow_control->send_window;
582 }
583 if (allowed > connection_flow_control->pacing_tokens) {
584 allowed = (size_t)connection_flow_control->pacing_tokens;
585 }
586 return allowed;
587}
