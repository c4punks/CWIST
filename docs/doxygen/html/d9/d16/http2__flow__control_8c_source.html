49static void cwist_http2_flow_control_adjust(cwist_http2_flow_control *flow_control) {
50 uint64_t delay_us;
51 uint64_t multiplier;
52 uint64_t rate_floor;
53
54 if (!flow_control->rtt_initialized) {
55 return;
56 }
57
58 delay_us = flow_control->srtt_us + (4U * flow_control->rttvar_us);
59 if (flow_control->rttvar_us > flow_control->srtt_us / 2U) {
60 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_UNSTABLE;
61 } else if (delay_us >= 200000U) {
62 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_HIGH_RTT;
63 } else {
64 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_NORMAL;
65 }
66
67 multiplier = delay_us / 200000U;
68 if (multiplier < 1U) {
69 multiplier = 1U;
70 }
71 if (flow_control->network_quality != CWIST_HTTP2_NETWORK_QUALITY_NORMAL) {
72 ++multiplier;
73 }
74
75 flow_control->target_window =
76 cwist_http2_window_clamp((uint64_t)flow_control->min_window * multiplier,
77 flow_control->min_window, flow_control->max_window);
78 flow_control->pacing_rate_bytes_per_sec = ((uint64_t)flow_control->target_window * 1000000U) /
79 (flow_control->srtt_us ? flow_control->srtt_us : 1U);
80 /* Floor: a single spiked RTT sample must not throttle the connection to
81 * a crawl. Overrating is harmless — the peer's flow-control windows
82 * still gate in-flight bytes; pacing only smooths bursts. */
83 rate_floor = (uint64_t)flow_control->min_window * 10U;
84 if (flow_control->pacing_rate_bytes_per_sec < rate_floor) {
85 flow_control->pacing_rate_bytes_per_sec = rate_floor;
86 }
87}

99static int cwist_http2_write_window_update(uint8_t *buffer, size_t buffer_len, size_t *written,
100 uint32_t stream_id, uint32_t increment) {
101 if (written != NULL) {
102 *written = 0;
103 }
104 /* A WINDOW_UPDATE frame is 9 bytes of header + 4 bytes of payload;
105 * accepting buffer_len < 13 used to write 4 bytes past the caller's
106 * buffer. */
107 if (buffer == NULL || written == NULL || buffer_len < 13U || increment == 0U) {
108 return -1;
109 }
110
111 buffer[0] = 0;
112 buffer[1] = 0;
113 buffer[2] = 4;
114 buffer[3] = 0x08;
115 buffer[4] = 0;
116 buffer[5] = (uint8_t)((stream_id >> 24) & 0x7fU);
117 buffer[6] = (uint8_t)(stream_id >> 16);
118 buffer[7] = (uint8_t)(stream_id >> 8);
119 buffer[8] = (uint8_t)stream_id;
120 buffer[9] = (uint8_t)(increment >> 24);
121 buffer[10] = (uint8_t)(increment >> 16);
122 buffer[11] = (uint8_t)(increment >> 8);
123 buffer[12] = (uint8_t)increment;
124 *written = 13U;
125 return 0;
126}

226void cwist_http2_flow_control_update_rtt(cwist_http2_flow_control *flow_control,
227 uint64_t rtt_sample_us) {
228 uint64_t difference;
229
230 if (flow_control == NULL || rtt_sample_us == 0U) {
231 return;
232 }
233 if (!flow_control->rtt_initialized) {
234 flow_control->srtt_us = rtt_sample_us;
235 flow_control->rttvar_us = rtt_sample_us / 2U;
236 flow_control->rtt_initialized = true;
237 } else {
238 difference = cwist_http2_abs_diff(flow_control->srtt_us, rtt_sample_us);
239 flow_control->rttvar_us = (3U * flow_control->rttvar_us + difference) / 4U;
240 flow_control->srtt_us = (7U * flow_control->srtt_us + rtt_sample_us) / 8U;
241 }
242 cwist_http2_flow_control_adjust(flow_control);
243}

331static uint32_t cwist_http2_flow_control_retune_target(uint32_t pending_update, uint64_t srtt_us,
332 uint64_t interval_us,
333 uint32_t current_target, uint32_t minimum,
334 uint32_t maximum) {
335 uint64_t bdp2;
336 uint64_t grow_cap;
337 uint64_t shrink_floor;
338 uint32_t target;
339
340 if (interval_us == 0) {
341 return current_target;
342 }
343 bdp2 = (2ULL * (uint64_t)pending_update * srtt_us) / interval_us;
344 target = cwist_http2_window_clamp(bdp2, minimum, maximum);
345 grow_cap = (uint64_t)current_target * 2U;
346 shrink_floor = current_target / 2U;
347 if ((uint64_t)target > grow_cap) {
348 target = cwist_http2_window_clamp(grow_cap, minimum, maximum);
349 } else if ((uint64_t)target < shrink_floor) {
350 target = cwist_http2_window_clamp(shrink_floor, minimum, maximum);
351 }
352 return target;
353}

392int cwist_http2_flow_control_maybe_window_update(cwist_http2_flow_control *flow_control,
393 uint8_t *buffer, size_t buffer_len,
394 size_t *written, uint64_t now_us, bool force) {
395 uint32_t increment;
396
397 if (written != NULL) {
398 *written = 0;
399 }
400 if (flow_control == NULL || flow_control->pending_update == 0U ||
401 (!force && flow_control->pending_update < flow_control->target_window / 2U)) {
402 return 0;
403 }
404

406 now_us > flow_control->last_window_update_us) {
407 flow_control->target_window = cwist_http2_flow_control_retune_target(
408 flow_control->pending_update, flow_control->srtt_us,

548size_t cwist_http2_flow_control_pacing_allowance(
549 cwist_http2_flow_control *connection_flow_control,
550 const cwist_http2_stream_flow_control *stream_flow_control, size_t requested, uint64_t now_us) {
551 uint64_t elapsed;
552 uint64_t added;
553 uint64_t fill_us;
554 size_t allowed;
555
556 if (connection_flow_control == NULL || stream_flow_control == NULL) {
557 return 0;
558 }
559 if (connection_flow_control->pacing_last_us == 0U) {
560 connection_flow_control->pacing_last_us = now_us;
561 } else if (now_us > connection_flow_control->pacing_last_us) {
562 elapsed = now_us - connection_flow_control->pacing_last_us;
563 /* The bucket tops out at target_window, so accruing beyond one
564 * window's worth of idle time buys nothing. Capping elapsed at the
565 * exact fill time keeps the multiplication inside uint64 no matter
566 * how long the connection idled or how large the rate is. */
567 fill_us = ((uint64_t)connection_flow_control->target_window * 1000000U) /
568 (connection_flow_control->pacing_rate_bytes_per_sec
569 ? connection_flow_control->pacing_rate_bytes_per_sec
570 : 1U) +
571 1U;
572 if (elapsed > fill_us) {
573 elapsed = fill_us;
574 }
575 added = (elapsed * connection_flow_control->pacing_rate_bytes_per_sec) / 1000000U;
576 connection_flow_control->pacing_tokens += added;
577 if (connection_flow_control->pacing_tokens > connection_flow_control->target_window) {
578 connection_flow_control->pacing_tokens = connection_flow_control->target_window;
579 }
580 connection_flow_control->pacing_last_us = now_us;
581 }
582
583 allowed = requested;
584 if (allowed > connection_flow_control->send_window) {
585 allowed = connection_flow_control->send_window;
586 }
587 if (allowed > stream_flow_control->send_window) {
588 allowed = stream_flow_control->send_window;
589 }
590 if (allowed > connection_flow_control->pacing_tokens) {
591 allowed = (size_t)connection_flow_control->pacing_tokens;
592 }
593 return allowed;
594}
