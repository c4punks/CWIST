163int cwist_grpc_parse_timeout(const char *value, uint64_t *out_ms) {
164 if (!value || !out_ms) return -1;
165 size_t len = strlen(value);
166 if (len < 2 || len > 9) return -1; /* up to 8 digits + unit */
167 char unit = value[len - 1];
168 uint64_t to_ms_num = 1, to_ms_den = 1;
169 switch (unit) {
170 case 'H': to_ms_num = 3600000; break;
171 case 'M': to_ms_num = 60000; break;
172 case 'S': to_ms_num = 1000; break;
173 case 'm': to_ms_num = 1; break;
174 case 'u':
175 to_ms_num = 1;
176 to_ms_den = 1000;
177 break;
178 case 'n':
179 to_ms_num = 1;
180 to_ms_den = 1000000;
181 break;
182 default: return -1;
183 }
184 uint64_t v = 0;
185 for (size_t i = 0; i + 1 < len; i++) {
186 if (!isdigit((unsigned char)value[i])) return -1;
187 v = v * 10 + (uint64_t)(value[i] - '0');
188 }
189 /* Round sub-millisecond units up so a positive timeout never becomes 0. */
190 uint64_t ms = (v * to_ms_num + to_ms_den - 1) / to_ms_den;
191 if (v > 0 && ms == 0) ms = 1;
192 *out_ms = ms;
193 return 0;
194}

238 z_stream zs;
239 memset(&zs, 0, sizeof(zs));
240 if (deflateInit2(&zs, Z_DEFAULT_COMPRESSION, Z_DEFLATED, 16 + MAX_WBITS, 8,
241 Z_DEFAULT_STRATEGY) != Z_OK)
242 return -1;

244 if (cap < 64) cap = 64;

248 return -1;

251 zs.avail_in = (uInt)in_len;
252 zs.next_out = buf;
253 zs.avail_out = (uInt)cap;

255 if (zrc != Z_STREAM_END) {
256 deflateEnd(&zs);
257 cwist_free(buf);

259 }
260 *out_len = cap - zs.avail_out;
261 deflateEnd(&zs);
262 *out = buf;
263 return 0;

278static int grpc_gzip_inflate(const uint8_t *in, size_t in_len, uint8_t **out, size_t *out_len) {
279 if (in_len > UINT_MAX) return -1;
280 z_stream zs;
281 memset(&zs, 0, sizeof(zs));
282 if (inflateInit2(&zs, 16 + MAX_WBITS) != Z_OK) return -1;
283 const size_t max_cap = (size_t)CWIST_HTTP_MAX_BODY_SIZE + 1;
284 size_t cap = 8192;
285 if (in_len <= (size_t)CWIST_HTTP_MAX_BODY_SIZE) {
286 size_t suggested = in_len * 3 + 1024;
287 if (suggested > cap) cap = suggested;
288 }
289 if (cap > max_cap) cap = max_cap;
290 uint8_t *buf = (uint8_t *)cwist_alloc(cap);
291 if (!buf) {
292 inflateEnd(&zs);
293 return -1;
294 }
295 zs.next_in = (Bytef *)in;
296 zs.avail_in = (uInt)in_len;
297 size_t used = 0;
298 int zrc = Z_OK;
299 while (zrc == Z_OK) {
300 if (used == cap) {
301 size_t next_cap = cap <= max_cap / 2 ? cap * 2 : max_cap;
302 if (next_cap > max_cap) next_cap = max_cap;
303 uint8_t *nb = (uint8_t *)cwist_realloc(buf, next_cap);
304 if (!nb) {
305 cwist_free(buf);
306 inflateEnd(&zs);
307 return -1;
308 }
309 buf = nb;
310 cap = next_cap;
311 }
312 zs.next_out = buf + used;
313 zs.avail_out = (uInt)(cap - used);
314 zrc = inflate(&zs, Z_NO_FLUSH);
315 used = cap - zs.avail_out;
316 if (used > (size_t)CWIST_HTTP_MAX_BODY_SIZE) {
317 cwist_free(buf);
318 inflateEnd(&zs);
319 return -2;
320 }
321 }
322 inflateEnd(&zs);
323 if (zrc != Z_STREAM_END) {
324 cwist_free(buf);
325 return -1;
326 }
327 *out = buf;
328 *out_len = used;
329 return 0;
330}

379 size_t *out_len) {

382 size_t used = 0;

668 const uint8_t *p = (const uint8_t *)frame;

671 if ((size_t)len > frame_len - 5) return -1;
672 if ((size_t)len != frame_len - 5) return -1;
673
674 out->compressed = p[0];
675 out->data = p + 5;
676 out->len = (size_t)len;
677 return 0;

721int cwist_grpc_encode_message(const void *payload, size_t payload_len, uint8_t compressed,
722 uint8_t **out, size_t *out_len) {
723 if (!out || !out_len || (payload_len > 0 && !payload)) return -1;
724 if (payload_len > UINT32_MAX) return -1;
725
726 uint8_t *buf = (uint8_t *)cwist_alloc(payload_len + 5);
727 if (!buf) return -1;
728 buf[0] = compressed ? 1 : 0;
729 buf[1] = (uint8_t)((payload_len >> 24) & 0xff);
730 buf[2] = (uint8_t)((payload_len >> 16) & 0xff);
731 buf[3] = (uint8_t)((payload_len >> 8) & 0xff);
732 buf[4] = (uint8_t)(payload_len & 0xff);
733 if (payload_len > 0) memcpy(buf + 5, payload, payload_len);
734
735 *out = buf;

778int cwist_grpc_decoder_feed(cwist_grpc_decoder *decoder, const void *data, size_t len,
779 cwist_grpc_message_callback callback, void *ctx) {
780 if (!decoder || (!data && len) || !callback) return -1;
781 const uint8_t *input = data;
782 while (len) {
783 if (decoder->header_len < sizeof(decoder->header)) {
784 size_t take = sizeof(decoder->header) - decoder->header_len;
785 if (take > len) take = len;
786 memcpy(decoder->header + decoder->header_len, input, take);
787 decoder->header_len += take;
788 input += take;
789 len -= take;
790 if (decoder->header_len < sizeof(decoder->header)) continue;
791 decoder->compressed = decoder->header[0];
792 decoder->payload_len = ((size_t)decoder->header[1] << 24) |
793 ((size_t)decoder->header[2] << 16) |
794 ((size_t)decoder->header[3] << 8) | (size_t)decoder->header[4];
795 if (decoder->compressed > 1 || decoder->payload_len > decoder->max_message_size)
796 return -1;
797 if (decoder->payload_len) {
798 decoder->payload = cwist_alloc(decoder->payload_len);
799 if (!decoder->payload) return -1;
800 }
801 }
802 size_t take = decoder->payload_len - decoder->payload_used;
803 if (take > len) take = len;
804 if (take) memcpy(decoder->payload + decoder->payload_used, input, take);
805 decoder->payload_used += take;
806 input += take;
807 len -= take;
808 if (decoder->payload_used != decoder->payload_len) continue;
809 cwist_grpc_message message = {decoder->compressed, decoder->payload, decoder->payload_len};
810 if (callback(ctx, &message) != 0) return -1;
811 cwist_free(decoder->payload);
812 decoder->payload = NULL;
813 decoder->payload_len = decoder->payload_used = decoder->header_len = 0;
814 }
815 return 0;
816}

880int cwist_grpc_stream_send(cwist_grpc_stream *stream, const void *payload, size_t payload_len) {
881 if (!stream) return -1;
882 int use_gzip = 0;
883 if (stream->session) {
884 use_gzip = (((cwist_grpc_session *)stream->session)->resp_encoding == 1);
885 } else if (stream->req) {
886 use_gzip = grpc_client_accepts_gzip(stream->req);
887 }
888
889 uint8_t *frame = NULL;
890 size_t frame_len = 0;
891 if (use_gzip && payload_len > 0) {
892 uint8_t *cbuf = NULL;
893 size_t clen = 0;
894 if (grpc_gzip_deflate(payload, payload_len, &cbuf, &clen) == 0) {
895 int enc_rc = cwist_grpc_encode_message(cbuf, clen, 1, &frame, &frame_len);
896 cwist_free(cbuf);
897 if (enc_rc != 0) {
898 stream->status = CWIST_GRPC_INTERNAL;
899 stream->status_message = "failed to encode gRPC stream message";
900 return -1;
901 }
902 } else {
903 if (cwist_grpc_encode_message(payload, payload_len, 0, &frame, &frame_len) != 0) {
904 stream->status = CWIST_GRPC_INTERNAL;
905 stream->status_message = "failed to encode gRPC stream message";
906 return -1;
907 }
908 }
909 } else {
910 if (cwist_grpc_encode_message(payload, payload_len, 0, &frame, &frame_len) != 0) {
911 stream->status = CWIST_GRPC_INTERNAL;
912 stream->status_message = "failed to encode gRPC stream message";
913 return -1;
914 }
915 }
916 if (stream->session) {
917 /* Transport-backed stream: emit a DATA frame immediately. */
918 if (!stream->write_frame ||
919 stream->write_frame(stream->write_frame_ctx, frame, frame_len, 0) != 0) {
920 cwist_free(frame);
921 stream->status = CWIST_GRPC_UNAVAILABLE;
922 stream->status_message = "failed to write gRPC DATA frame";
923 return -1;
924 }
925 cwist_free(frame);
926 return 0;
927 }
928 if (!stream->res || !stream->res->body) {
929 cwist_free(frame);
930 return -1;
931 }
932 if (stream->write_frame &&
933 stream->write_frame(stream->write_frame_ctx, frame, frame_len, 0) != 0) {
934 cwist_free(frame);
935 stream->status = CWIST_GRPC_UNAVAILABLE;
936 stream->status_message = "failed to write gRPC DATA frame";
937 return -1;
938 }
939 cwist_error_t err = cwist_sstring_append_len(stream->res->body, (char *)frame, frame_len);
940 cwist_free(frame);
941 bool appended = cwist_error_is_ok(&err);
942 cwist_error_dispose(&err);
943 if (!appended) {
944 stream->status = CWIST_GRPC_INTERNAL;
945 stream->status_message = "failed to append gRPC stream message";
946 return -1;
947 }
948 return 0;
949}

1133static void grpc_health_watch(cwist_grpc_stream *stream, void *ctx) {
1134 cwist_grpc_health_registry *reg = ctx;
1135 char *service = NULL;
1136 if (stream->session) {
1137 /* Transport path: the HealthCheckRequest arrives as the first
1138 * (and only) message; the client then half-closes. */
1139 cwist_grpc_message msg;
1140 int rc = cwist_grpc_stream_recv(stream, &msg);
1141 if (rc < 0) return; /* cancelled before the request arrived */
1142 if (rc > 0) service = grpc_health_service_copy(&msg, NULL);
1143 } else {
1144 cwist_grpc_message empty = {0, NULL, 0};
1145 const cwist_grpc_message *message = stream->message_count ? &stream->messages[0] : &empty;
1146 service = grpc_health_service_copy(message, NULL);
1147 }
1148
1149 /* Send the current status immediately, as the protocol requires. */
1150 if (grpc_health_send_status(stream, reg, service) != 0) goto out;
1151 if (!stream->session) goto out; /* buffered dispatch: single snapshot */
1152
1153 atomic_fetch_add(&reg->watchers, 1);
1154 for (;;) {
1155 int changed = 0;
1156 pthread_mutex_lock(&reg->mu);
1157 uint64_t gen = reg->generation;
1158 while (reg->generation == gen && !cwist_grpc_stream_cancelled(stream) &&
1159 cwist_grpc_stream_deadline_remaining_ms(stream) > 0) {
1160 /* Poll cancellation/deadline on a short tick; status changes
1161 * wake us immediately through the registry condvar. */
1162 struct timespec ts;
1163 clock_gettime(CLOCK_REALTIME, &ts);
1164 ts.tv_nsec += 100000000L;
1165 if (ts.tv_nsec >= 1000000000L) {
1166 ts.tv_sec++;
1167 ts.tv_nsec -= 1000000000L;
1168 }
1169 pthread_cond_timedwait(&reg->cond, &reg->mu, &ts);
1170 }
1171 changed = reg->generation != gen;
1172 pthread_mutex_unlock(&reg->mu);
1173 if (!changed) break; /* cancelled or deadline expired */
1174 if (grpc_health_send_status(stream, reg, service) != 0) break;
1175 }
1176 atomic_fetch_sub(&reg->watchers, 1);
1177out:
1178 cwist_free(service);
1179}

1702static void grpc_session_send_trailers(cwist_grpc_session *session, cwist_grpc_status_t status,
1703 const char *message) {
1704 char status_buf[16];
1705 snprintf(status_buf, sizeof(status_buf), "%d", (int)status);
1706 char pushback_buf[20];
1707 if (session->stream.retry_pushback_set)
1708 snprintf(pushback_buf, sizeof(pushback_buf), "%d", (int)session->stream.retry_pushback_ms);
1709 pthread_mutex_lock(&session->wmu);
1710 if (session->h2s && !__atomic_load_n(&session->trailers_sent, __ATOMIC_ACQUIRE)) {
1711 if (!session->headers_sent && status != CWIST_GRPC_OK) {
1712 /* Trailers-Only (PROTOCOL-HTTP2, gRFC A6): the whole response is
1713 * one HEADERS frame with END_STREAM. Response-Headers never went
1714 * out, so the RPC stays retryable for conforming clients. */
1715 cwist_http2_header hdrs[6];
1716 size_t count = 0;
1717 hdrs[count++] = (cwist_http2_header){"content-type", "application/grpc"};
1718 hdrs[count++] = (cwist_http2_header){"grpc-accept-encoding", "gzip, identity"};
1719 hdrs[count++] = (cwist_http2_header){"grpc-status", status_buf};
1720 if (message) hdrs[count++] = (cwist_http2_header){"grpc-message", message};
1721 if (session->stream.retry_pushback_set)
1722 hdrs[count++] = (cwist_http2_header){"grpc-retry-pushback-ms", pushback_buf};
1723 if (cwist_http2_stream_send_headers(session->h2s, 200, hdrs, count, 1) == 0)
1724 session->headers_sent = 1;
1725 __atomic_store_n(&session->trailers_sent, 1, __ATOMIC_RELEASE);
1726 } else if (grpc_session_send_headers_locked(session) == 0) {
1727 cwist_http2_header pairs[3];
1728 size_t count = 0;
1729 pairs[count].name = "grpc-status";
1730 pairs[count].value = status_buf;
1731 count++;
1732 if (message) {
1733 pairs[count].name = "grpc-message";
1734 pairs[count].value = message;
1735 count++;
1736 }
1737 if (session->stream.retry_pushback_set) {
1738 pairs[count].name = "grpc-retry-pushback-ms";
1739 pairs[count].value = pushback_buf;
1740 count++;
1741 }
1742 (void)cwist_http2_stream_send_trailers(session->h2s, pairs, count);
1743 __atomic_store_n(&session->trailers_sent, 1, __ATOMIC_RELEASE);
1744 } else {
1745 /* transport broken; do not retry */
1746 __atomic_store_n(&session->trailers_sent, 1, __ATOMIC_RELEASE);
1747 }
1748 }
1749 pthread_mutex_unlock(&session->wmu);
1750}

1808int cwist_grpc_stream_recv(cwist_grpc_stream *stream, cwist_grpc_message *out) {
1809 if (!stream || !out) return -1;
1810
1811 if (!stream->session) {
1812 /* Buffered path: pop the pre-decoded message array. */
1813 if (stream->deadline_ms && grpc_now_ms() >= stream->deadline_ms) {
1814 __atomic_store_n(&stream->cancelled, 1, __ATOMIC_RELEASE);
1815 return -1;
1816 }
1817 if (stream->recv_pos >= stream->message_count) return 0;
1818 *out = stream->messages[stream->recv_pos++];
1819 return 1;
1820 }
1821
1822 cwist_grpc_session *session = stream->session;
1823 pthread_mutex_lock(&session->mu);
1824 for (;;) {
1825 if (session->qhead) {
1826 grpc_qnode *node = session->qhead;
1827 session->qhead = node->next;
1828 if (!session->qhead) session->qtail = NULL;
1829 pthread_mutex_unlock(&session->mu);
1830 cwist_free(session->recv_buf);
1831 session->recv_buf = node->data;
1832 out->compressed = 0;
1833 out->data = node->data;
1834 out->len = node->len;
1835 cwist_free(node);
1836 return 1;
1837 }
1838 if (session->cancelled) {
1839 pthread_mutex_unlock(&session->mu);
1840 return -1;
1841 }
1842 if (session->eof) {
1843 pthread_mutex_unlock(&session->mu);
1844 return 0;
1845 }
1846 if (stream->deadline_ms) {
1847 uint64_t now = grpc_now_ms();
1848 if (now >= stream->deadline_ms) {
1849 session->cancelled = 1;
1850 __atomic_store_n(&stream->cancelled, 1, __ATOMIC_RELEASE);
1851 /* Report the real outcome so a handler that simply closes
1852 * with stream->status still emits DEADLINE_EXCEEDED. */
1853 stream->status = CWIST_GRPC_DEADLINE_EXCEEDED;
1854 stream->status_message = "deadline exceeded";
1855 pthread_mutex_unlock(&session->mu);
1856 return -1;
1857 }
1858 uint64_t left = stream->deadline_ms - now;
1859 struct timespec ts;
1860 struct timespec rt;
1861 clock_gettime(CLOCK_REALTIME, &rt);
1862 ts.tv_sec = rt.tv_sec + (time_t)(left / 1000);
1863 ts.tv_nsec = rt.tv_nsec + (long)((left % 1000) * 1000000);
1864 if (ts.tv_nsec >= 1000000000L) {
1865 ts.tv_sec++;
1866 ts.tv_nsec -= 1000000000L;
1867 }
1868 pthread_cond_timedwait(&session->cond, &session->mu, &ts);
1869 } else {
1870 pthread_cond_wait(&session->cond, &session->mu);
1871 }
1872 }
1873}

1946static void *grpc_h2_on_headers(void *conn_ctx, cwist_http_request *req, cwist_h2_stream *stream) {
1947 cwist_grpc_h2_conn_ctx *ctx = conn_ctx;
1948 if (!ctx || !ctx->app || !req || !req->path || !req->path->data) return NULL;
1949
1950 cwist_grpc_route *route = grpc_find_route(ctx->app, req->path->data);
1951 /* Unary-only builtins (health Check) and reflection stay on the buffered
1952 * dispatch path; the health Watch builtin streams live. */
1953 if (!route || !route->stream_handler || (route->builtin && route->builtin != 2)) return NULL;
1954
1955 const char *ct = grpc_header_get(req, "content-type");
1956 if (!grpc_content_type_is_grpc(ct)) return NULL;
1957
1958 int encoding = grpc_request_encoding(req);
1959 if (encoding < 0) return NULL; /* buffered dispatch replies UNIMPLEMENTED */
1960
1961 cwist_grpc_session *session = (cwist_grpc_session *)cwist_alloc(sizeof(*session));
1962 if (!session) return NULL;
1963 memset(session, 0, sizeof(*session));
1964 pthread_mutex_init(&session->mu, NULL);
1965 pthread_cond_init(&session->cond, NULL);
1966 pthread_mutex_init(&session->wmu, NULL);
1967 session->h2s = stream;
1968 session->conn_ctx = ctx;
1969 session->encoding = encoding;
1970 cwist_grpc_decoder_init(&session->decoder, 0);
1971 atomic_store(&session->refs, 2); /* transport + handler thread */
1972
1973 session->stream.req = req;
1974 session->stream.res = NULL;
1975 session->stream.messages = NULL;
1976 session->stream.message_count = 0;
1977 session->stream.status = CWIST_GRPC_OK;
1978 session->stream.status_message = NULL;
1979 session->stream.closed = 0;
1980 session->stream.write_frame = grpc_session_write_frame;
1981 session->stream.write_frame_ctx = session;
1982 session->stream.session = session;
1983 __atomic_store_n(&session->stream.cancelled, 0, __ATOMIC_RELEASE);
1984 session->stream.recv_pos = 0;
1985 const char *timeout = grpc_header_get(req, "grpc-timeout");
1986 if (timeout) {
1987 uint64_t timeout_ms = 0;
1988 if (cwist_grpc_parse_timeout(timeout, &timeout_ms) == 0)
1989 session->stream.deadline_ms = grpc_now_ms() + timeout_ms;
1990 }
1991 session->handler = route->stream_handler;
1992 session->user_ctx = route->user_ctx;
1993 session->resp_encoding = grpc_client_accepts_gzip(req) ? 1 : 0;
1994
1995 /* Response-Headers are deliberately not sent yet (gRFC A6): they go out
1996 * with the first DATA frame, or the call ends in Trailers-Only form so
1997 * conforming clients may retry the RPC. */
1998
1999 pthread_mutex_lock(&ctx->mu);
2000 ctx->refs++;
2001 session->next = ctx->sessions;
2002 ctx->sessions = session;
2003 pthread_mutex_unlock(&ctx->mu);
2004
2005 pthread_t tid;
2006 if (pthread_create(&tid, NULL, grpc_session_thread, session) != 0) {
2007 pthread_mutex_lock(&ctx->mu);
2008 ctx->refs--;
2009 cwist_grpc_session **pp = &ctx->sessions;
2010 while (*pp && *pp != session) pp = &(*pp)->next;
2011 if (*pp) *pp = session->next;
2012 pthread_mutex_unlock(&ctx->mu);
2013 pthread_mutex_destroy(&session->mu);
2014 pthread_cond_destroy(&session->cond);
2015 pthread_mutex_destroy(&session->wmu);
2016 cwist_grpc_decoder_destroy(&session->decoder);
2017 cwist_free(session);
2018 return NULL;
2019 }
2020 pthread_detach(tid);
2021 return session;
2022}
