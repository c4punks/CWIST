241int cwist_grpc_parse_timeout(const char *value, uint64_t *out_ms) {
242 if (!value || !out_ms) return -1;
243 size_t len = strlen(value);
244 if (len < 2 || len > 9) return -1; /* up to 8 digits + unit */
245 char unit = value[len - 1];
246 uint64_t to_ms_num = 1, to_ms_den = 1;
247 switch (unit) {
248 case 'H': to_ms_num = 3600000; break;
249 case 'M': to_ms_num = 60000; break;
250 case 'S': to_ms_num = 1000; break;
251 case 'm': to_ms_num = 1; break;
252 case 'u':
253 to_ms_num = 1;
254 to_ms_den = 1000;
255 break;
256 case 'n':
257 to_ms_num = 1;
258 to_ms_den = 1000000;
259 break;
260 default: return -1;
261 }
262 uint64_t v = 0;
263 for (size_t i = 0; i + 1 < len; i++) {
264 if (!isdigit((unsigned char)value[i])) return -1;
265 v = v * 10 + (uint64_t)(value[i] - '0');
266 }
267 /* Round sub-millisecond units up so a positive timeout never becomes 0. */
268 uint64_t ms = (v * to_ms_num + to_ms_den - 1) / to_ms_den;
269 if (v > 0 && ms == 0) ms = 1;
270 *out_ms = ms;
271 return 0;
272}

315static int grpc_gzip_deflate(const uint8_t *in, size_t in_len, uint8_t **out, size_t *out_len) {
316 z_stream zs;
317 memset(&zs, 0, sizeof(zs));
318 if (deflateInit2(&zs, Z_DEFAULT_COMPRESSION, Z_DEFLATED, 16 + MAX_WBITS, 8,
319 Z_DEFAULT_STRATEGY) != Z_OK)
320 return -1;
321 size_t cap = deflateBound(&zs, (uLong)in_len) + 32;
322 if (cap < 64) cap = 64;
323 uint8_t *buf = (uint8_t *)cwist_alloc(cap);
324 if (!buf) {
325 deflateEnd(&zs);
326 return -1;
327 }
328 zs.next_in = (Bytef *)in;
329 zs.avail_in = (uInt)in_len;
330 zs.next_out = buf;
331 zs.avail_out = (uInt)cap;
332 int zrc = deflate(&zs, Z_FINISH);
333 if (zrc != Z_STREAM_END) {
334 deflateEnd(&zs);
335 cwist_free(buf);
336 return -1;
337 }
338 *out_len = cap - zs.avail_out;
339 deflateEnd(&zs);
340 *out = buf;
341 return 0;
342}

356static int grpc_gzip_inflate(const uint8_t *in, size_t in_len, uint8_t **out, size_t *out_len) {
357 if (in_len > UINT_MAX) return -1;
358 z_stream zs;
359 memset(&zs, 0, sizeof(zs));
360 if (inflateInit2(&zs, 16 + MAX_WBITS) != Z_OK) return -1;
361 const size_t max_cap = (size_t)CWIST_HTTP_MAX_BODY_SIZE + 1;
362 size_t cap = 8192;
363 if (in_len <= (size_t)CWIST_HTTP_MAX_BODY_SIZE) {
364 size_t suggested = in_len * 3 + 1024;
365 if (suggested > cap) cap = suggested;
366 }
367 if (cap > max_cap) cap = max_cap;
368 uint8_t *buf = (uint8_t *)cwist_alloc(cap);
369 if (!buf) {
370 inflateEnd(&zs);
371 return -1;
372 }
373 zs.next_in = (Bytef *)in;
374 zs.avail_in = (uInt)in_len;
375 size_t used = 0;
376 int zrc = Z_OK;
377 while (zrc == Z_OK) {
378 if (used == cap) {
379 size_t next_cap = cap <= max_cap / 2 ? cap * 2 : max_cap;
380 if (next_cap > max_cap) next_cap = max_cap;
381 uint8_t *nb = (uint8_t *)cwist_realloc(buf, next_cap);
382 if (!nb) {
383 cwist_free(buf);
384 inflateEnd(&zs);
385 return -1;
386 }
387 buf = nb;
388 cap = next_cap;
389 }
390 zs.next_out = buf + used;
391 zs.avail_out = (uInt)(cap - used);
392 zrc = inflate(&zs, Z_NO_FLUSH);
393 used = cap - zs.avail_out;
394 if (used > (size_t)CWIST_HTTP_MAX_BODY_SIZE) {
395 cwist_free(buf);
396 inflateEnd(&zs);
397 return -2;
398 }
399 }
400 inflateEnd(&zs);
401 if (zrc != Z_STREAM_END) {
402 cwist_free(buf);
403 return -1;
404 }
405 *out = buf;
406 *out_len = used;
407 return 0;
408}

435static int grpc_b64_val(int c) {
436 if (c >= 'A' && c <= 'Z') return c - 'A';
437 if (c >= 'a' && c <= 'z') return c - 'a' + 26;
438 if (c >= '0' && c <= '9') return c - '0' + 52;
439 if (c == '+') return 62;

456static int grpc_base64_decode(const char *in, size_t in_len, uint8_t *out, size_t out_cap,
457 size_t *out_len) {
458 uint32_t acc = 0;
459 int bits = 0;
460 size_t used = 0;
461 for (size_t i = 0; i < in_len; i++) {
462 if (in[i] == '=') break;
463 int v = grpc_b64_val((unsigned char)in[i]);
464 if (v < 0) return -1;
465 acc = (acc << 6) | (uint32_t)v;
466 bits += 6;
467 if (bits >= 8) {
468 bits -= 8;
469 if (used >= out_cap) return -2;
470 out[used++] = (uint8_t)(acc >> bits);
471 }
472 }
473 *out_len = used;
474 return 0;
475}

635static void grpc_dispatch_stream_route(cwist_http_request *req, cwist_http_response *res,
636 const grpc_route_view *route) {
637 if (!route->stream_handler) {
638 cwist_grpc_set_error(res, CWIST_GRPC_UNIMPLEMENTED, "gRPC stream method not registered");
639 return;
640 }
641 if (grpc_validate_request(req, res) != 0) return;
642
643 size_t offset = 0;
644 size_t cap = 4;
645 size_t count = 0;
646 cwist_grpc_message *messages = (cwist_grpc_message *)cwist_alloc(cap * sizeof(*messages));
647 if (!messages) {
648 cwist_grpc_set_error(res, CWIST_GRPC_RESOURCE_EXHAUSTED,
649 "failed to allocate gRPC stream messages");
650 return;
651 }
652
653 while (offset < req->body->size) {
654 if (count == cap) {
655 cap *= 2;
656 cwist_grpc_message *next =
657 (cwist_grpc_message *)cwist_realloc(messages, cap * sizeof(*messages));
658 if (!next) {
659 cwist_free(messages);
660 cwist_grpc_set_error(res, CWIST_GRPC_RESOURCE_EXHAUSTED,
661 "failed to grow gRPC stream messages");
662 return;
663 }
664 messages = next;
665 }
666 if (cwist_grpc_decode_next_message(req->body->data, req->body->size, &offset,
667 &messages[count]) != 0) {
668 cwist_free(messages);
669 cwist_grpc_set_error(res, CWIST_GRPC_INVALID_ARGUMENT, "malformed gRPC stream frame");
670 return;
671 }
672 count++;
673 }
674
675 /* Decompress any compressed-flag messages when grpc-encoding: gzip. */
676 uint8_t **owned CWIST_DEFER_FREE = NULL;
677 int has_compressed = 0;
678 for (size_t i = 0; i < count; i++)
679 if (messages[i].compressed) {
680 has_compressed = 1;
681 break;
682 }
683 if (has_compressed) {
684 owned = (uint8_t **)cwist_alloc(count * sizeof(*owned));
685 if (!owned) {
686 cwist_free(messages);
687 cwist_grpc_set_error(res, CWIST_GRPC_RESOURCE_EXHAUSTED,
688 "failed to allocate inflate state");
689 return;
690 }
691 memset(owned, 0, count * sizeof(*owned));
692 for (size_t i = 0; i < count; i++) {
693 int inflate_rc = grpc_message_inflate(req, &messages[i], &owned[i]);
694 if (inflate_rc != 0) {
695 for (size_t j = 0; j < count; j++) cwist_free(owned[j]);
696 cwist_free(messages);
697 if (inflate_rc == -2)
698 cwist_grpc_set_error(res, CWIST_GRPC_RESOURCE_EXHAUSTED,
699 "decompressed gRPC message exceeds size limit");
700 else
701 cwist_grpc_set_error(res, CWIST_GRPC_UNIMPLEMENTED,
702 "unsupported compressed gRPC message");
703 return;
704 }
705 }
706 }
707
708 uint64_t deadline_ms = 0;
709 const char *timeout = grpc_header_get(req, "grpc-timeout");
710 if (timeout) {
711 uint64_t timeout_ms = 0;
712 if (cwist_grpc_parse_timeout(timeout, &timeout_ms) == 0)
713 deadline_ms = grpc_now_ms() + timeout_ms;
714 }
715
716 res->status_code = CWIST_HTTP_OK;

766int cwist_grpc_decode_message(const void *frame, size_t frame_len, cwist_grpc_message *out) {
767 if (!frame || !out || frame_len < 5) return -1;
768 const uint8_t *p = (const uint8_t *)frame;
769 uint32_t len =
770 ((uint32_t)p[1] << 24) | ((uint32_t)p[2] << 16) | ((uint32_t)p[3] << 8) | (uint32_t)p[4];
771 if ((size_t)len > frame_len - 5) return -1;
772 if ((size_t)len != frame_len - 5) return -1;
773
774 out->compressed = p[0];
775 out->data = p + 5;
776 out->len = (size_t)len;
777 return 0;
778}

793int cwist_grpc_decode_next_message(const void *frames, size_t frames_len, size_t *offset,
794 cwist_grpc_message *out) {
795 if (!frames || !offset || !out || *offset > frames_len) return -1;
796 size_t pos = *offset;
797 if (frames_len - pos < 5) return -1;
798
799 const uint8_t *p = (const uint8_t *)frames + pos;
800 uint32_t len =

802 if ((size_t)len > frames_len - pos - 5) return -1;
803
804 out->compressed = p[0];
805 out->data = p + 5;
806 out->len = (size_t)len;
807 *offset = pos + 5 + (size_t)len;
808 return 0;
809}

821int cwist_grpc_encode_message(const void *payload, size_t payload_len, uint8_t compressed,
822 uint8_t **out, size_t *out_len) {
823 if (!out || !out_len || (payload_len > 0 && !payload)) return -1;
824 if (payload_len > UINT32_MAX) return -1;
825
826 uint8_t *buf = (uint8_t *)cwist_alloc(payload_len + 5);
827 if (!buf) return -1;
828 buf[0] = compressed ? 1 : 0;
829 buf[1] = (uint8_t)((payload_len >> 24) & 0xff);
830 buf[2] = (uint8_t)((payload_len >> 16) & 0xff);
831 buf[3] = (uint8_t)((payload_len >> 8) & 0xff);
832 buf[4] = (uint8_t)(payload_len & 0xff);
833 if (payload_len > 0) memcpy(buf + 5, payload, payload_len);
834
835 *out = buf;
836 *out_len = payload_len + 5;
837 return 0;
838}

878int cwist_grpc_decoder_feed(cwist_grpc_decoder *decoder, const void *data, size_t len,
879 cwist_grpc_message_callback callback, void *ctx) {
880 if (!decoder || (!data && len) || !callback) return -1;
881 const uint8_t *input = data;
882 while (len) {
883 if (decoder->header_len < sizeof(decoder->header)) {
884 size_t take = sizeof(decoder->header) - decoder->header_len;
885 if (take > len) take = len;
886 memcpy(decoder->header + decoder->header_len, input, take);
887 decoder->header_len += take;
888 input += take;
889 len -= take;
890 if (decoder->header_len < sizeof(decoder->header)) continue;
891 decoder->compressed = decoder->header[0];
892 decoder->payload_len = ((size_t)decoder->header[1] << 24) |
893 ((size_t)decoder->header[2] << 16) |
894 ((size_t)decoder->header[3] << 8) | (size_t)decoder->header[4];
895 if (decoder->compressed > 1 || decoder->payload_len > decoder->max_message_size)
896 return -1;
897 if (decoder->payload_len) {
898 decoder->payload = cwist_alloc(decoder->payload_len);
899 if (!decoder->payload) return -1;
900 }
901 }
902 size_t take = decoder->payload_len - decoder->payload_used;
903 if (take > len) take = len;
904 if (take) memcpy(decoder->payload + decoder->payload_used, input, take);
905 decoder->payload_used += take;
906 input += take;
907 len -= take;
908 if (decoder->payload_used != decoder->payload_len) continue;
909 cwist_grpc_message message = {decoder->compressed, decoder->payload, decoder->payload_len};
910 if (callback(ctx, &message) != 0) return -1;
911 cwist_free(decoder->payload);
912 decoder->payload = NULL;
913 decoder->payload_len = decoder->payload_used = decoder->header_len = 0;
914 }
915 return 0;
916}

980int cwist_grpc_stream_send(cwist_grpc_stream *stream, const void *payload, size_t payload_len) {
981 if (!stream) return -1;
982 int use_gzip = 0;
983 if (stream->session) {
984 use_gzip = (((cwist_grpc_session *)stream->session)->resp_encoding == 1);
985 } else if (stream->req) {
986 use_gzip = grpc_client_accepts_gzip(stream->req);
987 }
988
989 uint8_t *frame = NULL;
990 size_t frame_len = 0;
991 if (use_gzip && payload_len > 0) {
992 uint8_t *cbuf = NULL;
993 size_t clen = 0;
994 if (grpc_gzip_deflate(payload, payload_len, &cbuf, &clen) == 0) {
995 int enc_rc = cwist_grpc_encode_message(cbuf, clen, 1, &frame, &frame_len);
996 cwist_free(cbuf);
997 if (enc_rc != 0) {
998 stream->status = CWIST_GRPC_INTERNAL;
999 stream->status_message = "failed to encode gRPC stream message";
1000 return -1;
1001 }
1002 } else {
1003 if (cwist_grpc_encode_message(payload, payload_len, 0, &frame, &frame_len) != 0) {
1004 stream->status = CWIST_GRPC_INTERNAL;
1005 stream->status_message = "failed to encode gRPC stream message";
1006 return -1;
1007 }
1008 }
1009 } else {
1010 if (cwist_grpc_encode_message(payload, payload_len, 0, &frame, &frame_len) != 0) {
1011 stream->status = CWIST_GRPC_INTERNAL;
1012 stream->status_message = "failed to encode gRPC stream message";
1013 return -1;
1014 }
1015 }
1016 if (stream->session) {
1017 /* Transport-backed stream: emit a DATA frame immediately. */
1018 if (!stream->write_frame ||
1019 stream->write_frame(stream->write_frame_ctx, frame, frame_len, 0) != 0) {
1020 cwist_free(frame);
1021 stream->status = CWIST_GRPC_UNAVAILABLE;
1022 stream->status_message = "failed to write gRPC DATA frame";
1023 return -1;
1024 }
1025 cwist_free(frame);
1026 return 0;
1027 }
1028 if (!stream->res || !stream->res->body) {
1029 cwist_free(frame);
1030 return -1;
1031 }
1032 if (stream->write_frame &&
1033 stream->write_frame(stream->write_frame_ctx, frame, frame_len, 0) != 0) {
1034 cwist_free(frame);
1035 stream->status = CWIST_GRPC_UNAVAILABLE;
1036 stream->status_message = "failed to write gRPC DATA frame";
1037 return -1;
1038 }
1039 cwist_error_t err = cwist_sstring_append_len(stream->res->body, (char *)frame, frame_len);
1040 cwist_free(frame);
1041 bool appended = cwist_error_is_ok(&err);
1042 cwist_error_dispose(&err);
1043 if (!appended) {
1044 stream->status = CWIST_GRPC_INTERNAL;
1045 stream->status_message = "failed to append gRPC stream message";
1046 return -1;
1047 }
1048 return 0;
1049}

1233static void grpc_health_watch(cwist_grpc_stream *stream, void *ctx) {
1234 cwist_grpc_health_registry *reg = ctx;
1235 char *service = NULL;
1236 int allocation_failed = 0;
1237 if (stream->session) {
1238 /* Transport path: the HealthCheckRequest arrives as the first
1239 * (and only) message; the client then half-closes. */
1240 cwist_grpc_message msg;
1241 int rc = cwist_grpc_stream_recv(stream, &msg);
1242 if (rc < 0) return; /* cancelled before the request arrived */
1243 if (rc > 0) service = grpc_health_service_copy(&msg, &allocation_failed);
1244 } else {
1245 cwist_grpc_message empty = {0, NULL, 0};
1246 const cwist_grpc_message *message = stream->message_count ? &stream->messages[0] : &empty;
1247 service = grpc_health_service_copy(message, &allocation_failed);
1248 }
1249 if (allocation_failed) {
1250 stream->status = CWIST_GRPC_INTERNAL;
1251 stream->status_message = "health request allocation failed";
1252 goto out;
1253 }
1254
1255 /* Send the current status immediately, as the protocol requires. */
1256 if (grpc_health_send_status(stream, reg, service) != 0) goto out;
1257 if (!stream->session) goto out; /* buffered dispatch: single snapshot */
1258
1259 atomic_fetch_add(&reg->watchers, 1);
1260 for (;;) {
1261 int changed = 0;
1262 pthread_mutex_lock(&reg->mu);
1263 uint64_t gen = reg->generation;
1264 while (reg->generation == gen && !reg->shutdown && !cwist_grpc_stream_cancelled(stream) &&
1265 cwist_grpc_stream_deadline_remaining_ms(stream) > 0) {
1266 /* Poll cancellation/deadline on a short tick; status changes
1267 * wake us immediately through the registry condvar. */
1268 struct timespec ts;
1269 clock_gettime(CLOCK_REALTIME, &ts);
1270 ts.tv_nsec += 100000000L;
1271 if (ts.tv_nsec >= 1000000000L) {
1272 ts.tv_sec++;
1273 ts.tv_nsec -= 1000000000L;
1274 }
1275 pthread_cond_timedwait(&reg->cond, &reg->mu, &ts);
1276 }
1277 changed = reg->generation != gen;
1278 int shutdown = reg->shutdown;
1279 pthread_mutex_unlock(&reg->mu);
1280 if (shutdown) {
1281 stream->status = CWIST_GRPC_UNAVAILABLE;
1282 stream->status_message = "health service unregistered";
1283 break;
1284 }
1285 if (!changed) break; /* cancelled or deadline expired */
1286 if (grpc_health_send_status(stream, reg, service) != 0) break;
1287 }
1288 atomic_fetch_sub(&reg->watchers, 1);
1289out:
1290 cwist_free(service);
1291}

1449 stream->status = CWIST_GRPC_RESOURCE_EXHAUSTED;

1451 }

1456 if (stream->status != CWIST_GRPC_OK) break;
1457 }
1458 pthread_rwlock_unlock(&grpc_routes_lock);
1459 if (stream->status == CWIST_GRPC_OK)

1462}

1496static int grpc_register_route(cwist_app *app, const char *service, const char *method,
1497 int streaming, cwist_grpc_unary_handler_func unary_handler,
1498 cwist_grpc_stream_handler_func stream_handler, void *user_ctx,
1499 int builtin) {
1500 if (!app || !service || !method) return -1;
1501 if ((!streaming && !unary_handler) || (streaming && !stream_handler)) return -1;
1502 size_t service_len = strlen(service);
1503 size_t method_len = strlen(method);
1504 if (service_len == 0 || method_len == 0) return -1;
1505
1506 size_t path_len = service_len + method_len + 3;
1507 char *path = (char *)cwist_alloc(path_len);
1508 if (!path) return -1;
1509 snprintf(path, path_len, "/%s/%s", service, method);
1510
1511 cwist_handler_func dispatch = streaming ? grpc_dispatch_stream : grpc_dispatch_unary;
1512 int rc = -1;
1513 pthread_rwlock_wrlock(&grpc_routes_lock);
1514 cwist_grpc_route *existing = grpc_find_route(app, path);
1515 if (existing) {
1516 /* A built-in owns its context; it must be unregistered, not overwritten. */
1517 if (existing->builtin != builtin || !cwist_app_post_checked(app, path, dispatch))
1518 goto out;
1519 existing->streaming = streaming;
1520 existing->handler = unary_handler;
1521 existing->stream_handler = stream_handler;
1522 existing->user_ctx = user_ctx;
1523 rc = 0;
1524 goto out;
1525 }
1526
1527 cwist_grpc_route *route = (cwist_grpc_route *)cwist_alloc(sizeof(*route));
1528 if (!route) goto out;
1529 memset(route, 0, sizeof(*route));
1530 route->path = path;
1531 route->streaming = streaming;
1532 route->builtin = builtin;
1533 route->handler = unary_handler;
1534 route->stream_handler = stream_handler;
1535 route->user_ctx = user_ctx;
1536 atomic_store(&route->refs, 1);
1537 if (!cwist_app_post_checked(app, path, dispatch)) {
1538 cwist_free(route);
1539 goto out;
1540 }
1541 if (builtin == 1 || builtin == 2) {
1542 cwist_grpc_health_registry *reg = user_ctx;
1543 atomic_fetch_add(&reg->refs, 1);
1544 }
1545 route->next = (cwist_grpc_route *)app->grpc_routes;
1546 app->grpc_routes = route;
1547 path = NULL; /* owned by the route */
1548 rc = 0;
1549out:
1550 pthread_rwlock_unlock(&grpc_routes_lock);
1551 cwist_free(path);
1552 return rc;
1553}

1646 while (route) {

1721static void grpc_session_release(cwist_grpc_session *session) {
1722 if (atomic_fetch_sub(&session->refs, 1) != 1) return;
1723 cwist_grpc_h2_conn_ctx *ctx = session->conn_ctx;
1724 pthread_mutex_lock(&ctx->mu);
1725 cwist_grpc_session **pp = &ctx->sessions;
1726 while (*pp && *pp != session) pp = &(*pp)->next;
1727 if (*pp) *pp = session->next;
1728 pthread_mutex_unlock(&ctx->mu);
1729 grpc_conn_ctx_release(ctx);
1730
1731 cwist_grpc_decoder_destroy(&session->decoder);
1732 while (session->qhead) {
1733 grpc_qnode *next = session->qhead->next;
1734 cwist_free(session->qhead->data);
1735 cwist_free(session->qhead);
1736 session->qhead = next;
1737 }
1738 cwist_free(session->recv_buf);
1739 if (session->stream.req) cwist_http_request_destroy(session->stream.req);
1740 pthread_mutex_destroy(&session->mu);
1741 pthread_cond_destroy(&session->cond);
1742 pthread_mutex_destroy(&session->wmu);
1743 cwist_free(session);
1744}

1756static int grpc_session_push(cwist_grpc_session *session, uint8_t *data, size_t len) {
1757 grpc_qnode *node = (grpc_qnode *)cwist_alloc(sizeof(*node));
1758 if (!node) return -1;
1759 node->data = data;
1760 node->len = len;
1761 node->next = NULL;
1762 pthread_mutex_lock(&session->mu);
1763 if (session->cancelled || session->eof) {
1764 pthread_mutex_unlock(&session->mu);
1765 cwist_free(data);
1766 cwist_free(node);
1767 return -1;
1768 }
1769 if (session->qtail)
1770 session->qtail->next = node;
1771 else
1772 session->qhead = node;
1773 session->qtail = node;
1774 pthread_cond_signal(&session->cond);
1775 pthread_mutex_unlock(&session->mu);
1776 return 0;
1777}

1860static void grpc_session_send_trailers(cwist_grpc_session *session, cwist_grpc_status_t status,
1861 const char *message) {
1862 char status_buf[16];
1863 snprintf(status_buf, sizeof(status_buf), "%d", (int)status);
1864 char pushback_buf[20];
1865 if (session->stream.retry_pushback_set)
1866 snprintf(pushback_buf, sizeof(pushback_buf), "%d", (int)session->stream.retry_pushback_ms);
1867 pthread_mutex_lock(&session->wmu);
1868 if (session->h2s && !__atomic_load_n(&session->trailers_sent, __ATOMIC_ACQUIRE)) {
1869 if (!session->headers_sent && status != CWIST_GRPC_OK) {
1870 /* Trailers-Only (PROTOCOL-HTTP2, gRFC A6): the whole response is
1871 * one HEADERS frame with END_STREAM. Response-Headers never went
1872 * out, so the RPC stays retryable for conforming clients. */
1873 cwist_http2_header hdrs[6];
1874 size_t count = 0;
1875 hdrs[count++] = (cwist_http2_header){"content-type", "application/grpc"};
1876 hdrs[count++] = (cwist_http2_header){"grpc-accept-encoding", "gzip, identity"};
1877 hdrs[count++] = (cwist_http2_header){"grpc-status", status_buf};
1878 if (message) hdrs[count++] = (cwist_http2_header){"grpc-message", message};
1879 if (session->stream.retry_pushback_set)
1880 hdrs[count++] = (cwist_http2_header){"grpc-retry-pushback-ms", pushback_buf};
1881 if (cwist_http2_stream_send_headers(session->h2s, 200, hdrs, count, 1) == 0)
1882 session->headers_sent = 1;
1883 __atomic_store_n(&session->trailers_sent, 1, __ATOMIC_RELEASE);
1884 } else if (grpc_session_send_headers_locked(session) == 0) {
1885 cwist_http2_header pairs[3];
1886 size_t count = 0;
1887 pairs[count].name = "grpc-status";
1888 pairs[count].value = status_buf;
1889 count++;
1890 if (message) {
1891 pairs[count].name = "grpc-message";
1892 pairs[count].value = message;
1893 count++;
1894 }
1895 if (session->stream.retry_pushback_set) {
1896 pairs[count].name = "grpc-retry-pushback-ms";
1897 pairs[count].value = pushback_buf;
1898 count++;
1899 }
1900 (void)cwist_http2_stream_send_trailers(session->h2s, pairs, count);
1901 __atomic_store_n(&session->trailers_sent, 1, __ATOMIC_RELEASE);
1902 } else {
1903 /* transport broken; do not retry */
1904 __atomic_store_n(&session->trailers_sent, 1, __ATOMIC_RELEASE);
1905 }
1906 }
1907 pthread_mutex_unlock(&session->wmu);
1908}

1967int cwist_grpc_stream_recv(cwist_grpc_stream *stream, cwist_grpc_message *out) {
1968 if (!stream || !out) return -1;
1969
1970 if (!stream->session) {
1971 /* Buffered path: pop the pre-decoded message array. */
1972 if (stream->deadline_ms && grpc_now_ms() >= stream->deadline_ms) {
1973 __atomic_store_n(&stream->cancelled, 1, __ATOMIC_RELEASE);
1974 return -1;
1975 }
1976 if (stream->recv_pos >= stream->message_count) return 0;
1977 *out = stream->messages[stream->recv_pos++];
1978 return 1;
1979 }
1980
1981 cwist_grpc_session *session = stream->session;
1982 pthread_mutex_lock(&session->mu);
1983 for (;;) {
1984 if (session->qhead) {
1985 grpc_qnode *node = session->qhead;
1986 session->qhead = node->next;
1987 if (!session->qhead) session->qtail = NULL;
1988 pthread_mutex_unlock(&session->mu);
1989 cwist_free(session->recv_buf);
1990 session->recv_buf = node->data;
1991 out->compressed = 0;
1992 out->data = node->data;
1993 out->len = node->len;
1994 cwist_free(node);
1995 return 1;
1996 }
1997 if (session->cancelled) {
1998 pthread_mutex_unlock(&session->mu);
1999 return -1;
2000 }
2001 if (session->eof) {
2002 pthread_mutex_unlock(&session->mu);
2003 return 0;
2004 }
2005 if (stream->deadline_ms) {
2006 uint64_t now = grpc_now_ms();
2007 if (now >= stream->deadline_ms) {
2008 session->cancelled = 1;
2009 __atomic_store_n(&stream->cancelled, 1, __ATOMIC_RELEASE);
2010 /* Report the real outcome so a handler that simply closes
2011 * with stream->status still emits DEADLINE_EXCEEDED. */
2012 stream->status = CWIST_GRPC_DEADLINE_EXCEEDED;
2013 stream->status_message = "deadline exceeded";
2014 pthread_mutex_unlock(&session->mu);
2015 return -1;
2016 }
2017 uint64_t left = stream->deadline_ms - now;
2018 struct timespec ts;
2019 struct timespec rt;
2020 clock_gettime(CLOCK_REALTIME, &rt);
2021 ts.tv_sec = rt.tv_sec + (time_t)(left / 1000);
2022 ts.tv_nsec = rt.tv_nsec + (long)((left % 1000) * 1000000);
2023 if (ts.tv_nsec >= 1000000000L) {
2024 ts.tv_sec++;
2025 ts.tv_nsec -= 1000000000L;
2026 }
2027 pthread_cond_timedwait(&session->cond, &session->mu, &ts);
2028 } else {
2029 pthread_cond_wait(&session->cond, &session->mu);
2030 }
2031 }
2032}

2105static void *grpc_h2_on_headers(void *conn_ctx, cwist_http_request *req, cwist_h2_stream *stream) {
2106 cwist_grpc_h2_conn_ctx *ctx = conn_ctx;
2107 if (!ctx || !ctx->app || !req || !req->path || !req->path->data) return NULL;
2108
2109 grpc_route_view view;
2110 cwist_grpc_route *route = grpc_route_acquire(ctx->app, req->path->data, &view);
2111 /* Unary-only builtins (health Check) and reflection stay on the buffered
2112 * dispatch path; the health Watch builtin streams live. */
2113 if (!route) return NULL;
2114 if (!view.stream_handler || (view.builtin && view.builtin != 2)) goto not_live;
2115
2116 const char *ct = grpc_header_get(req, "content-type");
2117 if (!grpc_content_type_is_grpc(ct)) goto not_live;
2118
2119 int encoding = grpc_request_encoding(req);
2120 if (encoding < 0) goto not_live; /* buffered dispatch replies UNIMPLEMENTED */
2121
2122 cwist_grpc_session *session = (cwist_grpc_session *)cwist_alloc(sizeof(*session));
2123 if (!session) goto not_live;
2124 memset(session, 0, sizeof(*session));
2125 pthread_mutex_init(&session->mu, NULL);
2126 pthread_cond_init(&session->cond, NULL);
2127 pthread_mutex_init(&session->wmu, NULL);
2128 session->h2s = stream;
2129 session->conn_ctx = ctx;
2130 session->encoding = encoding;
2131 cwist_grpc_decoder_init(&session->decoder, 0);
2132 atomic_store(&session->refs, 2); /* transport + handler thread */
2133
2134 session->stream.req = req;
2135 session->stream.res = NULL;
2136 session->stream.messages = NULL;
2137 session->stream.message_count = 0;
2138 session->stream.status = CWIST_GRPC_OK;
2139 session->stream.status_message = NULL;
2140 session->stream.closed = 0;
2141 session->stream.write_frame = grpc_session_write_frame;
2142 session->stream.write_frame_ctx = session;
2143 session->stream.session = session;
2144 __atomic_store_n(&session->stream.cancelled, 0, __ATOMIC_RELEASE);
2145 session->stream.recv_pos = 0;
2146 const char *timeout = grpc_header_get(req, "grpc-timeout");
2147 if (timeout) {
2148 uint64_t timeout_ms = 0;
2149 if (cwist_grpc_parse_timeout(timeout, &timeout_ms) == 0)
2150 session->stream.deadline_ms = grpc_now_ms() + timeout_ms;
2151 }
2152 session->handler = view.stream_handler;
2153 session->user_ctx = view.user_ctx;
2154 session->route = route; /* released by the handler thread */
2155 session->resp_encoding = grpc_client_accepts_gzip(req) ? 1 : 0;
2156
2157 /* Response-Headers are deliberately not sent yet (gRFC A6): they go out
2158 * with the first DATA frame, or the call ends in Trailers-Only form so
