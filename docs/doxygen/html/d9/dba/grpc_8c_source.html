146int cwist_grpc_parse_timeout(const char *value, uint64_t *out_ms) {
147 if (!value || !out_ms) return -1;
148 size_t len = strlen(value);
149 if (len < 2 || len > 9) return -1; /* up to 8 digits + unit */
150 char unit = value[len - 1];
151 uint64_t to_ms_num = 1, to_ms_den = 1;
152 switch (unit) {
153 case 'H': to_ms_num = 3600000; break;
154 case 'M': to_ms_num = 60000; break;
155 case 'S': to_ms_num = 1000; break;
156 case 'm': to_ms_num = 1; break;
157 case 'u':
158 to_ms_num = 1;
159 to_ms_den = 1000;
160 break;
161 case 'n':
162 to_ms_num = 1;
163 to_ms_den = 1000000;
164 break;
165 default: return -1;
166 }
167 uint64_t v = 0;
168 for (size_t i = 0; i + 1 < len; i++) {
169 if (!isdigit((unsigned char)value[i])) return -1;
170 v = v * 10 + (uint64_t)(value[i] - '0');
171 }
172 /* Round sub-millisecond units up so a positive timeout never becomes 0. */
173 uint64_t ms = (v * to_ms_num + to_ms_den - 1) / to_ms_den;
174 if (v > 0 && ms == 0) ms = 1;
175 *out_ms = ms;
176 return 0;
177}

220static int grpc_gzip_deflate(const uint8_t *in, size_t in_len, uint8_t **out, size_t *out_len) {
221 z_stream zs;
222 memset(&zs, 0, sizeof(zs));
223 if (deflateInit2(&zs, Z_DEFAULT_COMPRESSION, Z_DEFLATED, 16 + MAX_WBITS, 8,
224 Z_DEFAULT_STRATEGY) != Z_OK)
225 return -1;
226 size_t cap = deflateBound(&zs, (uLong)in_len) + 32;
227 if (cap < 64) cap = 64;
228 uint8_t *buf = (uint8_t *)cwist_alloc(cap);
229 if (!buf) {
230 deflateEnd(&zs);
231 return -1;
232 }
233 zs.next_in = (Bytef *)in;
234 zs.avail_in = (uInt)in_len;
235 zs.next_out = buf;
236 zs.avail_out = (uInt)cap;
237 int zrc = deflate(&zs, Z_FINISH);
238 if (zrc != Z_STREAM_END) {
239 deflateEnd(&zs);
240 cwist_free(buf);
241 return -1;
242 }
243 *out_len = cap - zs.avail_out;
244 deflateEnd(&zs);
245 *out = buf;
246 return 0;
247}

259static int grpc_gzip_inflate(const uint8_t *in, size_t in_len, uint8_t **out, size_t *out_len) {
260 z_stream zs;
261 memset(&zs, 0, sizeof(zs));
262 if (inflateInit2(&zs, 16 + MAX_WBITS) != Z_OK) return -1;
263 size_t cap = in_len * 3 + 1024;
264 if (cap < 8192) cap = 8192;
265 uint8_t *buf = (uint8_t *)cwist_alloc(cap);
266 if (!buf) {
267 inflateEnd(&zs);
268 return -1;
269 }
270 zs.next_in = (Bytef *)in;
271 zs.avail_in = (uInt)in_len;
272 size_t used = 0;
273 int zrc = Z_OK;
274 while (zrc == Z_OK) {
275 if (used == cap) {
276 cap *= 2;
277 uint8_t *nb = (uint8_t *)cwist_realloc(buf, cap);
278 if (!nb) {
279 cwist_free(buf);
280 inflateEnd(&zs);
281 return -1;
282 }
283 buf = nb;
284 }
285 zs.next_out = buf + used;
286 zs.avail_out = (uInt)(cap - used);
287 zrc = inflate(&zs, Z_NO_FLUSH);
288 used = cap - zs.avail_out;
289 }
290 inflateEnd(&zs);
291 if (zrc != Z_STREAM_END) {
292 cwist_free(buf);
293 return -1;
294 }
295 *out = buf;
296 *out_len = used;
297 return 0;
298}

345static int grpc_base64_decode(const char *in, size_t in_len, uint8_t *out, size_t out_cap,
346 size_t *out_len) {
347 uint32_t acc = 0;
348 int bits = 0;
349 size_t used = 0;
350 for (size_t i = 0; i < in_len; i++) {
351 if (in[i] == '=') break;
352 int v = grpc_b64_val((unsigned char)in[i]);
353 if (v < 0) return -1;
354 acc = (acc << 6) | (uint32_t)v;
355 bits += 6;
356 if (bits >= 8) {
357 bits -= 8;
358 if (used >= out_cap) return -2;
359 out[used++] = (uint8_t)(acc >> bits);
360 }
361 }
362 *out_len = used;
363 return 0;
364}

622int cwist_grpc_decode_message(const void *frame, size_t frame_len, cwist_grpc_message *out) {
623 if (!frame || !out || frame_len < 5) return -1;
624 const uint8_t *p = (const uint8_t *)frame;
625 uint32_t len =
626 ((uint32_t)p[1] << 24) | ((uint32_t)p[2] << 16) | ((uint32_t)p[3] << 8) | (uint32_t)p[4];
627 if ((size_t)len > frame_len - 5) return -1;
628 if ((size_t)len != frame_len - 5) return -1;
629
630 out->compressed = p[0];
631 out->data = p + 5;
632 out->len = (size_t)len;
633 return 0;

677int cwist_grpc_encode_message(const void *payload, size_t payload_len, uint8_t compressed,
678 uint8_t **out, size_t *out_len) {
679 if (!out || !out_len || (payload_len > 0 && !payload)) return -1;
680 if (payload_len > UINT32_MAX) return -1;
681
682 uint8_t *buf = (uint8_t *)cwist_alloc(payload_len + 5);
683 if (!buf) return -1;
684 buf[0] = compressed ? 1 : 0;
685 buf[1] = (uint8_t)((payload_len >> 24) & 0xff);
686 buf[2] = (uint8_t)((payload_len >> 16) & 0xff);
687 buf[3] = (uint8_t)((payload_len >> 8) & 0xff);
688 buf[4] = (uint8_t)(payload_len & 0xff);
689 if (payload_len > 0) memcpy(buf + 5, payload, payload_len);
690
691 *out = buf;
692 *out_len = payload_len + 5;
693 return 0;
694}

734int cwist_grpc_decoder_feed(cwist_grpc_decoder *decoder, const void *data, size_t len,
735 cwist_grpc_message_callback callback, void *ctx) {
736 if (!decoder || (!data && len) || !callback) return -1;
737 const uint8_t *input = data;
738 while (len) {
739 if (decoder->header_len < sizeof(decoder->header)) {
740 size_t take = sizeof(decoder->header) - decoder->header_len;
741 if (take > len) take = len;
742 memcpy(decoder->header + decoder->header_len, input, take);
743 decoder->header_len += take;
744 input += take;
745 len -= take;
746 if (decoder->header_len < sizeof(decoder->header)) continue;
747 decoder->compressed = decoder->header[0];
748 decoder->payload_len = ((size_t)decoder->header[1] << 24) |
749 ((size_t)decoder->header[2] << 16) |
750 ((size_t)decoder->header[3] << 8) | (size_t)decoder->header[4];
751 if (decoder->compressed > 1 || decoder->payload_len > decoder->max_message_size)
752 return -1;
753 if (decoder->payload_len) {
754 decoder->payload = cwist_alloc(decoder->payload_len);
755 if (!decoder->payload) return -1;
756 }
757 }
758 size_t take = decoder->payload_len - decoder->payload_used;
759 if (take > len) take = len;
760 if (take) memcpy(decoder->payload + decoder->payload_used, input, take);
761 decoder->payload_used += take;
762 input += take;
763 len -= take;
764 if (decoder->payload_used != decoder->payload_len) continue;
765 cwist_grpc_message message = {decoder->compressed, decoder->payload, decoder->payload_len};
766 if (callback(ctx, &message) != 0) return -1;
767 cwist_free(decoder->payload);
768 decoder->payload = NULL;
769 decoder->payload_len = decoder->payload_used = decoder->header_len = 0;
770 }
771 return 0;
772}

836int cwist_grpc_stream_send(cwist_grpc_stream *stream, const void *payload, size_t payload_len) {
837 if (!stream) return -1;
838 int use_gzip = 0;
839 if (stream->session) {
840 use_gzip = (((cwist_grpc_session *)stream->session)->resp_encoding == 1);
841 } else if (stream->req) {
842 use_gzip = grpc_client_accepts_gzip(stream->req);
843 }
844
845 uint8_t *frame = NULL;
846 size_t frame_len = 0;
847 if (use_gzip && payload_len > 0) {
848 uint8_t *cbuf = NULL;
849 size_t clen = 0;
850 if (grpc_gzip_deflate(payload, payload_len, &cbuf, &clen) == 0) {
851 int enc_rc = cwist_grpc_encode_message(cbuf, clen, 1, &frame, &frame_len);
852 cwist_free(cbuf);
853 if (enc_rc != 0) {
854 stream->status = CWIST_GRPC_INTERNAL;
855 stream->status_message = "failed to encode gRPC stream message";
856 return -1;
857 }
858 } else {
859 if (cwist_grpc_encode_message(payload, payload_len, 0, &frame, &frame_len) != 0) {
860 stream->status = CWIST_GRPC_INTERNAL;
861 stream->status_message = "failed to encode gRPC stream message";
862 return -1;
863 }
864 }
865 } else {
866 if (cwist_grpc_encode_message(payload, payload_len, 0, &frame, &frame_len) != 0) {
867 stream->status = CWIST_GRPC_INTERNAL;
868 stream->status_message = "failed to encode gRPC stream message";
869 return -1;
870 }
871 }
872 if (stream->session) {
873 /* Transport-backed stream: emit a DATA frame immediately. */
874 if (!stream->write_frame ||
875 stream->write_frame(stream->write_frame_ctx, frame, frame_len, 0) != 0) {
876 cwist_free(frame);
877 stream->status = CWIST_GRPC_UNAVAILABLE;
878 stream->status_message = "failed to write gRPC DATA frame";
879 return -1;
880 }
881 cwist_free(frame);
882 return 0;
883 }
884 if (!stream->res || !stream->res->body) {
885 cwist_free(frame);
886 return -1;
887 }
888 if (stream->write_frame &&
889 stream->write_frame(stream->write_frame_ctx, frame, frame_len, 0) != 0) {
890 cwist_free(frame);
891 stream->status = CWIST_GRPC_UNAVAILABLE;
892 stream->status_message = "failed to write gRPC DATA frame";
893 return -1;
894 }
895 cwist_error_t err = cwist_sstring_append_len(stream->res->body, (char *)frame, frame_len);
896 cwist_free(frame);
897 bool appended = cwist_error_is_ok(&err);
898 cwist_error_dispose(&err);
899 if (!appended) {
900 stream->status = CWIST_GRPC_INTERNAL;
901 stream->status_message = "failed to append gRPC stream message";
902 return -1;
903 }
904 return 0;
905}

1080static void grpc_health_watch(cwist_grpc_stream *stream, void *ctx) {
1081 cwist_grpc_health_registry *reg = ctx;
1082 char *service = NULL;
1083 if (stream->session) {
1084 /* Transport path: the HealthCheckRequest arrives as the first
1085 * (and only) message; the client then half-closes. */
1086 cwist_grpc_message msg;
1087 int rc = cwist_grpc_stream_recv(stream, &msg);
1088 if (rc < 0) return; /* cancelled before the request arrived */
1089 if (rc > 0) service = grpc_health_service_copy(&msg);
1090 } else {
1091 cwist_grpc_message empty = {0, NULL, 0};
1092 const cwist_grpc_message *message = stream->message_count ? &stream->messages[0] : &empty;
1093 service = grpc_health_service_copy(message);
1094 }
1095
1096 /* Send the current status immediately, as the protocol requires. */
1097 if (grpc_health_send_status(stream, reg, service) != 0) goto out;
1098 if (!stream->session) goto out; /* buffered dispatch: single snapshot */
1099
1100 atomic_fetch_add(&reg->watchers, 1);
1101 for (;;) {
1102 int changed = 0;
1103 pthread_mutex_lock(&reg->mu);
1104 uint64_t gen = reg->generation;
1105 while (reg->generation == gen && !cwist_grpc_stream_cancelled(stream) &&
1106 cwist_grpc_stream_deadline_remaining_ms(stream) > 0) {
1107 /* Poll cancellation/deadline on a short tick; status changes
1108 * wake us immediately through the registry condvar. */
1109 struct timespec ts;
1110 clock_gettime(CLOCK_REALTIME, &ts);
1111 ts.tv_nsec += 100000000L;
1112 if (ts.tv_nsec >= 1000000000L) {
1113 ts.tv_sec++;
1114 ts.tv_nsec -= 1000000000L;
1115 }
1116 pthread_cond_timedwait(&reg->cond, &reg->mu, &ts);
1117 }
1118 changed = reg->generation != gen;
1119 pthread_mutex_unlock(&reg->mu);
1120 if (!changed) break; /* cancelled or deadline expired */
1121 if (grpc_health_send_status(stream, reg, service) != 0) break;
1122 }
1123 atomic_fetch_sub(&reg->watchers, 1);
1124out:
1125 cwist_free(service);
1126}

1634static void grpc_session_send_trailers(cwist_grpc_session *session, cwist_grpc_status_t status,
1635 const char *message) {
1636 char status_buf[16];
1637 snprintf(status_buf, sizeof(status_buf), "%d", (int)status);
1638 char pushback_buf[20];
1639 if (session->stream.retry_pushback_set)
1640 snprintf(pushback_buf, sizeof(pushback_buf), "%d", (int)session->stream.retry_pushback_ms);
1641 pthread_mutex_lock(&session->wmu);
1642 if (session->h2s && !session->trailers_sent) {
1643 if (!session->headers_sent && status != CWIST_GRPC_OK) {
1644 /* Trailers-Only (PROTOCOL-HTTP2, gRFC A6): the whole response is
1645 * one HEADERS frame with END_STREAM. Response-Headers never went
1646 * out, so the RPC stays retryable for conforming clients. */
1647 cwist_http2_header hdrs[6];
1648 size_t count = 0;
1649 hdrs[count++] = (cwist_http2_header){"content-type", "application/grpc"};
1650 hdrs[count++] = (cwist_http2_header){"grpc-accept-encoding", "gzip, identity"};
1651 hdrs[count++] = (cwist_http2_header){"grpc-status", status_buf};
1652 if (message) hdrs[count++] = (cwist_http2_header){"grpc-message", message};
1653 if (session->stream.retry_pushback_set)
1654 hdrs[count++] = (cwist_http2_header){"grpc-retry-pushback-ms", pushback_buf};
1655 if (cwist_http2_stream_send_headers(session->h2s, 200, hdrs, count, 1) == 0)
1656 session->headers_sent = 1;
1657 session->trailers_sent = 1;
1658 } else if (grpc_session_send_headers_locked(session) == 0) {
1659 cwist_http2_header pairs[3];
1660 size_t count = 0;
1661 pairs[count].name = "grpc-status";
1662 pairs[count].value = status_buf;
1663 count++;
1664 if (message) {
1665 pairs[count].name = "grpc-message";
1666 pairs[count].value = message;
1667 count++;
1668 }
1669 if (session->stream.retry_pushback_set) {
1670 pairs[count].name = "grpc-retry-pushback-ms";
1671 pairs[count].value = pushback_buf;
1672 count++;
1673 }
1674 (void)cwist_http2_stream_send_trailers(session->h2s, pairs, count);
1675 session->trailers_sent = 1;
1676 } else {
1677 session->trailers_sent = 1; /* transport broken; do not retry */
1678 }
1679 }
1680 pthread_mutex_unlock(&session->wmu);
1681}

1738int cwist_grpc_stream_recv(cwist_grpc_stream *stream, cwist_grpc_message *out) {
1739 if (!stream || !out) return -1;
1740
1741 if (!stream->session) {
1742 /* Buffered path: pop the pre-decoded message array. */
1743 if (stream->deadline_ms && grpc_now_ms() >= stream->deadline_ms) {
1744 stream->cancelled = 1;
1745 return -1;
1746 }
1747 if (stream->recv_pos >= stream->message_count) return 0;
1748 *out = stream->messages[stream->recv_pos++];
1749 return 1;
1750 }
1751
1752 cwist_grpc_session *session = stream->session;
1753 pthread_mutex_lock(&session->mu);
1754 for (;;) {
1755 if (session->qhead) {
1756 grpc_qnode *node = session->qhead;
1757 session->qhead = node->next;
1758 if (!session->qhead) session->qtail = NULL;
1759 pthread_mutex_unlock(&session->mu);
1760 cwist_free(session->recv_buf);
1761 session->recv_buf = node->data;
1762 out->compressed = 0;
1763 out->data = node->data;
1764 out->len = node->len;
1765 cwist_free(node);
1766 return 1;
1767 }
1768 if (session->cancelled) {
1769 pthread_mutex_unlock(&session->mu);
1770 return -1;
1771 }
1772 if (session->eof) {
1773 pthread_mutex_unlock(&session->mu);
1774 return 0;
1775 }
1776 if (stream->deadline_ms) {
1777 uint64_t now = grpc_now_ms();
1778 if (now >= stream->deadline_ms) {
1779 session->cancelled = 1;
1780 stream->cancelled = 1;
1781 /* Report the real outcome so a handler that simply closes
1782 * with stream->status still emits DEADLINE_EXCEEDED. */
1783 stream->status = CWIST_GRPC_DEADLINE_EXCEEDED;
1784 stream->status_message = "deadline exceeded";
1785 pthread_mutex_unlock(&session->mu);
1786 return -1;
1787 }
1788 uint64_t left = stream->deadline_ms - now;
1789 struct timespec ts;
1790 struct timespec rt;
1791 clock_gettime(CLOCK_REALTIME, &rt);
1792 ts.tv_sec = rt.tv_sec + (time_t)(left / 1000);
1793 ts.tv_nsec = rt.tv_nsec + (long)((left % 1000) * 1000000);
1794 if (ts.tv_nsec >= 1000000000L) {
1795 ts.tv_sec++;
1796 ts.tv_nsec -= 1000000000L;
1797 }
1798 pthread_cond_timedwait(&session->cond, &session->mu, &ts);
1799 } else {
1800 pthread_cond_wait(&session->cond, &session->mu);
1801 }
1802 }
1803}

1874static void *grpc_h2_on_headers(void *conn_ctx, cwist_http_request *req, cwist_h2_stream *stream) {
1875 cwist_grpc_h2_conn_ctx *ctx = conn_ctx;
1876 if (!ctx || !ctx->app || !req || !req->path || !req->path->data) return NULL;
1877
1878 cwist_grpc_route *route = grpc_find_route(ctx->app, req->path->data);
1879 /* Unary-only builtins (health Check) and reflection stay on the buffered
1880 * dispatch path; the health Watch builtin streams live. */
1881 if (!route || !route->stream_handler || (route->builtin && route->builtin != 2)) return NULL;
1882
1883 const char *ct = grpc_header_get(req, "content-type");
1884 if (!grpc_content_type_is_grpc(ct)) return NULL;
1885
1886 int encoding = grpc_request_encoding(req);
1887 if (encoding < 0) return NULL; /* buffered dispatch replies UNIMPLEMENTED */
1888
1889 cwist_grpc_session *session = (cwist_grpc_session *)cwist_alloc(sizeof(*session));
1890 if (!session) return NULL;
1891 memset(session, 0, sizeof(*session));
1892 pthread_mutex_init(&session->mu, NULL);
1893 pthread_cond_init(&session->cond, NULL);
1894 pthread_mutex_init(&session->wmu, NULL);
1895 session->h2s = stream;
1896 session->conn_ctx = ctx;
1897 session->encoding = encoding;
1898 cwist_grpc_decoder_init(&session->decoder, 0);
1899 atomic_store(&session->refs, 2); /* transport + handler thread */
1900
1901 session->stream.req = req;
1902 session->stream.res = NULL;
1903 session->stream.messages = NULL;
1904 session->stream.message_count = 0;
1905 session->stream.status = CWIST_GRPC_OK;
1906 session->stream.status_message = NULL;
1907 session->stream.closed = 0;
1908 session->stream.write_frame = grpc_session_write_frame;
1909 session->stream.write_frame_ctx = session;
1910 session->stream.session = session;
1911 session->stream.cancelled = 0;
1912 session->stream.recv_pos = 0;
1913 const char *timeout = grpc_header_get(req, "grpc-timeout");
1914 if (timeout) {
1915 uint64_t timeout_ms = 0;
1916 if (cwist_grpc_parse_timeout(timeout, &timeout_ms) == 0)
1917 session->stream.deadline_ms = grpc_now_ms() + timeout_ms;
1918 }
1919 session->handler = route->stream_handler;
1920 session->user_ctx = route->user_ctx;
1921 session->resp_encoding = grpc_client_accepts_gzip(req) ? 1 : 0;
1922
1923 /* Response-Headers are deliberately not sent yet (gRFC A6): they go out
1924 * with the first DATA frame, or the call ends in Trailers-Only form so
1925 * conforming clients may retry the RPC. */
1926
1927 pthread_mutex_lock(&ctx->mu);
1928 ctx->refs++;
1929 session->next = ctx->sessions;
1930 ctx->sessions = session;
1931 pthread_mutex_unlock(&ctx->mu);
1932
1933 pthread_t tid;
1934 if (pthread_create(&tid, NULL, grpc_session_thread, session) != 0) {
1935 pthread_mutex_lock(&ctx->mu);
1936 ctx->refs--;
1937 cwist_grpc_session **pp = &ctx->sessions;
1938 while (*pp && *pp != session) pp = &(*pp)->next;
1939 if (*pp) *pp = session->next;
1940 pthread_mutex_unlock(&ctx->mu);
1941 pthread_mutex_destroy(&session->mu);
1942 pthread_cond_destroy(&session->cond);
1943 pthread_mutex_destroy(&session->wmu);
1944 cwist_grpc_decoder_destroy(&session->decoder);
1945 cwist_free(session);
1946 return NULL;
1947 }
1948 pthread_detach(tid);
1949 return session;
1950}
