51static void generate_request_id(char id[17]) {
52 static const char charset[] = "abcdefghijklmnopqrstuvwxyz0123456789";
53
54 pthread_mutex_lock(&rid_mutex);
55#ifndef CWIST_PLATFORM_NO_PROC
56 pid_t pid = getpid();
57 if (!rid_keyed || rid_pid != pid) {
58 cwist_generate_hash_seed(rid_key);
59 rid_pid = pid;
60 rid_keyed = true;
61 }
62#else
63 if (!rid_keyed) {
64 cwist_generate_hash_seed(rid_key);
65 rid_keyed = true;
66 }
67#endif
68 uint64_t n = rid_counter++;
69 uint8_t key[16];
70 memcpy(key, rid_key, sizeof(key));
71 pthread_mutex_unlock(&rid_mutex);
72
73 for (int half = 0; half < 2; half++) {
74 uint64_t in = (n << 1) | (uint64_t)half;
75 uint64_t v = siphash24(&in, sizeof(in), key);
76 for (int i = 0; i < 8; i++) {
77 id[half * 8 + i] = charset[v % (sizeof(charset) - 1)];
78 v /= (sizeof(charset) - 1);
79 }
80 }
81 id[16] = '\0';
82}

84static bool request_id_is_valid(const char *rid) {
85 if (!rid || !*rid) return false;
86 size_t len = strlen(rid);
87 if (len > 128) return false;
88 for (const unsigned char *p = (const unsigned char *)rid; *p; p++) {
89 if (!isalnum(*p) && *p != '.' && *p != '_' && *p != '-') return false;

164static char *access_log_escape(const char *input, bool json) {
165 if (!input) return cwist_strdup("-");
166 size_t len = strlen(input);
167 if (len > (SIZE_MAX - 1) / 6) return NULL;
168 char *escaped = cwist_alloc(len * 6 + 1);
169 if (!escaped) return NULL;
170 size_t out = 0;
171 for (size_t i = 0; i < len; i++) {
172 unsigned char c = (unsigned char)input[i];
173 if (c == '"' || c == '\\') {
174 escaped[out++] = '\\';
175 escaped[out++] = (char)c;
176 } else if (c < 0x20 || c == 0x7f) {
177 if (json) {
178 out += (size_t)snprintf(escaped + out, 7, "\\u%04x", c);
179 } else {
180 out += (size_t)snprintf(escaped + out, 5, "\\x%02x", c);
181 }
182 } else {
183 escaped[out++] = (char)c;
184 }
185 }
186 escaped[out] = '\0';
187 return escaped;
188}

463static void rate_limit_ip_key(const char *input, char output[46]) {
464#ifndef CWIST_PLATFORM_NO_SOCKETS

467 memset(addr.s6_addr + 8, 0, 8);

488static void cwist_mw_rate_limit_ip_handler(cwist_http_request *req, cwist_http_response *res,
489 cwist_handler_func next, int slot, int rpm) {
490 if (rpm <= 0) rpm = 60;
491
492 cwist_sstring *ip = cwist_http_request_client_ip(req);
493 if (!ip) {
494 next(req, res);
495 return;
496 }
497 char key[46];
498 rate_limit_ip_key(ip->data, key);
499
500 uint64_t now = ttak_get_tick_count();
501 pthread_mutex_lock(&rate_mutex);
502 ip_bucket_t *found = NULL, *free_slot = NULL, *oldest = NULL;
503 for (int i = 0; i < ip_bucket_count; i++) {
504 ip_bucket_t *entry = &ip_buckets[i];
505 if (!entry->active) {
506 if (!free_slot) free_slot = entry;
507 continue;
508 }
509 if (now - entry->last_used_ms > IP_BUCKET_IDLE_MS) {
510 entry->active = false;
511 if (!free_slot) free_slot = entry;
512 continue;
513 }
514 if (!oldest || entry->last_used_ms < oldest->last_used_ms) oldest = entry;
515 if (entry->slot == slot && strcmp(entry->ip, key) == 0) {
516 found = entry;
517 break;
518 }
519 }
520
521 if (!found) {
522 if (free_slot)
523 found = free_slot;
524 else if (ip_bucket_count < MAX_IP_BUCKETS)
525 found = &ip_buckets[ip_bucket_count++];
526 else
527 found = oldest;
528 if (found) {
529 snprintf(found->ip, sizeof(found->ip), "%s", key);
530 found->slot = slot;
531 found->active = true;
532 double rate = rpm / 60.0;
533 ttak_token_bucket_init(&found->bucket, rate, (double)rpm);
534 }
535 }
536 if (found) found->last_used_ms = now;
537
538 bool allowed = !found || ttak_token_bucket_consume(&found->bucket, 1.0);
539 /* Seconds until one token is back, from the bucket's own state and
540 * rounded up: "60 / rpm" rounded down, so clients retried early (#989). */
541 long retry_after = 1;
542 if (!allowed && found->bucket.refill_rate > 0) {
543 double missing = 1.0 - found->bucket.tokens;
544 if (missing < 0) missing = 0;
545 double secs = missing / found->bucket.refill_rate;
546 if (secs > 86400.0) secs = 86400.0;
547 retry_after = (long)secs;
548 if ((double)retry_after < secs) retry_after++;
549 if (retry_after < 1) retry_after = 1;
550 }
551 cwist_sstring_destroy(ip);
552 pthread_mutex_unlock(&rate_mutex);
553
554 if (!allowed) {
555 res->status_code = 429;
556 cwist_sstring_assign(res->body, "Too Many Requests");
557 char retry_buf[24];
558 snprintf(retry_buf, sizeof(retry_buf), "%ld", retry_after);
559 cwist_http_header_add(&res->headers, "Retry-After", retry_buf);
560 return;
561 }
562
563 next(req, res);
564}

673/*
674 * The decoded claims of a request reach downstream handlers through a
675 * per-thread stack of contexts, one per JWT middleware currently running,
676 * each tagged with its request. req->private_data is not used: inside the
677 * app's middleware chain it holds the chain's own state, which next() and
678 * the final route handler read.
679 *
680 * Each context is stack-allocated in the middleware frame and is only valid
681 * while the rest of the chain runs synchronously inside next(); claims are
682 * destroyed when next() returns, so no pointer to them escapes to
683 * asynchronous code.
684 */
685
686typedef struct cwist_jwt_ctx {
687 const cwist_http_request *req;
688 cwist_jwt_claims *claims;
689 struct cwist_jwt_ctx *outer;
690} cwist_jwt_ctx_t;

692#if (defined(__STDC_VERSION__) && __STDC_VERSION__ >= 201112L)
693static _Thread_local cwist_jwt_ctx_t *t_jwt_ctx = NULL;
694#elif defined(__GNUC__) || defined(__clang__)
695static __thread cwist_jwt_ctx_t *t_jwt_ctx = NULL;
696#endif

706static void cwist_mw_jwt_auth_handler(cwist_http_request *req, cwist_http_response *res,
707 cwist_handler_func next, const char *secret) {
708 /* Extract Bearer token from Authorization header */
709 char *auth_header = cwist_http_header_get(req->headers, "Authorization");
710 if (!auth_header) {
711 res->status_code = CWIST_HTTP_UNAUTHORIZED;
712 cwist_sstring_assign(res->body, "{\"error\":\"Missing Authorization header\"}");
713 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
714 cwist_http_header_add(&res->headers, "WWW-Authenticate", "Bearer realm=\"api\"");
715 return;
716 }
717
718 /* Expect "Bearer <token>" (scheme is case-insensitive per RFC 6750) */
719 if (strncasecmp(auth_header, "Bearer ", 7) != 0) {
720 res->status_code = CWIST_HTTP_UNAUTHORIZED;
721 cwist_sstring_assign(res->body, "{\"error\":\"Invalid Authorization scheme\"}");
722 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
723 cwist_http_header_add(&res->headers, "WWW-Authenticate",
724 "Bearer realm=\"api\", error=\"invalid_request\"");
725 return;
726 }
727
728 const char *token = auth_header + 7;
729 while (*token == ' ' || *token == '\t') token++;
730
731 cwist_jwt_claims *claims = cwist_jwt_verify(token, secret);
732 if (!claims) {
733 res->status_code = CWIST_HTTP_UNAUTHORIZED;
734 cwist_sstring_assign(res->body, "{\"error\":\"Invalid or expired token\"}");
735 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
736 cwist_http_header_add(&res->headers, "WWW-Authenticate",
737 "Bearer realm=\"api\", error=\"invalid_token\"");
738 return;
739 }
740
741 /* Publish the claims to downstream handlers on this thread for the
742 * duration of the rest of the chain. */
743 cwist_jwt_ctx_t ctx = {.req = req, .claims = claims, .outer = t_jwt_ctx};
744 t_jwt_ctx = &ctx;
745
746 if (next) next(req, res);
747
748 t_jwt_ctx = ctx.outer;
749 cwist_jwt_claims_destroy(claims);
750}

796cwist_middleware_func cwist_mw_jwt_auth(const char *secret) {
797#ifdef CWIST_JWT
798 if (!secret) return NULL;
799
800 pthread_mutex_lock(&s_jwt_cfg_mutex);
801
802 /* Reuse existing slot for the same secret */
803 for (int i = 0; i < s_jwt_cfg_count; i++) {
804 if (s_jwt_cfgs[i].secret && strcmp(s_jwt_cfgs[i].secret, secret) == 0) {
805 pthread_mutex_unlock(&s_jwt_cfg_mutex);
806 return s_jwt_wrappers[i];
807 }
808 }
809
810 if (s_jwt_cfg_count >= CWIST_JWT_MAX_SECRETS) {
811 pthread_mutex_unlock(&s_jwt_cfg_mutex);
812 fprintf(stderr, "[CWIST] cwist_mw_jwt_auth: maximum secret slots exceeded\n");
813 return NULL;
814 }
815
816 int slot = s_jwt_cfg_count++;
817 s_jwt_cfgs[slot].secret = secret;
818
819 pthread_mutex_unlock(&s_jwt_cfg_mutex);
820 return s_jwt_wrappers[slot];
821#else
822 /* JWT compiled out (CWIST_JWT=0): no wrapper can verify tokens. */
823 (void)secret;
824 return NULL;
825#endif
826}
