365static void cwist_mw_rate_limit_ip_handler(cwist_http_request *req, cwist_http_response *res,
366 cwist_handler_func next, int rpm) {
367 if (rpm <= 0) rpm = 60;
368
369 cwist_sstring *ip = cwist_get_client_ip_from_fd(req->client_fd);
370 if (!ip) {
371 next(req, res);
372 return;
373 }
374
375 pthread_mutex_lock(&rate_mutex);
376 ip_bucket_t *found = NULL;
377 for (int i = 0; i < ip_bucket_count; i++) {
378 if (ip_buckets[i].active && strcmp(ip_buckets[i].ip, ip->data) == 0) {
379 found = &ip_buckets[i];
380 break;
381 }
382 }
383
384 if (!found && ip_bucket_count < MAX_IP_BUCKETS) {
385 found = &ip_buckets[ip_bucket_count++];
386 strncpy(found->ip, ip->data, sizeof(found->ip) - 1);
387 found->ip[sizeof(found->ip) - 1] = '\0';
388 found->active = true;
389 double rate = rpm / 60.0;
390 ttak_token_bucket_init(&found->bucket, rate, (double)rpm);
391 }
392
393 /* Fail-open: if the IP table is full we cannot track this client, so
394 * allow the request rather than permanently blocking unknown IPs (which
395 * would let an attacker exhaust the 1024 slots with spoofed addresses
396 * and then deny service to all legitimate users). */
397 bool allowed = (found == NULL);
398 if (found) {
399 allowed = ttak_token_bucket_consume(&found->bucket, 1.0);
400 }
401 cwist_sstring_destroy(ip);
402 pthread_mutex_unlock(&rate_mutex);
403
404 if (!allowed) {
405 res->status_code = 429;
406 cwist_sstring_assign(res->body, "Too Many Requests");
407 char retry_buf[16];
408 snprintf(retry_buf, sizeof(retry_buf), "%d", 60 / rpm > 0 ? 60 / rpm : 1);
409 cwist_http_header_add(&res->headers, "Retry-After", retry_buf);
410 return;
411 }
412
413 next(req, res);
414}

531/*
532 * The decoded claims of a request reach downstream handlers through a
533 * per-thread stack of contexts, one per JWT middleware currently running,
534 * each tagged with its request. req->private_data is not used: inside the
535 * app's middleware chain it holds the chain's own state, which next() and
536 * the final route handler read.
537 *
538 * Each context is stack-allocated in the middleware frame and is only valid
539 * while the rest of the chain runs synchronously inside next(); claims are
540 * destroyed when next() returns, so no pointer to them escapes to
541 * asynchronous code.
542 */
543
544typedef struct cwist_jwt_ctx {
545 const cwist_http_request *req;
546 cwist_jwt_claims *claims;
547 struct cwist_jwt_ctx *outer;
548} cwist_jwt_ctx_t;
549
550#if (defined(__STDC_VERSION__) && __STDC_VERSION__ >= 201112L)
551static _Thread_local cwist_jwt_ctx_t *t_jwt_ctx = NULL;
552#elif defined(__GNUC__) || defined(__clang__)
553static __thread cwist_jwt_ctx_t *t_jwt_ctx = NULL;
554#endif

564static void cwist_mw_jwt_auth_handler(cwist_http_request *req, cwist_http_response *res,
565 cwist_handler_func next, const char *secret) {
566 /* Extract Bearer token from Authorization header */
567 char *auth_header = cwist_http_header_get(req->headers, "Authorization");
568 if (!auth_header) {
569 res->status_code = CWIST_HTTP_UNAUTHORIZED;
570 cwist_sstring_assign(res->body, "{\"error\":\"Missing Authorization header\"}");
571 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
572 return;
573 }
574
575 /* Expect "Bearer <token>" (scheme is case-insensitive per RFC 6750) */
576 if (strncasecmp(auth_header, "Bearer ", 7) != 0) {
577 res->status_code = CWIST_HTTP_UNAUTHORIZED;
578 cwist_sstring_assign(res->body, "{\"error\":\"Invalid Authorization scheme\"}");
579 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
580 return;
581 }
582
583 const char *token = auth_header + 7;
584 while (*token == ' ' || *token == '\t') token++;
585
586 cwist_jwt_claims *claims = cwist_jwt_verify(token, secret);
587 if (!claims) {
588 res->status_code = CWIST_HTTP_UNAUTHORIZED;
589 cwist_sstring_assign(res->body, "{\"error\":\"Invalid or expired token\"}");
590 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
591 return;
592 }
593
594 /* Publish the claims to downstream handlers on this thread for the
595 * duration of the rest of the chain. */
596 cwist_jwt_ctx_t ctx = {.req = req, .claims = claims, .outer = t_jwt_ctx};
597 t_jwt_ctx = &ctx;
598
599 if (next) next(req, res);
600
601 t_jwt_ctx = ctx.outer;
602 cwist_jwt_claims_destroy(claims);
603}

650#ifdef CWIST_JWT
651 if (!secret) return NULL;

653 pthread_mutex_lock(&s_jwt_cfg_mutex);
654
655 /* Reuse existing slot for the same secret */
656 for (int i = 0; i < s_jwt_cfg_count; i++) {
657 if (s_jwt_cfgs[i].secret && strcmp(s_jwt_cfgs[i].secret, secret) == 0) {
