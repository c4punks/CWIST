50static bool request_id_is_valid(const char *rid) {
51 if (!rid || !*rid) return false;
52 size_t len = strlen(rid);
53 if (len > 128) return false;
54 for (const unsigned char *p = (const unsigned char *)rid; *p; p++) {
55 if (!isalnum(*p) && *p != '.' && *p != '_' && *p != '-') return false;
56 }

376static void cwist_mw_rate_limit_ip_handler(cwist_http_request *req, cwist_http_response *res,
377 cwist_handler_func next, int rpm) {
378 if (rpm <= 0) rpm = 60;
379
380 cwist_sstring *ip = cwist_get_client_ip_from_fd(req->client_fd);
381 if (!ip) {
382 next(req, res);
383 return;
384 }
385
386 pthread_mutex_lock(&rate_mutex);
387 ip_bucket_t *found = NULL;
388 for (int i = 0; i < ip_bucket_count; i++) {
389 if (ip_buckets[i].active && strcmp(ip_buckets[i].ip, ip->data) == 0) {
390 found = &ip_buckets[i];
391 break;
392 }
393 }
394
395 if (!found && ip_bucket_count < MAX_IP_BUCKETS) {
396 found = &ip_buckets[ip_bucket_count++];
397 strncpy(found->ip, ip->data, sizeof(found->ip) - 1);
398 found->ip[sizeof(found->ip) - 1] = '\0';
399 found->active = true;
400 double rate = rpm / 60.0;
401 ttak_token_bucket_init(&found->bucket, rate, (double)rpm);
402 }
403
404 /* Fail-open: if the IP table is full we cannot track this client, so
405 * allow the request rather than permanently blocking unknown IPs (which
406 * would let an attacker exhaust the 1024 slots with spoofed addresses
407 * and then deny service to all legitimate users). */
408 bool allowed = (found == NULL);
409 if (found) {
410 allowed = ttak_token_bucket_consume(&found->bucket, 1.0);
411 }
412 cwist_sstring_destroy(ip);
413 pthread_mutex_unlock(&rate_mutex);
414
415 if (!allowed) {
416 res->status_code = 429;
417 cwist_sstring_assign(res->body, "Too Many Requests");
418 char retry_buf[16];
419 snprintf(retry_buf, sizeof(retry_buf), "%d", 60 / rpm > 0 ? 60 / rpm : 1);
420 cwist_http_header_add(&res->headers, "Retry-After", retry_buf);
421 return;
422 }
423
424 next(req, res);
425}

542/*
543 * The decoded claims of a request reach downstream handlers through a
544 * per-thread stack of contexts, one per JWT middleware currently running,
545 * each tagged with its request. req->private_data is not used: inside the
546 * app's middleware chain it holds the chain's own state, which next() and
547 * the final route handler read.
548 *
549 * Each context is stack-allocated in the middleware frame and is only valid
550 * while the rest of the chain runs synchronously inside next(); claims are
551 * destroyed when next() returns, so no pointer to them escapes to
552 * asynchronous code.
553 */
554
555typedef struct cwist_jwt_ctx {
556 const cwist_http_request *req;
557 cwist_jwt_claims *claims;
558 struct cwist_jwt_ctx *outer;
559} cwist_jwt_ctx_t;
560
561#if (defined(__STDC_VERSION__) && __STDC_VERSION__ >= 201112L)
562static _Thread_local cwist_jwt_ctx_t *t_jwt_ctx = NULL;
563#elif defined(__GNUC__) || defined(__clang__)
564static __thread cwist_jwt_ctx_t *t_jwt_ctx = NULL;
565#endif

575static void cwist_mw_jwt_auth_handler(cwist_http_request *req, cwist_http_response *res,
576 cwist_handler_func next, const char *secret) {
577 /* Extract Bearer token from Authorization header */
578 char *auth_header = cwist_http_header_get(req->headers, "Authorization");
579 if (!auth_header) {
580 res->status_code = CWIST_HTTP_UNAUTHORIZED;
581 cwist_sstring_assign(res->body, "{\"error\":\"Missing Authorization header\"}");
582 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
583 return;
584 }
585
586 /* Expect "Bearer <token>" (scheme is case-insensitive per RFC 6750) */
587 if (strncasecmp(auth_header, "Bearer ", 7) != 0) {
588 res->status_code = CWIST_HTTP_UNAUTHORIZED;
589 cwist_sstring_assign(res->body, "{\"error\":\"Invalid Authorization scheme\"}");
590 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
591 return;
592 }
593
594 const char *token = auth_header + 7;
595 while (*token == ' ' || *token == '\t') token++;
596
597 cwist_jwt_claims *claims = cwist_jwt_verify(token, secret);
598 if (!claims) {
599 res->status_code = CWIST_HTTP_UNAUTHORIZED;
600 cwist_sstring_assign(res->body, "{\"error\":\"Invalid or expired token\"}");
601 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
602 return;
603 }
604
605 /* Publish the claims to downstream handlers on this thread for the
606 * duration of the rest of the chain. */
607 cwist_jwt_ctx_t ctx = {.req = req, .claims = claims, .outer = t_jwt_ctx};
608 t_jwt_ctx = &ctx;
609
610 if (next) next(req, res);
611
612 t_jwt_ctx = ctx.outer;
613 cwist_jwt_claims_destroy(claims);
614}

660cwist_middleware_func cwist_mw_jwt_auth(const char *secret) {
661#ifdef CWIST_JWT
662 if (!secret) return NULL;
663
664 pthread_mutex_lock(&s_jwt_cfg_mutex);
665
666 /* Reuse existing slot for the same secret */
667 for (int i = 0; i < s_jwt_cfg_count; i++) {
668 if (s_jwt_cfgs[i].secret && strcmp(s_jwt_cfgs[i].secret, secret) == 0) {
669 pthread_mutex_unlock(&s_jwt_cfg_mutex);
670 return s_jwt_wrappers[i];
671 }
672 }
673
674 if (s_jwt_cfg_count >= CWIST_JWT_MAX_SECRETS) {
675 pthread_mutex_unlock(&s_jwt_cfg_mutex);
676 fprintf(stderr, "[CWIST] cwist_mw_jwt_auth: maximum secret slots exceeded\n");
677 return NULL;
678 }
679
680 int slot = s_jwt_cfg_count++;
681 s_jwt_cfgs[slot].secret = secret;
682
683 pthread_mutex_unlock(&s_jwt_cfg_mutex);
684 return s_jwt_wrappers[slot];
685#else
686 /* JWT compiled out (CWIST_JWT=0): no wrapper can verify tokens. */
687 (void)secret;
688 return NULL;
