428/* --- Event-driven (one-shot) connection path for the C1M reactor ---------
429 * The classic pool handler parks a worker thread on each keep-alive
430 * connection, capping concurrent connections at the thread count. The async
431 * path below never blocks on a read: the callback drains whatever arrived,
432 * serves every complete request, and rearms the fd, so one thread can hold
433 * hundreds of thousands of mostly-idle connections.
434 *
435 * Writes on the deferred-completion path are resumable: when a slow client
436 * saturates the socket, cwist_http_async_send_response deep-copies the unsent
437 * remainder and parks it on a one-shot POLLOUT slot instead of blocking the
438 * reactor thread in a poll() wait. Synchronous (non-deferred) responses
439 * still use the bounded poll wait inside cwist_http_send_response
440 * (CWIST_HTTP_TIMEOUT_MS). */
441typedef enum {
442 CWIST_ASYNC_CLOSE = 0, /* Close fd and release the connection. */
443 CWIST_ASYNC_REARM, /* Keep the connection; wait for more reads. */
444 CWIST_ASYNC_DETACH, /* Handler took ownership of fd (h2c, upgrades). */
445 CWIST_ASYNC_DEFER /* Completion/continuation owns fd and conn; leave untouched. */
446} cwist_async_action_t;
