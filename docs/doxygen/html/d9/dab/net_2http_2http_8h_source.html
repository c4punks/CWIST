454/* --- Event-driven (one-shot) connection path for the C1M reactor ---------
455 * The classic pool handler parks a worker thread on each keep-alive
456 * connection, capping concurrent connections at the thread count. The async
457 * path below never blocks on a read: the callback drains whatever arrived,
458 * serves every complete request, and rearms the fd, so one thread can hold
459 * hundreds of thousands of mostly-idle connections.
460 *
461 * Writes on the deferred-completion path are resumable: when a slow client
462 * saturates the socket, cwist_http_async_send_response deep-copies the unsent
463 * remainder and parks it on a one-shot POLLOUT slot instead of blocking the
464 * reactor thread in a poll() wait. Synchronous (non-deferred) responses
465 * still use the bounded poll wait inside cwist_http_send_response
466 * (CWIST_HTTP_TIMEOUT_MS). */
467typedef enum {
468 CWIST_ASYNC_CLOSE = 0, /* Close fd and release the connection. */
469 CWIST_ASYNC_REARM, /* Keep the connection; wait for more reads. */
470 CWIST_ASYNC_DETACH, /* Handler took ownership of fd (h2c, upgrades). */
471 CWIST_ASYNC_DEFER /* Completion/continuation owns fd and conn; leave untouched. */
472} cwist_async_action_t;

488 uint32_t worker_id; /* Assigned worker thread index for load tracking. */

492 /* RX receive path (Linux reactors only: a RECV SQE on an io_uring ring,
493 * or the reactor's own recv() on an epoll one; unused elsewhere).
494 * Wait-state discipline, all transitions on the
495 * reactor owner thread: at most one of {RECV SQE in flight, one-shot
496 * POLL armed} at any time, or NONE while buffered work is served.
497 * While rx_recv_inflight is true the stash buffer must not move or be
498 * consumed: the in-flight SQE references rbuf + len. */

502 * sends one request per idle period, so arming RECV first just burns an
503 * SQE before the POLL fallback). Cleared whenever a RECV stages bytes,
504 * so pipelining clients keep the SQE path. While set, waits go straight
505 * to the legacy POLL, making the steady-state op count identical to the
506 * legacy path for non-pipelining clients. */
507 bool rx_prefers_poll;
508 uint64_t rx_armed_ns; /* Latency-probe arm timestamp of the in-flight RECV. */
509 /* Env-gated kernel-arrival probe (CWIST_KERN_TS=1, Linux only, experiment

517 cwist_bdr_cursor_t bdr_cursor;
518 /* Hysteresis anchor for the idle stash shrink (CWIST_ASYNC_STASH_IDLE_SEC):
519 * monotonic time the rbuf/obuf capacity was last grown. While requests
520 * keep needing capacity the stash is never shrunk at re-arm; one idle
521 * window after the last growth it converges to the small floor. */
522 uint32_t stash_grown_sec;
