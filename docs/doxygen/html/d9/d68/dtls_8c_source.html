36int cwist_dtls_generate_cert(X509 **cert_out, EVP_PKEY **pkey_out) {
37 *cert_out = NULL;
38 *pkey_out = NULL;
39
40 EVP_PKEY_CTX *kctx = EVP_PKEY_CTX_new_id(EVP_PKEY_EC, NULL);
41 if (!kctx)
42 return -1;
43 if (EVP_PKEY_keygen_init(kctx) <= 0 || EVP_PKEY_CTX_set_ec_paramgen_curve_nid(kctx, NID_X9_62_prime256v1) <= 0) {
44 EVP_PKEY_CTX_free(kctx);
45 return -1;
46 }
47 EVP_PKEY *pkey = NULL;
48 if (EVP_PKEY_keygen(kctx, &pkey) <= 0) {
49 EVP_PKEY_CTX_free(kctx);
50 return -1;
51 }
52 EVP_PKEY_CTX_free(kctx);
53
54 X509 *cert = X509_new();
55 if (!cert) {
56 EVP_PKEY_free(pkey);
57 return -1;
58 }
59 X509_set_version(cert, 2);
60 ASN1_INTEGER_set(X509_get_serialNumber(cert), 1);
61 X509_gmtime_adj(X509_getm_notBefore(cert), 0);
62 X509_gmtime_adj(X509_getm_notAfter(cert), 60L * 60L * 24L * 30);
63 X509_set_pubkey(cert, pkey);
64 X509_NAME *name = X509_get_subject_name(cert);
65 X509_NAME_add_entry_by_txt(name, "CN", MBSTRING_ASC, (const uint8_t *)"cwist-webrtc", -1, -1, 0);
66 X509_set_issuer_name(cert, name);
67 if (X509_sign(cert, pkey, EVP_sha256()) == 0) {
68 X509_free(cert);
69 EVP_PKEY_free(pkey);
70 return -1;
71 }
72 *cert_out = cert;
73 *pkey_out = pkey;
74 return 0;
75}

90int cwist_dtls_fingerprint(X509 *cert, char *out, size_t cap) {
91 uint8_t der[2048];
92 int der_len = i2d_X509(cert, NULL);
93 if (der_len <= 0 || (size_t)der_len > sizeof(der))
94 return -1;
95 uint8_t *p = der;
96 i2d_X509(cert, &p);
97 uint8_t digest[EVP_MAX_MD_SIZE];
98 unsigned int dlen = 0;
99 if (!EVP_Digest(der, (size_t)der_len, digest, &dlen, EVP_sha256(), NULL))
100 return -1;
101 size_t off = 0;
102 for (unsigned int i = 0; i < dlen && off + 3 < cap; i++)
103 off += (size_t)snprintf(out + off, cap - off, "%s%02X", i ? ":" : "", digest[i]);
104 return off > 0 && off < cap ? 0 : -1;
105}

146/* ---- datagram BIO ----
147 *
148 * DTLS needs datagram semantics: one BIO_read returns exactly one datagram and
149 * one BIO_write is exactly one datagram. A memory BIO is a byte stream, so
150 * records written back to back get concatenated and a fixed-size read can cut
151 * one in half across two UDP packets. This BIO keeps the boundaries and also
152 * skips the extra copy through the memory buffer: reads hand over the datagram
153 * the event loop is processing, writes go straight to the ctx send batch. */
154
155static BIO_METHOD *g_dgram_method;
156static pthread_once_t g_dgram_once = PTHREAD_ONCE_INIT;
157
159static int dgram_bio_read(BIO *bio, char *out, int outl) {
160 struct cwist_webrtc_conn *conn = BIO_get_data(bio);
161 BIO_clear_retry_flags(bio);
162 if (!conn || !conn->dtls_in) {
163 BIO_set_retry_read(bio);
164 return -1;
165 }
166 size_t n = conn->dtls_in_len;
167 if (outl < 0 || n > (size_t)outl)
168 n = outl < 0 ? 0 : (size_t)outl; /* DTLS drops a truncated datagram. */
169 memcpy(out, conn->dtls_in, n);
170 conn->dtls_in = NULL;
171 conn->dtls_in_len = 0;
172 return (int)n;
173}

176static int dgram_bio_write(BIO *bio, const char *in, int inl) {
177 struct cwist_webrtc_conn *conn = BIO_get_data(bio);
178 BIO_clear_retry_flags(bio);
179 if (!conn || inl < 0) return -1;
180 cwist_webrtc_conn_dtls_out(conn, (const uint8_t *)in, (size_t)inl);
181 return inl;
182}

200static void dgram_method_init(void) {
201 BIO_METHOD *m = BIO_meth_new(BIO_get_new_index() | BIO_TYPE_SOURCE_SINK, "cwist-webrtc-dgram");
202 if (!m) return;
203 if (!BIO_meth_set_read(m, dgram_bio_read) || !BIO_meth_set_write(m, dgram_bio_write) ||
204 !BIO_meth_set_ctrl(m, dgram_bio_ctrl) || !BIO_meth_set_create(m, dgram_bio_create)) {
205 BIO_meth_free(m);
206 return;
207 }
208 g_dgram_method = m;
209}

211BIO *cwist_dtls_bio_new(struct cwist_webrtc_conn *conn) {
212 pthread_once(&g_dgram_once, dgram_method_init);
213 if (!g_dgram_method) return NULL;
214 BIO *bio = BIO_new(g_dgram_method);
215 if (bio) BIO_set_data(bio, conn);
216 return bio;
