79static char *base64_encode(const uint8_t *data, size_t len) {
80 size_t out_len = 4 * ((len + 2) / 3);
81 char *out = cwist_alloc(out_len + 1);
82 if (!out) return NULL;
83 for (size_t i = 0, j = 0; i < len; i += 3, j += 4) {
84 uint32_t v = ((uint32_t)data[i]) << 16;
85 if (i + 1 < len) v |= ((uint32_t)data[i + 1]) << 8;
86 if (i + 2 < len) v |= ((uint32_t)data[i + 2]);
87 out[j] = b64[(v >> 18) & 0x3F];
88 out[j + 1] = b64[(v >> 12) & 0x3F];
89 out[j + 2] = (i + 1 < len) ? b64[(v >> 6) & 0x3F] : '=';
90 out[j + 3] = (i + 2 < len) ? b64[v & 0x3F] : '=';
91 }
92 out[out_len] = '\0';
93 return out;
94}

105static int base64_decode(const char *in, uint8_t *out, size_t out_len) {
106 size_t in_len = strlen(in);
107 if (in_len % 4 != 0) return -1;
108 size_t pad = 0;
109 if (in_len > 0 && in[in_len - 1] == '=') pad++;
110 if (in_len > 1 && in[in_len - 2] == '=') pad++;
111 size_t len = (in_len / 4) * 3 - pad;
112 if (len > out_len) return -1;
113
114 int decode[256];
115 for (int i = 0; i < 256; i++) decode[i] = -1;
116 for (int i = 0; i < 64; i++) decode[(unsigned char)b64[i]] = i;
117
118 size_t j = 0;
119 for (size_t i = 0; i < in_len; i += 4) {
120 int a = decode[(unsigned char)in[i]];
121 int b = decode[(unsigned char)in[i + 1]];
122 int c = decode[(unsigned char)in[i + 2]];
123 int d = decode[(unsigned char)in[i + 3]];
124 if (a < 0 || b < 0) return -1;
125 uint32_t v = ((uint32_t)a << 18) | ((uint32_t)b << 12);
126 if (c >= 0) v |= ((uint32_t)c << 6);
127 if (d >= 0) v |= (uint32_t)d;
128 out[j++] = (v >> 16) & 0xFF;
129 if (c >= 0 && in[i + 2] != '=') out[j++] = (v >> 8) & 0xFF;
130 if (d >= 0 && in[i + 3] != '=') out[j++] = v & 0xFF;
131 }
132 return (int)j;
133}

146static bool hmac_sha256(const char *key, size_t key_len, const char *msg, size_t msg_len,
147 uint8_t out[32]) {

169static char *generate_secret(size_t len) {
170 char *secret = cwist_alloc(len * 2 + 1);
171 if (!secret) return NULL;
172 int fd = open("/dev/urandom", O_RDONLY);
173 if (fd >= 0) {
174 unsigned char *buf CWIST_DEFER_FREE = cwist_alloc(len);
175 if (!buf) {
176 close(fd);
177 cwist_free(secret);
178 return NULL;
179 }
180 ssize_t n = read(fd, buf, len);
181 close(fd);
182 if (n == (ssize_t)len) {
183 static const char hex[] = "0123456789abcdef";
184 for (size_t i = 0; i < len; i++) {
185 secret[i * 2] = hex[buf[i] >> 4];
186 secret[i * 2 + 1] = hex[buf[i] & 0x0F];
187 }
188 secret[len * 2] = '\0';
189 return secret;
190 }
191 }
192#ifdef __wasi__
193 /* WASI preview1 has no /dev/urandom preopened by default, but
194 * wasi-libc maps getentropy() onto __wasi_random_get. */
195 unsigned char *wbuf CWIST_DEFER_FREE = cwist_alloc(len);
196 if (wbuf && getentropy(wbuf, len) == 0) {
197 static const char hex[] = "0123456789abcdef";
198 for (size_t i = 0; i < len; i++) {
199 secret[i * 2] = hex[wbuf[i] >> 4];
200 secret[i * 2 + 1] = hex[wbuf[i] & 0x0F];
201 }
202 secret[len * 2] = '\0';
203 return secret;
204 }
205#endif
206#ifdef __EMSCRIPTEN__
207 if (fd < 0 && cwist_session_entropy_js(secret, (int)len) == 1) return secret;
208#endif
209 cwist_free(secret);
210 return NULL;
211}

289static bool verify_signature(cwist_app *app, const char *payload_b64, const char *sig_b64) {
290 uint8_t sig[32];
291 if (base64_decode(sig_b64, sig, sizeof(sig)) != 32) return false;
292 uint8_t expected[32];
293 if (!hmac_sha256(app->session_secret, strlen(app->session_secret), payload_b64,
294 strlen(payload_b64), expected)) {
295 return false;
296 }
297#if defined(__EMSCRIPTEN__) || defined(__wasi__)
298 /* No OpenSSL on WASM: use a portable constant-time loop. */
299 unsigned int diff = 0;
300 for (size_t i = 0; i < 32; i++) diff |= sig[i] ^ expected[i];
301 return diff == 0;
302#else
303 return CRYPTO_memcmp(sig, expected, 32) == 0;
304#endif
305}

347cwist_session_t *cwist_session_start(cwist_app *app, cwist_http_request *req,
348 cwist_http_response *res) {
349 if (!app || !req || !res) return NULL;
350 if (!app->session_secret && cwist_app_use_session(app, NULL) != 0) return NULL;
351
352 cwist_session_t *session = cwist_alloc(sizeof(*session));
353 if (!session) return NULL;
354 session->app = app;
355 session->req = req;
356 session->data = cwist_query_map_create();
357 session->modified = false;
358 session->invalidated = false;
359 session->loaded = false;
360
361 cwist_query_map *cookies = cwist_query_map_create();
362 if (cookies) {
363 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
364 cwist_cookie_parse(cookies, cookie_header);
365 const char *raw = cwist_cookie_get(cookies, app->session_name);
366 if (raw) {
367 char *copy = cwist_strdup(raw);
368 char *dot = strchr(copy, '.');
369 if (dot) {
370 *dot = '\0';
371 if (verify_signature(app, copy, dot + 1)) {
372 cwist_query_map_destroy(session->data);
373 session->data = parse_payload(copy);
374 if (!session->data) session->data = cwist_query_map_create();
375 session->loaded = true;
376 }
377 }
378 cwist_free(copy);
379 }
380 cwist_query_map_destroy(cookies);
381 }
382
383 req->session = session;
384 return session;
385}

457int cwist_session_commit(cwist_session_t *session, cwist_http_response *res) {
458 if (!session || !res) return -1;
459 if (!session->modified && session->loaded) return 0;
460
461 if (session->invalidated) {
462 cwist_cookie_delete(res, session->app->session_name);
463 return 0;
464 }
465
466 json_ctx_t ctx;
467 ctx.json = cJSON_CreateObject();
468 if (!ctx.json) return -1;
469 cwist_query_map_foreach(session->data, add_to_json, &ctx);
470 char *raw = cJSON_PrintUnformatted(ctx.json);
471 cJSON_Delete(ctx.json);
472 if (!raw) return -1;
473
474 char *payload_b64 = base64_encode((const uint8_t *)raw, strlen(raw));
475 free(raw);
476 if (!payload_b64) return -1;
477
478 uint8_t sig[32];
479 if (!hmac_sha256(session->app->session_secret, strlen(session->app->session_secret),
480 payload_b64, strlen(payload_b64), sig)) {
481 cwist_free(payload_b64);
482 return -1;
483 }
484 char *sig_b64 = base64_encode(sig, sizeof(sig));
485 if (!sig_b64) {
486 cwist_free(payload_b64);
487 return -1;
488 }
489
490 char cookie_value[16384];
491 snprintf(cookie_value, sizeof(cookie_value), "%s.%s", payload_b64, sig_b64);
492 cwist_free(payload_b64);
493 cwist_free(sig_b64);
494
495 cwist_cookie_options opts = {0};
496 opts.path = "/";
497 opts.max_age_seconds = session->app->session_max_age;
498 opts.http_only = true;
499 opts.same_site = "Lax";
500
501 int rc = cwist_cookie_set(res, session->app->session_name, cookie_value, &opts);
502 return rc;
503}
