106static char *session_base64_encode(const uint8_t *data, size_t len) {
107 size_t out_len = 4 * ((len + 2) / 3);
108 char *out = cwist_alloc(out_len + 1);
109 if (!out) return NULL;
110 for (size_t i = 0, j = 0; i < len; i += 3, j += 4) {
111 uint32_t v = ((uint32_t)data[i]) << 16;
112 if (i + 1 < len) v |= ((uint32_t)data[i + 1]) << 8;
113 if (i + 2 < len) v |= ((uint32_t)data[i + 2]);
114 out[j] = b64[(v >> 18) & 0x3F];
115 out[j + 1] = b64[(v >> 12) & 0x3F];
116 out[j + 2] = (i + 1 < len) ? b64[(v >> 6) & 0x3F] : '=';
117 out[j + 3] = (i + 2 < len) ? b64[v & 0x3F] : '=';
118 }
119 out[out_len] = '\0';
120 return out;
121}

132static int base64_decode(const char *in, uint8_t *out, size_t out_len) {
133 size_t in_len = strlen(in);
134 if (in_len % 4 != 0) return -1;
135 size_t pad = 0;
136 if (in_len > 0 && in[in_len - 1] == '=') pad++;
137 if (in_len > 1 && in[in_len - 2] == '=') pad++;
138 size_t len = (in_len / 4) * 3 - pad;
139 if (len > out_len) return -1;
140
141 int decode[256];
142 for (int i = 0; i < 256; i++) decode[i] = -1;
143 for (int i = 0; i < 64; i++) decode[(unsigned char)b64[i]] = i;
144
145 size_t j = 0;
146 for (size_t i = 0; i < in_len; i += 4) {
147 bool final = i + 4 == in_len;
148 int a = decode[(unsigned char)in[i]];
149 int b = decode[(unsigned char)in[i + 1]];
150 int c = decode[(unsigned char)in[i + 2]];
151 int d = decode[(unsigned char)in[i + 3]];
152 if (a < 0 || b < 0 || (!final && (c < 0 || d < 0))) return -1;
153 if (c < 0 && (in[i + 2] != '=' || in[i + 3] != '=' || (b & 0x0F) != 0)) return -1;
154 if (c >= 0 && d < 0 && (in[i + 3] != '=' || (c & 0x03) != 0)) return -1;
155
156 uint32_t v = ((uint32_t)a << 18) | ((uint32_t)b << 12);
157 if (c >= 0) v |= ((uint32_t)c << 6);
158 if (d >= 0) v |= (uint32_t)d;
159 out[j++] = (v >> 16) & 0xFF;
160 if (c >= 0) out[j++] = (v >> 8) & 0xFF;
161 if (d >= 0) out[j++] = v & 0xFF;
162 }
163 return j == len ? (int)j : -1;
164}

200static char *generate_secret(size_t len) {
201 char *secret = cwist_alloc(len * 2 + 1);
202 if (!secret) return NULL;
203 int fd = open("/dev/urandom", O_RDONLY);
204 if (fd >= 0) {
205 unsigned char *buf CWIST_DEFER_FREE = cwist_alloc(len);
206 if (!buf) {
207 close(fd);
208 cwist_free(secret);
209 return NULL;
210 }
211 size_t got = 0;
212 while (got < len) {
213 ssize_t n = read(fd, buf + got, len - got);
214 if (n > 0) {
215 got += (size_t)n;
216 } else if (n < 0 && errno == EINTR) {
217 continue;
218 } else {
219 break;
220 }
221 }
222 close(fd);
223 if (got == len) {
224 static const char hex[] = "0123456789abcdef";
225 for (size_t i = 0; i < len; i++) {
226 secret[i * 2] = hex[buf[i] >> 4];
227 secret[i * 2 + 1] = hex[buf[i] & 0x0F];
228 }
229 secret[len * 2] = '\0';
230 return secret;
231 }
232 }
233#ifdef __wasi__
234 /* WASI preview1 has no /dev/urandom preopened by default, but
235 * wasi-libc maps getentropy() onto __wasi_random_get. */
236 unsigned char *wbuf CWIST_DEFER_FREE = cwist_alloc(len);
237 if (wbuf && getentropy(wbuf, len) == 0) {
238 static const char hex[] = "0123456789abcdef";
239 for (size_t i = 0; i < len; i++) {
240 secret[i * 2] = hex[wbuf[i] >> 4];
241 secret[i * 2 + 1] = hex[wbuf[i] & 0x0F];
242 }
243 secret[len * 2] = '\0';
244 return secret;
245 }
246#endif
247#ifdef CWIST_BAREMETAL
248 unsigned char *pbuf CWIST_DEFER_FREE = cwist_alloc(len);
249 if (pbuf && cwist_port_entropy(pbuf, len) == 0) {
250 static const char hex[] = "0123456789abcdef";
251 for (size_t i = 0; i < len; i++) {
252 secret[i * 2] = hex[pbuf[i] >> 4];
253 secret[i * 2 + 1] = hex[pbuf[i] & 0x0F];
254 }
255 secret[len * 2] = '\0';
256 return secret;
257 }
258#endif
259#ifdef __EMSCRIPTEN__
260 if (fd < 0 && cwist_session_entropy_js(secret, (int)len) == 1) return secret;
261#endif
262 cwist_free(secret);
263 return NULL;
264}

276int cwist_app_use_session(cwist_app *app, const char *secret) {
277 if (!app) return -1;
278
279 /* Handle self-aliasing: if secret points to app->session_secret, it is already active */
280 if (secret && app->session_secret && secret == app->session_secret) {
281 if (!app->session_name) {
282 char *def_name = cwist_strdup(CWIST_SESSION_DEFAULT_NAME);
283 if (!def_name) return -1;
284 app->session_name = def_name;
285 }
286 if (app->session_max_age <= 0) app->session_max_age = CWIST_SESSION_DEFAULT_MAX_AGE;
287 return 0;
288 }
289
290 char *new_secret = NULL;
291 if (secret && *secret) {
292 new_secret = cwist_strdup(secret);
293 } else {
294 new_secret = generate_secret(32);
295 }
296 if (!new_secret) return -1;
297
298 char *new_name = NULL;
299 if (!app->session_name) {
300 new_name = cwist_strdup(CWIST_SESSION_DEFAULT_NAME);
301 if (!new_name) {
302 cwist_free(new_secret);
303 return -1;
304 }
305 }
306
307 if (app->session_secret) cwist_free(app->session_secret);
308 app->session_secret = new_secret;
309 if (new_name) {
310 app->session_name = new_name;
311 }
312 if (app->session_max_age <= 0) app->session_max_age = CWIST_SESSION_DEFAULT_MAX_AGE;
313 return 0;
314}

372static bool verify_signature(cwist_app *app, const char *payload_b64, const char *sig_b64) {
373 uint8_t sig[32];
374 if (base64_decode(sig_b64, sig, sizeof(sig)) != 32) return false;
375 uint8_t expected[32];
376 if (!session_hmac_sha256(app->session_secret, strlen(app->session_secret), payload_b64,
377 strlen(payload_b64), expected)) {
378 return false;
379 }
380#if !defined(CWIST_PLATFORM_HOSTED)
381 /* No OpenSSL on WASM: use a portable constant-time loop. */
382 unsigned int diff = 0;
383 for (size_t i = 0; i < 32; i++) diff |= sig[i] ^ expected[i];
384 return diff == 0;
385#else
386 return CRYPTO_memcmp(sig, expected, 32) == 0;
387#endif
388}

397static cwist_query_map *parse_payload(const char *payload_b64, time_t *expires_at,
398 char **session_id) {
399 *session_id = NULL;
400 uint8_t payload[8192];
401 int len = base64_decode(payload_b64, payload, sizeof(payload) - 1);
402 if (len < 0) return NULL;
403 payload[len] = '\0';
404 cJSON *json = cJSON_ParseWithOpts((const char *)payload, NULL, true);
405 if (!json || !cJSON_IsObject(json)) {
406 cJSON_Delete(json);
407 return NULL;
408 }
409 cJSON *exp = cJSON_GetObjectItemCaseSensitive(json, "__cwist_exp");
410 if (!cJSON_IsNumber(exp) || !cwist_number_double_is_finite(exp->valuedouble) ||
411 exp->valuedouble < 1 || exp->valuedouble > 9007199254740991.0) {
412 cJSON_Delete(json);
413 return NULL;
414 }
415 time_t expires = (time_t)exp->valuedouble;
416 if ((double)expires != exp->valuedouble || expires <= time(NULL)) {
417 cJSON_Delete(json);
418 return NULL;
419 }
420 cJSON *jti = cJSON_GetObjectItemCaseSensitive(json, "__cwist_jti");
421 if (jti && (!cJSON_IsString(jti) || !jti->valuestring || !*jti->valuestring)) {
422 cJSON_Delete(json);
423 return NULL;
424 }
425 cwist_query_map *map = cwist_query_map_create();
426 if (!map) {
427 cJSON_Delete(json);
428 return NULL;
429 }
430 cJSON *item = NULL;
431 cJSON_ArrayForEach(item, json) {
432 if (cJSON_IsString(item) && item->string && item->valuestring &&
433 strcmp(item->string, "__cwist_exp") != 0 && strcmp(item->string, "__cwist_jti") != 0) {
434 cwist_query_map_set(map, item->string, item->valuestring);
435 }
436 }
437 if (jti) {
438 *session_id = cwist_strdup(jti->valuestring);
439 if (!*session_id) {
440 cwist_query_map_destroy(map);
441 cJSON_Delete(json);
442 return NULL;
443 }
444 }
445 cJSON_Delete(json);
446 *expires_at = expires;
447 return map;
448}

450static bool session_is_revoked(cwist_app *app, const char *signature, const char *session_id) {
451 time_t now = time(NULL);
452 bool revoked = false;
453 pthread_mutex_lock(&g_session_revocations_lock);
454 session_revocation **link = &g_session_revocations;
455 while (*link) {
456 session_revocation *entry = *link;
457 if (entry->expires_at <= now) {
458 *link = entry->next;
459 cwist_free(entry->signature);
460 cwist_free(entry->session_id);
461 cwist_free(entry);
462 continue;
463 }
464 if (entry->app == app &&
465 ((session_id && entry->session_id && strcmp(entry->session_id, session_id) == 0) ||
466 (signature && entry->signature && strcmp(entry->signature, signature) == 0))) {
467 revoked = true;
468 }
469 link = &entry->next;
470 }
471 pthread_mutex_unlock(&g_session_revocations_lock);
472 return revoked;
473}

503cwist_session_t *cwist_session_start(cwist_app *app, cwist_http_request *req,
504 cwist_http_response *res) {
505 if (!app || !req || !res) return NULL;
506 if (!app->session_secret && cwist_app_use_session(app, NULL) != 0) return NULL;
507
508 cwist_session_t *session = cwist_alloc(sizeof(*session));
509 if (!session) return NULL;
510 session->app = app;
511 session->req = req;
512 session->data = cwist_query_map_create();
513 if (!session->data) {
514 cwist_free(session);
515 return NULL;
516 }
517 session->revocation = NULL;
518 session->modified = false;
519 session->invalidated = false;
520 session->loaded = false;
521
522 cwist_query_map *cookies = cwist_query_map_create();
523 if (cookies) {
524 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
525 cwist_cookie_parse(cookies, cookie_header);
526 const char *raw = cwist_cookie_get(cookies, app->session_name);
527 if (raw) {
528 /* On OOM the cookie is treated as absent: a fresh session. */
529 char *copy = cwist_strdup(raw);
530 char *dot = copy ? strchr(copy, '.') : NULL;
531 if (dot) {
532 *dot = '\0';
533 const char *signature = dot + 1;
534 if (verify_signature(app, copy, signature)) {
535 time_t expires_at;
536 char *session_id = NULL;
537 cwist_query_map *loaded = parse_payload(copy, &expires_at, &session_id);
538 if (loaded && !session_is_revoked(app, signature, session_id)) {
539 session_revocation *revocation = cwist_alloc(sizeof(*revocation));
540 char *signature_copy = cwist_strdup(signature);
541 char *id_copy = session_id ? cwist_strdup(session_id) : NULL;
542 if (revocation && signature_copy && (!session_id || id_copy)) {
543 cwist_gc_scope_disown(revocation);
544 cwist_gc_scope_disown(signature_copy);
545 if (id_copy) cwist_gc_scope_disown(id_copy);
546 revocation->app = app;
547 revocation->signature = signature_copy;
548 revocation->session_id = id_copy;
549 revocation->expires_at = expires_at;
550 revocation->next = NULL;
551 session->revocation = revocation;
552 session->session_id = session_id;
553 session_id = NULL;
554 cwist_query_map_destroy(session->data);
555 session->data = loaded;
556 session->loaded = true;
557 } else {
558 cwist_free(revocation);
559 cwist_free(signature_copy);
560 cwist_free(id_copy);
561 }
562 }
563 if (!session->loaded) {
564 cwist_query_map_destroy(loaded);
565 }
566 cwist_free(session_id);
567 }
568 }
569 cwist_free(copy);
570 }
571 cwist_query_map_destroy(cookies);
572 }
573
574 req->session = session;
575 return session;
576}

665int cwist_session_commit(cwist_session_t *session, cwist_http_response *res) {
666 if (!session || !res) return -1;
667 /* A session that was never set, deleted from or invalidated writes no
668 * cookie, so untouched responses stay cacheable and anonymous clients
669 * are not handed a session (#958). */
670 if (!session->modified) return 0;
671
672 if (session->invalidated) {
673 /* Delete with the attributes the cookie was set with. A __Host- or
674 * __Secure- name needs Secure, and some browsers keep a cookie whose
675 * deletion does not match its Path/Secure attributes (#957). */
676 cwist_cookie_options del = {0};
677 del.path = "/";
678 del.http_only = true;
679 del.secure = session->app->use_ssl;
680 del.same_site = "Lax";
681 int rc = cwist_cookie_delete_ex(res, session->app->session_name, &del);
682 if (rc == 0) {
683 session->modified = false;
684 session->loaded = false;
685 }
686 return rc;
687 }
688
689 if (!session->session_id) session->session_id = generate_secret(16);
690 if (!session->session_id) return -1;
691
692 json_ctx_t ctx = {.ok = true};
693 ctx.json = cJSON_CreateObject();
694 if (!ctx.json) return -1;
695 time_t expires_at = time(NULL) + session->app->session_max_age;
696 if (!cJSON_AddNumberToObject(ctx.json, "__cwist_exp", (double)expires_at) ||
697 !cJSON_AddStringToObject(ctx.json, "__cwist_jti", session->session_id)) {
698 cJSON_Delete(ctx.json);
699 return -1;
700 }
701 cwist_query_map_foreach(session->data, add_to_json, &ctx);
702 if (!ctx.ok) {
703 cJSON_Delete(ctx.json);
704 return -1;
705 }
706 char *raw = cJSON_PrintUnformatted(ctx.json);
707 cJSON_Delete(ctx.json);
708 if (!raw) return -1;
709
710 char *payload_b64 = session_base64_encode((const uint8_t *)raw, strlen(raw));
711 cJSON_free(raw);
712 if (!payload_b64) return -1;
713 if (strlen(payload_b64) > 8192) {
714 cwist_free(payload_b64);
715 return -1;
716 }
717
718 uint8_t sig[32];
719 if (!session_hmac_sha256(session->app->session_secret, strlen(session->app->session_secret),
720 payload_b64, strlen(payload_b64), sig)) {
721 cwist_free(payload_b64);
722 return -1;
723 }
724 char *sig_b64 = session_base64_encode(sig, sizeof(sig));
725 if (!sig_b64) {
726 cwist_free(payload_b64);
727 return -1;
728 }
729
730 size_t cookie_len = strlen(payload_b64) + strlen(sig_b64) + 1;
731 char *cookie_value = cwist_alloc(cookie_len + 1);
732 if (cookie_value) snprintf(cookie_value, cookie_len + 1, "%s.%s", payload_b64, sig_b64);
733 cwist_free(payload_b64);
734 cwist_free(sig_b64);
735 if (!cookie_value) return -1;
736 /* Browsers silently drop a cookie whose name and value exceed 4096
737 * bytes, so the session would be lost on the next request (#998). */
738 if (strlen(session->app->session_name) + 1 + strlen(cookie_value) >
739 CWIST_SESSION_MAX_COOKIE_BYTES) {
740 fprintf(stderr, "[CWIST] session cookie too large (%zu bytes); not sent\n",
741 strlen(cookie_value));
742 cwist_free(cookie_value);
743 return -1;
744 }
745
746 session_revocation *revocation = session->revocation;
747 bool fresh = revocation == NULL;
748 if (fresh) revocation = cwist_alloc(sizeof(*revocation));
749 char *signature_copy = revocation ? cwist_strdup(strstr(cookie_value, ".") + 1) : NULL;
750 char *id_copy =
751 revocation && !revocation->session_id ? cwist_strdup(session->session_id) : NULL;
752 if (!revocation || !signature_copy || (!revocation->session_id && !id_copy)) {
753 if (fresh) cwist_free(revocation);
754 cwist_free(signature_copy);
755 cwist_free(id_copy);
756 cwist_free(cookie_value);
757 return -1;
758 }
759 if (fresh) {
760 revocation->app = session->app;
761 revocation->next = NULL;
762 }
763
764 cwist_cookie_options opts = {0};
765 opts.path = "/";
766 opts.max_age_seconds = session->app->session_max_age;
767 opts.http_only = true;
768 opts.secure = session->app->use_ssl;
769 opts.same_site = "Lax";
770
771 int rc = cwist_cookie_set(res, session->app->session_name, cookie_value, &opts);
772 cwist_free(cookie_value);
773 if (rc != 0) {
774 if (fresh) cwist_free(revocation);
775 cwist_free(signature_copy);
776 cwist_free(id_copy);
777 return rc;
778 }
779
780 if (fresh) {
781 session->revocation = revocation;
782 cwist_gc_scope_disown(revocation);
783 } else {
784 cwist_free(revocation->signature);
785 }
786 revocation->signature = signature_copy;
787 cwist_gc_scope_disown(signature_copy);
788 if (id_copy) {
789 revocation->session_id = id_copy;
790 cwist_gc_scope_disown(id_copy);
791 }
792 revocation->expires_at = expires_at;
793 session->modified = false;
794 session->loaded = true;
795 return 0;
796}

817 cwist_handler_func next) {

819 if (!app) {

822 }
823 cwist_session_start(app, req, res);
824 next(req, res);

826 cwist_async_add_completion_hook(res->async, cwist_session_complete, NULL);
827 return;
828 }

831 }
832}
