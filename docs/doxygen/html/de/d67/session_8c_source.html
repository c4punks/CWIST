99static char *session_base64_encode(const uint8_t *data, size_t len) {
100 size_t out_len = 4 * ((len + 2) / 3);
101 char *out = cwist_alloc(out_len + 1);
102 if (!out) return NULL;
103 for (size_t i = 0, j = 0; i < len; i += 3, j += 4) {
104 uint32_t v = ((uint32_t)data[i]) << 16;
105 if (i + 1 < len) v |= ((uint32_t)data[i + 1]) << 8;
106 if (i + 2 < len) v |= ((uint32_t)data[i + 2]);
107 out[j] = b64[(v >> 18) & 0x3F];
108 out[j + 1] = b64[(v >> 12) & 0x3F];
109 out[j + 2] = (i + 1 < len) ? b64[(v >> 6) & 0x3F] : '=';
110 out[j + 3] = (i + 2 < len) ? b64[v & 0x3F] : '=';
111 }
112 out[out_len] = '\0';
113 return out;
114}

125static int base64_decode(const char *in, uint8_t *out, size_t out_len) {
126 size_t in_len = strlen(in);
127 if (in_len % 4 != 0) return -1;
128 size_t pad = 0;
129 if (in_len > 0 && in[in_len - 1] == '=') pad++;
130 if (in_len > 1 && in[in_len - 2] == '=') pad++;
131 size_t len = (in_len / 4) * 3 - pad;
132 if (len > out_len) return -1;
133
134 int decode[256];
135 for (int i = 0; i < 256; i++) decode[i] = -1;
136 for (int i = 0; i < 64; i++) decode[(unsigned char)b64[i]] = i;
137
138 size_t j = 0;
139 for (size_t i = 0; i < in_len; i += 4) {
140 bool final = i + 4 == in_len;
141 int a = decode[(unsigned char)in[i]];
142 int b = decode[(unsigned char)in[i + 1]];
143 int c = decode[(unsigned char)in[i + 2]];
144 int d = decode[(unsigned char)in[i + 3]];
145 if (a < 0 || b < 0 || (!final && (c < 0 || d < 0))) return -1;
146 if (c < 0 && (in[i + 2] != '=' || in[i + 3] != '=' || (b & 0x0F) != 0)) return -1;
147 if (c >= 0 && d < 0 && (in[i + 3] != '=' || (c & 0x03) != 0)) return -1;
148
149 uint32_t v = ((uint32_t)a << 18) | ((uint32_t)b << 12);
150 if (c >= 0) v |= ((uint32_t)c << 6);
151 if (d >= 0) v |= (uint32_t)d;
152 out[j++] = (v >> 16) & 0xFF;
153 if (c >= 0) out[j++] = (v >> 8) & 0xFF;
154 if (d >= 0) out[j++] = v & 0xFF;
155 }
156 return j == len ? (int)j : -1;
157}

170static bool session_hmac_sha256(const char *key, size_t key_len, const char *msg, size_t msg_len,
171 uint8_t out[32]) {

174#else

176 unsigned char *r =
177 HMAC(EVP_sha256(), key, (int)key_len, (const unsigned char *)msg, msg_len, out, &len);
178 return r != NULL && len == 32;

193static char *generate_secret(size_t len) {
194 char *secret = cwist_alloc(len * 2 + 1);
195 if (!secret) return NULL;
196 int fd = open("/dev/urandom", O_RDONLY);
197 if (fd >= 0) {
198 unsigned char *buf CWIST_DEFER_FREE = cwist_alloc(len);
199 if (!buf) {
200 close(fd);
201 cwist_free(secret);
202 return NULL;
203 }
204 ssize_t n = read(fd, buf, len);
205 close(fd);
206 if (n == (ssize_t)len) {
207 static const char hex[] = "0123456789abcdef";
208 for (size_t i = 0; i < len; i++) {
209 secret[i * 2] = hex[buf[i] >> 4];
210 secret[i * 2 + 1] = hex[buf[i] & 0x0F];
211 }
212 secret[len * 2] = '\0';
213 return secret;
214 }
215 }
216#ifdef __wasi__
217 /* WASI preview1 has no /dev/urandom preopened by default, but
218 * wasi-libc maps getentropy() onto __wasi_random_get. */
219 unsigned char *wbuf CWIST_DEFER_FREE = cwist_alloc(len);
220 if (wbuf && getentropy(wbuf, len) == 0) {
221 static const char hex[] = "0123456789abcdef";
222 for (size_t i = 0; i < len; i++) {
223 secret[i * 2] = hex[wbuf[i] >> 4];
224 secret[i * 2 + 1] = hex[wbuf[i] & 0x0F];
225 }
226 secret[len * 2] = '\0';
227 return secret;
228 }
229#endif
230#ifdef CWIST_BAREMETAL
231 unsigned char *pbuf CWIST_DEFER_FREE = cwist_alloc(len);
232 if (pbuf && cwist_port_entropy(pbuf, len) == 0) {
233 static const char hex[] = "0123456789abcdef";
234 for (size_t i = 0; i < len; i++) {
235 secret[i * 2] = hex[pbuf[i] >> 4];
236 secret[i * 2 + 1] = hex[pbuf[i] & 0x0F];
237 }
238 secret[len * 2] = '\0';
239 return secret;
240 }
241#endif
242#ifdef __EMSCRIPTEN__
243 if (fd < 0 && cwist_session_entropy_js(secret, (int)len) == 1) return secret;
244#endif
245 cwist_free(secret);
246 return NULL;
247}

259int cwist_app_use_session(cwist_app *app, const char *secret) {
260 if (!app) return -1;
261
262 /* Handle self-aliasing: if secret points to app->session_secret, it is already active */
263 if (secret && app->session_secret && secret == app->session_secret) {
264 if (!app->session_name) {
265 char *def_name = cwist_strdup(CWIST_SESSION_DEFAULT_NAME);
266 if (!def_name) return -1;
267 app->session_name = def_name;
268 }
269 if (app->session_max_age <= 0) app->session_max_age = CWIST_SESSION_DEFAULT_MAX_AGE;
270 return 0;
271 }
272
273 char *new_secret = NULL;
274 if (secret && *secret) {
275 new_secret = cwist_strdup(secret);
276 } else {
277 new_secret = generate_secret(32);
278 }
279 if (!new_secret) return -1;
280
281 char *new_name = NULL;
282 if (!app->session_name) {
283 new_name = cwist_strdup(CWIST_SESSION_DEFAULT_NAME);
284 if (!new_name) {
285 cwist_free(new_secret);
286 return -1;
287 }
288 }
289
290 if (app->session_secret) cwist_free(app->session_secret);
291 app->session_secret = new_secret;
292 if (new_name) {
293 app->session_name = new_name;
294 }
295 if (app->session_max_age <= 0) app->session_max_age = CWIST_SESSION_DEFAULT_MAX_AGE;
296 return 0;
297}

354static bool verify_signature(cwist_app *app, const char *payload_b64, const char *sig_b64) {
355 uint8_t sig[32];
356 if (base64_decode(sig_b64, sig, sizeof(sig)) != 32) return false;
357 uint8_t expected[32];
358 if (!session_hmac_sha256(app->session_secret, strlen(app->session_secret), payload_b64,
359 strlen(payload_b64), expected)) {
360 return false;
361 }
362#if !defined(CWIST_PLATFORM_HOSTED)
363 /* No OpenSSL on WASM: use a portable constant-time loop. */
364 unsigned int diff = 0;
365 for (size_t i = 0; i < 32; i++) diff |= sig[i] ^ expected[i];
366 return diff == 0;
367#else
368 return CRYPTO_memcmp(sig, expected, 32) == 0;
369#endif
370}

379static cwist_query_map *parse_payload(const char *payload_b64, time_t *expires_at,
380 char **session_id) {
381 *session_id = NULL;
382 uint8_t payload[8192];
383 int len = base64_decode(payload_b64, payload, sizeof(payload) - 1);
384 if (len < 0) return NULL;
385 payload[len] = '\0';
386 cJSON *json = cJSON_ParseWithOpts((const char *)payload, NULL, true);
387 if (!json || !cJSON_IsObject(json)) {
388 cJSON_Delete(json);
389 return NULL;
390 }
391 cJSON *exp = cJSON_GetObjectItemCaseSensitive(json, "__cwist_exp");
392 if (!cJSON_IsNumber(exp) || !cwist_number_double_is_finite(exp->valuedouble) ||
393 exp->valuedouble < 1 || exp->valuedouble > 9007199254740991.0) {
394 cJSON_Delete(json);
395 return NULL;
396 }
397 time_t expires = (time_t)exp->valuedouble;
398 if ((double)expires != exp->valuedouble || expires <= time(NULL)) {
399 cJSON_Delete(json);
400 return NULL;
401 }
402 cJSON *jti = cJSON_GetObjectItemCaseSensitive(json, "__cwist_jti");
403 if (jti && (!cJSON_IsString(jti) || !jti->valuestring || !*jti->valuestring)) {
404 cJSON_Delete(json);
405 return NULL;
406 }
407 cwist_query_map *map = cwist_query_map_create();
408 if (!map) {
409 cJSON_Delete(json);
410 return NULL;
411 }
412 cJSON *item = NULL;
413 cJSON_ArrayForEach(item, json) {
414 if (cJSON_IsString(item) && item->string && item->valuestring &&
415 strcmp(item->string, "__cwist_exp") != 0 && strcmp(item->string, "__cwist_jti") != 0) {
416 cwist_query_map_set(map, item->string, item->valuestring);
417 }
418 }
419 if (jti) {
420 *session_id = cwist_strdup(jti->valuestring);
421 if (!*session_id) {
422 cwist_query_map_destroy(map);
423 cJSON_Delete(json);
424 return NULL;
425 }
426 }
427 cJSON_Delete(json);
428 *expires_at = expires;
429 return map;
430}

432static bool session_is_revoked(cwist_app *app, const char *signature, const char *session_id) {
433 time_t now = time(NULL);
434 bool revoked = false;
435 pthread_mutex_lock(&g_session_revocations_lock);
436 session_revocation **link = &g_session_revocations;
437 while (*link) {
438 session_revocation *entry = *link;
439 if (entry->expires_at <= now) {
440 *link = entry->next;
441 cwist_free(entry->signature);
442 cwist_free(entry->session_id);
443 cwist_free(entry);
444 continue;
445 }
446 if (entry->app == app &&
447 ((session_id && entry->session_id && strcmp(entry->session_id, session_id) == 0) ||
448 (signature && entry->signature && strcmp(entry->signature, signature) == 0))) {
449 revoked = true;
450 }
451 link = &entry->next;
452 }
453 pthread_mutex_unlock(&g_session_revocations_lock);
454 return revoked;
455}

485cwist_session_t *cwist_session_start(cwist_app *app, cwist_http_request *req,
486 cwist_http_response *res) {
487 if (!app || !req || !res) return NULL;
488 if (!app->session_secret && cwist_app_use_session(app, NULL) != 0) return NULL;
489
490 cwist_session_t *session = cwist_alloc(sizeof(*session));
491 if (!session) return NULL;
492 session->app = app;
493 session->req = req;
494 session->data = cwist_query_map_create();
495 session->revocation = NULL;
496 session->modified = false;
497 session->invalidated = false;
498 session->loaded = false;
499
500 cwist_query_map *cookies = cwist_query_map_create();
501 if (cookies) {
502 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
503 cwist_cookie_parse(cookies, cookie_header);
504 const char *raw = cwist_cookie_get(cookies, app->session_name);
505 if (raw) {
506 /* On OOM the cookie is treated as absent: a fresh session. */
507 char *copy = cwist_strdup(raw);
508 char *dot = copy ? strchr(copy, '.') : NULL;
509 if (dot) {
510 *dot = '\0';
511 const char *signature = dot + 1;
512 if (verify_signature(app, copy, signature)) {
513 time_t expires_at;
514 char *session_id = NULL;
515 cwist_query_map *loaded = parse_payload(copy, &expires_at, &session_id);
516 if (loaded && !session_is_revoked(app, signature, session_id)) {
517 session_revocation *revocation = cwist_alloc(sizeof(*revocation));
518 char *signature_copy = cwist_strdup(signature);
519 char *id_copy = session_id ? cwist_strdup(session_id) : NULL;
520 if (revocation && signature_copy && (!session_id || id_copy)) {
521 cwist_gc_scope_disown(revocation);
522 cwist_gc_scope_disown(signature_copy);
523 if (id_copy) cwist_gc_scope_disown(id_copy);
524 revocation->app = app;
525 revocation->signature = signature_copy;
526 revocation->session_id = id_copy;
527 revocation->expires_at = expires_at;
528 revocation->next = NULL;
529 session->revocation = revocation;
530 session->session_id = session_id;
531 session_id = NULL;
532 cwist_query_map_destroy(session->data);
533 session->data = loaded;
534 session->loaded = true;
535 } else {
536 cwist_free(revocation);
537 cwist_free(signature_copy);
538 cwist_free(id_copy);
539 }
540 }
541 if (!session->loaded) {
542 cwist_query_map_destroy(loaded);
543 }
544 cwist_free(session_id);
545 }
546 }
547 cwist_free(copy);
548 }
549 cwist_query_map_destroy(cookies);
550 }
551
552 req->session = session;
553 return session;
554}

640int cwist_session_commit(cwist_session_t *session, cwist_http_response *res) {
641 if (!session || !res) return -1;
642 if (!session->modified && session->loaded) return 0;
643
644 if (session->invalidated) {
645 int rc = cwist_cookie_delete(res, session->app->session_name);
646 if (rc == 0) {
647 session->modified = false;
648 session->loaded = false;
649 }
650 return rc;
651 }
652
653 if (!session->session_id) session->session_id = generate_secret(16);
654 if (!session->session_id) return -1;
655
656 json_ctx_t ctx;
657 ctx.json = cJSON_CreateObject();
658 if (!ctx.json) return -1;
659 time_t expires_at = time(NULL) + session->app->session_max_age;
660 if (!cJSON_AddNumberToObject(ctx.json, "__cwist_exp", (double)expires_at) ||
661 !cJSON_AddStringToObject(ctx.json, "__cwist_jti", session->session_id)) {
662 cJSON_Delete(ctx.json);
663 return -1;
664 }
665 cwist_query_map_foreach(session->data, add_to_json, &ctx);
666 char *raw = cJSON_PrintUnformatted(ctx.json);
667 cJSON_Delete(ctx.json);
668 if (!raw) return -1;
669
670 char *payload_b64 = session_base64_encode((const uint8_t *)raw, strlen(raw));
671 free(raw);
672 if (!payload_b64) return -1;
673 if (strlen(payload_b64) > 8192) {
674 cwist_free(payload_b64);
675 return -1;
676 }
677
678 uint8_t sig[32];
679 if (!session_hmac_sha256(session->app->session_secret, strlen(session->app->session_secret),
680 payload_b64, strlen(payload_b64), sig)) {
681 cwist_free(payload_b64);
682 return -1;
683 }
684 char *sig_b64 = session_base64_encode(sig, sizeof(sig));
685 if (!sig_b64) {
686 cwist_free(payload_b64);
687 return -1;
688 }
689
690 size_t cookie_len = strlen(payload_b64) + strlen(sig_b64) + 1;
691 char *cookie_value = cwist_alloc(cookie_len + 1);
692 if (cookie_value) snprintf(cookie_value, cookie_len + 1, "%s.%s", payload_b64, sig_b64);
693 cwist_free(payload_b64);
694 cwist_free(sig_b64);
695 if (!cookie_value) return -1;
696
697 session_revocation *revocation = session->revocation;
698 bool fresh = revocation == NULL;
699 if (fresh) revocation = cwist_alloc(sizeof(*revocation));
700 char *signature_copy = revocation ? cwist_strdup(strstr(cookie_value, ".") + 1) : NULL;
701 char *id_copy =
702 revocation && !revocation->session_id ? cwist_strdup(session->session_id) : NULL;
703 if (!revocation || !signature_copy || (!revocation->session_id && !id_copy)) {
704 if (fresh) cwist_free(revocation);
705 cwist_free(signature_copy);
706 cwist_free(id_copy);
707 cwist_free(cookie_value);
708 return -1;
709 }
710 if (fresh) {
711 revocation->app = session->app;
712 revocation->next = NULL;
713 }
714
715 cwist_cookie_options opts = {0};
716 opts.path = "/";
717 opts.max_age_seconds = session->app->session_max_age;
718 opts.http_only = true;
719 opts.secure = session->app->use_ssl;
720 opts.same_site = "Lax";
721
722 int rc = cwist_cookie_set(res, session->app->session_name, cookie_value, &opts);
723 cwist_free(cookie_value);
724 if (rc != 0) {
725 if (fresh) cwist_free(revocation);
726 cwist_free(signature_copy);
727 cwist_free(id_copy);
728 return rc;
729 }
730
731 if (fresh) {
732 session->revocation = revocation;
