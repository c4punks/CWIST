83static char *session_base64_encode(const uint8_t *data, size_t len) {
84 size_t out_len = 4 * ((len + 2) / 3);
85 char *out = cwist_alloc(out_len + 1);
86 if (!out) return NULL;
87 for (size_t i = 0, j = 0; i < len; i += 3, j += 4) {
88 uint32_t v = ((uint32_t)data[i]) << 16;
89 if (i + 1 < len) v |= ((uint32_t)data[i + 1]) << 8;
90 if (i + 2 < len) v |= ((uint32_t)data[i + 2]);
91 out[j] = b64[(v >> 18) & 0x3F];
92 out[j + 1] = b64[(v >> 12) & 0x3F];
93 out[j + 2] = (i + 1 < len) ? b64[(v >> 6) & 0x3F] : '=';
94 out[j + 3] = (i + 2 < len) ? b64[v & 0x3F] : '=';
95 }
96 out[out_len] = '\0';
97 return out;
98}

109static int base64_decode(const char *in, uint8_t *out, size_t out_len) {
110 size_t in_len = strlen(in);
111 if (in_len % 4 != 0) return -1;
112 size_t pad = 0;
113 if (in_len > 0 && in[in_len - 1] == '=') pad++;
114 if (in_len > 1 && in[in_len - 2] == '=') pad++;
115 size_t len = (in_len / 4) * 3 - pad;
116 if (len > out_len) return -1;
117
118 int decode[256];
119 for (int i = 0; i < 256; i++) decode[i] = -1;
120 for (int i = 0; i < 64; i++) decode[(unsigned char)b64[i]] = i;
121
122 size_t j = 0;
123 for (size_t i = 0; i < in_len; i += 4) {
124 int a = decode[(unsigned char)in[i]];
125 int b = decode[(unsigned char)in[i + 1]];
126 int c = decode[(unsigned char)in[i + 2]];
127 int d = decode[(unsigned char)in[i + 3]];
128 if (a < 0 || b < 0) return -1;
129 uint32_t v = ((uint32_t)a << 18) | ((uint32_t)b << 12);
130 if (c >= 0) v |= ((uint32_t)c << 6);
131 if (d >= 0) v |= (uint32_t)d;
132 out[j++] = (v >> 16) & 0xFF;
133 if (c >= 0 && in[i + 2] != '=') out[j++] = (v >> 8) & 0xFF;
134 if (d >= 0 && in[i + 3] != '=') out[j++] = v & 0xFF;
135 }
136 return (int)j;
137}

150static bool session_hmac_sha256(const char *key, size_t key_len, const char *msg, size_t msg_len,
151 uint8_t out[32]) {

173static char *generate_secret(size_t len) {
174 char *secret = cwist_alloc(len * 2 + 1);
175 if (!secret) return NULL;
176 int fd = open("/dev/urandom", O_RDONLY);
177 if (fd >= 0) {
178 unsigned char *buf CWIST_DEFER_FREE = cwist_alloc(len);
179 if (!buf) {
180 close(fd);
181 cwist_free(secret);
182 return NULL;
183 }
184 ssize_t n = read(fd, buf, len);
185 close(fd);
186 if (n == (ssize_t)len) {
187 static const char hex[] = "0123456789abcdef";
188 for (size_t i = 0; i < len; i++) {
189 secret[i * 2] = hex[buf[i] >> 4];
190 secret[i * 2 + 1] = hex[buf[i] & 0x0F];
191 }
192 secret[len * 2] = '\0';
193 return secret;
194 }
195 }
196#ifdef __wasi__
197 /* WASI preview1 has no /dev/urandom preopened by default, but
198 * wasi-libc maps getentropy() onto __wasi_random_get. */
199 unsigned char *wbuf CWIST_DEFER_FREE = cwist_alloc(len);
200 if (wbuf && getentropy(wbuf, len) == 0) {
201 static const char hex[] = "0123456789abcdef";
202 for (size_t i = 0; i < len; i++) {
203 secret[i * 2] = hex[wbuf[i] >> 4];
204 secret[i * 2 + 1] = hex[wbuf[i] & 0x0F];
205 }
206 secret[len * 2] = '\0';
207 return secret;
208 }
209#endif
210#ifdef CWIST_BAREMETAL
211 unsigned char *pbuf CWIST_DEFER_FREE = cwist_alloc(len);
212 if (pbuf && cwist_port_entropy(pbuf, len) == 0) {
213 static const char hex[] = "0123456789abcdef";
214 for (size_t i = 0; i < len; i++) {
215 secret[i * 2] = hex[pbuf[i] >> 4];
216 secret[i * 2 + 1] = hex[pbuf[i] & 0x0F];
217 }
218 secret[len * 2] = '\0';
219 return secret;
220 }
221#endif
222#ifdef __EMSCRIPTEN__
223 if (fd < 0 && cwist_session_entropy_js(secret, (int)len) == 1) return secret;
224#endif
225 cwist_free(secret);
226 return NULL;
227}

308static bool verify_signature(cwist_app *app, const char *payload_b64, const char *sig_b64) {
309 uint8_t sig[32];
310 if (base64_decode(sig_b64, sig, sizeof(sig)) != 32) return false;
311 uint8_t expected[32];
312 if (!session_hmac_sha256(app->session_secret, strlen(app->session_secret), payload_b64,
313 strlen(payload_b64), expected)) {
314 return false;
315 }
316#if !defined(CWIST_PLATFORM_HOSTED)
317 /* No OpenSSL on WASM: use a portable constant-time loop. */
318 unsigned int diff = 0;
319 for (size_t i = 0; i < 32; i++) diff |= sig[i] ^ expected[i];
320 return diff == 0;
321#else
322 return CRYPTO_memcmp(sig, expected, 32) == 0;
323#endif
324}

333static cwist_query_map *parse_payload(const char *payload_b64) {
334 uint8_t payload[8192];
335 int len = base64_decode(payload_b64, payload, sizeof(payload) - 1);
336 if (len < 0) return NULL;
337 payload[len] = '\0';
338 cJSON *json = cJSON_Parse((const char *)payload);
339 if (!json) return NULL;
340 cwist_query_map *map = cwist_query_map_create();
341 if (!map) {
342 cJSON_Delete(json);
343 return NULL;
344 }
345 cJSON *item = NULL;
346 cJSON_ArrayForEach(item, json) {
347 if (cJSON_IsString(item) && item->string && item->valuestring) {
348 cwist_query_map_set(map, item->string, item->valuestring);
349 }
350 }
351 cJSON_Delete(json);
352 return map;
353}

366cwist_session_t *cwist_session_start(cwist_app *app, cwist_http_request *req,
367 cwist_http_response *res) {
368 if (!app || !req || !res) return NULL;
369 if (!app->session_secret && cwist_app_use_session(app, NULL) != 0) return NULL;
370
371 cwist_session_t *session = cwist_alloc(sizeof(*session));
372 if (!session) return NULL;
373 session->app = app;
374 session->req = req;
375 session->data = cwist_query_map_create();
376 session->modified = false;
377 session->invalidated = false;
378 session->loaded = false;
379
380 cwist_query_map *cookies = cwist_query_map_create();
381 if (cookies) {
382 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
383 cwist_cookie_parse(cookies, cookie_header);
384 const char *raw = cwist_cookie_get(cookies, app->session_name);
385 if (raw) {
386 /* On OOM the cookie is treated as absent: a fresh session. */
387 char *copy = cwist_strdup(raw);
388 char *dot = copy ? strchr(copy, '.') : NULL;
389 if (dot) {
390 *dot = '\0';
391 if (verify_signature(app, copy, dot + 1)) {
392 cwist_query_map_destroy(session->data);
393 session->data = parse_payload(copy);
394 if (!session->data) session->data = cwist_query_map_create();
395 session->loaded = true;
396 }
397 }
398 cwist_free(copy);
399 }
400 cwist_query_map_destroy(cookies);
401 }
402
403 req->session = session;
404 return session;
405}

477int cwist_session_commit(cwist_session_t *session, cwist_http_response *res) {
478 if (!session || !res) return -1;
479 if (!session->modified && session->loaded) return 0;
480
481 if (session->invalidated) {
482 cwist_cookie_delete(res, session->app->session_name);
483 return 0;
484 }
485
486 json_ctx_t ctx;
487 ctx.json = cJSON_CreateObject();
488 if (!ctx.json) return -1;
489 cwist_query_map_foreach(session->data, add_to_json, &ctx);
490 char *raw = cJSON_PrintUnformatted(ctx.json);
491 cJSON_Delete(ctx.json);
492 if (!raw) return -1;
493
494 char *payload_b64 = session_base64_encode((const uint8_t *)raw, strlen(raw));
495 free(raw);
496 if (!payload_b64) return -1;
497
498 uint8_t sig[32];
499 if (!session_hmac_sha256(session->app->session_secret, strlen(session->app->session_secret),
500 payload_b64, strlen(payload_b64), sig)) {
501 cwist_free(payload_b64);
502 return -1;
503 }
504 char *sig_b64 = session_base64_encode(sig, sizeof(sig));
505 if (!sig_b64) {
506 cwist_free(payload_b64);
507 return -1;
508 }
509
510 char cookie_value[16384];
511 snprintf(cookie_value, sizeof(cookie_value), "%s.%s", payload_b64, sig_b64);
512 cwist_free(payload_b64);
513 cwist_free(sig_b64);
514
515 cwist_cookie_options opts = {0};
516 opts.path = "/";
517 opts.max_age_seconds = session->app->session_max_age;
518 opts.http_only = true;
519 opts.same_site = "Lax";
520
521 int rc = cwist_cookie_set(res, session->app->session_name, cookie_value, &opts);
522 return rc;
523}
