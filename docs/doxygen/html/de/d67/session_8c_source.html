96static char *session_base64_encode(const uint8_t *data, size_t len) {
97 size_t out_len = 4 * ((len + 2) / 3);
98 char *out = cwist_alloc(out_len + 1);
99 if (!out) return NULL;
100 for (size_t i = 0, j = 0; i < len; i += 3, j += 4) {
101 uint32_t v = ((uint32_t)data[i]) << 16;
102 if (i + 1 < len) v |= ((uint32_t)data[i + 1]) << 8;
103 if (i + 2 < len) v |= ((uint32_t)data[i + 2]);
104 out[j] = b64[(v >> 18) & 0x3F];
105 out[j + 1] = b64[(v >> 12) & 0x3F];
106 out[j + 2] = (i + 1 < len) ? b64[(v >> 6) & 0x3F] : '=';
107 out[j + 3] = (i + 2 < len) ? b64[v & 0x3F] : '=';
108 }
109 out[out_len] = '\0';
110 return out;
111}

122static int base64_decode(const char *in, uint8_t *out, size_t out_len) {
123 size_t in_len = strlen(in);
124 if (in_len % 4 != 0) return -1;
125 size_t pad = 0;
126 if (in_len > 0 && in[in_len - 1] == '=') pad++;
127 if (in_len > 1 && in[in_len - 2] == '=') pad++;
128 size_t len = (in_len / 4) * 3 - pad;
129 if (len > out_len) return -1;
130
131 int decode[256];
132 for (int i = 0; i < 256; i++) decode[i] = -1;
133 for (int i = 0; i < 64; i++) decode[(unsigned char)b64[i]] = i;
134
135 size_t j = 0;
136 for (size_t i = 0; i < in_len; i += 4) {
137 int a = decode[(unsigned char)in[i]];
138 int b = decode[(unsigned char)in[i + 1]];
139 int c = decode[(unsigned char)in[i + 2]];
140 int d = decode[(unsigned char)in[i + 3]];
141 if (a < 0 || b < 0) return -1;
142 uint32_t v = ((uint32_t)a << 18) | ((uint32_t)b << 12);
143 if (c >= 0) v |= ((uint32_t)c << 6);
144 if (d >= 0) v |= (uint32_t)d;
145 out[j++] = (v >> 16) & 0xFF;
146 if (c >= 0 && in[i + 2] != '=') out[j++] = (v >> 8) & 0xFF;
147 if (d >= 0 && in[i + 3] != '=') out[j++] = v & 0xFF;
148 }
149 return (int)j;
150}

186static char *generate_secret(size_t len) {
187 char *secret = cwist_alloc(len * 2 + 1);
188 if (!secret) return NULL;
189 int fd = open("/dev/urandom", O_RDONLY);
190 if (fd >= 0) {
191 unsigned char *buf CWIST_DEFER_FREE = cwist_alloc(len);
192 if (!buf) {
193 close(fd);
194 cwist_free(secret);
195 return NULL;
196 }
197 ssize_t n = read(fd, buf, len);
198 close(fd);
199 if (n == (ssize_t)len) {
200 static const char hex[] = "0123456789abcdef";
201 for (size_t i = 0; i < len; i++) {
202 secret[i * 2] = hex[buf[i] >> 4];
203 secret[i * 2 + 1] = hex[buf[i] & 0x0F];
204 }
205 secret[len * 2] = '\0';
206 return secret;
207 }
208 }
209#ifdef __wasi__
210 /* WASI preview1 has no /dev/urandom preopened by default, but
211 * wasi-libc maps getentropy() onto __wasi_random_get. */
212 unsigned char *wbuf CWIST_DEFER_FREE = cwist_alloc(len);
213 if (wbuf && getentropy(wbuf, len) == 0) {
214 static const char hex[] = "0123456789abcdef";
215 for (size_t i = 0; i < len; i++) {
216 secret[i * 2] = hex[wbuf[i] >> 4];
217 secret[i * 2 + 1] = hex[wbuf[i] & 0x0F];
218 }
219 secret[len * 2] = '\0';
220 return secret;
221 }
222#endif
223#ifdef CWIST_BAREMETAL
224 unsigned char *pbuf CWIST_DEFER_FREE = cwist_alloc(len);
225 if (pbuf && cwist_port_entropy(pbuf, len) == 0) {
226 static const char hex[] = "0123456789abcdef";
227 for (size_t i = 0; i < len; i++) {
228 secret[i * 2] = hex[pbuf[i] >> 4];
229 secret[i * 2 + 1] = hex[pbuf[i] & 0x0F];
230 }
231 secret[len * 2] = '\0';
232 return secret;
233 }
234#endif
235#ifdef __EMSCRIPTEN__
236 if (fd < 0 && cwist_session_entropy_js(secret, (int)len) == 1) return secret;
237#endif
238 cwist_free(secret);
239 return NULL;
240}

321static bool verify_signature(cwist_app *app, const char *payload_b64, const char *sig_b64) {
322 uint8_t sig[32];
323 if (base64_decode(sig_b64, sig, sizeof(sig)) != 32) return false;
324 uint8_t expected[32];
325 if (!session_hmac_sha256(app->session_secret, strlen(app->session_secret), payload_b64,
326 strlen(payload_b64), expected)) {
327 return false;
328 }
329#if !defined(CWIST_PLATFORM_HOSTED)
330 /* No OpenSSL on WASM: use a portable constant-time loop. */
331 unsigned int diff = 0;
332 for (size_t i = 0; i < 32; i++) diff |= sig[i] ^ expected[i];
333 return diff == 0;
334#else
335 return CRYPTO_memcmp(sig, expected, 32) == 0;
336#endif
337}

346static cwist_query_map *parse_payload(const char *payload_b64, time_t *expires_at) {
347 uint8_t payload[8192];
348 int len = base64_decode(payload_b64, payload, sizeof(payload) - 1);
349 if (len < 0) return NULL;
350 payload[len] = '\0';
351 cJSON *json = cJSON_Parse((const char *)payload);
352 if (!json) return NULL;
353 cJSON *exp = cJSON_GetObjectItemCaseSensitive(json, "__cwist_exp");
354 if (!cJSON_IsNumber(exp) || !isfinite(exp->valuedouble) || exp->valuedouble < 1 ||
355 exp->valuedouble > 9007199254740991.0) {
356 cJSON_Delete(json);
357 return NULL;
358 }
359 time_t expires = (time_t)exp->valuedouble;
360 if ((double)expires != exp->valuedouble || expires <= time(NULL)) {
361 cJSON_Delete(json);
362 return NULL;
363 }
364 cwist_query_map *map = cwist_query_map_create();
365 if (!map) {
366 cJSON_Delete(json);
367 return NULL;
368 }
369 cJSON *item = NULL;
370 cJSON_ArrayForEach(item, json) {
371 if (cJSON_IsString(item) && item->string && item->valuestring &&
372 strcmp(item->string, "__cwist_exp") != 0 && strcmp(item->string, "__cwist_jti") != 0) {
373 cwist_query_map_set(map, item->string, item->valuestring);
374 }
375 }
376 cJSON_Delete(json);
377 *expires_at = expires;
378 return map;
379}

381static bool session_is_revoked(cwist_app *app, const char *signature) {
382 time_t now = time(NULL);
383 bool revoked = false;
384 pthread_mutex_lock(&g_session_revocations_lock);
385 session_revocation **link = &g_session_revocations;
386 while (*link) {
387 session_revocation *entry = *link;
388 if (entry->expires_at <= now) {
389 *link = entry->next;
390 cwist_free(entry->signature);
391 cwist_free(entry);
392 continue;
393 }
394 if (entry->app == app && strcmp(entry->signature, signature) == 0) revoked = true;
395 link = &entry->next;
396 }
397 pthread_mutex_unlock(&g_session_revocations_lock);
398 return revoked;
399}

428cwist_session_t *cwist_session_start(cwist_app *app, cwist_http_request *req,
429 cwist_http_response *res) {
430 if (!app || !req || !res) return NULL;
431 if (!app->session_secret && cwist_app_use_session(app, NULL) != 0) return NULL;
432
433 cwist_session_t *session = cwist_alloc(sizeof(*session));
434 if (!session) return NULL;
435 session->app = app;
436 session->req = req;
437 session->data = cwist_query_map_create();
438 session->revocation = NULL;
439 session->modified = false;
440 session->invalidated = false;
441 session->loaded = false;
442
443 cwist_query_map *cookies = cwist_query_map_create();
444 if (cookies) {
445 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
446 cwist_cookie_parse(cookies, cookie_header);
447 const char *raw = cwist_cookie_get(cookies, app->session_name);
448 if (raw) {
449 /* On OOM the cookie is treated as absent: a fresh session. */
450 char *copy = cwist_strdup(raw);
451 char *dot = copy ? strchr(copy, '.') : NULL;
452 if (dot) {
453 *dot = '\0';
454 const char *signature = dot + 1;
455 if (verify_signature(app, copy, signature) && !session_is_revoked(app, signature)) {
456 time_t expires_at;
457 cwist_query_map *loaded = parse_payload(copy, &expires_at);
458 session_revocation *revocation = cwist_alloc(sizeof(*revocation));
459 char *signature_copy = cwist_strdup(signature);
460 if (loaded && revocation && signature_copy) {
461 cwist_gc_scope_disown(revocation);
462 cwist_gc_scope_disown(signature_copy);
463 revocation->app = app;
464 revocation->signature = signature_copy;
465 revocation->expires_at = expires_at;
466 revocation->next = NULL;
467 session->revocation = revocation;
468 cwist_query_map_destroy(session->data);
469 session->data = loaded;
470 session->loaded = true;
471 } else {
472 cwist_query_map_destroy(loaded);
473 cwist_free(revocation);
474 cwist_free(signature_copy);
475 }
476 }
477 }
478 cwist_free(copy);
479 }
480 cwist_query_map_destroy(cookies);
481 }
482
483 req->session = session;
484 return session;
485}

569int cwist_session_commit(cwist_session_t *session, cwist_http_response *res) {
570 if (!session || !res) return -1;
571 if (!session->modified && session->loaded) return 0;
572
573 if (session->invalidated) {
574 cwist_cookie_delete(res, session->app->session_name);
575 return 0;
576 }
577
578 json_ctx_t ctx;
579 ctx.json = cJSON_CreateObject();
580 if (!ctx.json) return -1;
581 time_t expires_at = time(NULL) + session->app->session_max_age;
582 char *session_id = generate_secret(16);
583 if (!session_id || !cJSON_AddNumberToObject(ctx.json, "__cwist_exp", (double)expires_at) ||
584 !cJSON_AddStringToObject(ctx.json, "__cwist_jti", session_id)) {
585 cwist_free(session_id);
586 cJSON_Delete(ctx.json);
587 return -1;
588 }
589 cwist_free(session_id);
590 cwist_query_map_foreach(session->data, add_to_json, &ctx);
591 char *raw = cJSON_PrintUnformatted(ctx.json);
592 cJSON_Delete(ctx.json);
593 if (!raw) return -1;
594
595 char *payload_b64 = session_base64_encode((const uint8_t *)raw, strlen(raw));
596 free(raw);
597 if (!payload_b64) return -1;
598 if (strlen(payload_b64) > 8192) {
599 cwist_free(payload_b64);
600 return -1;
601 }
602
603 uint8_t sig[32];
604 if (!session_hmac_sha256(session->app->session_secret, strlen(session->app->session_secret),
605 payload_b64, strlen(payload_b64), sig)) {
606 cwist_free(payload_b64);
607 return -1;
608 }
609 char *sig_b64 = session_base64_encode(sig, sizeof(sig));
610 if (!sig_b64) {
611 cwist_free(payload_b64);
612 return -1;
613 }
614 if (!session->revocation) {
615 session_revocation *revocation = cwist_alloc(sizeof(*revocation));
616 char *signature_copy = cwist_strdup(sig_b64);
617 if (!revocation || !signature_copy) {
618 cwist_free(revocation);
619 cwist_free(signature_copy);
620 cwist_free(payload_b64);
621 cwist_free(sig_b64);
622 return -1;
623 }
624 cwist_gc_scope_disown(revocation);
625 cwist_gc_scope_disown(signature_copy);
626 revocation->app = session->app;
627 revocation->signature = signature_copy;
628 revocation->expires_at = expires_at;
629 revocation->next = NULL;
630 session->revocation = revocation;
631 }
632
633 size_t cookie_len = strlen(payload_b64) + strlen(sig_b64) + 1;
634 char *cookie_value = cwist_alloc(cookie_len + 1);
635 if (!cookie_value) {
636 cwist_free(payload_b64);
637 cwist_free(sig_b64);
638 return -1;
639 }
640 snprintf(cookie_value, cookie_len + 1, "%s.%s", payload_b64, sig_b64);
641 cwist_free(payload_b64);
642 cwist_free(sig_b64);
643
644 cwist_cookie_options opts = {0};
645 opts.path = "/";
646 opts.max_age_seconds = session->app->session_max_age;
647 opts.http_only = true;
648 opts.secure = session->app->use_ssl;
649 opts.same_site = "Lax";
650
651 int rc = cwist_cookie_set(res, session->app->session_name, cookie_value, &opts);
652 cwist_free(cookie_value);
653 return rc;
654}
