174cwist_arena_t *cwist_arena_create(size_t generation_bytes) {
175 cwist_arena_tls_t *tls = cwist_arena_tls_get();
176 if (!tls) return NULL;
177
178 size_t bytes = generation_bytes ? generation_bytes : CWIST_ARENA_DEFAULT_GENERATION_BYTES;
179
180 /* Only default-sized buffers participate in the recycle cache. */
181 void *buffer = NULL;
182 if (bytes == CWIST_ARENA_DEFAULT_GENERATION_BYTES && tls->buf_count > 0) {
183 buffer = tls->bufs[--tls->buf_count];
184 }
185 if (!buffer) {
186 buffer = ttak_mem_alloc_with_flags_raw(bytes, __TTAK_UNSAFE_MEM_FOREVER__,
187 ttak_get_tick_count(), tls->env.config.alloc_flags);
188 if (!buffer) return NULL;
189 }
190
191 cwist_arena_t *arena = tls->struct_count > 0
192 ? tls->structs[--tls->struct_count]
193 : (cwist_arena_t *)calloc(1, sizeof(cwist_arena_t));
194 if (!arena) {
195 if (bytes == CWIST_ARENA_DEFAULT_GENERATION_BYTES &&
196 tls->buf_count < CWIST_ARENA_CACHE_MAX) {
197 tls->bufs[tls->buf_count++] = buffer;
198 } else {
199 ttak_mem_free(buffer);
200 }
201 return NULL;
202 }
203
204 arena->gen.base = buffer;
205 arena->gen.capacity = bytes;
206 arena->gen.used = 0;
207 arena->gen.epoch_id = ++tls->epoch_seq;
208 arena->generation_bytes = bytes;
209 arena->owner_tid = pthread_self();
210 return arena;
211}

213void *cwist_arena_alloc(cwist_arena_t *arena, size_t size) {
214 if (!arena || !size || !arena->gen.base) return NULL;
215 /* Plain 16-byte-aligned bump over the generation buffer. Skipping
216 * ttak_arena_generation_claim avoids its per-claim cache-line padding
217 * and scatter offset; exhaustion still returns NULL so callers fall
218 * back to the heap exactly as before. */
219 if (size > SIZE_MAX - 15u) return NULL;
220 size = (size + 15u) & ~15u;
221 if (size > arena->gen.capacity - arena->gen.used) return NULL;
222 void *ptr = (uint8_t *)arena->gen.base + arena->gen.used;
223 arena->gen.used += size;
224 return ptr;
225}

234void cwist_arena_destroy(cwist_arena_t *arena) {
235 if (!arena) return;
236 void *buffer = arena->gen.base;
237 size_t bytes = arena->generation_bytes;
238 bool default_sized = (bytes == CWIST_ARENA_DEFAULT_GENERATION_BYTES);
239
240 /* If destroyed on a foreign thread (e.g. cross-thread async completion),
241 * release directly to avoid polluting foreign thread-local caches. */
242 bool is_owner = pthread_equal(arena->owner_tid, pthread_self());
243 cwist_arena_tls_t *tls = is_owner ? cwist_arena_tls_get() : NULL;
244
245 if (tls && tls->struct_count < CWIST_ARENA_CACHE_MAX) {
246 /* Recycle the arena struct itself along with the buffer. */
247 tls->structs[tls->struct_count++] = arena;
248 } else {
249 free(arena);
250 }
251
252 if (!buffer) return;
253 if (tls && default_sized && tls->buf_count < CWIST_ARENA_CACHE_MAX) {
254 /* Recycle: the next request reuses this buffer with zero GC work. */
255 tls->bufs[tls->buf_count++] = buffer;
256 } else {
257 ttak_mem_free(buffer);
258 }
259}
