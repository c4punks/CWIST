37 size_t capacity = generation_bytes ? generation_bytes : CWIST_ARENA_DEFAULT_GENERATION_BYTES;
38 cwist_arena_t *arena = struct_cache_count > 0
39 ? struct_cache[--struct_cache_count]
40 : (cwist_arena_t *)calloc(1, sizeof(cwist_arena_t));

204cwist_arena_t *cwist_arena_create(size_t generation_bytes) {
205 cwist_arena_tls_t *tls = cwist_arena_tls_get();
206 if (!tls) return NULL;
207
208 size_t bytes = generation_bytes ? generation_bytes : CWIST_ARENA_DEFAULT_GENERATION_BYTES;
209
210 /* Only default-sized buffers participate in the recycle cache. */
211 void *buffer = NULL;
212 if (bytes == CWIST_ARENA_DEFAULT_GENERATION_BYTES && tls->buf_count > 0) {
213 buffer = tls->bufs[--tls->buf_count];
214 }
215 if (!buffer) {
216 buffer = ttak_mem_alloc_with_flags_raw(bytes, __TTAK_UNSAFE_MEM_FOREVER__,
217 ttak_get_tick_count(), tls->env.config.alloc_flags);
218 if (!buffer) return NULL;
219 }
220
221 cwist_arena_t *arena = tls->struct_count > 0
222 ? tls->structs[--tls->struct_count]
223 : (cwist_arena_t *)calloc(1, sizeof(cwist_arena_t));
224 if (!arena) {
225 if (bytes == CWIST_ARENA_DEFAULT_GENERATION_BYTES &&
226 tls->buf_count < CWIST_ARENA_CACHE_MAX) {
227 tls->bufs[tls->buf_count++] = buffer;
228 } else {
229 ttak_mem_free(buffer);
230 }
231 return NULL;
232 }
233
234 arena->gen.base = buffer;
235 arena->gen.capacity = bytes;
236 arena->gen.used = 0;
237 arena->gen.epoch_id = ++tls->epoch_seq;
238 arena->generation_bytes = bytes;
239 arena->owner_tid = pthread_self();
240 return arena;
241}

243void *cwist_arena_alloc(cwist_arena_t *arena, size_t size) {
244 if (!arena || !size || !arena->gen.base) return NULL;
245 /* Plain 16-byte-aligned bump over the generation buffer. Skipping
246 * ttak_arena_generation_claim avoids its per-claim cache-line padding
247 * and scatter offset; exhaustion still returns NULL so callers fall
248 * back to the heap exactly as before. */
249 if (size > SIZE_MAX - 15u) return NULL;
250 size = (size + 15u) & ~15u;
251 if (size > arena->gen.capacity - arena->gen.used) return NULL;
252 void *ptr = (uint8_t *)arena->gen.base + arena->gen.used;
253 arena->gen.used += size;
254 return ptr;
255}

264void cwist_arena_destroy(cwist_arena_t *arena) {
265 if (!arena) return;
266 void *buffer = arena->gen.base;
267 size_t bytes = arena->generation_bytes;
268 bool default_sized = (bytes == CWIST_ARENA_DEFAULT_GENERATION_BYTES);
269
270 /* If destroyed on a foreign thread (e.g. cross-thread async completion),
271 * release directly to avoid polluting foreign thread-local caches. */
272 bool is_owner = pthread_equal(arena->owner_tid, pthread_self());
273 cwist_arena_tls_t *tls = is_owner ? cwist_arena_tls_get() : NULL;
274
275 if (tls && tls->struct_count < CWIST_ARENA_CACHE_MAX) {
276 /* Recycle the arena struct itself along with the buffer. */
277 tls->structs[tls->struct_count++] = arena;
278 } else {
279 free(arena);
280 }
281
282 if (!buffer) return;
283 if (tls && default_sized && tls->buf_count < CWIST_ARENA_CACHE_MAX) {
284 /* Recycle: the next request reuses this buffer with zero GC work. */
285 tls->bufs[tls->buf_count++] = buffer;
286 } else {
287 ttak_mem_free(buffer);
288 }
289}
