113/* --- Async defer completion queue (internal; used by net/http/async.c) ---
114 *
115 * A handler on the HTTP/2 path may defer its response with
116 * cwist_async_defer(). The completing worker thread never writes frames
117 * itself (single-threaded HPACK/flow-control invariant); it enqueues the
118 * finished exchange on the per-connection queue and pokes the wake fd, and
119 * the connection thread drains the queue and emits HEADERS/DATA. The queue
120 * is refcounted: cwist_async_defer acquires a reference so a completion that
121 * lands after connection teardown is discarded safely instead of touching
122 * freed connection state. */
123typedef struct cwist_h2_async_queue cwist_h2_async_queue;
