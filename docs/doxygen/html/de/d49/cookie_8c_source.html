35char *cwist_cookie_encode(const char *value) {
36 if (!value) return NULL;
37 size_t len = strlen(value);
38 if (len > (SIZE_MAX - 1) / 3) return NULL;
39 size_t max = len * 3 + 1;
40 char *out = cwist_alloc(max);
41 if (!out) return NULL;
42 size_t j = 0;
43 for (size_t i = 0; i < len; i++) {
44 unsigned char c = (unsigned char)value[i];
45 if (needs_url_encode((char)c)) {
46 snprintf(out + j, max - j, "%%%02X", c);
47 j += 3;
48 } else {
49 out[j++] = (char)c;
50 }
51 }
52 out[j] = '\0';
53 return out;
54}

56int cwist_cookie_decode(const char *in, char *out, size_t out_len) {
57 if (!in || !out || out_len == 0) return -1;
58 size_t len = strlen(in);
59 size_t j = 0;
60 for (size_t i = 0; i < len; i++) {
61 if (j >= out_len - 1) return -1;
62 if (in[i] == '%' && i + 2 < len && isxdigit((unsigned char)in[i + 1]) &&
63 isxdigit((unsigned char)in[i + 2])) {
64 unsigned int hex;
65 if (sscanf(in + i + 1, "%2x", &hex) != 1) {
66 out[j++] = in[i];
67 continue;
68 }
69 out[j++] = (char)hex;
70 i += 2;
71 } else if (in[i] == '+') {
72 out[j++] = ' ';
73 } else {
74 out[j++] = in[i];
75 }
76 }
77 out[j] = '\0';
78 return (int)j;
79}

81void cwist_cookie_parse(cwist_query_map *map, const char *header) {
82 if (!map || !header || !*header) return;
83
84 char *buf = cwist_strdup(header);
85 if (!buf) return;
86
87 char *save = NULL;
88 char *pair = strtok_r(buf, ";", &save);
89 while (pair) {
90 /* trim leading whitespace */
91 while (*pair == ' ' || *pair == '\t') pair++;
92 char *eq = strchr(pair, '=');
93 if (eq) {
94 *eq = '\0';
95 char *name = pair;
96 char *value = eq + 1;
97 /* trim trailing whitespace on name */
98 size_t nlen = strlen(name);
99 while (nlen > 0 && (name[nlen - 1] == ' ' || name[nlen - 1] == '\t')) {
100 name[--nlen] = '\0';
101 }
102
103 if (nlen > 0) {
104 char *decoded = cwist_alloc(strlen(value) + 1);
105 if (decoded) {
106 if (cwist_cookie_decode(value, decoded, strlen(value) + 1) >= 0) {
107 cwist_query_map_set(map, name, decoded);
108 }
109 cwist_free(decoded);
110 }
111 }
112 }
113 pair = strtok_r(NULL, ";", &save);
114 }
115 cwist_free(buf);
116}

180int cwist_cookie_set(cwist_http_response *res, const char *name, const char *value,
181 const cwist_cookie_options *opts) {
182 if (!res || !cookie_name_is_valid(name) || !cookie_options_are_valid(opts)) return -1;
183
184 char *encoded = value ? cwist_cookie_encode(value) : cwist_strdup("");
185 if (!encoded) return -1;
186
187 cwist_sstring *cookie = cwist_sstring_create();
188 if (!cookie) {
189 cwist_free(encoded);
190 return -1;
191 }
192
193 bool ok = cookie_append(cookie, name) == 0 && cookie_append(cookie, "=") == 0 &&
194 cookie_append(cookie, encoded) == 0;
195 cwist_free(encoded);
196
197 if (ok && opts) {
198 if (opts->path) ok = cookie_append_attr(cookie, "; Path=", opts->path) == 0;
199 if (ok && opts->domain) ok = cookie_append_attr(cookie, "; Domain=", opts->domain) == 0;
200 if (ok && opts->max_age_seconds >= 0) {
201 char buf[64];
202 snprintf(buf, sizeof(buf), "; Max-Age=%d", opts->max_age_seconds);
203 ok = cookie_append(cookie, buf) == 0;
204 }
205 if (ok && opts->http_only) ok = cookie_append(cookie, "; HttpOnly") == 0;
206 if (ok && opts->secure) ok = cookie_append(cookie, "; Secure") == 0;
207 if (ok && opts->same_site)
208 ok = cookie_append_attr(cookie, "; SameSite=", opts->same_site) == 0;
209 }
210
211 int rc = ok ? add_set_cookie(res, cookie->data) : -1;
212 cwist_sstring_destroy(cookie);
213 return rc;
214}
