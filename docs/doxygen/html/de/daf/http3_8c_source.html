345static void cwist_h3_hset_track(cwist_http3_context *ctx, cwist_h3_hset_t *hset) {
346 if (!ctx || !hset) return;
347 hset->owner = ctx;
348 hset->next = (cwist_h3_hset_t *)ctx->hsets;
349 hset->prev = (cwist_h3_hset_t **)&ctx->hsets;
350 if (hset->next) hset->next->prev = &hset->next;
351 ctx->hsets = hset;
352}

358static void cwist_h3_hset_untrack(cwist_h3_hset_t *hset) {
359 if (!hset || !hset->owner) return;
360 *hset->prev = hset->next;
361 if (hset->next) hset->next->prev = hset->prev;
362 hset->next = NULL;
363 hset->prev = NULL;
364 hset->owner = NULL;
365}

413static struct lsxpack_header *cwist_h3_hsi_prepare(void *hset_p, struct lsxpack_header *xhdr,
414 size_t req_space) {
415 cwist_h3_hset_t *hset = hset_p;
416 if (!hset) return NULL;
417
418 if (xhdr) {
419 if (req_space > LSXPACK_MAX_STRLEN || xhdr->name_offset < 0 ||
420 (size_t)xhdr->name_offset >= sizeof(hset->decode_buf) ||
421 req_space > sizeof(hset->decode_buf) - (size_t)xhdr->name_offset) {
422 fprintf(stderr,
423 "[HTTP/3] Rejecting oversized QPACK header resize (space=%zu, offset=%d)\n",
424 req_space, (int)xhdr->name_offset);
425 return NULL;
426 }
427 xhdr->val_len = (lsxpack_strlen_t)req_space;
428 return xhdr;
429 }
430
431 if (hset->count >= H3_MAX_HEADERS) return NULL;
432 if (req_space > sizeof(hset->decode_buf) - hset->decode_off) return NULL;
433 lsxpack_header_prepare_decode(&hset->headers[hset->count], hset->decode_buf, hset->decode_off,
434 sizeof(hset->decode_buf) - hset->decode_off);
435 return &hset->headers[hset->count];
436}

446static int cwist_h3_hsi_process_header(void *hset_p, struct lsxpack_header *xhdr) {
447 cwist_h3_hset_t *hset = hset_p;
448 /* A NULL header marks the end of a header block. */
449 if (!hset || !xhdr) return 0;
450
451 /* The QPACK decoder exposes the exact storage used by this completed
452 * header. */
453 size_t total = lsxpack_header_get_dec_size(xhdr);
454 if (total > sizeof(hset->decode_buf) - hset->decode_off) {
455 fprintf(stderr, "[HTTP/3] Rejecting oversized QPACK header (size=%zu, used=%zu)\n", total,
456 hset->decode_off);
457 return -1;
458 }
459 hset->decode_off += total;
460 hset->count++;
461 return 0;
462}

516static void h3_setup_cmsg(struct msghdr *msg, char *cbuf, size_t cbuf_sz,
517 const struct lsquic_out_spec *spec, uint16_t gso_seg) {
518 msg->msg_control = cbuf;
519 msg->msg_controllen = cbuf_sz;
520 memset(cbuf, 0, cbuf_sz);
521 size_t ctl_len = 0;
522
523#if defined(__linux__) && defined(UDP_SEGMENT)
524 if (gso_seg > 0) {
525 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
526 cmsg->cmsg_level = SOL_UDP;
527 cmsg->cmsg_type = UDP_SEGMENT;
528 cmsg->cmsg_len = CMSG_LEN(sizeof(uint16_t));
529 memcpy(CMSG_DATA(cmsg), &gso_seg, sizeof(uint16_t));
530 ctl_len += CMSG_SPACE(sizeof(uint16_t));
531 }
532#else
533 (void)gso_seg;
534#endif
535
536 if (spec->local_sa && spec->dest_sa) {
537 if (spec->dest_sa->sa_family == AF_INET && spec->local_sa->sa_family == AF_INET) {
538 struct in_addr addr = ((const struct sockaddr_in *)spec->local_sa)->sin_addr;
539 if (addr.s_addr != INADDR_ANY) {
540#if defined(__linux__) && defined(IP_PKTINFO)
541 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
542 cmsg->cmsg_level = IPPROTO_IP;
543 cmsg->cmsg_type = IP_PKTINFO;
544 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in_pktinfo));
545 struct in_pktinfo info = {0};
546 info.ipi_spec_dst = addr;
547 memcpy(CMSG_DATA(cmsg), &info, sizeof(info));
548 ctl_len += CMSG_SPACE(sizeof(struct in_pktinfo));
549#elif defined(IP_SENDSRCADDR)
550 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
551 cmsg->cmsg_level = IPPROTO_IP;
552 cmsg->cmsg_type = IP_SENDSRCADDR;
553 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in_addr));
554 memcpy(CMSG_DATA(cmsg), &addr, sizeof(addr));
555 ctl_len += CMSG_SPACE(sizeof(struct in_addr));
556#endif
557 }
558 } else if (spec->dest_sa->sa_family == AF_INET6 && spec->local_sa->sa_family == AF_INET6) {
559 const struct in6_addr *addr6 =
560 &((const struct sockaddr_in6 *)spec->local_sa)->sin6_addr;
561 if (memcmp(addr6, &in6addr_any, sizeof(struct in6_addr)) != 0) {
562#if defined(IPV6_PKTINFO)
563 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
564 cmsg->cmsg_level = IPPROTO_IPV6;
565 cmsg->cmsg_type = IPV6_PKTINFO;
566 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in6_pktinfo));
567 struct in6_pktinfo info6 = {0};
568 info6.ipi6_addr = *addr6;
569 memcpy(CMSG_DATA(cmsg), &info6, sizeof(info6));
570 ctl_len += CMSG_SPACE(sizeof(struct in6_pktinfo));
571#endif
572 }
573 }
574 }
575
576#if defined(CWIST_H3_HAVE_ECN_CMSG)
577 if (spec->ecn && spec->dest_sa) {
578 if (spec->dest_sa->sa_family == AF_INET) {
579 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
580 cmsg->cmsg_level = IPPROTO_IP;
581 cmsg->cmsg_type = IP_TOS;
582 int tos = spec->ecn;
583 cmsg->cmsg_len = CMSG_LEN(sizeof(tos));
584 memcpy(CMSG_DATA(cmsg), &tos, sizeof(tos));
585 ctl_len += CMSG_SPACE(sizeof(tos));
586 }
587#if defined(IPV6_TCLASS)
588 else if (spec->dest_sa->sa_family == AF_INET6) {
589 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
590 cmsg->cmsg_level = IPPROTO_IPV6;
591 cmsg->cmsg_type = IPV6_TCLASS;
592 int tos = spec->ecn;
593 cmsg->cmsg_len = CMSG_LEN(sizeof(tos));
594 memcpy(CMSG_DATA(cmsg), &tos, sizeof(tos));
595 ctl_len += CMSG_SPACE(sizeof(tos));
596 }
597#endif
598 }
599#endif
600
601 msg->msg_controllen = ctl_len;
602 if (ctl_len == 0) {
603 msg->msg_control = NULL;
604 }
605}

666 return false;
667 }

712 memset(msg, 0, sizeof(*msg));
713 msg->msg_name = (void *)spec->dest_sa;
714 msg->msg_namelen = (spec->dest_sa && spec->dest_sa->sa_family == AF_INET)
715 ? sizeof(struct sockaddr_in)
716 : sizeof(struct sockaddr_in6);
717 msg->msg_iov = (struct iovec *)spec->iov;
718 msg->msg_iovlen = spec->iovlen;
719 h3_setup_cmsg(msg, ctrl_bufs[k], sizeof(ctrl_bufs[k]), spec, 0);
720 msgs[k].msg_len = 0;
721 }
722
723 int res = sendmmsg(udp_fd, msgs, batch, MSG_DONTWAIT);
724 if (res > 0) {
725 i += (unsigned)res;
726 if ((unsigned)res < batch) break;
727 continue;
728 }
729 if (res < 0) {
730 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
731 if (i > 0) return (int)i;
732 int nw = h3_send_one(udp_fd, &specs[i]);
733 if (nw >= 0) {
734 i++;
735 continue;
736 }
737 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
738 return -1;
739 }
740 break;
741 }
742 return (int)i;
743}

761static int cwist_h3_packets_out(void *ctx, const struct lsquic_out_spec *specs, unsigned n_specs) {
762 int udp_fd = *(int *)ctx;
763#if defined(__linux__)
764 /* One-time GSO probe: check env var on first call. */
765 if (__builtin_expect(h3_gso_state < 0, 0)) {
766 const char *no_gso = getenv("CWIST_H3_NO_GSO");
767 h3_gso_state = (no_gso && no_gso[0] == '1') ? 1 : 0;
768 }
769
770 if (!h3_gso_state) {
771 /* GSO fast path: coalesce runs of equal-size datagrams to the same
772 * peer into one sendmsg with UDP_SEGMENT. */
773 unsigned i = 0;
774 while (i < n_specs) {
775 size_t seg = h3_spec_len(&specs[i]);
776 unsigned run = 1;
777 unsigned niov = specs[i].iovlen;
778 if (seg > 0 && seg <= 65535) {
779 /* Total payload must fit in one UDP datagram (65535 bytes);
780 * 65536/seg could allow an exactly-64KiB super-packet whose
781 * sendmsg fails with EMSGSIZE and would disable GSO. */
782 unsigned max_segs = (unsigned)(65535 / seg);
783 if (max_segs > 48) max_segs = 48;
784 while (i + run < n_specs && run < max_segs && specs[i + run].ecn == specs[i].ecn &&
785 h3_same_dest(&specs[i], &specs[i + run]) &&
786 h3_spec_len(&specs[i + run]) == seg && niov + specs[i + run].iovlen <= 128) {
787 niov += specs[i + run].iovlen;
788 run++;
789 }
790 }
791 if (run >= 2) {
792 struct iovec iov[128];
793 unsigned n = 0;
794 for (unsigned k = 0; k < run; k++)
795 for (unsigned m = 0; m < specs[i + k].iovlen; m++)
796 iov[n++] = specs[i + k].iov[m];
797
798 char ctrl[512];
799 struct msghdr msg = {0};
800 msg.msg_name = (void *)specs[i].dest_sa;
801 msg.msg_namelen = (specs[i].dest_sa && specs[i].dest_sa->sa_family == AF_INET)
802 ? sizeof(struct sockaddr_in)
803 : sizeof(struct sockaddr_in6);
804 msg.msg_iov = iov;
805 msg.msg_iovlen = n;
806 h3_setup_cmsg(&msg, ctrl, sizeof(ctrl), &specs[i], (uint16_t)seg);
807
808 ssize_t nw = sendmsg(udp_fd, &msg, MSG_DONTWAIT);
809 if (nw >= 0) {
810 i += run;
811 continue;
812 }
813 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
814 /* GSO rejected: permanently disable and fall back to sendmmsg. */
815 h3_gso_state = 1;
816 return h3_sendmmsg_batch(udp_fd, specs, i, n_specs);
817 }
818 /* Single packet: send directly. */
819 int nw = h3_send_one(udp_fd, &specs[i]);
820 if (nw < 0) {
821 if (errno == EAGAIN || errno == EWOULDBLOCK) return (int)i;
822 if (errno == ECONNREFUSED || errno == ENETUNREACH || errno == EHOSTUNREACH ||
823 errno == EMSGSIZE) {
824 i++;
825 continue;
826 }
827 return (i > 0) ? (int)i : -1;
828 }
829 i++;
830 }
831 return (int)i;
832 }
833
834 /* sendmmsg fallback path (GSO disabled). */
835 return h3_sendmmsg_batch(udp_fd, specs, 0, n_specs);
836#else
837 unsigned i = 0;
838 for (; i < n_specs; ++i) {
839 int nw = h3_send_one(udp_fd, &specs[i]);
840 if (nw < 0) {
841 if (errno == EAGAIN || errno == EWOULDBLOCK) return (int)i;
842 if (errno == ECONNREFUSED || errno == ENETUNREACH || errno == EHOSTUNREACH ||
843 errno == EMSGSIZE)
844 continue;
845 return (i > 0) ? (int)i : -1;
846 }
847 }
848 return (int)i;
849#endif
850}

1193static int h3_seq_append_and_feed(h3_stream_ctx_t *st, const unsigned char *data, size_t len) {
1194 if (len > SIZE_MAX - st->seq_len) return -1;
1195 size_t need = st->seq_len + len;
1196 if (need > st->seq_cap) {
1197 size_t cap = st->seq_cap ? st->seq_cap : 4096;
1198 while (cap < need) {
1199 if (cap > (CWIST_SEQ_HEADER_SIZE + UINT16_MAX) / 2) {
1200 cap = CWIST_SEQ_HEADER_SIZE + UINT16_MAX;
1201 break;
1202 }
1203 cap *= 2;
1204 }
1205 if (cap < need || cap > CWIST_SEQ_HEADER_SIZE + UINT16_MAX) return -1;
1206 unsigned char *tmp = realloc(st->seq_buf, cap);
1207 if (!tmp) return -1;
1208 st->seq_buf = tmp;
1209 st->seq_cap = cap;
1210 }
1211 memcpy(st->seq_buf + st->seq_len, data, len);
1212 st->seq_len += len;
1213
1214 while (st->seq_len >= CWIST_SEQ_HEADER_SIZE) {
1215 size_t payload_len = ((size_t)st->seq_buf[4] << 8) | st->seq_buf[5];
1216 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + payload_len;
1217 if (payload_len == 0 || st->seq_len < chunk_len) break;
1218 cwist_seq_chunk_t chunk;
1219 if (!cwist_seq_chunk_parse(st->seq_buf, chunk_len, &chunk)) return -1;
1220 if (!st->body_assembler) {
1221 st->body_assembler = cwist_seq_assembler_create_limited(CWIST_HTTP_MAX_BODY_SIZE);
1222 if (!st->body_assembler) return -1;
1223 }
1224 if (!cwist_seq_assembler_feed(st->body_assembler, &chunk)) return -1;
1225 memmove(st->seq_buf, st->seq_buf + chunk_len, st->seq_len - chunk_len);
1226 st->seq_len -= chunk_len;
1227 }
1228 return 0;
1229}

1269static bool h3_process_stream_headers(lsquic_stream_t *stream, h3_stream_ctx_t *st) {
1270 if (st->headers_done) return true;
1271 void *hset = lsquic_stream_get_hset(stream);
1272 if (!hset) return false;
1273 cwist_h3_hset_t *hs = (cwist_h3_hset_t *)hset;
1274 /* lsxpack exposes counted slices, not C strings, so each header needs a
1275 * NUL-terminated scratch copy to hand to strcmp()/h3_apply_header().
1276 * These used to be a malloc(name_len+1)/malloc(value_len+1) pair freed
1277 * at the bottom of every loop iteration - up to two heap round trips
1278 * per header, tens of them on a real request. The bounds below
1279 * (name_len <= 1024, value_len <= H3_DECODE_BUF_SIZE - 1) are already
1280 * enforced before anything touches these buffers, so a single
1281 * reusable pair sized to those same bounds, declared once outside the
1282 * loop, replaces all of that: h3_apply_header() copies out of them
1283 * before the next iteration overwrites them (see below), so reuse is
1284 * safe. */
1285 char name_scratch[1024 + 1];
1286 char value_scratch[H3_DECODE_BUF_SIZE];
1287 for (size_t i = 0; i < hs->count; ++i) {
1288 const struct lsxpack_header *xhdr = &hs->headers[i];
1289 const char *raw_name = lsxpack_header_get_name(xhdr);
1290 const char *raw_value = lsxpack_header_get_value(xhdr);
1291 size_t name_len = xhdr->name_len;
1292 size_t value_len = xhdr->val_len;
1293 if (raw_name && raw_value && name_len > 0 && name_len <= 1024 &&
1294 value_len <= H3_DECODE_BUF_SIZE - 1) {
1295 char *name = name_scratch;
1296 char *value = value_scratch;
1297 memcpy(name, raw_name, name_len);
1298 name[name_len] = '\0';
1299 memcpy(value, raw_value, value_len);
1300 value[value_len] = '\0';
1301 if (name[0] == ':') {
1302 /* RFC 9114 Section 4.3.1: pseudo-headers precede regular
1303 * headers, appear at most once, and only the defined set is
1304 * valid in requests. */
1305 unsigned bit = 0;
1306 if (strcmp(name, ":method") == 0)
1307 bit = H3_PSEUDO_METHOD;
1308 else if (strcmp(name, ":scheme") == 0)
1309 bit = H3_PSEUDO_SCHEME;
1310 else if (strcmp(name, ":path") == 0)
1311 bit = H3_PSEUDO_PATH;
1312 else if (strcmp(name, ":authority") == 0)
1313 bit = H3_PSEUDO_AUTHORITY;
1314 else if (strcmp(name, ":protocol") == 0)
1315 bit = H3_PSEUDO_PROTOCOL;
1316 if (bit == 0 || st->seen_regular_header || (st->pseudo_seen & bit)) {
1317 h3_reject_malformed_request(stream, st);
1318 return false;
1319 }
1320 st->pseudo_seen |= bit;
1321 if (bit == H3_PSEUDO_METHOD && strcmp(value, "CONNECT") == 0) {
1322 st->is_connect = 1;
1323 }
1324 if (bit == H3_PSEUDO_PATH && value_len == 0) {
1325 st->saw_empty_path = 1;
1326 }
1327 } else {
1328 st->seen_regular_header = 1;
1329 }
1330 h3_apply_header(st->req, name, value);
1331#ifdef CWIST_WEBTRANSPORT
1332 if (strcmp(name, ":protocol") == 0 && strcmp(value, "webtransport") == 0) {
1333 if (st->req && st->req->method == CWIST_HTTP_CONNECT) {
1334 st->is_webtransport = 1;
1335 }
1336 }
1337#endif
1338 }
1339 }
1340 /* RFC 9114 Section 4.3.1 completeness rules, evaluated once the whole
1341 * header block has been seen. */
1342 if (st->is_connect && !(st->pseudo_seen & H3_PSEUDO_PROTOCOL)) {
1343 /* Plain CONNECT: :scheme and :path MUST be omitted, :authority is
1344 * required. */
1345 if ((st->pseudo_seen & (H3_PSEUDO_SCHEME | H3_PSEUDO_PATH)) ||
1346 !(st->pseudo_seen & H3_PSEUDO_AUTHORITY)) {
1347 h3_reject_malformed_request(stream, st);
1348 return false;
1349 }
1350 } else if (st->is_connect) {
1351 /* Extended CONNECT (RFC 9220, e.g. WebTransport): :scheme, :path and
1352 * :authority are all required. */
1353 if (!(st->pseudo_seen & H3_PSEUDO_SCHEME) || !(st->pseudo_seen & H3_PSEUDO_PATH) ||
1354 !(st->pseudo_seen & H3_PSEUDO_AUTHORITY)) {
1355 h3_reject_malformed_request(stream, st);
1356 return false;
1357 }
1358 } else {
1359 /* Normal request: :method, :scheme and :path are mandatory,
1360 * :authority (or an equivalent Host header) is required for https,
1361 * and :protocol is only legal on extended CONNECT. */
1362 if (!(st->pseudo_seen & H3_PSEUDO_METHOD) || !(st->pseudo_seen & H3_PSEUDO_SCHEME) ||
1363 !(st->pseudo_seen & H3_PSEUDO_PATH) || (st->pseudo_seen & H3_PSEUDO_PROTOCOL) ||
1364 (!(st->pseudo_seen & H3_PSEUDO_AUTHORITY) &&
1365 !cwist_http_header_get(st->req->headers, "host"))) {
1366 h3_reject_malformed_request(stream, st);
1367 return false;
1368 }
1369 /* An empty :path is only legal for OPTIONS (asterisk-form). */
1370 if (st->saw_empty_path && (!st->req || st->req->method != CWIST_HTTP_OPTIONS)) {
1371 h3_reject_malformed_request(stream, st);
1372 return false;
1373 }
1374 }
1375 st->headers_done = 1;
1376 char *seq_header = cwist_http_header_get(st->req->headers, "x-cwist-sequenced-data");
1377 st->sequenced_data =
1378 seq_header && (strcmp(seq_header, "1") == 0 || strcasecmp(seq_header, "true") == 0);
1379 return true;
1380}

1408static void cwist_h3_on_read(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
1409 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
1410 if (!st) return;
1411
1412 unsigned char buf[8192];
1413 ssize_t nread;
1414
1415 h3_process_stream_headers(stream, st);
1416 if (st->malformed) return;
1417
1418 while ((nread = lsquic_stream_read(stream, buf, sizeof(buf))) > 0) {
1419 if (st->sequenced_data) {
1420 if (h3_seq_append_and_feed(st, buf, (size_t)nread) != 0) {
1421 lsquic_stream_close(stream);
1422 return;
1423 }
1424 continue;
1425 }
1426 size_t need = st->body_len + (size_t)nread;
1427 if (need > CWIST_HTTP_MAX_BODY_SIZE) {
1428 /* Body too large: answer 413 with a small JSON body instead of a
1429 * bare stream close, so the client sees an explicit status
1430 * (RFC 9110 Section 15.5.14). */
1431 if (!st->res) st->res = cwist_http_response_create();
1432 if (st->res) {
1433 st->res->status_code = 413; /* Payload Too Large */
1434 if (st->res->body) {
1435 cwist_sstring_assign(st->res->body, "{\"error\":\"payload too large\"}");
1436 }
1437 if (st->res->headers) {
1438 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1439 }
1440 st->response_ready = 1;
1441 lsquic_stream_wantread(stream, 0);
1442 lsquic_stream_shutdown(stream, 0);
1443 lsquic_stream_wantwrite(stream, 1);
1444 } else {
1445 lsquic_stream_close(stream);
1446 }
1447 return;
1448 }
1449 if (need > st->body_cap) {
1450 size_t new_cap = st->body_cap ? st->body_cap * 2 : 4096;
1451 while (new_cap < need) new_cap *= 2;
1452 char *tmp = realloc(st->body, new_cap);
1453 if (!tmp) {
1454 lsquic_stream_close(stream);
1455 return;
1456 }
1457 st->body = tmp;
1458 st->body_cap = new_cap;
1459 }
1460 memcpy(st->body + st->body_len, buf, (size_t)nread);
1461 st->recv_xor ^= h3_xor_bytes((const unsigned char *)buf, (size_t)nread);
1462 st->body_len += (size_t)nread;
1463 }
1464
1465 if (nread == 0) {
1466 /* End of stream (FIN received) */
1467 h3_process_stream_headers(stream, st);
1468
1469 if (st->malformed) return;
1470
1471 if (!st->headers_done) {
1472 /* Malformed request: no headers before FIN */
1473 st->res = cwist_http_response_create();
1474 if (st->res) {
1475 st->res->status_code = CWIST_HTTP_BAD_REQUEST;
1476 if (st->res->body) {
1477 cwist_sstring_assign(st->res->body, "{\"error\":\"malformed request\"}");
1478 }
1479 if (st->res->headers) {
1480 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1481 }
1482 }
1483 st->response_ready = 1;
1484 lsquic_stream_wantread(stream, 0);
1485 lsquic_stream_shutdown(stream, 0);
1486 lsquic_stream_wantwrite(stream, 1);
1487 return;
1488 }
1489
1490 if (st->sequenced_data) {
1491 const uint8_t *assembled = NULL;
1492 size_t assembled_len = 0;
1493 if (st->seq_len != 0 || !st->body_assembler ||
1494 !cwist_seq_assembler_get_data(st->body_assembler, &assembled, &assembled_len)) {
1495 /* Do not dispatch a partial request. */
1496 st->res = cwist_http_response_create();
1497 if (st->res) {
1498 st->res->status_code = CWIST_HTTP_BAD_REQUEST;
1499 cwist_http_header_add(&st->res->headers, "x-cwist-retry", "1");
1500 if (st->res->body) {
1501 cwist_sstring_assign(st->res->body,
1502 "{\"error\":\"incomplete sequenced body, retry\"}");
1503 }
1504 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1505 }
1506 st->response_ready = 1;
1507 lsquic_stream_wantread(stream, 0);
1508 lsquic_stream_shutdown(stream, 0);
1509 lsquic_stream_wantwrite(stream, 1);
1510 return;
1511 }
1512 if (assembled_len == 0) {
1513 /* realloc(p, 0) may legally return NULL; do not mistake an
1514 * empty assembled body for an allocation failure (which used
1515 * to reset the stream instead of dispatching the request). */
1516 free(st->body);
1517 st->body = NULL;
1518 st->body_len = 0;
1519 st->body_cap = 0;
1520 } else {
1521 st->body = realloc(st->body, assembled_len);
1522 if (!st->body) {
1523 lsquic_stream_close(stream);
1524 return;
1525 }
1526 memcpy(st->body, assembled, assembled_len);
1527 st->body_len = assembled_len;
1528 st->body_cap = assembled_len;
1529 }
1530 }
1531
1532 if (st->body_len > 0 && st->req && st->req->body) {
1533 cwist_sstring_assign_len(st->req->body, st->body, st->body_len);
1534 }
1535
1536 st->res = cwist_http_response_create();
1537 if (st->res && st->req) {
1538 cwist_http3_context *h3_ctx = h3_shared_ctx(lsquic_stream_conn(stream));
1539#ifdef CWIST_WEBTRANSPORT
1540 if (st->is_webtransport && h3_ctx && h3_ctx->wt_handler) {
1541 char *host = cwist_http_header_get(st->req->headers, "host");
1542 char *origin = cwist_http_header_get(st->req->headers, "origin");
1543 struct lsquic_wt_connect_info info = {
1544 .wtci_authority = host,
1545 .wtci_path = st->req->path ? st->req->path->data : NULL,
1546 .wtci_origin = origin,
1547 .wtci_protocol = NULL,
1548 .wtci_draft = lsquic_wt_peer_draft(lsquic_stream_conn(stream)),
1549 };
1550 struct lsquic_wt_accept_params params = {
1551 .wtap_status = LSQUIC_WTAP_STATUS_DEFAULT,
1552 .wtap_wt_if = &cwist_h3_wt_if,
1553 .wtap_wt_if_ctx = st,
1554 .wtap_connect_info = &info,
1555 .wtap_datagram_send_mode = LSQUIC_HTTP_DG_SEND_DEFAULT,
1556 };
1557 if (lsquic_wt_accept(stream, &params) == 0) {
1558 st->wt_taken = 1;
1559 lsquic_stream_wantread(stream, 0);
1560 lsquic_stream_shutdown(stream, 0);
1561 lsquic_stream_wantwrite(stream, 0);
1562 return;
1563 }
1564 /* Accept failed: falling through with the default 200/empty
1565 * response used to send the client a silent 0-byte body. */
1566 CWIST_LOG_WARN("[HTTP/3] WebTransport accept failed; answering 500");
1567 st->res->status_code = CWIST_HTTP_INTERNAL_ERROR;
1568 if (st->res->body && st->res->body->size == 0) {
1569 cwist_sstring_assign(st->res->body,
1570 "{\"error\":\"webtransport accept failed\"}");
1571 }
1572 if (st->res->headers) {
1573 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1574 }
1575 } else
1576#endif
1577 if (h3_ctx && h3_ctx->handler) {
1578 /* 0-RTT replay guard: requests delivered while the QUIC
1579 * handshake is still in progress arrived as early data.
1580 * lsquic (this baseline) exposes no
1581 * lsquic_conn_is_early_data_accepted()-style query, so the
1582 * guard uses the public lsquic_conn_status() instead. */
1583 bool is_early_data = h3_ctx->early_data_enabled && h3_ctx->early_data_guard &&
1584 lsquic_conn_status(lsquic_stream_conn(stream), NULL, 0) ==
1585 LSCONN_ST_HSK_IN_PROGRESS;
1586 if (is_early_data && st->req && !h3_method_is_idempotent(st->req->method)) {
1587 /* RFC 8470: refuse replayable non-idempotent early data
1588 * so the client retries after the handshake. Attach a
1589 * body: without one the response goes out as HEADERS+FIN
1590 * with no content-length, which some clients surface as
1591 * an empty (0-byte) reply instead of status 425. */
1592 st->res->status_code = 425; /* Too Early */
1593 if (st->res->body) {
1594 cwist_sstring_assign(st->res->body,
1595 "{\"error\":\"too early, retry after handshake\"}");
1596 }
1597 if (st->res->headers) {
1598 cwist_http_header_add(&st->res->headers, "Content-Type",
1599 "application/json");
1600 }
1601 } else {
1602 h3_ctx->handler(h3_ctx->user_ctx, st->req, st->res);
1603 }
1604 }
1605 }
1606 st->response_ready = 1;
1607 lsquic_stream_wantread(stream, 0);
1608 lsquic_stream_shutdown(stream, 0);
1609 lsquic_stream_wantwrite(stream, 1);
1610 } else if (nread < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
1611 lsquic_stream_close(stream);
1612 }
1613}

1651static void cwist_h3_on_write(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
1652 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
1653 if (!st || !st->response_ready) return;
1654 if (!st->res) {
1655 /* Response allocation failed upstream; reset instead of dereferencing
1656 * NULL (which used to take down every stream on the connection). */
1657 lsquic_stream_close(stream);
1658 return;
1659 }
1660
1661 if (st->write_state == 0) {
1662 struct lsxpack_header headers_arr[H3_MAX_RESPONSE_HEADERS];
1663 char hbuf[H3_RESPONSE_HEADER_BUF_SIZE];
1664 size_t hbuf_off = 0;
1665 size_t hdr_count = 0;
1666
1667 bool bodyless = h3_status_forbids_body(st->res->status_code);
1668 bool is_head = st->req && st->req->method == CWIST_HTTP_HEAD;
1669
1670 /* Handler attached a body source that cannot produce bytes
1671 * (NULL pointer body / invalid file fd with a non-zero length).
1672 * Sending HEADERS would announce (or imply) a body we cannot
1673 * deliver; close the stream so the client sees an explicit error
1674 * instead of a silent empty 200. */
1675 if (!bodyless && !is_head &&
1676 ((st->res->is_ptr_body && !st->res->ptr_body && st->res->ptr_body_len > 0) ||
1677 (st->res->use_file_stream && st->res->file_stream_fd < 0 &&
1678 st->res->file_stream_len > 0))) {
1679 CWIST_LOG_WARN("[HTTP/3] unusable response body source; resetting stream");
1680 lsquic_stream_close(stream);
1681 return;
1682 }
1683
1684 /* :status */
1685 if (st->res->status_code < 200 || st->res->status_code > 999) {
1686 /* RFC 9110 Section 15: a final response needs a status in
1687 * 200-999 (RFC 9114 Section 4.1.2 requires three digits).
1688 * A handler that leaves the status unset, picks 1xx (an
1689 * interim response, illegal as the complete answer), or an
1690 * out-of-range value gets coerced to 500 instead of putting
1691 * a malformed :status on the wire. */
1692 CWIST_LOG_WARN("[HTTP/3] invalid status %d; coercing to 500", st->res->status_code);
1693 st->res->status_code = CWIST_HTTP_INTERNAL_ERROR;
1694 bodyless = false;
1695 if (!is_head && st->res->body && st->res->body->size == 0 && !st->res->is_ptr_body &&
1696 !st->res->use_file_stream) {
1697 cwist_sstring_assign(st->res->body, "{\"error\":\"internal server error\"}");
1698 }
1699 }
1700 int status_code = st->res->status_code;
1701 char status_str[16];
1702 snprintf(status_str, sizeof(status_str), "%d", status_code);
1703 size_t slen = strlen(status_str);
1704 if (hdr_count < H3_MAX_RESPONSE_HEADERS && slen > 0 &&
1705 hbuf_off + 7 + 2 + slen <= sizeof(hbuf)) {
1706 memcpy(hbuf + hbuf_off, ":status", 7);
1707 memcpy(hbuf + hbuf_off + 9, status_str, slen);
1708 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0, 7, 9, slen);
1709 hbuf_off += 9 + slen;
1710 hdr_count++;
1711 }
1712
1713 /* content-length */
1714 size_t body_len = 0;
1715 if (st->res->use_file_stream) {
1716 if (st->res->file_stream_fd >= 0) body_len = st->res->file_stream_len;
1717 } else if (st->res->is_ptr_body)
1718 body_len = st->res->ptr_body ? st->res->ptr_body_len : 0;
1719 else if (st->res->body)
1720 body_len = st->res->body->size;
1721
1722 const char *user_cl =
1723 st->res->headers ? cwist_http_header_get(st->res->headers, "content-length") : NULL;
1724
1725 if (!bodyless && (body_len > 0 || is_head) && hdr_count < H3_MAX_RESPONSE_HEADERS) {
1726 /* For HEAD, content-length describes the would-be GET body: keep
1727 * the handler-provided value when present (e.g. static file size
1728 * with an empty body), otherwise compute it like a GET. For all
1729 * other methods the computed body length is authoritative: it is
1730 * exactly what the write path below will send, so trusting a
1731 * divergent handler value would announce bytes never delivered. */
1732 char cl_str[32];
1733 const char *cl_val = NULL;
1734 if (is_head && user_cl) {
1735 if (h3_content_length_is_valid(user_cl)) {
1736 cl_val = user_cl;
1737 } else {
1738 /* Malformed handler content-length must not reach the
1739 * wire; fall back to the computed body length. */
1740 CWIST_LOG_WARN("[HTTP/3] dropping invalid HEAD content-length '%s'", user_cl);
1741 }
1742 }
1743 if (!cl_val) {
1744 snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
1745 cl_val = cl_str;
1746 }
1747 if (cl_val && cl_val[0] != '\0') {
1748 size_t cl_name_len = 14;
1749 size_t cl_val_len = strlen(cl_val);
1750 size_t total = cl_name_len + 2 + cl_val_len;
1751 if (cl_val_len > 0 && hbuf_off + total <= sizeof(hbuf)) {
1752 memcpy(hbuf + hbuf_off, "content-length", cl_name_len);
1753 memcpy(hbuf + hbuf_off + cl_name_len + 2, cl_val, cl_val_len);
1754 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0,
1755 cl_name_len, cl_name_len + 2, cl_val_len);
1756 hbuf_off += total;
1757 hdr_count++;
1758 }
1759 }
1760 }
1761
1762 /* content-type (if body is present or HEAD request) */
1763 if (st->res->headers && (body_len > 0 || is_head)) {
1764 char *ct = cwist_http_header_get(st->res->headers, "content-type");
1765 if (ct && ct[0] != '\0' && hdr_count < H3_MAX_RESPONSE_HEADERS) {
1766 size_t klen = 12;
1767 size_t vlen = strlen(ct);
1768 if (vlen > 0 && hbuf_off + klen + 2 + vlen <= sizeof(hbuf)) {
1769 memcpy(hbuf + hbuf_off, "content-type", klen);
1770 memcpy(hbuf + hbuf_off + klen + 2, ct, vlen);
1771 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0, klen,
1772 klen + 2, vlen);
1773 hbuf_off += klen + 2 + vlen;
1774 hdr_count++;
1775 }
1776 }
1777 }
1778
1779 /* user headers (skip content-length/content-type already handled) */
1780 cwist_http_header_node *node = st->res->headers;
1781 while (node && hdr_count < H3_MAX_RESPONSE_HEADERS) {
1782 if (node->key && node->key->data && node->key->size > 0 && node->value &&
1783 node->value->data) {
1784 char h3_name[256];
1785 if (cwist_http3_normalize_response_header_name(node->key->data, h3_name,
1786 sizeof(h3_name)) != 0 ||
1787 h3_name[0] == '\0' || strlen(node->value->data) != node->value->size ||
1788 !cwist_http3_response_header_value_is_safe(node->value->data)) {
1789 node = node->next;
1790 continue;
1791 }
1792
1793 if (strcmp(h3_name, "content-length") == 0 ||
1794 strcmp(h3_name, "content-type") == 0) {
1795 node = node->next;
1796 continue;
1797 }
1798 /* RFC 9114 section 4.2: connection-specific fields are malformed in
1799 * HTTP/3; te is only allowed with the value "trailers". */
1800 if (strcmp(h3_name, "connection") == 0 || strcmp(h3_name, "keep-alive") == 0 ||
1801 strcmp(h3_name, "proxy-connection") == 0 ||
1802 strcmp(h3_name, "transfer-encoding") == 0 || strcmp(h3_name, "upgrade") == 0 ||
1803 (strcmp(h3_name, "te") == 0 &&
1804 strcasecmp(node->value->data, "trailers") != 0)) {
1805 node = node->next;
1806 continue;
1807 }
1808
1809 size_t klen = strlen(h3_name);
1810 size_t vlen = node->value->size;
1811 if (klen > 0 && hbuf_off + klen + 2 + vlen <= sizeof(hbuf)) {
1812 memcpy(hbuf + hbuf_off, h3_name, klen);
1813 memcpy(hbuf + hbuf_off + klen + 2, node->value->data, vlen);
1814 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0, klen,
1815 klen + 2, vlen);
1816 hbuf_off += klen + 2 + vlen;
1817 hdr_count++;
1818 }
1819 }
1820 node = node->next;
1821 }
1822
1823 if (status_code == CWIST_HTTP_OK && !bodyless && !is_head && body_len == 0) {
1824 /* Legal per RFC 9110, but a 200 with no body and no
1825 * content-length usually means a handler forgot to assign one;
1826 * make it diagnosable in the logs. */
1827 CWIST_LOG_WARN("[HTTP/3] handler produced 200 with empty body");
1828 }
1829
1830 size_t initial_body_len = 0;
1831 if (st->res->use_file_stream && st->res->file_stream_fd >= 0) {
1832 initial_body_len = st->res->file_stream_len;
1833 } else if (st->res->is_ptr_body && st->res->ptr_body) {
1834 initial_body_len = st->res->ptr_body_len;
1835 } else if (st->res->body) {
1836 initial_body_len = st->res->body->size;
1837 }
1838 int eos = (bodyless || is_head || initial_body_len == 0) ? 1 : 0;
1839
1840 lsquic_http_headers_t headers = {
1841 .count = (unsigned)hdr_count,
1842 .headers = headers_arr,
1843 };
1844 if (lsquic_stream_send_headers(stream, &headers, eos) != 0) {
1845 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1846 lsquic_stream_wantwrite(stream, 1);
1847 return;
1848 }
1849 lsquic_stream_close(stream);
1850 return;
1851 }
1852 if (eos) {
1853 /* NB: the eos argument of lsquic_stream_send_headers is ignored
1854 * for IETF QUIC, so an empty body must be finished with an
1855 * explicit shutdown(1) here. Relying on eos leaves every
1856 * empty-body response (204s, 304s, HEAD, error statuses) without
1857 * FIN, which browsers surface as a protocol error — timing- and
1858 * RTT-dependent because of header-block stashing under low cwnd. */
1859 lsquic_stream_shutdown(stream, 1);
1860 st->write_state = 2;
1861 lsquic_stream_wantread(stream, 1);
1862 return;
1863 }
1864 st->write_state = 1;
1865 lsquic_stream_wantwrite(stream, 1);
1866 return;
1867 }
1868
1869 if (st->write_state == 1 && st->res) {
1870 bool bodyless = h3_status_forbids_body(st->res->status_code);
1871 bool is_head = st->req && st->req->method == CWIST_HTTP_HEAD;
1872 size_t body_len = 0;
1873 const char *body_data = NULL;
1874
1875 if (st->res->use_file_stream) {
1876 /* Fill the congestion window on each callback: loop chunk writes
1877 * until the stream says EAGAIN instead of one 64 KiB chunk per
1878 * tick - but cap how many synchronous pread()s a single callback
1879 * does. This thread also drives lsquic_engine_process_conns()
1880 * and packet I/O for every other connection; with an unbounded
1881 * loop, one big response on a wide-open congestion window turns
1882 * into dozens of blocking pread()s back to back (worse under
1883 * disk contention/page faults than the SSD-idle case), starving
1884 * every other connection's ACKs/packets for that whole stretch.
1885 * Re-arming wantwrite and returning after H3_FILE_CHUNKS_PER_TICK
1886 * chunks gives the event loop a chance to service other
1887 * connections between bursts; lsquic calls this back again on
1888 * the next tick to keep going from where body_sent left off. */
1889#define H3_FILE_CHUNKS_PER_TICK 4
1890 int chunks_this_tick = 0;
1891 while (st->res->file_stream_fd >= 0 && st->body_sent < st->res->file_stream_len) {
1892 if (chunks_this_tick >= H3_FILE_CHUNKS_PER_TICK) {
1893 lsquic_stream_wantwrite(stream, 1);
1894 return;
1895 }
1896 static __thread char file_buf[65536];
1897 off_t offset = st->res->file_stream_offset + (off_t)st->body_sent;
1898 size_t to_read = st->res->file_stream_len - st->body_sent;
1899 if (to_read > sizeof(file_buf)) to_read = sizeof(file_buf);
1900 ssize_t nr = pread(st->res->file_stream_fd, file_buf, to_read, offset);
1901 if (nr > 0) {
1902 chunks_this_tick++;
1903 ssize_t nw = lsquic_stream_write(stream, file_buf, (size_t)nr);
1904 if (nw < 0) {
1905 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1906 lsquic_stream_wantwrite(stream, 1);
1907 return;
1908 }
1909 lsquic_stream_close(stream);
1910 return;
1911 }
1912 if (nw == 0) {
1913 lsquic_stream_wantwrite(stream, 1);
1914 return;
1915 }
1916 st->send_xor ^= h3_xor_bytes((const unsigned char *)file_buf, (size_t)nw);
1917 st->body_sent += (size_t)nw;
1918 if (nw < nr) {
1919 /* Stream buffer full for now; come back when writable. */
1920 lsquic_stream_wantwrite(stream, 1);
1921 return;
1922 }
1923 continue;
1924 } else if (nr == 0) {
1925 /* Short EOF (file truncated after fstat): the announced
1926 * content-length can no longer be honoured, and finishing
1927 * anyway would be a malformed-message error at the
1928 * client. Reset the stream instead. */
1929 lsquic_stream_close(stream);
1930 return;
1931 } else {
1932 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1933 /* Retry next tick */
1934 lsquic_stream_wantwrite(stream, 1);
1935 return;
1936 }
1937 lsquic_stream_close(stream);
1938 return;
1939 }
1940 }
1941#undef H3_FILE_CHUNKS_PER_TICK
1942 if (st->res->file_stream_fd >= 0) body_len = st->res->file_stream_len;
1943 } else if (st->res->is_ptr_body) {
1944 if (!st->res->ptr_body && st->res->ptr_body_len > 0) {
1945 /* Announced a body we cannot produce; spinning here with
1946 * wantwrite armed would hang the stream forever. */
1947 lsquic_stream_close(stream);
1948 return;
1949 }
1950 body_len = st->res->ptr_body_len;
1951 body_data = (const char *)st->res->ptr_body;
1952 } else if (st->res->body) {
1953 body_len = st->res->body->size;
1954 body_data = st->res->body->data;
1955 }
1956
1957 /* Skip payload writing entirely if body is empty, forbidden, or already fully sent */
1958 if (bodyless || is_head || body_len == 0 || st->body_sent >= body_len) {
1959 st->write_state = 2;
1960 lsquic_stream_shutdown(stream, 1);
1961 lsquic_stream_wantread(stream, 1);
1962 return;
1963 }
1964
1965 while (body_data && body_len > 0 && st->body_sent < body_len) {
1966 ssize_t n =
1967 lsquic_stream_write(stream, body_data + st->body_sent, body_len - st->body_sent);
1968 if (n > 0) {
1969 st->send_xor ^=
1970 h3_xor_bytes((const unsigned char *)(body_data + st->body_sent), (size_t)n);
1971 st->body_sent += (size_t)n;
1972 if (st->body_sent < body_len) {
1973 lsquic_stream_wantwrite(stream, 1);
1974 }
1975 continue;
1976 }
1977 if (n < 0) {
1978 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1979 lsquic_stream_wantwrite(stream, 1);
1980 return;
1981 }
1982 lsquic_stream_close(stream);
1983 return;
1984 }
1985 /* n == 0: stream buffer full for now; re-arm wantwrite and come back */
1986 lsquic_stream_wantwrite(stream, 1);
1987 return;
1988 }
1989
1990 if (st->body_sent >= body_len) {
1991 st->write_state = 2;
1992 lsquic_stream_shutdown(stream, 1);
1993 lsquic_stream_wantread(stream, 1);
1994 } else {
1995 lsquic_stream_wantwrite(stream, 1);
1996 }
1997 }

2269/* --- Shared session ticket key -------------------------------------------
2270 * Session resumption (and therefore 0-RTT early data) requires the server to
2271 * issue resumable tickets. Mirrors the prefork-safe pattern used by the
2272 * HTTPS stack (src/net/http/https.c): one random key generated at context
2273 * creation and installed via an explicit ticket-key callback, so tickets
2274 * issued by one worker resume on any other (SO_REUSEPORT scatters
2275 * connections across workers). The key lives for the process lifetime.
2276 * ------------------------------------------------------------------------- */
2277typedef struct cwist_h3_ticket_key {
2278 unsigned char name[16]; /* key_name sent in the ticket */
2279 unsigned char aes_key[32]; /* AES-256-CBC encryption key */
2280 unsigned char hmac_key[32]; /* HMAC-SHA-256 MAC key */
2281} cwist_h3_ticket_key;

2304static int cwist_h3_ticket_key_cb(SSL *ssl, uint8_t *key_name, uint8_t *iv, EVP_CIPHER_CTX *ectx,
2305 HMAC_CTX *hctx, int encrypt) {
2306 SSL_CTX *ssl_ctx = ssl ? SSL_get_SSL_CTX(ssl) : NULL;
2307 const cwist_h3_ticket_key *key = NULL;
2308 if (ssl_ctx && g_h3_ticket_key_ex_data_idx >= 0) {
2309 key =
2310 (const cwist_h3_ticket_key *)SSL_CTX_get_ex_data(ssl_ctx, g_h3_ticket_key_ex_data_idx);
2311 }
2312 if (!key) return 0;
2313
2314 if (encrypt) {
2315 memcpy(key_name, key->name, sizeof(key->name));
2316 if (RAND_bytes(iv, 16) != 1) return 0;
2317 if (EVP_EncryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
2318 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1)
2319 return 0;
2320 return 1;
2321 }
2322
2323 if (memcmp(key_name, key->name, sizeof(key->name)) != 0) {
2324 return 0; /* Unknown key: decline the ticket, do a full handshake. */
2325 }
2326 if (EVP_DecryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
2327 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1) return 0;
2328 return 1;
2329}

2361static void cwist_h3_free_session_ticket_key(SSL_CTX *ssl_ctx) {

2364 * cwist_h3_ticket_key_ex_data_init() via the pthread_once below -

2366 * (line ~2196), which can run on a thread that never called
2367 * cwist_h3_setup_session_tickets()/this same pthread_once itself (e.g.

2373 * this gate before reading the index. */

2377 if (key) {

3566int cwist_http3_send_datagram(void *conn, const void *data, size_t len) {
3567 lsquic_conn_t *c = (lsquic_conn_t *)conn;
3568 if (!c || !data || len == 0) return -1;
3569
3570 h3_conn_ctx_t *cc = (h3_conn_ctx_t *)lsquic_conn_get_ctx(c);
3571 if (!cc) return -1;
3572
3573 h3_dgram_node_t *node = (h3_dgram_node_t *)malloc(sizeof(*node));
3574 if (!node) return -1;
3575 node->data = malloc(len);
3576 if (!node->data) {
3577 free(node);
3578 return -1;
3579 }
3580 memcpy(node->data, data, len);
3581 node->len = len;
3582 node->next = NULL;
3583
3584 /* Append to this connection's own queue - concurrent sends on other
3585 * connections touch their own h3_conn_ctx_t and never see this lock,
3586 * and concurrent sends on *this* connection now queue instead of one
3587 * clobbering (free()ing) the other's buffer. */
3588 pthread_mutex_lock(&cc->dgram_lock);
3589 if (cc->dgram_tail) {
3590 cc->dgram_tail->next = node;
3591 } else {
3592 cc->dgram_head = node;
3593 }
3594 cc->dgram_tail = node;
3595 pthread_mutex_unlock(&cc->dgram_lock);
3596
3597 lsquic_conn_want_datagram_write(c, 1);
3598 return 0;
3599}
