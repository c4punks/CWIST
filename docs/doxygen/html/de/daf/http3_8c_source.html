184 * services. Previously every connection shared the single, unlocked
185 * cwist_http3_context pointer directly, which put datagram state
186 * (g_h3_dgram) in a process-wide global: one connection's pending
187 * datagram could be freed/overwritten by a concurrent send on a

264static void cwist_wt_handle_free(cwist_wt_handle_t *handle) {
265 if (!handle) return;
266 while (handle->first_child) {
267 cwist_wt_handle_t *next = handle->first_child;
268 handle->first_child = next->next_sibling;
269 next->magic = 0;
270 next->ptr = NULL;
271 next->parent = NULL;
272 next->first_child = NULL;
273 next->next_sibling = NULL;
274 free(next);
275 }
276 if (handle->parent) {
277 cwist_wt_handle_t **link = &handle->parent->first_child;
278 while (*link) {
279 if (*link == handle) {
280 *link = handle->next_sibling;
281 break;
282 }
283 link = &(*link)->next_sibling;
284 }
285 }
286 handle->magic = 0;
287 handle->ptr = NULL;
288 handle->parent = NULL;
289 handle->first_child = NULL;
290 handle->next_sibling = NULL;
291 free(handle);
292}

351/* Browsers routinely send more than 64 HTTP/3 headers once Client Hints,
352 * security metadata and cookies are included. Keep a firm per-stream cap,
353 * but leave enough headroom that a late Cookie or :path is never discarded. */
354#define H3_MAX_HEADERS 256
355#define H3_DECODE_BUF_SIZE 131072
356#define H3_MAX_RESPONSE_HEADERS 256
357#define H3_RESPONSE_HEADER_BUF_SIZE 32768

359typedef struct cwist_h3_hset {
360 lsquic_stream_t *stream;
361 struct cwist_h3_hset *next; /* intrusive list of live hsets per ctx */
362 struct cwist_h3_hset **prev; /* link to the pointer that points at us */
363 cwist_http3_context *owner; /* tracking context, NULL if untracked */
364 struct lsxpack_header headers[H3_MAX_HEADERS];
365 size_t count;
366 char decode_buf[H3_DECODE_BUF_SIZE];
367 size_t decode_off;
368} cwist_h3_hset_t;

377static void cwist_h3_hset_track(cwist_http3_context *ctx, cwist_h3_hset_t *hset) {
378 if (!ctx || !hset) return;
379 hset->owner = ctx;
380 hset->next = (cwist_h3_hset_t *)ctx->hsets;
381 hset->prev = (cwist_h3_hset_t **)&ctx->hsets;
382 if (hset->next) hset->next->prev = &hset->next;
383 ctx->hsets = hset;
384}
385
390static void cwist_h3_hset_untrack(cwist_h3_hset_t *hset) {
391 if (!hset || !hset->owner) return;
392 *hset->prev = hset->next;
393 if (hset->next) hset->next->prev = hset->prev;
394 hset->next = NULL;
395 hset->prev = NULL;
396 hset->owner = NULL;
397}

445static struct lsxpack_header *cwist_h3_hsi_prepare(void *hset_p, struct lsxpack_header *xhdr,
446 size_t req_space) {
447 cwist_h3_hset_t *hset = hset_p;
448 if (!hset) return NULL;
449
450 if (xhdr) {
451 if (req_space > LSXPACK_MAX_STRLEN || xhdr->name_offset < 0 ||
452 (size_t)xhdr->name_offset >= sizeof(hset->decode_buf) ||
453 req_space > sizeof(hset->decode_buf) - (size_t)xhdr->name_offset) {
454 fprintf(stderr,
455 "[HTTP/3] Rejecting oversized QPACK header resize (space=%zu, offset=%d)\n",
456 req_space, (int)xhdr->name_offset);
457 return NULL;
458 }
459 xhdr->val_len = (lsxpack_strlen_t)req_space;
460 return xhdr;
461 }
462
463 if (hset->count >= H3_MAX_HEADERS) return NULL;
464 if (req_space > sizeof(hset->decode_buf) - hset->decode_off) return NULL;
465 lsxpack_header_prepare_decode(&hset->headers[hset->count], hset->decode_buf, hset->decode_off,
466 sizeof(hset->decode_buf) - hset->decode_off);
467 return &hset->headers[hset->count];
468}

478static int cwist_h3_hsi_process_header(void *hset_p, struct lsxpack_header *xhdr) {
479 cwist_h3_hset_t *hset = hset_p;
480 /* A NULL header marks the end of a header block. */
481 if (!hset || !xhdr) return 0;
482
483 /* The QPACK decoder exposes the exact storage used by this completed
484 * header. */
485 size_t total = lsxpack_header_get_dec_size(xhdr);
486 if (total > sizeof(hset->decode_buf) - hset->decode_off) {
487 fprintf(stderr, "[HTTP/3] Rejecting oversized QPACK header (size=%zu, used=%zu)\n", total,
488 hset->decode_off);
489 return -1;
490 }
491 hset->decode_off += total;
492 hset->count++;
493 return 0;
494}

564static void h3_setup_cmsg(struct msghdr *msg, char *cbuf, size_t cbuf_sz,
565 const struct lsquic_out_spec *spec, uint16_t gso_seg) {
566 msg->msg_control = cbuf;
567 msg->msg_controllen = cbuf_sz;
568 memset(cbuf, 0, cbuf_sz);
569 size_t ctl_len = 0;
570
571#if defined(__linux__) && defined(UDP_SEGMENT)
572 if (gso_seg > 0) {
573 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
574 cmsg->cmsg_level = SOL_UDP;
575 cmsg->cmsg_type = UDP_SEGMENT;
576 cmsg->cmsg_len = CMSG_LEN(sizeof(uint16_t));
577 memcpy(CMSG_DATA(cmsg), &gso_seg, sizeof(uint16_t));
578 ctl_len += CMSG_SPACE(sizeof(uint16_t));
579 }
580#else
581 (void)gso_seg;
582#endif
583
584 if (spec->local_sa && spec->dest_sa) {
585 if (spec->dest_sa->sa_family == AF_INET && spec->local_sa->sa_family == AF_INET) {
586 struct in_addr addr = ((const struct sockaddr_in *)spec->local_sa)->sin_addr;
587 if (addr.s_addr != INADDR_ANY) {
588#if defined(__linux__) && defined(IP_PKTINFO)
589 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
590 cmsg->cmsg_level = IPPROTO_IP;
591 cmsg->cmsg_type = IP_PKTINFO;
592 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in_pktinfo));
593 struct in_pktinfo info = {0};
594 info.ipi_spec_dst = addr;
595 memcpy(CMSG_DATA(cmsg), &info, sizeof(info));
596 ctl_len += CMSG_SPACE(sizeof(struct in_pktinfo));
597#elif defined(IP_SENDSRCADDR)
598 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
599 cmsg->cmsg_level = IPPROTO_IP;
600 cmsg->cmsg_type = IP_SENDSRCADDR;
601 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in_addr));
602 memcpy(CMSG_DATA(cmsg), &addr, sizeof(addr));
603 ctl_len += CMSG_SPACE(sizeof(struct in_addr));
604#endif
605 }
606 } else if (spec->dest_sa->sa_family == AF_INET6 && spec->local_sa->sa_family == AF_INET6) {
607 const struct in6_addr *addr6 =
608 &((const struct sockaddr_in6 *)spec->local_sa)->sin6_addr;
609 if (memcmp(addr6, &in6addr_any, sizeof(struct in6_addr)) != 0) {
610#if defined(IPV6_PKTINFO)
611 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
612 cmsg->cmsg_level = IPPROTO_IPV6;
613 cmsg->cmsg_type = IPV6_PKTINFO;
614 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in6_pktinfo));
615 struct in6_pktinfo info6 = {0};
616 info6.ipi6_addr = *addr6;
617 memcpy(CMSG_DATA(cmsg), &info6, sizeof(info6));
618 ctl_len += CMSG_SPACE(sizeof(struct in6_pktinfo));
619#endif
620 }
621 }
622 }
623
624#if defined(CWIST_H3_HAVE_ECN_CMSG)
625 if (spec->ecn && spec->dest_sa) {
626 if (spec->dest_sa->sa_family == AF_INET) {
627 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
628 cmsg->cmsg_level = IPPROTO_IP;
629 cmsg->cmsg_type = IP_TOS;
630 int tos = spec->ecn;
631 cmsg->cmsg_len = CMSG_LEN(sizeof(tos));
632 memcpy(CMSG_DATA(cmsg), &tos, sizeof(tos));
633 ctl_len += CMSG_SPACE(sizeof(tos));
634 }
635#if defined(IPV6_TCLASS)
636 else if (spec->dest_sa->sa_family == AF_INET6) {
637 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
638 cmsg->cmsg_level = IPPROTO_IPV6;
639 cmsg->cmsg_type = IPV6_TCLASS;
640 int tos = spec->ecn;
641 cmsg->cmsg_len = CMSG_LEN(sizeof(tos));
642 memcpy(CMSG_DATA(cmsg), &tos, sizeof(tos));
643 ctl_len += CMSG_SPACE(sizeof(tos));
644 }
645#endif
646 }
647#endif
648
649 msg->msg_controllen = ctl_len;
650 if (ctl_len == 0) {
651 msg->msg_control = NULL;
652 }
653}

698static bool h3_same_dest(const struct lsquic_out_spec *a, const struct lsquic_out_spec *b) {
699 if (!a->dest_sa || !b->dest_sa) return false;
700 if (a->dest_sa->sa_family != b->dest_sa->sa_family) return false;
701 if (a->local_sa != b->local_sa) {
702 if (!a->local_sa || !b->local_sa) return false;
703 if (a->local_sa->sa_family != b->local_sa->sa_family) return false;
704 if (a->local_sa->sa_family == AF_INET) {
705 const struct sockaddr_in *x = (const struct sockaddr_in *)a->local_sa;
706 const struct sockaddr_in *y = (const struct sockaddr_in *)b->local_sa;

748static int h3_sendmmsg_batch(int udp_fd, const struct lsquic_out_spec *specs, unsigned i,
749 unsigned n_specs) {
750 struct mmsghdr msgs[64];

752

754 unsigned batch = n_specs - i;
755 if (batch > 64) batch = 64;

758 const struct lsquic_out_spec *spec = &specs[i + k];
759 struct msghdr *msg = &msgs[k].msg_hdr;
760 memset(msg, 0, sizeof(*msg));
761 msg->msg_name = (void *)spec->dest_sa;
762 msg->msg_namelen = (spec->dest_sa && spec->dest_sa->sa_family == AF_INET)
763 ? sizeof(struct sockaddr_in)
764 : sizeof(struct sockaddr_in6);
765 msg->msg_iov = (struct iovec *)spec->iov;
766 msg->msg_iovlen = spec->iovlen;
767 h3_setup_cmsg(msg, ctrl_bufs[k], sizeof(ctrl_bufs[k]), spec, 0);
768 msgs[k].msg_len = 0;
769 }
770

772 if (res > 0) {

777 if (res < 0) {
778 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
779 if (i > 0) return (int)i;

781 if (nw >= 0) {
782 i++;
783 continue;
784 }
785 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
786 return -1;
787 }
788 break;
789 }
790 return (int)i;

809static int cwist_h3_packets_out(void *ctx, const struct lsquic_out_spec *specs, unsigned n_specs) {
810 int udp_fd = *(int *)ctx;
811#if defined(__linux__)
812 /* One-time GSO probe: check env var on first call. */
813 if (__builtin_expect(h3_gso_state < 0, 0)) {
814 const char *no_gso = getenv("CWIST_H3_NO_GSO");
815 h3_gso_state = (no_gso && no_gso[0] == '1') ? 1 : 0;
816 }
817
818 if (!h3_gso_state) {
819 /* GSO fast path: coalesce runs of equal-size datagrams to the same
820 * peer into one sendmsg with UDP_SEGMENT. */
821 unsigned i = 0;
822 while (i < n_specs) {
823 size_t seg = h3_spec_len(&specs[i]);
824 unsigned run = 1;
825 unsigned niov = specs[i].iovlen;
826 if (seg > 0 && seg <= 65535) {
827 /* Total payload must fit in one UDP datagram (65535 bytes);
828 * 65536/seg could allow an exactly-64KiB super-packet whose
829 * sendmsg fails with EMSGSIZE and would disable GSO. */
830 unsigned max_segs = (unsigned)(65535 / seg);
831 if (max_segs > 48) max_segs = 48;
832 while (i + run < n_specs && run < max_segs && specs[i + run].ecn == specs[i].ecn &&
833 h3_same_dest(&specs[i], &specs[i + run]) &&
834 h3_spec_len(&specs[i + run]) == seg && niov + specs[i + run].iovlen <= 128) {
835 niov += specs[i + run].iovlen;
836 run++;
837 }
838 }
839 if (run >= 2) {
840 struct iovec iov[128];
841 unsigned n = 0;
842 for (unsigned k = 0; k < run; k++)
843 for (unsigned m = 0; m < specs[i + k].iovlen; m++)
844 iov[n++] = specs[i + k].iov[m];
845
846 char ctrl[512];
847 struct msghdr msg = {0};
848 msg.msg_name = (void *)specs[i].dest_sa;
849 msg.msg_namelen = (specs[i].dest_sa && specs[i].dest_sa->sa_family == AF_INET)
850 ? sizeof(struct sockaddr_in)
851 : sizeof(struct sockaddr_in6);
852 msg.msg_iov = iov;
853 msg.msg_iovlen = n;
854 h3_setup_cmsg(&msg, ctrl, sizeof(ctrl), &specs[i], (uint16_t)seg);
855
856 ssize_t nw = sendmsg(udp_fd, &msg, MSG_DONTWAIT);
857 if (nw >= 0) {
858 i += run;
859 continue;
860 }
861 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
862 /* GSO rejected: permanently disable and fall back to sendmmsg. */
863 h3_gso_state = 1;
864 return h3_sendmmsg_batch(udp_fd, specs, i, n_specs);
865 }
866 /* Single packet: send directly. */
867 int nw = h3_send_one(udp_fd, &specs[i]);
868 if (nw < 0) {
869 if (errno == EAGAIN || errno == EWOULDBLOCK) return (int)i;
870 if (errno == ECONNREFUSED || errno == ENETUNREACH || errno == EHOSTUNREACH ||
871 errno == EMSGSIZE) {
872 i++;
873 continue;
874 }
875 return (i > 0) ? (int)i : -1;
876 }
877 i++;
878 }
879 return (int)i;
880 }
881
882 /* sendmmsg fallback path (GSO disabled). */
883 return h3_sendmmsg_batch(udp_fd, specs, 0, n_specs);
884#else
885 unsigned i = 0;
886 for (; i < n_specs; ++i) {
887 int nw = h3_send_one(udp_fd, &specs[i]);
888 if (nw < 0) {
889 if (errno == EAGAIN || errno == EWOULDBLOCK) return (int)i;
890 if (errno == ECONNREFUSED || errno == ENETUNREACH || errno == EHOSTUNREACH ||
891 errno == EMSGSIZE)
892 continue;
893 return (i > 0) ? (int)i : -1;
894 }
895 }
896 return (int)i;
897#endif
898}

1219static int h3_seq_append_and_feed(h3_stream_ctx_t *st, const unsigned char *data, size_t len) {
1220 if (len > SIZE_MAX - st->seq_len) return -1;
1221 size_t need = st->seq_len + len;
1222 if (need > st->seq_cap) {
1223 size_t cap = st->seq_cap ? st->seq_cap : 4096;
1224 while (cap < need) {
1225 if (cap > (CWIST_SEQ_HEADER_SIZE + UINT16_MAX) / 2) {
1226 cap = CWIST_SEQ_HEADER_SIZE + UINT16_MAX;
1227 break;
1228 }
1229 cap *= 2;
1230 }
1231 if (cap < need || cap > CWIST_SEQ_HEADER_SIZE + UINT16_MAX) return -1;
1232 unsigned char *tmp = realloc(st->seq_buf, cap);
1233 if (!tmp) return -1;
1234 st->seq_buf = tmp;
1235 st->seq_cap = cap;
1236 }
1237 memcpy(st->seq_buf + st->seq_len, data, len);
1238 st->seq_len += len;
1239
1240 while (st->seq_len >= CWIST_SEQ_HEADER_SIZE) {
1241 size_t payload_len = ((size_t)st->seq_buf[4] << 8) | st->seq_buf[5];
1242 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + payload_len;
1243 if (payload_len == 0 || st->seq_len < chunk_len) break;
1244 cwist_seq_chunk_t chunk;
1245 if (!cwist_seq_chunk_parse(st->seq_buf, chunk_len, &chunk)) return -1;
1246 if (!st->body_assembler) {
1247 st->body_assembler = cwist_seq_assembler_create_limited(CWIST_HTTP_MAX_BODY_SIZE);
1248 if (!st->body_assembler) return -1;
1249 }
1250 if (!cwist_seq_assembler_feed(st->body_assembler, &chunk)) return -1;
1251 memmove(st->seq_buf, st->seq_buf + chunk_len, st->seq_len - chunk_len);
1252 st->seq_len -= chunk_len;
1253 }
1254 return 0;
1255}

1288static bool h3_process_stream_headers(lsquic_stream_t *stream, h3_stream_ctx_t *st) {
1289 if (st->headers_done) return true;
1290 void *hset = lsquic_stream_get_hset(stream);
1291 if (!hset) return false;
1292 cwist_h3_hset_t *hs = (cwist_h3_hset_t *)hset;
1293 /* lsxpack exposes counted slices, not C strings, so each header needs a
1294 * NUL-terminated scratch copy to hand to strcmp()/h3_apply_header().
1295 * These used to be a malloc(name_len+1)/malloc(value_len+1) pair freed
1296 * at the bottom of every loop iteration - up to two heap round trips
1297 * per header, tens of them on a real request. The bounds below
1298 * (name_len <= 1024, value_len <= H3_DECODE_BUF_SIZE - 1) are already
1299 * enforced before anything touches these buffers, so a single
1300 * reusable pair sized to those same bounds, declared once outside the
1301 * loop, replaces all of that: h3_apply_header() copies out of them
1302 * before the next iteration overwrites them (see below), so reuse is
1303 * safe. */
1304 char name_scratch[1024 + 1];
1305 char value_scratch[H3_DECODE_BUF_SIZE];
1306 for (size_t i = 0; i < hs->count; ++i) {
1307 const struct lsxpack_header *xhdr = &hs->headers[i];
1308 const char *raw_name = lsxpack_header_get_name(xhdr);
1309 const char *raw_value = lsxpack_header_get_value(xhdr);
1310 size_t name_len = xhdr->name_len;
1311 size_t value_len = xhdr->val_len;
1312 if (raw_name && raw_value && name_len > 0 && name_len <= 1024 &&
1313 value_len <= H3_DECODE_BUF_SIZE - 1) {
1314 char *name = name_scratch;
1315 char *value = value_scratch;
1316 memcpy(name, raw_name, name_len);
1317 name[name_len] = '\0';
1318 memcpy(value, raw_value, value_len);
1319 value[value_len] = '\0';
1320 if (name[0] == ':') {
1321 /* RFC 9114 Section 4.3.1: pseudo-headers precede regular
1322 * headers, appear at most once, and only the defined set is
1323 * valid in requests. */
1324 unsigned bit = 0;
1325 if (strcmp(name, ":method") == 0)
1326 bit = H3_PSEUDO_METHOD;
1327 else if (strcmp(name, ":scheme") == 0)
1328 bit = H3_PSEUDO_SCHEME;
1329 else if (strcmp(name, ":path") == 0)
1330 bit = H3_PSEUDO_PATH;
1331 else if (strcmp(name, ":authority") == 0)
1332 bit = H3_PSEUDO_AUTHORITY;
1333 else if (strcmp(name, ":protocol") == 0)
1334 bit = H3_PSEUDO_PROTOCOL;
1335 if (bit == 0 || st->seen_regular_header || (st->pseudo_seen & bit)) {
1336 h3_reject_malformed_request(stream, st);
1337 return false;
1338 }
1339 st->pseudo_seen |= bit;
1340 if (bit == H3_PSEUDO_METHOD && strcmp(value, "CONNECT") == 0) {
1341 st->is_connect = 1;
1342 }
1343 if (bit == H3_PSEUDO_PATH && value_len == 0) {
1344 st->saw_empty_path = 1;
1345 }
1346 } else {
1347 st->seen_regular_header = 1;
1348 }
1349 h3_apply_header(st->req, name, value);
1350#ifdef CWIST_WEBTRANSPORT_SERVER
1351 /* :protocol may precede :method; is_connect is checked once the
1352 * whole block is in (below). */
1353 if (strcmp(name, ":protocol") == 0 && strcmp(value, "webtransport") == 0) {
1354 st->is_webtransport = 1;
1355 }
1356#endif
1357 }
1358 }
1359 /* RFC 9114 Section 4.3.1 completeness rules, evaluated once the whole
1360 * header block has been seen. */
1361 if (st->is_connect && !(st->pseudo_seen & H3_PSEUDO_PROTOCOL)) {
1362 /* Plain CONNECT: :scheme and :path MUST be omitted, :authority is
1363 * required. */
1364 if ((st->pseudo_seen & (H3_PSEUDO_SCHEME | H3_PSEUDO_PATH)) ||
1365 !(st->pseudo_seen & H3_PSEUDO_AUTHORITY)) {
1366 h3_reject_malformed_request(stream, st);
1367 return false;
1368 }
1369 } else if (st->is_connect) {
1370 /* Extended CONNECT (RFC 9220, e.g. WebTransport): :scheme, :path and
1371 * :authority are all required. */
1372 if (!(st->pseudo_seen & H3_PSEUDO_SCHEME) || !(st->pseudo_seen & H3_PSEUDO_PATH) ||
1373 !(st->pseudo_seen & H3_PSEUDO_AUTHORITY)) {
1374 h3_reject_malformed_request(stream, st);
1375 return false;
1376 }
1377 } else {
1378 /* Normal request: :method, :scheme and :path are mandatory,
1379 * :authority (or an equivalent Host header) is required for https,
1380 * and :protocol is only legal on extended CONNECT. */
1381 if (!(st->pseudo_seen & H3_PSEUDO_METHOD) || !(st->pseudo_seen & H3_PSEUDO_SCHEME) ||
1382 !(st->pseudo_seen & H3_PSEUDO_PATH) || (st->pseudo_seen & H3_PSEUDO_PROTOCOL) ||
1383 (!(st->pseudo_seen & H3_PSEUDO_AUTHORITY) &&
1384 !cwist_http_header_get(st->req->headers, "host"))) {
1385 h3_reject_malformed_request(stream, st);
1386 return false;
1387 }
1388 /* An empty :path is only legal for OPTIONS (asterisk-form). */
1389 if (st->saw_empty_path && (!st->req || st->req->method != CWIST_HTTP_OPTIONS)) {
1390 h3_reject_malformed_request(stream, st);
1391 return false;
1392 }
1393 }
1394#ifdef CWIST_WEBTRANSPORT_SERVER
1395 if (!st->is_connect) st->is_webtransport = 0;
1396#endif
1397 st->headers_done = 1;
1398 char *seq_header = cwist_http_header_get(st->req->headers, "x-cwist-sequenced-data");
1399 st->sequenced_data =
1400 seq_header && (strcmp(seq_header, "1") == 0 || strcasecmp(seq_header, "true") == 0);
1401 return true;
1402}

1631static void cwist_h3_on_read(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
1632 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
1633 if (!st) return;
1634
1635 unsigned char buf[8192];
1636 ssize_t nread;
1637
1638#ifdef CWIST_WT_STOCK_LSQUIC
1639 if (h3_wt_adopt_stream(stream, st)) {
1640 if (st->wt_stream) h3_wt_stream_on_read(stream, st);
1641 return;
1642 }
1643 if (st->wt_session) {
1644 h3_wt_session_on_read(stream);
1645 return;
1646 }
1647#endif
1648 h3_process_stream_headers(stream, st);
1649 if (st->malformed) return;
1650#ifdef CWIST_WT_STOCK_LSQUIC
1651 /* An extended CONNECT never sends FIN, so the session is opened as soon
1652 * as its header block is in rather than at end of stream. */
1653 if (st->headers_done && st->is_webtransport && !st->wt_dispatched) {
1654 cwist_http3_context *h3_ctx = h3_shared_ctx(lsquic_stream_conn(stream));
1655 if (h3_ctx && h3_ctx->wt_handler) {
1656 h3_wt_open_session(stream, st, h3_ctx);
1657 return;
1658 }
1659 }
1660#endif
1661
1662 while ((nread = lsquic_stream_read(stream, buf, sizeof(buf))) > 0) {
1663 if (st->sequenced_data) {
1664 if (h3_seq_append_and_feed(st, buf, (size_t)nread) != 0) {
1665 lsquic_stream_close(stream);
1666 return;
1667 }
1668 continue;
1669 }
1670 size_t need = st->body_len + (size_t)nread;
1671 if (need > CWIST_HTTP_MAX_BODY_SIZE) {
1672 /* Body too large: answer 413 with a small JSON body instead of a
1673 * bare stream close, so the client sees an explicit status
1674 * (RFC 9110 Section 15.5.14). */
1675 if (!st->res) st->res = cwist_http_response_create();
1676 if (st->res) {
1677 h3_json_error(st->res, 413 /* Payload Too Large */,
1678 "{\"error\":\"payload too large\"}");
1679 st->response_ready = 1;
1680 lsquic_stream_wantread(stream, 0);
1681 lsquic_stream_shutdown(stream, 0);
1682 lsquic_stream_wantwrite(stream, 1);
1683 } else {
1684 lsquic_stream_close(stream);
1685 }
1686 return;
1687 }
1688 if (need > st->body_cap) {
1689 size_t new_cap = st->body_cap ? st->body_cap * 2 : 4096;
1690 while (new_cap < need) new_cap *= 2;
1691 char *tmp = realloc(st->body, new_cap);
1692 if (!tmp) {
1693 lsquic_stream_close(stream);
1694 return;
1695 }
1696 st->body = tmp;
1697 st->body_cap = new_cap;
1698 }
1699 memcpy(st->body + st->body_len, buf, (size_t)nread);
1700 st->recv_xor ^= h3_xor_bytes((const unsigned char *)buf, (size_t)nread);
1701 st->body_len += (size_t)nread;
1702 }
1703
1704 if (nread == 0) {
1705 /* End of stream (FIN received) */
1706 h3_process_stream_headers(stream, st);
1707
1708 if (st->malformed) return;
1709
1710 if (!st->headers_done) {
1711 /* Malformed request: no headers before FIN */
1712 st->res = cwist_http_response_create();
1713 if (st->res) {
1714 h3_json_error(st->res, CWIST_HTTP_BAD_REQUEST, "{\"error\":\"malformed request\"}");
1715 }
1716 st->response_ready = 1;
1717 lsquic_stream_wantread(stream, 0);
1718 lsquic_stream_shutdown(stream, 0);
1719 lsquic_stream_wantwrite(stream, 1);
1720 return;
1721 }
1722
1723 if (st->sequenced_data) {
1724 const uint8_t *assembled = NULL;
1725 size_t assembled_len = 0;
1726 if (st->seq_len != 0 || !st->body_assembler ||
1727 !cwist_seq_assembler_get_data(st->body_assembler, &assembled, &assembled_len)) {
1728 /* Do not dispatch a partial request. */
1729 st->res = cwist_http_response_create();
1730 if (st->res) {
1731 cwist_http_header_add(&st->res->headers, "x-cwist-retry", "1");
1732 h3_json_error(st->res, CWIST_HTTP_BAD_REQUEST,
1733 "{\"error\":\"incomplete sequenced body, retry\"}");
1734 }
1735 st->response_ready = 1;
1736 lsquic_stream_wantread(stream, 0);
1737 lsquic_stream_shutdown(stream, 0);
1738 lsquic_stream_wantwrite(stream, 1);
1739 return;
1740 }
1741 if (assembled_len == 0) {
1742 /* realloc(p, 0) may legally return NULL; do not mistake an
1743 * empty assembled body for an allocation failure (which used
1744 * to reset the stream instead of dispatching the request). */
1745 free(st->body);
1746 st->body = NULL;
1747 st->body_len = 0;
1748 st->body_cap = 0;
1749 } else {
1750 st->body = realloc(st->body, assembled_len);
1751 if (!st->body) {
1752 lsquic_stream_close(stream);
1753 return;
1754 }
1755 memcpy(st->body, assembled, assembled_len);
1756 st->body_len = assembled_len;
1757 st->body_cap = assembled_len;
1758 }
1759 }
1760
1761 if (st->body_len > 0 && st->req && st->req->body) {
1762 cwist_sstring_assign_len(st->req->body, st->body, st->body_len);
1763 }
1764
1765 st->res = cwist_http_response_create();
1766 if (st->res && st->req) {
1767 cwist_http3_context *h3_ctx = h3_shared_ctx(lsquic_stream_conn(stream));
1768#ifdef CWIST_WT_SESSION_API
1769 if (st->is_webtransport && h3_ctx && h3_ctx->wt_handler) {
1770 char *host = cwist_http_header_get(st->req->headers, "host");

1773 .wtci_authority = host,
1774 .wtci_path = st->req->path ? st->req->path->data : NULL,
1775 .wtci_origin = origin,
1776 .wtci_protocol = NULL,
1777 .wtci_draft = lsquic_wt_peer_draft(lsquic_stream_conn(stream)),
1778 };

1780 .wtap_status = LSQUIC_WTAP_STATUS_DEFAULT,
1781 .wtap_wt_if = &cwist_h3_wt_if,
1782 .wtap_wt_if_ctx = st,

1784 .wtap_datagram_send_mode = LSQUIC_HTTP_DG_SEND_DEFAULT,

1789 lsquic_stream_shutdown(stream, 0);
1790 lsquic_stream_wantwrite(stream, 0);

1799 "{\"error\":\"webtransport accept failed\"}");
1800 }

1802 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1803 }
1804 } else
1805#endif

1807 /* 0-RTT replay guard: requests delivered while the QUIC
1808 * handshake is still in progress arrived as early data.
1809 * lsquic (this baseline) exposes no
1810 * lsquic_conn_is_early_data_accepted()-style query, so the
1811 * guard uses the public lsquic_conn_status() instead. */
1812 bool is_early_data = h3_ctx->early_data_enabled && h3_ctx->early_data_guard &&
1813 lsquic_conn_status(lsquic_stream_conn(stream), NULL, 0) ==

1816 /* RFC 8470: refuse replayable non-idempotent early data
1817 * so the client retries after the handshake. Attach a
1818 * body: without one the response goes out as HEADERS+FIN
1819 * with no content-length, which some clients surface as
1820 * an empty (0-byte) reply instead of status 425. */
1821 h3_json_error(st->res, 425 /* Too Early */,
1822 "{\"error\":\"too early, retry after handshake\"}");
1823 } else {
1824 h3_ctx->handler(h3_ctx->user_ctx, st->req, st->res);
1825 }
1826 }
1827 }
1828 st->response_ready = 1;
1829 lsquic_stream_wantread(stream, 0);
1830 lsquic_stream_shutdown(stream, 0);
1831 lsquic_stream_wantwrite(stream, 1);
1832 } else if (nread < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
1833#ifdef CWIST_WT_STOCK_LSQUIC
1834 /* The read that runs lsquic's frame filter over a WebTransport
1835 * stream's signal fails; the stream is tagged by then. */
1836 if (h3_wt_adopt_stream(stream, st)) {
1837 if (st->wt_stream) h3_wt_stream_on_read(stream, st);
1838 return;
1839 }
1840#endif
1841 lsquic_stream_close(stream);
1842 }

1881static void cwist_h3_on_write(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
1882 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
1883 if (!st || !st->response_ready) return;
1884 if (!st->res) {
1885 /* Response allocation failed upstream; reset instead of dereferencing
1886 * NULL (which used to take down every stream on the connection). */
1887 lsquic_stream_close(stream);
1888 return;
1889 }
1890
1891 if (st->write_state == 0) {
1892 struct lsxpack_header headers_arr[H3_MAX_RESPONSE_HEADERS];
1893 char hbuf[H3_RESPONSE_HEADER_BUF_SIZE];
1894 h3_hdr_out hdrs = {
1895 .arr = headers_arr, .max = H3_MAX_RESPONSE_HEADERS, .buf = hbuf, .cap = sizeof(hbuf)};
1896
1897 bool bodyless = h3_status_forbids_body(st->res->status_code);
1898 bool is_head = st->req && st->req->method == CWIST_HTTP_HEAD;
1899
1900 /* Handler attached a body source that cannot produce bytes
1901 * (NULL pointer body / invalid file fd with a non-zero length).
1902 * Sending HEADERS would announce (or imply) a body we cannot
1903 * deliver; close the stream so the client sees an explicit error
1904 * instead of a silent empty 200. */
1905 if (!bodyless && !is_head &&
1906 ((st->res->is_ptr_body && !st->res->ptr_body && st->res->ptr_body_len > 0) ||
1907 (st->res->use_file_stream && st->res->file_stream_fd < 0 &&
1908 st->res->file_stream_len > 0))) {
1909 CWIST_LOG_WARN("[HTTP/3] unusable response body source; resetting stream");
1910 lsquic_stream_close(stream);
1911 return;
1912 }
1913
1914 /* :status */
1915 if (st->res->status_code < 200 || st->res->status_code > 999) {
1916 /* RFC 9110 Section 15: a final response needs a status in
1917 * 200-999 (RFC 9114 Section 4.1.2 requires three digits).
1918 * A handler that leaves the status unset, picks 1xx (an
1919 * interim response, illegal as the complete answer), or an
1920 * out-of-range value gets coerced to 500 instead of putting
1921 * a malformed :status on the wire. */
1922 CWIST_LOG_WARN("[HTTP/3] invalid status %d; coercing to 500", st->res->status_code);
1923 st->res->status_code = CWIST_HTTP_INTERNAL_ERROR;
1924 bodyless = false;
1925 if (!is_head && st->res->body && st->res->body->size == 0 && !st->res->is_ptr_body &&
1926 !st->res->use_file_stream) {
1927 cwist_sstring_assign(st->res->body, "{\"error\":\"internal server error\"}");
1928 }
1929 }
1930 int status_code = st->res->status_code;
1931 char status_str[16];
1932 snprintf(status_str, sizeof(status_str), "%d", status_code);
1933 if (status_str[0] != '\0') h3_hdr_out_add_str(&hdrs, ":status", status_str);
1934
1935 /* content-length */
1936 size_t body_len = 0;
1937 if (st->res->use_file_stream) {
1938 if (st->res->file_stream_fd >= 0) body_len = st->res->file_stream_len;
1939 } else if (st->res->is_ptr_body)
1940 body_len = st->res->ptr_body ? st->res->ptr_body_len : 0;
1941 else if (st->res->body)
1942 body_len = st->res->body->size;
1943
1944 const char *user_cl =
1945 st->res->headers ? cwist_http_header_get(st->res->headers, "content-length") : NULL;
1946
1947 if (!bodyless && (body_len > 0 || is_head) && hdrs.count < hdrs.max) {
1948 /* For HEAD, content-length describes the would-be GET body: keep
1949 * the handler-provided value when present (e.g. static file size
1950 * with an empty body), otherwise compute it like a GET. For all
1951 * other methods the computed body length is authoritative: it is
1952 * exactly what the write path below will send, so trusting a
1953 * divergent handler value would announce bytes never delivered. */
1954 char cl_str[32];
1955 const char *cl_val = NULL;
1956 if (is_head && user_cl) {
1957 if (h3_content_length_is_valid(user_cl)) {
1958 cl_val = user_cl;
1959 } else {
1960 /* Malformed handler content-length must not reach the
1961 * wire; fall back to the computed body length. */
1962 CWIST_LOG_WARN("[HTTP/3] dropping invalid HEAD content-length '%s'", user_cl);
1963 }
1964 }
1965 if (!cl_val) {
1966 snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
1967 cl_val = cl_str;
1968 }
1969 if (cl_val && cl_val[0] != '\0') h3_hdr_out_add_str(&hdrs, "content-length", cl_val);
1970 }
1971
1972 /* content-type (if body is present or HEAD request) */
1973 if (st->res->headers && (body_len > 0 || is_head)) {
1974 char *ct = cwist_http_header_get(st->res->headers, "content-type");
1975 if (ct && ct[0] != '\0') h3_hdr_out_add_str(&hdrs, "content-type", ct);
1976 }
1977
1978 /* user headers (skip content-length/content-type already handled) */
1979 cwist_http_header_node *node = st->res->headers;
1980 while (node && hdrs.count < hdrs.max) {
1981 if (node->key && node->key->data && node->key->size > 0 && node->value &&
1982 node->value->data) {
1983 char h3_name[256];
1984 if (cwist_http3_normalize_response_header_name(node->key->data, h3_name,
1985 sizeof(h3_name)) != 0 ||
1986 h3_name[0] == '\0' || strlen(node->value->data) != node->value->size ||
1987 !cwist_http3_response_header_value_is_safe(node->value->data)) {
1988 node = node->next;
1989 continue;
1990 }
1991
1992 if (strcmp(h3_name, "content-length") == 0 ||
1993 strcmp(h3_name, "content-type") == 0) {
1994 node = node->next;
1995 continue;
1996 }
1997 /* RFC 9114 section 4.2: connection-specific fields are malformed in
1998 * HTTP/3; te is only allowed with the value "trailers". */
1999 if (strcmp(h3_name, "connection") == 0 || strcmp(h3_name, "keep-alive") == 0 ||
2000 strcmp(h3_name, "proxy-connection") == 0 ||
2001 strcmp(h3_name, "transfer-encoding") == 0 || strcmp(h3_name, "upgrade") == 0 ||
2002 (strcmp(h3_name, "te") == 0 &&
2003 strcasecmp(node->value->data, "trailers") != 0)) {
2004 node = node->next;
2005 continue;
2006 }
2007
2008 h3_hdr_out_add(&hdrs, h3_name, strlen(h3_name), node->value->data,
2009 node->value->size);
2010 }
2011 node = node->next;
2012 }
2013
2014 bool empty_ok = false;
2015#ifdef CWIST_WT_STOCK_LSQUIC
2016 empty_ok = st->wt_session != NULL;
2017#endif
2018 if (status_code == CWIST_HTTP_OK && !bodyless && !is_head && body_len == 0 && !empty_ok) {
2019 /* Legal per RFC 9110, but a 200 with no body and no
2020 * content-length usually means a handler forgot to assign one;
2021 * make it diagnosable in the logs. */
2022 CWIST_LOG_WARN("[HTTP/3] handler produced 200 with empty body");
2023 }
2024
2025 size_t initial_body_len = 0;
2026 if (st->res->use_file_stream && st->res->file_stream_fd >= 0) {
2027 initial_body_len = st->res->file_stream_len;
2028 } else if (st->res->is_ptr_body && st->res->ptr_body) {
2029 initial_body_len = st->res->ptr_body_len;
2030 } else if (st->res->body) {
2031 initial_body_len = st->res->body->size;
2032 }
2033 int eos = (bodyless || is_head || initial_body_len == 0) ? 1 : 0;
2034#ifdef CWIST_WT_STOCK_LSQUIC
2035 if (st->wt_session) eos = 0;
2036#endif
2037
2038 lsquic_http_headers_t headers = {
2039 .count = (unsigned)hdrs.count,
2040 .headers = headers_arr,
2041 };
2042 if (lsquic_stream_send_headers(stream, &headers, eos) != 0) {
2043 if (errno == EAGAIN || errno == EWOULDBLOCK) {
2044 lsquic_stream_wantwrite(stream, 1);
2045 return;
2046 }
2047 lsquic_stream_close(stream);
2048 return;
2049 }
2050#ifdef CWIST_WT_STOCK_LSQUIC
2051 if (st->wt_session) {
2052 /* Session accepted: the stream stays open until either side
2053 * closes it (write_state 3). */
2054 st->write_state = 3;
2055 lsquic_stream_wantwrite(stream, 0);
2056 lsquic_stream_flush(stream);
2057 return;
2058 }
2059#endif
2060 if (eos) {
2061 /* NB: the eos argument of lsquic_stream_send_headers is ignored
2062 * for IETF QUIC, so an empty body must be finished with an
2063 * explicit shutdown(1) here. Relying on eos leaves every
2064 * empty-body response (204s, 304s, HEAD, error statuses) without
2065 * FIN, which browsers surface as a protocol error — timing- and
2066 * RTT-dependent because of header-block stashing under low cwnd. */
2067 lsquic_stream_shutdown(stream, 1);
2068 st->write_state = 2;
2069 lsquic_stream_wantread(stream, 1);
2070 return;
2071 }
2072 st->write_state = 1;
2073 lsquic_stream_wantwrite(stream, 1);
2074 return;
2075 }
2076
2077 if (st->write_state == 1 && st->res) {
2078 bool bodyless = h3_status_forbids_body(st->res->status_code);
2079 bool is_head = st->req && st->req->method == CWIST_HTTP_HEAD;
2080 size_t body_len = 0;
2081 const char *body_data = NULL;
2082
2083 if (st->res->use_file_stream) {
2084 /* Fill the congestion window on each callback: loop chunk writes
2085 * until the stream says EAGAIN instead of one 64 KiB chunk per
2086 * tick - but cap how many synchronous pread()s a single callback
2087 * does. This thread also drives lsquic_engine_process_conns()
2088 * and packet I/O for every other connection; with an unbounded
2089 * loop, one big response on a wide-open congestion window turns
2090 * into dozens of blocking pread()s back to back (worse under
2091 * disk contention/page faults than the SSD-idle case), starving
2092 * every other connection's ACKs/packets for that whole stretch.
2093 * Re-arming wantwrite and returning after H3_FILE_CHUNKS_PER_TICK
2094 * chunks gives the event loop a chance to service other
2095 * connections between bursts; lsquic calls this back again on
2096 * the next tick to keep going from where body_sent left off. */
2097#define H3_FILE_CHUNKS_PER_TICK 4
2098 int chunks_this_tick = 0;
2099 while (st->res->file_stream_fd >= 0 && st->body_sent < st->res->file_stream_len) {
2100 if (chunks_this_tick >= H3_FILE_CHUNKS_PER_TICK) {
2101 lsquic_stream_wantwrite(stream, 1);
2102 return;
2103 }
2104 static __thread char file_buf[65536];
2105 off_t offset = st->res->file_stream_offset + (off_t)st->body_sent;
2106 size_t to_read = st->res->file_stream_len - st->body_sent;
2107 if (to_read > sizeof(file_buf)) to_read = sizeof(file_buf);
2108 ssize_t nr = pread(st->res->file_stream_fd, file_buf, to_read, offset);
2109 if (nr > 0) {
2110 chunks_this_tick++;
2111 ssize_t nw = lsquic_stream_write(stream, file_buf, (size_t)nr);
2112 if (nw < 0) {
2113 if (errno == EAGAIN || errno == EWOULDBLOCK) {
2114 lsquic_stream_wantwrite(stream, 1);
2115 return;
2116 }
2117 lsquic_stream_close(stream);
2118 return;
2119 }
2120 if (nw == 0) {
2121 lsquic_stream_wantwrite(stream, 1);
2122 return;
2123 }
2124 st->send_xor ^= h3_xor_bytes((const unsigned char *)file_buf, (size_t)nw);
2125 st->body_sent += (size_t)nw;
2126 if (nw < nr) {
2127 /* Stream buffer full for now; come back when writable. */
2128 lsquic_stream_wantwrite(stream, 1);
2129 return;
2130 }
2131 continue;
2132 } else if (nr == 0) {
2133 /* Short EOF (file truncated after fstat): the announced
2134 * content-length can no longer be honoured, and finishing
2135 * anyway would be a malformed-message error at the
2136 * client. Reset the stream instead. */
2137 lsquic_stream_close(stream);
2138 return;
2139 } else {
2140 if (errno == EAGAIN || errno == EWOULDBLOCK) {
2141 /* Retry next tick */
2142 lsquic_stream_wantwrite(stream, 1);
2143 return;
2144 }
2145 lsquic_stream_close(stream);
2146 return;
2147 }
2148 }
2149#undef H3_FILE_CHUNKS_PER_TICK
2150 if (st->res->file_stream_fd >= 0) body_len = st->res->file_stream_len;
2151 } else if (st->res->is_ptr_body) {
2152 if (!st->res->ptr_body && st->res->ptr_body_len > 0) {
2153 /* Announced a body we cannot produce; spinning here with
2154 * wantwrite armed would hang the stream forever. */
2155 lsquic_stream_close(stream);
2156 return;
2157 }
2158 body_len = st->res->ptr_body_len;
2159 body_data = (const char *)st->res->ptr_body;
2160 } else if (st->res->body) {
2161 body_len = st->res->body->size;
2162 body_data = st->res->body->data;
2163 }
2164
2165 /* Skip payload writing entirely if body is empty, forbidden, or already fully sent */
2166 if (bodyless || is_head || body_len == 0 || st->body_sent >= body_len) {
2167 st->write_state = 2;
2168 lsquic_stream_shutdown(stream, 1);
2169 lsquic_stream_wantread(stream, 1);
2170 return;
2171 }
2172
2173 while (body_data && body_len > 0 && st->body_sent < body_len) {
2174 ssize_t n =
2175 lsquic_stream_write(stream, body_data + st->body_sent, body_len - st->body_sent);
2176 if (n > 0) {
2177 st->send_xor ^=
2178 h3_xor_bytes((const unsigned char *)(body_data + st->body_sent), (size_t)n);
2179 st->body_sent += (size_t)n;
2180 if (st->body_sent < body_len) {
2181 lsquic_stream_wantwrite(stream, 1);
2182 }
2183 continue;
2184 }
2185 if (n < 0) {
2186 if (errno == EAGAIN || errno == EWOULDBLOCK) {
2187 lsquic_stream_wantwrite(stream, 1);
2188 return;
2189 }
2190 lsquic_stream_close(stream);
2191 return;
2192 }
2193 /* n == 0: stream buffer full for now; re-arm wantwrite and come back */
2194 lsquic_stream_wantwrite(stream, 1);
2195 return;
2196 }
2197
2198 if (st->body_sent >= body_len) {
2199 st->write_state = 2;
2200 lsquic_stream_shutdown(stream, 1);
2201 lsquic_stream_wantread(stream, 1);
2202 } else {
2203 lsquic_stream_wantwrite(stream, 1);
2204 }
2205 }

2519static int cwist_h3_ticket_key_cb(SSL *ssl, uint8_t *key_name, uint8_t *iv, EVP_CIPHER_CTX *ectx,
2520 HMAC_CTX *hctx, int encrypt) {
2521 SSL_CTX *ssl_ctx = ssl ? SSL_get_SSL_CTX(ssl) : NULL;
2522 const cwist_h3_ticket_key *key = NULL;
2523 if (ssl_ctx && g_h3_ticket_key_ex_data_idx >= 0) {
2524 key =
2525 (const cwist_h3_ticket_key *)SSL_CTX_get_ex_data(ssl_ctx, g_h3_ticket_key_ex_data_idx);
2526 }
2527 if (!key) return 0;
2528
2529 if (encrypt) {
2530 memcpy(key_name, key->name, sizeof(key->name));
2531 if (RAND_bytes(iv, 16) != 1) return 0;
2532 if (EVP_EncryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
2533 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1)
2534 return 0;
2535 return 1;
2536 }
2537
2538 if (memcmp(key_name, key->name, sizeof(key->name)) != 0) {
2539 return 0; /* Unknown key: decline the ticket, do a full handshake. */
2540 }
2541 if (EVP_DecryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
2542 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1) return 0;
2543 return 1;
2544}

3135 atomic_load(&g_cwist_running)) {
