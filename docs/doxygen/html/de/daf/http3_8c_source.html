345static void cwist_h3_hset_track(cwist_http3_context *ctx, cwist_h3_hset_t *hset) {
346 if (!ctx || !hset) return;
347 hset->owner = ctx;
348 hset->next = (cwist_h3_hset_t *)ctx->hsets;
349 hset->prev = (cwist_h3_hset_t **)&ctx->hsets;
350 if (hset->next) hset->next->prev = &hset->next;
351 ctx->hsets = hset;
352}

358static void cwist_h3_hset_untrack(cwist_h3_hset_t *hset) {
359 if (!hset || !hset->owner) return;
360 *hset->prev = hset->next;
361 if (hset->next) hset->next->prev = hset->prev;
362 hset->next = NULL;
363 hset->prev = NULL;
364 hset->owner = NULL;
365}

413static struct lsxpack_header *cwist_h3_hsi_prepare(void *hset_p, struct lsxpack_header *xhdr,
414 size_t req_space) {
415 cwist_h3_hset_t *hset = hset_p;
416 if (!hset) return NULL;
417
418 if (xhdr) {
419 if (req_space > LSXPACK_MAX_STRLEN || xhdr->name_offset < 0 ||
420 (size_t)xhdr->name_offset >= sizeof(hset->decode_buf) ||
421 req_space > sizeof(hset->decode_buf) - (size_t)xhdr->name_offset) {
422 fprintf(stderr,
423 "[HTTP/3] Rejecting oversized QPACK header resize (space=%zu, offset=%d)\n",
424 req_space, (int)xhdr->name_offset);
425 return NULL;
426 }
427 xhdr->val_len = (lsxpack_strlen_t)req_space;
428 return xhdr;
429 }
430
431 if (hset->count >= H3_MAX_HEADERS) return NULL;
432 if (req_space > sizeof(hset->decode_buf) - hset->decode_off) return NULL;
433 lsxpack_header_prepare_decode(&hset->headers[hset->count], hset->decode_buf, hset->decode_off,
434 sizeof(hset->decode_buf) - hset->decode_off);
435 return &hset->headers[hset->count];
436}

446static int cwist_h3_hsi_process_header(void *hset_p, struct lsxpack_header *xhdr) {
447 cwist_h3_hset_t *hset = hset_p;
448 /* A NULL header marks the end of a header block. */
449 if (!hset || !xhdr) return 0;
450
451 /* The QPACK decoder exposes the exact storage used by this completed
452 * header. */
453 size_t total = lsxpack_header_get_dec_size(xhdr);
454 if (total > sizeof(hset->decode_buf) - hset->decode_off) {
455 fprintf(stderr, "[HTTP/3] Rejecting oversized QPACK header (size=%zu, used=%zu)\n", total,
456 hset->decode_off);
457 return -1;
458 }
459 hset->decode_off += total;
460 hset->count++;
461 return 0;
462}

516static void h3_setup_cmsg(struct msghdr *msg, char *cbuf, size_t cbuf_sz,
517 const struct lsquic_out_spec *spec, uint16_t gso_seg) {
518 msg->msg_control = cbuf;
519 msg->msg_controllen = cbuf_sz;
520 memset(cbuf, 0, cbuf_sz);
521 size_t ctl_len = 0;
522
523#if defined(__linux__) && defined(UDP_SEGMENT)
524 if (gso_seg > 0) {
525 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
526 cmsg->cmsg_level = SOL_UDP;
527 cmsg->cmsg_type = UDP_SEGMENT;
528 cmsg->cmsg_len = CMSG_LEN(sizeof(uint16_t));
529 memcpy(CMSG_DATA(cmsg), &gso_seg, sizeof(uint16_t));
530 ctl_len += CMSG_SPACE(sizeof(uint16_t));
531 }
532#else
533 (void)gso_seg;
534#endif
535
536 if (spec->local_sa && spec->dest_sa) {
537 if (spec->dest_sa->sa_family == AF_INET && spec->local_sa->sa_family == AF_INET) {
538 struct in_addr addr = ((const struct sockaddr_in *)spec->local_sa)->sin_addr;
539 if (addr.s_addr != INADDR_ANY) {
540#if defined(__linux__) && defined(IP_PKTINFO)
541 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
542 cmsg->cmsg_level = IPPROTO_IP;
543 cmsg->cmsg_type = IP_PKTINFO;
544 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in_pktinfo));
545 struct in_pktinfo info = {0};
546 info.ipi_spec_dst = addr;
547 memcpy(CMSG_DATA(cmsg), &info, sizeof(info));
548 ctl_len += CMSG_SPACE(sizeof(struct in_pktinfo));
549#elif defined(IP_SENDSRCADDR)
550 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
551 cmsg->cmsg_level = IPPROTO_IP;
552 cmsg->cmsg_type = IP_SENDSRCADDR;
553 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in_addr));
554 memcpy(CMSG_DATA(cmsg), &addr, sizeof(addr));
555 ctl_len += CMSG_SPACE(sizeof(struct in_addr));
556#endif
557 }
558 } else if (spec->dest_sa->sa_family == AF_INET6 && spec->local_sa->sa_family == AF_INET6) {
559 const struct in6_addr *addr6 =
560 &((const struct sockaddr_in6 *)spec->local_sa)->sin6_addr;
561 if (memcmp(addr6, &in6addr_any, sizeof(struct in6_addr)) != 0) {
562#if defined(IPV6_PKTINFO)
563 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
564 cmsg->cmsg_level = IPPROTO_IPV6;
565 cmsg->cmsg_type = IPV6_PKTINFO;
566 cmsg->cmsg_len = CMSG_LEN(sizeof(struct in6_pktinfo));
567 struct in6_pktinfo info6 = {0};
568 info6.ipi6_addr = *addr6;
569 memcpy(CMSG_DATA(cmsg), &info6, sizeof(info6));
570 ctl_len += CMSG_SPACE(sizeof(struct in6_pktinfo));
571#endif
572 }
573 }
574 }
575
576#if defined(CWIST_H3_HAVE_ECN_CMSG)
577 if (spec->ecn && spec->dest_sa) {
578 if (spec->dest_sa->sa_family == AF_INET) {
579 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
580 cmsg->cmsg_level = IPPROTO_IP;
581 cmsg->cmsg_type = IP_TOS;
582 int tos = spec->ecn;
583 cmsg->cmsg_len = CMSG_LEN(sizeof(tos));
584 memcpy(CMSG_DATA(cmsg), &tos, sizeof(tos));
585 ctl_len += CMSG_SPACE(sizeof(tos));
586 }
587#if defined(IPV6_TCLASS)
588 else if (spec->dest_sa->sa_family == AF_INET6) {
589 struct cmsghdr *cmsg = (struct cmsghdr *)(cbuf + ctl_len);
590 cmsg->cmsg_level = IPPROTO_IPV6;
591 cmsg->cmsg_type = IPV6_TCLASS;
592 int tos = spec->ecn;
593 cmsg->cmsg_len = CMSG_LEN(sizeof(tos));
594 memcpy(CMSG_DATA(cmsg), &tos, sizeof(tos));
595 ctl_len += CMSG_SPACE(sizeof(tos));
596 }
597#endif
598 }
599#endif
600
601 msg->msg_controllen = ctl_len;
602 if (ctl_len == 0) {
603 msg->msg_control = NULL;
604 }
605}

666 return false;
667 }

712 memset(msg, 0, sizeof(*msg));
713 msg->msg_name = (void *)spec->dest_sa;
714 msg->msg_namelen = (spec->dest_sa && spec->dest_sa->sa_family == AF_INET)
715 ? sizeof(struct sockaddr_in)
716 : sizeof(struct sockaddr_in6);
717 msg->msg_iov = (struct iovec *)spec->iov;
718 msg->msg_iovlen = spec->iovlen;
719 h3_setup_cmsg(msg, ctrl_bufs[k], sizeof(ctrl_bufs[k]), spec, 0);
720 msgs[k].msg_len = 0;
721 }
722
723 int res = sendmmsg(udp_fd, msgs, batch, MSG_DONTWAIT);
724 if (res > 0) {
725 i += (unsigned)res;
726 if ((unsigned)res < batch) break;
727 continue;
728 }
729 if (res < 0) {
730 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
731 if (i > 0) return (int)i;
732 int nw = h3_send_one(udp_fd, &specs[i]);
733 if (nw >= 0) {
734 i++;
735 continue;
736 }
737 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
738 return -1;
739 }
740 break;
741 }
742 return (int)i;
743}
744#endif

761static int cwist_h3_packets_out(void *ctx, const struct lsquic_out_spec *specs, unsigned n_specs) {
762 int udp_fd = *(int *)ctx;
763#if defined(__linux__)
764 /* One-time GSO probe: check env var on first call. */
765 if (__builtin_expect(h3_gso_state < 0, 0)) {
766 const char *no_gso = getenv("CWIST_H3_NO_GSO");
767 h3_gso_state = (no_gso && no_gso[0] == '1') ? 1 : 0;
768 }
769
770 if (!h3_gso_state) {
771 /* GSO fast path: coalesce runs of equal-size datagrams to the same
772 * peer into one sendmsg with UDP_SEGMENT. */
773 unsigned i = 0;
774 while (i < n_specs) {
775 size_t seg = h3_spec_len(&specs[i]);
776 unsigned run = 1;
777 unsigned niov = specs[i].iovlen;
778 if (seg > 0 && seg <= 65535) {
779 /* Total payload must fit in one UDP datagram (65535 bytes);
780 * 65536/seg could allow an exactly-64KiB super-packet whose
781 * sendmsg fails with EMSGSIZE and would disable GSO. */
782 unsigned max_segs = (unsigned)(65535 / seg);
783 if (max_segs > 48) max_segs = 48;
784 while (i + run < n_specs && run < max_segs && specs[i + run].ecn == specs[i].ecn &&
785 h3_same_dest(&specs[i], &specs[i + run]) &&
786 h3_spec_len(&specs[i + run]) == seg && niov + specs[i + run].iovlen <= 128) {
787 niov += specs[i + run].iovlen;
788 run++;
789 }
790 }
791 if (run >= 2) {
792 struct iovec iov[128];
793 unsigned n = 0;
794 for (unsigned k = 0; k < run; k++)
795 for (unsigned m = 0; m < specs[i + k].iovlen; m++)
796 iov[n++] = specs[i + k].iov[m];
797
798 char ctrl[512];
799 struct msghdr msg = {0};
800 msg.msg_name = (void *)specs[i].dest_sa;
801 msg.msg_namelen = (specs[i].dest_sa && specs[i].dest_sa->sa_family == AF_INET)
802 ? sizeof(struct sockaddr_in)
803 : sizeof(struct sockaddr_in6);
804 msg.msg_iov = iov;
805 msg.msg_iovlen = n;
806 h3_setup_cmsg(&msg, ctrl, sizeof(ctrl), &specs[i], (uint16_t)seg);
807
808 ssize_t nw = sendmsg(udp_fd, &msg, MSG_DONTWAIT);
809 if (nw >= 0) {
810 i += run;
811 continue;
812 }
813 if (errno == EAGAIN || errno == EWOULDBLOCK) break;
814 /* GSO rejected: permanently disable and fall back to sendmmsg. */
815 h3_gso_state = 1;
816 return h3_sendmmsg_batch(udp_fd, specs, i, n_specs);
817 }
818 /* Single packet: send directly. */
819 int nw = h3_send_one(udp_fd, &specs[i]);
820 if (nw < 0) {
821 if (errno == EAGAIN || errno == EWOULDBLOCK) return (int)i;
822 if (errno == ECONNREFUSED || errno == ENETUNREACH || errno == EHOSTUNREACH ||
823 errno == EMSGSIZE) {
824 i++;
825 continue;
826 }
827 return (i > 0) ? (int)i : -1;
828 }
829 i++;
830 }
831 return (int)i;
832 }
833
834 /* sendmmsg fallback path (GSO disabled). */
835 return h3_sendmmsg_batch(udp_fd, specs, 0, n_specs);
836#else
837 unsigned i = 0;
838 for (; i < n_specs; ++i) {
839 int nw = h3_send_one(udp_fd, &specs[i]);
840 if (nw < 0) {
841 if (errno == EAGAIN || errno == EWOULDBLOCK) return (int)i;
842 if (errno == ECONNREFUSED || errno == ENETUNREACH || errno == EHOSTUNREACH ||
843 errno == EMSGSIZE)
844 continue;
845 return (i > 0) ? (int)i : -1;
846 }
847 }
848 return (int)i;
849#endif
850}

1106 }
1107 } else if (strcmp(name, ":scheme") == 0) {
1108 /* RFC 9114: silently ignore pseudo-headers we don't need to expose */
1109 } else if (strcmp(name, "content-length") == 0) {

1111 unsigned long long cl = strtoull(value, &endptr, 10);

1113 req->content_length = (size_t)cl;

1115 cwist_http_header_add(&req->headers, name, value);
1116 } else if (strcmp(name, "content-type") == 0) {

1118 } else if (strcmp(name, "cookie") == 0) {
1119 cwist_http_header_add(&req->headers, name, value);
1120 } else if (strcmp(name, "authorization") == 0) {
1121 cwist_http_header_add(&req->headers, name, value);
1122 } else if (strcmp(name, "accept") == 0) {
1123 cwist_http_header_add(&req->headers, name, value);
1124 } else if (strcmp(name, "user-agent") == 0) {
1125 cwist_http_header_add(&req->headers, name, value);
1126 } else if (strcmp(name, "accept-encoding") == 0) {

1128 } else if (strcmp(name, "accept-language") == 0) {
1129 cwist_http_header_add(&req->headers, name, value);
1130 } else if (strcmp(name, "referer") == 0 || strcmp(name, "referrer") == 0) {

1161static int h3_seq_append_and_feed(h3_stream_ctx_t *st, const unsigned char *data, size_t len) {
1162 if (len > SIZE_MAX - st->seq_len) return -1;
1163 size_t need = st->seq_len + len;
1164 if (need > st->seq_cap) {
1165 size_t cap = st->seq_cap ? st->seq_cap : 4096;
1166 while (cap < need) {
1167 if (cap > (CWIST_SEQ_HEADER_SIZE + UINT16_MAX) / 2) {
1168 cap = CWIST_SEQ_HEADER_SIZE + UINT16_MAX;
1169 break;
1170 }
1171 cap *= 2;
1172 }
1173 if (cap < need || cap > CWIST_SEQ_HEADER_SIZE + UINT16_MAX) return -1;
1174 unsigned char *tmp = realloc(st->seq_buf, cap);
1175 if (!tmp) return -1;
1176 st->seq_buf = tmp;
1177 st->seq_cap = cap;
1178 }
1179 memcpy(st->seq_buf + st->seq_len, data, len);
1180 st->seq_len += len;
1181
1182 while (st->seq_len >= CWIST_SEQ_HEADER_SIZE) {
1183 size_t payload_len = ((size_t)st->seq_buf[4] << 8) | st->seq_buf[5];
1184 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + payload_len;
1185 if (payload_len == 0 || st->seq_len < chunk_len) break;
1186 cwist_seq_chunk_t chunk;
1187 if (!cwist_seq_chunk_parse(st->seq_buf, chunk_len, &chunk)) return -1;
1188 if (!st->body_assembler) {
1189 st->body_assembler = cwist_seq_assembler_create_limited(CWIST_HTTP_MAX_BODY_SIZE);
1190 if (!st->body_assembler) return -1;
1191 }
1192 if (!cwist_seq_assembler_feed(st->body_assembler, &chunk)) return -1;
1193 memmove(st->seq_buf, st->seq_buf + chunk_len, st->seq_len - chunk_len);
1194 st->seq_len -= chunk_len;
1195 }
1196 return 0;
1197}

1237static bool h3_process_stream_headers(lsquic_stream_t *stream, h3_stream_ctx_t *st) {
1238 if (st->headers_done) return true;
1239 void *hset = lsquic_stream_get_hset(stream);
1240 if (!hset) return false;
1241 cwist_h3_hset_t *hs = (cwist_h3_hset_t *)hset;
1242 /* lsxpack exposes counted slices, not C strings, so each header needs a
1243 * NUL-terminated scratch copy to hand to strcmp()/h3_apply_header().
1244 * These used to be a malloc(name_len+1)/malloc(value_len+1) pair freed
1245 * at the bottom of every loop iteration - up to two heap round trips
1246 * per header, tens of them on a real request. The bounds below
1247 * (name_len <= 1024, value_len <= H3_DECODE_BUF_SIZE - 1) are already
1248 * enforced before anything touches these buffers, so a single
1249 * reusable pair sized to those same bounds, declared once outside the
1250 * loop, replaces all of that: h3_apply_header() copies out of them
1251 * before the next iteration overwrites them (see below), so reuse is
1252 * safe. */
1253 char name_scratch[1024 + 1];
1254 char value_scratch[H3_DECODE_BUF_SIZE];
1255 for (size_t i = 0; i < hs->count; ++i) {
1256 const struct lsxpack_header *xhdr = &hs->headers[i];
1257 const char *raw_name = lsxpack_header_get_name(xhdr);
1258 const char *raw_value = lsxpack_header_get_value(xhdr);
1259 size_t name_len = xhdr->name_len;
1260 size_t value_len = xhdr->val_len;
1261 if (raw_name && raw_value && name_len > 0 && name_len <= 1024 &&
1262 value_len <= H3_DECODE_BUF_SIZE - 1) {
1263 char *name = name_scratch;
1264 char *value = value_scratch;
1265 memcpy(name, raw_name, name_len);
1266 name[name_len] = '\0';
1267 memcpy(value, raw_value, value_len);
1268 value[value_len] = '\0';
1269 if (name[0] == ':') {
1270 /* RFC 9114 Section 4.3.1: pseudo-headers precede regular
1271 * headers, appear at most once, and only the defined set is
1272 * valid in requests. */
1273 unsigned bit = 0;
1274 if (strcmp(name, ":method") == 0)
1275 bit = H3_PSEUDO_METHOD;
1276 else if (strcmp(name, ":scheme") == 0)
1277 bit = H3_PSEUDO_SCHEME;
1278 else if (strcmp(name, ":path") == 0)
1279 bit = H3_PSEUDO_PATH;
1280 else if (strcmp(name, ":authority") == 0)
1281 bit = H3_PSEUDO_AUTHORITY;
1282 else if (strcmp(name, ":protocol") == 0)
1283 bit = H3_PSEUDO_PROTOCOL;
1284 if (bit == 0 || st->seen_regular_header || (st->pseudo_seen & bit)) {
1285 h3_reject_malformed_request(stream, st);
1286 return false;
1287 }
1288 st->pseudo_seen |= bit;
1289 if (bit == H3_PSEUDO_METHOD && strcmp(value, "CONNECT") == 0) {
1290 st->is_connect = 1;
1291 }
1292 if (bit == H3_PSEUDO_PATH && value_len == 0) {
1293 st->saw_empty_path = 1;
1294 }
1295 } else {
1296 st->seen_regular_header = 1;
1297 }
1298 h3_apply_header(st->req, name, value);
1299#ifdef CWIST_WEBTRANSPORT
1300 if (strcmp(name, ":protocol") == 0 && strcmp(value, "webtransport") == 0) {
1301 if (st->req && st->req->method == CWIST_HTTP_CONNECT) {
1302 st->is_webtransport = 1;
1303 }
1304 }
1305#endif
1306 }
1307 }
1308 /* RFC 9114 Section 4.3.1 completeness rules, evaluated once the whole
1309 * header block has been seen. */
1310 if (st->is_connect && !(st->pseudo_seen & H3_PSEUDO_PROTOCOL)) {
1311 /* Plain CONNECT: :scheme and :path MUST be omitted, :authority is
1312 * required. */
1313 if ((st->pseudo_seen & (H3_PSEUDO_SCHEME | H3_PSEUDO_PATH)) ||
1314 !(st->pseudo_seen & H3_PSEUDO_AUTHORITY)) {
1315 h3_reject_malformed_request(stream, st);
1316 return false;
1317 }
1318 } else if (st->is_connect) {
1319 /* Extended CONNECT (RFC 9220, e.g. WebTransport): :scheme, :path and
1320 * :authority are all required. */
1321 if (!(st->pseudo_seen & H3_PSEUDO_SCHEME) || !(st->pseudo_seen & H3_PSEUDO_PATH) ||
1322 !(st->pseudo_seen & H3_PSEUDO_AUTHORITY)) {
1323 h3_reject_malformed_request(stream, st);
1324 return false;
1325 }
1326 } else {
1327 /* Normal request: :method, :scheme and :path are mandatory,
1328 * :authority (or an equivalent Host header) is required for https,
1329 * and :protocol is only legal on extended CONNECT. */
1330 if (!(st->pseudo_seen & H3_PSEUDO_METHOD) || !(st->pseudo_seen & H3_PSEUDO_SCHEME) ||
1331 !(st->pseudo_seen & H3_PSEUDO_PATH) || (st->pseudo_seen & H3_PSEUDO_PROTOCOL) ||
1332 (!(st->pseudo_seen & H3_PSEUDO_AUTHORITY) &&
1333 !cwist_http_header_get(st->req->headers, "host"))) {
1334 h3_reject_malformed_request(stream, st);
1335 return false;
1336 }
1337 /* An empty :path is only legal for OPTIONS (asterisk-form). */
1338 if (st->saw_empty_path && (!st->req || st->req->method != CWIST_HTTP_OPTIONS)) {
1339 h3_reject_malformed_request(stream, st);
1340 return false;
1341 }
1342 }
1343 st->headers_done = 1;
1344 char *seq_header = cwist_http_header_get(st->req->headers, "x-cwist-sequenced-data");
1345 st->sequenced_data =
1346 seq_header && (strcmp(seq_header, "1") == 0 || strcasecmp(seq_header, "true") == 0);
1347 return true;
1348}

1376static void cwist_h3_on_read(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
1377 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
1378 if (!st) return;
1379
1380 unsigned char buf[8192];
1381 ssize_t nread;
1382
1383 h3_process_stream_headers(stream, st);
1384 if (st->malformed) return;
1385
1386 while ((nread = lsquic_stream_read(stream, buf, sizeof(buf))) > 0) {
1387 if (st->sequenced_data) {
1388 if (h3_seq_append_and_feed(st, buf, (size_t)nread) != 0) {
1389 lsquic_stream_close(stream);
1390 return;
1391 }
1392 continue;
1393 }
1394 size_t need = st->body_len + (size_t)nread;
1395 if (need > CWIST_HTTP_MAX_BODY_SIZE) {
1396 /* Body too large: answer 413 with a small JSON body instead of a
1397 * bare stream close, so the client sees an explicit status
1398 * (RFC 9110 Section 15.5.14). */
1399 if (!st->res) st->res = cwist_http_response_create();
1400 if (st->res) {
1401 st->res->status_code = 413; /* Payload Too Large */
1402 if (st->res->body) {
1403 cwist_sstring_assign(st->res->body, "{\"error\":\"payload too large\"}");
1404 }
1405 if (st->res->headers) {
1406 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1407 }
1408 st->response_ready = 1;
1409 lsquic_stream_wantread(stream, 0);
1410 lsquic_stream_shutdown(stream, 0);
1411 lsquic_stream_wantwrite(stream, 1);
1412 } else {
1413 lsquic_stream_close(stream);
1414 }
1415 return;
1416 }
1417 if (need > st->body_cap) {
1418 size_t new_cap = st->body_cap ? st->body_cap * 2 : 4096;
1419 while (new_cap < need) new_cap *= 2;
1420 char *tmp = realloc(st->body, new_cap);
1421 if (!tmp) {
1422 lsquic_stream_close(stream);
1423 return;
1424 }
1425 st->body = tmp;
1426 st->body_cap = new_cap;
1427 }
1428 memcpy(st->body + st->body_len, buf, (size_t)nread);
1429 st->recv_xor ^= h3_xor_bytes((const unsigned char *)buf, (size_t)nread);
1430 st->body_len += (size_t)nread;
1431 }
1432
1433 if (nread == 0) {
1434 /* End of stream (FIN received) */
1435 h3_process_stream_headers(stream, st);
1436
1437 if (st->malformed) return;
1438
1439 if (!st->headers_done) {
1440 /* Malformed request: no headers before FIN */
1441 st->res = cwist_http_response_create();
1442 if (st->res) {
1443 st->res->status_code = CWIST_HTTP_BAD_REQUEST;
1444 if (st->res->body) {
1445 cwist_sstring_assign(st->res->body, "{\"error\":\"malformed request\"}");
1446 }
1447 if (st->res->headers) {
1448 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1449 }
1450 }
1451 st->response_ready = 1;
1452 lsquic_stream_wantread(stream, 0);
1453 lsquic_stream_shutdown(stream, 0);
1454 lsquic_stream_wantwrite(stream, 1);
1455 return;
1456 }
1457
1458 if (st->sequenced_data) {
1459 const uint8_t *assembled = NULL;
1460 size_t assembled_len = 0;
1461 if (st->seq_len != 0 || !st->body_assembler ||
1462 !cwist_seq_assembler_get_data(st->body_assembler, &assembled, &assembled_len)) {
1463 /* Do not dispatch a partial request. */
1464 st->res = cwist_http_response_create();
1465 if (st->res) {
1466 st->res->status_code = CWIST_HTTP_BAD_REQUEST;
1467 cwist_http_header_add(&st->res->headers, "x-cwist-retry", "1");
1468 if (st->res->body) {
1469 cwist_sstring_assign(st->res->body,
1470 "{\"error\":\"incomplete sequenced body, retry\"}");
1471 }
1472 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1473 }
1474 st->response_ready = 1;
1475 lsquic_stream_wantread(stream, 0);
1476 lsquic_stream_shutdown(stream, 0);
1477 lsquic_stream_wantwrite(stream, 1);
1478 return;
1479 }
1480 if (assembled_len == 0) {
1481 /* realloc(p, 0) may legally return NULL; do not mistake an
1482 * empty assembled body for an allocation failure (which used
1483 * to reset the stream instead of dispatching the request). */
1484 free(st->body);
1485 st->body = NULL;
1486 st->body_len = 0;
1487 st->body_cap = 0;
1488 } else {
1489 st->body = realloc(st->body, assembled_len);
1490 if (!st->body) {
1491 lsquic_stream_close(stream);
1492 return;
1493 }
1494 memcpy(st->body, assembled, assembled_len);
1495 st->body_len = assembled_len;
1496 st->body_cap = assembled_len;
1497 }
1498 }
1499
1500 if (st->body_len > 0 && st->req && st->req->body) {
1501 cwist_sstring_assign_len(st->req->body, st->body, st->body_len);
1502 }
1503
1504 st->res = cwist_http_response_create();
1505 if (st->res && st->req) {
1506 cwist_http3_context *h3_ctx = h3_shared_ctx(lsquic_stream_conn(stream));
1507#ifdef CWIST_WEBTRANSPORT
1508 if (st->is_webtransport && h3_ctx && h3_ctx->wt_handler) {
1509 char *host = cwist_http_header_get(st->req->headers, "host");
1510 char *origin = cwist_http_header_get(st->req->headers, "origin");
1511 struct lsquic_wt_connect_info info = {
1512 .wtci_authority = host,
1513 .wtci_path = st->req->path ? st->req->path->data : NULL,
1514 .wtci_origin = origin,
1515 .wtci_protocol = NULL,
1516 .wtci_draft = lsquic_wt_peer_draft(lsquic_stream_conn(stream)),
1517 };
1518 struct lsquic_wt_accept_params params = {
1519 .wtap_status = LSQUIC_WTAP_STATUS_DEFAULT,
1520 .wtap_wt_if = &cwist_h3_wt_if,
1521 .wtap_wt_if_ctx = st,
1522 .wtap_connect_info = &info,
1523 .wtap_datagram_send_mode = LSQUIC_HTTP_DG_SEND_DEFAULT,
1524 };
1525 if (lsquic_wt_accept(stream, &params) == 0) {
1526 st->wt_taken = 1;
1527 lsquic_stream_wantread(stream, 0);
1528 lsquic_stream_shutdown(stream, 0);
1529 lsquic_stream_wantwrite(stream, 0);
1530 return;
1531 }
1532 /* Accept failed: falling through with the default 200/empty
1533 * response used to send the client a silent 0-byte body. */
1534 CWIST_LOG_WARN("[HTTP/3] WebTransport accept failed; answering 500");
1535 st->res->status_code = CWIST_HTTP_INTERNAL_ERROR;
1536 if (st->res->body && st->res->body->size == 0) {
1537 cwist_sstring_assign(st->res->body,
1538 "{\"error\":\"webtransport accept failed\"}");
1539 }
1540 if (st->res->headers) {
1541 cwist_http_header_add(&st->res->headers, "Content-Type", "application/json");
1542 }
1543 } else
1544#endif
1545 if (h3_ctx && h3_ctx->handler) {
1546 /* 0-RTT replay guard: requests delivered while the QUIC
1547 * handshake is still in progress arrived as early data.
1548 * lsquic (this baseline) exposes no
1549 * lsquic_conn_is_early_data_accepted()-style query, so the
1550 * guard uses the public lsquic_conn_status() instead. */
1551 bool is_early_data = h3_ctx->early_data_enabled && h3_ctx->early_data_guard &&
1552 lsquic_conn_status(lsquic_stream_conn(stream), NULL, 0) ==
1553 LSCONN_ST_HSK_IN_PROGRESS;
1554 if (is_early_data && st->req && !h3_method_is_idempotent(st->req->method)) {
1555 /* RFC 8470: refuse replayable non-idempotent early data
1556 * so the client retries after the handshake. Attach a
1557 * body: without one the response goes out as HEADERS+FIN
1558 * with no content-length, which some clients surface as
1559 * an empty (0-byte) reply instead of status 425. */
1560 st->res->status_code = 425; /* Too Early */
1561 if (st->res->body) {
1562 cwist_sstring_assign(st->res->body,
1563 "{\"error\":\"too early, retry after handshake\"}");
1564 }
1565 if (st->res->headers) {
1566 cwist_http_header_add(&st->res->headers, "Content-Type",
1567 "application/json");
1568 }
1569 } else {
1570 h3_ctx->handler(h3_ctx->user_ctx, st->req, st->res);
1571 }
1572 }
1573 }
1574 st->response_ready = 1;
1575 lsquic_stream_wantread(stream, 0);
1576 lsquic_stream_shutdown(stream, 0);
1577 lsquic_stream_wantwrite(stream, 1);
1578 } else if (nread < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
1579 lsquic_stream_close(stream);
1580 }
1581}

1591 for (const char *p = cl + 1; *p; ++p) {
1592 if (*p < '0' || *p > '9') return false;
1593 }
1594 return true;

1619static void cwist_h3_on_write(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
1620 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
1621 if (!st || !st->response_ready) return;
1622 if (!st->res) {
1623 /* Response allocation failed upstream; reset instead of dereferencing
1624 * NULL (which used to take down every stream on the connection). */
1625 lsquic_stream_close(stream);
1626 return;
1627 }
1628
1629 if (st->write_state == 0) {
1630 struct lsxpack_header headers_arr[H3_MAX_RESPONSE_HEADERS];
1631 char hbuf[H3_RESPONSE_HEADER_BUF_SIZE];
1632 size_t hbuf_off = 0;
1633 size_t hdr_count = 0;
1634
1635 bool bodyless = h3_status_forbids_body(st->res->status_code);
1636 bool is_head = st->req && st->req->method == CWIST_HTTP_HEAD;
1637
1638 /* Handler attached a body source that cannot produce bytes
1639 * (NULL pointer body / invalid file fd with a non-zero length).
1640 * Sending HEADERS would announce (or imply) a body we cannot
1641 * deliver; close the stream so the client sees an explicit error
1642 * instead of a silent empty 200. */
1643 if (!bodyless && !is_head &&
1644 ((st->res->is_ptr_body && !st->res->ptr_body && st->res->ptr_body_len > 0) ||
1645 (st->res->use_file_stream && st->res->file_stream_fd < 0 &&
1646 st->res->file_stream_len > 0))) {
1647 CWIST_LOG_WARN("[HTTP/3] unusable response body source; resetting stream");
1648 lsquic_stream_close(stream);
1649 return;
1650 }
1651
1652 /* :status */
1653 if (st->res->status_code < 200 || st->res->status_code > 999) {
1654 /* RFC 9110 Section 15: a final response needs a status in
1655 * 200-999 (RFC 9114 Section 4.1.2 requires three digits).
1656 * A handler that leaves the status unset, picks 1xx (an
1657 * interim response, illegal as the complete answer), or an
1658 * out-of-range value gets coerced to 500 instead of putting
1659 * a malformed :status on the wire. */
1660 CWIST_LOG_WARN("[HTTP/3] invalid status %d; coercing to 500", st->res->status_code);
1661 st->res->status_code = CWIST_HTTP_INTERNAL_ERROR;
1662 bodyless = false;
1663 if (!is_head && st->res->body && st->res->body->size == 0 && !st->res->is_ptr_body &&
1664 !st->res->use_file_stream) {
1665 cwist_sstring_assign(st->res->body, "{\"error\":\"internal server error\"}");
1666 }
1667 }
1668 int status_code = st->res->status_code;
1669 char status_str[16];
1670 snprintf(status_str, sizeof(status_str), "%d", status_code);
1671 size_t slen = strlen(status_str);
1672 if (hdr_count < H3_MAX_RESPONSE_HEADERS && slen > 0 &&
1673 hbuf_off + 7 + 2 + slen <= sizeof(hbuf)) {
1674 memcpy(hbuf + hbuf_off, ":status", 7);
1675 memcpy(hbuf + hbuf_off + 9, status_str, slen);
1676 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0, 7, 9, slen);
1677 hbuf_off += 9 + slen;
1678 hdr_count++;
1679 }
1680
1681 /* content-length */
1682 size_t body_len = 0;
1683 if (st->res->use_file_stream) {
1684 if (st->res->file_stream_fd >= 0) body_len = st->res->file_stream_len;
1685 } else if (st->res->is_ptr_body)
1686 body_len = st->res->ptr_body ? st->res->ptr_body_len : 0;
1687 else if (st->res->body)
1688 body_len = st->res->body->size;
1689
1690 const char *user_cl =
1691 st->res->headers ? cwist_http_header_get(st->res->headers, "content-length") : NULL;
1692
1693 if (!bodyless && (body_len > 0 || is_head) && hdr_count < H3_MAX_RESPONSE_HEADERS) {
1694 /* For HEAD, content-length describes the would-be GET body: keep
1695 * the handler-provided value when present (e.g. static file size
1696 * with an empty body), otherwise compute it like a GET. For all
1697 * other methods the computed body length is authoritative: it is
1698 * exactly what the write path below will send, so trusting a
1699 * divergent handler value would announce bytes never delivered. */
1700 char cl_str[32];
1701 const char *cl_val = NULL;
1702 if (is_head && user_cl) {
1703 if (h3_content_length_is_valid(user_cl)) {
1704 cl_val = user_cl;
1705 } else {
1706 /* Malformed handler content-length must not reach the
1707 * wire; fall back to the computed body length. */
1708 CWIST_LOG_WARN("[HTTP/3] dropping invalid HEAD content-length '%s'", user_cl);
1709 }
1710 }
1711 if (!cl_val) {
1712 snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
1713 cl_val = cl_str;
1714 }
1715 if (cl_val && cl_val[0] != '\0') {
1716 size_t cl_name_len = 14;
1717 size_t cl_val_len = strlen(cl_val);
1718 size_t total = cl_name_len + 2 + cl_val_len;
1719 if (cl_val_len > 0 && hbuf_off + total <= sizeof(hbuf)) {
1720 memcpy(hbuf + hbuf_off, "content-length", cl_name_len);
1721 memcpy(hbuf + hbuf_off + cl_name_len + 2, cl_val, cl_val_len);
1722 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0,
1723 cl_name_len, cl_name_len + 2, cl_val_len);
1724 hbuf_off += total;
1725 hdr_count++;
1726 }
1727 }
1728 }
1729
1730 /* content-type (if body is present or HEAD request) */
1731 if (st->res->headers && (body_len > 0 || is_head)) {
1732 char *ct = cwist_http_header_get(st->res->headers, "content-type");
1733 if (ct && ct[0] != '\0' && hdr_count < H3_MAX_RESPONSE_HEADERS) {
1734 size_t klen = 12;
1735 size_t vlen = strlen(ct);
1736 if (vlen > 0 && hbuf_off + klen + 2 + vlen <= sizeof(hbuf)) {
1737 memcpy(hbuf + hbuf_off, "content-type", klen);
1738 memcpy(hbuf + hbuf_off + klen + 2, ct, vlen);
1739 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0, klen,
1740 klen + 2, vlen);
1741 hbuf_off += klen + 2 + vlen;
1742 hdr_count++;
1743 }
1744 }
1745 }
1746
1747 /* user headers (skip content-length/content-type already handled) */
1748 cwist_http_header_node *node = st->res->headers;
1749 while (node && hdr_count < H3_MAX_RESPONSE_HEADERS) {
1750 if (node->key && node->key->data && node->key->size > 0 && node->value &&
1751 node->value->data) {
1752 char h3_name[256];
1753 if (cwist_http3_normalize_response_header_name(node->key->data, h3_name,
1754 sizeof(h3_name)) != 0 ||
1755 h3_name[0] == '\0' || strlen(node->value->data) != node->value->size ||
1756 !cwist_http3_response_header_value_is_safe(node->value->data)) {
1757 node = node->next;
1758 continue;
1759 }
1760
1761 if (strcmp(h3_name, "content-length") == 0 ||
1762 strcmp(h3_name, "content-type") == 0) {
1763 node = node->next;
1764 continue;
1765 }
1766 /* RFC 9114 section 4.2: connection-specific fields are malformed in
1767 * HTTP/3; te is only allowed with the value "trailers". */
1768 if (strcmp(h3_name, "connection") == 0 || strcmp(h3_name, "keep-alive") == 0 ||
1769 strcmp(h3_name, "proxy-connection") == 0 ||
1770 strcmp(h3_name, "transfer-encoding") == 0 || strcmp(h3_name, "upgrade") == 0 ||
1771 (strcmp(h3_name, "te") == 0 &&
1772 strcasecmp(node->value->data, "trailers") != 0)) {
1773 node = node->next;
1774 continue;
1775 }
1776
1777 size_t klen = strlen(h3_name);
1778 size_t vlen = node->value->size;
1779 if (klen > 0 && hbuf_off + klen + 2 + vlen <= sizeof(hbuf)) {
1780 memcpy(hbuf + hbuf_off, h3_name, klen);
1781 memcpy(hbuf + hbuf_off + klen + 2, node->value->data, vlen);
1782 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off, 0, klen,
1783 klen + 2, vlen);
1784 hbuf_off += klen + 2 + vlen;
1785 hdr_count++;
1786 }
1787 }
1788 node = node->next;
1789 }
1790
1791 if (status_code == CWIST_HTTP_OK && !bodyless && !is_head && body_len == 0) {
1792 /* Legal per RFC 9110, but a 200 with no body and no
1793 * content-length usually means a handler forgot to assign one;
1794 * make it diagnosable in the logs. */
1795 CWIST_LOG_WARN("[HTTP/3] handler produced 200 with empty body");
1796 }
1797
1798 size_t initial_body_len = 0;
1799 if (st->res->use_file_stream && st->res->file_stream_fd >= 0) {
1800 initial_body_len = st->res->file_stream_len;
1801 } else if (st->res->is_ptr_body && st->res->ptr_body) {
1802 initial_body_len = st->res->ptr_body_len;
1803 } else if (st->res->body) {
1804 initial_body_len = st->res->body->size;
1805 }
1806 int eos = (bodyless || is_head || initial_body_len == 0) ? 1 : 0;
1807
1808 lsquic_http_headers_t headers = {
1809 .count = (unsigned)hdr_count,
1810 .headers = headers_arr,
1811 };
1812 if (lsquic_stream_send_headers(stream, &headers, eos) != 0) {
1813 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1814 lsquic_stream_wantwrite(stream, 1);
1815 return;
1816 }
1817 lsquic_stream_close(stream);
1818 return;
1819 }
1820 if (eos) {
1821 /* NB: the eos argument of lsquic_stream_send_headers is ignored
1822 * for IETF QUIC, so an empty body must be finished with an
1823 * explicit shutdown(1) here. Relying on eos leaves every
1824 * empty-body response (204s, 304s, HEAD, error statuses) without
1825 * FIN, which browsers surface as a protocol error — timing- and
1826 * RTT-dependent because of header-block stashing under low cwnd. */
1827 lsquic_stream_shutdown(stream, 1);
1828 st->write_state = 2;
1829 lsquic_stream_wantread(stream, 1);
1830 return;
1831 }
1832 st->write_state = 1;
1833 lsquic_stream_wantwrite(stream, 1);
1834 return;
1835 }
1836
1837 if (st->write_state == 1 && st->res) {
1838 bool bodyless = h3_status_forbids_body(st->res->status_code);
1839 bool is_head = st->req && st->req->method == CWIST_HTTP_HEAD;
1840 size_t body_len = 0;
1841 const char *body_data = NULL;
1842
1843 if (st->res->use_file_stream) {
1844 /* Fill the congestion window on each callback: loop chunk writes
1845 * until the stream says EAGAIN instead of one 64 KiB chunk per
1846 * tick - but cap how many synchronous pread()s a single callback
1847 * does. This thread also drives lsquic_engine_process_conns()
1848 * and packet I/O for every other connection; with an unbounded
1849 * loop, one big response on a wide-open congestion window turns
1850 * into dozens of blocking pread()s back to back (worse under
1851 * disk contention/page faults than the SSD-idle case), starving
1852 * every other connection's ACKs/packets for that whole stretch.
1853 * Re-arming wantwrite and returning after H3_FILE_CHUNKS_PER_TICK
1854 * chunks gives the event loop a chance to service other
1855 * connections between bursts; lsquic calls this back again on
1856 * the next tick to keep going from where body_sent left off. */
1857#define H3_FILE_CHUNKS_PER_TICK 4
1858 int chunks_this_tick = 0;
1859 while (st->res->file_stream_fd >= 0 && st->body_sent < st->res->file_stream_len) {
1860 if (chunks_this_tick >= H3_FILE_CHUNKS_PER_TICK) {
1861 lsquic_stream_wantwrite(stream, 1);
1862 return;
1863 }
1864 static __thread char file_buf[65536];
1865 off_t offset = st->res->file_stream_offset + (off_t)st->body_sent;
1866 size_t to_read = st->res->file_stream_len - st->body_sent;
1867 if (to_read > sizeof(file_buf)) to_read = sizeof(file_buf);
1868 ssize_t nr = pread(st->res->file_stream_fd, file_buf, to_read, offset);
1869 if (nr > 0) {
1870 chunks_this_tick++;
1871 ssize_t nw = lsquic_stream_write(stream, file_buf, (size_t)nr);
1872 if (nw < 0) {
1873 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1874 lsquic_stream_wantwrite(stream, 1);
1875 return;
1876 }
1877 lsquic_stream_close(stream);
1878 return;
1879 }
1880 if (nw == 0) {
1881 lsquic_stream_wantwrite(stream, 1);
1882 return;
1883 }
1884 st->send_xor ^= h3_xor_bytes((const unsigned char *)file_buf, (size_t)nw);
1885 st->body_sent += (size_t)nw;
1886 if (nw < nr) {
1887 /* Stream buffer full for now; come back when writable. */
1888 lsquic_stream_wantwrite(stream, 1);
1889 return;
1890 }
1891 continue;
1892 } else if (nr == 0) {
1893 /* Short EOF (file truncated after fstat): the announced
1894 * content-length can no longer be honoured, and finishing
1895 * anyway would be a malformed-message error at the
1896 * client. Reset the stream instead. */
1897 lsquic_stream_close(stream);
1898 return;
1899 } else {
1900 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1901 /* Retry next tick */
1902 lsquic_stream_wantwrite(stream, 1);
1903 return;
1904 }
1905 lsquic_stream_close(stream);
1906 return;
1907 }
1908 }
1909#undef H3_FILE_CHUNKS_PER_TICK
1910 if (st->res->file_stream_fd >= 0) body_len = st->res->file_stream_len;
1911 } else if (st->res->is_ptr_body) {
1912 if (!st->res->ptr_body && st->res->ptr_body_len > 0) {
1913 /* Announced a body we cannot produce; spinning here with
1914 * wantwrite armed would hang the stream forever. */
1915 lsquic_stream_close(stream);
1916 return;
1917 }
1918 body_len = st->res->ptr_body_len;
1919 body_data = (const char *)st->res->ptr_body;
1920 } else if (st->res->body) {
1921 body_len = st->res->body->size;
1922 body_data = st->res->body->data;
1923 }
1924
1925 /* Skip payload writing entirely if body is empty, forbidden, or already fully sent */
1926 if (bodyless || is_head || body_len == 0 || st->body_sent >= body_len) {
1927 st->write_state = 2;
1928 lsquic_stream_shutdown(stream, 1);
1929 lsquic_stream_wantread(stream, 1);
1930 return;
1931 }
1932
1933 while (body_data && body_len > 0 && st->body_sent < body_len) {
1934 ssize_t n =
1935 lsquic_stream_write(stream, body_data + st->body_sent, body_len - st->body_sent);
1936 if (n > 0) {
1937 st->send_xor ^=
1938 h3_xor_bytes((const unsigned char *)(body_data + st->body_sent), (size_t)n);
1939 st->body_sent += (size_t)n;
1940 if (st->body_sent < body_len) {
1941 lsquic_stream_wantwrite(stream, 1);
1942 }
1943 continue;
1944 }
1945 if (n < 0) {
1946 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1947 lsquic_stream_wantwrite(stream, 1);
1948 return;
1949 }
1950 lsquic_stream_close(stream);
1951 return;
1952 }
1953 /* n == 0: stream buffer full for now; re-arm wantwrite and come back */
1954 lsquic_stream_wantwrite(stream, 1);
1955 return;
1956 }
1957
1958 if (st->body_sent >= body_len) {
1959 st->write_state = 2;
1960 lsquic_stream_shutdown(stream, 1);
1961 lsquic_stream_wantread(stream, 1);
1962 } else {
1963 lsquic_stream_wantwrite(stream, 1);
1964 }
1965 }
1966}

2223/* --- Shared session ticket key -------------------------------------------
2224 * Session resumption (and therefore 0-RTT early data) requires the server to
2225 * issue resumable tickets. Mirrors the prefork-safe pattern used by the
2226 * HTTPS stack (src/net/http/https.c): one random key generated at context
2227 * creation and installed via an explicit ticket-key callback, so tickets
2228 * issued by one worker resume on any other (SO_REUSEPORT scatters
2229 * connections across workers). The key lives for the process lifetime.
2230 * ------------------------------------------------------------------------- */
2231typedef struct cwist_h3_ticket_key {
2232 unsigned char name[16]; /* key_name sent in the ticket */
2233 unsigned char aes_key[32]; /* AES-256-CBC encryption key */
2234 unsigned char hmac_key[32]; /* HMAC-SHA-256 MAC key */
2235} cwist_h3_ticket_key;

2258static int cwist_h3_ticket_key_cb(SSL *ssl, uint8_t *key_name, uint8_t *iv, EVP_CIPHER_CTX *ectx,
2259 HMAC_CTX *hctx, int encrypt) {
2260 SSL_CTX *ssl_ctx = ssl ? SSL_get_SSL_CTX(ssl) : NULL;
2261 const cwist_h3_ticket_key *key = NULL;
2262 if (ssl_ctx && g_h3_ticket_key_ex_data_idx >= 0) {
2263 key =
2264 (const cwist_h3_ticket_key *)SSL_CTX_get_ex_data(ssl_ctx, g_h3_ticket_key_ex_data_idx);
2265 }
2266 if (!key) return 0;
2267
2268 if (encrypt) {
2269 memcpy(key_name, key->name, sizeof(key->name));
2270 if (RAND_bytes(iv, 16) != 1) return 0;
2271 if (EVP_EncryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
2272 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1)
2273 return 0;
2274 return 1;
2275 }
2276
2277 if (memcmp(key_name, key->name, sizeof(key->name)) != 0) {
2278 return 0; /* Unknown key: decline the ticket, do a full handshake. */
2279 }
2280 if (EVP_DecryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
2281 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1) return 0;
2282 return 1;
2283}

3491 lsquic_conn_t *c = (lsquic_conn_t *)conn;
3492 if (!c || !data || len == 0) return -1;
3493

3495 if (!cc) return -1;
3496

3498 if (!node) return -1;

3500 if (!node->data) {
3501 free(node);
3502 return -1;

3505 node->len = len;
3506 node->next = NULL;
3507
3508 /* Append to this connection's own queue - concurrent sends on other
3509 * connections touch their own h3_conn_ctx_t and never see this lock,

3514 cc->dgram_tail->next = node;
3515 } else {
3516 cc->dgram_head = node;
3517 }
3518 cc->dgram_tail = node;
3519 pthread_mutex_unlock(&cc->dgram_lock);
3520
3521 lsquic_conn_want_datagram_write(c, 1);
3522 return 0;
3523}
