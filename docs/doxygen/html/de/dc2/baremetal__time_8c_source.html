61time_t timegm(struct tm *tm) {
62 long long y = (long long)tm->tm_year + 1900;
63 int m = tm->tm_mon + 1;
64 y -= m <= 2;
65 long long era = (y >= 0 ? y : y - 399) / 400;
66 unsigned yoe = (unsigned)(y - era * 400);
67 unsigned doy = (153u * (unsigned)(m > 2 ? m - 3 : m + 9) + 2u) / 5u + (unsigned)tm->tm_mday - 1u;
68 unsigned doe = yoe * 365u + yoe / 4u - yoe / 100u + doy;
69 long long days = era * 146097LL + (long long)doe - 719468LL;
70 return (time_t)(days * 86400LL + tm->tm_hour * 3600LL + tm->tm_min * 60LL + tm->tm_sec);
71}
