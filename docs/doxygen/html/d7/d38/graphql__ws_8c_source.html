49static bool gql_ws_subprotocol_offered(const char *value) {
50 static const char protocol[] = "graphql-transport-ws";
51 if (!value) return false;
52 const char *p = value;
53 while (*p) {
54 while (*p == ' ' || *p == '\t' || *p == ',') p++;
55 const char *start = p;
56 while (*p && *p != ',') p++;
57 const char *end = p;
58 while (end > start && (end[-1] == ' ' || end[-1] == '\t')) end--;
59 if ((size_t)(end - start) == sizeof(protocol) - 1 &&
60 memcmp(start, protocol, sizeof(protocol) - 1) == 0)
61 return true;
62 if (*p) p++;
63 }
64 return false;
65}

81bool cwist_graphql_ws_upgrade(cwist_http_request *req, int client_fd) {
82 if (!req || client_fd < 0) return false;
83 cwist_http_response *res = cwist_http_response_create();
84 if (!res) return false;
85 bool ok = cwist_graphql_ws_upgrade_response(req, res);
86 if (!ok) {
87 if (res->status_code != CWIST_HTTP_OK) cwist_http_send_response(client_fd, res);
88 cwist_http_response_destroy(res);
89 return false;
90 }

93 return false;
94 }
95 cwist_http_response_destroy(res);
96 req->upgraded = true;
97 return true;

447#define GQL_WS_MAX_PENDING_BYTES (1024u * 1024u)
448
449/* Append a `next` message to the connection's queue (caller holds g_broker_lock). */
450static bool gql_ws_pending_push(gql_ws_conn_t *conn, char *msg) {
451 size_t msg_len = strlen(msg);
452 if (msg_len > GQL_WS_MAX_PENDING_BYTES ||
453 conn->pending_bytes > GQL_WS_MAX_PENDING_BYTES - msg_len)
454 return false;
455 if (conn->pending_count == conn->pending_cap) {
456 size_t new_cap = conn->pending_cap ? conn->pending_cap * 2 : 8;
457 char **np = (char **)cwist_alloc_array(new_cap, sizeof(char *));
458 if (!np) return false;
459 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(np);
460 if (conn->pending_count > 0)
461 memcpy(np, conn->pending, conn->pending_count * sizeof(char *));
462 cwist_free(conn->pending);
463 conn->pending = np;
464 conn->pending_cap = new_cap;
465 }
466 conn->pending[conn->pending_count++] = msg;
467 conn->pending_bytes += msg_len;
468 return true;
469}

471/* Serialize one `next` message for @p op (malloc'd string, or NULL on OOM). */
472static char *gql_ws_build_next(const gql_ws_op_t *op, const cJSON *payload) {
473 char *printed = NULL;
474 cJSON *msg = cJSON_CreateObject();
475 cJSON *pl = cJSON_CreateObject();
476 cJSON *data = cJSON_CreateObject();
477 cJSON *value = payload ? cJSON_Duplicate(payload, true) : cJSON_CreateNull();
478 if (!msg || !pl || !data || !value) goto out;
479 cJSON_AddStringToObject(msg, "type", "next");
480 cJSON_AddStringToObject(msg, "id", op->id);
481 cJSON_AddItemToObject(data, op->field, value);
482 value = NULL;
483 cJSON_AddItemToObject(pl, "data", data);
484 data = NULL;
485 cJSON_AddItemToObject(msg, "payload", pl);
486 pl = NULL;
487 printed = cJSON_PrintUnformatted(msg);
488out:
489 cJSON_Delete(value);
490 cJSON_Delete(data);
491 cJSON_Delete(pl);
492 cJSON_Delete(msg);
493 return printed;
494}

496/* Runs on the reactor thread; drains the pending queue to the wire. */
497static void gql_ws_flush_cb(void *ctx) {
498 gql_ws_flush_t *flush = (gql_ws_flush_t *)ctx;
499 gql_ws_conn_t *conn = flush->conn;
500 char **msgs = NULL;
501 size_t count = 0;
502
503 pthread_mutex_lock(&g_broker_lock);
504 /* Teardown runs on this same thread, so conn->ws is stable here: once
505 * dead, the socket is gone and the queue was already freed. A NULL ws
506 * (attach still finalizing) leaves the queue for a later flush. */
507 if (!conn->dead && conn->ws && conn->pending_count > 0) {
508 count = conn->pending_count;
509 msgs = (char **)cwist_alloc_array(count, sizeof(char *));
510 if (msgs) {
511 memcpy(msgs, conn->pending, count * sizeof(char *));
512 conn->pending_count = 0;
513 conn->pending_bytes = 0;
514 } else {
515 count = 0; /* OOM: leave queued for the next flush */
516 }
517 }
518 conn->flush_pending = false;
519 pthread_mutex_unlock(&g_broker_lock);
520
521 for (size_t i = 0; i < count; i++) {
522 if (cwist_websocket_async_send(conn->ws, CWIST_WS_FRAME_TEXT, (const uint8_t *)msgs[i],
523 strlen(msgs[i])) != 0) {
524 cwist_websocket_async_close(conn->ws);
525 }
526 cwist_free(msgs[i]);
527 }
528 cwist_free(msgs);
529
530 gql_ws_conn_unref(conn);
531 cwist_free(flush);
532}

534/* Queue a flush node for @p conn; takes a reference released by the callback. */
535static void gql_ws_post_flush(gql_ws_conn_t *conn) {
536 pthread_mutex_lock(&g_broker_lock);
537 if (conn->flush_pending) {
538 pthread_mutex_unlock(&g_broker_lock);
539 return;
540 }
541 conn->flush_pending = true;
542 pthread_mutex_unlock(&g_broker_lock);
543
544 gql_ws_flush_t *flush = (gql_ws_flush_t *)cwist_alloc(sizeof(*flush));
545 if (!flush) {
546 pthread_mutex_lock(&g_broker_lock);
547 conn->flush_pending = false;
548 pthread_mutex_unlock(&g_broker_lock);
549 return;
550 }
551 flush->conn = conn;
552 flush->node.cb = gql_ws_flush_cb;
553 flush->node.ctx = flush;
554 gql_ws_conn_ref(conn);
555 if (!cwist_reactor_post(conn->reactor, &flush->node)) {
556 pthread_mutex_lock(&g_broker_lock);
557 conn->flush_pending = false;
558 pthread_mutex_unlock(&g_broker_lock);
559 gql_ws_conn_unref(conn);
560 cwist_free(flush);
561 }
562}

564size_t cwist_graphql_publish(const char *topic, const cJSON *payload) {
565 if (!topic || topic[0] == '\0') return 0;
566
567 gql_ws_conn_t **conns = NULL;
568 size_t conn_count = 0, conn_cap = 0;
569 size_t delivered = 0;
570
571 pthread_mutex_lock(&g_broker_lock);
572 for (gql_ws_op_t *op = g_broker_ops; op; op = op->next) {
573 if (!gql_src_has_topic(op->src, topic)) continue;
574 char *msg = gql_ws_build_next(op, payload);
575 if (!msg) continue;
576 if (!gql_ws_pending_push(op->conn, msg)) {
577 cwist_free(msg);
578 continue;
579 }
580 delivered++;
581 bool seen = false;
582 for (size_t i = 0; i < conn_count; i++) {
583 if (conns[i] == op->conn) {
584 seen = true;
585 break;
586 }
587 }
588 if (!seen) {
589 if (conn_count == conn_cap) {
590 size_t new_cap = conn_cap ? conn_cap * 2 : 4;
591 gql_ws_conn_t **nc = (gql_ws_conn_t **)cwist_alloc_array(new_cap, sizeof(*nc));
592 if (!nc) continue; /* queue stays; drained by an earlier flush */
593 if (conn_count > 0) memcpy(nc, conns, conn_count * sizeof(*nc));
594 cwist_free(conns);
595 conns = nc;
596 conn_cap = new_cap;
597 }
598 conns[conn_count] = op->conn;
599 gql_ws_conn_ref(op->conn);
600 conn_count++;
601 }
602 }
603 pthread_mutex_unlock(&g_broker_lock);
604
605 for (size_t i = 0; i < conn_count; i++) {
606 gql_ws_post_flush(conns[i]); /* adopts the reference */
607 gql_ws_conn_unref(conns[i]); /* release the publish-side reference */
608 }
609 cwist_free(conns);
610 return delivered;
611}

617static void gql_ws_handle_subscribe(cwist_websocket_async *ws, gql_ws_conn_t *conn,
618 const cJSON *msg) {
619 const cJSON *id = cJSON_GetObjectItemCaseSensitive(msg, "id");
620 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(msg, "payload");
621 const cJSON *query =
622 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "query") : NULL;
623 const cJSON *operation_name =
624 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "operationName") : NULL;
625 const cJSON *variables =
626 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "variables") : NULL;
627 const cJSON *extensions =
628 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "extensions") : NULL;
629 if (!cJSON_IsString(id) || !id->valuestring || id->valuestring[0] == '\0' ||
630 !cJSON_IsString(query) || !query->valuestring ||
631 (operation_name && !cJSON_IsNull(operation_name) && !cJSON_IsString(operation_name)) ||
632 (variables && !cJSON_IsNull(variables) && !cJSON_IsObject(variables)) ||
633 (extensions && !cJSON_IsNull(extensions) && !cJSON_IsObject(extensions))) {
634 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
635 return;
636 }
637 const char *op_id = id->valuestring;
638 if (strlen(query->valuestring) > CWIST_GRAPHQL_MAX_QUERY_SIZE) {
639 gql_ws_send_error(ws, op_id, "Subscription query exceeds the size limit");
640 return;
641 }
642
643 pthread_mutex_lock(&g_broker_lock);
644 bool duplicate = false;
645 for (gql_ws_op_t *op = g_broker_ops; op; op = op->next) {
646 if (op->conn == conn && strcmp(op->id, op_id) == 0) {
647 duplicate = true;
648 break;
649 }
650 }
651 pthread_mutex_unlock(&g_broker_lock);
652 if (duplicate) {
653 /* graphql-ws: a duplicate subscriber id is a connection-level
654 * violation, not an operation-level one. */
655 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_DUPLICATE_ID);
656 return;
657 }
658
659 char field[128];
660 cJSON *args = NULL;
661 if (!cwist_graphql_parse_subscription(query->valuestring, variables, field, sizeof(field),
662 &args)) {
663 gql_ws_send_error(ws, op_id, "Expected a 'subscription' operation with one root field");
664 return;
665 }
666
667 gql_sub_field_t *sf = NULL;
668 for (gql_sub_field_t *f = conn->gws->subs; f; f = f->next) {
669 if (strcmp(f->name, field) == 0) {
670 sf = f;
671 break;
672 }
673 }
674 if (!sf) {
675 cJSON_Delete(args);
676 char buf[192];
677 snprintf(buf, sizeof(buf), "Cannot subscribe field '%s' on schema", field);
678 gql_ws_send_error(ws, op_id, buf);
679 return;
680 }
681
682 cwist_graphql_event_source_t *src = sf->fn(args, variables, sf->ctx);
683 cJSON_Delete(args);
684 if (!src) {
685 gql_ws_send_error(ws, op_id, "Subscription resolver rejected the operation");
686 return;
687 }
688
689 gql_ws_op_t *op = (gql_ws_op_t *)cwist_alloc(sizeof(*op));
690 if (!op) {
691 cwist_graphql_event_source_destroy(src);
692 gql_ws_send_error(ws, op_id, "Out of memory");
693 return;
694 }
695 op->id = cwist_strdup(op_id);
696 op->field = cwist_strdup(field);
697 if (!op->id || !op->field) {
698 cwist_free(op->id);
699 cwist_free(op->field);
700 cwist_free(op);
701 cwist_graphql_event_source_destroy(src);
702 gql_ws_send_error(ws, op_id, "Out of memory");
703 return;
704 }
705 op->src = src;
706 op->conn = conn;
707
708 pthread_mutex_lock(&g_broker_lock);
709 op->next = g_broker_ops;
710 g_broker_ops = op;
711 pthread_mutex_unlock(&g_broker_lock);
712}

714static bool gql_ws_optional_record_valid(const cJSON *msg) {
715 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(msg, "payload");
716 return !payload || cJSON_IsNull(payload) || cJSON_IsObject(payload);
717}
718
719static void gql_ws_handle_keepalive(cwist_websocket_async *ws, const cJSON *msg, bool ping) {
720 if (!gql_ws_optional_record_valid(msg)) {
721 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
722 return;
723 }
724 if (!ping) return;
725
726 cJSON *pong = cJSON_CreateObject();
727 if (!pong) {
728 cwist_websocket_async_close(ws);
729 return;
730 }
731 cJSON_AddStringToObject(pong, "type", "pong");
732 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(msg, "payload");
733 if (payload) cJSON_AddItemToObject(pong, "payload", cJSON_Duplicate(payload, true));
734 gql_ws_send_json(ws, pong);
735}
736
737static void gql_ws_dispatch(cwist_websocket_async *ws, gql_ws_conn_t *conn, cJSON *msg) {
738 const cJSON *type = cJSON_GetObjectItemCaseSensitive(msg, "type");
739 if (!cJSON_IsString(type) || !type->valuestring) {
740 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
741 return;
742 }
743 const char *t = type->valuestring;
744
745 if (strcmp(t, "ping") == 0 || strcmp(t, "pong") == 0) {
746 gql_ws_handle_keepalive(ws, msg, strcmp(t, "ping") == 0);
747 return;
748 }
749
750 if (!conn->init_done) {
751 if (strcmp(t, "connection_init") == 0) {
752 if (!gql_ws_optional_record_valid(msg)) {
753 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
754 return;
755 }
756 cwist_reactor_timer_cancel(conn->reactor, &conn->init_timer);
757 conn->init_done = true;
758 cJSON *ack = cJSON_CreateObject();
759 if (!ack) {
760 cwist_websocket_async_close(ws);
761 return;
762 }
763 cJSON_AddStringToObject(ack, "type", "connection_ack");
764 gql_ws_send_json(ws, ack);
765 } else {
766 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_UNAUTHORIZED);
767 }
768 return;
769 }
770 if (strcmp(t, "connection_init") == 0) {
771 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_TOO_MANY_INIT);
772 } else if (strcmp(t, "subscribe") == 0) {
773 gql_ws_handle_subscribe(ws, conn, msg);
774 } else if (strcmp(t, "complete") == 0) {
775 const cJSON *id = cJSON_GetObjectItemCaseSensitive(msg, "id");
776 if (!cJSON_IsString(id) || !id->valuestring) {
777 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
778 return;
779 }
780 pthread_mutex_lock(&g_broker_lock);
781 for (gql_ws_op_t *op = g_broker_ops; op; op = op->next) {
782 if (op->conn == conn && strcmp(op->id, id->valuestring) == 0) {
783 gql_ws_op_unlink(op);
784 gql_ws_op_free(op);
785 break;
786 }
787 }
788 pthread_mutex_unlock(&g_broker_lock);
789 /* A client `complete` gets no response; the operation is simply gone. */
790 } else {
791 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
792 }
793}

799static void gql_ws_init_timeout(void *ctx) {
800 gql_ws_conn_t *conn = ctx;
801 if (!conn->dead && !conn->init_done && conn->ws)
802 cwist_websocket_async_close_code(conn->ws, GQL_WS_CLOSE_INIT_TIMEOUT);
803}
804
805static void gql_ws_arm_init_timeout(void *ctx) {
806 gql_ws_conn_t *conn = ctx;
807 if (!conn->dead && !conn->init_done && conn->ws &&
808 !cwist_reactor_timer_arm(conn->reactor, &conn->init_timer, conn->init_timeout_us))
809 cwist_websocket_async_close_code(conn->ws, GQL_WS_CLOSE_INIT_TIMEOUT);
810 gql_ws_conn_unref(conn);
811}

813static gql_ws_conn_t *gql_ws_conn_create(cwist_graphql_ws_t *gws, cwist_reactor_t *reactor) {
814 gql_ws_conn_t *conn = (gql_ws_conn_t *)cwist_alloc(sizeof(*conn));
815 if (!conn) return NULL;
816 gql_ws_endpoint_ref(gws);
817 conn->gws = gws;
818 conn->ws = NULL;
819 conn->reactor = reactor;
820 atomic_init(&conn->refs, 1); /* owned by the protocol layer until teardown */
821 conn->init_done = false;
822 conn->dead = false;
823 conn->pending = NULL;
824 conn->pending_count = 0;
825 conn->pending_cap = 0;
826 conn->pending_bytes = 0;
827 conn->flush_pending = false;
828 cwist_reactor_timer_init(&conn->init_timer, gql_ws_init_timeout, conn);
829 return conn;
830}

832/* Shared by on_close (teardown) and the attach-failure path: mark dead, purge
833 * broker operations and the pending queue, release the protocol reference.
834 * Posted callbacks still in flight keep the state alive and no-op on dispatch. */
835static void gql_ws_conn_dispose(gql_ws_conn_t *conn) {
836 cwist_reactor_timer_cancel(conn->reactor, &conn->init_timer);
837 pthread_mutex_lock(&g_broker_lock);
838 conn->dead = true;
839 conn->ws = NULL;
840 gql_ws_conn_purge_ops(conn);
841 for (size_t i = 0; i < conn->pending_count; i++) cwist_free(conn->pending[i]);
842 conn->pending_count = 0;
843 conn->pending_bytes = 0;
844 pthread_mutex_unlock(&g_broker_lock);
845 gql_ws_conn_unref(conn);
846}

848/* WS layer teardown hook (reactor thread). */
849static void gql_ws_on_close(cwist_websocket_async *ws, void *user_data) {
850 (void)ws;
851 gql_ws_conn_dispose((gql_ws_conn_t *)user_data);
852}

854static void gql_ws_on_message(cwist_websocket_async *ws, cwist_ws_frame *frame, void *user_data) {
855 gql_ws_conn_t *conn = (gql_ws_conn_t *)user_data;
856 if (frame->opcode == CWIST_WS_FRAME_PING || frame->opcode == CWIST_WS_FRAME_PONG) {
857 cwist_websocket_frame_destroy(frame);
858 return;
859 }
860 if (frame->opcode != CWIST_WS_FRAME_TEXT) {
861 cwist_websocket_frame_destroy(frame);
862 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
863 return;
864 }
865 cJSON *msg = cJSON_ParseWithLength((const char *)frame->payload, frame->payload_len);
866 cwist_websocket_frame_destroy(frame);
867 if (!cJSON_IsObject(msg)) {
868 cJSON_Delete(msg);
869 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
870 return;
871 }
872 gql_ws_dispatch(ws, conn, msg);
873 cJSON_Delete(msg);
874}

876bool cwist_graphql_ws_attach_ex(cwist_graphql_ws_t *gws, int fd, cwist_reactor_t *reactor,
877 const uint8_t *initial, size_t initial_len,
878 uint64_t init_timeout_ms) {
879 if (!gws || fd < 0 || !reactor) return false;
880
881 gql_ws_conn_t *conn = gql_ws_conn_create(gws, reactor);
882 if (!conn) {
883 close(fd);
884 return false;
885 }
886
887 /* on_close releases the protocol state at teardown. */
888 cwist_websocket_async *ws = cwist_websocket_async_attach_ex(
889 fd, reactor, gql_ws_on_message, gql_ws_on_close, conn, initial, initial_len);
890 if (!ws) {
891 /* Attach failed after any inline initial-bytes delivery; no on_close
892 * will ever run, so clean up here. */
893 gql_ws_conn_dispose(conn);
894 return false;
895 }
896
897 pthread_mutex_lock(&g_broker_lock);
898 conn->ws = ws;
899 pthread_mutex_unlock(&g_broker_lock);
900
901 if (init_timeout_ms > 0) {
902 conn->init_timeout_us = init_timeout_ms * 1000;
903 conn->init_post.cb = gql_ws_arm_init_timeout;
904 conn->init_post.ctx = conn;
905 gql_ws_conn_ref(conn);
906 if (!cwist_reactor_post(reactor, &conn->init_post)) {
907 gql_ws_conn_unref(conn);
908 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_INIT_TIMEOUT);
909 }
910 }
911
912 /* Drain anything queued by inline initial-bytes delivery. */
913 gql_ws_post_flush(conn);
914 return true;
915}
