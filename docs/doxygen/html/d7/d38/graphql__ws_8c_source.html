76bool cwist_graphql_event_source_add_topic(cwist_graphql_event_source_t *src, const char *topic) {
77 if (!src || !topic || topic[0] == '\0') return false;
78 for (size_t i = 0; i < src->topic_count; i++) {
79 if (strcmp(src->topics[i], topic) == 0) return true; /* idempotent */
80 }
81 if (src->topic_count == src->topic_cap) {
82 size_t new_cap = src->topic_cap ? src->topic_cap * 2 : 4;
83 char **nt = (char **)cwist_alloc_array(new_cap, sizeof(char *));
84 if (!nt) return false;
85 if (src->topic_count > 0) memcpy(nt, src->topics, src->topic_count * sizeof(char *));
86 cwist_free(src->topics);
87 src->topics = nt;
88 src->topic_cap = new_cap;
89 }
90 char *copy = cwist_strdup(topic);
91 if (!copy) return false;
92 src->topics[src->topic_count++] = copy;
93 return true;
94}

389#define GQL_WS_MAX_PENDING_BYTES (1024u * 1024u)
390
391/* Append a `next` message to the connection's queue (caller holds g_broker_lock). */
392static bool gql_ws_pending_push(gql_ws_conn_t *conn, char *msg) {
393 size_t msg_len = strlen(msg);
394 if (msg_len > GQL_WS_MAX_PENDING_BYTES ||
395 conn->pending_bytes > GQL_WS_MAX_PENDING_BYTES - msg_len)
396 return false;
397 if (conn->pending_count == conn->pending_cap) {
398 size_t new_cap = conn->pending_cap ? conn->pending_cap * 2 : 8;
399 char **np = (char **)cwist_alloc_array(new_cap, sizeof(char *));
400 if (!np) return false;
401 if (conn->pending_count > 0)
402 memcpy(np, conn->pending, conn->pending_count * sizeof(char *));
403 cwist_free(conn->pending);
404 conn->pending = np;
405 conn->pending_cap = new_cap;
406 }
407 conn->pending[conn->pending_count++] = msg;
408 conn->pending_bytes += msg_len;
409 return true;
410}

412/* Serialize one `next` message for @p op (malloc'd string, or NULL on OOM). */
413static char *gql_ws_build_next(const gql_ws_op_t *op, const cJSON *payload) {
414 char *printed = NULL;
415 cJSON *msg = cJSON_CreateObject();
416 cJSON *pl = cJSON_CreateObject();
417 cJSON *data = cJSON_CreateObject();
418 cJSON *value = payload ? cJSON_Duplicate(payload, true) : cJSON_CreateNull();
419 if (!msg || !pl || !data || !value) goto out;
420 cJSON_AddStringToObject(msg, "type", "next");
421 cJSON_AddStringToObject(msg, "id", op->id);
422 cJSON_AddItemToObject(data, op->field, value);
423 value = NULL;
424 cJSON_AddItemToObject(pl, "data", data);
425 data = NULL;
426 cJSON_AddItemToObject(msg, "payload", pl);
427 pl = NULL;
428 printed = cJSON_PrintUnformatted(msg);
429out:
430 cJSON_Delete(value);
431 cJSON_Delete(data);
432 cJSON_Delete(pl);
433 cJSON_Delete(msg);
434 return printed;
435}

437/* Runs on the reactor thread; drains the pending queue to the wire. */
438static void gql_ws_flush_cb(void *ctx) {
439 gql_ws_flush_t *flush = (gql_ws_flush_t *)ctx;
440 gql_ws_conn_t *conn = flush->conn;
441 char **msgs = NULL;
442 size_t count = 0;
443
444 pthread_mutex_lock(&g_broker_lock);
445 /* Teardown runs on this same thread, so conn->ws is stable here: once
446 * dead, the socket is gone and the queue was already freed. A NULL ws
447 * (attach still finalizing) leaves the queue for a later flush. */
448 if (!conn->dead && conn->ws && conn->pending_count > 0) {
449 count = conn->pending_count;
450 msgs = (char **)cwist_alloc_array(count, sizeof(char *));
451 if (msgs) {
452 memcpy(msgs, conn->pending, count * sizeof(char *));
453 conn->pending_count = 0;
454 conn->pending_bytes = 0;
455 } else {
456 count = 0; /* OOM: leave queued for the next flush */
457 }
458 }
459 conn->flush_pending = false;
460 pthread_mutex_unlock(&g_broker_lock);
461
462 for (size_t i = 0; i < count; i++) {
463 if (cwist_websocket_async_send(conn->ws, CWIST_WS_FRAME_TEXT, (const uint8_t *)msgs[i],
464 strlen(msgs[i])) != 0) {
465 cwist_websocket_async_close(conn->ws);
466 }
467 cwist_free(msgs[i]);
468 }
469 cwist_free(msgs);
470
471 gql_ws_conn_unref(conn);
472 cwist_free(flush);
473}

475/* Queue a flush node for @p conn; takes a reference released by the callback. */
476static void gql_ws_post_flush(gql_ws_conn_t *conn) {
477 pthread_mutex_lock(&g_broker_lock);
478 if (conn->flush_pending) {
479 pthread_mutex_unlock(&g_broker_lock);
480 return;
481 }
482 conn->flush_pending = true;
483 pthread_mutex_unlock(&g_broker_lock);
484
485 gql_ws_flush_t *flush = (gql_ws_flush_t *)cwist_alloc(sizeof(*flush));
486 if (!flush) {
487 pthread_mutex_lock(&g_broker_lock);
488 conn->flush_pending = false;
489 pthread_mutex_unlock(&g_broker_lock);
490 return;
491 }
492 flush->conn = conn;
493 flush->node.cb = gql_ws_flush_cb;
494 flush->node.ctx = flush;
495 gql_ws_conn_ref(conn);
496 if (!cwist_reactor_post(conn->reactor, &flush->node)) {
497 pthread_mutex_lock(&g_broker_lock);
498 conn->flush_pending = false;
499 pthread_mutex_unlock(&g_broker_lock);
500 gql_ws_conn_unref(conn);
501 cwist_free(flush);
502 }
503}

505size_t cwist_graphql_publish(const char *topic, const cJSON *payload) {
506 if (!topic || topic[0] == '\0') return 0;
507
508 gql_ws_conn_t **conns = NULL;
509 size_t conn_count = 0, conn_cap = 0;
510 size_t delivered = 0;
511
512 pthread_mutex_lock(&g_broker_lock);
513 for (gql_ws_op_t *op = g_broker_ops; op; op = op->next) {
514 if (!gql_src_has_topic(op->src, topic)) continue;
515 char *msg = gql_ws_build_next(op, payload);
516 if (!msg) continue;
517 if (!gql_ws_pending_push(op->conn, msg)) {
518 cwist_free(msg);
519 continue;
520 }
521 delivered++;
522 bool seen = false;
523 for (size_t i = 0; i < conn_count; i++) {
524 if (conns[i] == op->conn) {
525 seen = true;
526 break;
527 }
528 }
529 if (!seen) {
530 if (conn_count == conn_cap) {
531 size_t new_cap = conn_cap ? conn_cap * 2 : 4;
532 gql_ws_conn_t **nc = (gql_ws_conn_t **)cwist_alloc_array(new_cap, sizeof(*nc));
533 if (!nc) continue; /* queue stays; drained by an earlier flush */
534 if (conn_count > 0) memcpy(nc, conns, conn_count * sizeof(*nc));
535 cwist_free(conns);
536 conns = nc;
537 conn_cap = new_cap;
538 }
539 conns[conn_count] = op->conn;
540 gql_ws_conn_ref(op->conn);
541 conn_count++;
542 }
543 }
544 pthread_mutex_unlock(&g_broker_lock);
545
546 for (size_t i = 0; i < conn_count; i++) {
547 gql_ws_post_flush(conns[i]); /* adopts the reference */
548 gql_ws_conn_unref(conns[i]); /* release the publish-side reference */
549 }
550 cwist_free(conns);
551 return delivered;
552}

558static void gql_ws_handle_subscribe(cwist_websocket_async *ws, gql_ws_conn_t *conn,
559 const cJSON *msg) {
560 const cJSON *id = cJSON_GetObjectItemCaseSensitive(msg, "id");
561 const cJSON *payload = cJSON_GetObjectItemCaseSensitive(msg, "payload");
562 const cJSON *query =
563 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "query") : NULL;
564 const cJSON *operation_name =
565 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "operationName") : NULL;
566 const cJSON *variables =
567 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "variables") : NULL;
568 const cJSON *extensions =
569 cJSON_IsObject(payload) ? cJSON_GetObjectItemCaseSensitive(payload, "extensions") : NULL;
570 if (!cJSON_IsString(id) || !id->valuestring || id->valuestring[0] == '\0' ||
571 !cJSON_IsString(query) || !query->valuestring ||
572 (operation_name && !cJSON_IsNull(operation_name) && !cJSON_IsString(operation_name)) ||
573 (variables && !cJSON_IsNull(variables) && !cJSON_IsObject(variables)) ||
574 (extensions && !cJSON_IsNull(extensions) && !cJSON_IsObject(extensions))) {
575 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
576 return;
577 }
578 const char *op_id = id->valuestring;
579 if (strlen(query->valuestring) > CWIST_GRAPHQL_MAX_QUERY_SIZE) {
580 gql_ws_send_error(ws, op_id, "Subscription query exceeds the size limit");
581 return;
582 }
583
584 pthread_mutex_lock(&g_broker_lock);
585 bool duplicate = false;
586 for (gql_ws_op_t *op = g_broker_ops; op; op = op->next) {
587 if (op->conn == conn && strcmp(op->id, op_id) == 0) {
588 duplicate = true;
589 break;
590 }
591 }
592 pthread_mutex_unlock(&g_broker_lock);
593 if (duplicate) {
594 /* graphql-ws: a duplicate subscriber id is a connection-level
595 * violation, not an operation-level one. */
596 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_DUPLICATE_ID);
597 return;
598 }
599
600 char field[128];
601 cJSON *args = NULL;
602 if (!cwist_graphql_parse_subscription(query->valuestring, variables, field, sizeof(field),
603 &args)) {
604 gql_ws_send_error(ws, op_id, "Expected a 'subscription' operation with one root field");
605 return;
606 }
607
608 gql_sub_field_t *sf = NULL;
609 for (gql_sub_field_t *f = conn->gws->subs; f; f = f->next) {
610 if (strcmp(f->name, field) == 0) {
611 sf = f;
612 break;
613 }
614 }
615 if (!sf) {
616 cJSON_Delete(args);
617 char buf[192];
618 snprintf(buf, sizeof(buf), "Cannot subscribe field '%s' on schema", field);
619 gql_ws_send_error(ws, op_id, buf);
620 return;
621 }
622
623 cwist_graphql_event_source_t *src = sf->fn(args, variables, sf->ctx);
624 cJSON_Delete(args);
625 if (!src) {
626 gql_ws_send_error(ws, op_id, "Subscription resolver rejected the operation");
627 return;
628 }
629
630 gql_ws_op_t *op = (gql_ws_op_t *)cwist_alloc(sizeof(*op));
631 if (!op) {
632 cwist_graphql_event_source_destroy(src);
633 gql_ws_send_error(ws, op_id, "Out of memory");
634 return;
635 }
636 op->id = cwist_strdup(op_id);
637 op->field = cwist_strdup(field);
638 if (!op->id || !op->field) {
639 cwist_free(op->id);
640 cwist_free(op->field);
641 cwist_free(op);
642 cwist_graphql_event_source_destroy(src);
643 gql_ws_send_error(ws, op_id, "Out of memory");
644 return;
645 }
646 op->src = src;
647 op->conn = conn;
648
649 pthread_mutex_lock(&g_broker_lock);
650 op->next = g_broker_ops;
651 g_broker_ops = op;
652 pthread_mutex_unlock(&g_broker_lock);
653}

678static void gql_ws_dispatch(cwist_websocket_async *ws, gql_ws_conn_t *conn, cJSON *msg) {
679 const cJSON *type = cJSON_GetObjectItemCaseSensitive(msg, "type");
680 if (!cJSON_IsString(type) || !type->valuestring) {
681 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
682 return;
683 }
684 const char *t = type->valuestring;
685
686 if (strcmp(t, "ping") == 0 || strcmp(t, "pong") == 0) {
687 gql_ws_handle_keepalive(ws, msg, strcmp(t, "ping") == 0);
688 return;
689 }
690
691 if (!conn->init_done) {
692 if (strcmp(t, "connection_init") == 0) {
693 if (!gql_ws_optional_record_valid(msg)) {
694 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
695 return;
696 }
697 conn->init_done = true;
698 cJSON *ack = cJSON_CreateObject();
699 if (!ack) {
700 cwist_websocket_async_close(ws);
701 return;
702 }
703 cJSON_AddStringToObject(ack, "type", "connection_ack");
704 gql_ws_send_json(ws, ack);
705 } else {
706 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_UNAUTHORIZED);
707 }
708 return;
709 }
710 if (strcmp(t, "connection_init") == 0) {
711 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_TOO_MANY_INIT);
712 } else if (strcmp(t, "subscribe") == 0) {
713 gql_ws_handle_subscribe(ws, conn, msg);
714 } else if (strcmp(t, "complete") == 0) {
715 const cJSON *id = cJSON_GetObjectItemCaseSensitive(msg, "id");
716 if (!cJSON_IsString(id) || !id->valuestring) {
717 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
718 return;
719 }
720 pthread_mutex_lock(&g_broker_lock);
721 for (gql_ws_op_t *op = g_broker_ops; op; op = op->next) {
722 if (op->conn == conn && strcmp(op->id, id->valuestring) == 0) {
723 gql_ws_op_unlink(op);
724 gql_ws_op_free(op);
725 break;
726 }
727 }
728 pthread_mutex_unlock(&g_broker_lock);
729 /* A client `complete` gets no response; the operation is simply gone. */
730 } else {
731 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
732 }
733}

739static gql_ws_conn_t *gql_ws_conn_create(cwist_graphql_ws_t *gws, cwist_reactor_t *reactor) {
740 gql_ws_conn_t *conn = (gql_ws_conn_t *)cwist_alloc(sizeof(*conn));
741 if (!conn) return NULL;
742 gql_ws_endpoint_ref(gws);
743 conn->gws = gws;
744 conn->ws = NULL;
745 conn->reactor = reactor;
746 atomic_init(&conn->refs, 1); /* owned by the protocol layer until teardown */
747 conn->init_done = false;
748 conn->dead = false;
749 conn->pending = NULL;
750 conn->pending_count = 0;
751 conn->pending_cap = 0;
752 conn->pending_bytes = 0;
753 conn->flush_pending = false;
754 return conn;
755}
756
757/* Shared by on_close (teardown) and the attach-failure path: mark dead, purge
758 * broker operations and the pending queue, release the protocol reference.
759 * Flush nodes still in flight keep the state alive and no-op on dispatch. */
760static void gql_ws_conn_dispose(gql_ws_conn_t *conn) {
761 pthread_mutex_lock(&g_broker_lock);
762 conn->dead = true;
763 conn->ws = NULL;
764 gql_ws_conn_purge_ops(conn);
765 for (size_t i = 0; i < conn->pending_count; i++) cwist_free(conn->pending[i]);
766 conn->pending_count = 0;
767 conn->pending_bytes = 0;
768 pthread_mutex_unlock(&g_broker_lock);
769 gql_ws_conn_unref(conn);
770}

772/* WS layer teardown hook (reactor thread). */
773static void gql_ws_on_close(cwist_websocket_async *ws, void *user_data) {
774 (void)ws;
775 gql_ws_conn_dispose((gql_ws_conn_t *)user_data);
776}

778static void gql_ws_on_message(cwist_websocket_async *ws, cwist_ws_frame *frame, void *user_data) {
779 gql_ws_conn_t *conn = (gql_ws_conn_t *)user_data;
780 if (frame->opcode == CWIST_WS_FRAME_PING || frame->opcode == CWIST_WS_FRAME_PONG) {
781 cwist_websocket_frame_destroy(frame);
782 return;
783 }
784 if (frame->opcode != CWIST_WS_FRAME_TEXT) {
785 cwist_websocket_frame_destroy(frame);
786 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
787 return;
788 }
789 cJSON *msg = cJSON_ParseWithLength((const char *)frame->payload, frame->payload_len);
790 cwist_websocket_frame_destroy(frame);
791 if (!cJSON_IsObject(msg)) {
792 cJSON_Delete(msg);
793 cwist_websocket_async_close_code(ws, GQL_WS_CLOSE_BAD_REQUEST);
794 return;
795 }
796 gql_ws_dispatch(ws, conn, msg);
797 cJSON_Delete(msg);
798}

800bool cwist_graphql_ws_attach(cwist_graphql_ws_t *gws, int fd, cwist_reactor_t *reactor,
801 const uint8_t *initial, size_t initial_len) {
802 if (!gws || fd < 0 || !reactor) return false;
803
804 gql_ws_conn_t *conn = gql_ws_conn_create(gws, reactor);
805 if (!conn) {
806 close(fd);
807 return false;
808 }
809
810 /* on_close releases the protocol state at teardown. */
811 cwist_websocket_async *ws = cwist_websocket_async_attach_ex(
812 fd, reactor, gql_ws_on_message, gql_ws_on_close, conn, initial, initial_len);
813 if (!ws) {
814 /* Attach failed after any inline initial-bytes delivery; no on_close
815 * will ever run, so clean up here. */
816 gql_ws_conn_dispose(conn);
817 return false;
818 }
819
820 pthread_mutex_lock(&g_broker_lock);
821 conn->ws = ws;
822 pthread_mutex_unlock(&g_broker_lock);
823
824 /* Drain anything queued by inline initial-bytes delivery. */
825 gql_ws_post_flush(conn);
826 return true;
827}
