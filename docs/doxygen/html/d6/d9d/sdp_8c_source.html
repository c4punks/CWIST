22static void copy_attr_value(const char *line, const char *prefix, char *dst, size_t cap) {
23 size_t plen = strlen(prefix);
24 if (strncmp(line, prefix, plen) != 0)

43int cwist_sdp_parse(const char *sdp, size_t len, cwist_sdp_info *info) {
44 memset(info, 0, sizeof(*info));
45 char line[512];
46 size_t pos = 0;
47 while (pos < len) {
48 size_t n = strcspn(sdp + pos, "\r\n");
49 if (n >= sizeof(line))
50 n = sizeof(line) - 1;
51 memcpy(line, sdp + pos, n);
52 line[n] = '\0';
53 pos += n;
54 while (pos < len && (sdp[pos] == '\r' || sdp[pos] == '\n'))
55 pos++;
56
57 if (strncmp(line, "a=ice-ufrag:", 12) == 0)
58 copy_attr_value(line, "a=ice-ufrag:", info->ice_ufrag, sizeof(info->ice_ufrag));
59 else if (strncmp(line, "a=ice-pwd:", 10) == 0)
60 copy_attr_value(line, "a=ice-pwd:", info->ice_pwd, sizeof(info->ice_pwd));
61 else if (strncmp(line, "a=fingerprint:", 14) == 0)
62 copy_attr_value(line, "a=fingerprint:sha-256 ", info->fingerprint,
63 sizeof(info->fingerprint));
64 else if (strncmp(line, "a=setup:", 8) == 0)
65 copy_attr_value(line, "a=setup:", info->setup, sizeof(info->setup));
66 else if (strncmp(line, "a=mid:", 6) == 0)
67 copy_attr_value(line, "a=mid:", info->mid, sizeof(info->mid));
68 else if (strcmp(line, "a=ice-lite") == 0)
69 info->has_lite = 1;
70 }

72 return -1;
