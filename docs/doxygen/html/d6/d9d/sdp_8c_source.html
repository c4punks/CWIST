19static int copy_attr_value(const char *line, const char *prefix, char *dst, size_t cap) {
20 size_t plen = strlen(prefix);
21 if (strncmp(line, prefix, plen) != 0) return -1;
22 const char *v = line + plen;
23 size_t n = strcspn(v, "\r\n");
24 if (n >= cap) return -1;
25 memcpy(dst, v, n);
26 dst[n] = '\0';
27 return 0;

30static int sdp_parse_port(const char *line, uint16_t *port) {
31 const char *p = line + 12;
32 if (!*p) return -1;
33 unsigned value = 0;
34 for (; *p; p++) {
35 if (*p < '0' || *p > '9') return -1;
36 value = value * 10 + (unsigned)(*p - '0');
37 if (value > 65535) return -1;
38 }
39 if (value == 0) return -1;
40 *port = (uint16_t)value;
41 return 0;
42}

119 if (value[0] && strcmp(value, line + 8) != 0) return -1;

138 }
139 if ((required & (SDP_V | SDP_O | SDP_S | SDP_T)) != (SDP_V | SDP_O | SDP_S | SDP_T) ||
140 info->section_count == 0)
141 return -1;

143 for (size_t i = 0; i < info->section_count; i++) {
144 const cwist_sdp_section *section = &info->sections[i];
145 if (!section->is_data) continue;

147 if (section->ice_ufrag[0])

157 strcmp(info->setup, "actpass") != 0))
158 return -1;
159 return 0;

175static int write_session(char *out, size_t cap, const char *fingerprint, const char *ufrag,
176 const char *pwd, const char *mid, const char *setup, int with_lite,
177 long long sess_id) {
178 int n = snprintf(out, cap,
179 "v=0\r\n"

181 "s=-\r\n"
182 "t=0 0\r\n"
183 "a=group:BUNDLE %s\r\n"
184 "m=application 9 UDP/DTLS/SCTP webrtc-datachannel\r\n"

191 "a=setup:%s\r\n"

211static bool sdp_bundle_has_mid(const char *group, const char *mid) {
212 if (!group || !mid || !*mid) return false;
213 size_t len = strlen(mid);
214 const char *p = group;
215 while (*p) {
216 while (*p == ' ' || *p == '\t') p++;
217 const char *start = p;
218 while (*p && *p != ' ' && *p != '\t') p++;
219 if ((size_t)(p - start) == len && memcmp(start, mid, len) == 0) return true;
220 }
221 return false;

224int cwist_sdp_write_answer_for_offer(char *out, size_t cap, const char *fingerprint,
225 const char *ufrag, const char *pwd,
226 const cwist_sdp_info *offer, const char *host, uint16_t port) {
227 if (!out || !offer || !host || offer->section_count == 0) return -1;
228 unsigned long long session_id;
229 if (RAND_bytes((unsigned char *)&session_id, sizeof(session_id)) != 1) return -1;
230
231 char bundle[64] = "";
232 for (size_t i = 0; i < offer->section_count; i++) {
233 const cwist_sdp_section *section = &offer->sections[i];
234 if (!section->is_data) continue;
235 if (sdp_bundle_has_mid(offer->bundle_group, section->mid)) {
236 size_t used = 0;
237 if (sdp_append(bundle, sizeof(bundle), &used, "a=group:BUNDLE %s\r\n", section->mid) < 0)
238 return -1;
239 }
240 break;
241 }
242
243 size_t off = 0;
244 if (sdp_append(out, cap, &off, "v=0\r\no=- %llu 2 IN IP4 127.0.0.1\r\ns=-\r\nt=0 0\r\n%s",
245 session_id, bundle) < 0)
246 return -1;
247
248 const char *setup = strcasecmp(offer->setup, "passive") == 0 ? "active" : "passive";
249 int accepted = 0;
250 for (size_t i = 0; i < offer->section_count; i++) {
251 const cwist_sdp_section *section = &offer->sections[i];
252 char mid_attribute[40] = "";
253 if (section->mid[0])
254 snprintf(mid_attribute, sizeof(mid_attribute), "a=mid:%s\r\n", section->mid);
255 if (section->is_data && accepted == 0) {
256 if (sdp_append(out, cap, &off,
257 "m=application 9 UDP/DTLS/SCTP webrtc-datachannel\r\n"
258 "c=IN IP4 0.0.0.0\r\n"
259 "%s"
260 "a=ice-ufrag:%s\r\n"
261 "a=ice-pwd:%s\r\n"
262 "a=ice-lite\r\n"
263 "a=fingerprint:sha-256 %s\r\n"
264 "a=setup:%s\r\n"
265 "a=candidate:1 1 UDP 2122260223 %s %u typ host\r\n"
266 "a=end-of-candidates\r\n"
267 "a=sctp-port:%u\r\n"
268 "a=max-message-size:262144\r\n",
269 mid_attribute, ufrag, pwd, fingerprint, setup, host, (unsigned)port,
270 (unsigned)section->sctp_port) < 0)
271 return -1;
272 accepted++;
273 } else if (sdp_append(out, cap, &off, "m=%s 0 %s %s\r\nc=IN IP4 0.0.0.0\r\n%s",
274 section->media, section->proto, section->fmt, mid_attribute) < 0) {
275 return -1;
276 }
277 }
278 return accepted == 1 ? 0 : -1;
279}
