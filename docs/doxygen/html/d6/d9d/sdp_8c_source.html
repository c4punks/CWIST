19static int copy_attr_value(const char *line, const char *prefix, char *dst, size_t cap) {
20 size_t plen = strlen(prefix);
21 if (strncmp(line, prefix, plen) != 0) return -1;
22 const char *v = line + plen;
23 size_t n = strcspn(v, "\r\n");
24 if (n >= cap) return -1;
25 memcpy(dst, v, n);
26 dst[n] = '\0';
27 return 0;
28}

39int cwist_sdp_parse(const char *sdp, size_t len, cwist_sdp_info *info) {
40 if (!sdp || !info) return -1;
41 memset(info, 0, sizeof(*info));
42 char line[512];
43 size_t pos = 0;
44 int current = -1;
45 while (pos < len) {
46 size_t line_len = 0;
47 while (pos + line_len < len && sdp[pos + line_len] != '\r' && sdp[pos + line_len] != '\n')
48 line_len++;
49 if (line_len >= sizeof(line)) return -1;
50 memcpy(line, sdp + pos, line_len);
51 line[line_len] = '\0';
52 pos += line_len;
53 while (pos < len && (sdp[pos] == '\r' || sdp[pos] == '\n')) pos++;
54
55 if (strncmp(line, "m=", 2) == 0) {
56 if (info->section_count == CWIST_SDP_MAX_SECTIONS) return -1;
57 current = (int)info->section_count++;
58 cwist_sdp_section *section = &info->sections[current];
59 if (sscanf(line, "m=%15s %*s %31s %31s", section->media, section->proto,
60 section->fmt) != 3)
61 return -1;
62 section->is_data = strcmp(section->media, "application") == 0 &&
63 strcmp(section->proto, "UDP/DTLS/SCTP") == 0 &&
64 strcmp(section->fmt, "webrtc-datachannel") == 0;
65 } else if (strncmp(line, "a=ice-ufrag:", 12) == 0) {
66 char *value = current >= 0 ? info->sections[current].ice_ufrag : info->ice_ufrag;
67 if (copy_attr_value(line, "a=ice-ufrag:", value, CWIST_WEBRTC_ICE_CREDENTIAL_CAP) < 0)
68 return -1;
69 } else if (strncmp(line, "a=ice-pwd:", 10) == 0) {
70 char *value = current >= 0 ? info->sections[current].ice_pwd : info->ice_pwd;
71 if (copy_attr_value(line, "a=ice-pwd:", value, CWIST_WEBRTC_ICE_CREDENTIAL_CAP) < 0)
72 return -1;
73 } else if (strncmp(line, "a=fingerprint:", 14) == 0) {
74 const char *value = line + 14;
75 while (*value == ' ' || *value == '\t') value++;
76 const char *algorithm_end = strchr(value, ' ');
77 if (!algorithm_end) return -1;
78 if ((size_t)(algorithm_end - value) == 7 && strncasecmp(value, "sha-256", 7) == 0) {
79 value = algorithm_end + 1;
80 while (*value == ' ' || *value == '\t') value++;
81 char *fingerprint =
82 current >= 0 ? info->sections[current].fingerprint : info->fingerprint;
83 if (copy_attr_value(value, "", fingerprint, 128) < 0) return -1;
84 }
85 } else if (strncmp(line, "a=setup:", 8) == 0) {
86 char *value = current >= 0 ? info->sections[current].setup : info->setup;
87 if (copy_attr_value(line, "a=setup:", value, 16) < 0) return -1;
88 } else if (strncmp(line, "a=mid:", 6) == 0) {
89 char mid[16];
90 if (copy_attr_value(line, "a=mid:", mid, sizeof(mid)) < 0) return -1;
91 if (current >= 0) {
92 cwist_sdp_section *section = &info->sections[current];
93 snprintf(section->mid, sizeof(section->mid), "%s", mid);
94 if (section->is_data && info->mid[0] == '\0')
95 snprintf(info->mid, sizeof(info->mid), "%s", mid);
96 }
97 } else if (strcmp(line, "a=ice-lite") == 0) {
98 info->has_lite = 1;
99 }
100 }
101 for (size_t i = 0; i < info->section_count; i++) {
102 const cwist_sdp_section *section = &info->sections[i];
103 if (!section->is_data) continue;
104 if (section->ice_ufrag[0])
105 memcpy(info->ice_ufrag, section->ice_ufrag, sizeof(info->ice_ufrag));
106 if (section->ice_pwd[0]) memcpy(info->ice_pwd, section->ice_pwd, sizeof(info->ice_pwd));
107 if (section->fingerprint[0])
108 memcpy(info->fingerprint, section->fingerprint, sizeof(info->fingerprint));
109 if (section->setup[0]) memcpy(info->setup, section->setup, sizeof(info->setup));
110 break;
111 }
112 if (info->ice_ufrag[0] == '\0' || info->ice_pwd[0] == '\0') return -1;
113 return 0;
114}

129static int write_session(char *out, size_t cap, const char *fingerprint, const char *ufrag,
130 const char *pwd, const char *mid, const char *setup, int with_lite,
131 long long sess_id) {
132 int n = snprintf(out, cap,
133 "v=0\r\n"

135 "s=-\r\n"
136 "t=0 0\r\n"
137 "a=group:BUNDLE %s\r\n"
138 "m=application 9 UDP/DTLS/SCTP webrtc-datachannel\r\n"
139 "c=IN IP4 0.0.0.0\r\n"
140 "a=mid:%s\r\n"
141 "a=ice-ufrag:%s\r\n"
142 "a=ice-pwd:%s\r\n"
143 "%s"
144 "a=fingerprint:sha-256 %s\r\n"
145 "a=setup:%s\r\n"
146 "a=sctp-port:5000\r\n"
147 "a=max-message-size:262144\r\n",
148 sess_id, mid, mid, ufrag, pwd, with_lite ? "a=ice-lite\r\n" : "", fingerprint,
149 setup);
150 if (n < 0 || (size_t)n >= cap) return -1;
151 return 0;

155static int sdp_append(char *out, size_t cap, size_t *offset, const char *fmt, ...) {
156 va_list ap;
157 va_start(ap, fmt);
158 int n = vsnprintf(out + *offset, cap - *offset, fmt, ap);
159 va_end(ap);
160 if (n < 0 || (size_t)n >= cap - *offset) return -1;
161 *offset += (size_t)n;
162 return 0;
163}

165int cwist_sdp_write_answer_for_offer(char *out, size_t cap, const char *fingerprint,
166 const char *ufrag, const char *pwd,
167 const cwist_sdp_info *offer, const char *host, uint16_t port) {
168 if (!out || !offer || !host || offer->section_count == 0) return -1;
169 unsigned long long session_id;
170 if (RAND_bytes((unsigned char *)&session_id, sizeof(session_id)) != 1) return -1;
171
172 /* Only the accepted data section is bundled: an m-section rejected in
173 * the answer (port 0) must not be in its BUNDLE group (RFC 8843 section
174 * 7.3.3), and libwebrtc refuses such an answer (#993). */
175 char bundle[CWIST_SDP_MAX_SECTIONS * 20];
176 bundle[0] = '\0';
177 for (size_t i = 0; i < offer->section_count; i++) {
178 if (!offer->sections[i].is_data) continue;
179 char fallback[sizeof(size_t) * 3 + 1];
180 const char *mid = offer->sections[i].mid;
181 if (!mid[0]) {
182 snprintf(fallback, sizeof(fallback), "%zu", i);
183 mid = fallback;
184 }
185 size_t used = 0;
186 if (sdp_append(bundle, sizeof(bundle), &used, "%s", mid) < 0) return -1;
187 break;
188 }
189
190 size_t off = 0;
191 if (sdp_append(out, cap, &off,
192 "v=0\r\no=- %llu 2 IN IP4 127.0.0.1\r\ns=-\r\nt=0 0\r\na=group:BUNDLE %s\r\n",
193 session_id, bundle) < 0)
194 return -1;
195
196 const char *setup = strcasecmp(offer->setup, "passive") == 0 ? "active" : "passive";
197 int accepted = 0;
198 for (size_t i = 0; i < offer->section_count; i++) {
199 const cwist_sdp_section *section = &offer->sections[i];
200 char fallback[sizeof(size_t) * 3 + 1];
201 const char *mid = section->mid;
202 if (!mid[0]) {
203 snprintf(fallback, sizeof(fallback), "%zu", i);
204 mid = fallback;
205 }
206 if (section->is_data && accepted == 0) {
207 if (sdp_append(out, cap, &off,
208 "m=application 9 UDP/DTLS/SCTP webrtc-datachannel\r\n"
209 "c=IN IP4 0.0.0.0\r\n"
210 "a=mid:%s\r\n"
211 "a=ice-ufrag:%s\r\n"
212 "a=ice-pwd:%s\r\n"
213 "a=ice-lite\r\n"
214 "a=fingerprint:sha-256 %s\r\n"
215 "a=setup:%s\r\n"
216 "a=candidate:1 1 UDP 2122260223 %s %u typ host\r\n"
217 "a=end-of-candidates\r\n"
218 "a=sctp-port:5000\r\n"
219 "a=max-message-size:262144\r\n",
220 mid, ufrag, pwd, fingerprint, setup, host, (unsigned)port) < 0)
221 return -1;
222 accepted++;
223 } else if (sdp_append(out, cap, &off, "m=%s 0 %s %s\r\nc=IN IP4 0.0.0.0\r\na=mid:%s\r\n",
224 section->media, section->proto, section->fmt, mid) < 0) {
225 return -1;
226 }
227 }
228 return accepted == 1 ? 0 : -1;
229}
