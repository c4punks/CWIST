50static size_t write_callback(void *contents, size_t size, size_t nmemb, void *userp) {
51 size_t total = size * nmemb;
52 response_body_t *body = (response_body_t *)userp;
53 /* Refuse bodies larger than the server-side cap to prevent an OOM DoS
54 * from a malicious or misbehaving server. */
55 if (body->size + total > CWIST_HTTP_MAX_BODY_SIZE) return 0;
56 size_t need = body->size + total;
57 if (need > body->cap) {
58 size_t new_cap = body->cap ? body->cap * 2 : 4096;
59 while (new_cap < need) new_cap *= 2;
60 char *tmp = realloc(body->data, new_cap);
61 if (!tmp) return 0;
62 body->data = tmp;
63 body->cap = new_cap;
64 }
65 memcpy(body->data + body->size, contents, total);
66 body->size += total;
67 return total;
68}

86static size_t header_callback(char *ptr, size_t size, size_t nmemb, void *userp) {
87 size_t total = size * nmemb;
88 response_headers_t *rh = (response_headers_t *)userp;
89
90 /* libcurl sends headers as "Name: value\r\n" or "HTTP/1.1 200 OK\r\n" */
91 if (total > 5 && strncasecmp(ptr, "HTTP/", 5) == 0) {
92 /* Status line: parse status code */
93 char *p = ptr;
94 while (*p && !isspace((unsigned char)*p)) p++;
95 while (*p && isspace((unsigned char)*p)) p++;
96 rh->status_code = strtol(p, NULL, 10);
97 /* Each status line starts a new response: a followed redirect hop
98 * or a 1xx interim reply. Keep only the final response's headers,
99 * and apply the header cap per response. */
100 cwist_http_header_free_all(rh->headers);
101 rh->headers = NULL;
102 rh->header_count = 0;
103 return total;
104 }
105
106 /* Guard against header-count exhaustion attacks. */
107 if (rh->header_count >= 200) return 0;
108
109 /* Find colon separator */
110 char *colon = memchr(ptr, ':', total);
111 if (!colon) return total;
112
113 size_t name_len = (size_t)(colon - ptr);
114 if (name_len == 0) return total;
115
116 char *value = colon + 1;
117 size_t value_len = total - name_len - 1;
118
119 /* Trim leading spaces in value */
120 while (value_len > 0 && isspace((unsigned char)*value)) {
121 value++;
122 value_len--;
123 }
124 /* Trim trailing whitespace (\r\n) */
125 while (value_len > 0 && isspace((unsigned char)value[value_len - 1])) {
126 value_len--;
127 }
128
129 if (name_len > 0 && value_len > 0) {
130 cwist_scratch_t name_s CWIST_SCRATCH_DEFER = {0};
131 char *name = cwist_scratch_alloc(&name_s, name_len + 1);
132 if (!name) return total;
133 memcpy(name, ptr, name_len);
134 name[name_len] = '\0';
135
136 cwist_scratch_t val_s CWIST_SCRATCH_DEFER = {0};
137 char *val = cwist_scratch_alloc(&val_s, value_len + 1);
138 if (!val) return total;
139 memcpy(val, value, value_len);
140 val[value_len] = '\0';
141
142 cwist_http_header_node *node = cwist_alloc(sizeof(*node));
143 if (!node) return total;
144 node->key = cwist_sstring_create();
145 node->value = cwist_sstring_create();
146 if (!node->key || !node->value) {
147 cwist_sstring_destroy(node->key);
148 cwist_sstring_destroy(node->value);
149 cwist_free(node);
150 return total;
151 }
152 cwist_sstring_assign(node->key, name);
153 cwist_sstring_assign(node->value, val);
154 node->next = rh->headers;
155 rh->headers = node;
156 rh->header_count++;
157 }
158
159 return total;
160}

336cwist_error_t cwist_http_client_request(cwist_http_client *client, const char *url,
337 cwist_http_method_t method, cwist_http_header_node *headers,
338 const char *body, size_t body_len,
339 cwist_http_response **out_response) {
340 cwist_error_t err = make_error(CWIST_ERR_INT16);
341 if (!client || !client->curl || !url || !out_response) {
342 err.error.err_i16 = -1;
343 return err;
344 }
345
346 *out_response = NULL;
347
348 CURL *curl = client->curl;
349
350 /* Reset handle for reuse */
351 curl_easy_reset(curl);
352
353 /* Re-apply persistent settings after reset */
354 curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, client->follow_redirects ? 1L : 0L);
355 curl_easy_setopt(curl, CURLOPT_TIMEOUT_MS, (long)client->timeout_ms);
356 curl_easy_setopt(curl, CURLOPT_TCP_KEEPALIVE, 1L);
357 curl_easy_setopt(curl, CURLOPT_ACCEPT_ENCODING, "");
358 curl_easy_setopt(curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");
359 if (client->ca_bundle) {
360 curl_easy_setopt(curl, CURLOPT_CAINFO, client->ca_bundle);
361 }
362 if (client->altsvc_enabled) {
363 curl_easy_setopt(curl, CURLOPT_ALTSVC_CTRL,
364 (long)(CURLALTSVC_H1 | CURLALTSVC_H2 | CURLALTSVC_H3));
365 curl_easy_setopt(curl, CURLOPT_ALTSVC, client->altsvc_db);
366 }
367
368 /* URL */
369 curl_easy_setopt(curl, CURLOPT_URL, url);
370
371 /* Method */
372 const char *method_str = cwist_http_method_to_string(method);
373 if (strcmp(method_str, "GET") == 0) {
374 curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L);
375 } else if (strcmp(method_str, "POST") == 0) {
376 curl_easy_setopt(curl, CURLOPT_POST, 1L);
377 } else if (strcmp(method_str, "PUT") == 0) {
378 curl_easy_setopt(curl, CURLOPT_UPLOAD, 0L);
379 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PUT");
380 } else if (strcmp(method_str, "DELETE") == 0) {
381 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "DELETE");
382 } else if (strcmp(method_str, "PATCH") == 0) {
383 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PATCH");
384 } else if (strcmp(method_str, "HEAD") == 0) {
385 curl_easy_setopt(curl, CURLOPT_NOBODY, 1L);
386 } else if (strcmp(method_str, "OPTIONS") == 0) {
387 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "OPTIONS");
388 } else {
389 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method_str);
390 }
391
392 /* Request headers */
393 struct curl_slist *curl_headers = NULL;
394 cwist_http_header_node *node = headers;
395 while (node) {
396 if (node->key && node->key->data && node->value && node->value->data) {
397 char header_line[4096];
398 int n = snprintf(header_line, sizeof(header_line), "%s: %s", node->key->data,
399 node->value->data);
400 if (n > 0 && (size_t)n < sizeof(header_line)) {
401 curl_headers = curl_slist_append(curl_headers, header_line);
402 }
403 }
404 node = node->next;
405 }
406 if (curl_headers) {
407 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, curl_headers);
408 }
409
410 /* Request body */
411 if (body && body_len > 0) {
412 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE_LARGE, (curl_off_t)body_len);
413 curl_easy_setopt(curl, CURLOPT_COPYPOSTFIELDS, body);
414 }
415
416 /* Response accumulators */
417 response_body_t resp_body = {0};
418 response_headers_t resp_hdrs = {0};
419
420 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
421 curl_easy_setopt(curl, CURLOPT_WRITEDATA, &resp_body);
422 curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, header_callback);
423 curl_easy_setopt(curl, CURLOPT_HEADERDATA, &resp_hdrs);
424
425 /* Prefer HTTP/2, allow downgrade to HTTP/1.1 */
426 curl_easy_setopt(curl, CURLOPT_HTTP_VERSION, CURL_HTTP_VERSION_2_0);
427
428 /* Perform */
429 CURLcode res = curl_easy_perform(curl);
430
431 /* Clean up request headers */
432 if (curl_headers) {
433 curl_slist_free_all(curl_headers);
434 }
435
436 if (res != CURLE_OK) {
437 free(resp_body.data);
438 cwist_http_header_free_all(resp_hdrs.headers);
439 err.error.err_i16 = -1;
440 return err;
441 }
442
443 /* Build response object */
444 cwist_http_response *response = cwist_http_response_create();
445 if (!response) {
446 free(resp_body.data);
447 cwist_http_header_free_all(resp_hdrs.headers);
448 err.error.err_i16 = -1;
449 return err;
450 }
451
452 response->status_code = (cwist_http_status_t)resp_hdrs.status_code;
453 if (resp_body.data && resp_body.size > 0) {
454 cwist_sstring_assign_len(response->body, resp_body.data, resp_body.size);
455 }
456
457 /* Transfer collected headers */
458 response->headers = resp_hdrs.headers;
459
460 /* Populate status text based on code */
461 switch (response->status_code) {
462 case CWIST_HTTP_OK: cwist_sstring_assign(response->status_text, "OK"); break;
463 case CWIST_HTTP_CREATED: cwist_sstring_assign(response->status_text, "Created"); break;
464 case CWIST_HTTP_NO_CONTENT:
465 cwist_sstring_assign(response->status_text, "No Content");
466 break;
467 case CWIST_HTTP_BAD_REQUEST:
468 cwist_sstring_assign(response->status_text, "Bad Request");
469 break;
470 case CWIST_HTTP_UNAUTHORIZED:
471 cwist_sstring_assign(response->status_text, "Unauthorized");
472 break;
473 case CWIST_HTTP_FORBIDDEN: cwist_sstring_assign(response->status_text, "Forbidden"); break;
474 case CWIST_HTTP_NOT_FOUND: cwist_sstring_assign(response->status_text, "Not Found"); break;
475 case CWIST_HTTP_INTERNAL_ERROR:
476 cwist_sstring_assign(response->status_text, "Internal Server Error");
477 break;
478 case CWIST_HTTP_NOT_IMPLEMENTED:
479 cwist_sstring_assign(response->status_text, "Not Implemented");
480 break;
481 case CWIST_HTTP_SERVICE_UNAVAILABLE:
482 cwist_sstring_assign(response->status_text, "Service Unavailable");
483 break;
484 default: cwist_sstring_assign(response->status_text, ""); break;
485 }
486
487 free(resp_body.data);
488 *out_response = response;
489 err.error.err_i16 = 0;
490 return err;
491}
