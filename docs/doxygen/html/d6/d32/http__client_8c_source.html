58static size_t write_callback(void *contents, size_t size, size_t nmemb, void *userp) {
59 if (size && nmemb > SIZE_MAX / size) return 0;
60 size_t total = size * nmemb;
61 if (total == 0) return 0;
62 response_body_t *body = (response_body_t *)userp;
63 /* Refuse bodies larger than the server-side cap to prevent an OOM DoS
64 * from a malicious or misbehaving server. */
65 if (total > CWIST_HTTP_MAX_BODY_SIZE - body->size) return 0;
66 size_t need = body->size + total;
67 if (need > body->cap) {
68 size_t new_cap = body->cap ? body->cap * 2 : 4096;
69 while (new_cap < need) new_cap *= 2;
70 char *tmp = realloc(body->data, new_cap);
71 if (!tmp) return 0;
72 body->data = tmp;
73 body->cap = new_cap;
74 }
75 memcpy(body->data + body->size, contents, total);
76 body->size += total;
77 return total;
78}

96static size_t header_callback(char *ptr, size_t size, size_t nmemb, void *userp) {
97 size_t total = size * nmemb;
98 response_headers_t *rh = (response_headers_t *)userp;
99
100 /* libcurl sends headers as "Name: value\r\n" or "HTTP/1.1 200 OK\r\n" */
101 if (total > 5 && strncasecmp(ptr, "HTTP/", 5) == 0) {
102 /* Status line: parse the three-digit status code within this buffer. */
103 size_t pos = 0;
104 while (pos < total && !isspace((unsigned char)ptr[pos])) pos++;
105 while (pos < total && isspace((unsigned char)ptr[pos])) pos++;
106 long status_code = 0;
107 for (int i = 0; i < 3 && pos < total && isdigit((unsigned char)ptr[pos]); i++, pos++)
108 status_code = status_code * 10 + (ptr[pos] - '0');
109 rh->status_code = status_code;
110 /* Each status line starts a new response: a followed redirect hop
111 * or a 1xx interim reply. Keep only the final response's headers,
112 * and apply the header cap per response. */
113 cwist_http_header_free_all(rh->headers);
114 rh->headers = NULL;
115 rh->header_count = 0;
116 return total;
117 }
118
119 /* Guard against header-count exhaustion attacks. */
120 if (rh->header_count >= 200) return 0;
121
122 /* Find colon separator */
123 char *colon = memchr(ptr, ':', total);
124 if (!colon) return total;
125
126 size_t name_len = (size_t)(colon - ptr);
127 if (name_len == 0) return total;
128
129 char *value = colon + 1;
130 size_t value_len = total - name_len - 1;
131
132 /* Trim leading spaces in value */
133 while (value_len > 0 && isspace((unsigned char)*value)) {
134 value++;
135 value_len--;
136 }
137 /* Trim trailing whitespace (\r\n) */
138 while (value_len > 0 && isspace((unsigned char)value[value_len - 1])) {
139 value_len--;
140 }
141
142 cwist_scratch_t name_s CWIST_SCRATCH_DEFER = {0};
143 char *name = cwist_scratch_alloc(&name_s, name_len + 1);
144 if (!name) {
145 rh->oom = true;
146 return 0;
147 }
148 memcpy(name, ptr, name_len);
149 name[name_len] = '\0';
150
151 cwist_scratch_t val_s CWIST_SCRATCH_DEFER = {0};
152 char *val = cwist_scratch_alloc(&val_s, value_len + 1);
153 if (!val) {
154 rh->oom = true;
155 return 0;
156 }
157 memcpy(val, value, value_len);
158 val[value_len] = '\0';
159
160 cwist_http_header_node *node = cwist_alloc(sizeof(*node));
161 if (!node) {
162 rh->oom = true;
163 return 0;
164 }
165 node->key = cwist_sstring_create();
166 node->value = cwist_sstring_create();
167 if (!node->key || !node->value) {
168 cwist_sstring_destroy(node->key);
169 cwist_sstring_destroy(node->value);
170 cwist_free(node);
171 rh->oom = true;
172 return 0;
173 }
174 cwist_error_t key_err = cwist_sstring_assign(node->key, name);
175 bool key_ok = cwist_error_is_ok(&key_err);
176 cwist_error_dispose(&key_err);
177 cwist_error_t value_err = cwist_sstring_assign(node->value, val);
178 bool value_ok = cwist_error_is_ok(&value_err);
179 cwist_error_dispose(&value_err);
180 if (!key_ok || !value_ok) {
181 cwist_sstring_destroy(node->key);
182 cwist_sstring_destroy(node->value);
183 cwist_free(node);
184 rh->oom = true;
185 return 0;
186 }
187 node->next = rh->headers;
188 rh->headers = node;
189 rh->header_count++;
190
191 return total;
192}

221cwist_http_client *cwist_http_client_create(void) {
222 cwist_curl_global_acquire();
223
224 cwist_http_client *client = cwist_alloc(sizeof(*client));
225 if (!client) {
226 cwist_curl_global_release();
227 return NULL;
228 }
229
230 client->curl = curl_easy_init();
231 if (!client->curl) {
232 cwist_free(client);
233 cwist_curl_global_release();
234 return NULL;
235 }

237 curl_easy_cleanup(client->curl);
238 cwist_free(client);
239 cwist_curl_global_release();
240 return NULL;

243 client->follow_redirects = 1;
244 client->timeout_ms = 30000;
245 client->ca_bundle = NULL;
246 client->altsvc_enabled = 0;
247 client->altsvc_db = NULL;
248 client->file_root = NULL;
249
250 curl_easy_setopt(client->curl, CURLOPT_FOLLOWLOCATION, 1L);
251 curl_easy_setopt(client->curl, CURLOPT_MAXREDIRS, 10L);
252 curl_easy_setopt(client->curl, CURLOPT_TIMEOUT_MS, 30000L);
253 curl_easy_setopt(client->curl, CURLOPT_TCP_KEEPALIVE, 1L);
254 curl_easy_setopt(client->curl, CURLOPT_ACCEPT_ENCODING, "");
255 curl_easy_setopt(client->curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");

468static cwist_error_t http_client_request_locked(cwist_http_client *client, const char *url,
469 cwist_http_method_t method,
470 cwist_http_header_node *headers, const char *body,
471 size_t body_len,
472 cwist_http_response **out_response) {
473 cwist_error_t err = make_error(CWIST_ERR_INT16);
474 if (!client || !client->curl || !url || !out_response) {
475 err.error.err_i16 = -1;
476 return err;
477 }
478
479 *out_response = NULL;
480
481 if (strncasecmp(url, "file://", 7) == 0) {
482 err.error.err_i16 = cwist_http_client_file_request(url, method, client->timeout_ms,
483 client->file_root, out_response);
484 return err;
485 }
486
487 CURL *curl = client->curl;
488
489 /* Reset handle for reuse */
490 curl_easy_reset(curl);
491
492 /* Re-apply persistent settings after reset */
493 curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, client->follow_redirects ? 1L : 0L);
494 curl_easy_setopt(curl, CURLOPT_MAXREDIRS, 10L);
495#if LIBCURL_VERSION_NUM >= 0x075500 /* CURLOPT_PROTOCOLS_STR is an enum, not a macro */
496 curl_easy_setopt(curl, CURLOPT_PROTOCOLS_STR, "http,https");
497 curl_easy_setopt(curl, CURLOPT_REDIR_PROTOCOLS_STR, "http,https");
498#else
499 curl_easy_setopt(curl, CURLOPT_PROTOCOLS, (long)(CURLPROTO_HTTP | CURLPROTO_HTTPS));
500 curl_easy_setopt(curl, CURLOPT_REDIR_PROTOCOLS, (long)(CURLPROTO_HTTP | CURLPROTO_HTTPS));
501#endif
502 curl_easy_setopt(curl, CURLOPT_TIMEOUT_MS, (long)client->timeout_ms);
503 curl_easy_setopt(curl, CURLOPT_TCP_KEEPALIVE, 1L);
504 curl_easy_setopt(curl, CURLOPT_ACCEPT_ENCODING, "");
505 curl_easy_setopt(curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");
506 if (client->ca_bundle) {
507 curl_easy_setopt(curl, CURLOPT_CAINFO, client->ca_bundle);
508 }
509 if (client->altsvc_enabled) {
510 curl_easy_setopt(curl, CURLOPT_ALTSVC_CTRL,
511 (long)(CURLALTSVC_H1 | CURLALTSVC_H2 | CURLALTSVC_H3));
512 curl_easy_setopt(curl, CURLOPT_ALTSVC, client->altsvc_db);
513 }
514
515 /* URL */
516 curl_easy_setopt(curl, CURLOPT_URL, url);
517
518 /* Method */
519 const char *method_str = cwist_http_method_to_string(method);
520 if (strcmp(method_str, "GET") == 0) {
521 curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L);
522 } else if (strcmp(method_str, "POST") == 0) {
523 curl_easy_setopt(curl, CURLOPT_POST, 1L);
524 } else if (strcmp(method_str, "PUT") == 0) {
525 curl_easy_setopt(curl, CURLOPT_UPLOAD, 0L);
526 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PUT");
527 } else if (strcmp(method_str, "DELETE") == 0) {
528 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "DELETE");
529 } else if (strcmp(method_str, "PATCH") == 0) {
530 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PATCH");
531 } else if (strcmp(method_str, "HEAD") == 0) {
532 curl_easy_setopt(curl, CURLOPT_NOBODY, 1L);
533 } else if (strcmp(method_str, "OPTIONS") == 0) {
534 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "OPTIONS");
535 } else {
536 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method_str);
537 }
538
539 /* Request headers */
540 struct curl_slist *curl_headers = NULL;
541 cwist_http_header_node *node = headers;
542 while (node) {
543 if (node->key && node->key->data && node->value && node->value->data) {
544 /* libcurl drops "Name: " (nothing after the colon) as a request
545 * to remove the header; "Name;" is its syntax for an empty value. */
546 bool empty = node->value->data[strspn(node->value->data, " \t")] == '\0';
547 const char *value = empty ? "" : node->value->data;
548 const char *sep = empty ? ";" : ": ";
549 int needed = snprintf(NULL, 0, "%s%s%s", node->key->data, sep, value);
550 char *header_line = needed >= 0 ? cwist_alloc((size_t)needed + 1) : NULL;
551 if (!header_line) {
552 curl_slist_free_all(curl_headers);
553 err.error.err_i16 = -1;
554 return err;
555 }
556 snprintf(header_line, (size_t)needed + 1, "%s%s%s", node->key->data, sep, value);
557 struct curl_slist *appended = curl_slist_append(curl_headers, header_line);
558 cwist_free(header_line);
559 if (!appended) {
560 curl_slist_free_all(curl_headers);
561 err.error.err_i16 = -1;
562 return err;
563 }
564 curl_headers = appended;
565 }
566 node = node->next;
567 }
568 if (curl_headers) {
569 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, curl_headers);
570 /* libcurl cannot filter arbitrary custom headers per redirect origin.
571 * Do not forward caller credentials or other custom metadata to a
572 * redirect target; header-free requests retain automatic redirects. */
573 curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, 0L);
574 }
575
576 /* Request body */
577 if (body && body_len > 0) {
578 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE_LARGE, (curl_off_t)body_len);
579 curl_easy_setopt(curl, CURLOPT_COPYPOSTFIELDS, body);
580 } else if (strcmp(method_str, "POST") == 0) {
581 /* Without POSTFIELDS libcurl reads the POST body from its default
582 * read callback, which is stdin (#955). Send an empty body. */
583 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE_LARGE, (curl_off_t)0);
584 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, "");
585 }
586 if (strcmp(method_str, "GET") == 0 && body && body_len > 0)
587 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "GET");
588 if (strcmp(method_str, "HEAD") == 0) curl_easy_setopt(curl, CURLOPT_NOBODY, 1L);
589
590 /* Response accumulators */
591 response_body_t resp_body = {0};
592 response_headers_t resp_hdrs = {0};
593
594 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
595 curl_easy_setopt(curl, CURLOPT_WRITEDATA, &resp_body);
596 curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, header_callback);
597 curl_easy_setopt(curl, CURLOPT_HEADERDATA, &resp_hdrs);
598
599 /* Prefer HTTP/2, allow downgrade to HTTP/1.1 */
600 curl_easy_setopt(curl, CURLOPT_HTTP_VERSION, CURL_HTTP_VERSION_2_0);
601
602 /* Perform */
603 CURLcode res = curl_easy_perform(curl);
604
605 /* Clean up request headers */
606 if (curl_headers) {
607 curl_slist_free_all(curl_headers);
608 }
609
610 if (res != CURLE_OK || resp_hdrs.oom) {
611 free(resp_body.data);
612 cwist_http_header_free_all(resp_hdrs.headers);
613 err.error.err_i16 = -1;
614 return err;
615 }
616
617 /* Build response object */
618 cwist_http_response *response = cwist_http_response_create();
619 if (!response) {
620 free(resp_body.data);
621 cwist_http_header_free_all(resp_hdrs.headers);
622 err.error.err_i16 = -1;
623 return err;
624 }
625
626 response->status_code = (cwist_http_status_t)resp_hdrs.status_code;
627 if (resp_body.data && resp_body.size > 0) {
628 cwist_error_t body_err =
629 cwist_sstring_assign_len(response->body, resp_body.data, resp_body.size);
630 if (!cwist_error_is_ok(&body_err)) {
631 cwist_error_dispose(&body_err);
632 free(resp_body.data);
633 cwist_http_header_free_all(resp_hdrs.headers);
634 cwist_http_response_destroy(response);
635 err.error.err_i16 = -1;
636 return err;
637 }
638 cwist_error_dispose(&body_err);
639 }
640
641 /* Transfer collected headers */
642 response->headers = resp_hdrs.headers;
643
644 /* Populate status text based on code */
645 switch (response->status_code) {
646 case CWIST_HTTP_OK: cwist_sstring_assign(response->status_text, "OK"); break;
647 case CWIST_HTTP_CREATED: cwist_sstring_assign(response->status_text, "Created"); break;
648 case CWIST_HTTP_NO_CONTENT:
649 cwist_sstring_assign(response->status_text, "No Content");
650 break;
651 case CWIST_HTTP_BAD_REQUEST:
652 cwist_sstring_assign(response->status_text, "Bad Request");
653 break;
654 case CWIST_HTTP_UNAUTHORIZED:
655 cwist_sstring_assign(response->status_text, "Unauthorized");
656 break;
657 case CWIST_HTTP_FORBIDDEN: cwist_sstring_assign(response->status_text, "Forbidden"); break;
658 case CWIST_HTTP_NOT_FOUND: cwist_sstring_assign(response->status_text, "Not Found"); break;
659 case CWIST_HTTP_INTERNAL_ERROR:
660 cwist_sstring_assign(response->status_text, "Internal Server Error");
661 break;
662 case CWIST_HTTP_NOT_IMPLEMENTED:
663 cwist_sstring_assign(response->status_text, "Not Implemented");
664 break;
665 case CWIST_HTTP_SERVICE_UNAVAILABLE:
666 cwist_sstring_assign(response->status_text, "Service Unavailable");
667 break;
668 default: cwist_sstring_assign(response->status_text, ""); break;
669 }
670
671 free(resp_body.data);
672 *out_response = response;
673 err.error.err_i16 = 0;
674 return err;
675}
