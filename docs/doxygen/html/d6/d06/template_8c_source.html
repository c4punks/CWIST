69static void path_dirname(const char *path, char *buf, size_t buf_len) {
70 if (!path || buf_len == 0) return;
71
72 const char *last = NULL;
73 for (const char *p = path; *p; p++) {
74 if (*p == '/' || *p == '\\') last = p;
75 }
76
77 if (!last) {
78 /* No separator found - directory is the current directory. */
79 if (buf_len >= 2) {
80 buf[0] = '.';
81 buf[1] = '\0';
82 } else {
83 buf[0] = '\0';
84 }
85 return;
86 }
87
88 size_t len = (size_t)(last - path);
89 if (len == 0) {
90 /* Path starts with a separator, e.g. "/file.html". */
91 len = 1;
92 }
93 if (len >= buf_len) len = buf_len - 1;
94 memcpy(buf, path, len);
95 buf[len] = '\0';
96}

110static void resolve_include_path(const char *base_dir, const char *include_path, char *buf,
111 size_t buf_len) {
112 if (!include_path || buf_len == 0) {
113 if (buf_len > 0) buf[0] = '\0';
114 return;
115 }
116
117 /* Absolute path: use as-is. */
118 if (include_path[0] == '/' || include_path[0] == '\\' ||
119 (include_path[0] != '\0' && include_path[1] == ':')) {
120 snprintf(buf, buf_len, "%s", include_path);
121 return;
122 }
123
124 /* Relative path: prepend base_dir when available. */
125 if (base_dir && base_dir[0] != '\0' && !(base_dir[0] == '.' && base_dir[1] == '\0')) {
126 snprintf(buf, buf_len, "%s/%s", base_dir, include_path);
127 } else {
128 snprintf(buf, buf_len, "%s", include_path);
129 }
130}

132static template_tag_kind classify_command(const char *cmd, size_t len) {
133 static const struct {
134 const char *name;
135 template_tag_kind kind;
136 } commands[] = {
137 {"if", TAG_IF}, {"else", TAG_ELSE}, {"endif", TAG_ENDIF},
138 {"for", TAG_FOR}, {"endfor", TAG_ENDFOR}, {"include", TAG_INCLUDE},
139 };
140 for (size_t i = 0; i < sizeof(commands) / sizeof(commands[0]); i++) {
141 if (strlen(commands[i].name) == len && strncmp(cmd, commands[i].name, len) == 0)
142 return commands[i].kind;
143 }
144 return TAG_NONE;
145}

153static const char *tag_type(const char *tag) {
154 if (!tag || tag[0] != '{') return NULL;
155 switch (tag[1]) {
156 case '{': return "variable";
157 case '%':
158 switch (tag_command(tag + 2)) {
159 case TAG_IF: return "if";
160 case TAG_FOR: return "for";
161 case TAG_INCLUDE: return "include";
162 default: return "control";
163 }
164 default: return NULL;

180static const char *find_block_end(const char *start, const char *opening,
181 template_tag_kind open_kind, const char **else_pos,
182 const template_source *source) {
183 const template_tag_kind close_kind = open_kind == TAG_IF ? TAG_ENDIF : TAG_ENDFOR;
184 int nested = 0;
185 const char *scan = start;
186 const char *end;
187 while (*scan) {
188 if (scan[0] != '{') {
189 scan++;
190 continue;
191 }
192 switch (scan[1]) {
193 case '#':
194 end = strstr(scan + 2, "#}");
195 if (!end) goto unclosed_block;
196 scan = end + 2;
197 break;
198 case '{':
199 end = strstr(scan + 2, "}}");
200 if (!end) goto unclosed_tag;
201 scan = end + 2;
202 break;
203 case '%': {
204 end = strstr(scan + 2, "%}");
205 if (!end) goto unclosed_tag;
206 template_tag_kind kind = tag_command(scan + 2);
207 if (kind == open_kind) {
208 nested++;
209 } else if (kind == close_kind) {
210 if (nested == 0) return scan;
211 nested--;
212 } else if (kind == TAG_ELSE && else_pos && !*else_pos && nested == 0) {
213 *else_pos = scan;
214 }
215 scan = end + 2;
216 break;
217 }
218 default: scan++; break;
219 }
220 }
221
222unclosed_block:
223 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_UNCLOSED_BLOCK);
224 return NULL;
225unclosed_tag:
226 template_report_error(source, scan, CWIST_TEMPLATE_ERROR_UNCLOSED_TAG);
227 return NULL;
228}

231static bool json_truthy(const cJSON *value) {
232 if (!value) return false;
233 switch (value->type & 0xFF) {

235 case cJSON_String: return value->valuestring && value->valuestring[0] != '\0';
236 case cJSON_Object:

238 case cJSON_Number: return value->valuedouble != 0;

257static void html_escape(const char *value, char *out, size_t out_len) {
258 if (!value || out_len == 0) return;
259 size_t i, j;
260 for (i = 0, j = 0; value[i] && j < out_len - 1; i++) {
261 const char *repl = NULL;
262 switch (value[i]) {
263 case '&': repl = "&amp;"; break;
264 case '<': repl = "&lt;"; break;
265 case '>': repl = "&gt;"; break;
266 case '"': repl = "&quot;"; break;
267 case '\'': repl = "&#39;"; break;
268 default: break;
269 }
270 if (repl) {
271 size_t rlen = strlen(repl);
272 if (j + rlen >= out_len) break;
273 memcpy(out + j, repl, rlen);
274 j += rlen;
275 } else {
276 out[j++] = value[i];
277 }
278 }
279 out[j] = '\0';
280}

290static void apply_filter(const char *value, const char *filter, char *out, size_t out_len) {
291 if (!filter || out_len == 0) return;
292 if (strcmp(filter, "upper") == 0) {
293 size_t i;
294 for (i = 0; i < out_len - 1 && value && value[i]; i++)
295 out[i] = (char)toupper((unsigned char)value[i]);
296 out[i] = '\0';
297 } else if (strcmp(filter, "lower") == 0) {
298 size_t i;
299 for (i = 0; i < out_len - 1 && value && value[i]; i++)
300 out[i] = (char)tolower((unsigned char)value[i]);
301 out[i] = '\0';
302 } else if (strcmp(filter, "escape") == 0 || strcmp(filter, "e") == 0) {
303 html_escape(value, out, out_len);
304 } else if (strcmp(filter, "trim") == 0) {
305 if (!value || value[0] == '\0') {
306 out[0] = '\0';
307 return;
308 }
309 const char *s = value;
310 while (isspace((unsigned char)*s)) s++;
311 /* Guard against all-whitespace input: s may have advanced past the
312 * last character, making e = value + vlen - 1 < s. */
313 size_t vlen = strlen(value);
314 const char *e = value + vlen - 1;
315 while (e > s && isspace((unsigned char)*e)) e--;
316 size_t len = (s > e) ? 0 : (size_t)(e - s + 1);
317 if (len >= out_len) len = out_len - 1;
318 memcpy(out, s, len);
319 out[len] = '\0';
320 } else if (strcmp(filter, "length") == 0 || strcmp(filter, "len") == 0) {
321 snprintf(out, out_len, "%zu", value ? strlen(value) : 0);
322 } else if (strncmp(filter, "default", 7) == 0) {
323 const char *arg = NULL;
324 const char *lp = strchr(filter, '(');
325 const char *rp = strrchr(filter, ')');
326 if (lp && rp && rp > lp) {
327 arg = lp + 1;
328 }
329 if (value && value[0]) {
330 snprintf(out, out_len, "%s", value);
331 } else if (arg) {
332 size_t arg_len = (size_t)(rp - arg);
333 if (arg_len >= out_len) arg_len = out_len - 1;
334 memcpy(out, arg, arg_len);
335 out[arg_len] = '\0';
336 } else {
337 out[0] = '\0';
338 }
339 } else {
340 if (value) {
341 snprintf(out, out_len, "%s", value);
342 } else {
343 out[0] = '\0';
344 }
345 }
346}

354static const cJSON *get_value_from_context(const cJSON *context, const char *key) {
355 if (!context || !key) return NULL;
356
357 // Handle the special case of "." referring to the current context in a loop
358 if (strcmp(key, ".") == 0) {
359 return context;
360 }
361
362 char *key_copy = cwist_strdup(key);
363 if (!key_copy) return NULL;
364 char *ptr_to_free = key_copy;
365 char *saveptr = NULL;
366 char *token = strtok_r(key_copy, ".", &saveptr);
367 const cJSON *current = context;
368
369 while (token != NULL) {
370 if (!cJSON_IsObject(current)) {
371 current = NULL;
372 break;
373 }
374 current = cJSON_GetObjectItem(current, token);
375 if (!current) break;
376 token = strtok_r(NULL, ".", &saveptr);
377 }
378
379 cwist_free(ptr_to_free);
380 return current;
381}

399static cwist_sstring *render_internal(const char **template_str, const cJSON *context, int depth,
400 const template_source *source) {
401 if (!template_str || !*template_str) {
402 template_report_error(source, NULL, CWIST_TEMPLATE_ERROR_INVALID_ARGUMENT);
403 return NULL;
404 }
405
406 cwist_sstring *output = cwist_sstring_create();
407 if (!output) {
408 template_report_error(source, *template_str, CWIST_TEMPLATE_ERROR_NO_MEMORY);
409 return NULL;
410 }
411 const char *p = *template_str;
412 const char *start = p;
413
414 while (*p) {
415 if (p[0] != '{') {
416 p++;
417 continue;
418 }
419 const char *opening = p;
420
421 switch (p[1]) {
422 case '#': /* Comment: {# ... #} -- skip silently. */
423 cwist_sstring_append_len(output, start, p - start);
424 p += 2;
425 while (*p && !(p[0] == '#' && p[1] == '}')) p++;
426 if (*p) p += 2;
427 start = p;
428 break;
429
430 case '{': { /* Variable: {{ key [| filter] }} */
431 cwist_sstring_append_len(output, start, p - start);
432 p += 2;
433 const char *var_start = p;
434 while (*p && (p[0] != '}' || p[1] != '}')) p++;
435 if (!*p) {
436 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_UNCLOSED_TAG);
437 goto fail;
438 }
439
440 char var_buf[256] = {0};
441 size_t var_len = (size_t)(p - var_start);
442 if (var_len >= sizeof(var_buf)) var_len = sizeof(var_buf) - 1;
443 memcpy(var_buf, var_start, var_len);
444
445 char *trimmed_var = var_buf;
446 while (*trimmed_var == ' ') trimmed_var++;
447 char *end = trimmed_var + strlen(trimmed_var);
448 while (end > trimmed_var && end[-1] == ' ') *--end = '\0';
449
450 char *filter = NULL;
451 char *pipe = strstr(trimmed_var, "|");
452 if (pipe) {
453 char *ve = pipe;
454 while (ve > trimmed_var && isspace((unsigned char)ve[-1])) *--ve = '\0';
455 *pipe = '\0';
456 filter = pipe + 1;
457 while (*filter == ' ') filter++;
458 char *fe = filter + strlen(filter);
459 while (fe > filter && fe[-1] == ' ') *--fe = '\0';
460 }
461
462 const cJSON *value = get_value_from_context(context, trimmed_var);
463 char num_str[64] = {0};
464 const char *raw = value ? json_scalar_text(value, num_str, sizeof(num_str)) : NULL;
465 if (raw && filter) {
466 char filtered[512] = {0};
467 apply_filter(raw, filter, filtered, sizeof(filtered));
468 cwist_sstring_append(output, filtered);
469 } else if (raw) {
470 cwist_sstring_append(output, raw);
471 }
472 p += 2;
473 start = p;
474 break;
475 }
476
477 case '%': { /* Tag: {% ... %} */
478 cwist_sstring_append_len(output, start, p - start);
479 p += 2;
480 const char *tag_start = p;
481 while (*p && (p[0] != '%' || p[1] != '}')) p++;
482 if (!*p) {
483 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_UNCLOSED_TAG);
484 goto fail;
485 }
486
487 char tag[256] = {0};
488 size_t tag_len = (size_t)(p - tag_start);
489 if (tag_len >= sizeof(tag)) tag_len = sizeof(tag) - 1;
490 memcpy(tag, tag_start, tag_len);
491 p += 2; /* past "%}": start of a block body, or of the text after the tag */
492
493 char *tag_saveptr = NULL;
494 char *cmd = strtok_r(tag, TAG_DELIMS, &tag_saveptr);
495 template_tag_kind kind = cmd ? classify_command(cmd, strlen(cmd)) : TAG_NONE;
496
497 switch (kind) {
498 case TAG_IF: {
499 char *tok = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
500 bool negate = tok && strcmp(tok, "not") == 0;
501 if (negate) tok = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
502 bool cond = json_truthy(tok ? get_value_from_context(context, tok) : NULL);
503 if (negate) cond = !cond;
504
505 const char *else_pos = NULL;
506 const char *block_end =
507 find_block_end(p, opening, TAG_IF, &else_pos, source);
508 if (!block_end) goto fail;
509
510 const char *branch = NULL;
511 if (cond)
512 branch = p;
513 else if (else_pos)
514 branch = strstr(else_pos, "%}") + 2;
515 if (branch) {
516 if (depth >= CWIST_TEMPLATE_MAX_DEPTH) {
517 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_DEPTH);
518 goto fail;
519 }
520 if (!append_block(output, branch, context, depth + 1, source))
521 goto fail;
522 }
523 p = strstr(block_end, "%}") + 2;
524 break;
525 }
526
527 case TAG_FOR: {
528 char *item_name = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
529 strtok_r(NULL, TAG_DELIMS, &tag_saveptr); /* "in" */
530 char *array_name = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
531 const cJSON *array = (array_name && context)
532 ? get_value_from_context(context, array_name)
533 : NULL;
534
535 const char *block_end = find_block_end(p, opening, TAG_FOR, NULL, source);
536 if (!block_end) goto fail;
537
538 if (item_name && cJSON_IsArray(array)) {
539 if (cJSON_GetArraySize(array) > 0 &&
540 depth >= CWIST_TEMPLATE_MAX_DEPTH) {
541 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_DEPTH);
542 goto fail;
543 }
544 cJSON *item;
545 cJSON_ArrayForEach(item, array) {
546 cJSON *loop_context = cJSON_Duplicate(context, 1);
547 cJSON *loop_item = cJSON_Duplicate(item, 1);
548 if (!loop_context || !loop_item ||
549 !cJSON_AddItemToObject(loop_context, item_name, loop_item)) {
550 template_report_error(source, opening,
551 CWIST_TEMPLATE_ERROR_NO_MEMORY);
552 cJSON_Delete(loop_item);
553 cJSON_Delete(loop_context);
554 goto fail;
555 }
556 bool ok = append_block(output, p, loop_context, depth + 1, source);
557 cJSON_Delete(loop_context);
558 if (!ok) goto fail;
559 }
560 }
561 p = strstr(block_end, "%}") + 2;
562 break;
563 }
564
565 case TAG_INCLUDE: {
566 char *path = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
567 if (!path) break;
568 while (*path == ' ' || *path == '\t' || *path == '\n') path++;
569 char *end = path + strlen(path);
570 while (end > path && (end[-1] == ' ' || end[-1] == '\t' ||
571 end[-1] == '\n' || end[-1] == '"'))
572 *--end = '\0';
573 if (path[0] == '"') path++;
574 if (!path[0]) break;
575
576 /* Resolve against the including file's directory so relative
577 * includes do not depend on the process working directory. */
578 char resolved[4096];
579 resolve_include_path(source->base_dir, path, resolved, sizeof(resolved));
580 cwist_sstring *included =
581 render_file_internal(resolved, context, depth + 1, source, opening);
582 if (!included) goto fail;
583 cwist_sstring_append(output, included->data);
584 cwist_sstring_destroy(included);
585 break;
586 }
587
588 case TAG_ELSE:
589 case TAG_ENDIF:
590 case TAG_ENDFOR: *template_str = p; return output;
591
592 case TAG_NONE: break;
593 }
594 start = p;
595 break;
596 }
597
598 default: p++; break;
599 }
600 }
601
602 cwist_sstring_append_len(output, start, p - start);
603 *template_str = p;
604 return output;
605
606fail:
607 cwist_sstring_destroy(output);
608 return NULL;
609}

683static TEMPLATE_NOINLINE FILE *open_confined(const char *root, const char *path,
684 cwist_template_error_code_t *code) {
685#if defined(CWIST_BAREMETAL)
686 /* No realpath() in newlib; canonical_root() already refused the root. */
687 (void)root;
688 (void)path;
689 *code = CWIST_TEMPLATE_ERROR_INVALID_ARGUMENT;
690 return NULL;
691#else
692 char canonical[PATH_MAX];
693 if (!realpath(path, canonical)) {
694 *code = CWIST_TEMPLATE_ERROR_IO;
695 return NULL;
696 }
697 size_t root_len = strlen(root);
698 bool inside = root_len == 1 || /* the root is "/" */
699 (strncmp(canonical, root, root_len) == 0 &&
700 (canonical[root_len] == '/' || canonical[root_len] == '\0'));
701 if (!inside) {
702 *code = CWIST_TEMPLATE_ERROR_FORBIDDEN;
703 return NULL;
704 }
705 FILE *f = fopen(canonical, "rb");
706 if (!f) *code = CWIST_TEMPLATE_ERROR_IO;
707 return f;
708#endif
709}

711static cwist_sstring *render_file_internal(const char *file_path, const cJSON *context, int depth,
712 const template_source *parent, const char *include_tag) {
713 if (!file_path || depth > CWIST_TEMPLATE_MAX_DEPTH) {
714 template_report_error(parent, include_tag,
715 file_path ? CWIST_TEMPLATE_ERROR_DEPTH
716 : CWIST_TEMPLATE_ERROR_INVALID_ARGUMENT);
717 return NULL;
718 }
719 cwist_template_error_code_t code = CWIST_TEMPLATE_ERROR_IO;
720 FILE *f = include_tag && parent->include_root
721 ? open_confined(parent->include_root, file_path, &code)
722 : fopen(file_path, "rb");
723 if (!f) {
724 template_report_error(parent, include_tag, code);
725 if (code == CWIST_TEMPLATE_ERROR_IO) perror("Failed to open template file");
726 return NULL;
727 }
728
729 char *template_str = NULL;
730 long len;
731 if (fseek(f, 0, SEEK_END) != 0 || (len = ftell(f)) < 0 || fseek(f, 0, SEEK_SET) != 0)
732 goto fail_close;
733
734 template_str = cwist_alloc((size_t)len + 1);
735 if (!template_str) {
736 code = CWIST_TEMPLATE_ERROR_NO_MEMORY;
737 goto fail_close;
738 }
739
740 size_t read_len = fread(template_str, 1, (size_t)len, f);
741 bool read_failed = read_len != (size_t)len || ferror(f);
742 if (fclose(f) != 0 || read_failed) goto fail;
743 template_str[read_len] = '\0';
744
745 char dir_buf[4096];
746 path_dirname(file_path, dir_buf, sizeof(dir_buf));
747
