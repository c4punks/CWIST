82static void path_dirname(const char *path, char *buf, size_t buf_len) {
83 if (!path || buf_len == 0) return;
84
85 const char *last = NULL;
86 for (const char *p = path; *p; p++) {
87 if (*p == '/' || *p == '\\') last = p;
88 }
89
90 if (!last) {
91 /* No separator found - directory is the current directory. */
92 if (buf_len >= 2) {
93 buf[0] = '.';
94 buf[1] = '\0';
95 } else {
96 buf[0] = '\0';
97 }
98 return;
99 }
100
101 size_t len = (size_t)(last - path);
102 if (len == 0) {
103 /* Path starts with a separator, e.g. "/file.html". */
104 len = 1;
105 }
106 if (len >= buf_len) len = buf_len - 1;
107 memcpy(buf, path, len);
108 buf[len] = '\0';
109}

123static void resolve_include_path(const char *base_dir, const char *include_path, char *buf,
124 size_t buf_len) {
125 if (!include_path || buf_len == 0) {
126 if (buf_len > 0) buf[0] = '\0';
127 return;
128 }
129
130 /* Absolute path: use as-is. */
131 if (include_path[0] == '/' || include_path[0] == '\\' ||
132 (include_path[0] != '\0' && include_path[1] == ':')) {
133 snprintf(buf, buf_len, "%s", include_path);
134 return;
135 }
136
137 /* Relative path: prepend base_dir when available. */
138 if (base_dir && base_dir[0] != '\0' && !(base_dir[0] == '.' && base_dir[1] == '\0')) {
139 snprintf(buf, buf_len, "%s/%s", base_dir, include_path);
140 } else {
141 snprintf(buf, buf_len, "%s", include_path);
142 }
143}

145static template_tag_kind classify_command(const char *cmd, size_t len) {
146 static const struct {
147 const char *name;
148 template_tag_kind kind;
149 } commands[] = {
150 {"if", TAG_IF}, {"else", TAG_ELSE}, {"endif", TAG_ENDIF},
151 {"for", TAG_FOR}, {"endfor", TAG_ENDFOR}, {"include", TAG_INCLUDE},
152 {"extends", TAG_EXTENDS}, {"block", TAG_BLOCK}, {"endblock", TAG_ENDBLOCK},
153 };
154 for (size_t i = 0; i < sizeof(commands) / sizeof(commands[0]); i++) {
155 if (strlen(commands[i].name) == len && strncmp(cmd, commands[i].name, len) == 0)
156 return commands[i].kind;
157 }
158 return TAG_NONE;
159}

169 switch (tag[1]) {

196static const char *find_block_end(const char *start, const char *opening,
197 template_tag_kind open_kind, const char **else_pos,
198 const template_source *source) {
199 const template_tag_kind close_kind =
200 open_kind == TAG_IF ? TAG_ENDIF : (open_kind == TAG_FOR ? TAG_ENDFOR : TAG_ENDBLOCK);
201 int nested = 0;
202 const char *scan = start;
203 const char *end;
204 while (*scan) {
205 if (scan[0] != '{') {
206 scan++;
207 continue;
208 }
209 switch (scan[1]) {
210 case '#':
211 end = strstr(scan + 2, "#}");
212 if (!end) goto unclosed_block;
213 scan = end + 2;
214 break;
215 case '{':
216 end = strstr(scan + 2, "}}");
217 if (!end) goto unclosed_tag;
218 scan = end + 2;
219 break;
220 case '%': {
221 end = strstr(scan + 2, "%}");
222 if (!end) goto unclosed_tag;
223 template_tag_kind kind = tag_command(scan + 2);
224 if (kind == open_kind) {
225 nested++;
226 } else if (kind == close_kind) {
227 if (nested == 0) return scan;
228 nested--;
229 } else if (kind == TAG_ELSE && else_pos && !*else_pos && nested == 0) {
230 *else_pos = scan;
231 }
232 scan = end + 2;
233 break;
234 }
235 default: scan++; break;
236 }
237 }
238
239unclosed_block:
240 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_UNCLOSED_BLOCK);
241 return NULL;
242unclosed_tag:
243 template_report_error(source, scan, CWIST_TEMPLATE_ERROR_UNCLOSED_TAG);
244 return NULL;
245}

275static cwist_sstring *apply_filter(const char *value, const char *filter) {
276 if (!filter) return NULL;
277 cwist_sstring *out = cwist_sstring_create();
278 if (!out) return NULL;
279
280 if (strcmp(filter, "upper") == 0 || strcmp(filter, "lower") == 0) {
281 bool upper = strcmp(filter, "upper") == 0;
282 if (value) {
283 const char *p = value;
284 while (*p) {
285 char chunk[256];
286 size_t len = strlen(p) < sizeof(chunk) ? strlen(p) : sizeof(chunk);
287 for (size_t i = 0; i < len; i++) {
288 unsigned char c = (unsigned char)p[i];
289 chunk[i] = (char)(upper ? toupper(c) : tolower(c));
290 }
291 cwist_sstring_append_len(out, chunk, len);
292 p += len;
293 }
294 }
295 } else if (strcmp(filter, "escape") == 0 || strcmp(filter, "e") == 0) {
296 cwist_sstring_append_escaped(out, value);
297 } else if (strcmp(filter, "trim") == 0) {
298 if (value && *value) {
299 const char *s = value;
300 while (isspace((unsigned char)*s)) s++;
301 size_t vlen = strlen(value);
302 const char *e = value + vlen - 1;
303 while (e > s && isspace((unsigned char)*e)) e--;
304 if (s <= e) cwist_sstring_append_len(out, s, (size_t)(e - s + 1));
305 }
306 } else if (strcmp(filter, "length") == 0 || strcmp(filter, "len") == 0) {
307 /* Code points, not bytes: UTF-8 continuation bytes are not counted
308 * (#986). Arrays and objects are counted in apply_filters(). */
309 size_t count = 0;
310 for (const unsigned char *c = (const unsigned char *)(value ? value : ""); *c; c++)
311 if ((*c & 0xC0) != 0x80) count++;
312 char length[64];
313 snprintf(length, sizeof(length), "%zu", count);
314 cwist_sstring_append(out, length);
315 } else if (value) {
316 cwist_sstring_append(out, value);
317 }
318 return out;
319}

325static char *next_filter(char **cursor) {
326 char *p = *cursor;
327 if (!p) return NULL;
328 while (isspace((unsigned char)*p) || *p == '|') p++;
329 if (!*p) {
330 *cursor = NULL;
331 return NULL;
332 }
333 char *start = p;
334 char quote = 0;
335 for (; *p; p++) {
336 if (quote) {
337 if (*p == quote) quote = 0;
338 } else if (*p == '"' || *p == '\'') {
339 quote = *p;
340 } else if (*p == '|') {
341 break;
342 }
343 }
344 *cursor = *p ? p + 1 : NULL;
345 *p = '\0';
346 char *end = p;
347 while (end > start && isspace((unsigned char)end[-1])) *--end = '\0';
348 return start;
349}

356static bool default_argument(const char *filter, const char **arg, size_t *arg_len) {
357 if (strncmp(filter, "default", 7) != 0) return false;
358 const char *lp = filter + 7;
359 while (isspace((unsigned char)*lp)) lp++;
360 if (*lp != '(') return false;
361 const char *rp = strrchr(lp, ')');
362 if (!rp) return false;
363 const char *s = lp + 1;
364 const char *e = rp;
365 while (s < e && isspace((unsigned char)*s)) s++;
366 while (e > s && isspace((unsigned char)e[-1])) e--;
367 if (e - s >= 2 && (*s == '"' || *s == '\'') && e[-1] == *s) {
368 s++;
369 e--;
370 }
371 *arg = s;
372 *arg_len = (size_t)(e - s);
373 return true;

385static cwist_sstring *apply_filters(const char *value, const cJSON *node, char *filters) {
386 bool have = value != NULL;
387 cwist_sstring *current = cwist_sstring_create();
388 if (!current) return NULL;
389 cwist_sstring_append(current, value ? value : "");
390
391 bool autoescape = true;
392 bool escaped = false;
393 bool first = true;
394 char *cursor = filters;
395 char *filter;
396 while ((filter = next_filter(&cursor)) != NULL) {
397 const char *arg = NULL;
398 size_t arg_len = 0;
399 if (strcmp(filter, "safe") == 0 || strcmp(filter, "raw") == 0) {
400 autoescape = false;
401 } else if (default_argument(filter, &arg, &arg_len)) {
402 /* Applies to a missing variable as well as an empty one (#985). */
403 if (!have || !current->data || !current->data[0]) {
404 cwist_sstring_destroy(current);
405 current = cwist_sstring_create();
406 if (!current) return NULL;
407 cwist_sstring_assign(current, "");
408 cwist_sstring_append_len(current, arg, arg_len);
409 have = true;
410 }
411 } else if (first && (strcmp(filter, "length") == 0 || strcmp(filter, "len") == 0) &&
412 (cJSON_IsArray(node) || cJSON_IsObject(node) || !have)) {
413 /* Item count for arrays and objects, 0 for a missing value (#986). */
414 char length[64];
415 snprintf(length, sizeof(length), "%d", node ? cJSON_GetArraySize(node) : 0);
416 cwist_sstring_destroy(current);
417 current = cwist_sstring_create();
418 if (!current) return NULL;
419 cwist_sstring_assign(current, length);
420 have = true;
421 } else if (have) {
422 cwist_sstring *next = apply_filter(current->data, filter);
423 if (!next) {
424 cwist_sstring_destroy(current);
425 return NULL;
426 }
427 cwist_sstring_destroy(current);
428 current = next;
429 /* Sticky: once escaped, later filters (upper, trim, ...) work
430 * on markup, and escaping it again would print "&amp;lt;". */
431 if (strcmp(filter, "escape") == 0 || strcmp(filter, "e") == 0) escaped = true;
432 }
433 first = false;
434 }
435 if (!have) {
436 cwist_sstring_destroy(current);
437 return NULL;
438 }
439 if (autoescape && !escaped) {
440 cwist_sstring *next = cwist_sstring_create();
441 if (!next) {
442 cwist_sstring_destroy(current);
443 return NULL;
444 }
445 cwist_sstring_append_escaped(next, current->data);
446 cwist_sstring_destroy(current);
447 current = next;
448 }
449 return current;
450}

458static const cJSON *get_value_from_context(const cJSON *context, const char *key) {
459 if (!context || !key) return NULL;
460
461 // Handle the special case of "." referring to the current context in a loop
462 if (strcmp(key, ".") == 0) {
463 return context;
464 }
465
466 char *key_copy = cwist_strdup(key);
467 if (!key_copy) return NULL;
468 char *ptr_to_free = key_copy;
469 char *saveptr = NULL;
470 char *token = strtok_r(key_copy, ".", &saveptr);
471 const cJSON *current = context;
472
473 while (token != NULL) {
474 if (!cJSON_IsObject(current)) {
475 current = NULL;
476 break;
477 }
478 current = cJSON_GetObjectItem(current, token);
479 if (!current) break;
480 token = strtok_r(NULL, ".", &saveptr);
481 }
482
483 cwist_free(ptr_to_free);
484 return current;
485}

734 if (filtered) {
735 cwist_sstring_append(output, filtered->data);
736 cwist_sstring_destroy(filtered);
737 }
738 }
739 p += 2;
740 start = p;
741 break;
742 }
743
744 case '%': { /* Tag: {% ... %} */
745 cwist_sstring_append_len(output, start, p - start);
746 p += 2;
747 const char *tag_start = p;
748 while (*p && (p[0] != '%' || p[1] != '}')) p++;
749 if (!*p) {
750 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_UNCLOSED_TAG);
751 goto fail;
752 }
753
754 char tag[256] = {0};
755 size_t tag_len = (size_t)(p - tag_start);
756 if (tag_len >= sizeof(tag)) tag_len = sizeof(tag) - 1;
757 memcpy(tag, tag_start, tag_len);
758 p += 2; /* past "%}": start of a block body, or of the text after the tag */
759
760 char *tag_saveptr = NULL;
761 char *cmd = strtok_r(tag, TAG_DELIMS, &tag_saveptr);
762 template_tag_kind kind = cmd ? classify_command(cmd, strlen(cmd)) : TAG_NONE;
763
764 switch (kind) {
765 case TAG_IF: {
766 char *tok = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
767 bool negate = tok && strcmp(tok, "not") == 0;
768 if (negate) tok = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
769 bool cond = json_truthy(tok ? get_value_from_context(context, tok) : NULL);
770 if (negate) cond = !cond;
771
772 const char *else_pos = NULL;
773 const char *block_end =
774 find_block_end(p, opening, TAG_IF, &else_pos, source);
775 if (!block_end) goto fail;
776
777 const char *branch = NULL;
778 if (cond)
779 branch = p;
780 else if (else_pos)
781 branch = strstr(else_pos, "%}") + 2;
782 if (branch) {
783 if (depth >= CWIST_TEMPLATE_MAX_DEPTH) {
784 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_DEPTH);
785 goto fail;
786 }
787 if (!append_block(output, branch, context, depth + 1, source))
788 goto fail;
789 }
790 p = strstr(block_end, "%}") + 2;
791 break;
792 }
793
794 case TAG_FOR: {
795 char *item_name = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
796 strtok_r(NULL, TAG_DELIMS, &tag_saveptr); /* "in" */
797 char *array_name = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
798 const cJSON *array = (array_name && context)
799 ? get_value_from_context(context, array_name)
800 : NULL;
801
802 const char *block_end = find_block_end(p, opening, TAG_FOR, NULL, source);
803 if (!block_end) goto fail;
804
805 if (item_name && cJSON_IsArray(array)) {
806 if (cJSON_GetArraySize(array) > 0 &&
807 depth >= CWIST_TEMPLATE_MAX_DEPTH) {
808 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_DEPTH);
809 goto fail;
810 }
811 cJSON *item;
812 cJSON_ArrayForEach(item, array) {
813 cJSON *loop_context = cJSON_Duplicate(context, 1);
814 cJSON *loop_item = cJSON_Duplicate(item, 1);
815 if (!loop_context || !loop_item ||
816 !cJSON_AddItemToObject(loop_context, item_name, loop_item)) {
817 template_report_error(source, opening,
818 CWIST_TEMPLATE_ERROR_NO_MEMORY);
819 cJSON_Delete(loop_item);
820 cJSON_Delete(loop_context);
821 goto fail;
822 }
823 bool ok = append_block(output, p, loop_context, depth + 1, source);
824 cJSON_Delete(loop_context);
825 if (!ok) goto fail;
826 }
827 }
828 p = strstr(block_end, "%}") + 2;
829 break;
830 }
831
832 case TAG_INCLUDE: {
833 char *path = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
834 if (!path) break;
835 while (*path == ' ' || *path == '\t' || *path == '\n') path++;
836 char *end = path + strlen(path);
837 while (end > path && (end[-1] == ' ' || end[-1] == '\t' ||
838 end[-1] == '\n' || end[-1] == '"'))
839 *--end = '\0';
840 if (path[0] == '"') path++;
841 if (!path[0]) break;
842
843 /* Resolve against the including file's directory so relative
844 * includes do not depend on the process working directory. */
845 char resolved[4096];
846 resolve_include_path(source->base_dir, path, resolved, sizeof(resolved));
847 cwist_sstring *included =
848 render_file_internal(resolved, context, depth + 1, source, opening);
849 if (!included) goto fail;
850 cwist_sstring_append(output, included->data);
851 cwist_sstring_destroy(included);
852 break;
853 }
854
855 case TAG_BLOCK: {
856 char *bname = strtok_r(NULL, TAG_DELIMS, &tag_saveptr);
857 const char *block_end = find_block_end(p, opening, TAG_BLOCK, NULL, source);
858 if (!block_end) goto fail;
859
860 const template_block_entry *override =
861 bname ? find_block_override(source->blocks, bname) : NULL;
862 if (override && override->content) {
863 if (depth >= CWIST_TEMPLATE_MAX_DEPTH) {
864 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_DEPTH);
865 goto fail;
866 }
867 if (!append_block(output, override->content, context, depth + 1,
868 source))
869 goto fail;
870 } else {
871 if (depth >= CWIST_TEMPLATE_MAX_DEPTH) {
872 template_report_error(source, opening, CWIST_TEMPLATE_ERROR_DEPTH);
873 goto fail;
874 }
875 if (!append_block(output, p, context, depth + 1, source)) goto fail;
876 }
877 p = strstr(block_end, "%}") + 2;
878 break;
879 }
880
881 case TAG_EXTENDS: break;
882
883 case TAG_ELSE:
884 case TAG_ENDIF:
885 case TAG_ENDFOR:
886 case TAG_ENDBLOCK: *template_str = p; return output;
887
888 case TAG_NONE: break;
889 }
890 start = p;
891 break;
892 }
893
894 default: p++; break;
895 }
896 }
897
898 cwist_sstring_append_len(output, start, p - start);
899 *template_str = p;
900 return output;
901
902fail:
903 cwist_sstring_destroy(output);
904 return NULL;
905}

979static TEMPLATE_NOINLINE FILE *open_confined(const char *root, const char *path,
980 cwist_template_error_code_t *code) {
981#if defined(CWIST_BAREMETAL)
982 /* No realpath() in newlib; canonical_root() already refused the root. */
983 (void)root;
984 (void)path;
985 *code = CWIST_TEMPLATE_ERROR_INVALID_ARGUMENT;
986 return NULL;
987#else
988 char canonical[PATH_MAX];
989 if (!realpath(path, canonical)) {
990 *code = CWIST_TEMPLATE_ERROR_IO;
991 return NULL;
992 }
993 if (!cwist_file_path_is_inside(canonical, root)) {
994 *code = CWIST_TEMPLATE_ERROR_FORBIDDEN;
995 return NULL;
996 }
997 int fd = cwist_file_open_confined(root, canonical);
998 if (fd < 0) {
999 *code = CWIST_TEMPLATE_ERROR_FORBIDDEN;
1000 return NULL;
1001 }
1002 FILE *f = fdopen(fd, "rb");
1003 if (!f) {
1004 close(fd);
1005 *code = CWIST_TEMPLATE_ERROR_IO;
1006 }
1007 return f;
1008#endif
1009}

1011static cwist_sstring *render_file_internal(const char *file_path, const cJSON *context, int depth,
1012 const template_source *parent, const char *include_tag) {
1013 if (!file_path || depth > CWIST_TEMPLATE_MAX_DEPTH) {
1014 template_report_error(parent, include_tag,
1015 file_path ? CWIST_TEMPLATE_ERROR_DEPTH
1016 : CWIST_TEMPLATE_ERROR_INVALID_ARGUMENT);
1017 return NULL;
1018 }
1019 cwist_template_error_code_t code = CWIST_TEMPLATE_ERROR_IO;
1020 FILE *f = include_tag && parent->include_root
1021 ? open_confined(parent->include_root, file_path, &code)
1022 : fopen(file_path, "rb");
1023 if (!f) {
1024 template_report_error(parent, include_tag, code);
1025 if (code == CWIST_TEMPLATE_ERROR_IO) perror("Failed to open template file");
1026 return NULL;
1027 }
1028
1029 char *template_str = NULL;
1030 long len;
1031 if (fseek(f, 0, SEEK_END) != 0 || (len = ftell(f)) < 0 || fseek(f, 0, SEEK_SET) != 0)
1032 goto fail_close;
1033
1034 template_str = cwist_alloc((size_t)len + 1);
1035 if (!template_str) {
1036 code = CWIST_TEMPLATE_ERROR_NO_MEMORY;
1037 goto fail_close;
1038 }
1039
1040 size_t read_len = fread(template_str, 1, (size_t)len, f);
1041 bool read_failed = read_len != (size_t)len || ferror(f);
1042 if (fclose(f) != 0 || read_failed) goto fail;
1043 template_str[read_len] = '\0';
1044
1045 char dir_buf[4096];
1046 path_dirname(file_path, dir_buf, sizeof(dir_buf));
1047
1048 const template_source source = {template_str, file_path, dir_buf, parent->include_root,
1049 parent->error, parent->blocks, true};
1050 const char *p = template_str;
1051 cwist_sstring *rendered = render_internal(&p, context, depth, &source);
1052 cwist_free(template_str);
1053 return rendered;
1054
1055fail_close:
1056 fclose(f);
1057fail:
1058 template_report_error(parent, include_tag, code);
1059 cwist_free(template_str);
1060 return NULL;
1061}
