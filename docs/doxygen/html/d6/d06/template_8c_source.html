23static void html_escape(const char *value, char *out, size_t out_len) {
24 if (!value || out_len == 0) return;
25 size_t i, j;
26 for (i = 0, j = 0; value[i] && j < out_len - 1; i++) {
27 const char *repl = NULL;
28 switch (value[i]) {
29 case '&': repl = "&amp;"; break;
30 case '<': repl = "&lt;"; break;
31 case '>': repl = "&gt;"; break;
32 case '"': repl = "&quot;"; break;
33 case '\'': repl = "&#39;"; break;
34 default: break;
35 }
36 if (repl) {
37 size_t rlen = strlen(repl);
38 if (j + rlen >= out_len) break;
39 memcpy(out + j, repl, rlen);
40 j += rlen;
41 } else {
42 out[j++] = value[i];
43 }
44 }
45 out[j] = '\0';
46}

56static void apply_filter(const char *value, const char *filter, char *out, size_t out_len) {
57 if (!filter || out_len == 0) return;
58 if (strcmp(filter, "upper") == 0) {
59 size_t i;
60 for (i = 0; i < out_len - 1 && value && value[i]; i++)
61 out[i] = (char)toupper((unsigned char)value[i]);
62 out[i] = '\0';
63 } else if (strcmp(filter, "lower") == 0) {
64 size_t i;
65 for (i = 0; i < out_len - 1 && value && value[i]; i++)
66 out[i] = (char)tolower((unsigned char)value[i]);
67 out[i] = '\0';
68 } else if (strcmp(filter, "escape") == 0 || strcmp(filter, "e") == 0) {
69 html_escape(value, out, out_len);
70 } else if (strcmp(filter, "trim") == 0) {
71 if (!value || value[0] == '\0') {
72 out[0] = '\0';
73 return;
74 }
75 const char *s = value;
76 while (isspace((unsigned char)*s)) s++;
77 /* Guard against all-whitespace input: s may have advanced past the
78 * last character, making e = value + vlen - 1 < s. */
79 size_t vlen = strlen(value);
80 const char *e = value + vlen - 1;
81 while (e > s && isspace((unsigned char)*e)) e--;
82 size_t len = (s > e) ? 0 : (size_t)(e - s + 1);
83 if (len >= out_len) len = out_len - 1;
84 memcpy(out, s, len);
85 out[len] = '\0';
86 } else if (strcmp(filter, "length") == 0 || strcmp(filter, "len") == 0) {
87 snprintf(out, out_len, "%zu", value ? strlen(value) : 0);
88 } else if (strncmp(filter, "default", 7) == 0) {
89 const char *arg = NULL;
90 const char *lp = strchr(filter, '(');
91 const char *rp = strrchr(filter, ')');
92 if (lp && rp && rp > lp) {
93 arg = lp + 1;
94 }
95 if (value && value[0]) {
96 snprintf(out, out_len, "%s", value);
97 } else if (arg) {
98 size_t arg_len = (size_t)(rp - arg);
99 if (arg_len >= out_len) arg_len = out_len - 1;
100 memcpy(out, arg, arg_len);
101 out[arg_len] = '\0';
102 } else {
103 out[0] = '\0';
104 }
105 } else {
106 if (value) {
107 snprintf(out, out_len, "%s", value);
108 } else {
109 out[0] = '\0';
110 }
111 }

121 if (!context || !key) return NULL;

123 // Handle the special case of "." referring to the current context in a loop
124 if (strcmp(key, ".") == 0) {

137 current = NULL;
