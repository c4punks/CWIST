96/* Random non-zero id of `nibbles` lowercase hex digits. */
97static void otel_random_id(char *out, size_t nibbles) {
98 static const char hex[] = "0123456789abcdef";
99 uint8_t bytes[OTEL_TRACE_ID_HEX / 2];
100 do {
101 if (RAND_bytes(bytes, (int)(nibbles / 2)) != 1) memset(bytes, 0x5a, sizeof(bytes));
102 for (size_t i = 0; i < nibbles / 2; i++) {
103 out[2 * i] = hex[bytes[i] >> 4];
104 out[2 * i + 1] = hex[bytes[i] & 0xF];
105 }
106 out[nibbles] = '\0';
107 } while (otel_is_zero(out, nibbles));
108}

110/* W3C trace-context: "vv-<32 hex trace-id>-<16 hex parent-id>-ff", version
111 * not "ff", ids not all zero. Later versions may append "-..." fields. */
112static bool otel_parse_traceparent(const char *h, char *trace_id, char *parent_id) {
113 if (!h || strlen(h) < 55) return false;
114 if (!otel_is_hex(h, 2) || strncmp(h, "ff", 2) == 0 || h[2] != '-') return false;
115 if (!otel_is_hex(h + 3, OTEL_TRACE_ID_HEX) || otel_is_zero(h + 3, OTEL_TRACE_ID_HEX) ||
116 h[35] != '-')
117 return false;
118 if (!otel_is_hex(h + 36, OTEL_SPAN_ID_HEX) || otel_is_zero(h + 36, OTEL_SPAN_ID_HEX) ||
119 h[52] != '-')
120 return false;
121 if (!otel_is_hex(h + 53, 2) || (h[55] != '\0' && (strncmp(h, "00", 2) == 0 || h[55] != '-')))
122 return false;
123 memcpy(trace_id, h + 3, OTEL_TRACE_ID_HEX);
124 trace_id[OTEL_TRACE_ID_HEX] = '\0';
125 memcpy(parent_id, h + 36, OTEL_SPAN_ID_HEX);
126 parent_id[OTEL_SPAN_ID_HEX] = '\0';
127 return true;
128}

137/* OTLP/HTTP JSON body for `n` spans (uint64 fields are strings per the
138 * protobuf JSON mapping). */
139static char *otel_encode_spans(const otel_span_t *spans, size_t n) {
140 cJSON *root = cJSON_CreateObject();
141 cJSON *rs = cJSON_CreateObject();
142 cJSON_AddItemToArray(cJSON_AddArrayToObject(root, "resourceSpans"), rs);
143 cJSON *rattrs = cJSON_AddArrayToObject(cJSON_AddObjectToObject(rs, "resource"), "attributes");
144 cJSON_AddItemToArray(
145 rattrs, otel_attr("service.name", cJSON_CreateString(g_otel.service_name), "stringValue"));
146 cJSON *ss = cJSON_CreateObject();
147 cJSON_AddItemToArray(cJSON_AddArrayToObject(rs, "scopeSpans"), ss);
148 cJSON_AddStringToObject(cJSON_AddObjectToObject(ss, "scope"), "name", "cwist");
149 cJSON *out = cJSON_AddArrayToObject(ss, "spans");
150
151 for (size_t i = 0; i < n; i++) {
152 const otel_span_t *s = &spans[i];
153 const char *method = cwist_http_method_to_string(s->method);
154 char num[24];
155 cJSON *span = cJSON_CreateObject();
156 cJSON_AddStringToObject(span, "traceId", s->trace_id);
157 cJSON_AddStringToObject(span, "spanId", s->span_id);
158 if (s->parent_span_id[0]) cJSON_AddStringToObject(span, "parentSpanId", s->parent_span_id);
159 cJSON_AddStringToObject(span, "name", method);
160 cJSON_AddNumberToObject(span, "kind", 2); /* SPAN_KIND_SERVER */
161 snprintf(num, sizeof(num), "%llu", (unsigned long long)s->start_ns);
162 cJSON_AddStringToObject(span, "startTimeUnixNano", num);
163 snprintf(num, sizeof(num), "%llu", (unsigned long long)s->end_ns);
164 cJSON_AddStringToObject(span, "endTimeUnixNano", num);
165 cJSON *attrs = cJSON_AddArrayToObject(span, "attributes");
166 cJSON_AddItemToArray(
167 attrs, otel_attr("http.request.method", cJSON_CreateString(method), "stringValue"));
168 cJSON_AddItemToArray(attrs,
169 otel_attr("url.path", cJSON_CreateString(s->path), "stringValue"));
170 snprintf(num, sizeof(num), "%d", s->status);
171 cJSON_AddItemToArray(
172 attrs, otel_attr("http.response.status_code", cJSON_CreateString(num), "intValue"));
173 if (s->status >= 500)
174 cJSON_AddNumberToObject(cJSON_AddObjectToObject(span, "status"), "code", 2);
175 cJSON_AddItemToArray(out, span);
176 }
177 char *body = cJSON_PrintUnformatted(root);
178 cJSON_Delete(root);
179 return body;
180}

195static void *otel_exporter_main(void *arg) {
196 (void)arg;
197 cwist_http_client *client = cwist_http_client_create();
198 otel_span_t *batch = cwist_alloc_array(g_otel.batch, sizeof(*batch));
199 if (client) cwist_http_client_set_timeout_ms(client, OTEL_EXPORT_TIMEOUT_MS);
200
201 pthread_mutex_lock(&g_otel_lock);
202 for (;;) {
203 while (g_otel.count == 0) pthread_cond_wait(&g_otel_work, &g_otel_lock);
204 size_t n = g_otel.count < g_otel.batch ? g_otel.count : g_otel.batch;
205 for (size_t i = 0; i < n; i++) {
206 if (batch) batch[i] = g_otel.ring[g_otel.head];
207 g_otel.head = (g_otel.head + 1) % g_otel.cap;
208 }
209 g_otel.count -= n;
210 pthread_mutex_unlock(&g_otel_lock);
211 if (client && batch) otel_export_batch(client, batch, n);
212 pthread_mutex_lock(&g_otel_lock);
213 }
214 return NULL;
215}

225/* The parent exports its own queued spans; the child starts empty and spawns

227static void otel_atfork_child(void) {
228 g_otel.head = g_otel.count = 0;
229 g_otel.exporter_alive = false;

235static void otel_enqueue(const otel_span_t *span) {
236 pthread_mutex_lock(&g_otel_lock);
237 if (!g_otel.exporter_alive)
238 g_otel.exporter_alive =
239 pthread_create(&g_otel.exporter, NULL, otel_exporter_main, NULL) == 0;
240 if (g_otel.exporter_alive) {
241 if (g_otel.count == g_otel.cap) { /* full: drop the oldest */
242 g_otel.head = (g_otel.head + 1) % g_otel.cap;
243 g_otel.count--;
244 }
245 g_otel.ring[(g_otel.head + g_otel.count) % g_otel.cap] = *span;
246 g_otel.count++;
247 pthread_cond_signal(&g_otel_work);
248 }
249 pthread_mutex_unlock(&g_otel_lock);
250}

252static void otel_middleware(cwist_http_request *req, cwist_http_response *res,
253 cwist_handler_func next) {
254 otel_span_t span = {.method = req->method, .start_ns = otel_unix_nanos()};
255 if (!otel_parse_traceparent(cwist_http_header_get(req->headers, "traceparent"), span.trace_id,
256 span.parent_span_id))
257 otel_random_id(span.trace_id, OTEL_TRACE_ID_HEX);
258 otel_random_id(span.span_id, OTEL_SPAN_ID_HEX);
259
260 otel_ctx ctx = {.req = req, .trace_id = span.trace_id, .outer = t_otel_ctx};
261 t_otel_ctx = &ctx;
262 next(req, res);
263 t_otel_ctx = ctx.outer;
264
265 span.end_ns = otel_unix_nanos();
266 span.status = (int)res->status_code;
267 snprintf(span.path, sizeof(span.path), "%s",
268 req->path && req->path->data ? req->path->data : "/");
269 otel_enqueue(&span);
270}

272cwist_middleware_func cwist_mw_otel_trace(const cwist_otel_cfg_t *cfg) {
273 pthread_mutex_lock(&g_otel_lock);
274 if (!g_otel.configured) {
275 const char *endpoint = cfg && cfg->endpoint ? cfg->endpoint : OTEL_DEFAULT_ENDPOINT;
276 const char *service = cfg && cfg->service_name ? cfg->service_name : OTEL_DEFAULT_SERVICE;
277 size_t cap = cfg && cfg->queue_cap ? cfg->queue_cap : OTEL_DEFAULT_QUEUE;
278 size_t batch = cfg && cfg->batch_size ? cfg->batch_size : OTEL_DEFAULT_BATCH;
279 g_otel.endpoint = cwist_strdup(endpoint);
280 g_otel.service_name = cwist_strdup(service);
281 g_otel.ring = cwist_alloc_array(cap, sizeof(*g_otel.ring));
282 if (g_otel.endpoint && g_otel.service_name && g_otel.ring &&
283 pthread_atfork(otel_atfork_prepare, otel_atfork_parent, otel_atfork_child) == 0) {
284 g_otel.cap = cap;
285 g_otel.batch = batch < cap ? batch : cap;
286 g_otel.configured = true;
287 } else {
288 cwist_free(g_otel.endpoint);
289 cwist_free(g_otel.service_name);
290 cwist_free(g_otel.ring);
291 g_otel.endpoint = g_otel.service_name = NULL;
292 g_otel.ring = NULL;
293 }
294 }
295 bool ok = g_otel.configured;
296 pthread_mutex_unlock(&g_otel_lock);
297 return ok ? otel_middleware : NULL;
298}
