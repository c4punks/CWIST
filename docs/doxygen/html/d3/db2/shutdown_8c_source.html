56void cwist_shutdown_request(void) {
57 atomic_store(&g_cwist_running, 0);
58 /* Take each descriptor atomically so a second request (another thread,
59 * a signal, a repeated call) cannot close a number that has meanwhile
60 * been reused for an unrelated file. */
61 int fd = __atomic_exchange_n(&g_cwist_listen_fd, -1, __ATOMIC_SEQ_CST);
62 if (fd >= 0) {
63#ifndef __wasi__
64 /* close() alone does not interrupt an accept() already blocked in
65 * another thread on Linux; shutdown() makes it return (EINVAL),
66 * which the accept loops treat as the end of the server. */
67 shutdown(fd, SHUT_RDWR);
68#endif
69 close(fd);
70 }
71 int udp = __atomic_exchange_n(&g_cwist_udp_fd, -1, __ATOMIC_SEQ_CST);
72 if (udp >= 0) {
73 close(udp);
74 }
75}

107void cwist_shutdown_install_handlers(void) {
108 struct sigaction sa, old_term, old_int;
109 sigemptyset(&sa.sa_mask);
110 sa.sa_flags = SA_RESTART;
111 sa.sa_handler = cwist_shutdown_handler;
112 bool term_ok = sigaction(SIGTERM, &sa, &old_term) == 0;
113 bool int_ok = sigaction(SIGINT, &sa, &old_int) == 0;
114 if (!g_prev_saved && term_ok && int_ok) {
115 g_prev_sigterm = old_term;
116 g_prev_sigint = old_int;
117 g_prev_saved = true;
118 }
119}

129void cwist_shutdown_restore_handlers(void) {

131 sigaction(SIGTERM, &g_prev_sigterm, NULL);
132 sigaction(SIGINT, &g_prev_sigint, NULL);
133 g_prev_saved = false;
