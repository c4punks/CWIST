74static size_t cwist_cert_fetch_write_cb(void *ptr, size_t size, size_t nmemb, void *userdata) {
75 if (size != 0 && nmemb > SIZE_MAX / size) return 0;
76
77 size_t total = size * nmemb;
78 cwist_cert_fetch_buffer *buf = (cwist_cert_fetch_buffer *)userdata;
79 if (total > CWIST_AIA_MAX_RESPONSE_BYTES || buf->len > CWIST_AIA_MAX_RESPONSE_BYTES - total) {
80 return 0;
81 }
82
83 size_t need = buf->len + total;
84 if (need > buf->cap) {
85 size_t new_cap = buf->cap ? buf->cap * 2 : 4096;
86 while (new_cap < need) {
87 if (new_cap > CWIST_AIA_MAX_RESPONSE_BYTES / 2) {
88 new_cap = CWIST_AIA_MAX_RESPONSE_BYTES;
89 break;
90 }
91 new_cap *= 2;
92 }
93
94 uint8_t *tmp = (uint8_t *)realloc(buf->data, new_cap);
95 if (!tmp) return 0;
96 buf->data = tmp;
97 buf->cap = new_cap;
98 }
99
100 memcpy(buf->data + buf->len, ptr, total);
101 buf->len += total;
102 return total;
103}

199static STACK_OF(X509) * cwist_parse_pem_x509(const uint8_t *data, size_t len) {
200 if (!data || len == 0 || len > INT_MAX) return NULL;
201
202 BIO *bio = BIO_new_mem_buf(data, (int)len);
203 if (!bio) return NULL;
204
205 STACK_OF(X509) *certs = sk_X509_new_null();
206 if (!certs) {
207 BIO_free(bio);
208 return NULL;
209 }
210
211 X509 *cert = NULL;
212 while ((cert = PEM_read_bio_X509(bio, NULL, NULL, NULL)) != NULL) {
213 if (!sk_X509_push(certs, cert)) {
214 X509_free(cert);
215 sk_X509_pop_free(certs, X509_free);
216 BIO_free(bio);
217 return NULL;
218 }
219 }
220 ERR_clear_error();
221 BIO_free(bio);
222
223 if (sk_X509_num(certs) == 0) {
224 sk_X509_free(certs);
225 return NULL;
226 }
227 return certs;

322static bool cwist_x509_is_valid_issuer(X509 *issuer, X509 *subject) {
323 if (!issuer || !subject || X509_cmp(issuer, subject) == 0) return false;
324 if (X509_check_ca(issuer) != 1) return false;
325 if (X509_check_issued(issuer, subject) != X509_V_OK) return false;
326
327 EVP_PKEY *issuer_key = X509_get0_pubkey(issuer);
328 return issuer_key && X509_verify(subject, issuer_key) == 1;

365static bool cwist_ctx_contains_cert(SSL_CTX *ssl_ctx, X509 *cert) {
366 if (!ssl_ctx || !cert) return false;
367
368 X509 *leaf = SSL_CTX_get0_certificate(ssl_ctx);
369 if (leaf && X509_cmp(leaf, cert) == 0) return true;
370
371 STACK_OF(X509) *chain = NULL;
372 if (SSL_CTX_get0_chain_certs(ssl_ctx, &chain) == 1 && chain) {
373 for (size_t i = 0; i < sk_X509_num(chain); i++) {
374 if (X509_cmp(sk_X509_value(chain, i), cert) == 0) return true;
375 }
376 }
377 return false;

428static char *cwist_dup_aia_uri(const ACCESS_DESCRIPTION *desc) {
429 if (!desc || OBJ_obj2nid(desc->method) != NID_ad_ca_issuers || !desc->location) {
430 return NULL;
431 }
432
433 int name_type = -1;
434 ASN1_IA5STRING *uri = (ASN1_IA5STRING *)GENERAL_NAME_get0_value(desc->location, &name_type);
435 if (name_type != GEN_URI || !uri) return NULL;
436
437 int uri_len = ASN1_STRING_length(uri);
438 if (uri_len <= 0) return NULL;
439
440 const unsigned char *uri_data = ASN1_STRING_get0_data(uri);
441 if (!uri_data || memchr(uri_data, '\0', (size_t)uri_len)) return NULL;
442
443 char *url = (char *)malloc((size_t)uri_len + 1);
444 if (!url) return NULL;
445 memcpy(url, uri_data, (size_t)uri_len);
446 url[uri_len] = '\0';
447
448 if (!cwist_url_has_http_scheme(url)) {

537unsigned long cwist_tls_sess_cache_size(void) {
538 static _Atomic long cached = -1;
539 long v = atomic_load_explicit(&cached, memory_order_relaxed);
540 if (v < 0) {

568int cwist_tls_autoload_intermediates(SSL_CTX *ssl_ctx) {
569 if (!ssl_ctx) return -1;
570
571 int added = 0;
572 for (int depth = 0; depth < CWIST_AIA_MAX_CHAIN_DEPTH; depth++) {
573 X509 *tail = cwist_ctx_chain_tail(ssl_ctx);
574 if (!tail || cwist_x509_is_self_signed(tail)) break;
575
576 X509 *issuer = cwist_fetch_issuer_from_aia(tail);
577 if (!issuer) break;
578
579 if (cwist_x509_is_self_signed(issuer)) {
580 X509_free(issuer);
581 break;
582 }
583
584 if (cwist_ctx_contains_cert(ssl_ctx, issuer)) {
585 X509_free(issuer);
586 break;
587 }
588
589 if (SSL_CTX_add1_chain_cert(ssl_ctx, issuer) != 1) {
590 X509_free(issuer);
591 return -1;
592 }
593
594 X509_free(issuer);
595 added++;
596 }
597
598 return added;
599}
