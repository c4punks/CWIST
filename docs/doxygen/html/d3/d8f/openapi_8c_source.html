49static cwist_http_method_t method_from_string(const char *method) {
50 static const cwist_http_method_t all[] = {
51 CWIST_HTTP_GET, CWIST_HTTP_POST, CWIST_HTTP_PUT, CWIST_HTTP_DELETE,
52 CWIST_HTTP_PATCH, CWIST_HTTP_HEAD, CWIST_HTTP_OPTIONS,
53 };
54 for (size_t i = 0; i < sizeof(all) / sizeof(all[0]); i++) {
55 if (strcasecmp(method, cwist_http_method_to_string(all[i])) == 0) return all[i];
56 }
57 return CWIST_HTTP_UNKNOWN;
58}

103/* "/users/:id" -> "/users/{id}", with one required path parameter per segment. */
104static char *openapi_path(const char *route, cJSON *params) {
105 size_t len = strlen(route);
106 char *out = cwist_alloc(len * 2 + 1);
107 if (!out) return NULL;
108 char *w = out;
109 for (const char *r = route; *r;) {
110 if (*r != ':' || (r != route && r[-1] != '/')) {
111 *w++ = *r++;
112 continue;
113 }
114 const char *name = ++r;
115 while (*r && *r != '/') r++;
116 *w++ = '{';
117 memcpy(w, name, (size_t)(r - name));
118 w += r - name;
119 *w++ = '}';
120
121 char *pname = cwist_strndup(name, (size_t)(r - name));
122 cJSON *p = cJSON_CreateObject();
123 cJSON_AddStringToObject(p, "name", pname ? pname : "");
124 cwist_free(pname);
125 cJSON_AddStringToObject(p, "in", "path");
126 cJSON_AddBoolToObject(p, "required", 1);
127 cJSON_AddStringToObject(cJSON_AddObjectToObject(p, "schema"), "type", "string");
128 cJSON_AddItemToArray(params, p);
129 }
130 *w = '\0';
131 return out;
132}

139static void add_operation(void *ctx, cwist_http_method_t method, const char *route) {
140 build_ctx *b = ctx;
141 if (method == CWIST_HTTP_CONNECT || method == CWIST_HTTP_UNKNOWN) return;
142 if (method == CWIST_HTTP_GET && strcmp(route, b->doc->json_path) == 0) return;
143
144 cJSON *params = cJSON_CreateArray();
145 char *path = openapi_path(route, params);
146 if (!path) {
147 cJSON_Delete(params);
148 return;
149 }
150 cJSON *item = cJSON_GetObjectItemCaseSensitive(b->paths, path);
151 if (!item) item = cJSON_AddObjectToObject(b->paths, path);
152 cwist_free(path);
153
154 char verb[16];
155 snprintf(verb, sizeof(verb), "%s", cwist_http_method_to_string(method));
156 for (char *c = verb; *c; c++) *c = (char)(*c | 0x20);
157
158 cJSON *op = cJSON_AddObjectToObject(item, verb);
159 const openapi_note *note = find_note(b->doc, method, route);
160 if (note && note->summary) cJSON_AddStringToObject(op, "summary", note->summary);
161 if (note && note->tags) {
162 cJSON *tags = cJSON_AddArrayToObject(op, "tags");
163 char *list = cwist_strdup(note->tags);
164 char *save = NULL;
165 for (char *t = list ? strtok_r(list, ",", &save) : NULL; t; t = strtok_r(NULL, ",", &save)) {
166 while (*t == ' ') t++;
167 if (*t) cJSON_AddItemToArray(tags, cJSON_CreateString(t));
168 }
169 cwist_free(list);
170 }
171 if (cJSON_GetArraySize(params) > 0)
172 cJSON_AddItemToObject(op, "parameters", params);
173 else
174 cJSON_Delete(params);
175 cJSON *ok = cJSON_AddObjectToObject(cJSON_AddObjectToObject(op, "responses"), "200");
176 cJSON_AddStringToObject(ok, "description", "OK");
177}

179static void openapi_handler(void *ctx, cwist_http_request *req, cwist_http_response *res) {
180 (void)req;
181 const openapi_doc *doc = ctx;
182
183 cJSON *root = cJSON_CreateObject();
184 cJSON_AddStringToObject(root, "openapi", "3.1.0");
185 cJSON *info = cJSON_AddObjectToObject(root, "info");
186 cJSON_AddStringToObject(info, "title", doc->title ? doc->title : "CWIST API");
187 cJSON_AddStringToObject(info, "version", doc->version ? doc->version : "1.0.0");
188 if (doc->description) cJSON_AddStringToObject(info, "description", doc->description);
189 cJSON *server = cJSON_CreateObject();
190 cJSON_AddStringToObject(server, "url", doc->server_url ? doc->server_url : "/");
191 cJSON_AddItemToArray(cJSON_AddArrayToObject(root, "servers"), server);
192
193 build_ctx b = {.doc = doc, .paths = cJSON_AddObjectToObject(root, "paths")};
194 pthread_mutex_lock(&g_docs_mu);
195 cwist_app_foreach_route(doc->app, add_operation, &b);
196 pthread_mutex_unlock(&g_docs_mu);
197
198 char *json = cJSON_PrintUnformatted(root);
199 cJSON_Delete(root);
200 if (!json) {
201 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
202 return;
203 }
204 cwist_sstring_assign(res->body, json);
205 cJSON_free(json);
206 res->status_code = CWIST_HTTP_OK;
207 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
208}

210void cwist_app_enable_openapi(cwist_app *app, const char *json_path,
211 const cwist_openapi_meta_t *meta) {
212 if (!app) return;
213 pthread_mutex_lock(&g_docs_mu);
214 bool enabled = find_doc(app) != NULL;
215 pthread_mutex_unlock(&g_docs_mu);
216 if (enabled) return;
217
218 openapi_doc *doc = cwist_alloc(sizeof(*doc));
219 if (!doc) return;
220 doc->app = app;
221 doc->json_path = cwist_strdup(json_path ? json_path : "/openapi.json");
222 if (meta) {
223 doc->title = dup_or_null(meta->title);
224 doc->version = dup_or_null(meta->version);
225 doc->description = dup_or_null(meta->description);
226 doc->server_url = dup_or_null(meta->server_url);
227 }
228 if (!doc->json_path) {
229 cwist_free(doc);
230 return;
231 }
232
233 pthread_mutex_lock(&g_docs_mu);
234 doc->next = g_docs;
235 g_docs = doc;
236 pthread_mutex_unlock(&g_docs_mu);
237
238 cwist_error_t err = cwist_app_get_ex(app, doc->json_path, openapi_handler, doc, doc_destroy);
239 if (!cwist_error_is_ok(&err)) doc_destroy(doc);
240}

242void cwist_openapi_annotate(cwist_app *app, const char *method, const char *path,
243 const char *summary, const char *tags) {
244 if (!app || !method || !path) return;
245 cwist_http_method_t m = method_from_string(method);
246 if (m == CWIST_HTTP_UNKNOWN) return;
247
248 pthread_mutex_lock(&g_docs_mu);
249 openapi_doc *doc = find_doc(app);
250 openapi_note *note = doc ? (openapi_note *)find_note(doc, m, path) : NULL;
251 if (doc && !note && (note = cwist_alloc(sizeof(*note)))) {
252 note->method = m;
253 note->path = cwist_strdup(path);
254 if (!note->path) {
255 cwist_free(note);
256 note = NULL;
257 } else {
258 note->next = doc->notes;
259 doc->notes = note;
260 }
261 }
262 if (note) {
263 cwist_free(note->summary);
264 cwist_free(note->tags);
265 note->summary = dup_or_null(summary);
266 note->tags = dup_or_null(tags);
267 }
268 pthread_mutex_unlock(&g_docs_mu);
269}
