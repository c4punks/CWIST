56static char *normalize_prefix(const char *url_prefix) {
57 if (!url_prefix || url_prefix[0] != '/') return NULL;
58 size_t len = strlen(url_prefix);
59 while (len > 0 && url_prefix[len - 1] == '/') len--;
60 for (size_t i = 0; i < len; i++) {
61 unsigned char c = (unsigned char)url_prefix[i];
62 if (c <= 0x20 || c == 0x7f || c == '?' || c == '#' || c == '%') return NULL;
63 }
64 char *out = (char *)cwist_alloc(len + 1);
65 if (!out) return NULL;
66 memcpy(out, url_prefix, len);
67 out[len] = '\0';
68 return out;
69}

72static bool name_is_valid(const char *name) {
73 if (!name || !*name || name[0] == '/') return false;
74 const char *seg = name;
75 for (const char *p = name;; p++) {
76 char c = *p;
77 if (c == '/' || c == '\0') {
78 size_t seg_len = (size_t)(p - seg);
79 if (seg_len == 0) return false;
80 if (seg_len == 1 && seg[0] == '.') return false;
81 if (seg_len == 2 && seg[0] == '.' && seg[1] == '.') return false;
82 if (c == '\0') return true;
83 seg = p + 1;
84 continue;
85 }
86 bool ok = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') ||
87 c == '.' || c == '-' || c == '_' || c == '~';
88 if (!ok) return false;
89 }
90}

93static size_t extension_offset(const char *name) {
94 const char *base = strrchr(name, '/');
95 base = base ? base + 1 : name;
96 const char *dot = strrchr(base, '.');
97 /* A leading dot names a dotfile, not an extension. */
98 if (!dot || dot == base) return strlen(name);
99 return (size_t)(dot - name);

103static const char *guess_content_type(const char *name) {
104 static const struct {
105 const char *ext;
106 const char *type;
107 } table[] = {
108 {".css", "text/css; charset=utf-8"},
109 {".js", "text/javascript; charset=utf-8"},
110 {".mjs", "text/javascript; charset=utf-8"},
111 {".json", "application/json"},
112 {".map", "application/json"},
113 {".html", "text/html; charset=utf-8"},
114 {".htm", "text/html; charset=utf-8"},
115 {".txt", "text/plain; charset=utf-8"},
116 {".svg", "image/svg+xml"},
117 {".png", "image/png"},
118 {".jpg", "image/jpeg"},
119 {".jpeg", "image/jpeg"},
120 {".gif", "image/gif"},
121 {".webp", "image/webp"},
122 {".ico", "image/x-icon"},
123 {".woff", "font/woff"},
124 {".woff2", "font/woff2"},
125 {".wasm", "application/wasm"},
126 };
127 const char *ext = name + extension_offset(name);
128 for (size_t i = 0; i < sizeof(table) / sizeof(table[0]); i++) {
129 const char *a = ext, *b = table[i].ext;
130 while (*a && *b && (*a | 0x20) == *b) {
131 a++;
132 b++;
133 }
134 if (*a == '\0' && *b == '\0') return table[i].type;
135 }
136 return "application/octet-stream";
137}

140static void content_hash(const void *data, size_t len, char out[ASSET_HASH_HEX + 1]) {
141 static const char hex[] = "0123456789abcdef";
142 uint8_t digest[CWIST_SHA256_DIGEST_LEN];
143 cwist_sha256_ctx ctx;
144 cwist_sha256_init(&ctx);
145 if (len > 0) cwist_sha256_update(&ctx, (const uint8_t *)data, len);
146 cwist_sha256_final(&ctx, digest);
147 for (size_t i = 0; i < ASSET_HASH_HEX / 2; i++) {
148 out[2 * i] = hex[digest[i] >> 4];
149 out[2 * i + 1] = hex[digest[i] & 0xf];
150 }
151 out[ASSET_HASH_HEX] = '\0';
152}

206static struct cwist_asset *asset_build(const cwist_asset_registry *reg, const char *name,
207 unsigned char *data, size_t len, const char *content_type,
208 const char *hash) {
209 struct cwist_asset *asset = (struct cwist_asset *)cwist_alloc(sizeof(*asset));
210 if (!asset) {
211 cwist_free(data);
212 return NULL;
213 }
214 asset->data = data;
215 asset->len = len;
216
217 size_t ext = extension_offset(name);
218 char *stem = (char *)cwist_alloc(ext + 1);
219 if (stem) {
220 memcpy(stem, name, ext);
221 stem[ext] = '\0';
222 }
223 char *stem_hash = stem ? join3(stem, ".", hash) : NULL;
224 cwist_free(stem);
225
226 asset->name = cwist_strdup(name);
227 asset->hashed_name = stem_hash ? join3(stem_hash, name + ext, "") : NULL;
228 cwist_free(stem_hash);
229 asset->url = asset->hashed_name ? join3(reg->prefix, "/", asset->hashed_name) : NULL;
230 asset->content_type = cwist_strdup(content_type ? content_type : guess_content_type(name));
231 snprintf(asset->etag, sizeof(asset->etag), "\"%s\"", hash);
232
233 if (!asset->name || !asset->hashed_name || !asset->url || !asset->content_type) {
234 asset_free(asset);
235 return NULL;
236 }
237 return asset;
238}

249static cwist_error_t asset_publish(cwist_app *app, const char *name, unsigned char *data,
250 size_t len, const char *content_type) {
251 cwist_asset_registry *reg = registry_get(app, true);
252 if (!reg) {
253 cwist_free(data);
254 return asset_result(-1);
255 }
256
257 char hash[ASSET_HASH_HEX + 1];
258 content_hash(data, len, hash);
259
260 /* The newest asset under this name already has this content and type:
261 * nothing to publish. An older match does not count, since the name
262 * currently resolves to something else. */
263 const char *type = content_type ? content_type : guess_content_type(name);
264 for (struct cwist_asset *a = atomic_load_explicit(&reg->head, memory_order_acquire); a;
265 a = a->next) {
266 if (strcmp(a->name, name) != 0) continue;
267 if (memcmp(a->etag + 1, hash, ASSET_HASH_HEX) == 0 && a->len == len &&
268 (len == 0 || memcmp(a->data, data, len) == 0) && strcmp(a->content_type, type) == 0) {
269 cwist_free(data);
270 return asset_result(0);
271 }
272 break;
273 }
274
275 struct cwist_asset *asset = asset_build(reg, name, data, len, content_type, hash);
276 if (!asset) return asset_result(-1);
277
278 struct cwist_asset *head = atomic_load_explicit(&reg->head, memory_order_relaxed);
279 do {
280 asset->next = head;
281 } while (!atomic_compare_exchange_weak_explicit(&reg->head, &head, asset, memory_order_release,
282 memory_order_relaxed));
283 return asset_result(0);
284}

325cwist_error_t cwist_app_asset_add_file(cwist_app *app, const char *name, const char *path,
326 const char *content_type) {
327 if (!app || !name_is_valid(name) || !path) return asset_result(-1);
328 if (content_type && (!*content_type || !header_value_is_valid(content_type))) {
329 return asset_result(-1);
330 }
331 FILE *f = fopen(path, "rb");
332 if (!f) return asset_result(-1);
333
334 size_t cap = 4096, len = 0;
335 unsigned char *buf = (unsigned char *)cwist_alloc(cap);
336 bool ok = buf != NULL;
337 while (ok) {
338 if (len == cap) {
339 unsigned char *grown = (unsigned char *)cwist_realloc(buf, cap * 2);
340 if (!grown) {
341 ok = false;
342 break;
343 }
344 buf = grown;
345 cap *= 2;
346 }
347 size_t n = fread(buf + len, 1, cap - len, f);
348 len += n;
349 if (n == 0) {
350 ok = !ferror(f);
351 break;
352 }
353 }
354 fclose(f);
355 if (!ok) {
356 cwist_free(buf);
357 return asset_result(-1);
358 }
359 return asset_publish(app, name, buf, len, content_type);
360}

392bool cwist_assets_match(cwist_app *app, const cwist_http_request *req, cwist_asset_match *out) {
393 if (!app || !app->assets || !req || !req->path || !req->path->data) return false;
394 if (req->method != CWIST_HTTP_GET && req->method != CWIST_HTTP_HEAD) return false;
395 cwist_asset_registry *reg = (cwist_asset_registry *)app->assets;
396
397 const char *path = req->path->data;
398 size_t plen = strlen(reg->prefix);
399 if (strncmp(path, reg->prefix, plen) != 0 || path[plen] != '/') return false;
400 const char *rest = path + plen + 1;
401
402 const struct cwist_asset *logical = NULL;
403 for (struct cwist_asset *a = atomic_load_explicit(&reg->head, memory_order_acquire); a;
404 a = a->next) {
405 if (strcmp(a->hashed_name, rest) == 0) {
406 out->asset = a;
407 out->immutable = true;
408 return true;
409 }
410 if (!logical && strcmp(a->name, rest) == 0) logical = a;
411 }
412 if (!logical) return false;
413 out->asset = logical;
414 out->immutable = false;
415 return true;

419static bool etag_matches(const char *if_none_match, const char *etag) {
420 size_t elen = strlen(etag);
421 const char *p = if_none_match;
422 while (*p) {
423 while (*p == ' ' || *p == '\t' || *p == ',') p++;
424 if (!*p) break;
425 const char *start = p;
426 while (*p && *p != ',') p++;
427 const char *end = p;
428 while (end > start && (end[-1] == ' ' || end[-1] == '\t')) end--;
429 if (end - start == 1 && *start == '*') return true;
430 if (end - start > 2 && start[0] == 'W' && start[1] == '/') start += 2;
431 if ((size_t)(end - start) == elen && memcmp(start, etag, elen) == 0) return true;
432 }
433 return false;
434}

455void cwist_assets_respond(cwist_http_request *req, cwist_http_response *res,
456 const cwist_asset_match *match) {
457 if (!req || !res) return;
458 if (!match || !match->asset) {
459 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
460 return;
461 }
462 const struct cwist_asset *asset = match->asset;
463 const char *cache = match->immutable ? CWIST_ASSET_IMMUTABLE_CACHE : "no-cache";
464 const char *inm = cwist_http_header_get(req->headers, "If-None-Match");
465 bool not_modified = inm && etag_matches(inm, asset->etag);
466
467 bool ok = add_header(res, "ETag", asset->etag) && add_header(res, "Cache-Control", cache);
468 if (ok && !not_modified) ok = add_header(res, "Content-Type", asset->content_type);
469 if (!ok) {
470 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
471 return;
472 }
473
474 if (not_modified) {
475 res->status_code = CWIST_HTTP_NOT_MODIFIED;
476 cwist_error_t err = cwist_sstring_assign(res->body, "");
477 cwist_error_dispose(&err);
478 return;
479 }
480 res->status_code = CWIST_HTTP_OK;
481 /* Assets live until the app is destroyed, so no release hook is needed. */
482 cwist_http_response_set_body_ptr(res, asset->data, asset->len);
483}

494int cwist_assets_clone(void **dst, const void *src) {
495 *dst = NULL;
496 if (!src) return 0;
497 const cwist_asset_registry *from = (const cwist_asset_registry *)src;
498 cwist_asset_registry *reg = (cwist_asset_registry *)cwist_alloc(sizeof(*reg));
499 if (!reg) return -1;
500 reg->prefix = cwist_strdup(from->prefix);
501 atomic_init(&reg->head, NULL);
502 if (!reg->prefix) {
503 cwist_free(reg);
504 return -1;
505 }
506
507 /* Keep the newest-first order: append each copy at the tail. */
508 struct cwist_asset *head = NULL, **tail = &head;
509 bool ok = true;
510 for (struct cwist_asset *a =
511 atomic_load_explicit(&((cwist_asset_registry *)from)->head, memory_order_acquire);
512 a && ok; a = a->next) {
513 unsigned char *data = (unsigned char *)cwist_alloc(a->len ? a->len : 1);
514 if (!data) {
515 ok = false;
516 break;
517 }
518 if (a->len) memcpy(data, a->data, a->len);
519 char hash[ASSET_HASH_HEX + 1];
520 memcpy(hash, a->etag + 1, ASSET_HASH_HEX);
521 hash[ASSET_HASH_HEX] = '\0';
522 struct cwist_asset *copy = asset_build(reg, a->name, data, a->len, a->content_type, hash);
523 if (!copy) {
524 ok = false;
525 break;
526 }
527 copy->next = NULL;
528 *tail = copy;
529 tail = &copy->next;
530 }
531 atomic_init(&reg->head, head);
532 if (!ok) {
533 cwist_assets_destroy(reg);
534 return -1;
535 }
536 *dst = reg;
537 return 0;
538}
