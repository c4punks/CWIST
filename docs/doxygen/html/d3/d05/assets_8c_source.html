56static char *normalize_prefix(const char *url_prefix) {
57 if (!url_prefix || url_prefix[0] != '/') return NULL;
58 size_t len = strlen(url_prefix);
59 while (len > 0 && url_prefix[len - 1] == '/') len--;
60
61 for (size_t i = 0; i < len; i++) {
62 unsigned char c = (unsigned char)url_prefix[i];
63 if (c <= 0x20 || c == 0x7f || c == '?' || c == '#' || c == '%') return NULL;
64 }
65
66 char *out = (char *)cwist_alloc(len + 1);
67 if (!out) return NULL;
68
69 memcpy(out, url_prefix, len);
70 out[len] = '\0';
71 return out;
72}

75static bool name_is_valid(const char *name) {
76 if (!name || !*name || name[0] == '/') return false;
77 const char *seg = name;
78 for (const char *p = name;; p++) {
79 char c = *p;
80 if (c == '/' || c == '\0') {
81 size_t seg_len = (size_t)(p - seg);
82 if (seg_len == 0) return false;
83 if (seg_len == 1 && seg[0] == '.') return false;
84 if (seg_len == 2 && seg[0] == '.' && seg[1] == '.') return false;
85 if (c == '\0') return true;
86 seg = p + 1;
87 continue;
88 }
89 bool ok = (c >= 'a' && c <= 'z')
90 || (c >= 'A' && c <= 'Z')
91 || (c >= '0' && c <= '9')
92 || c == '.' || c == '-'
93 || c == '_' || c == '~';
94 if (!ok) return false;
95 }
96}

99static size_t extension_offset(const char *name) {
100 const char *base = strrchr(name, '/');
101 base = base ? base + 1 : name;
102 const char *dot = strrchr(base, '.');
103 /* A leading dot names a dotfile, not an extension. */
104 if (!dot || dot == base) return strlen(name);
105 return (size_t)(dot - name);

109static const char *guess_content_type(const char *name) {
110 static const struct {
111 const char *ext;
112 const char *type;
113 } table[] = {
114 {".css", "text/css; charset=utf-8"},
115 {".js", "text/javascript; charset=utf-8"},
116 {".mjs", "text/javascript; charset=utf-8"},
117 {".json", "application/json"},
118 {".map", "application/json"},
119 {".html", "text/html; charset=utf-8"},
120 {".htm", "text/html; charset=utf-8"},
121 {".txt", "text/plain; charset=utf-8"},
122 {".svg", "image/svg+xml"},
123 {".png", "image/png"},
124 {".jpg", "image/jpeg"},
125 {".jpeg", "image/jpeg"},
126 {".gif", "image/gif"},
127 {".webp", "image/webp"},
128 {".ico", "image/x-icon"},
129 {".woff", "font/woff"},
130 {".woff2", "font/woff2"},
131 {".wasm", "application/wasm"},
132 };
133 const char *ext = name + extension_offset(name);
134 for (size_t i = 0; i < sizeof(table) / sizeof(table[0]); i++) {
135 const char *a = ext, *b = table[i].ext;
136 while (*a && *b && (*a | 0x20) == *b) {
137 a++;
138 b++;
139 }
140 if (*a == '\0' && *b == '\0') return table[i].type;
141 }
142 return "application/octet-stream";
143}

146static void content_hash(const void *data, size_t len, char out[ASSET_HASH_HEX + 1]) {
147 static const char hex[] = "0123456789abcdef";
148 uint8_t digest[CWIST_SHA256_DIGEST_LEN];
149 cwist_sha256_ctx ctx;
150 cwist_sha256_init(&ctx);
151 if (len > 0) cwist_sha256_update(&ctx, (const uint8_t *)data, len);
152 cwist_sha256_final(&ctx, digest);
153 for (size_t i = 0; i < ASSET_HASH_HEX / 2; i++) {
154 out[2 * i] = hex[digest[i] >> 4];
155 out[2 * i + 1] = hex[digest[i] & 0xf];
156 }
157 out[ASSET_HASH_HEX] = '\0';
158}

212static struct cwist_asset *asset_build(const cwist_asset_registry *reg, const char *name,
213 unsigned char *data, size_t len, const char *content_type,
214 const char *hash) {
215 struct cwist_asset *asset = (struct cwist_asset *)cwist_alloc(sizeof(*asset));
216 if (!asset) {
217 cwist_free(data);
218 return NULL;
219 }
220 asset->data = data;
221 asset->len = len;
222
223 size_t ext = extension_offset(name);
224 char *stem = (char *)cwist_alloc(ext + 1);
225 if (stem) {
226 memcpy(stem, name, ext);
227 stem[ext] = '\0';
228 }
229 char *stem_hash = stem ? join3(stem, ".", hash) : NULL;
230 cwist_free(stem);
231
232 asset->name = cwist_strdup(name);
233 asset->hashed_name = stem_hash ? join3(stem_hash, name + ext, "") : NULL;
234 cwist_free(stem_hash);
235 asset->url = asset->hashed_name ? join3(reg->prefix, "/", asset->hashed_name) : NULL;
236 asset->content_type = cwist_strdup(content_type ? content_type : guess_content_type(name));
237 snprintf(asset->etag, sizeof(asset->etag), "\"%s\"", hash);
238
239 if (!asset->name || !asset->hashed_name || !asset->url || !asset->content_type) {
240 asset_free(asset);
241 return NULL;
242 }
243 return asset;
244}

255static cwist_error_t asset_publish(cwist_app *app, const char *name, unsigned char *data,
256 size_t len, const char *content_type) {
257 cwist_asset_registry *reg = registry_get(app, true);
258 if (!reg) {
259 cwist_free(data);
260 return asset_result(-1);
261 }
262
263 char hash[ASSET_HASH_HEX + 1];
264 content_hash(data, len, hash);
265
266 /* The newest asset under this name already has this content and type:
267 * nothing to publish. An older match does not count, since the name
268 * currently resolves to something else. */
269 const char *type = content_type ? content_type : guess_content_type(name);
270 for (struct cwist_asset *a = atomic_load_explicit(&reg->head, memory_order_acquire); a;
271 a = a->next) {
272 if (strcmp(a->name, name) != 0) continue;
273 if (memcmp(a->etag + 1, hash, ASSET_HASH_HEX) == 0 && a->len == len &&
274 (len == 0 || memcmp(a->data, data, len) == 0) && strcmp(a->content_type, type) == 0) {
275 cwist_free(data);
276 return asset_result(0);
277 }
278 break;
279 }
280
281 struct cwist_asset *asset = asset_build(reg, name, data, len, content_type, hash);
282 if (!asset) return asset_result(-1);
283
284 struct cwist_asset *head = atomic_load_explicit(&reg->head, memory_order_relaxed);
285 do {
286 asset->next = head;
287 } while (!atomic_compare_exchange_weak_explicit(&reg->head, &head, asset, memory_order_release,
288 memory_order_relaxed));
289 return asset_result(0);
290}

336cwist_error_t cwist_app_asset_add_file(cwist_app *app, const char *name, const char *path,
337 const char *content_type) {
338 if (!app || !name_is_valid(name) || !path) return asset_result(-1);
339 if (content_type && (!*content_type || !header_value_is_valid(content_type))) {
340 return asset_result(-1);
341 }
342
343 FILE *f = fopen(path, "rb");
344 if (!f) return asset_result(-1);
345
346 size_t cap = 4096, len = 0;
347 unsigned char *buf = (unsigned char *)cwist_alloc(cap);
348 bool ok = buf != NULL;
349
350 while (ok) {
351 if (len == cap) {
352 unsigned char *grown = (unsigned char *)cwist_realloc(buf, cap * 2);
353 if (!grown) {
354 ok = false;
355 break;
356 }
357 buf = grown;
358 cap *= 2;
359 }
360 size_t n = fread(buf + len, 1, cap - len, f);
361 len += n;
362 if (n == 0) {
363 ok = !ferror(f);
364 break;
365 }
366 }
367 fclose(f);
368
369 if (!ok) {
370 cwist_free(buf);
371 return asset_result(-1);
372 }
373
374 return asset_publish(app, name, buf, len, content_type);
375}

410bool cwist_assets_match(cwist_app *app, const cwist_http_request *req, cwist_asset_match *out) {
411 if (!app || !app->assets || !req || !req->path || !req->path->data) return false;
412 if (req->method != CWIST_HTTP_GET && req->method != CWIST_HTTP_HEAD) return false;
413 cwist_asset_registry *reg = (cwist_asset_registry *)app->assets;
414
415 const char *path = req->path->data;
416 size_t plen = strlen(reg->prefix);
417 if (strncmp(path, reg->prefix, plen) != 0 || path[plen] != '/') return false;
418 const char *rest = path + plen + 1;
419
420 const struct cwist_asset *logical = NULL;
421 for (struct cwist_asset *a = atomic_load_explicit(&reg->head, memory_order_acquire); a;
422 a = a->next) {
423 if (strcmp(a->hashed_name, rest) == 0) {
424 out->asset = a;
425 out->immutable = true;
426 return true;
427 }
428 if (!logical && strcmp(a->name, rest) == 0) logical = a;
429 }
430
431 if (!logical) return false;
432 out->asset = logical;
433 out->immutable = false;
434
435 return true;
436}

439static bool etag_matches(const char *if_none_match, const char *etag) {
440 size_t elen = strlen(etag);
441 const char *p = if_none_match;
442 while (*p) {
443 while (*p == ' ' || *p == '\t' || *p == ',') p++;
444 if (!*p) break;
445 const char *start = p;
446 while (*p && *p != ',') p++;
447 const char *end = p;
448 while (end > start && (end[-1] == ' ' || end[-1] == '\t')) end--;
449 if (end - start == 1 && *start == '*') return true;
450 if (end - start > 2 && start[0] == 'W' && start[1] == '/') start += 2;
451 if ((size_t)(end - start) == elen && memcmp(start, etag, elen) == 0) return true;
452 }
453 return false;
454}

475void cwist_assets_respond(cwist_http_request *req, cwist_http_response *res,
476 const cwist_asset_match *match) {
477 if (!req || !res) return;
478 if (!match || !match->asset) {
479 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
480 return;
481 }
482 const struct cwist_asset *asset = match->asset;
483 const char *cache = match->immutable ? CWIST_ASSET_IMMUTABLE_CACHE : "no-cache";
484 const char *inm = cwist_http_header_get(req->headers, "If-None-Match");
485 bool not_modified = inm && etag_matches(inm, asset->etag);
486
487 bool ok = assets_add_header(res, "ETag", asset->etag) && assets_add_header(res, "Cache-Control", cache);
488 if (ok && !not_modified) ok = assets_add_header(res, "Content-Type", asset->content_type);
489 if (!ok) {
490 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
491 return;
492 }
493
494 if (not_modified) {
495 res->status_code = CWIST_HTTP_NOT_MODIFIED;
496 cwist_error_t err = cwist_sstring_assign(res->body, "");
497 cwist_error_dispose(&err);
498 return;
499 }
500 res->status_code = CWIST_HTTP_OK;
501 /* Assets live until the app is destroyed, so no release hook is needed. */
502 cwist_http_response_set_body_ptr(res, asset->data, asset->len);
503}

515 *dst = NULL;
516 if (!src) return 0;
517
518 const cwist_asset_registry *from = (const cwist_asset_registry *)src;
519 cwist_asset_registry *reg = (cwist_asset_registry *)cwist_alloc(sizeof(*reg));
520 if (!reg) return -1;

522 reg->prefix = cwist_strdup(from->prefix);
523 atomic_init(&reg->head, NULL);
524
525 if (!reg->prefix) {

528 }

531 struct cwist_asset *head = NULL, **tail = &head;
532 bool ok = true;

534 atomic_load_explicit(&((cwist_asset_registry *)from)->head, memory_order_acquire);
535 a && ok; a = a->next) {
536 unsigned char *data = (unsigned char *)cwist_alloc(a->len ? a->len : 1);
537 if (!data) {
538 ok = false;
539 break;
540 }
541
542 if (a->len) memcpy(data, a->data, a->len);
543

545 memcpy(hash, a->etag + 1, ASSET_HASH_HEX);
