46static char *join3(const char *a, const char *b, const char *c) {
47 size_t la = strlen(a), lb = strlen(b), lc = strlen(c);

49 if (!out) return NULL;
50 memcpy(out, a, la);
51 memcpy(out + la, b, lb);
52 memcpy(out + la + lb, c, lc + 1);

57static char *normalize_prefix(const char *url_prefix) {
58 if (!url_prefix || url_prefix[0] != '/') return NULL;
59 size_t len = strlen(url_prefix);
60 while (len > 0 && url_prefix[len - 1] == '/') len--;
61
62 for (size_t i = 0; i < len; i++) {
63 unsigned char c = (unsigned char)url_prefix[i];
64 if (c <= 0x20 || c == 0x7f || c == '?' || c == '#' || c == '%') return NULL;
65 }
66
67 char *out = (char *)cwist_alloc(len + 1);
68 if (!out) return NULL;
69
70 memcpy(out, url_prefix, len);
71 out[len] = '\0';
72 return out;
73}

77static bool name_is_valid(const char *name) {
78 if (!name || !*name || name[0] == '/') return false;
79 const char *seg = name;
80 for (const char *p = name;; p++) {
81 char c = *p;
82 if (c == '/' || c == '\0') {
83 size_t seg_len = (size_t)(p - seg);
84 if (seg_len == 0) return false;
85 if (seg_len == 1 && seg[0] == '.') return false;
86 if (seg_len == 2 && seg[0] == '.' && seg[1] == '.') return false;
87 if (c == '\0') return true;
88 seg = p + 1;
89 continue;
90 }
91 bool ok = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') ||
92 c == '.' || c == '-' || c == '_' || c == '~';
93 if (!ok) return false;
94 }
95}

98static size_t extension_offset(const char *name) {
99 const char *base = strrchr(name, '/');
100 base = base ? base + 1 : name;
101 const char *dot = strrchr(base, '.');
102 /* A leading dot names a dotfile, not an extension. */
103 if (!dot || dot == base) return strlen(name);
104 return (size_t)(dot - name);
105}

108static const char *guess_content_type(const char *name) {
109 static const struct {
110 const char *ext;
111 const char *type;
112 } table[] = {
113 {".css", "text/css; charset=utf-8"},
114 {".js", "text/javascript; charset=utf-8"},
115 {".mjs", "text/javascript; charset=utf-8"},
116 {".json", "application/json"},
117 {".map", "application/json"},
118 {".html", "text/html; charset=utf-8"},
119 {".htm", "text/html; charset=utf-8"},
120 {".txt", "text/plain; charset=utf-8"},
121 {".svg", "image/svg+xml"},
122 {".png", "image/png"},
123 {".jpg", "image/jpeg"},
124 {".jpeg", "image/jpeg"},
125 {".gif", "image/gif"},
126 {".webp", "image/webp"},
127 {".ico", "image/x-icon"},
128 {".woff", "font/woff"},
129 {".woff2", "font/woff2"},
130 {".wasm", "application/wasm"},
131 };
132 const char *ext = name + extension_offset(name);
133 for (size_t i = 0; i < sizeof(table) / sizeof(table[0]); i++) {
134 const char *a = ext, *b = table[i].ext;
135 while (*a && *b && (*a | 0x20) == *b) {
136 a++;
137 b++;
138 }
139 if (*a == '\0' && *b == '\0') return table[i].type;
140 }
141 return "application/octet-stream";
142}

145static void content_hash(const void *data, size_t len, char out[ASSET_HASH_HEX + 1]) {
146 static const char hex[] = "0123456789abcdef";
147 uint8_t digest[CWIST_SHA256_DIGEST_LEN];
148 cwist_sha256_ctx ctx;
149 cwist_sha256_init(&ctx);
150 if (len > 0) cwist_sha256_update(&ctx, (const uint8_t *)data, len);
151 cwist_sha256_final(&ctx, digest);
152 for (size_t i = 0; i < ASSET_HASH_HEX / 2; i++) {
153 out[2 * i] = hex[digest[i] >> 4];
154 out[2 * i + 1] = hex[digest[i] & 0xf];
155 }
156 out[ASSET_HASH_HEX] = '\0';
157}

177 pthread_mutex_unlock(&g_registry_lock);

230static struct cwist_asset *asset_build(const cwist_asset_registry *reg, const char *name,
231 unsigned char *data, size_t len, const char *content_type,
232 const char *hash) {
233 struct cwist_asset *asset = (struct cwist_asset *)cwist_alloc(sizeof(*asset));
234 if (!asset) {
235 cwist_free(data);
236 return NULL;
237 }
238 asset->data = data;
239 asset->len = len;
240
241 size_t ext = extension_offset(name);
242 char *stem = (char *)cwist_alloc(ext + 1);
243 if (stem) {
244 memcpy(stem, name, ext);
245 stem[ext] = '\0';
246 }
247 char *stem_hash = stem ? join3(stem, ".", hash) : NULL;
248 cwist_free(stem);
249
250 asset->name = cwist_strdup(name);
251 asset->hashed_name = stem_hash ? join3(stem_hash, name + ext, "") : NULL;
252 cwist_free(stem_hash);
253 asset->url = asset->hashed_name ? join3(reg->prefix, "/", asset->hashed_name) : NULL;
254 asset->content_type = cwist_strdup(content_type ? content_type : guess_content_type(name));
255 snprintf(asset->etag, sizeof(asset->etag), "\"%s\"", hash);
256
257 if (!asset->name || !asset->hashed_name || !asset->url || !asset->content_type) {
258 asset_free(asset);
259 return NULL;
260 }
261 return asset;
262}

273static cwist_error_t asset_publish(cwist_app *app, const char *name, unsigned char *data,
274 size_t len, const char *content_type) {
275 cwist_asset_registry *reg = registry_get(app, true);
276 if (!reg) {
277 cwist_free(data);
278 return asset_result(-1);
279 }
280
281 char hash[ASSET_HASH_HEX + 1];
282 content_hash(data, len, hash);
283
284 /* The newest asset under this name already has this content and type:
285 * nothing to publish. An older match does not count, since the name
286 * currently resolves to something else. */
287 const char *type = content_type ? content_type : guess_content_type(name);
288 for (struct cwist_asset *a = atomic_load_explicit(&reg->head, memory_order_acquire); a;
289 a = a->next) {
290 if (strcmp(a->name, name) != 0) continue;
291 if (memcmp(a->etag + 1, hash, ASSET_HASH_HEX) == 0 && a->len == len &&
292 (len == 0 || memcmp(a->data, data, len) == 0) && strcmp(a->content_type, type) == 0) {
293 cwist_free(data);
294 return asset_result(0);
295 }
296 break;
297 }
298
299 pthread_mutex_lock(&g_registry_lock);
300 struct cwist_asset *asset = asset_build(reg, name, data, len, content_type, hash);
301 pthread_mutex_unlock(&g_registry_lock);
302 if (!asset) return asset_result(-1);
303
304 struct cwist_asset *head = atomic_load_explicit(&reg->head, memory_order_relaxed);
305 do {
306 asset->next = head;
307 } while (!atomic_compare_exchange_weak_explicit(&reg->head, &head, asset, memory_order_release,
308 memory_order_relaxed));
309 return asset_result(0);
310}

353cwist_error_t cwist_app_asset_add_file(cwist_app *app, const char *name, const char *path,
354 const char *content_type) {
355 if (!app || !name_is_valid(name) || !path) return asset_result(-1);
356 if (content_type && (!*content_type || !header_value_is_valid(content_type))) {
357 return asset_result(-1);
358 }
359
360 FILE *f = fopen(path, "rb");
361 if (!f) return asset_result(-1);
362
363 size_t cap = 4096, len = 0;
364 unsigned char *buf = (unsigned char *)cwist_alloc(cap);
365 bool ok = buf != NULL;
366
367 while (ok) {
368 if (len == cap) {
369 unsigned char *grown = (unsigned char *)cwist_realloc(buf, cap * 2);
370 if (!grown) {
371 ok = false;
372 break;
373 }
374 buf = grown;
375 cap *= 2;
376 }
377 size_t n = fread(buf + len, 1, cap - len, f);
378 len += n;
379 if (n == 0) {
380 ok = !ferror(f);
381 break;
382 }
383 }
384 fclose(f);
385
386 if (!ok) {
387 cwist_free(buf);
388 return asset_result(-1);
389 }
390
391 return asset_publish(app, name, buf, len, content_type);
392}

427bool cwist_assets_match(cwist_app *app, const cwist_http_request *req, cwist_asset_match *out) {
428 if (!app || !req || !req->path || !req->path->data) return false;
429 if (req->method != CWIST_HTTP_GET && req->method != CWIST_HTTP_HEAD) return false;
430 cwist_asset_registry *reg = registry_get(app, false);
431 if (!reg) return false;
432
433 const char *path = req->path->data;
434 size_t plen = strlen(reg->prefix);
435 if (strncmp(path, reg->prefix, plen) != 0 || path[plen] != '/') return false;
436 const char *rest = path + plen + 1;
437
438 const struct cwist_asset *logical = NULL;
439 for (struct cwist_asset *a = atomic_load_explicit(&reg->head, memory_order_acquire); a;
440 a = a->next) {
441 if (strcmp(a->hashed_name, rest) == 0) {
442 out->asset = a;
443 out->immutable = true;
444 return true;
445 }
446 if (!logical && strcmp(a->name, rest) == 0) logical = a;
447 }
448
449 if (!logical) return false;
450 out->asset = logical;
451 out->immutable = false;
452
453 return true;
454}

457bool cwist_assets_etag_matches(const char *if_none_match, const char *etag) {
458 size_t elen = strlen(etag);
459 const char *p = if_none_match;
460 while (*p) {
461 while (*p == ' ' || *p == '\t' || *p == ',') p++;
462 if (!*p) break;
463 const char *start = p;
464 while (*p && *p != ',') p++;
465 const char *end = p;
466 while (end > start && (end[-1] == ' ' || end[-1] == '\t')) end--;
467 if (end - start == 1 && *start == '*') return true;
468 if (end - start > 2 && start[0] == 'W' && start[1] == '/') start += 2;
469 if ((size_t)(end - start) == elen && memcmp(start, etag, elen) == 0) return true;
470 }
471 return false;
472}

493void cwist_assets_respond(cwist_http_request *req, cwist_http_response *res,
494 const cwist_asset_match *match) {
495 if (!req || !res) return;
496 if (!match || !match->asset) {
497 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
498 return;
499 }
500 const struct cwist_asset *asset = match->asset;
501 const char *cache = match->immutable ? CWIST_ASSET_IMMUTABLE_CACHE : "no-cache";
502 const char *inm = cwist_http_header_get(req->headers, "If-None-Match");
503 bool not_modified = inm && cwist_assets_etag_matches(inm, asset->etag);
504
505 bool ok = assets_add_header(res, "ETag", asset->etag) &&
506 assets_add_header(res, "Cache-Control", cache);
507 if (ok && !not_modified) ok = assets_add_header(res, "Content-Type", asset->content_type);
508 if (!ok) {
509 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
510 return;
511 }
512
513 if (not_modified) {
514 res->status_code = CWIST_HTTP_NOT_MODIFIED;
515 cwist_error_t err = cwist_sstring_assign(res->body, "");
516 cwist_error_dispose(&err);
517 return;
518 }
519 res->status_code = CWIST_HTTP_OK;
520 /* Assets live until the app is destroyed, so no release hook is needed. */
521 cwist_http_response_set_body_ptr(res, asset->data, asset->len);
522}

533int cwist_assets_clone(void **dst, const void *src) {
534 *dst = NULL;
535 if (!src) return 0;
536
537 const cwist_asset_registry *from = (const cwist_asset_registry *)src;
538 pthread_mutex_lock(&g_registry_lock);
539 cwist_asset_registry *reg = (cwist_asset_registry *)cwist_alloc(sizeof(*reg));
540 if (!reg) {
541 pthread_mutex_unlock(&g_registry_lock);
542 return -1;
543 }
544
545 reg->prefix = cwist_strdup(from->prefix);
546 atomic_init(&reg->head, NULL);
547
548 if (!reg->prefix) {
549 cwist_free(reg);
550 pthread_mutex_unlock(&g_registry_lock);
551 return -1;
552 }
553
554 /* Keep the newest-first order: append each copy at the tail. */
555 struct cwist_asset *head = NULL, **tail = &head;
556 bool ok = true;
557 for (struct cwist_asset *a =
558 atomic_load_explicit(&((cwist_asset_registry *)from)->head, memory_order_acquire);
559 a && ok; a = a->next) {
560 unsigned char *data = (unsigned char *)cwist_alloc(a->len ? a->len : 1);
561 if (!data) {
562 ok = false;
563 break;
564 }
565
566 if (a->len) memcpy(data, a->data, a->len);
567
568 char hash[ASSET_HASH_HEX + 1];
569 memcpy(hash, a->etag + 1, ASSET_HASH_HEX);
570 hash[ASSET_HASH_HEX] = '\0';
571 struct cwist_asset *copy = asset_build(reg, a->name, data, a->len, a->content_type, hash);
572
573 if (!copy) {
574 ok = false;
575 break;
576 }
577
578 copy->next = NULL;
579 *tail = copy;
580 tail = &copy->next;
581 }
582
583 atomic_init(&reg->head, head);
584 if (!ok) {
585 cwist_assets_destroy(reg);
586 pthread_mutex_unlock(&g_registry_lock);
587 return -1;
588 }
589
590 *dst = reg;
591 pthread_mutex_unlock(&g_registry_lock);
592 return 0;
593}
