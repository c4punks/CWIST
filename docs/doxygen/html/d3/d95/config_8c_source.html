66void cwist_config_set(cwist_config *cfg, const char *key, const char *value) {
67 if (!cfg || !key) return;
68 size_t idx = cwist_config_hash(key) % cfg->bucket_count;
69 cwist_config_bucket *curr = cfg->buckets[idx];
70 while (curr) {
71 if (strcmp(curr->key, key) == 0) {
72 /* Keep the old value if the copy fails, so get() never sees NULL. */
73 char *copy = cwist_strdup(value ? value : "");
74 if (!copy) return;
75 cwist_free(curr->value);
76 curr->value = copy;
77 return;
78 }
79 curr = curr->next;
80 }
81 cwist_config_bucket *bucket = (cwist_config_bucket *)cwist_alloc(sizeof(cwist_config_bucket));
82 if (!bucket) return;
83 bucket->key = cwist_strdup(key);
84 bucket->value = cwist_strdup(value ? value : "");
85 if (!bucket->key || !bucket->value) {
86 cwist_free(bucket->key);

88 cwist_free(bucket);

184void cwist_config_load_file(cwist_config *cfg, const char *path) {
185 if (!cfg || !path) return;
186 FILE *f = fopen(path, "r");
187 if (!f) return;
188 char line[1024];
189 while (fgets(line, sizeof(line), f)) {
190 char *p = line;
191 while (isspace((unsigned char)*p)) p++;
192 if (*p == '#' || *p == '\0') continue;
193 char *eq = strchr(p, '=');
194 if (!eq) continue;
195 *eq = '\0';
196 char *key = p;
197 char *value = eq + 1;
198 // trim key
199 char *ke = key + strlen(key) - 1;
200 while (ke > key && isspace((unsigned char)*ke)) *ke-- = '\0';
201 // trim value
202 while (isspace((unsigned char)*value)) value++;
203 char *ve = value + strlen(value) - 1;
204 while (ve > value && isspace((unsigned char)*ve)) *ve-- = '\0';
205 // unquote value
206 if (value[0] == '"' && ve == value + strlen(value) - 1 && *ve == '"') {
207 value++;
208 *ve = '\0';
209 }
210 cwist_config_set(cfg, key, value);
211 }
212 fclose(f);
213}
