173void cwist_config_load_file(cwist_config *cfg, const char *path) {
174 if (!cfg || !path) return;
175 FILE *f = fopen(path, "r");
176 if (!f) return;
177 char line[1024];
178 while (fgets(line, sizeof(line), f)) {
179 char *p = line;
180 while (isspace((unsigned char)*p)) p++;
181 if (*p == '#' || *p == '\0') continue;
182 char *eq = strchr(p, '=');
183 if (!eq) continue;
184 *eq = '\0';
185 char *key = p;
186 char *value = eq + 1;
187 // trim key
188 char *ke = key + strlen(key) - 1;
189 while (ke > key && isspace((unsigned char)*ke)) *ke-- = '\0';
190 // trim value
191 while (isspace((unsigned char)*value)) value++;
192 char *ve = value + strlen(value) - 1;
193 while (ve > value && isspace((unsigned char)*ve)) *ve-- = '\0';
194 // unquote value
195 if (value[0] == '"' && ve == value + strlen(value) - 1 && *ve == '"') {
196 value++;
197 *ve = '\0';
198 }
199 cwist_config_set(cfg, key, value);
200 }
201 fclose(f);
