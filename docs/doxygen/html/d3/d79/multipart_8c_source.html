58static int mp_parse_headers(mp_parse_ctx *ctx) {
59 ctx->header_field[ctx->header_field_len] = '\0';
60 ctx->header_value[ctx->header_value_len] = '\0';
61
62 if (strcasecmp(ctx->header_field, "Content-Disposition") == 0) {
63 const char *name_tag = strstr(ctx->header_value, "name=\"");
64 if (name_tag) {
65 name_tag += 6;
66 const char *name_end = strchr(name_tag, '"');
67 if (name_end) {
68 size_t nlen = (size_t)(name_end - name_tag);
69 if (nlen >= sizeof(ctx->name)) return 1;
70 memcpy(ctx->name, name_tag, nlen);
71 ctx->name[nlen] = '\0';
72 }
73 }
74 const char *filename_tag = strstr(ctx->header_value, "filename=\"");
75 if (filename_tag) {
76 filename_tag += 10;
77 const char *filename_end = strchr(filename_tag, '"');
78 if (filename_end) {
79 size_t flen = (size_t)(filename_end - filename_tag);
80 if (flen >= sizeof(ctx->filename)) return 1;
81 memcpy(ctx->filename, filename_tag, flen);
82 ctx->filename[flen] = '\0';
83 }
84 }
85 } else if (strcasecmp(ctx->header_field, "Content-Type") == 0) {
86 size_t vlen = strlen(ctx->header_value);
87 if (vlen >= sizeof(ctx->content_type)) return 1;
88 strcpy(ctx->content_type, ctx->header_value);
89 }
90
91 ctx->header_field_len = 0;
92 ctx->header_value_len = 0;
93 ctx->have_value = false;
94 return 0;

142static int mp_on_part_data(multipart_parser *p, const char *at, size_t len) {
143 mp_parse_ctx *ctx = (mp_parse_ctx *)multipart_parser_get_data(p);
144 if (len == 0) return 0;
145 if (ctx->data_len + len > ctx->data_cap) {
146 size_t new_cap = (ctx->data_len + len) * 2;
147 if (new_cap < 256) new_cap = 256;
148 char *new_data = (char *)cwist_alloc(new_cap);
149 if (!new_data) return 1;
150 if (ctx->data) {
151 memcpy(new_data, ctx->data, ctx->data_len);
152 cwist_free(ctx->data);
153 }
154 ctx->data = new_data;
155 ctx->data_cap = new_cap;
156 }
157 memcpy(ctx->data + ctx->data_len, at, len);
158 ctx->data_len += len;
159 return 0;
160}

217cwist_multipart_result *cwist_multipart_parse(const char *body, size_t body_len,
218 const char *boundary) {
219 if (!body || body_len == 0 || !boundary) return NULL;
220
221 cwist_multipart_result *result =
222 (cwist_multipart_result *)cwist_alloc(sizeof(cwist_multipart_result));
223 if (!result) return NULL;
224 result->fields = NULL;
225
226 mp_parse_ctx ctx = {0};
227 ctx.result = result;
228
229 multipart_parser_settings settings = {
230 .on_header_field = mp_on_header_field,
231 .on_header_value = mp_on_header_value,
232 .on_headers_complete = mp_on_headers_complete,
233 .on_part_data_begin = mp_on_part_data_begin,
234 .on_part_data = mp_on_part_data,
235 .on_part_data_end = mp_on_part_data_end,
236 };
237
238 /* multipart-parser-c expects the leading dashes in the boundary. */
239 size_t blen = strlen(boundary);
240 cwist_scratch_t parser_boundary_s CWIST_SCRATCH_DEFER = {0};
241 char *parser_boundary = (char *)cwist_scratch_alloc(&parser_boundary_s, blen + 3);
242 if (!parser_boundary) {
243 cwist_free(result);
244 return NULL;
245 }
246 memcpy(parser_boundary, "--", 2);
247 memcpy(parser_boundary + 2, boundary, blen);
248 parser_boundary[blen + 2] = '\0';
249
250 multipart_parser *parser = multipart_parser_init(parser_boundary, &settings);
251 if (!parser) {
252 cwist_free(result);
253 return NULL;
254 }
255 multipart_parser_set_data(parser, &ctx);
256 size_t consumed = multipart_parser_execute(parser, body, body_len);
257 multipart_parser_free(parser);
258
259 /* Truncated body: on_part_data_end never fired, so release the in-flight
260 * part buffer here. */
261 if (ctx.data) {
262 cwist_free(ctx.data);
263 ctx.data = NULL;
264 }
265
266 /* The parser stops early on malformed input; report it as NULL per the
267 * documented contract. */
268 if (consumed != body_len) {
269 cwist_multipart_result_destroy(result);
270 return NULL;
271 }
272
273 return result;
274}

311char *cwist_multipart_extract_boundary(const char *content_type) {
312 if (!content_type) return NULL;
313 const char *p = content_type;
314 while (*p) {
315 while (*p && (isspace((unsigned char)*p) || *p == ';')) p++;
316 if (strncasecmp(p, "boundary", 8) == 0) {
317 p += 8;
318 while (*p && isspace((unsigned char)*p)) p++;
319 if (*p == '=') {
320 p++;
321 while (*p && isspace((unsigned char)*p)) p++;
322 const char *start = p;
323 if (*p == '"') {
324 start = ++p;
325 while (*p && *p != '"') p++;
326 } else {
327 while (*p && *p != ';' && !isspace((unsigned char)*p)) p++;
328 }
329 size_t len = (size_t)(p - start);
330 char *boundary = (char *)cwist_alloc(len + 1);
331 if (boundary) {
332 memcpy(boundary, start, len);
333 boundary[len] = '\0';
334 }
335 return boundary;
336 }
337 }
338 while (*p && *p != ';') p++;
339 if (*p == ';') p++;
340 }
341 return NULL;
