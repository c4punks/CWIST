61static int mp_disposition_param(const char *value, const char *key, char *out, size_t cap) {
62 size_t key_len = strlen(key);
63 const char *p = value;
64 while (*p) {
65 while (isspace((unsigned char)*p) || *p == ';') p++;
66 if (strncasecmp(p, key, key_len) != 0) {
67 while (*p && *p != ';') p++;
68 continue;
69 }
70 const char *name_end = p + key_len;
71 if (*name_end != '=' && !isspace((unsigned char)*name_end)) {
72 while (*p && *p != ';') p++;
73 continue;
74 }
75 p = name_end;
76 while (isspace((unsigned char)*p)) p++;
77 if (*p != '=') {
78 while (*p && *p != ';') p++;
79 continue;
80 }
81 p++;
82 while (isspace((unsigned char)*p)) p++;
83
84 size_t out_len = 0;
85 if (*p == '"') {
86 p++;
87 bool closed = false;
88 while (*p) {
89 if (*p == '"') {
90 closed = true;
91 break;
92 }
93 if (*p == '\\') {
94 p++;
95 if (!*p) return -1;
96 }
97 if (out_len + 1 >= cap) return -1;
98 out[out_len++] = *p++;
99 }
100 if (!closed) return -1;
101 } else {
102 while (*p && *p != ';' && !isspace((unsigned char)*p)) {
103 if (!mp_token_char((unsigned char)*p) || out_len + 1 >= cap) return -1;
104 out[out_len++] = *p++;
105 }
106 }
107 out[out_len] = '\0';
108 return 1;
109 }
110 return 0;
111}

114static int mp_parse_headers(mp_parse_ctx *ctx) {
115 ctx->header_field[ctx->header_field_len] = '\0';
116 ctx->header_value[ctx->header_value_len] = '\0';
117
118 if (strcasecmp(ctx->header_field, "Content-Disposition") == 0) {
119 if (mp_disposition_param(ctx->header_value, "name", ctx->name, sizeof(ctx->name)) < 0 ||
120 mp_disposition_param(ctx->header_value, "filename", ctx->filename,
121 sizeof(ctx->filename)) < 0)
122 return 1;
123 } else if (strcasecmp(ctx->header_field, "Content-Type") == 0) {
124 size_t vlen = strlen(ctx->header_value);
125 if (vlen >= sizeof(ctx->content_type)) return 1;
126 strcpy(ctx->content_type, ctx->header_value);
127 }
128
129 ctx->header_field_len = 0;
130 ctx->header_value_len = 0;
131 ctx->have_value = false;
132 return 0;

180static int mp_on_part_data(multipart_parser *p, const char *at, size_t len) {
181 mp_parse_ctx *ctx = (mp_parse_ctx *)multipart_parser_get_data(p);
182 if (len == 0) return 0;
183 if (ctx->data_len + len > ctx->data_cap) {
184 size_t new_cap = (ctx->data_len + len) * 2;
185 if (new_cap < 256) new_cap = 256;
186 char *new_data = (char *)cwist_alloc(new_cap);
187 if (!new_data) return 1;
188 if (ctx->data) {
189 memcpy(new_data, ctx->data, ctx->data_len);
190 cwist_free(ctx->data);
191 }
192 ctx->data = new_data;
193 ctx->data_cap = new_cap;
194 }
195 memcpy(ctx->data + ctx->data_len, at, len);
196 ctx->data_len += len;
197 return 0;
198}

256 const char *boundary) {

271 .on_part_data_begin = mp_on_part_data_begin,

277 size_t blen = strlen(boundary);

279 char *parser_boundary = (char *)cwist_scratch_alloc(&parser_boundary_s, blen + 3);
280 if (!parser_boundary) {

282 return NULL;
283 }
284 memcpy(parser_boundary, "--", 2);
285 memcpy(parser_boundary + 2, boundary, blen);
286 parser_boundary[blen + 2] = '\0';
287
288 multipart_parser *parser = multipart_parser_init(parser_boundary, &settings);
289 if (!parser) {
290 cwist_free(result);
291 return NULL;
292 }
293 multipart_parser_set_data(parser, &ctx);
294 size_t consumed = multipart_parser_execute(parser, body, body_len);
295 multipart_parser_free(parser);
296
297 /* Truncated body: on_part_data_end never fired, so release the in-flight
298 * part buffer here. */
299 if (ctx.data) {
300 cwist_free(ctx.data);
301 ctx.data = NULL;
302 }
303
304 /* The parser stops early on malformed input; report it as NULL per the
305 * documented contract. */
306 if (consumed != body_len) {
307 cwist_multipart_result_destroy(result);
308 return NULL;
309 }
310
311 return result;

349char *cwist_multipart_extract_boundary(const char *content_type) {
350 if (!content_type) return NULL;
351 const char *p = content_type;
352 while (*p) {
353 while (*p && (isspace((unsigned char)*p) || *p == ';')) p++;
354 if (strncasecmp(p, "boundary", 8) == 0) {
355 p += 8;
356 while (*p && isspace((unsigned char)*p)) p++;
357 if (*p == '=') {
358 p++;
359 while (*p && isspace((unsigned char)*p)) p++;
360 const char *start = p;
361 if (*p == '"') {
362 start = ++p;
363 while (*p && *p != '"') p++;
364 } else {
365 while (*p && *p != ';' && !isspace((unsigned char)*p)) p++;
366 }
367 size_t len = (size_t)(p - start);
368 char *boundary = (char *)cwist_alloc(len + 1);
369 if (boundary) {
370 memcpy(boundary, start, len);
371 boundary[len] = '\0';
372 }
373 return boundary;
374 }
375 }
376 while (*p && *p != ';') p++;
377 if (*p == ';') p++;
378 }
