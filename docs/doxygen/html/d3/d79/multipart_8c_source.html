49static int mp_on_header_field(multipart_parser *p, const char *at, size_t len) {

93static void mp_parse_headers(mp_parse_ctx *ctx) {
94 if (ctx->header_field_len >= sizeof(ctx->header_field))
95 ctx->header_field_len = sizeof(ctx->header_field) - 1;
96 ctx->header_field[ctx->header_field_len] = '\0';
97
98 if (ctx->header_value_len >= sizeof(ctx->header_value))
99 ctx->header_value_len = sizeof(ctx->header_value) - 1;
100 ctx->header_value[ctx->header_value_len] = '\0';
101
102 if (strcasecmp(ctx->header_field, "Content-Disposition") == 0) {
103 const char *name_tag = strstr(ctx->header_value, "name=\"");
104 if (name_tag) {
105 name_tag += 6;
106 const char *name_end = strchr(name_tag, '"');
107 if (name_end) {
108 size_t nlen = (size_t)(name_end - name_tag);
109 if (nlen < sizeof(ctx->name)) {
110 memcpy(ctx->name, name_tag, nlen);
111 ctx->name[nlen] = '\0';
112 }
113 }
114 }
115 const char *filename_tag = strstr(ctx->header_value, "filename=\"");
116 if (filename_tag) {
117 filename_tag += 10;
118 const char *filename_end = strchr(filename_tag, '"');
119 if (filename_end) {
120 size_t flen = (size_t)(filename_end - filename_tag);
121 if (flen < sizeof(ctx->filename)) {
122 memcpy(ctx->filename, filename_tag, flen);
123 ctx->filename[flen] = '\0';
124 }
125 }
126 }
127 } else if (strcasecmp(ctx->header_field, "Content-Type") == 0) {
128 size_t vlen = strlen(ctx->header_value);
129 if (vlen < sizeof(ctx->content_type)) strcpy(ctx->content_type, ctx->header_value);
130 }
131
132 ctx->header_field_len = 0;
133 ctx->header_value_len = 0;
134 ctx->have_value = false;
135}

194static int mp_on_part_data(multipart_parser *p, const char *at, size_t len) {
195 mp_parse_ctx *ctx = (mp_parse_ctx *)multipart_parser_get_data(p);
196 if (len == 0) return 0;
197 if (ctx->data_len + len > ctx->data_cap) {
198 size_t new_cap = (ctx->data_len + len) * 2;
199 if (new_cap < 256) new_cap = 256;
200 char *new_data = (char *)cwist_alloc(new_cap);
201 if (!new_data) return 1;
202 if (ctx->data) {
203 memcpy(new_data, ctx->data, ctx->data_len);
204 cwist_free(ctx->data);
205 }
206 ctx->data = new_data;
207 ctx->data_cap = new_cap;
208 }
209 memcpy(ctx->data + ctx->data_len, at, len);
210 ctx->data_len += len;
211 return 0;
212}

269cwist_multipart_result *cwist_multipart_parse(const char *body, size_t body_len,
270 const char *boundary) {
271 if (!body || body_len == 0 || !boundary) return NULL;
272
273 cwist_multipart_result *result =
274 (cwist_multipart_result *)cwist_alloc(sizeof(cwist_multipart_result));
275 if (!result) return NULL;
276 result->fields = NULL;
277
278 mp_parse_ctx ctx = {0};
279 ctx.result = result;
280
281 multipart_parser_settings settings = {
282 .on_header_field = mp_on_header_field,
283 .on_header_value = mp_on_header_value,
284 .on_headers_complete = mp_on_headers_complete,
285 .on_part_data_begin = mp_on_part_data_begin,
286 .on_part_data = mp_on_part_data,
287 .on_part_data_end = mp_on_part_data_end,
288 };
289
290 /* multipart-parser-c expects the leading dashes in the boundary. */
291 size_t blen = strlen(boundary);
292 cwist_scratch_t parser_boundary_s CWIST_SCRATCH_DEFER = {0};
293 char *parser_boundary = (char *)cwist_scratch_alloc(&parser_boundary_s, blen + 3);
294 if (!parser_boundary) {
295 cwist_free(result);
296 return NULL;
297 }
298 memcpy(parser_boundary, "--", 2);
299 memcpy(parser_boundary + 2, boundary, blen);
300 parser_boundary[blen + 2] = '\0';
301
302 multipart_parser *parser = multipart_parser_init(parser_boundary, &settings);
303 if (!parser) {
304 cwist_free(result);
305 return NULL;
306 }
307 multipart_parser_set_data(parser, &ctx);
308 size_t consumed = multipart_parser_execute(parser, body, body_len);
309 multipart_parser_free(parser);
310
311 /* Truncated body: on_part_data_end never fired, so release the in-flight
312 * part buffer here. */
313 if (ctx.data) {
314 cwist_free(ctx.data);
315 ctx.data = NULL;
316 }
317
318 /* The parser stops early on malformed input; report it as NULL per the
319 * documented contract. */
320 if (consumed != body_len) {
321 cwist_multipart_result_destroy(result);
322 return NULL;
323 }
324
325 return result;
326}

363char *cwist_multipart_extract_boundary(const char *content_type) {
364 if (!content_type) return NULL;
365 const char *p = content_type;
366 while (*p) {
367 while (*p && (isspace((unsigned char)*p) || *p == ';')) p++;
368 if (strncasecmp(p, "boundary", 8) == 0) {
369 p += 8;
370 while (*p && isspace((unsigned char)*p)) p++;
371 if (*p == '=') {
372 p++;
373 while (*p && isspace((unsigned char)*p)) p++;
374 const char *start = p;
375 if (*p == '"') {
376 start = ++p;
377 while (*p && *p != '"') p++;
378 } else {
379 while (*p && *p != ';' && !isspace((unsigned char)*p)) p++;
380 }
381 size_t len = (size_t)(p - start);
382 char *boundary = (char *)cwist_alloc(len + 1);
383 if (boundary) {
384 memcpy(boundary, start, len);
385 boundary[len] = '\0';
386 }
387 return boundary;
388 }
389 }
390 while (*p && *p != ';') p++;
391 if (*p == ';') p++;
392 }
393 return NULL;
