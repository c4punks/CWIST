20typedef enum {
21 LATIN_GF2, /* q = 2^e: L_k(r,c) = k*r + c in GF(2^e) */
22 LATIN_PRIME, /* q = p: L_k(r,c) = (k*r + c) mod p */
23 LATIN_PRIME_DIGITS, /* q = p^e, p odd, e >= 2: the prime rule per base-p digit */

93static uint32_t gf2_mul(uint32_t a, uint32_t b, uint32_t e, uint32_t poly) {
94 uint64_t r = 0;
95 uint64_t x = a;
96 const uint64_t top = UINT64_C(1) << e;
97 while (b) {
98 if (b & 1u) r ^= x;
99 b >>= 1;
100 x <<= 1;
101 if (x & top) x ^= poly;
102 }
103 return (uint32_t)r;
104}

107static uint32_t component_symbol(const latin_component_t *cp, uint32_t k, uint32_t r, uint32_t c) {
108 switch (cp->kind) {
109 case LATIN_GF2:
110 return gf2_mul(k + 1, r, cp->e, cp->poly) ^ c;
111 case LATIN_PRIME:
112 return (uint32_t)(((uint64_t)(k + 1) * r + c) % cp->p);
113 case LATIN_PRIME_DIGITS: {
114 uint32_t out = 0, stride = 1;
115 for (uint32_t i = 0; i < cp->e; i++) {
116 uint32_t rd = r % cp->p, cd = c % cp->p;
117 out += (uint32_t)(((uint64_t)(k + 1) * rd + cd) % cp->p) * stride;
118 r /= cp->p;
119 c /= cp->p;
120 stride *= cp->p;
121 }
122 return out;
123 }
124 case LATIN_CHOI9: {
125 const uint8_t *t = (k == 0) ? k_choi9_a : k_choi9_b;
126 return (uint32_t)t[r * 9 + c] - 1u;
127 }
128 }
129 return 0;
130}

133 switch (cp->kind) {

149cwist_latin_family_t *cwist_latin_family_create(uint32_t order, uint32_t min_squares) {
150 if (order < 2 || order > CWIST_LATIN_MAX_ORDER) return NULL;
151 if (min_squares == 0) min_squares = 1;
152
153 latin_component_t comp[LATIN_MAX_COMPONENTS];
154 uint32_t ncomp = 0;
155 uint32_t count = UINT32_MAX;
156 uint32_t n = order;
157
158 /* Trial division: at most sqrt(2^31) ~ 46341 steps, once per create. */
159 for (uint32_t p = 2; n > 1; p = (p == 2) ? 3 : p + 2) {
160 if ((uint64_t)p * p > n) p = n; /* remaining n is prime */
161 if (n % p != 0) continue;
162 uint32_t e = 0, q = 1;
163 while (n % p == 0) {
164 n /= p;
165 q *= p;
166 e++;
167 }
168 latin_component_t *cp = &comp[ncomp++];
169 memset(cp, 0, sizeof(*cp));
170 cp->q = q;
171 cp->p = p;
172 cp->e = e;
173 if (p == 2) {
174 cp->kind = LATIN_GF2;
175 cp->poly = k_gf2_poly[e];
176 cp->count = q - 1;
177 } else if (e == 1) {
178 cp->kind = LATIN_PRIME;
179 cp->count = p - 1;
180 } else if (q == 9) {
181 cp->kind = LATIN_CHOI9;
182 cp->count = 2;
183 } else {
184 cp->kind = LATIN_PRIME_DIGITS;
185 cp->count = p - 1;
186 }
187 if (cp->count < count) count = cp->count;
188 }
189
190 if (count < min_squares) return NULL;
191
192 cwist_latin_family_t *f = cwist_alloc(sizeof(*f));
193 if (!f) return NULL;
194 memset(f, 0, sizeof(*f));
195 f->order = order;
196 f->count = count;
197 f->ncomp = ncomp;
198 memcpy(f->comp, comp, ncomp * sizeof(comp[0]));
199
200 size_t used = 0;
201 for (uint32_t i = 0; i < ncomp && used < sizeof(f->construction); i++) {
202 char part[32];
203 component_describe(&comp[i], part, sizeof(part));
204 int w = snprintf(f->construction + used, sizeof(f->construction) - used, "%s%s",
205 i ? " x " : "", part);
206 if (w < 0) break;
207 used += (size_t)w;
208 }
209 return f;
210}

228uint32_t cwist_latin_family_symbol(const cwist_latin_family_t *family, uint32_t square,
229 uint32_t row, uint32_t col) {
230 if (!family || square >= family->count || row >= family->order || col >= family->order) {
231 return UINT32_MAX;
232 }
233 /* MacNeish product: split row and column into mixed-radix digits, one
234 * per component, and recombine the per-component symbols. */
235 uint32_t out = 0, stride = 1;
236 for (uint32_t i = 0; i < family->ncomp; i++) {
237 const latin_component_t *cp = &family->comp[i];
238 uint32_t s = component_symbol(cp, square, row % cp->q, col % cp->q);
239 out += s * stride;
240 row /= cp->q;
241 col /= cp->q;
242 stride *= cp->q;
243 }
244 return out;
245}

255uint32_t cwist_latin_oa_level(const cwist_latin_family_t *family, uint64_t run, uint32_t factor) {

257 return UINT32_MAX;
258 }

260 uint32_t c = (uint32_t)(run % family->order);
