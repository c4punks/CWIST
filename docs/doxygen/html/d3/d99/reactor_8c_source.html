291static void latency_probe_record(uint64_t *buckets, uint64_t *count, uint64_t *sum_us,
292 uint64_t *max_us, uint64_t *over_5ms, uint64_t sample_us) {
293 int lo = 0, hi = LATENCY_PROBE_BUCKETS - 1;
294 while (lo < hi) { /* first bucket whose upper bound exceeds the sample */
295 int mid = (lo + hi) / 2;
296 if (sample_us < latency_probe_bounds_us[mid])
297 hi = mid;
298 else
299 lo = mid + 1;
300 }
301 buckets[lo]++;
302 (*count)++;
303 *sum_us += sample_us;
304 if (sample_us > *max_us) *max_us = sample_us;
305 if (sample_us > 5000) (*over_5ms)++;
306}

317static uint64_t latency_probe_percentile(const uint64_t *buckets, uint64_t count,
318 uint64_t pct_mille) {
319 uint64_t target = (count * pct_mille + 999) / 1000;

323 if (cumulative >= target)
324 return i + 1 < LATENCY_PROBE_BUCKETS ? latency_probe_bounds_us[i] : UINT64_MAX;

326 return UINT64_MAX;

386bool cwist_reactor_recv_arm(cwist_reactor_t *reactor, int fd, void *buf, unsigned len, void *conn,
387 uint64_t *armed_ns) {
388 if (!reactor || fd < 0 || !buf || len == 0 || !conn) return false;
389 if (!cwist_reactor_rx_supported(reactor)) return false;
390 /* Tag-bit discipline: ev_ctx pointers come from cwist_alloc (malloc
391 * backed, at least max_align_t aligned), so the low bit is always zero
392 * there and one here. Refuse a misaligned connection pointer rather
393 * than aliasing into the ev_ctx namespace. */
394 if (((uintptr_t)conn & 1u) != 0) return false;
395
396 struct io_uring_sqe sqe;
397 memset(&sqe, 0, sizeof(sqe));
398 sqe.opcode = IORING_OP_RECV;
399 sqe.fd = fd;
400 sqe.addr = (uint64_t)(uintptr_t)buf;
401 sqe.len = len;
402 sqe.user_data = (uint64_t)(uintptr_t)conn | 1u;
403 if (armed_ns && latency_probe_enabled()) {
404 struct timespec ts;
405 clock_gettime(CLOCK_MONOTONIC, &ts);
406 *armed_ns = (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
407 }

410 * submits immediately so remote workers wake. Data SQEs carry no ev_ctx
411 * slot, so deferred_ctxs[] holds NULL for them and flush_deferred knows
412 * the connection pointer rides in the tagged user_data. */
413 if (reactor->dispatching && pthread_equal(pthread_self(), reactor->owner) &&

418 reactor->deferred_ctxs[slot] = NULL;

422}

433static reactor_event_ctx_t *alloc_reactor_ctx(cwist_reactor_t *r, int fd, cwist_reactor_cb_t cb,
434 const void *payload, size_t payload_size) {
435 if (!r || payload_size > CWIST_REACTOR_PAYLOAD_SIZE) return NULL;
436 pthread_mutex_lock(&r->pool_lock);
437 reactor_event_ctx_t *ev_ctx = r->free_head;
438 if (ev_ctx) r->free_head = (reactor_event_ctx_t *)ev_ctx->ctx;
439 pthread_mutex_unlock(&r->pool_lock);
440
441 if (!ev_ctx) {
442 /* Free list empty: grow a chunk. Allocate and link it outside the
443 * lock so the ~40 KiB setup does not serialize every other submitter
444 * on this reactor; under the lock the chunk is pushed onto the free
445 * list wholesale and one slot popped. Chunks are never freed until
446 * destroy, so a race with another grower only means two chunks. */
447 reactor_slot_chunk_t *chunk =
448 cwist_alloc(sizeof(*chunk) + REACTOR_CHUNK_EVENTS * sizeof(reactor_event_ctx_t));
449 if (!chunk) return NULL;
450 for (uint32_t i = 0; i + 1 < REACTOR_CHUNK_EVENTS; i++) {
451 chunk->slots[i].ctx = &chunk->slots[i + 1];
452 }
453 chunk->slots[REACTOR_CHUNK_EVENTS - 1].ctx = NULL;
454
455 pthread_mutex_lock(&r->pool_lock);
456 chunk->next = r->chunks;
457 r->chunks = chunk;
458 chunk->slots[REACTOR_CHUNK_EVENTS - 1].ctx = r->free_head;
459 r->free_head = &chunk->slots[0];
460 ev_ctx = r->free_head;
461 r->free_head = (reactor_event_ctx_t *)ev_ctx->ctx;
462 pthread_mutex_unlock(&r->pool_lock);
463 }
464 if (!ev_ctx) return NULL;
465
466 memset(ev_ctx, 0, sizeof(*ev_ctx));
467 ev_ctx->fd = fd;
468 ev_ctx->cb = cb;
469 ev_ctx->ctx = ev_ctx->payload;
470 if (payload && payload_size > 0) {
471 memcpy(ev_ctx->payload, payload, payload_size);
472 }
473 return ev_ctx;
474}

484bool cwist_reactor_post(cwist_reactor_t *r, cwist_reactor_post_t *node) {
485 if (!r || !node || !node->cb) return false;
486 cwist_reactor_post_t *head = atomic_load_explicit(&r->post_head, memory_order_relaxed);
487 do {
488 node->next = head;
489 } while (!atomic_compare_exchange_weak_explicit(&r->post_head, &head, node,
490 memory_order_release, memory_order_relaxed));
491 /* Wake only on the first post of a pending batch. A non-empty stack
492 * already has a wake pending; another write would be redundant. */
493 if (head == NULL && r->wake_wr >= 0) {
494 uint64_t one = 1;
495 ssize_t ign = write(r->wake_wr, &one, sizeof(one));
496 (void)ign; /* EAGAIN means the run thread is already awake. */
497 }
498 return true;
499}

501/* Pop the whole MPSC stack and run the callbacks oldest-first. Only ever
502 * called by the reactor's run thread (or destroy, after it has stopped). */
503static void reactor_drain_posts(cwist_reactor_t *r) {
504 cwist_reactor_post_t *list =
505 atomic_exchange_explicit(&r->post_head, NULL, memory_order_acquire);
506 cwist_reactor_post_t *rev = NULL;
507 while (list) {
508 cwist_reactor_post_t *next = list->next;
509 list->next = rev;
510 rev = list;
511 list = next;
512 }
513 while (rev) {
514 cwist_reactor_post_t *next = rev->next;
515 rev->cb(rev->ctx);
516 rev = next;
517 }
518}

542static void timer_heap_up(cwist_reactor_t *r, uint32_t i) {
543 cwist_reactor_timer_t *t = r->timer_heap[i];
544 while (i > 0) {
545 uint32_t parent = (i - 1) / 2;
546 if (r->timer_heap[parent]->deadline_ns <= t->deadline_ns) break;
547 timer_heap_place(r, i, r->timer_heap[parent]);
548 i = parent;
549 }

553static void timer_heap_down(cwist_reactor_t *r, uint32_t i) {
554 cwist_reactor_timer_t *t = r->timer_heap[i];
555 for (;;) {
556 uint32_t child = 2 * i + 1;
557 if (child >= r->timer_n) break;
558 if (child + 1 < r->timer_n &&
559 r->timer_heap[child + 1]->deadline_ns < r->timer_heap[child]->deadline_ns)
560 child++;
561 if (t->deadline_ns <= r->timer_heap[child]->deadline_ns) break;
562 timer_heap_place(r, i, r->timer_heap[child]);
563 i = child;
564 }
565 timer_heap_place(r, i, t);
566}

568static void timer_heap_remove(cwist_reactor_t *r, cwist_reactor_timer_t *t) {
569 uint32_t i = t->heap_slot - 1;
570 t->heap_slot = 0;

572 if (i == r->timer_n) return;
573 timer_heap_place(r, i, last);

575 timer_heap_up(r, i);

617 * entry run, so a callback that re-arms itself with a zero delay waits for
618 * the next round instead of spinning here. */
619static void reactor_run_timers(cwist_reactor_t *r) {
620 if (r->timer_n == 0) return;
621 uint64_t now = reactor_now_ns();
622 uint32_t budget = r->timer_n;
623 while (budget-- > 0 && r->timer_n > 0 && r->timer_heap[0]->deadline_ns <= now) {
624 cwist_reactor_timer_t *t = r->timer_heap[0];
625 timer_heap_remove(r, t);
626 t->cb(t->ctx);
627 }
628}

630/* Poll wait for this round: the time to the earliest timer, capped at the
631 * idle wait. */
632static uint64_t reactor_wait_ns(const cwist_reactor_t *r) {
633 if (r->timer_n == 0) return REACTOR_IDLE_WAIT_NS;
634 uint64_t now = reactor_now_ns();
635 uint64_t deadline = r->timer_heap[0]->deadline_ns;
636 if (deadline <= now) return 0;
637 uint64_t wait = deadline - now;
638 return wait < REACTOR_IDLE_WAIT_NS ? wait : REACTOR_IDLE_WAIT_NS;
639}

641static void reactor_wake_cb(int fd, void *ctx) {
642 cwist_reactor_t *r = *(cwist_reactor_t *const *)ctx;
643 uint64_t buf[8];
644 while (read(fd, buf, sizeof(buf)) > 0) {
645 }
646 reactor_drain_posts(r);
647 /* One-shot slots are recycled after firing: re-arm for the next post. */
648 if (!cwist_reactor_add(r, fd, reactor_wake_cb, &r, sizeof(r))) {
649 int wr = r->wake_wr;
650 r->wake_fd = -1;
651 r->wake_wr = -1;
652 close(fd);
653 if (wr >= 0 && wr != fd) close(wr);
654 }
655}

657cwist_reactor_t *cwist_reactor_create(void) {
658 cwist_reactor_t *r = cwist_alloc(sizeof(cwist_reactor_t));
659 if (!r) return NULL;
660 memset(r, 0, sizeof(cwist_reactor_t));
661 __atomic_store_n(&r->running, false, __ATOMIC_RELEASE);
662 pthread_mutex_init(&r->pool_lock, NULL);
663#ifdef __linux__
664 pthread_mutex_init(&r->impl.sq_lock, NULL);
665#endif
666
667#ifdef __linux__
668 r->impl.use_epoll = false;
669 /* CWIST_REACTOR_BACKEND=epoll forces the epoll path for A/B measurement;
670 * any other value (or unset) keeps io_uring as the default. */
671 {
672 const char *backend = getenv("CWIST_REACTOR_BACKEND");
673 if (backend && strcmp(backend, "epoll") == 0) r->impl.use_epoll = true;
674 }
675 struct io_uring_params p;
676 memset(&p, 0, sizeof(p));
677 int fd = r->impl.use_epoll ? -1 : sys_io_uring_setup(4096, &p);
678 if (fd >= 0) {
679 r->impl.ring_fd = fd;
680 r->impl.sq_ring_sz = sq_ring_size(&p);
681 r->impl.cq_ring_sz = cq_ring_size(&p);
682 r->impl.sqes_sz = p.sq_entries * sizeof(struct io_uring_sqe);
683
684 void *sq_ptr = mmap_ring(fd, r->impl.sq_ring_sz, IORING_OFF_SQ_RING);
685 void *cq_ptr = mmap_ring(fd, r->impl.cq_ring_sz, IORING_OFF_CQ_RING);
686 void *sqes_ptr = mmap_ring(fd, r->impl.sqes_sz, IORING_OFF_SQES);
687
688 if (sq_ptr && cq_ptr && sqes_ptr) {
689 r->impl.sq_head = (uint32_t *)((char *)sq_ptr + p.sq_off.head);
690 r->impl.sq_tail = (uint32_t *)((char *)sq_ptr + p.sq_off.tail);
691 r->impl.sq_ring_mask = (uint32_t *)((char *)sq_ptr + p.sq_off.ring_mask);
692 r->impl.sq_array = (uint32_t *)((char *)sq_ptr + p.sq_off.array);
693 r->impl.sqes = sqes_ptr;
694 r->impl.sq_entries = p.sq_entries;
695
696 r->impl.cq_head = (uint32_t *)((char *)cq_ptr + p.cq_off.head);
697 r->impl.cq_tail = (uint32_t *)((char *)cq_ptr + p.cq_off.tail);
698 r->impl.cq_ring_mask = (uint32_t *)((char *)cq_ptr + p.cq_off.ring_mask);
699 r->impl.cqes = (struct io_uring_cqe *)((char *)cq_ptr + p.cq_off.cqes);
700 r->impl.cq_entries = p.cq_entries;
701 r->impl.active = true;
702
703 /* Identity SQE mapping is fixed for the ring's lifetime; fill it
704 * once here instead of rewriting sq_array on every submission. */
705 for (uint32_t i = 0; i < p.sq_entries; i++) r->impl.sq_array[i] = i;
706 } else {
707 if (sq_ptr) munmap(sq_ptr, r->impl.sq_ring_sz);
708 if (cq_ptr) munmap(cq_ptr, r->impl.cq_ring_sz);
709 if (sqes_ptr) munmap(sqes_ptr, r->impl.sqes_sz);
710 close(fd);
711 r->impl.use_epoll = true;
712 }
713 } else {
714 r->impl.use_epoll = true;
715 }
716
717 if (r->impl.use_epoll) {
718 r->impl.epoll_fd = epoll_create1(0);
719 if (r->impl.epoll_fd < 0) {
720 pthread_mutex_destroy(&r->impl.sq_lock);
721 pthread_mutex_destroy(&r->pool_lock);
722 cwist_free(r);
723 return NULL;
724 }
725 }
726#elif defined(__APPLE__) || defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__)
727 r->impl.kq_fd = kqueue();
728 if (r->impl.kq_fd < 0) {
729 pthread_mutex_destroy(&r->pool_lock);
730 cwist_free(r);
731 return NULL;
732 }
733#endif
734 atomic_init(&r->post_head, NULL);
735#ifdef __linux__
736 r->wake_fd = eventfd(0, EFD_NONBLOCK | EFD_CLOEXEC);
737 r->wake_wr = r->wake_fd;
738#else
739 r->wake_fd = -1;
740 r->wake_wr = -1;
741 {
742 int pfds[2];
743 if (pipe(pfds) == 0) {
744 for (int i = 0; i < 2; i++) {
745 int fl = fcntl(pfds[i], F_GETFL, 0);
746 if (fl >= 0) fcntl(pfds[i], F_SETFL, fl | O_NONBLOCK);
747 }
748 r->wake_fd = pfds[0];
749 r->wake_wr = pfds[1];
750 }
751 }
752#endif
753 /* Poll the wake fd on every backend. IORING_REGISTER_EVENTFD notifies
754 * an external eventfd when CQEs arrive; writes to it do not create CQEs
755 * or interrupt io_uring_enter. The one-shot poll is re-armed by
756 * reactor_wake_cb, including when posts arrive during a dispatch. */
757 if (r->wake_fd >= 0 && !cwist_reactor_add(r, r->wake_fd, reactor_wake_cb, &r, sizeof(r))) {
758 /* Wake best-effort: the run loop still drains the stack each round. */
759 close(r->wake_fd);
760 if (r->wake_wr != r->wake_fd) close(r->wake_wr);
761 r->wake_fd = -1;
762 r->wake_wr = -1;
763 }
764 return r;
765}

767void cwist_reactor_destroy(cwist_reactor_t *reactor) {
768 if (!reactor) return;
769#ifdef __linux__
770 latency_probe_dump(reactor);
771#endif
772 /* Run any completions posted after the run thread parked for good. */
773 reactor_drain_posts(reactor);
774 if (reactor->wake_fd >= 0) close(reactor->wake_fd);
775 if (reactor->wake_wr >= 0 && reactor->wake_wr != reactor->wake_fd) close(reactor->wake_wr);
776#ifdef __linux__
777 if (!reactor->impl.use_epoll) {
778 /* Teardown absorbed from io_uring_backend.c: unmap all three rings. */
779 if (reactor->impl.sqes) munmap(reactor->impl.sqes, reactor->impl.sqes_sz);
780 if (reactor->impl.cqes) {
781 munmap((char *)reactor->impl.cqes -
782 (reactor->impl.cq_ring_sz -
783 reactor->impl.cq_entries * sizeof(struct io_uring_cqe)),
784 reactor->impl.cq_ring_sz);
785 }
786 if (reactor->impl.sq_array) {
787 munmap((char *)reactor->impl.sq_array -
788 (reactor->impl.sq_ring_sz - reactor->impl.sq_entries * sizeof(uint32_t)),
789 reactor->impl.sq_ring_sz);
790 }
791 close(reactor->impl.ring_fd);
792 } else {
793 close(reactor->impl.epoll_fd);
794 }
795#elif defined(__APPLE__) || defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__)
796 close(reactor->impl.kq_fd);
797#endif
798 pthread_mutex_destroy(&reactor->pool_lock);
799#ifdef __linux__
800 pthread_mutex_destroy(&reactor->impl.sq_lock);
801#endif
802 cwist_free(reactor->timer_heap);
803 reactor_slot_chunk_t *chunk = reactor->chunks;
804 while (chunk) {
805 reactor_slot_chunk_t *next = chunk->next;
806 cwist_free(chunk);
807 chunk = next;
808 }
809 cwist_free(reactor);
810}

813/* Submit one SQE immediately: the reactor never batches or defers submission,
814 * so a woken worker always sees the event on its next wait. */
815static bool uring_submit(cwist_reactor_t *reactor, struct io_uring_sqe *out_sqe) {
816 bool ok = false;
817 pthread_mutex_lock(&reactor->impl.sq_lock);
818 uint32_t tail = *reactor->impl.sq_tail;
819 uint32_t head = __atomic_load_n(reactor->impl.sq_head, __ATOMIC_ACQUIRE);
820 if (tail - head < reactor->impl.sq_entries) {
821 uint32_t index = tail & *reactor->impl.sq_ring_mask;
822 struct io_uring_sqe *sqe = &reactor->impl.sqes[index];
823 memcpy(sqe, out_sqe, sizeof(*sqe));
824 __atomic_store_n(reactor->impl.sq_tail, tail + 1, __ATOMIC_RELEASE);
825 if (sys_io_uring_enter(reactor->impl.ring_fd, 1, 0, 0, NULL) < 0) {
826 if (getenv("CWIST_ASYNC_DEBUG")) {
827 static _Atomic long dbg_enter_fail;
828 long n = atomic_fetch_add(&dbg_enter_fail, 1) + 1;
829 if (n <= 5 || n % 10000 == 0)
830 fprintf(stderr,
831 "[reactor] uring enter failed fd=%d errno=%d(%s) sq=%u/%u total=%ld\n",
832 out_sqe->fd, errno, strerror(errno), tail - head,
833 reactor->impl.sq_entries, n);
834 }
835 __atomic_store_n(reactor->impl.sq_tail, tail, __ATOMIC_RELEASE);
836 } else {
837 ok = true;
838 }
839 } else if (getenv("CWIST_ASYNC_DEBUG")) {
840 static _Atomic long dbg_sq_full;
841 long n = atomic_fetch_add(&dbg_sq_full, 1) + 1;
842 if (n <= 5 || n % 10000 == 0)
843 fprintf(stderr, "[reactor] SQ full fd=%d depth=%u/%u total=%ld\n", out_sqe->fd,
844 tail - head, reactor->impl.sq_entries, n);
845 }
846 pthread_mutex_unlock(&reactor->impl.sq_lock);
847 return ok;
848}

851#ifdef __linux__
852/* Flush the deferred SQE queue with a single io_uring_enter for the whole
853 * batch. On batch failure (SQ momentarily full from concurrent cross-thread
854 * submissions) fall back to per-SQE submits with a short retry; a final
855 * failure closes the connection and recycles its slot, matching every
856 * caller's add-failure path. */
857static bool uring_submit_batch(cwist_reactor_t *reactor, struct io_uring_sqe *batch, uint32_t n) {
858 bool ok = false;
859 pthread_mutex_lock(&reactor->impl.sq_lock);
860 uint32_t tail = *reactor->impl.sq_tail;
861 uint32_t head = __atomic_load_n(reactor->impl.sq_head, __ATOMIC_ACQUIRE);
862 if (tail - head + n <= reactor->impl.sq_entries) {
863 for (uint32_t i = 0; i < n; i++) {
864 uint32_t index = (tail + i) & *reactor->impl.sq_ring_mask;
865 memcpy(&reactor->impl.sqes[index], &batch[i], sizeof(batch[i]));
866 }
867 __atomic_store_n(reactor->impl.sq_tail, tail + n, __ATOMIC_RELEASE);
868 if (sys_io_uring_enter(reactor->impl.ring_fd, n, 0, 0, NULL) >= 0) {
869 ok = true;
870 } else {
871 __atomic_store_n(reactor->impl.sq_tail, tail, __ATOMIC_RELEASE);
872 }
873 }
874 pthread_mutex_unlock(&reactor->impl.sq_lock);
875 return ok;
876}

878static void flush_deferred(cwist_reactor_t *reactor) {
879 uint32_t n = reactor->deferred_n;
880 reactor->deferred_n = 0;
881 if (n == 0) return;
882 if (uring_submit_batch(reactor, reactor->deferred_sqes, n)) return;
883 for (uint32_t i = 0; i < n; i++) {
884 struct io_uring_sqe *sqe = &reactor->deferred_sqes[i];
885 reactor_event_ctx_t *ev_ctx = reactor->deferred_ctxs[i];
886 int attempt;
887 for (attempt = 0; attempt < 100; attempt++) {
888 if (uring_submit(reactor, sqe)) break;
889 struct timespec ts = {.tv_sec = 0, .tv_nsec = 1000 * 1000};
890 nanosleep(&ts, NULL); /* SQ drains without our help; wait it out */
891 }
892 if (attempt == 100) {
893 if (getenv("CWIST_ASYNC_DEBUG")) {
894 fprintf(stderr, "[reactor] deferred submit failed fd=%d; closing\n", (int)sqe->fd);
895 }
896 if (ev_ctx) {
897 close((int)sqe->fd);
898 free_reactor_ctx(reactor, ev_ctx);
899 } else if (reactor->rx_cb) {
900 /* Tagged RECV SQE: the connection owns the fd and shell, so
901 * report the cancellation and let its handler clean up. */
902 reactor->rx_cb((void *)(sqe->user_data & ~(uint64_t)1u), -ECANCELED);
903 }
904 }
905 }
906}

908/* Copy the deferred SQEs into the SQ without an io_uring_enter: the run
909 * loop's next wait enter carries to_submit, so a dispatch round costs one
910 * enter total instead of wait-enter + flush-enter. On SQ contention fall
911 * back to the immediate flush so re-arms never stall. */
912static void queue_deferred(cwist_reactor_t *reactor) {
913 uint32_t n = reactor->deferred_n;
914 if (n == 0) return;
915 pthread_mutex_lock(&reactor->impl.sq_lock);
916 uint32_t tail = *reactor->impl.sq_tail;
917 uint32_t head = __atomic_load_n(reactor->impl.sq_head, __ATOMIC_ACQUIRE);
918 if (tail - head + n <= reactor->impl.sq_entries) {
919 for (uint32_t i = 0; i < n; i++) {
920 uint32_t index = (tail + i) & *reactor->impl.sq_ring_mask;
921 memcpy(&reactor->impl.sqes[index], &reactor->deferred_sqes[i],
922 sizeof(reactor->deferred_sqes[i]));
923 }
924 __atomic_store_n(reactor->impl.sq_tail, tail + n, __ATOMIC_RELEASE);
925 reactor->deferred_n = 0;
926 reactor->sq_unsubmitted += n;
927 }
928 pthread_mutex_unlock(&reactor->impl.sq_lock);
929 if (reactor->deferred_n) flush_deferred(reactor);
930}

933static bool reactor_add_common(cwist_reactor_t *reactor, int fd, cwist_reactor_cb_t cb,
934 const void *payload, size_t payload_size, bool for_write) {
935 if (!reactor || fd < 0) return false;
936 reactor_event_ctx_t *ev_ctx = alloc_reactor_ctx(reactor, fd, cb, payload, payload_size);
937 if (!ev_ctx) return false;
938
939#ifdef __linux__
940 if (latency_probe_enabled()) {
941 struct timespec ts;
942 clock_gettime(CLOCK_MONOTONIC, &ts);
943 ev_ctx->armed_ns = (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
944 }
945 if (!reactor->impl.use_epoll) {
946 /* One-shot POLL_ADD: multishot (IORING_POLL_ADD_MULTI) was rejected
947 * because the slot is recycled after firing, so a persistent poll
948 * would re-dispatch into a recycled slot. */
949 struct io_uring_sqe sqe;
950 memset(&sqe, 0, sizeof(sqe));
951 sqe.opcode = IORING_OP_POLL_ADD;
952 sqe.fd = fd;
953 sqe.poll_events = for_write ? POLLOUT : POLLIN;
954 sqe.user_data = (uint64_t)ev_ctx;
955 /* Run-thread re-arms defer to the batch flush; everything else
956 * submits immediately so remote workers wake from their wait. */
957 if (reactor->dispatching && pthread_equal(pthread_self(), reactor->owner) &&
958 reactor->deferred_n <
959 (uint32_t)(sizeof(reactor->deferred_sqes) / sizeof(reactor->deferred_sqes[0]))) {
960 uint32_t slot = reactor->deferred_n++;
961 reactor->deferred_sqes[slot] = sqe;
962 reactor->deferred_ctxs[slot] = ev_ctx;
963 return true;
964 }
965 if (uring_submit(reactor, &sqe)) {
966 return true;
967 }
968 free_reactor_ctx(reactor, ev_ctx);
969 return false;
970 } else {
971 struct epoll_event ev;
972 ev.events = (for_write ? EPOLLOUT : EPOLLIN) | EPOLLET | EPOLLONESHOT;
973 ev.data.ptr = ev_ctx;
974 if (epoll_ctl(reactor->impl.epoll_fd, EPOLL_CTL_ADD, fd, &ev) == 0) {
975 return true;
976 }
977 if (errno == EEXIST) {
978 if (epoll_ctl(reactor->impl.epoll_fd, EPOLL_CTL_MOD, fd, &ev) == 0) {
979 return true;
980 }
981 }
982 free_reactor_ctx(reactor, ev_ctx);
983 return false;
984 }
985#elif defined(__APPLE__) || defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__)
986 struct kevent change;
987 EV_SET(&change, fd, for_write ? EVFILT_WRITE : EVFILT_READ, EV_ADD | EV_CLEAR | EV_ONESHOT, 0,
988 0, ev_ctx);
989 if (kevent(reactor->impl.kq_fd, &change, 1, NULL, 0, NULL) == 0) {
990 return true;
991 }
992 free_reactor_ctx(reactor, ev_ctx);
993 return false;
994#endif
995 free_reactor_ctx(reactor, ev_ctx);
996 return false;
997}

1077/* CWIST_REACTOR_DRAIN_CHUNK: cooperative-queuing knob (see the comment at
1078 * its call site). Values > 0 bound how many connection callbacks run before
1079 * foreign-thread posts are drained. 0 disables the mid-batch drain (legacy
1080 * behavior). Unset or invalid defaults to 64 so a foreign-thread completion
1081 * does not queue behind a full CQE batch by default. Cached after the first
1082 * read like the other env knobs in this file -- the racy recompute is benign
1083 * (same result every time). */
1084static uint32_t reactor_drain_chunk(void) {
1085 static _Atomic int cached = -1;
1086 int v = atomic_load_explicit(&cached, memory_order_relaxed);
1087 if (v < 0) {
1088 const char *s = getenv("CWIST_REACTOR_DRAIN_CHUNK");
1089 if (s) {
1090 long parsed = strtol(s, NULL, 10);
1091 v = (parsed >= 0 && parsed < INT_MAX) ? (int)parsed : 64;
1092 } else {
1093 v = 64;
1094 }
1095 atomic_store_explicit(&cached, v, memory_order_relaxed);
1096 }
1097 return (uint32_t)v;
1098}

1100/* CQ grace: how long a reactor loop peeks its completion ring (pure
1101 * userspace, no syscall) before committing to the kernel wait. Completions
1102 * that land while a dispatch batch is being processed -- the common shape
1103 * under keep-alive load, where the peer's next request often arrives within
1104 * microseconds of the response -- would otherwise force a fresh wake each;
1105 * the grace folds them into the current one. 0 disables the spin.
1106 * CWIST_REACTOR_CQ_GRACE_NS overrides the 20 us default. */
1107static uint64_t reactor_cq_grace_ns(void) {
1108 static _Atomic long cached = -1;
1109 long v = atomic_load_explicit(&cached, memory_order_relaxed);
1110 if (v < 0) {
1111 const char *s = getenv("CWIST_REACTOR_CQ_GRACE_NS");
1112 if (s && *s) {
1113 char *end = NULL;
1114 long parsed = strtol(s, &end, 10);
1115 v = (end != s && parsed >= 0 && parsed < 1000000000L) ? parsed : 20000;
1116 } else {
1117 v = 20000;
1118 }
1119 atomic_store_explicit(&cached, v, memory_order_relaxed);
1120 }
1121 return (uint64_t)v;

1124/* Round budget: an io_uring dispatch round drains every CQE in the ring, and
1125 * a busy round can hold hundreds of connections' callbacks. Callbacks re-arm
1126 * their connections into the deferred SQE array, which only reaches the
1127 * kernel when the round ends -- so a request that arrived right after its
1128 * connection was re-armed waits out the whole batch, which is where the
1129 * arm-to-dispatch tail comes from (issue #166). The budget bounds that wait
1130 * without breaking the round: past it, parked re-arms are flushed to the
1131 * kernel at the next check instead of at round end (one extra enter per
1132 * check, only while a round is long). The round still runs to completion,
1133 * so the stop-at-round-boundary contract is unchanged. The check rides its
1134 * own cadence (every 64 callbacks), independent of CWIST_REACTOR_DRAIN_CHUNK.
1135 * 0 disables the bound. CWIST_REACTOR_ROUND_BUDGET_US overrides the 2000 us
1136 * default. */
1137static uint64_t reactor_round_budget_us(void) {
1138 static _Atomic long cached = -1;
1139 long v = atomic_load_explicit(&cached, memory_order_relaxed);
1140 if (v < 0) {
1141 const char *s = getenv("CWIST_REACTOR_ROUND_BUDGET_US");
1142 if (s && *s) {
1143 char *end = NULL;
1144 long parsed = strtol(s, &end, 10);
1145 v = (end != s && parsed >= 0 && parsed < 3600000000L) ? parsed : 2000;
1146 } else {
1147 v = 2000;
1148 }
1149 atomic_store_explicit(&cached, v, memory_order_relaxed);
1150 }
1151 return (uint64_t)v;
1152}

1175 * the SQ lock, then wait in a separate call. The submit enter
1176 * MUST hold the lock: uring_submit/uring_submit_batch roll the
1177 * SQ tail back when their own enter fails, and a concurrent
1178 * unlocked enter here could consume that SQE first, leaving
1179 * sq_tail behind sq_head. tail < head reads as a permanently
1180 * full SQ to every later submit (tail - head underflows past
1181 * sq_entries), so the reactor goes deaf: no re-arm, no accept,
1182 * no completions - the server stops answering while every thread
1183 * looks idle. With all SQ consumers under the lock, a failed
1184 * enter means the SQE was definitely not consumed and the
1185 * rollback is exact.
1186 *
1187 * The wait is bounded (like the epoll path's 100 ms poll)
1188 * because the shutdown handler runs with SA_RESTART: an
1189 * unbounded enter would be restarted after SIGTERM and hang an
1190 * idle reactor forever. It passes to_submit=0, so it consumes
1191 * nothing and needs no lock. */
1192 /* CQ grace: completions that land while the previous batch is
1193 * dispatching would otherwise force a fresh wake. After the
1194 * locked submit enter below, peek the CQ ring from userspace for
1195 * a bounded window; completions caught in time are dispatched in
1196 * the current round and the wait enter is skipped. Re-arms are
1197 * safe either way: the submit enter above already flushed them.
1198 * The deadline bounds the spin; on expiry (or when disabled) the
1199 * code falls through to the bounded wait below. */
1200 bool cq_pending = false;

1203 __atomic_load_n(reactor->impl.cq_head, __ATOMIC_ACQUIRE);

1205 struct timespec gs;
1206 clock_gettime(CLOCK_MONOTONIC, &gs);
1207 uint64_t g_end = (uint64_t)gs.tv_sec * 1000000000ull +
1208 (uint64_t)gs.tv_nsec + cq_grace_ns;
1209 uint32_t spins = 0;

1211 reactor_cpu_relax();

1213 clock_gettime(CLOCK_MONOTONIC, &gs);
1214 uint64_t now = (uint64_t)gs.tv_sec * 1000000000ull +
1215 (uint64_t)gs.tv_nsec;

1217 }
1218 ph = __atomic_load_n(reactor->impl.cq_head, __ATOMIC_ACQUIRE);
1219 }
1220 cq_pending = (ph != *reactor->impl.cq_tail);
1221 }
1222 }

1225 const uint64_t wait_ns = reactor_wait_ns(reactor);
1226 const struct __kernel_timespec idle_ts = {
1227 .tv_sec = (long long)(wait_ns / 1000000000ull),
1228 .tv_nsec = (long long)(wait_ns % 1000000000ull)};

1230 reactor->sq_unsubmitted = 0;
1231 if (to_submit > 0) {
1232 pthread_mutex_lock(&reactor->impl.sq_lock);
1233 int sret = sys_io_uring_enter(reactor->impl.ring_fd, to_submit, 0, 0, NULL);
1234 pthread_mutex_unlock(&reactor->impl.sq_lock);
1235 if (sret < 0) {
1236 /* Submission state is unknown on EINTR; anything else
1237 * failed before consuming. Either way the SQEs are
1238 * still in the SQ (the kernel caps resubmission at what
1239 * is actually there), so restore the count and let the
1240 * next round retry. */
1241 reactor->sq_unsubmitted += to_submit;
1242 }
1243 }
1244 if (!cq_pending) {
1245 int ret = sys_io_uring_enter_timeout(reactor->impl.ring_fd, 0, 1,
1246 IORING_ENTER_GETEVENTS, &idle_ts);
1247 if (ret < 0) {
1248 if (errno == EINTR) continue;
1249 if (errno == ETIME) continue;
1250 break;
1251 }
1252 }

1254 uint32_t tail = *reactor->impl.cq_tail;
1255 if (head == tail) {
1256 continue;
1257 }
1258 reactor->dispatching = true;
1259 const uint64_t round_budget_us = reactor_round_budget_us();
1260 uint64_t round_start_ns = 0;
1261 if (round_budget_us) {
1262 struct timespec ts;
1263 clock_gettime(CLOCK_MONOTONIC, &ts);
1264 round_start_ns = (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
1265 }
1266 /* Cooperative queuing: a busy round can carry hundreds of ready
1267 * connections in one CQE batch (the ring is 4096 deep). Foreign-
1268 * thread completions (cwist_async_defer, background jobs) queue
1269 * onto post_head via cwist_reactor_post() and used to wait for
1270 * reactor_drain_posts() at the *top* of the next round -- i.e.
1271 * behind this entire batch, even if the post arrived while we
1272 * were only a few callbacks in. CWIST_REACTOR_DRAIN_CHUNK bounds
1273 * how many connection callbacks run before posts are drained, so
1274 * a foreign-thread completion's own tail latency stops scaling
1275 * with how many *other* connections happened to be ready in the
1276 * same wake. The default (64) applies the bound out of the box;
1277 * 0 restores the legacy single-drain-at-end behavior. */

1279 uint32_t since_drain = 0;
1280 uint32_t since_budget_check = 0;
1281 while (head != tail) {
1282 struct io_uring_cqe *cqe = &reactor->impl.cqes[head & *reactor->impl.cq_ring_mask];
1283 uint64_t user_data = cqe->user_data;
1284 if (user_data & 1u) {
1285 /* Tagged data-path SQE (RX-uring RECV): the connection
1286 * pointer rides in the remaining bits. Dispatch on any
1287 * res, including negative ones: -EAGAIN is the POLL
1288 * fallback signal and 0 is EOF. Latency probe samples
1289 * are recorded by the handler itself (it owns the arm
1290 * timestamp), not by the ev_ctx wrapper below. */
1291 if (reactor->rx_cb) {
1292 reactor->rx_cb((void *)(user_data & ~(uint64_t)1u), cqe->res);
1293 }

1296 if (ev_ctx) {
1297 if (cqe->res >= 0) {

1299 struct timespec ts;
1300 clock_gettime(CLOCK_MONOTONIC, &ts);
1301 uint64_t t0 =
1302 (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;

1408 if (!reactor) return;

1410 __atomic_store_n(&reactor->running, false, __ATOMIC_RELEASE);

1412 * ring sleeps until a CQE arrives: the kernel ignores the enter
1413 * timeout for SQPOLL rings, so with no pending SQEs the wait never
1414 * returns and the pthread_join() in the pool destroy path hangs the
1415 * process (observed as a worker child surviving SIGTERM until SIGKILL,
1416 * wedging the supervisor's shutdown waitpid). Nudge the registered
1417 * wake fd so the run thread re-checks its shutdown flags. */
1418 if (reactor->wake_wr >= 0) {
1419 uint64_t one = 1;
1420 ssize_t ign = write(reactor->wake_wr, &one, sizeof(one));
1421 (void)ign; /* EAGAIN: a wake is already pending, which is fine. */
1422 }
