374static void latency_probe_record(uint64_t *buckets, uint64_t *count, uint64_t *sum_us,
375 uint64_t *max_us, uint64_t *over_5ms, uint64_t sample_us) {
376 int lo = 0, hi = LATENCY_PROBE_BUCKETS - 1;
377 while (lo < hi) { /* first bucket whose upper bound exceeds the sample */
378 int mid = (lo + hi) / 2;
379 if (sample_us < latency_probe_bounds_us[mid])
380 hi = mid;
381 else
382 lo = mid + 1;
383 }
384 buckets[lo]++;
385 (*count)++;
386 *sum_us += sample_us;
387 if (sample_us > *max_us) *max_us = sample_us;
388 if (sample_us > 5000) (*over_5ms)++;
389}

400static uint64_t latency_probe_percentile(const uint64_t *buckets, uint64_t count,
401 uint64_t pct_mille) {
402 uint64_t target = (count * pct_mille + 999) / 1000;

480/* Per-fd epoll registration state, process-wide (an fd belongs to one
481 * reactor at a time). It serves two purposes:
482 *
483 * - A hint for the one-shot path: a re-arm of an fd that is already
484 * registered goes straight to EPOLL_CTL_MOD instead of failing an ADD with
485 * EEXIST first, which used to cost two epoll_ctl calls per keep-alive
486 * request.
487 * - The RX-epoll receive path (reactor_rx.h): the fd stays registered
488 * edge-triggered, with no EPOLLONESHOT, and the reactor does the recv() into
489 * the armed buffer itself. A steady keep-alive request then costs a recv
490 * and a send, with no epoll_ctl.
491 *
492 * state is the only field touched off the run thread (by cwist_reactor_add,
493 * _mod and _del, which take an fd out of RX mode); the rest belongs to the
494 * run thread of the reactor that armed the fd. Every state value is only a
495 * hint for epoll_ctl: a stale one costs one failed call (ENOENT/EEXIST) and
496 * the fallback, never a lost registration, except RX, which recv_arm trusts
497 * only for the same connection pointer (see cwist_reactor_fd_forget). */
498enum { EPFD_UNKNOWN = 0, EPFD_ONESHOT = 1, EPFD_RX = 2 };

510#define EPFD_CHUNK_BITS 10
511#define EPFD_CHUNK (1u << EPFD_CHUNK_BITS)
512#define EPFD_MAX_FDS (1u << 22)
513static _Atomic(epoll_fd_slot_t *) g_epfd_chunks[EPFD_MAX_FDS / EPFD_CHUNK];

516static epoll_fd_slot_t *epfd_slot(int fd, bool create) {
517 if (fd < 0 || (unsigned)fd >= EPFD_MAX_FDS) return NULL;
518 _Atomic(epoll_fd_slot_t *) *chunk = &g_epfd_chunks[(unsigned)fd >> EPFD_CHUNK_BITS];
519 epoll_fd_slot_t *c = atomic_load_explicit(chunk, memory_order_acquire);
520 if (!c) {
521 if (!create) return NULL;
522 /* Chunks live as long as the process: reachable from the table and
523 * reused by every later fd in their range. */
524 epoll_fd_slot_t *fresh = cwist_alloc(EPFD_CHUNK * sizeof(*fresh));
525 if (!fresh) return NULL;
526 memset(fresh, 0, EPFD_CHUNK * sizeof(*fresh));
527 if (atomic_compare_exchange_strong_explicit(chunk, &c, fresh, memory_order_acq_rel,
528 memory_order_acquire)) {
529 c = fresh;
530 } else {
531 cwist_free(fresh); /* Another thread installed it first; c holds that one. */
532 }
533 }
534 return &c[(unsigned)fd & (EPFD_CHUNK - 1)];
535}

562static bool rx_ready_push(cwist_reactor_t *r, int fd, epoll_fd_slot_t *slot) {
563 if (slot->queued) return true;
564 if (r->rx_ready_n == r->rx_ready_cap) {
565 uint32_t cap = r->rx_ready_cap ? r->rx_ready_cap * 2 : 64;
566 int *grown = cwist_alloc(cap * sizeof(*grown));
567 if (!grown) return false;
568 if (r->rx_ready_n) memcpy(grown, r->rx_ready, r->rx_ready_n * sizeof(*grown));
569 cwist_free(r->rx_ready);
570 r->rx_ready = grown;
571 r->rx_ready_cap = cap;
572 }
573 r->rx_ready[r->rx_ready_n++] = fd;
574 slot->queued = true;
575 return true;
576}

580static void rx_epoll_recv(cwist_reactor_t *r, int fd, epoll_fd_slot_t *slot) {
581 ssize_t n;
582 do {
583 n = recv(fd, slot->buf, slot->len, MSG_DONTWAIT);
584 } while (n < 0 && errno == EINTR);
585 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
586 slot->maybe_more = false;
587 return;
588 }
589 int res = n >= 0 ? (int)n : -errno;
590 /* A full buffer may have left bytes behind; a short read drained the
591 * socket, so anything newer raises a fresh edge. */
592 slot->maybe_more = n > 0 && (size_t)n == slot->len;
593 slot->armed = false;
594 /* rx_cb may close and free the connection, or re-arm (possibly queueing
595 * this fd again); the slot itself stays valid. */
596 if (r->rx_cb) r->rx_cb(slot->conn, res);
597}

600static void rx_epoll_event(cwist_reactor_t *r, int fd) {
601 epoll_fd_slot_t *slot = epfd_slot(fd, false);
602 if (!slot || atomic_load_explicit(&slot->state, memory_order_relaxed) != EPFD_RX) return;
603 if (!slot->armed) {
604 /* The connection is between receives (serving, or parked on a
605 * deferred response). Remember the edge for the next arm. */
606 slot->maybe_more = true;
607 return;
608 }
609 rx_epoll_recv(r, fd, slot);
610}

614static void rx_epoll_drain_ready(cwist_reactor_t *r) {
615 uint32_t n = r->rx_ready_n;
616 for (uint32_t i = 0; i < n; i++) {
617 int fd = r->rx_ready[i];
618 epoll_fd_slot_t *slot = epfd_slot(fd, false);
619 if (!slot) continue;
620 slot->queued = false;
621 if (atomic_load_explicit(&slot->state, memory_order_relaxed) != EPFD_RX || !slot->armed) {
622 continue;
623 }
624 rx_epoll_recv(r, fd, slot);
625 }
626 /* Keep what the callbacks queued during this pass. */
627 memmove(r->rx_ready, r->rx_ready + n, (r->rx_ready_n - n) * sizeof(*r->rx_ready));
628 r->rx_ready_n -= n;

632static bool rx_epoll_arm(cwist_reactor_t *reactor, int fd, void *buf, unsigned len, void *conn) {
633 /* Off the run thread the slot would race the run thread's receive; the
634 * caller falls back to the one-shot POLL, whose callback arms from the
635 * run thread. */
636 if (!reactor_on_run_thread(reactor)) return false;
637 epoll_fd_slot_t *slot = epfd_slot(fd, true);
638 if (!slot) return false;
639 bool registered =
640 atomic_load_explicit(&slot->state, memory_order_relaxed) == EPFD_RX && slot->conn == conn;
641 slot->buf = buf;
642 slot->len = len;
643 slot->conn = conn;
644 if (!registered) {
645 /* First arm, or the fd went through the one-shot path since. MOD and
646 * ADD both report an fd that is already readable, so no edge is lost
647 * across the switch. */
648 struct epoll_event ev;
649 ev.events = EPOLLIN | EPOLLRDHUP | EPOLLET;
650 ev.data.u64 = ((uint64_t)(unsigned)fd << 1) | 1u;
651 if (!epoll_register(reactor->impl.epoll_fd, fd, &ev)) return false;
652 atomic_store_explicit(&slot->state, EPFD_RX, memory_order_relaxed);
653 slot->maybe_more = false;
654 slot->armed = true;
655 return true;
656 }
657 slot->armed = true;
658 if (slot->maybe_more && !rx_ready_push(reactor, fd, slot)) {
659 slot->armed = false;
660 return false;
661 }
662 return true;
663}

669bool cwist_reactor_recv_arm(cwist_reactor_t *reactor, int fd, void *buf, unsigned len, void *conn,
670 uint64_t *armed_ns) {
671 if (!reactor || fd < 0 || !buf || len == 0 || !conn) return false;
672 if (!cwist_reactor_rx_supported(reactor)) return false;
673 if (reactor->impl.use_epoll) {
674 if (armed_ns && latency_probe_enabled()) {
675 struct timespec ts;
676 clock_gettime(CLOCK_MONOTONIC, &ts);
677 *armed_ns = (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
678 }
679 return rx_epoll_arm(reactor, fd, buf, len, conn);
680 }
681 /* Tag-bit discipline: ev_ctx pointers come from cwist_alloc (malloc
682 * backed, at least max_align_t aligned), so the low bit is always zero
683 * there and one here. Refuse a misaligned connection pointer rather
684 * than aliasing into the ev_ctx namespace. */
685 if (((uintptr_t)conn & 1u) != 0) return false;
686
687 struct io_uring_sqe sqe;
688 memset(&sqe, 0, sizeof(sqe));
689 sqe.opcode = IORING_OP_RECV;
690 sqe.fd = fd;
691 sqe.addr = (uint64_t)(uintptr_t)buf;
692 sqe.len = len;
693 sqe.user_data = (uint64_t)(uintptr_t)conn | 1u;
694 if (armed_ns && latency_probe_enabled()) {
695 struct timespec ts;
696 clock_gettime(CLOCK_MONOTONIC, &ts);
697 *armed_ns = (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
698 }
699
700 /* Run-thread re-arms defer into the batch flush; everything else
701 * submits immediately so remote workers wake. Data SQEs carry no ev_ctx
702 * slot, so deferred_ctxs[] holds NULL for them and flush_deferred knows
703 * the connection pointer rides in the tagged user_data. */
704 if (reactor_on_run_thread(reactor) && reactor->dispatching &&
705 reactor->deferred_n <
706 (uint32_t)(sizeof(reactor->deferred_sqes) / sizeof(reactor->deferred_sqes[0]))) {
707 uint32_t slot = reactor->deferred_n++;
708 reactor->deferred_sqes[slot] = sqe;
709 reactor->deferred_ctxs[slot] = NULL;
710 return true;
711 }
712 return uring_submit(reactor, &sqe);
713}

782static reactor_event_ctx_t *alloc_reactor_ctx(cwist_reactor_t *r, int fd, cwist_reactor_cb_t cb,
783 const void *payload, size_t payload_size) {
784 if (!r || payload_size > CWIST_REACTOR_PAYLOAD_SIZE) return NULL;
785 pthread_mutex_lock(&r->pool_lock);
786 reactor_event_ctx_t *ev_ctx = r->free_head;
787 if (ev_ctx) r->free_head = (reactor_event_ctx_t *)ev_ctx->ctx;
788 pthread_mutex_unlock(&r->pool_lock);
789
790 if (!ev_ctx) {
791 /* Free list empty: grow a chunk. Allocate and link it outside the
792 * lock so the ~40 KiB setup does not serialize every other submitter
793 * on this reactor; under the lock the chunk is pushed onto the free
794 * list wholesale and one slot popped. Chunks are never freed until
795 * destroy, so a race with another grower only means two chunks. */
796 reactor_slot_chunk_t *chunk =
797 cwist_alloc(sizeof(*chunk) + REACTOR_CHUNK_EVENTS * sizeof(reactor_event_ctx_t));
798 if (!chunk) return NULL;
799 for (uint32_t i = 0; i + 1 < REACTOR_CHUNK_EVENTS; i++) {
800 chunk->slots[i].ctx = &chunk->slots[i + 1];
801 }
802 chunk->slots[REACTOR_CHUNK_EVENTS - 1].ctx = NULL;
803
804 pthread_mutex_lock(&r->pool_lock);
805 chunk->next = r->chunks;
806 r->chunks = chunk;
807 chunk->slots[REACTOR_CHUNK_EVENTS - 1].ctx = r->free_head;
808 r->free_head = &chunk->slots[0];
809 ev_ctx = r->free_head;
810 r->free_head = (reactor_event_ctx_t *)ev_ctx->ctx;
811 pthread_mutex_unlock(&r->pool_lock);
812 }
813 if (!ev_ctx) return NULL;
814
815 memset(ev_ctx, 0, sizeof(*ev_ctx));
816 ev_ctx->fd = fd;
817 ev_ctx->cb = cb;
818 ev_ctx->ctx = ev_ctx->payload;
819 if (payload && payload_size > 0) {
820 memcpy(ev_ctx->payload, payload, payload_size);
821 }
822 return ev_ctx;
823}

833bool cwist_reactor_post(cwist_reactor_t *r, cwist_reactor_post_t *node) {
834 if (!r || !node || !node->cb) return false;
835 cwist_reactor_post_t *head = atomic_load_explicit(&r->post_head, memory_order_relaxed);
836 do {
837 node->next = head;
838 } while (!atomic_compare_exchange_weak_explicit(&r->post_head, &head, node,
839 memory_order_release, memory_order_relaxed));
840 /* Wake only on the first post of a pending batch. A non-empty stack
841 * already has a wake pending; another write would be redundant. */
842 if (head == NULL && r->wake_wr >= 0) {
843 uint64_t one = 1;
844 ssize_t ign = write(r->wake_wr, &one, sizeof(one));
845 (void)ign; /* EAGAIN means the run thread is already awake. */
846 }
847 return true;
848}

850/* Pop the whole MPSC stack and run the callbacks oldest-first. Only ever
851 * called by the reactor's run thread (or destroy, after it has stopped). */
852static void reactor_drain_posts(cwist_reactor_t *r) {
853 cwist_reactor_post_t *list =
854 atomic_exchange_explicit(&r->post_head, NULL, memory_order_acquire);
855 cwist_reactor_post_t *rev = NULL;
856 while (list) {
857 cwist_reactor_post_t *next = list->next;
858 list->next = rev;
859 rev = list;
860 list = next;
861 }
862 while (rev) {
863 cwist_reactor_post_t *next = rev->next;
864 rev->cb(rev->ctx);
865 rev = next;
866 }
867}

877 * re-checked between waits (see cwist_reactor_run). */

880static uint64_t reactor_now_ns(void) {

886static void timer_heap_place(cwist_reactor_t *r, uint32_t i, cwist_reactor_timer_t *t) {
887 r->timer_heap[i] = t;
888 t->heap_slot = i + 1;
889}

891static void timer_heap_up(cwist_reactor_t *r, uint32_t i) {
892 cwist_reactor_timer_t *t = r->timer_heap[i];
893 while (i > 0) {
894 uint32_t parent = (i - 1) / 2;
895 if (r->timer_heap[parent]->deadline_ns <= t->deadline_ns) break;
896 timer_heap_place(r, i, r->timer_heap[parent]);
897 i = parent;
898 }
899 timer_heap_place(r, i, t);
900}
901
902static void timer_heap_down(cwist_reactor_t *r, uint32_t i) {
903 cwist_reactor_timer_t *t = r->timer_heap[i];
904 for (;;) {
905 uint32_t child = 2 * i + 1;
906 if (child >= r->timer_n) break;
907 if (child + 1 < r->timer_n &&
908 r->timer_heap[child + 1]->deadline_ns < r->timer_heap[child]->deadline_ns)
909 child++;
910 if (t->deadline_ns <= r->timer_heap[child]->deadline_ns) break;
911 timer_heap_place(r, i, r->timer_heap[child]);
912 i = child;
913 }
914 timer_heap_place(r, i, t);
915}

917static void timer_heap_remove(cwist_reactor_t *r, cwist_reactor_timer_t *t) {
918 uint32_t i = t->heap_slot - 1;
919 t->heap_slot = 0;
920 cwist_reactor_timer_t *last = r->timer_heap[--r->timer_n];
921 if (i == r->timer_n) return;
922 timer_heap_place(r, i, last);
923 if (i > 0 && r->timer_heap[(i - 1) / 2]->deadline_ns > last->deadline_ns)
924 timer_heap_up(r, i);
925 else
926 timer_heap_down(r, i);
927}

937bool cwist_reactor_timer_arm(cwist_reactor_t *r, cwist_reactor_timer_t *timer, uint64_t delay_us) {
938 if (!r || !timer || !timer->cb) return false;
939 if (timer->heap_slot) timer_heap_remove(r, timer);
940 if (r->timer_n == r->timer_cap) {
941 uint32_t cap = r->timer_cap ? r->timer_cap * 2 : 64;
942 cwist_reactor_timer_t **heap = cwist_alloc(cap * sizeof(*heap));
943 if (!heap) return false;
944 if (r->timer_n) memcpy(heap, r->timer_heap, r->timer_n * sizeof(*heap));
945 cwist_free(r->timer_heap);
946 r->timer_heap = heap;
947 r->timer_cap = cap;
948 }
949 timer->deadline_ns = reactor_now_ns() + delay_us * 1000ull;
950 r->timer_n++;
951 timer_heap_place(r, r->timer_n - 1, timer);
952 timer_heap_up(r, r->timer_n - 1);
953 return true;
954}

965/* Fire every timer whose deadline has passed. At most the timers armed on
966 * entry run, so a callback that re-arms itself with a zero delay waits for
967 * the next round instead of spinning here. */
968static void reactor_run_timers(cwist_reactor_t *r) {
969 if (r->timer_n == 0) return;
970 uint64_t now = reactor_now_ns();
971 uint32_t budget = r->timer_n;
972 while (budget-- > 0 && r->timer_n > 0 && r->timer_heap[0]->deadline_ns <= now) {
973 cwist_reactor_timer_t *t = r->timer_heap[0];
974 timer_heap_remove(r, t);
975 t->cb(t->ctx);
976 }
977}

979/* Poll wait for this round: the time to the earliest timer, capped at the
980 * idle wait. */
981static uint64_t reactor_wait_ns(const cwist_reactor_t *r) {
982 if (r->timer_n == 0) return REACTOR_IDLE_WAIT_NS;
983 uint64_t now = reactor_now_ns();
984 uint64_t deadline = r->timer_heap[0]->deadline_ns;
985 if (deadline <= now) return 0;
986 uint64_t wait = deadline - now;
987 return wait < REACTOR_IDLE_WAIT_NS ? wait : REACTOR_IDLE_WAIT_NS;
988}

990static void reactor_wake_cb(int fd, void *ctx) {
991 cwist_reactor_t *r = *(cwist_reactor_t *const *)ctx;
992 uint64_t buf[8];
993 while (read(fd, buf, sizeof(buf)) > 0) {
994 }
995 reactor_drain_posts(r);
996 /* One-shot slots are recycled after firing: re-arm for the next post. */
997 if (!cwist_reactor_add(r, fd, reactor_wake_cb, &r, sizeof(r))) {
998 int wr = r->wake_wr;
999 r->wake_fd = -1;
1000 r->wake_wr = -1;
1001 close(fd);
1002 if (wr >= 0 && wr != fd) close(wr);
1003 }
1004}

1006cwist_reactor_t *cwist_reactor_create(void) {
1007 cwist_reactor_t *r = cwist_alloc(sizeof(cwist_reactor_t));
1008 if (!r) return NULL;
1009 memset(r, 0, sizeof(cwist_reactor_t));
1010 __atomic_store_n(&r->running, false, __ATOMIC_RELEASE);
1011 pthread_mutex_init(&r->pool_lock, NULL);
1012#ifdef __linux__
1013 pthread_mutex_init(&r->impl.sq_lock, NULL);
1014#endif
1015
1016#ifdef __linux__
1017 r->impl.use_epoll = false;
1018 /* CWIST_REACTOR_BACKEND=epoll forces the epoll path for A/B measurement;
1019 * any other value (or unset) keeps io_uring as the default. */
1020 {
1021 const char *backend = getenv("CWIST_REACTOR_BACKEND");
1022 if (backend && strcmp(backend, "epoll") == 0) r->impl.use_epoll = true;
1023 }
1024 struct io_uring_params p;
1025 memset(&p, 0, sizeof(p));
1026 int fd = r->impl.use_epoll ? -1 : sys_io_uring_setup(4096, &p);
1027 if (fd >= 0) {
1028 r->impl.ring_fd = fd;
1029 r->impl.sq_ring_sz = sq_ring_size(&p);
1030 r->impl.cq_ring_sz = cq_ring_size(&p);
1031 r->impl.sqes_sz = p.sq_entries * sizeof(struct io_uring_sqe);
1032
1033 void *sq_ptr = mmap_ring(fd, r->impl.sq_ring_sz, IORING_OFF_SQ_RING);
1034 void *cq_ptr = mmap_ring(fd, r->impl.cq_ring_sz, IORING_OFF_CQ_RING);
1035 void *sqes_ptr = mmap_ring(fd, r->impl.sqes_sz, IORING_OFF_SQES);
1036
1037 if (sq_ptr && cq_ptr && sqes_ptr) {
1038 r->impl.sq_head = (uint32_t *)((char *)sq_ptr + p.sq_off.head);
1039 r->impl.sq_tail = (uint32_t *)((char *)sq_ptr + p.sq_off.tail);
1040 r->impl.sq_ring_mask = (uint32_t *)((char *)sq_ptr + p.sq_off.ring_mask);
1041 r->impl.sq_array = (uint32_t *)((char *)sq_ptr + p.sq_off.array);
1042 r->impl.sqes = sqes_ptr;
1043 r->impl.sq_entries = p.sq_entries;
1044
1045 r->impl.cq_head = (uint32_t *)((char *)cq_ptr + p.cq_off.head);
1046 r->impl.cq_tail = (uint32_t *)((char *)cq_ptr + p.cq_off.tail);
1047 r->impl.cq_ring_mask = (uint32_t *)((char *)cq_ptr + p.cq_off.ring_mask);
1048 r->impl.cqes = (struct io_uring_cqe *)((char *)cq_ptr + p.cq_off.cqes);
1049 r->impl.cq_entries = p.cq_entries;
1050 r->impl.active = true;
1051
1052 /* Identity SQE mapping is fixed for the ring's lifetime; fill it
1053 * once here instead of rewriting sq_array on every submission. */
1054 for (uint32_t i = 0; i < p.sq_entries; i++) r->impl.sq_array[i] = i;
1055 } else {
1056 if (sq_ptr) munmap(sq_ptr, r->impl.sq_ring_sz);
1057 if (cq_ptr) munmap(cq_ptr, r->impl.cq_ring_sz);
1058 if (sqes_ptr) munmap(sqes_ptr, r->impl.sqes_sz);
1059 close(fd);
1060 r->impl.use_epoll = true;
1061 }
1062 } else {
1063 r->impl.use_epoll = true;
1064 }
1065
1066 if (r->impl.use_epoll) {
1067 r->impl.epoll_fd = epoll_create1(0);
1068 if (r->impl.epoll_fd < 0) {
1069 pthread_mutex_destroy(&r->impl.sq_lock);
1070 pthread_mutex_destroy(&r->pool_lock);
1071 cwist_free(r);
1072 return NULL;
1073 }
1074 }
1075#elif defined(__APPLE__) || defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__)
1076 r->impl.kq_fd = kqueue();
1077 if (r->impl.kq_fd < 0) {
1078 pthread_mutex_destroy(&r->pool_lock);
1079 cwist_free(r);
1080 return NULL;
1081 }
1082#endif
1083 atomic_init(&r->post_head, NULL);
1084#ifdef __linux__
1085 r->wake_fd = eventfd(0, EFD_NONBLOCK | EFD_CLOEXEC);
1086 r->wake_wr = r->wake_fd;
1087#else
1088 r->wake_fd = -1;
1089 r->wake_wr = -1;
1090 {
1091 int pfds[2];
1092 if (pipe(pfds) == 0) {
1093 for (int i = 0; i < 2; i++) {
1094 int fl = fcntl(pfds[i], F_GETFL, 0);
1095 if (fl >= 0) fcntl(pfds[i], F_SETFL, fl | O_NONBLOCK);
1096 }
1097 r->wake_fd = pfds[0];
1098 r->wake_wr = pfds[1];
1099 }
1100 }
1101#endif
1102 /* Poll the wake fd on every backend. IORING_REGISTER_EVENTFD notifies
1103 * an external eventfd when CQEs arrive; writes to it do not create CQEs
1104 * or interrupt io_uring_enter. The one-shot poll is re-armed by
1105 * reactor_wake_cb, including when posts arrive during a dispatch. */
1106 if (r->wake_fd >= 0 && !cwist_reactor_add(r, r->wake_fd, reactor_wake_cb, &r, sizeof(r))) {
1107 /* Wake best-effort: the run loop still drains the stack each round. */
1108 close(r->wake_fd);
1109 if (r->wake_wr != r->wake_fd) close(r->wake_wr);
1110 r->wake_fd = -1;
1111 r->wake_wr = -1;
1112 }
1113 return r;
1114}

1116void cwist_reactor_destroy(cwist_reactor_t *reactor) {
1117 if (!reactor) return;
1118#ifdef __linux__
1119 latency_probe_dump(reactor);
1120#endif
1121 /* Run any completions posted after the run thread parked for good. */
1122 reactor_drain_posts(reactor);
1123 if (reactor->wake_fd >= 0) close(reactor->wake_fd);
1124 if (reactor->wake_wr >= 0 && reactor->wake_wr != reactor->wake_fd) close(reactor->wake_wr);
1125#ifdef __linux__
1126 if (!reactor->impl.use_epoll) {
1127 /* Teardown absorbed from io_uring_backend.c: unmap all three rings. */
1128 if (reactor->impl.sqes) munmap(reactor->impl.sqes, reactor->impl.sqes_sz);
1129 if (reactor->impl.cqes) {
1130 munmap((char *)reactor->impl.cqes -
1131 (reactor->impl.cq_ring_sz -
1132 reactor->impl.cq_entries * sizeof(struct io_uring_cqe)),
1133 reactor->impl.cq_ring_sz);
1134 }
1135 if (reactor->impl.sq_array) {
1136 munmap((char *)reactor->impl.sq_array -
1137 (reactor->impl.sq_ring_sz - reactor->impl.sq_entries * sizeof(uint32_t)),
1138 reactor->impl.sq_ring_sz);
1139 }
1140 close(reactor->impl.ring_fd);
1141 } else {

1170/* Submit one SQE immediately: the reactor never batches or defers submission,
1171 * so a woken worker always sees the event on its next wait. */
1172static bool uring_submit(cwist_reactor_t *reactor, struct io_uring_sqe *out_sqe) {
1173 bool ok = false;
1174 pthread_mutex_lock(&reactor->impl.sq_lock);
1175 uint32_t tail = *reactor->impl.sq_tail;
1176 uint32_t head = __atomic_load_n(reactor->impl.sq_head, __ATOMIC_ACQUIRE);
1177 if (tail - head < reactor->impl.sq_entries) {
1178 uint32_t index = tail & *reactor->impl.sq_ring_mask;
1179 struct io_uring_sqe *sqe = &reactor->impl.sqes[index];
1180 memcpy(sqe, out_sqe, sizeof(*sqe));
1181 reactor_tsan_release(&reactor->impl);
1182 __atomic_store_n(reactor->impl.sq_tail, tail + 1, __ATOMIC_RELEASE);
1183 if (sys_io_uring_enter(reactor->impl.ring_fd, 1, 0, 0, NULL) < 0) {
1184 if (cwist_async_debug_enabled()) {
1185 static _Atomic long dbg_enter_fail;
1186 long n = atomic_fetch_add(&dbg_enter_fail, 1) + 1;
1187 if (n <= 5 || n % 10000 == 0)
1188 fprintf(stderr,
1189 "[reactor] uring enter failed fd=%d errno=%d(%s) sq=%u/%u total=%ld\n",
1190 out_sqe->fd, errno, strerror(errno), tail - head,
1191 reactor->impl.sq_entries, n);
1192 }
1193 __atomic_store_n(reactor->impl.sq_tail, tail, __ATOMIC_RELEASE);
1194 } else {
1195 ok = true;
1196 }
1197 } else if (cwist_async_debug_enabled()) {
1198 static _Atomic long dbg_sq_full;
1199 long n = atomic_fetch_add(&dbg_sq_full, 1) + 1;
1200 if (n <= 5 || n % 10000 == 0)
1201 fprintf(stderr, "[reactor] SQ full fd=%d depth=%u/%u total=%ld\n", out_sqe->fd,
1202 tail - head, reactor->impl.sq_entries, n);
1203 }
1204 pthread_mutex_unlock(&reactor->impl.sq_lock);
1205 return ok;
1206}

1210/* Flush the deferred SQE queue with a single io_uring_enter for the whole
1211 * batch. On batch failure (SQ momentarily full from concurrent cross-thread
1212 * submissions) fall back to per-SQE submits with a short retry; a final
1213 * failure closes the connection and recycles its slot, matching every
1214 * caller's add-failure path. */
1215static bool uring_submit_batch(cwist_reactor_t *reactor, struct io_uring_sqe *batch, uint32_t n) {
1216 bool ok = false;
1217 pthread_mutex_lock(&reactor->impl.sq_lock);
1218 uint32_t tail = *reactor->impl.sq_tail;
1219 uint32_t head = __atomic_load_n(reactor->impl.sq_head, __ATOMIC_ACQUIRE);
1220 if (tail - head + n <= reactor->impl.sq_entries) {
1221 for (uint32_t i = 0; i < n; i++) {
1222 uint32_t index = (tail + i) & *reactor->impl.sq_ring_mask;
1223 memcpy(&reactor->impl.sqes[index], &batch[i], sizeof(batch[i]));
1224 }
1225 reactor_tsan_release(&reactor->impl);
1226 __atomic_store_n(reactor->impl.sq_tail, tail + n, __ATOMIC_RELEASE);
1227 if (sys_io_uring_enter(reactor->impl.ring_fd, n, 0, 0, NULL) >= 0) {
1228 ok = true;
1229 } else {
1230 __atomic_store_n(reactor->impl.sq_tail, tail, __ATOMIC_RELEASE);
1231 }
1232 }
1233 pthread_mutex_unlock(&reactor->impl.sq_lock);
1234 return ok;
1235}

1237static void flush_deferred(cwist_reactor_t *reactor) {
1238 uint32_t n = reactor->deferred_n;
1239 reactor->deferred_n = 0;
1240 if (n == 0) return;
1241 if (uring_submit_batch(reactor, reactor->deferred_sqes, n)) return;
1242 for (uint32_t i = 0; i < n; i++) {
1243 struct io_uring_sqe *sqe = &reactor->deferred_sqes[i];
1244 reactor_event_ctx_t *ev_ctx = reactor->deferred_ctxs[i];
1245 int attempt;
1246 for (attempt = 0; attempt < 100; attempt++) {
1247 if (uring_submit(reactor, sqe)) break;
1248 struct timespec ts = {.tv_sec = 0, .tv_nsec = 1000 * 1000};
1249 nanosleep(&ts, NULL); /* SQ drains without our help; wait it out */
1250 }
1251 if (attempt == 100) {
1252 if (cwist_async_debug_enabled()) {
1253 fprintf(stderr, "[reactor] deferred submit failed fd=%d; closing\n", (int)sqe->fd);
1254 }
1255 if (ev_ctx) {
1256 close((int)sqe->fd);
1257 free_reactor_ctx(reactor, ev_ctx);
1258 } else if (reactor->rx_cb) {
1259 /* Tagged RECV SQE: the connection owns the fd and shell, so
1260 * report the cancellation and let its handler clean up. */
1261 reactor->rx_cb((void *)(sqe->user_data & ~(uint64_t)1u), -ECANCELED);
1262 }
1263 }
1264 }
1265}

1269 * enter total instead of wait-enter + flush-enter. On SQ contention fall

1287 }

1293static bool reactor_add_common(cwist_reactor_t *reactor, int fd, cwist_reactor_cb_t cb,

1297 if (!ev_ctx) return false;

1299#ifdef __linux__
1300 if (latency_probe_enabled()) {
1301 struct timespec ts;
1302 clock_gettime(CLOCK_MONOTONIC, &ts);
1303 ev_ctx->armed_ns = (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
1304 }
1305 if (!reactor->impl.use_epoll) {

1318 reactor->deferred_n <

1326 return true;
1327 }

1353 free_reactor_ctx(reactor, ev_ctx);

1579void cwist_reactor_run(cwist_reactor_t *reactor) {
1580 if (!reactor) return;
1581 __atomic_store_n(&reactor->running, true, __ATOMIC_RELEASE);
1582 t_run_reactor = reactor;
1583
1584#ifdef __linux__
1585 if (!reactor->impl.use_epoll) {
1586 const uint64_t cq_grace_ns = reactor_cq_grace_ns();
1587 while (reactor_running(reactor)) {
1588 reactor_drain_posts(reactor);
1589 reactor_run_timers(reactor);
1590 /* Submit the re-arms queued by the previous dispatch batch under
1591 * the SQ lock, then wait in a separate call. The submit enter
1592 * MUST hold the lock: uring_submit/uring_submit_batch roll the
1593 * SQ tail back when their own enter fails, and a concurrent
1594 * unlocked enter here could consume that SQE first, leaving
1595 * sq_tail behind sq_head. tail < head reads as a permanently
1596 * full SQ to every later submit (tail - head underflows past
1597 * sq_entries), so the reactor goes deaf: no re-arm, no accept,
1598 * no completions - the server stops answering while every thread
1599 * looks idle. With all SQ consumers under the lock, a failed
1600 * enter means the SQE was definitely not consumed and the
1601 * rollback is exact.
1602 *
1603 * The wait is bounded (like the epoll path's 100 ms poll)
1604 * because the shutdown handler runs with SA_RESTART: an
1605 * unbounded enter would be restarted after SIGTERM and hang an
1606 * idle reactor forever. It passes to_submit=0, so it consumes
1607 * nothing and needs no lock. */
1608 /* CQ grace: completions that land while the previous batch is
1609 * dispatching would otherwise force a fresh wake. After the
1610 * locked submit enter below, peek the CQ ring from userspace for
1611 * a bounded window; completions caught in time are dispatched in
1612 * the current round and the wait enter is skipped. Re-arms are
1613 * safe either way: the submit enter above already flushed them.
1614 * The deadline bounds the spin; on expiry (or when disabled) the
1615 * code falls through to the bounded wait below. */
1616 bool cq_pending = false;
1617 if (cq_grace_ns > 0) {
1618 uint32_t ph =
1619 __atomic_load_n(reactor->impl.cq_head, __ATOMIC_ACQUIRE);
1620 if (ph == *reactor->impl.cq_tail) {
1621 struct timespec gs;
1622 clock_gettime(CLOCK_MONOTONIC, &gs);
1623 uint64_t g_end = (uint64_t)gs.tv_sec * 1000000000ull +
1624 (uint64_t)gs.tv_nsec + cq_grace_ns;
1625 uint32_t spins = 0;
1626 while (ph == *reactor->impl.cq_tail) {
1627 reactor_cpu_relax();
1628 if (((++spins) & 31u) == 0) {
1629 clock_gettime(CLOCK_MONOTONIC, &gs);
1630 uint64_t now = (uint64_t)gs.tv_sec * 1000000000ull +
1631 (uint64_t)gs.tv_nsec;
1632 if (now >= g_end) break;
1633 }
1634 ph = __atomic_load_n(reactor->impl.cq_head, __ATOMIC_ACQUIRE);
1635 }
1636 cq_pending = (ph != *reactor->impl.cq_tail);
1637 }
1638 }
