121static void ctx_tx_flush(cwist_webrtc_ctx *ctx) {
122 uint32_t n = ctx->tx_n;
123 ctx->tx_n = 0;
124 if (n == 0 || ctx->udp_fd < 0) return;
125#ifdef __linux__
126 struct mmsghdr msgs[CWIST_WEBRTC_TX_BATCH];
127 struct iovec iov[CWIST_WEBRTC_TX_BATCH];
128 for (uint32_t i = 0; i < n; i++) {
129 iov[i].iov_base = ctx->tx_bufs + (size_t)i * CWIST_WEBRTC_DGRAM_MAX;
130 iov[i].iov_len = ctx->tx_len[i];
131 memset(&msgs[i].msg_hdr, 0, sizeof(msgs[i].msg_hdr));
132 msgs[i].msg_hdr.msg_name = &ctx->tx_addr[i];
133 msgs[i].msg_hdr.msg_namelen = sizeof(ctx->tx_addr[i]);
134 msgs[i].msg_hdr.msg_iov = &iov[i];
135 msgs[i].msg_hdr.msg_iovlen = 1;
136 }
137 uint32_t off = 0;
138 while (off < n) {
139 int sent = sendmmsg(ctx->udp_fd, msgs + off, n - off, MSG_DONTWAIT);
140 if (sent > 0) {
141 off += (uint32_t)sent;
142 continue;
143 }
144 if (sent < 0 && errno == EINTR) continue;
145 /* EAGAIN/ENOBUFS or a per-destination error: UDP is lossy anyway and
146 * DTLS/SCTP retransmit, so skip this datagram and keep going. */
147 off++;
148 }
149#else
150 for (uint32_t i = 0; i < n; i++) {
151 sendto(ctx->udp_fd, ctx->tx_bufs + (size_t)i * CWIST_WEBRTC_DGRAM_MAX, ctx->tx_len[i], 0,
152 (const struct sockaddr *)&ctx->tx_addr[i], sizeof(ctx->tx_addr[i]));
153 }
154#endif
155}

159 const struct sockaddr_in *to) {

178static void conn_flush_pending(struct cwist_webrtc_conn *conn) {
179 while (conn->pending_head && conn->sctp_ready) {
180 cwist_webrtc_msg *m = conn->pending_head;
181 int rc = cwist_sctp_send(conn, m->channel, m->data, m->len, m->is_string);
182 if (rc == 1) break; /* send buffer full: retried after the next inbound packet */
183 conn->pending_head = m->next;
184 if (!conn->pending_head) conn->pending_tail = NULL;
185 atomic_fetch_sub_explicit(&conn->buffered, m->len, memory_order_relaxed);
186 cwist_free(m);

191static void conn_service(struct cwist_webrtc_conn *conn) {
192 if (atomic_load(&conn->detached)) return;
193 if (conn->state == CWIST_CONN_ESTABLISHED && !conn->close_requested) {
194 if (cwist_sctp_conn_drain(conn) < 0) conn->close_requested = true;
195 if (!conn->close_requested && !conn->sctp_ready && cwist_sctp_assoc_established(conn)) {
196 conn->sctp_ready = true;
197 conn->notified_open = true;
198 atomic_fetch_add(&conn->ctx->ready_count, 1);
199 }
200 if (!conn->close_requested) conn_flush_pending(conn);
201 }
202 if (conn->close_requested) conn_teardown(conn);

206static void ctx_leave(cwist_webrtc_ctx *ctx, cwist_webrtc_ctx *prev) {
207 if (prev != ctx) {
208 /* Servicing can dirty more conns (a message handler sending on
209 * another conn), so loop until the list stays empty. */
210 while (ctx->dirty) {
211 struct cwist_webrtc_conn *list = ctx->dirty;
212 ctx->dirty = NULL;
213 while (list) {
214 struct cwist_webrtc_conn *next = list->dirty_next;
215 list->dirty = false;
216 conn_service(list);
217 cwist_webrtc_conn_release(list);
218 list = next;
219 }
220 }
221 ctx_tx_flush(ctx);
222 tl_ctx = prev;
223 ctx_release(ctx);
224 }

248static void table_grow(cwist_webrtc_ctx *ctx) {
249 uint32_t nb = ctx->nbuckets * 2;
250 struct cwist_webrtc_conn **b = cwist_alloc(nb * sizeof(*b));
251 if (!b) return; /* keep the old, longer chains */
252 uint32_t old_n = ctx->nbuckets;
253 struct cwist_webrtc_conn **old = ctx->buckets;
254 ctx->buckets = b;
255 ctx->nbuckets = nb;
256 for (uint32_t i = 0; i < old_n; i++) {
257 struct cwist_webrtc_conn *c = old[i];
258 while (c) {
259 struct cwist_webrtc_conn *next = c->hnext;
260 uint32_t h = conn_hash(ctx, &c->remote);
261 c->hnext = b[h];
262 b[h] = c;
263 c = next;
264 }
265 }
266 cwist_free(old);
267}

321static void conn_teardown(struct cwist_webrtc_conn *conn) {
322 if (atomic_exchange(&conn->detached, true)) return;
323 cwist_webrtc_ctx *ctx = conn->ctx;
324 conn->state = CWIST_CONN_CLOSED;
325 cwist_reactor_timer_cancel(ctx->reactor, &conn->timer);
326 if (conn->sctp_ready) atomic_fetch_sub(&ctx->ready_count, 1);
327 conn->sctp_ready = false;
328 if (conn->sctp_sock || conn->sctp_acc) {
329 cwist_sctp_conn_close(conn);
330 if (--ctx->sctp_assocs == 0) cwist_reactor_timer_cancel(ctx->reactor, &ctx->sctp_timer);
331 }
332 if (conn->ssl) {
333 SSL_free(conn->ssl);
334 conn->ssl = NULL;
335 }
336 while (conn->pending_head) {
337 cwist_webrtc_msg *m = conn->pending_head;
338 conn->pending_head = m->next;
339 atomic_fetch_sub_explicit(&conn->buffered, m->len, memory_order_relaxed);
340 cwist_free(m);
341 }
342 conn->pending_tail = NULL;
343 bool notify = conn->notified_open;
344 conn->notified_open = false;
345 if (notify && ctx->close_cb) ctx->close_cb(conn, ctx->close_user);
346 if (conn->registered) {
347 table_remove(ctx, conn);
348 cwist_webrtc_conn_release(conn); /* the table's reference */
349 }

388static void conn_dtls_input(struct cwist_webrtc_conn *conn, const uint8_t *buf, size_t len) {
389 conn->dtls_in = buf;
390 conn->dtls_in_len = len;
391 if (conn->state == CWIST_CONN_DTLS) {
392 int rc = SSL_do_handshake(conn->ssl);
393 if (rc == 1) {
394 conn_on_dtls_up(conn);
395 } else if (SSL_get_error(conn->ssl, rc) == SSL_ERROR_SSL) {
396 conn->close_requested = true;
397 } else {
398 conn_rearm_timer(conn); /* a flight arrived: new retransmit deadline */
399 }
400 }
401 if (conn->state == CWIST_CONN_ESTABLISHED && !conn->close_requested) {
402 for (;;) {
403 int n = SSL_read(conn->ssl, conn->ctx->plain_buf, CWIST_WEBRTC_MAX_MESSAGE);
404 if (n > 0) {
405 cwist_sctp_conn_input(conn, conn->ctx->plain_buf, (size_t)n);
406 continue;
407 }
408 int err = SSL_get_error(conn->ssl, n);
409 if (err == SSL_ERROR_ZERO_RETURN || err == SSL_ERROR_SSL ||
410 (err == SSL_ERROR_SYSCALL && n == 0))
411 conn->close_requested = true;
412 break;
413 }
414 }
415 conn->dtls_in = NULL;
416 conn->dtls_in_len = 0;
417 conn_mark_dirty(conn);
418}

436static void conn_rearm_timer(struct cwist_webrtc_conn *conn) {
437 cwist_webrtc_ctx *ctx = conn->ctx;
438 uint64_t now = ctx->now_ns ? ctx->now_ns : mono_ns();
439 uint64_t delay_us;
440 switch (conn->state) {
441 case CWIST_CONN_STUN:
442 if (conn->parked) {
443 uint64_t end = conn->created_ns + PARK_TIMEOUT_MS * 1000000ull;
444 delay_us = end > now ? (end - now) / 1000 : 0;
445 } else if (!conn->is_server_role) {
446 delay_us = (uint64_t)conn->stun_rto_ms * 1000;
447 } else {
448 uint64_t end = conn->created_ns + HANDSHAKE_TIMEOUT_MS * 1000000ull;
449 delay_us = end > now ? (end - now) / 1000 : 0;
450 }
451 break;
452 case CWIST_CONN_DTLS: {
453 struct timeval tv;
454 uint64_t end = conn->created_ns + HANDSHAKE_TIMEOUT_MS * 1000000ull;
455 uint64_t hs_us = end > now ? (end - now) / 1000 : 0;
456 delay_us = hs_us;
457 if (DTLSv1_get_timeout(conn->ssl, &tv) == 1) {
458 uint64_t t = (uint64_t)tv.tv_sec * 1000000ull + (uint64_t)tv.tv_usec;
459 if (t < delay_us) delay_us = t;
460 }
461 break;
462 }
463 case CWIST_CONN_ESTABLISHED: {
464 uint64_t end = conn->last_rx_ns + LIVENESS_TIMEOUT_MS * 1000000ull;
465 delay_us = end > now ? (end - now) / 1000 : 0;
466 break;
467 }
468 default: return;
469 }
470 cwist_reactor_timer_arm(ctx->reactor, &conn->timer, delay_us);
471}

474static void conn_timer_cb(void *arg) {
475 struct cwist_webrtc_conn *conn = arg;
476 cwist_webrtc_ctx *ctx = conn->ctx;
477 cwist_webrtc_ctx *prev = ctx_enter(ctx);
478 uint64_t now = ctx->now_ns;
479 uint64_t age_ms = (now - conn->created_ns) / 1000000ull;
480 switch (conn->state) {
481 case CWIST_CONN_STUN:
482 if (conn->parked || conn->is_server_role) {
483 if (age_ms >= (conn->parked ? PARK_TIMEOUT_MS : HANDSHAKE_TIMEOUT_MS))
484 conn->close_requested = true;
485 } else if (conn->stun_attempts >= STUN_MAX_ATTEMPTS) {
486 conn->close_requested = true;
487 } else {
488 conn_send_stun_request(conn);
489 conn->stun_rto_ms = conn->stun_rto_ms * 2 > STUN_RTO_MAX_MS ? STUN_RTO_MAX_MS
490 : conn->stun_rto_ms * 2;
491 }
492 break;
493 case CWIST_CONN_DTLS:
494 if (age_ms >= HANDSHAKE_TIMEOUT_MS) {
495 conn->close_requested = true;
496 } else if (DTLSv1_handle_timeout(conn->ssl) < 0) {
497 conn->close_requested = true;
498 }
499 break;
500 case CWIST_CONN_ESTABLISHED:
501 if ((now - conn->last_rx_ns) / 1000000ull >= LIVENESS_TIMEOUT_MS)
502 conn->close_requested = true;
503 break;
504 default: break;
505 }
506 if (conn->close_requested)
507 conn_mark_dirty(conn);
508 else
509 conn_rearm_timer(conn);
510 ctx_leave(ctx, prev);
511}

599static void handle_stun(cwist_webrtc_ctx *ctx, const uint8_t *buf, size_t len,
600 const struct sockaddr_in *remote) {
601 if (cwist_ice_stun_is_binding_request(buf, len)) {
602 if (!atomic_load(&ctx->ice_lite_server))
603 return; /* no offer answered yet: not an ICE-lite endpoint */
604 if (!cwist_ice_stun_validate_request(buf, len, ctx->ice_pwd))
605 return; /* bad MESSAGE-INTEGRITY: drop (RFC 8445 would send a 400) */
606 uint8_t resp[256];
607 int rlen =
608 cwist_ice_stun_build_response(resp, sizeof(resp), buf, len, remote, ctx->ice_pwd);
609 if (rlen < 0) return;
610 ctx_tx_queue(ctx, resp, (size_t)rlen, remote);
611
612 bool nominate = cwist_ice_stun_has_use_candidate(buf, len);
613 struct cwist_webrtc_conn *conn = table_find(ctx, remote);
614 if (conn) {
615 conn->last_rx_ns = ctx->now_ns; /* consent check keeps it alive */
616 } else if (nominate) {
617 /* Adopt the conn parked by cwist_webrtc_handle_offer() when the
618 * USERNAME's remote ufrag matches; otherwise the request is still
619 * authenticated with our password, so start a fresh conn. */
620 char rfrag[64] = "";
621 cwist_ice_stun_get_remote_ufrag(buf, len, rfrag, sizeof(rfrag));
622 for (struct cwist_webrtc_conn *c = ctx->parked; c; c = c->hnext) {
623 if (rfrag[0] && strcmp(c->peer_ufrag, rfrag) == 0) {
624 conn = c;
625 break;
626 }
627 }
628 if (conn) {
629 table_remove(ctx, conn); /* the parked list's reference moves */
630 } else {
631 conn = conn_new(ctx, NULL, true);
632 if (!conn) return;
633 }
634 conn->remote = *remote;
635 conn->last_rx_ns = ctx->now_ns;
636 table_insert(ctx, conn);
637 }
638 if (conn && nominate && conn->state == CWIST_CONN_STUN) conn_start_dtls(conn);
639 return;
640 }
641 /* STUN response: only client-role conns expect these. */
642 struct cwist_webrtc_conn *conn = table_find(ctx, remote);
643 if (!conn || conn->is_server_role || conn->state != CWIST_CONN_STUN) return;
644 struct sockaddr_in mapped;
645 if (cwist_ice_stun_parse_response(buf, len, conn->stun_txid, conn->peer_pwd, &mapped)) {
646 conn->last_rx_ns = ctx->now_ns;
647 conn_start_dtls(conn);
648 }
649}

652static void handle_packet(cwist_webrtc_ctx *ctx, const uint8_t *buf, size_t len,
653 const struct sockaddr_in *remote) {
654 if (len == 0) return;
655 if (buf[0] <= 3) {
656 if (cwist_ice_stun_is_message(buf, len)) handle_stun(ctx, buf, len, remote);
657 return;
658 }
659 if (buf[0] < 20 || buf[0] > 63) return; /* not DTLS (RTP/RTCP or garbage) */
660 struct cwist_webrtc_conn *conn = table_find(ctx, remote);
661 if (!conn || !conn->ssl || atomic_load(&conn->detached)) return;
662 conn->last_rx_ns = ctx->now_ns;
663 conn_dtls_input(conn, buf, len);
664}

689static void udp_readable(int fd, void *payload) {
690 cwist_webrtc_ctx *ctx = *(cwist_webrtc_ctx **)payload;
691 ctx->udp_armed = false;
692 if (ctx->closed) {
693 /* Woken by teardown: the slot is consumed, so the fd can go. */
694 ctx_close_fd(ctx);
695 if (!ctx->owns_reactor) ctx_release(ctx);
696 return;
697 }
698 cwist_webrtc_ctx *prev = ctx_enter(ctx);
699 int handled = 0;
700 while (handled < RX_BUDGET) {
701#ifdef __linux__
702 struct mmsghdr msgs[CWIST_WEBRTC_RX_BATCH];
703 struct iovec iov[CWIST_WEBRTC_RX_BATCH];
704 struct sockaddr_in from[CWIST_WEBRTC_RX_BATCH];
705 for (int i = 0; i < CWIST_WEBRTC_RX_BATCH; i++) {
706 iov[i].iov_base = ctx->rx_bufs + (size_t)i * CWIST_WEBRTC_DGRAM_MAX;
707 iov[i].iov_len = CWIST_WEBRTC_DGRAM_MAX;
708 memset(&msgs[i].msg_hdr, 0, sizeof(msgs[i].msg_hdr));
709 msgs[i].msg_hdr.msg_name = &from[i];
710 msgs[i].msg_hdr.msg_namelen = sizeof(from[i]);
711 msgs[i].msg_hdr.msg_iov = &iov[i];
712 msgs[i].msg_hdr.msg_iovlen = 1;
713 }
714 int n = recvmmsg(fd, msgs, CWIST_WEBRTC_RX_BATCH, MSG_DONTWAIT, NULL);
715 if (n < 0 && errno == EINTR) continue;
716 if (n <= 0) break;
717 for (int i = 0; i < n; i++) {
718 if (msgs[i].msg_hdr.msg_flags & MSG_TRUNC) continue;
719 if (msgs[i].msg_hdr.msg_namelen < sizeof(struct sockaddr_in) ||
720 from[i].sin_family != AF_INET)
721 continue;
722 handle_packet(ctx, iov[i].iov_base, msgs[i].msg_len, &from[i]);
723 }
724 handled += n;
725 if (n < CWIST_WEBRTC_RX_BATCH) break;
726#else
727 struct sockaddr_in from;
728 socklen_t flen = sizeof(from);
729 ssize_t n = recvfrom(fd, ctx->rx_bufs, CWIST_WEBRTC_DGRAM_MAX, MSG_DONTWAIT,
730 (struct sockaddr *)&from, &flen);
731 if (n < 0 && errno == EINTR) continue;
732 if (n <= 0) break;
733 if (flen >= sizeof(struct sockaddr_in) && from.sin_family == AF_INET)
734 handle_packet(ctx, ctx->rx_bufs, (size_t)n, &from);
735 handled++;
736#endif
737 }
738 if (handled > 0) ctx_note_activity(ctx);
739 ctx_leave(ctx, prev);
740 if (!ctx->closed) {
741 /* The slot that fired held a ref; the new slot takes its own. */
742 ctx_arm_udp_slot(ctx);
743 }
744 if (!ctx->owns_reactor) ctx_release(ctx);
745}

750static void detect_host_ip(char *out, size_t cap) {
751 snprintf(out, cap, "127.0.0.1");
752 struct ifaddrs *ifas = NULL;
753 if (getifaddrs(&ifas) < 0) return;
754 for (struct ifaddrs *ifa = ifas; ifa; ifa = ifa->ifa_next) {
755 if (!ifa->ifa_addr || ifa->ifa_addr->sa_family != AF_INET) continue;
756 struct sockaddr_in *sin = (struct sockaddr_in *)ifa->ifa_addr;
757 uint32_t a = ntohl(sin->sin_addr.s_addr);
758 if (((a >> 24) & 0xFF) == 127) continue;
759 snprintf(out, cap, "%u.%u.%u.%u", (a >> 24) & 0xFF, (a >> 16) & 0xFF, (a >> 8) & 0xFF,
760 a & 0xFF);
761 break;
762 }
763 freeifaddrs(ifas);
764}

769static void ctx_release(cwist_webrtc_ctx *ctx) {
770 if (atomic_fetch_sub_explicit(&ctx->refs, 1, memory_order_acq_rel) != 1) return;
771 ctx_release_resources(ctx);
772 pthread_mutex_destroy(&ctx->free_mu);
773 pthread_cond_destroy(&ctx->free_cv);
774 cwist_free(ctx);
775}

778static void ctx_release_resources(cwist_webrtc_ctx *ctx) {
779 if (ctx->server_ssl_ctx) SSL_CTX_free(ctx->server_ssl_ctx);
780 if (ctx->client_ssl_ctx) SSL_CTX_free(ctx->client_ssl_ctx);
781 if (ctx->cert) X509_free(ctx->cert);
782 if (ctx->pkey) EVP_PKEY_free(ctx->pkey);
783 ctx->server_ssl_ctx = ctx->client_ssl_ctx = NULL;
784 ctx->cert = NULL;
785 ctx->pkey = NULL;
786 cwist_free(ctx->buckets);
787 cwist_free(ctx->rx_bufs);
788 cwist_free(ctx->plain_buf);
789 cwist_free(ctx->sctp_buf);
790 cwist_free(ctx->tx_bufs);
791 ctx->buckets = NULL;
792 ctx->rx_bufs = ctx->plain_buf = ctx->sctp_buf = ctx->tx_bufs = NULL;
793}

799static void ctx_teardown_owner(cwist_webrtc_ctx *ctx) {
800 if (ctx->closed) return;
801 cwist_webrtc_ctx *prev = ctx_enter(ctx);
802 while (ctx->parked) conn_teardown(ctx->parked);
803 for (uint32_t i = 0; i < ctx->nbuckets; i++) {
804 while (ctx->buckets[i]) conn_teardown(ctx->buckets[i]);
805 }
806 cwist_reactor_timer_cancel(ctx->reactor, &ctx->sctp_timer);
807 ctx_tx_flush(ctx); /* SCTP ABORTs from the closes */
808 ctx->closed = true;
809 ctx_leave(ctx, prev); /* dirty conns are detached: only their refs drop */
810}

813static cwist_webrtc_ctx *ctx_create(cwist_reactor_t *reactor, bool owns_reactor, uint16_t port) {
814 if (!reactor || cwist_sctp_global_init() < 0) return NULL;
815 cwist_webrtc_ctx *ctx = cwist_malloc(sizeof(*ctx));
816 if (!ctx) return NULL;
817 ctx->reactor = reactor;
818 ctx->owns_reactor = owns_reactor;
819 ctx->owner_pid = getpid();
820 atomic_init(&ctx->refs, 1);
821 atomic_init(&ctx->ice_lite_server, 0);
822 atomic_init(&ctx->ready_count, 0);
823 ctx->udp_fd = -1;
824 pthread_mutex_init(&ctx->free_mu, NULL);
825 pthread_cond_init(&ctx->free_cv, NULL);
826 cwist_reactor_timer_init(&ctx->sctp_timer, ctx_sctp_timer_cb, ctx);
827
828 ctx->nbuckets = 64;
829 ctx->buckets = cwist_alloc(ctx->nbuckets * sizeof(*ctx->buckets));
830 ctx->rx_bufs = cwist_alloc((size_t)CWIST_WEBRTC_RX_BATCH * CWIST_WEBRTC_DGRAM_MAX);
831 ctx->tx_bufs = cwist_alloc((size_t)CWIST_WEBRTC_TX_BATCH * CWIST_WEBRTC_DGRAM_MAX);
832 ctx->plain_buf = cwist_alloc(CWIST_WEBRTC_MAX_MESSAGE);
833 ctx->sctp_buf = cwist_alloc(CWIST_WEBRTC_MAX_MESSAGE);
834 if (!ctx->buckets || !ctx->rx_bufs || !ctx->tx_bufs || !ctx->plain_buf || !ctx->sctp_buf)
835 goto fail;
836
837 ctx->udp_fd = socket(AF_INET, SOCK_DGRAM, 0);
838 if (ctx->udp_fd < 0) goto fail;
839 int flags = fcntl(ctx->udp_fd, F_GETFL, 0);
840 fcntl(ctx->udp_fd, F_SETFL, flags | O_NONBLOCK);
841 int buf = UDP_SOCKBUF;
842 setsockopt(ctx->udp_fd, SOL_SOCKET, SO_RCVBUF, &buf, sizeof(buf));
843 setsockopt(ctx->udp_fd, SOL_SOCKET, SO_SNDBUF, &buf, sizeof(buf));
844 struct sockaddr_in addr;
845 memset(&addr, 0, sizeof(addr));
846 addr.sin_family = AF_INET;
847 addr.sin_addr.s_addr = htonl(INADDR_ANY);
848 addr.sin_port = htons(port);
849 if (bind(ctx->udp_fd, (struct sockaddr *)&addr, sizeof(addr)) < 0) goto fail;
850 socklen_t alen = sizeof(addr);
851 if (getsockname(ctx->udp_fd, (struct sockaddr *)&addr, &alen) < 0) goto fail;
852 ctx->port = ntohs(addr.sin_port);
853
854 if (cwist_dtls_generate_cert(&ctx->cert, &ctx->pkey) < 0) goto fail;
855 if (cwist_dtls_fingerprint(ctx->cert, ctx->fingerprint, sizeof(ctx->fingerprint)) < 0)
856 goto fail;
857 ctx->server_ssl_ctx = cwist_dtls_ctx_new(1, ctx->cert, ctx->pkey);
858 ctx->client_ssl_ctx = cwist_dtls_ctx_new(0, NULL, NULL);
859 if (!ctx->server_ssl_ctx || !ctx->client_ssl_ctx) goto fail;
860
861 cwist_ice_random_creds(ctx->ice_ufrag, sizeof(ctx->ice_ufrag), ctx->ice_pwd,
862 sizeof(ctx->ice_pwd));
863 detect_host_ip(ctx->host_ip, sizeof(ctx->host_ip));
864 return ctx;
865
866fail:
867 if (ctx->udp_fd >= 0) close(ctx->udp_fd);
868 ctx_release(ctx);
869 return NULL;
870}

912static void ctx_teardown_on_reactor(cwist_webrtc_ctx *ctx) {
913 ctx_teardown_owner(ctx);
914 if (ctx->udp_armed && ctx->udp_fd >= 0) {
915 struct sockaddr_in self;
916 memset(&self, 0, sizeof(self));
917 self.sin_family = AF_INET;
918 self.sin_port = htons(ctx->port);
919 self.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
920 uint8_t b = 0;
921 if (sendto(ctx->udp_fd, &b, 1, 0, (struct sockaddr *)&self, sizeof(self)) == 1) return;
922 /* Could not wake it: cancel the slot and close here. */
923 cwist_reactor_del(ctx->reactor, ctx->udp_fd);
924 ctx->udp_armed = false;
925 ctx_release(ctx); /* the slot's reference */
926 }
927 ctx_close_fd(ctx);
928}

944void cwist_webrtc_ctx_free(cwist_webrtc_ctx *ctx) {
945 if (!ctx) return;
946 if (getpid() != ctx->owner_pid) {
947 /* A forked child's copy: its reactor thread exists only in the
948 * parent, and the sockets and memory are the parent's to release.
949 * Leave the copy alone; it goes away with this process. */
950 return;
951 }
952 if (ctx->owns_reactor) {
953 if (ctx->thread_running) {
954 ctx->free_post.cb = ctx_stop_post_cb;
955 ctx->free_post.ctx = ctx;
956 cwist_reactor_post(ctx->reactor, &ctx->free_post);
957 pthread_join(ctx->thread, NULL);
958 }
959 /* The loop is gone (or never ran): finish here. Destroying the
960 * reactor drains leftover posts, which only drop references. */
961 ctx_teardown_owner(ctx);
962 if (ctx->udp_fd >= 0) {
963 close(ctx->udp_fd);
964 ctx->udp_fd = -1;
965 }
966 cwist_reactor_destroy(ctx->reactor);
967 ctx->reactor = NULL;
968 ctx_release(ctx);
969 return;
970 }
971 if (tl_ctx == ctx) {
972 /* Called from one of our own callbacks: tear down inline; the read
973 * slot's callback finishes the fd close after we return. */
974 ctx_teardown_on_reactor(ctx);
975 } else {
976 ctx_retain(ctx);
977 ctx->free_post.cb = ctx_free_post_cb;
978 ctx->free_post.ctx = ctx;
979 cwist_reactor_post(ctx->reactor, &ctx->free_post);
980 pthread_mutex_lock(&ctx->free_mu);
981 while (!ctx->fd_closed) pthread_cond_wait(&ctx->free_cv, &ctx->free_mu);
982 pthread_mutex_unlock(&ctx->free_mu);
983 }
984 ctx_release(ctx);
985}

1027 struct cwist_webrtc_conn *conn = arg;

1030 cwist_webrtc_conn_release(conn);

1042int cwist_webrtc_handle_offer(cwist_webrtc_ctx *ctx, const char *offer, char *answer_out,
1043 size_t out_len) {
1044 cwist_sdp_info info;
1045 if (!ctx || !offer || cwist_sdp_parse(offer, strlen(offer), &info) < 0) return -1;
1046 if (getpid() != ctx->owner_pid)
1047 return -1; /* inherited across fork(): this process has no loop for it */
1048 const char *mid = info.mid[0] ? info.mid : "0";
1049 if (cwist_sdp_write_answer(answer_out, out_len, ctx->fingerprint, ctx->ice_ufrag, ctx->ice_pwd,
1050 mid, ctx->host_ip, ctx->port) < 0)
1051 return -1;
1052 struct cwist_webrtc_conn *conn = conn_new(ctx, NULL, true);
1053 if (!conn) return -1;
1054 snprintf(conn->peer_ufrag, sizeof(conn->peer_ufrag), "%s", info.ice_ufrag);
1055 snprintf(conn->peer_pwd, sizeof(conn->peer_pwd), "%s", info.ice_pwd);
1056 atomic_store(&ctx->ice_lite_server, 1);
1057 conn->reg_post.cb = park_post_cb;
1058 conn->reg_post.ctx = conn;
1059 cwist_reactor_post(ctx->reactor, &conn->reg_post);
1060 return 0;
1061}

1112static void kick_post_cb(void *arg) {
1113 struct cwist_webrtc_conn *conn = arg;
1114 /* acq_rel: producers that see the emptied inbox re-post kick_post, which
1115 * must happen after the reactor's reads of it for this run. */
1116 cwist_webrtc_msg *list = atomic_exchange_explicit(&conn->inbox, NULL, memory_order_acq_rel);
1117 cwist_webrtc_msg *rev = NULL;
1118 while (list) {
1119 cwist_webrtc_msg *next = list->next;
1120 list->next = rev;
1121 rev = list;
1122 list = next;
1123 }
1124 if (atomic_load(&conn->detached)) {
1125 while (rev) {
1126 cwist_webrtc_msg *next = rev->next;
1127 atomic_fetch_sub_explicit(&conn->buffered, rev->len, memory_order_relaxed);
1128 cwist_free(rev);
1129 rev = next;
1130 }
1131 } else if (rev) {
1132 cwist_webrtc_ctx *prev = ctx_enter(conn->ctx);
1133 ctx_note_activity(conn->ctx);
1134 while (rev) {
1135 cwist_webrtc_msg *next = rev->next;
1136 conn_enqueue(conn, rev);
1137 rev = next;
1138 }
1139 conn_mark_dirty(conn);
1140 ctx_leave(conn->ctx, prev);
1141 }
1142 cwist_webrtc_conn_release(conn); /* the kick's reference */
1143}

1149static bool conn_reserve(struct cwist_webrtc_conn *conn, size_t len) {
1150 size_t before = atomic_fetch_add_explicit(&conn->buffered, len, memory_order_relaxed);
1151 if (before + len > CWIST_WEBRTC_MAX_BUFFERED && before > 0) {
1152 atomic_fetch_sub_explicit(&conn->buffered, len, memory_order_relaxed);
1153 return false;
1154 }
1155 return true;

1161 (len > 0 && !data))

1168 if (rc == 0) return 0;
1169 if (rc < 0) return -1;
1170 }

1172 cwist_webrtc_msg *m = cwist_alloc(sizeof(*m) + len);

1176 }

1179 m->channel = channel_id;
1180 m->is_string = (uint8_t)is_string;
1181 if (len) memcpy(m->data, data, len);

1183 conn_mark_dirty(conn);
1184 return 0;
1185 }
1186 if (!conn_reserve(conn, len)) return -1;
1187 cwist_webrtc_msg *m = cwist_alloc(sizeof(*m) + len);
1188 if (!m) {
1189 atomic_fetch_sub_explicit(&conn->buffered, len, memory_order_relaxed);
1190 return -1;
1191 }
1192 m->conn = conn;
1193 m->len = (uint32_t)len;
1194 m->channel = channel_id;
1195 m->is_string = (uint8_t)is_string;
1196 if (len) memcpy(m->data, data, len);
1197 cwist_webrtc_msg *head = atomic_load_explicit(&conn->inbox, memory_order_acquire);
1198 do {
1199 m->next = head;
1200 } while (!atomic_compare_exchange_weak_explicit(&conn->inbox, &head, m, memory_order_acq_rel,
1201 memory_order_acquire));

1203 /* First message of a batch: wake the owner. The kick node is reused;
1204 * the reactor reads its link before running it, so re-posting from

1208 conn->kick_post.ctx = conn;
1209 cwist_reactor_post(conn->ctx->reactor, &conn->kick_post);
1210 }

1212}

1216 struct cwist_webrtc_conn *conn = arg;
1217 if (!atomic_load(&conn->detached)) {
1218 cwist_webrtc_ctx *prev = ctx_enter(conn->ctx);
1219 conn->close_requested = true;
1220 conn_mark_dirty(conn);
1221 ctx_leave(conn->ctx, prev);
1222 }
1223 atomic_store(&conn->close_posted, false);
1224 cwist_webrtc_conn_release(conn);
1225}

1227void cwist_webrtc_conn_close(cwist_webrtc_conn *conn) {
1228 if (!conn || atomic_load(&conn->detached)) return;
1229 if (cwist_webrtc_on_owner(conn->ctx)) {
1230 /* Deferred to the end of the round: the caller may be inside a
1231 * message callback that is still reading this conn's SCTP socket. */
1232 conn->close_requested = true;
1233 conn_mark_dirty(conn);
1234 return;
1235 }
1236 if (atomic_exchange(&conn->close_posted, true)) return;
1237 cwist_webrtc_conn_retain(conn);
1238 conn->close_post.cb = close_post_cb;
1239 conn->close_post.ctx = conn;
1240 cwist_reactor_post(conn->ctx->reactor, &conn->close_post);
