53typedef struct {

87static void write_all(int fd, const char *p, size_t len) {
88 while (len > 0) {
89 ssize_t n = write(fd, p, len);
90 if (n < 0) {
91 if (errno == EINTR) continue;
92 return;
93 }
94 p += n;
95 len -= (size_t)n;
96 }
97}

102static void *log_writer_main(void *arg) {
103 (void)arg;
104 pthread_mutex_lock(&g_lock);
105 for (;;) {
106 while (rings_empty() && !g_stopping) pthread_cond_wait(&g_work, &g_lock);
107 if (rings_empty()) break; /* stopping and drained */
108
109 for (int i = 0; i < LOG_ASYNC_STREAMS; i++) {
110 log_ring_t *r = &g_rings[i];
111 while (r->head != r->tail) {
112 size_t off = r->tail % r->cap;
113 size_t len = r->head - r->tail;
114 if (len > r->cap - off) len = r->cap - off; /* up to the wrap */
115 const char *p = r->buf + off;
116 pthread_mutex_unlock(&g_lock);
117 write_all(i == 0 ? STDOUT_FILENO : STDERR_FILENO, p, len);
118 pthread_mutex_lock(&g_lock);
119 r->tail += len;
120 pthread_cond_broadcast(&g_space);
121 }
122 }
123 }
124 pthread_mutex_unlock(&g_lock);
125 return NULL;
126}

175bool cwist_log_async_start(size_t buffer_bytes) {
176 if (buffer_bytes == 0) buffer_bytes = LOG_ASYNC_DEFAULT_BYTES;
177
178 pthread_mutex_lock(&g_lock);
179 if (g_active) {
180 pthread_mutex_unlock(&g_lock);
181 return true;
182 }
183 if (!g_hooks_installed) {
184 if (pthread_atfork(atfork_prepare, atfork_parent, atfork_child) != 0 ||
185 atexit(atexit_stop) != 0) {
186 pthread_mutex_unlock(&g_lock);
187 return false;
188 }
189 g_hooks_installed = true;
190 }
191 for (int i = 0; i < LOG_ASYNC_STREAMS; i++) {
192 log_ring_t *r = &g_rings[i];
193 r->buf = cwist_alloc(buffer_bytes);
194 if (!r->buf) {
195 for (int j = 0; j < i; j++) {
196 cwist_free(g_rings[j].buf);
197 g_rings[j].buf = NULL;
198 }
199 pthread_mutex_unlock(&g_lock);
200 return false;
201 }
202 r->cap = buffer_bytes;
203 r->head = r->tail = 0;
204 }
205 g_stopping = false;
206 if (!spawn_writer()) {
207 for (int i = 0; i < LOG_ASYNC_STREAMS; i++) {
208 cwist_free(g_rings[i].buf);
209 g_rings[i].buf = NULL;
210 }
211 pthread_mutex_unlock(&g_lock);
212 return false;
213 }
214 g_active = true;
215 pthread_mutex_unlock(&g_lock);
216 return true;
217}

228void cwist_log_async_stop(void) {
229 pthread_mutex_lock(&g_lock);
230 if (!g_active) {
231 pthread_mutex_unlock(&g_lock);
232 return;
233 }
234 g_active = false; /* new lines go synchronous from here on */
235 g_stopping = true;
236 bool join = g_writer_alive;
237 pthread_cond_broadcast(&g_work);
238 pthread_cond_broadcast(&g_space); /* callers waiting for space fall back */
239 pthread_mutex_unlock(&g_lock);
240
241 if (join) pthread_join(g_writer, NULL);
242
243 pthread_mutex_lock(&g_lock);
244 g_writer_alive = false;
245 for (int i = 0; i < LOG_ASYNC_STREAMS; i++) {
246 cwist_free(g_rings[i].buf);
247 g_rings[i].buf = NULL;
248 g_rings[i].cap = g_rings[i].head = g_rings[i].tail = 0;
249 }
250 pthread_mutex_unlock(&g_lock);
251}

262bool cwist_log_async_submit(int fd, const char *line, size_t len) {
263 if (len == 0) return true;
264 pthread_mutex_lock(&g_lock);
265 log_ring_t *r = ring_for_fd(fd);
266 if (!g_active || !r || len > r->cap) {
267 pthread_mutex_unlock(&g_lock);
268 return false;
269 }
270 if (!g_writer_alive && !spawn_writer()) { /* first line in a forked child */
271 pthread_mutex_unlock(&g_lock);
272 return false;
273 }
274 while (g_active && r->cap - (r->head - r->tail) < len) pthread_cond_wait(&g_space, &g_lock);
275 if (!g_active) { /* stopped while waiting */
276 pthread_mutex_unlock(&g_lock);
277 return false;
278 }
279
280 size_t off = r->head % r->cap;
281 size_t first = r->cap - off;
282 if (first > len) first = len;
283 memcpy(r->buf + off, line, first);
284 memcpy(r->buf, line + first, len - first);
285 bool was_empty = rings_empty();
286 r->head += len;
287 if (was_empty) pthread_cond_signal(&g_work);
288 pthread_mutex_unlock(&g_lock);
289 return true;
290}
