32static char *csrf_generate_token(void) {
33 unsigned char bytes[CWIST_CSRF_TOKEN_BYTES];
34 int fd = open("/dev/urandom", O_RDONLY | O_CLOEXEC);
35 if (fd < 0) return NULL;
36 size_t got = 0;
37 while (got < sizeof(bytes)) {
38 ssize_t n = read(fd, bytes + got, sizeof(bytes) - got);
39 if (n > 0) {
40 got += (size_t)n;
41 continue;
42 }
43 if (n < 0 && errno == EINTR) continue;
44 close(fd);
45 return NULL;
46 }
47 close(fd);
48 static const char hex[] = "0123456789abcdef";
49 char *token = cwist_alloc(CWIST_CSRF_TOKEN_CHARS + 1);
50 if (!token) return NULL;
51 for (size_t i = 0; i < sizeof(bytes); ++i) {
52 token[2 * i] = hex[bytes[i] >> 4];
53 token[2 * i + 1] = hex[bytes[i] & 15U];
54 }
55 token[CWIST_CSRF_TOKEN_CHARS] = '\0';
56 return token;
57}

66static bool csrf_equal(const char *a, const char *b) {
67 if (!a || !b) return false;
68 size_t alen = strlen(a), blen = strlen(b);
69 size_t diff = alen ^ blen;
70 size_t limit = alen > blen ? alen : blen;
71 for (size_t i = 0; i < limit; ++i) {
72 unsigned char ac = i < alen ? (unsigned char)a[i] : 0;
73 unsigned char bc = i < blen ? (unsigned char)b[i] : 0;
74 diff |= (size_t)(ac ^ bc);
75 }
76 return diff == 0;
77}

84static int hex_value(unsigned char c) {
85 if (c >= '0' && c <= '9') return c - '0';
86 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
87 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
88 return -1;

102static char *csrf_form_value(const cwist_http_request *req) {
103 if (!req || !req->body || !req->body->data) return NULL;
104 const char *body = req->body->data;
105 size_t len = req->body->size;
106 size_t pos = 0;
107 while (pos < len) {
108 size_t field_start = pos;
109 while (pos < len && body[pos] != '&') ++pos;
110 size_t field_end = pos++;
111 const char *eq = memchr(body + field_start, '=', field_end - field_start);
112 size_t key_len = eq ? (size_t)(eq - (body + field_start)) : field_end - field_start;
113 if (!((key_len == 5 && memcmp(body + field_start, "_csrf", 5) == 0) ||
114 (key_len == 10 && memcmp(body + field_start, "csrf_token", 10) == 0)))
115 continue;
116 const char *value = eq ? eq + 1 : "";
117 size_t value_len = eq ? field_end - (size_t)(value - body) : 0;
118 char *decoded = cwist_alloc(value_len + 1);
119 if (!decoded) return NULL;
120 size_t out = 0;
121 for (size_t i = 0; i < value_len; ++i) {
122 unsigned char c = (unsigned char)value[i];
123 if (c == '+')
124 decoded[out++] = ' ';
125 else if (c == '%' && i + 2 < value_len) {
126 int hi = hex_value((unsigned char)value[i + 1]);
127 int lo = hex_value((unsigned char)value[i + 2]);
128 if (hi < 0 || lo < 0) {
129 cwist_free(decoded);
130 return NULL;
131 }
132 decoded[out++] = (char)((hi << 4) | lo);
133 i += 2;
134 } else
135 decoded[out++] = (char)c;
136 }
137 decoded[out] = '\0';
138 return decoded;
139 }
140 return NULL;
141}

175static void csrf_handler(cwist_http_request *req, cwist_http_response *res,
176 cwist_handler_func next) {
177 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
178 cwist_query_map *cookies = cwist_query_map_create();
179 const char *cookie = cookies ? cwist_cookie_get(cookies, CWIST_CSRF_COOKIE_NAME) : NULL;
180 if (cookies) {
181 cwist_cookie_parse(cookies, cookie_header);
182 cookie = cwist_cookie_get(cookies, CWIST_CSRF_COOKIE_NAME);
183 }
184 if (req->csrf_token) {
185 cwist_free(req->csrf_token);
186 req->csrf_token = NULL;
187 }
188 if (!cookie) {
189 char *token = csrf_generate_token();
190 if (!token) {
191 if (cookies) cwist_query_map_destroy(cookies);
192 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
193 return;
194 }
195 req->csrf_token = token;
196 csrf_issue_cookie(req, res, token);
197 if (cookies) cwist_query_map_destroy(cookies);
198 if (csrf_safe_method(req->method))
199 next(req, res);
200 else {
201 res->status_code = CWIST_HTTP_FORBIDDEN;
202 cwist_sstring_assign(res->body, "CSRF token missing");
203 }
204 return;
205 }
206 req->csrf_token = cwist_strdup(cookie);
207 if (csrf_safe_method(req->method)) {
208 cwist_query_map_destroy(cookies);
209 next(req, res);
210 return;
211 }
212 const char *header = cwist_http_header_get(req->headers, "X-CSRF-Token");
213 char *form = header ? NULL : csrf_form_value(req);
214 bool valid = csrf_equal(header ? header : form, cookie);
215 if (form) cwist_free(form);
216 cwist_query_map_destroy(cookies);
217 if (!valid) {
218 res->status_code = CWIST_HTTP_FORBIDDEN;
219 cwist_sstring_assign(res->body, "CSRF token invalid");
220 return;
221 }
222 next(req, res);
223}
