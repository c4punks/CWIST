36static char *csrf_generate_token(void) {
37 unsigned char bytes[CWIST_CSRF_TOKEN_BYTES];
38 int fd = open("/dev/urandom", O_RDONLY | O_CLOEXEC);
39 if (fd < 0) return NULL;
40 size_t got = 0;
41 while (got < sizeof(bytes)) {
42 ssize_t n = read(fd, bytes + got, sizeof(bytes) - got);
43 if (n > 0) {
44 got += (size_t)n;
45 continue;
46 }
47 if (n < 0 && errno == EINTR) continue;
48 close(fd);
49 return NULL;
50 }
51 close(fd);
52 static const char hex[] = "0123456789abcdef";
53 char *token = cwist_alloc(CWIST_CSRF_TOKEN_CHARS + 1);
54 if (!token) return NULL;
55 for (size_t i = 0; i < sizeof(bytes); ++i) {
56 token[2 * i] = hex[bytes[i] >> 4];
57 token[2 * i + 1] = hex[bytes[i] & 15U];
58 }
59 token[CWIST_CSRF_TOKEN_CHARS] = '\0';
60 return token;
61}

81static bool csrf_sign(const cwist_http_request *req, const char *nonce, char signature[65]) {
82 const char *secret = csrf_secret(req);
83 if (!secret) return false;
84 unsigned char digest[32];
85 unsigned int digest_len = 0;
86 if (!HMAC(EVP_sha256(), secret, (int)strlen(secret), (const unsigned char *)nonce,
87 CWIST_CSRF_TOKEN_CHARS, digest, &digest_len))
88 return false;
89 static const char hex[] = "0123456789abcdef";
90 for (size_t i = 0; i < sizeof(digest); i++) {
91 signature[2 * i] = hex[digest[i] >> 4];
92 signature[2 * i + 1] = hex[digest[i] & 15U];
93 }
94 signature[64] = '\0';
95 return true;
96}

105static bool csrf_equal(const char *a, const char *b) {
106 if (!a || !b) return false;
107 size_t alen = strlen(a), blen = strlen(b);
108 size_t diff = alen ^ blen;
109 size_t limit = alen > blen ? alen : blen;
110 for (size_t i = 0; i < limit; ++i) {
111 unsigned char ac = i < alen ? (unsigned char)a[i] : 0;
112 unsigned char bc = i < blen ? (unsigned char)b[i] : 0;
113 diff |= (size_t)(ac ^ bc);
114 }
115 return diff == 0;
116}

118static bool csrf_token_valid(const cwist_http_request *req, const char *token) {
119 if (!token || strlen(token) != CWIST_CSRF_TOKEN_CHARS * 2 + 1 ||
120 token[CWIST_CSRF_TOKEN_CHARS] != '.')
121 return false;
122 char signature[65];
123 return csrf_sign(req, token, signature) && csrf_equal(token + CWIST_CSRF_TOKEN_CHARS + 1,
124 signature);

132static int hex_value(unsigned char c) {
133 if (c >= '0' && c <= '9') return c - '0';
134 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
135 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
136 return -1;
137}

139static char *csrf_multipart_value(const cwist_http_request *req) {
140 const char *content_type = cwist_http_header_get(req->headers, "Content-Type");
141 char *boundary = cwist_multipart_extract_boundary(content_type);
142 if (!boundary) return NULL;
143 cwist_multipart_result *parts = cwist_multipart_parse(req->body->data, req->body->size, boundary);
144 cwist_free(boundary);
145 if (!parts) return NULL;
146 char *value = NULL;
147 for (cwist_multipart_field *field = parts->fields; field; field = field->next) {
148 if (!field->name || (strcmp(field->name, "_csrf") != 0 &&
149 strcmp(field->name, "csrf_token") != 0))
150 continue;
151 value = cwist_alloc(field->data_len + 1);
152 if (value) {
153 if (field->data_len) memcpy(value, field->data, field->data_len);
154 value[field->data_len] = '\0';
155 }
156 break;
157 }
158 cwist_multipart_result_destroy(parts);
159 return value;
160}

173static char *csrf_form_value(const cwist_http_request *req) {
174 if (!req || !req->body || !req->body->data) return NULL;
175 const char *content_type = cwist_http_header_get(req->headers, "Content-Type");
176 if (content_type && strncasecmp(content_type, "multipart/form-data", 19) == 0)
177 return csrf_multipart_value(req);
178 if (content_type &&
179 strncasecmp(content_type, "application/x-www-form-urlencoded", 33) != 0)
180 return NULL;
181 const char *body = req->body->data;
182 size_t len = req->body->size;
183 size_t pos = 0;
184 while (pos < len) {
185 size_t field_start = pos;
186 while (pos < len && body[pos] != '&') ++pos;
187 size_t field_end = pos++;
188 const char *eq = memchr(body + field_start, '=', field_end - field_start);
189 size_t key_len = eq ? (size_t)(eq - (body + field_start)) : field_end - field_start;
190 if (!((key_len == 5 && memcmp(body + field_start, "_csrf", 5) == 0) ||
191 (key_len == 10 && memcmp(body + field_start, "csrf_token", 10) == 0)))
192 continue;
193 const char *value = eq ? eq + 1 : "";
194 size_t value_len = eq ? field_end - (size_t)(value - body) : 0;
195 char *decoded = cwist_alloc(value_len + 1);
196 if (!decoded) return NULL;
197 size_t out = 0;
198 for (size_t i = 0; i < value_len; ++i) {
199 unsigned char c = (unsigned char)value[i];
200 if (c == '+')
201 decoded[out++] = ' ';
202 else if (c == '%' && i + 2 < value_len) {
203 int hi = hex_value((unsigned char)value[i + 1]);
204 int lo = hex_value((unsigned char)value[i + 2]);
205 if (hi < 0 || lo < 0) {
206 cwist_free(decoded);
207 return NULL;
208 }
209 decoded[out++] = (char)((hi << 4) | lo);
210 i += 2;
211 } else
212 decoded[out++] = (char)c;
213 }
214 decoded[out] = '\0';
215 return decoded;
216 }
217 return NULL;
218}

253static void csrf_handler(cwist_http_request *req, cwist_http_response *res,
254 cwist_handler_func next) {
255 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
256 const char *cookie_name = req && req->app && req->app->use_ssl ? "__Host-csrf" : CWIST_CSRF_COOKIE_NAME;
257 cwist_query_map *cookies = cwist_query_map_create();
258 const char *cookie = cookies ? cwist_cookie_get(cookies, cookie_name) : NULL;
259 if (cookies) {
260 cwist_cookie_parse(cookies, cookie_header);
261 cookie = cwist_cookie_get(cookies, cookie_name);
262 }
263 if (req->csrf_token) {
264 cwist_free(req->csrf_token);
265 req->csrf_token = NULL;
266 }
267 if (!cookie || !csrf_token_valid(req, cookie)) {
268 char *nonce = csrf_generate_token();
269 char signature[65];
270 if (!nonce || !csrf_sign(req, nonce, signature)) {
271 cwist_free(nonce);
272 if (cookies) cwist_query_map_destroy(cookies);
273 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
274 return;
275 }
276 char token[CWIST_CSRF_TOKEN_CHARS * 2 + 2];
277 snprintf(token, sizeof(token), "%s.%s", nonce, signature);
278 cwist_free(nonce);
279 req->csrf_token = cwist_strdup(token);
280 csrf_issue_cookie(req, res, token);
281 if (cookies) cwist_query_map_destroy(cookies);
282 if (csrf_safe_method(req->method))
283 next(req, res);
284 else {
285 res->status_code = CWIST_HTTP_FORBIDDEN;
286 cwist_sstring_assign(res->body, "CSRF token missing");
287 }
288 return;
289 }
290 req->csrf_token = cwist_strdup(cookie);
291 if (csrf_safe_method(req->method)) {
292 cwist_query_map_destroy(cookies);
293 next(req, res);
294 return;
295 }
296 const char *header = cwist_http_header_get(req->headers, "X-CSRF-Token");
297 char *form = header ? NULL : csrf_form_value(req);
298 bool valid = csrf_equal(header ? header : form, cookie);
299 if (form) cwist_free(form);
300 cwist_query_map_destroy(cookies);
301 if (!valid) {
302 res->status_code = CWIST_HTTP_FORBIDDEN;
303 cwist_sstring_assign(res->body, "CSRF token invalid");
304 return;
305 }
306 next(req, res);
307}
