12static inline void cwist_http_async_disown_string(cwist_sstring *str) {
13 if (!str) return;
14 if (!str->borrows_buffer) cwist_gc_scope_disown(str->data);
15 if (str->owns_storage) cwist_gc_scope_disown(str);
16}

18static inline void cwist_http_async_disown_headers(cwist_http_header_node *node) {
19 for (; node; node = node->next) {
20 cwist_http_async_disown_string(node->key);
21 cwist_http_async_disown_string(node->value);
22 if (!node->arena_owned) cwist_gc_scope_disown(node);
23 }

26static inline void cwist_http_async_disown_map(cwist_query_map *map) {
27 /* Arena maps have no heap fallback and their destructor is a no-op. */
28 if (!map || map->arena) return;
29 if (map->buckets) {
30 for (size_t i = 0; i < map->size; ++i) {
31 for (cwist_query_bucket *node = map->buckets[i]; node; node = node->next) {
32 cwist_gc_scope_disown(node->key);
33 cwist_gc_scope_disown(node->value);
34 cwist_gc_scope_disown(node);
35 }
36 }
37 cwist_gc_scope_disown(map->buckets);
38 }
39 cwist_gc_scope_disown(map);
40}

61static inline void cwist_http_async_disown_response(cwist_http_response *res) {
62 if (!res || !cwist_full_gc_enabled()) return;
63 cwist_http_async_disown_string(res->version);
64 cwist_http_async_disown_string(res->status_text);
65 cwist_http_async_disown_string(res->body);
66 cwist_http_async_disown_headers(res->headers);
67 cwist_gc_scope_disown(res->alt_svc);
68 /* Cleanup, not GC, owns a managed payload. Do not inspect the opaque
69 * context or change the callback; caller transfers its dependencies. */
70 if (res->is_ptr_body && res->ptr_body_cleanup) cwist_gc_scope_disown((void *)res->ptr_body);
71 if (!res->arena || !cwist_arena_owns(res->arena, res)) cwist_gc_scope_disown(res);
72}
