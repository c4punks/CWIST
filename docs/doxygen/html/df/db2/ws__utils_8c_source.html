180static void SHA1Update(SHA1_CTX *context, const uint8_t *data, uint32_t len) {
181 uint32_t i, j;
182 j = context->count[0];
183 if ((context->count[0] += len << 3) < j) context->count[1]++;
184 context->count[1] += (len >> 29);
185 j = (j >> 3) & 63;
186 if ((j + len) > 63) {
187 memcpy(&context->buffer[j], data, (i = 64 - j));
188 /* GCC's -Wstringop-overread fires on this classic buffered-SHA1
189 * shape: the block buffer is provably full here (j + 64 - j bytes),
190 * and the loop below only runs while i + 63 < len. */
191#pragma GCC diagnostic push
192#pragma GCC diagnostic ignored "-Wstringop-overread"
193 SHA1Transform(context->state, context->buffer);
194 for (; i + 63 < len; i += 64) {
195 SHA1Transform(context->state, &data[i]);
196 }
197#pragma GCC diagnostic pop
198 j = 0;
199 } else
200 i = 0;
201 memcpy(&context->buffer[j], &data[i], len - i);
202}
