74static void append_json_escaped_str(cwist_sstring *buf, const char *str) {
75 if (!str) return;
76 cwist_sstring_append(buf, "\"");
77 const char *start = str;
78 const char *p = str;
79 while (*p) {
80 const char *esc = NULL;
81 char ucode[8];
82 switch (*p) {
83 case '"': esc = "\\\""; break;
84 case '\\': esc = "\\\\"; break;
85 case '\b': esc = "\\b"; break;
86 case '\f': esc = "\\f"; break;
87 case '\n': esc = "\\n"; break;
88 case '\r': esc = "\\r"; break;
89 case '\t': esc = "\\t"; break;
90 default:
91 if ((unsigned char)*p < 0x20) {
92 snprintf(ucode, sizeof(ucode), "\\u%04x", (unsigned char)*p);
93 esc = ucode;
94 }
95 break;
96 }
97 if (esc) {
98 if (p > start) {
99 char tmp[4096];
100 size_t seg_len = (size_t)(p - start);
101 while (seg_len > 0) {
102 size_t chunk = seg_len < sizeof(tmp) - 1 ? seg_len : sizeof(tmp) - 1;
103 memcpy(tmp, start, chunk);
104 tmp[chunk] = '\0';
105 cwist_sstring_append(buf, tmp);
106 start += chunk;
107 seg_len -= chunk;
108 }
109 }
110 cwist_sstring_append(buf, esc);
111 start = p + 1;
112 }
113 p++;
114 }
115 if (p > start) {
116 cwist_sstring_append(buf, start);
117 }
118 cwist_sstring_append(buf, "\"");
119}
