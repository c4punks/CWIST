96static void append_json_escaped_str(cwist_json_builder *b, const char *str) {
97 if (!str) return;
98 jb_append(b, "\"");
99 const char *start = str;
100 const char *p = str;
101 while (*p) {
102 const char *esc = NULL;
103 char ucode[8];
104 switch (*p) {
105 case '"': esc = "\\\""; break;
106 case '\\': esc = "\\\\"; break;
107 case '\b': esc = "\\b"; break;
108 case '\f': esc = "\\f"; break;
109 case '\n': esc = "\\n"; break;
110 case '\r': esc = "\\r"; break;
111 case '\t': esc = "\\t"; break;
112 default:
113 if ((unsigned char)*p < 0x20) {
114 snprintf(ucode, sizeof(ucode), "\\u%04x", (unsigned char)*p);
115 esc = ucode;
116 }
117 break;
118 }
119 if (esc) {
120 if (p > start) {
121 char tmp[4096];
122 size_t seg_len = (size_t)(p - start);
123 while (seg_len > 0) {
124 size_t chunk = seg_len < sizeof(tmp) - 1 ? seg_len : sizeof(tmp) - 1;
125 memcpy(tmp, start, chunk);
126 tmp[chunk] = '\0';
127 jb_append(b, tmp);
128 start += chunk;
129 seg_len -= chunk;
130 }
131 }
132 jb_append(b, esc);
133 start = p + 1;
134 }
135 p++;
136 }
137 if (p > start) {
138 jb_append(b, start);
139 }
140 jb_append(b, "\"");
141}
