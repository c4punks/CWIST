70static int recv_all(int fd, void *buf, size_t n)
71{
72 size_t total = 0;
73 char *p = (char *)buf;
74 while (total < n) {
75 ssize_t r = recv(fd, p + total, n - total, MSG_WAITALL);
76 if (r <= 0) return -1;
77 total += (size_t)r;
78 }
79 return 0;
80}

86static int send_all(int fd, const void *buf, size_t n)
87{
88 size_t total = 0;
89 const char *p = (const char *)buf;
90 while (total < n) {
91 ssize_t w = send(fd, p + total, n - total, 0);
92 if (w <= 0) return -1;
93 total += (size_t)w;
94 }
95 return 0;
96}

101static void free_worker_ctx(cwist_orm_worker_ctx_t *ctx)
102{
103 /* The context is allocated by the caller's thread and freed by the
104 * detached worker, so it stays on the raw allocator: a cwist_alloc
105 * scope sweep on the caller side must not reclaim it. */
106 if (!ctx) return;
107 free(ctx->db_path);

125static int socket_query_callback(void *data, int argc, char **argv,
126 char **azColName)
127{
128 query_accumulator_t *acc = (query_accumulator_t *)data;
129 cJSON *row = cJSON_CreateObject();
130 if (!row) return 1; /* abort */
131
132 for (int i = 0; i < argc; i++) {
133 if (argv[i]) {
134 /* Attempt numeric conversion for cleaner JSON. */
135 char *endptr = NULL;
136 errno = 0;
137 long long ll = strtoll(argv[i], &endptr, 10);
138 if (endptr && *endptr == '\0' && errno == 0) {
139 cJSON_AddNumberToObject(row, azColName[i], (double)ll);
140 } else {
141 double d = strtod(argv[i], &endptr);
142 if (endptr && *endptr == '\0' && errno == 0) {
143 cJSON_AddNumberToObject(row, azColName[i], d);
144 } else {
145 cJSON_AddStringToObject(row, azColName[i], argv[i]);
146 }
147 }
148 } else {
149 cJSON_AddNullToObject(row, azColName[i]);
150 }
151 }
152
153 cJSON_AddItemToArray(acc->rows, row);
154 return 0;
155}

167static void *cwist_orm_socket_worker(void *arg)
168{
169 cwist_orm_worker_ctx_t *ctx = (cwist_orm_worker_ctx_t *)arg;
170 if (!ctx) return NULL;
171
172 sqlite3 *db = NULL;
173 int rc = sqlite3_open(ctx->db_path, &db);
174 if (rc != SQLITE_OK) {
175 fprintf(stderr, "[ORM-SOCKET] sqlite3_open failed: %s\n",
176 sqlite3_errmsg(db));
177 sqlite3_close(db);
178 close(ctx->fd);
179 free_worker_ctx(ctx);
180 return NULL;
181 }
182
183 /* Speed up concurrent access semantics for single-writer scenarios. */
184 sqlite3_exec(db, "PRAGMA journal_mode=WAL;", NULL, NULL, NULL);
185
186 for (;;) {
187 /* ---- read request length (4 bytes, big-endian) ---- */
188 uint32_t net_len = 0;
189 if (recv_all(ctx->fd, &net_len, sizeof(net_len)) != 0) break;
190 uint32_t sql_len = ntohl(net_len);
191 if (sql_len == 0 || sql_len > 16 * 1024 * 1024) break; /* sanity */
192
193 /* ---- read SQL text ---- */
194 char *sql = (char *)cwist_alloc(sql_len + 1);
195 if (!sql) break;
196 if (recv_all(ctx->fd, sql, sql_len) != 0) {
197 cwist_free(sql);
198 break;
199 }
200 sql[sql_len] = '\0';
201
202 /* ---- execute ----
203 * sql comes from the in-process peer, not from an external client;
204 * see the trust boundary note at the top of this file. */
205 query_accumulator_t acc;
206 acc.rows = cJSON_CreateArray();
207 char *errmsg = NULL;
208
209 rc = sqlite3_exec(db, sql, socket_query_callback, &acc, &errmsg);
210 cwist_free(sql);
211
212 /* ---- build response payload ---- */
213 cJSON *resp = cJSON_CreateObject();
214 if (!resp) {
215 cJSON_Delete(acc.rows);
216 if (errmsg) sqlite3_free(errmsg);
217 break;
218 }
219
220 cJSON_AddNumberToObject(resp, "status", (double)rc);
221 if (errmsg) {
222 cJSON_AddStringToObject(resp, "error", errmsg);
223 sqlite3_free(errmsg);
224 } else {
225 cJSON_AddNullToObject(resp, "error");
226 }
227 cJSON_AddItemToObject(resp, "rows", acc.rows);
228
229 char *payload = cJSON_PrintUnformatted(resp);
230 cJSON_Delete(resp);
231 if (!payload) break;
232
233 uint32_t payload_len = (uint32_t)strlen(payload);
234 if (payload_len > UINT32_MAX - 1) {
235 cJSON_free(payload);
236 break;
237 }
238
239 /* ---- send framed response ---- */
240 uint32_t net_payload = htonl(payload_len);
241 if (send_all(ctx->fd, &net_payload, sizeof(net_payload)) != 0) {
242 cJSON_free(payload);
243 break;
244 }
245 if (send_all(ctx->fd, payload, payload_len) != 0) {
246 cJSON_free(payload);
247 break;
248 }
249 cJSON_free(payload);
250 }
251
252 sqlite3_close(db);
253 close(ctx->fd);
254 free_worker_ctx(ctx);
255 return NULL;
256}

275int cwist_orm_socketpair_cloexec(int domain, int type, int protocol, int sv[2]) {
276 int res = -1;
277#if defined(SOCK_CLOEXEC) && SOCK_CLOEXEC != 0
278 res = socketpair(domain, type | SOCK_CLOEXEC, protocol, sv);
279#endif
280 if (res != 0) {
281 res = socketpair(domain, type & ~SOCK_CLOEXEC, protocol, sv);
282 if (res == 0) {
283 set_cloexec(sv[0]);
284 set_cloexec(sv[1]);
285 }
286 } else {
287 set_cloexec(sv[0]);
288 set_cloexec(sv[1]);
289 }
290 return res;
291}

294int cwist_db_transfer_sqlite_to_socket(const char *db_path)
295{
296 if (!db_path) {
297 errno = EINVAL;
298 return -1;
299 }
300
301 int sv[2];
302 /* SOCK_CLOEXEC is not available on every BSD-family system (macOS lacks
303 * it); the helper falls back to fcntl(FD_CLOEXEC) there. */
304 if (cwist_orm_socketpair_cloexec(AF_UNIX, SOCK_STREAM, 0, sv) != 0) {
305 return -1;
306 }
307
308 cwist_orm_worker_ctx_t *ctx =
309 (cwist_orm_worker_ctx_t *)calloc(1, sizeof(*ctx));
310 if (!ctx) {
311 close(sv[0]);
312 close(sv[1]);
313 return -1;
314 }
315
316 ctx->fd = sv[1];
317 ctx->db_path = strdup(db_path);
318 if (!ctx->db_path) {
319 free(ctx);
320 close(sv[0]);
321 close(sv[1]);
322 return -1;
323 }
324
325 pthread_t tid;
326 pthread_attr_t attr;
327 pthread_attr_init(&attr);
328 pthread_attr_setdetachstate(&attr, PTHREAD_CREATE_DETACHED);
329
330 int rc = pthread_create(&tid, &attr, cwist_orm_socket_worker, ctx);
331 pthread_attr_destroy(&attr);
332
333 if (rc != 0) {
334 free_worker_ctx(ctx);
335 close(sv[0]);
336 close(sv[1]);
337 errno = rc;
338 return -1;
339 }
340
341 return sv[0]; /* caller owns this end */
342}
