32 size_t *sent_out) {

42

84 int flags, size_t *total_sent) {
85 if (!iov || iovcnt <= 0) {

87 return CWIST_WRITE_DONE;
88 }

91#if defined(MSG_NOSIGNAL)
92 send_flags |= MSG_NOSIGNAL;
93#endif
94
95#ifdef CWIST_WASI_SOCKETS
96 /* wasi:sockets (0.2) has no sendmsg(2): coalesce and send once.
97 * This is the speculative path - the blocking drain in
98 * cwist_http_sendmsg_all() re-drives whatever is left. */
99 size_t expected_wasi = 0;
100 for (int i = 0; i < iovcnt; ++i) expected_wasi += iov[i].iov_len;
101 char *wasi_buf = cwist_alloc(expected_wasi ? expected_wasi : 1);
102 if (!wasi_buf) {
103 if (total_sent) *total_sent = 0;
104 return CWIST_WRITE_ERR;
105 }
106 size_t wasi_off = 0;
107 for (int i = 0; i < iovcnt; ++i) {
108 memcpy(wasi_buf + wasi_off, iov[i].iov_base, iov[i].iov_len);
109 wasi_off += iov[i].iov_len;
110 }
111 ssize_t n;
112 do {
113 n = send(fd, wasi_buf, expected_wasi, send_flags);
114 } while (n < 0 && errno == EINTR);
115 cwist_free(wasi_buf);
116#else
117 struct msghdr msg = {.msg_iov = iov, .msg_iovlen = (size_t)iovcnt};
118
119 ssize_t n;
120 do {
121 n = sendmsg(fd, &msg, send_flags);
122 } while (n < 0 && errno == EINTR);
123#endif
124
125 if (n >= 0) {
126 if (total_sent) *total_sent = (size_t)n;
127
128 /* Calculate total expected payload length across all iovec items */
129 size_t expected = 0;
130 for (int i = 0; i < iovcnt; ++i) {
131 expected += iov[i].iov_len;
132 }
133
134 if ((size_t)n >= expected) {
135 return CWIST_WRITE_DONE;
136 }
137 return CWIST_WRITE_PENDING;
138 }
139
140 if (errno == EAGAIN || errno == EWOULDBLOCK) {
141 if (total_sent) *total_sent = 0;
142 return CWIST_WRITE_PENDING;
143 }

145 return CWIST_WRITE_ERR;

161uint32_t cwist_sched_p2c_select_worker(const _Atomic uint32_t *worker_loads, uint32_t num_workers) {
162 if (!worker_loads || num_workers <= 1) return 0;
163
164 static _Thread_local uint32_t seed = 0x9e3779b9;
165 if (__builtin_expect(seed == 0, 0)) {
166 seed = (uint32_t)(uintptr_t)&seed;
167 if (seed == 0) seed = 0x9e3779b9;
168 }
169
170 /* Fast xorshift32 PRNG */
171 seed ^= seed << 13;
172 seed ^= seed >> 17;
173 seed ^= seed << 5;
174 uint32_t w1 = seed % num_workers;
175
176 seed ^= seed << 13;
177 seed ^= seed >> 17;
178 seed ^= seed << 5;
179 uint32_t w2 = seed % num_workers;
180
181 if (w1 == w2) {
182 w2 = (w1 + 1) % num_workers;
183 }
184
185 uint32_t load1 = atomic_load_explicit(&worker_loads[w1], memory_order_relaxed);
186 uint32_t load2 = atomic_load_explicit(&worker_loads[w2], memory_order_relaxed);
187
188 return (load1 <= load2) ? w1 : w2;
189}
