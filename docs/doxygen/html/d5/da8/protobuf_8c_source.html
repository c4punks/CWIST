42int cwist_pb_writer_reserve(cwist_pb_writer *w, size_t extra) {
43 if (!w) return -1;
44 if (extra > SIZE_MAX - w->len) return -1;
45 size_t need = w->len + extra;
46 if (need <= w->cap) return 0;
47
48 size_t cap = w->cap ? w->cap : 64;
49 while (cap < need) {
50 if (cap > SIZE_MAX / 2) {
51 cap = need;
52 break;
53 }
54 cap *= 2;
55 }
56 uint8_t *next = (uint8_t *)cwist_realloc(w->data, cap);
57 if (!next) return -1;
58 w->data = next;
59 w->cap = cap;
60 return 0;
61}

69int cwist_pb_write_varint(cwist_pb_writer *w, uint64_t value) {

258int cwist_pb_read_varint(cwist_pb_reader *r, uint64_t *out) {
259 if (!r || !out) return -1;
260 uint64_t value = 0;
261 unsigned shift = 0;
262 while (r->pos < r->len && shift <= 63) {
263 uint8_t byte = r->data[r->pos++];
264 value |= ((uint64_t)(byte & 0x7f)) << shift;
265 if ((byte & 0x80) == 0) {
266 *out = value;
267 return 0;
268 }
269 shift += 7;
270 }
271 return -1;
272}

284int cwist_pb_read_field(cwist_pb_reader *r, cwist_pb_field *field) {
285 if (!r || !field || r->pos >= r->len) return 0;
286 memset(field, 0, sizeof(*field));
287
288 uint64_t key = 0;
289 if (cwist_pb_read_varint(r, &key) != 0) return -1;
290 field->number = (uint32_t)(key >> 3);
291 field->wire_type = (cwist_pb_wire_type_t)(key & 0x07);
292 if (field->number == 0) return -1;
293
294 switch (field->wire_type) {
295 case CWIST_PB_VARINT:
296 if (cwist_pb_read_varint(r, &field->varint) != 0) return -1;
297 return 1;
298 case CWIST_PB_LEN: {
299 uint64_t len = 0;
300 if (cwist_pb_read_varint(r, &len) != 0) return -1;
301 if (len > SIZE_MAX || (size_t)len > r->len - r->pos) return -1;
302 field->bytes = r->data + r->pos;
303 field->len = (size_t)len;
304 r->pos += field->len;
305 return 1;
306 }
307 case CWIST_PB_32BIT:
308 if (r->len - r->pos < 4) return -1;
309 field->bytes = r->data + r->pos;
310 field->len = 4;
311 r->pos += 4;
312 return 1;
313 case CWIST_PB_64BIT:
314 if (r->len - r->pos < 8) return -1;
315 field->bytes = r->data + r->pos;
316 field->len = 8;
317 r->pos += 8;
318 return 1;
319 default: return -1;
320 }
321}
