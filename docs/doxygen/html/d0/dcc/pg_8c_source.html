43cwist_pg_t *cwist_pg_connect(const char *conninfo) {
44 if (!conninfo || conninfo[0] == '\0') return NULL;
45
46 cwist_pg_t *pg = (cwist_pg_t *)cwist_alloc(sizeof(*pg));
47 if (!pg) return NULL;
48 memset(pg, 0, sizeof(*pg));
49
50 pg->conn = PQconnectdb(conninfo);
51 if (!pg->conn) {
52 cwist_free(pg);
53 return NULL;
54 }
55
56 if (PQstatus(pg->conn) != CONNECTION_OK) {
57 set_last_error(pg, PQerrorMessage(pg->conn));
58 PQfinish(pg->conn);
59 cwist_free(pg);
60 return NULL;
61 }
62
63 set_last_error(pg, NULL);
64 return pg;
65}

81cwist_error_t cwist_pg_exec(cwist_pg_t *pg, const char *sql) {
82 cwist_error_t err = make_error(CWIST_ERR_INT16);
83 if (!pg || !pg->conn || !sql) {
84 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
85 return err;
86 }
87
88 PGresult *res = PQexec(pg->conn, sql);
89 if (!res) {
90 set_last_error(pg, PQerrorMessage(pg->conn));
91 err.error.err_i16 = CWIST_ERROR_IO;
92 return err;
93 }
94
95 ExecStatusType status = PQresultStatus(res);
96 if (status != PGRES_COMMAND_OK && status != PGRES_TUPLES_OK) {
97 set_last_error(pg, PQresultErrorMessage(res));
98 PQclear(res);
99 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
100 return err;
101 }
102
103 PQclear(res);
104 set_last_error(pg, NULL);
105 err.error.err_i16 = CWIST_SUCCESS;
106 return err;
107}

143 if (PQgetisnull(res, i, j)) {

145 } else {
146 const char *val = PQgetvalue(res, i, j);
147 cJSON_AddStringToObject(row, field_name, val);
148 }
149 }
150 cJSON_AddItemToArray(rows, row);
151 }
152 *out_rows = rows;
153 }

155 PQclear(res);
