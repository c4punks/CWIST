52static int pg_wait_until(cwist_pg_t *pg, short events, long long deadline) {
53 int fd = PQsocket(pg->conn);
54 if (fd < 0) return -1;
55 for (;;) {
56 long long remaining = deadline - pg_now_ms();
57 if (remaining <= 0) {
58 set_last_error(pg, "database operation timed out");
59 return 0;
60 }
61 struct pollfd pfd = {.fd = fd, .events = events, .revents = 0};
62 int rc = poll(&pfd, 1, remaining > 2147483647 ? 2147483647 : (int)remaining);
63 if (rc > 0) return 1;
64 if (rc == 0) {
65 set_last_error(pg, "database operation timed out");
66 return 0;
67 }
68 if (errno != EINTR) {
69 set_last_error(pg, strerror(errno));
70 return -1;
71 }
72 }
73}

91static PGresult *pg_results(cwist_pg_t *pg, const char *sql, int n_params,
92 const char *const *param_values) {
93 long long deadline = pg_now_ms() + 30000;
94 int sent = n_params >= 0 ? PQsendQueryParams(pg->conn, sql, n_params, NULL,
95 param_values, NULL, NULL, 0)
96 : PQsendQuery(pg->conn, sql);
97 if (sent != 1) {
98 set_last_error(pg, PQerrorMessage(pg->conn));
99 return NULL;
100 }
101 if (pg_flush(pg, deadline) < 0) return NULL;
102
103 PGresult *last = NULL;
104 for (;;) {
105 while (PQisBusy(pg->conn)) {
106 if (!PQconsumeInput(pg->conn)) {
107 set_last_error(pg, PQerrorMessage(pg->conn));
108 goto fail;
109 }
110 if (pg_wait_until(pg, POLLIN, deadline) != 1) goto fail;
111 }
112 PGresult *next = PQgetResult(pg->conn);
113 if (!next) return last;
114 PQclear(last);
115 last = next;
116 }
117
118fail:
119 PQclear(last);
120 return NULL;
121}

123cwist_pg_t *cwist_pg_connect(const char *conninfo) {
124 if (!conninfo || conninfo[0] == '\0') return NULL;
125
126 cwist_pg_t *pg = (cwist_pg_t *)cwist_alloc(sizeof(*pg));
127 if (!pg) return NULL;
128 memset(pg, 0, sizeof(*pg));
129
130 pg->conn = PQconnectStart(conninfo);
131 if (!pg->conn) {
132 cwist_free(pg);
133 return NULL;
134 }
135 long long deadline = pg_now_ms() + 5000;
136 for (;;) {
137 PostgresPollingStatusType status = PQconnectPoll(pg->conn);
138 if (status == PGRES_POLLING_OK) break;
139 if (status == PGRES_POLLING_FAILED) {
140 set_last_error(pg, PQerrorMessage(pg->conn));
141 pg_abort_connection(pg);
142 cwist_free(pg);
143 return NULL;
144 }
145 short events = status == PGRES_POLLING_READING ? POLLIN : POLLOUT;
146 if (pg_wait_until(pg, events, deadline) != 1) {
147 pg_abort_connection(pg);
148 cwist_free(pg);
149 return NULL;
150 }
151 }
152 if (PQsetnonblocking(pg->conn, 1) != 0) {
153 set_last_error(pg, PQerrorMessage(pg->conn));
154 pg_abort_connection(pg);
155 cwist_free(pg);
156 return NULL;
157 }
158
159 set_last_error(pg, NULL);
160 return pg;
161}

177cwist_error_t cwist_pg_exec(cwist_pg_t *pg, const char *sql) {
178 cwist_error_t err = make_error(CWIST_ERR_INT16);
179 if (!pg || !pg->conn || !sql) {
180 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
181 return err;
182 }
183
184 PGresult *res = pg_results(pg, sql, -1, NULL);
185 if (!res) {
186 pg_abort_connection(pg);
187 err.error.err_i16 = CWIST_ERROR_IO;
188 return err;
189 }
190
191 ExecStatusType status = PQresultStatus(res);
192 if (status != PGRES_COMMAND_OK && status != PGRES_TUPLES_OK) {
193 set_last_error(pg, PQresultErrorMessage(res));
194 PQclear(res);
195 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
196 return err;
197 }
198
199 PQclear(res);
200 set_last_error(pg, NULL);
201 err.error.err_i16 = CWIST_SUCCESS;
202 return err;
203}

205cwist_error_t cwist_pg_exec_params(cwist_pg_t *pg, const char *sql, int n_params,
206 const char *const *param_values, cJSON **out_rows) {
207 cwist_error_t err = make_error(CWIST_ERR_INT16);
208 cJSON *rows = NULL;
209 if (out_rows) *out_rows = NULL;
210 if (!pg || !pg->conn || !sql || n_params < 0 || (n_params > 0 && !param_values)) {
211 err.error.err_i16 = CWIST_ERROR_INVALID_PARAM;
212 return err;
213 }
214
215 PGresult *res = pg_results(pg, sql, n_params, param_values);
216 if (!res) {
217 pg_abort_connection(pg);
218 err.error.err_i16 = CWIST_ERROR_IO;
219 return err;
220 }
221
222 ExecStatusType status = PQresultStatus(res);
223 if (status != PGRES_COMMAND_OK && status != PGRES_TUPLES_OK) {
224 set_last_error(pg, PQresultErrorMessage(res));
225 PQclear(res);
226 err.error.err_i16 = CWIST_ERROR_PROTOCOL;
227 return err;
228 }
229
230 if (out_rows) {
231 rows = cJSON_CreateArray();
232 if (!rows) goto oom;
233 int n_rows = PQntuples(res);
234 int n_fields = PQnfields(res);
235
236 for (int i = 0; i < n_rows; i++) {
237 cJSON *row = cJSON_CreateObject();
238 if (!row) goto oom;
239 for (int j = 0; j < n_fields; j++) {
240 const char *field_name = PQfname(res, j);
241 cJSON *cell = PQgetisnull(res, i, j)
242 ? cJSON_AddNullToObject(row, field_name)
243 : cJSON_AddStringToObject(row, field_name, PQgetvalue(res, i, j));
244 if (!cell) {
245 cJSON_Delete(row);
246 goto oom;
247 }
248 }
249 if (!cJSON_AddItemToArray(rows, row)) {
250 cJSON_Delete(row);
251 goto oom;
252 }
253 }
254 *out_rows = rows;
255 }
256
257 PQclear(res);
258 set_last_error(pg, NULL);
259 err.error.err_i16 = CWIST_SUCCESS;
260 return err;
261
262oom:
263 cJSON_Delete(rows);
264 PQclear(res);
265 set_last_error(pg, "out of memory");
266 err.error.err_i16 = CWIST_ERROR_NOMEM;
267 return err;
268}
