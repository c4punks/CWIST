23/* Report an exchange whose public deferred/async fields disagree with what
24 * cwist_async_defer() recorded. Prints once per process; with
25 * CWIST_ASYNC_STRICT=1 it aborts so the misuse leaves a core. */
26static void cwist_async_report_misuse(const char *what) {
27 const char *strict = getenv("CWIST_ASYNC_STRICT");
28 bool abort_now = strict && strict[0] == '1';
29 if (!atomic_exchange(&g_misuse_reported, true) || abort_now) {
30 fprintf(stderr,
31 "cwist: async misuse: %s; producers must not write res->deferred or "
32 "res->async (see cwist_async_wait_dispatched())\n",
33 what);
34 }
35 if (abort_now) abort();
36}
