105static bool seq_auth_chunk_valid(const uint8_t *data, size_t len, cwist_seq_chunk_t *out) {
106 if (!data || len < CWIST_SEQ_HEADER_SIZE || !out) return false;
107 out->seq = read_u16(data);
108 out->total = read_u16(data + 2);
109 out->payload_len = read_u16(data + 4);
110 out->chunk_size = read_u16(data + 6);
111 if (!out->seq || !out->total || out->total > CWIST_SEQ_MAX_CHUNKS || out->seq > out->total ||
112 !out->payload_len || !out->chunk_size || out->payload_len > out->chunk_size ||
113 (out->seq != out->total && out->payload_len != out->chunk_size) ||
114 len < (size_t)CWIST_SEQ_AUTH_HEADER_SIZE + out->payload_len)
115 return false;
116 out->payload = data + CWIST_SEQ_AUTH_HEADER_SIZE;
117 return true;
118}

138static bool auth_tag(const cwist_seq_auth_context_t *ctx, const uint8_t *header_without_tag,
139 size_t header_len, const uint8_t *payload, size_t payload_len,
140 uint8_t tag[SEQ_AUTH_TAG_CAP]) {
141#ifdef SEQ_AUTH_WASM_CRYPTO
142 /* Streaming HMAC-SHA256 (RFC 2104) over session_id || header || payload.
143 * The 32-byte key never exceeds the 64-byte block size, so no key
144 * pre-hashing is needed. */
145 uint8_t k[64] = {0};
146 memcpy(k, ctx->key, sizeof(ctx->key));
147 uint8_t ipad[64], opad[64];
148 for (size_t i = 0; i < 64; i++) {
149 ipad[i] = (uint8_t)(k[i] ^ 0x36);
150 opad[i] = (uint8_t)(k[i] ^ 0x5c);
151 }
152 cwist_sha256_ctx c;
153 uint8_t inner[CWIST_SHA256_DIGEST_LEN];
154 cwist_sha256_init(&c);
155 cwist_sha256_update(&c, ipad, sizeof(ipad));
156 cwist_sha256_update(&c, ctx->session_id, sizeof(ctx->session_id));
157 cwist_sha256_update(&c, header_without_tag, header_len);
158 cwist_sha256_update(&c, payload, payload_len);
159 cwist_sha256_final(&c, inner);
160 cwist_sha256_init(&c);
161 cwist_sha256_update(&c, opad, sizeof(opad));
162 cwist_sha256_update(&c, inner, sizeof(inner));
163 cwist_sha256_final(&c, tag);
164 SEQ_AUTH_CLEANSE(ipad, sizeof(ipad));
165 SEQ_AUTH_CLEANSE(opad, sizeof(opad));
166 SEQ_AUTH_CLEANSE(inner, sizeof(inner));
167 return true;
168#else
169 HMAC_CTX *h = HMAC_CTX_new();
170 unsigned int tag_len = 0;
171 bool ok = h && HMAC_Init_ex(h, ctx->key, sizeof(ctx->key), EVP_sha256(), NULL) == 1 &&
172 HMAC_Update(h, ctx->session_id, sizeof(ctx->session_id)) == 1 &&
173 HMAC_Update(h, header_without_tag, header_len) == 1 &&
174 HMAC_Update(h, payload, payload_len) == 1 && HMAC_Final(h, tag, &tag_len) == 1 &&
175 tag_len >= CWIST_SEQ_AUTH_TAG_SIZE;
176 HMAC_CTX_free(h);
177 return ok;
178#endif
179}

237cwist_seq_auth_context_create(const uint8_t key[CWIST_SEQ_AUTH_KEY_SIZE],
238 const uint8_t session_id[CWIST_SEQ_AUTH_SESSION_ID_SIZE],
239 size_t replay_window) {
240 if (!key || !session_id || replay_window == 0 || replay_window > SIZE_MAX / 28) return NULL;
241 cwist_seq_auth_context_t *ctx = cwist_alloc(sizeof(*ctx));
242 if (!ctx) return NULL;
243 memset(ctx, 0, sizeof(*ctx));
244 if (pthread_mutex_init(&ctx->lock, NULL) != 0) {
245 cwist_free(ctx);
246 return NULL;
247 }
248 ctx->nonces = cwist_alloc_array(replay_window, CWIST_SEQ_AUTH_NONCE_SIZE);
249 ctx->completed_ids = cwist_alloc_array(replay_window, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE);
250 if (!ctx->nonces || !ctx->completed_ids) {
251 cwist_seq_auth_context_destroy(ctx);
252 return NULL;
253 }
254 memcpy(ctx->key, key, sizeof(ctx->key));
255 memcpy(ctx->session_id, session_id, sizeof(ctx->session_id));
256 ctx->capacity = replay_window;
257 return ctx;
258}

374bool cwist_seq_auth_unwrap(cwist_seq_auth_context_t *ctx, const uint8_t *data, size_t len,
375 cwist_seq_auth_metadata_t *metadata, cwist_seq_chunk_t *chunk) {
376 cwist_seq_chunk_t parsed;
377 if (!ctx || !metadata || !chunk || !seq_auth_chunk_valid(data, len, &parsed) ||
378 len != (size_t)CWIST_SEQ_AUTH_HEADER_SIZE + parsed.payload_len)
379 return false;
380 const uint8_t *message_id = data + CWIST_SEQ_HEADER_SIZE;
381 const uint8_t *nonce = message_id + CWIST_SEQ_AUTH_MESSAGE_ID_SIZE;
382 const uint8_t *tag = nonce + CWIST_SEQ_AUTH_NONCE_SIZE;
383 uint8_t expected[SEQ_AUTH_TAG_CAP];
384 if (!auth_tag(ctx, data,
385 CWIST_SEQ_HEADER_SIZE + CWIST_SEQ_AUTH_MESSAGE_ID_SIZE +
386 CWIST_SEQ_AUTH_NONCE_SIZE,
387 parsed.payload, parsed.payload_len, expected) ||
388 SEQ_AUTH_MEMCMP(tag, expected, CWIST_SEQ_AUTH_TAG_SIZE) != 0)
389 return false;
390 pthread_mutex_lock(&ctx->lock);
391 bool replayed =
392 cache_contains(ctx->nonces, ctx->nonce_count, CWIST_SEQ_AUTH_NONCE_SIZE, nonce) ||
393 cache_contains(ctx->completed_ids, ctx->completed_count, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE,
394 message_id);
395 if (replayed) {
396 pthread_mutex_unlock(&ctx->lock);
397 return false;
398 }
399 cache_add(ctx->nonces, &ctx->nonce_count, &ctx->nonce_next, ctx->capacity,
400 CWIST_SEQ_AUTH_NONCE_SIZE, nonce);
401 pthread_mutex_unlock(&ctx->lock);
402 memcpy(metadata->message_id, message_id, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE);
403 memcpy(metadata->nonce, nonce, CWIST_SEQ_AUTH_NONCE_SIZE);
404 *chunk = parsed;
405 return true;
406}
