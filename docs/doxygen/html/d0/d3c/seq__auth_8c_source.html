40static int seq_auth_memcmp(const uint8_t *a, const uint8_t *b, size_t len) {
41 uint8_t diff = 0;

101static bool seq_auth_chunk_valid(const uint8_t *data, size_t len, cwist_seq_chunk_t *out) {
102 if (!data || len < CWIST_SEQ_HEADER_SIZE || !out) return false;
103 out->seq = read_u16(data);
104 out->total = read_u16(data + 2);
105 out->payload_len = read_u16(data + 4);
106 out->chunk_size = read_u16(data + 6);
107 if (!out->seq || !out->total || out->total > CWIST_SEQ_MAX_CHUNKS || out->seq > out->total ||
108 !out->payload_len || !out->chunk_size || out->payload_len > out->chunk_size ||
109 (out->seq != out->total && out->payload_len != out->chunk_size) ||
110 len < (size_t)CWIST_SEQ_AUTH_HEADER_SIZE + out->payload_len)
111 return false;
112 out->payload = data + CWIST_SEQ_AUTH_HEADER_SIZE;
113 return true;
114}

134static bool auth_tag(const cwist_seq_auth_context_t *ctx, const uint8_t *header_without_tag,
135 size_t header_len, const uint8_t *payload, size_t payload_len,
136 uint8_t tag[SEQ_AUTH_TAG_CAP]) {
137#ifdef SEQ_AUTH_WASM_CRYPTO
138 /* Streaming HMAC-SHA256 (RFC 2104) over session_id || header || payload.
139 * The 32-byte key never exceeds the 64-byte block size, so no key
140 * pre-hashing is needed. */
141 uint8_t k[64] = {0};
142 memcpy(k, ctx->key, sizeof(ctx->key));
143 uint8_t ipad[64], opad[64];
144 for (size_t i = 0; i < 64; i++) {
145 ipad[i] = (uint8_t)(k[i] ^ 0x36);
146 opad[i] = (uint8_t)(k[i] ^ 0x5c);
147 }
148 cwist_sha256_ctx c;
149 uint8_t inner[CWIST_SHA256_DIGEST_LEN];
150 cwist_sha256_init(&c);
151 cwist_sha256_update(&c, ipad, sizeof(ipad));
152 cwist_sha256_update(&c, ctx->session_id, sizeof(ctx->session_id));
153 cwist_sha256_update(&c, header_without_tag, header_len);
154 cwist_sha256_update(&c, payload, payload_len);
155 cwist_sha256_final(&c, inner);
156 cwist_sha256_init(&c);
157 cwist_sha256_update(&c, opad, sizeof(opad));
158 cwist_sha256_update(&c, inner, sizeof(inner));
159 cwist_sha256_final(&c, tag);
160 SEQ_AUTH_CLEANSE(ipad, sizeof(ipad));
161 SEQ_AUTH_CLEANSE(opad, sizeof(opad));
162 SEQ_AUTH_CLEANSE(inner, sizeof(inner));
163 return true;
164#else
165 HMAC_CTX *h = HMAC_CTX_new();
166 unsigned int tag_len = 0;
167 bool ok = h && HMAC_Init_ex(h, ctx->key, sizeof(ctx->key), EVP_sha256(), NULL) == 1 &&
168 HMAC_Update(h, ctx->session_id, sizeof(ctx->session_id)) == 1 &&
169 HMAC_Update(h, header_without_tag, header_len) == 1 &&
170 HMAC_Update(h, payload, payload_len) == 1 && HMAC_Final(h, tag, &tag_len) == 1 &&
171 tag_len >= CWIST_SEQ_AUTH_TAG_SIZE;
172 HMAC_CTX_free(h);
173 return ok;
174#endif
175}

233cwist_seq_auth_context_create(const uint8_t key[CWIST_SEQ_AUTH_KEY_SIZE],
234 const uint8_t session_id[CWIST_SEQ_AUTH_SESSION_ID_SIZE],
235 size_t replay_window) {
236 if (!key || !session_id || replay_window == 0 || replay_window > SIZE_MAX / 28) return NULL;
237 cwist_seq_auth_context_t *ctx = cwist_alloc(sizeof(*ctx));
238 if (!ctx) return NULL;
239 memset(ctx, 0, sizeof(*ctx));
240 if (pthread_mutex_init(&ctx->lock, NULL) != 0) {
241 cwist_free(ctx);
242 return NULL;
243 }
244 ctx->nonces = cwist_alloc_array(replay_window, CWIST_SEQ_AUTH_NONCE_SIZE);
245 ctx->completed_ids = cwist_alloc_array(replay_window, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE);
246 if (!ctx->nonces || !ctx->completed_ids) {
247 cwist_seq_auth_context_destroy(ctx);
248 return NULL;
249 }
250 memcpy(ctx->key, key, sizeof(ctx->key));
251 memcpy(ctx->session_id, session_id, sizeof(ctx->session_id));
252 ctx->capacity = replay_window;
253 return ctx;
254}

334 CWIST_SEQ_HEADER_SIZE + CWIST_SEQ_AUTH_MESSAGE_ID_SIZE +

337 cwist_free(wire);

341 CWIST_SEQ_AUTH_NONCE_SIZE,

367bool cwist_seq_auth_unwrap(cwist_seq_auth_context_t *ctx, const uint8_t *data, size_t len,
368 cwist_seq_auth_metadata_t *metadata, cwist_seq_chunk_t *chunk) {
369 cwist_seq_chunk_t parsed;
370 if (!ctx || !metadata || !chunk || !seq_auth_chunk_valid(data, len, &parsed) ||
371 len != (size_t)CWIST_SEQ_AUTH_HEADER_SIZE + parsed.payload_len)
372 return false;
373 const uint8_t *message_id = data + CWIST_SEQ_HEADER_SIZE;
374 const uint8_t *nonce = message_id + CWIST_SEQ_AUTH_MESSAGE_ID_SIZE;
375 const uint8_t *tag = nonce + CWIST_SEQ_AUTH_NONCE_SIZE;
376 uint8_t expected[SEQ_AUTH_TAG_CAP];
377 if (!auth_tag(ctx, data,
378 CWIST_SEQ_HEADER_SIZE + CWIST_SEQ_AUTH_MESSAGE_ID_SIZE +
379 CWIST_SEQ_AUTH_NONCE_SIZE,
380 parsed.payload, parsed.payload_len, expected) ||
381 SEQ_AUTH_MEMCMP(tag, expected, CWIST_SEQ_AUTH_TAG_SIZE) != 0)
382 return false;
383 pthread_mutex_lock(&ctx->lock);
384 bool replayed =
385 cache_contains(ctx->nonces, ctx->nonce_count, CWIST_SEQ_AUTH_NONCE_SIZE, nonce) ||
386 cache_contains(ctx->completed_ids, ctx->completed_count, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE,
387 message_id);
388 if (replayed) {
389 pthread_mutex_unlock(&ctx->lock);
390 return false;
391 }
392 cache_add(ctx->nonces, &ctx->nonce_count, &ctx->nonce_next, ctx->capacity,
393 CWIST_SEQ_AUTH_NONCE_SIZE, nonce);
394 pthread_mutex_unlock(&ctx->lock);
395 memcpy(metadata->message_id, message_id, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE);
396 memcpy(metadata->nonce, nonce, CWIST_SEQ_AUTH_NONCE_SIZE);
397 *chunk = parsed;
398 return true;
399}
