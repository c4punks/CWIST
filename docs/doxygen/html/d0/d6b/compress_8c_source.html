89static int zlib_compress(void *state, const char *in, size_t in_len, char *out, size_t *out_len,
90 int flush) {
91 z_stream *zs = (z_stream *)state;
92 zs->next_in = (Bytef *)in;
93 zs->avail_in = (uInt)in_len;
94 zs->next_out = (Bytef *)out;
95 zs->avail_out = (uInt)(*out_len);
96 int rc = deflate(zs, flush ? Z_FINISH : Z_NO_FLUSH);
97 /* All-or-nothing: the caller advances the input by the full chunk, so a
98 * partially consumed chunk would silently drop the tail of the body.
99 * Report an error instead; the middleware then serves the body
100 * uncompressed. */
101 if (flush) {
102 if (rc != Z_STREAM_END) return -1;
103 } else {
104 if (rc != Z_OK || zs->avail_in != 0) return -1;
105 }
106 *out_len = (size_t)(*out_len - zs->avail_out);
107 return 0;
108}

118static int zlib_finish(void *state, char *out, size_t *out_len) {
119 z_stream *zs = (z_stream *)state;
120 zs->next_in = NULL;
121 zs->avail_in = 0;
122 zs->next_out = (Bytef *)out;
123 zs->avail_out = (uInt)(*out_len);
124 int rc = deflate(zs, Z_FINISH);
125 /* Z_BUF_ERROR here means the trailer did not fit; accepting it would
126 * emit a truncated gzip stream. */
127 if (rc != Z_STREAM_END) return -1;
128 *out_len = (size_t)(*out_len - zs->avail_out);
129 return 0;

238static int brotli_compress(void *state, const char *in, size_t in_len, char *out, size_t *out_len,
239 int flush) {
240 cwist_brotli_state_t *bs = (cwist_brotli_state_t *)state;
241 if (!bs || !out || !out_len) return -1;
242
243 if (!flush) {
244 if (in_len == 0) {
245 *out_len = 0;
246 return 0;
247 }
248 if (brotli_ensure_accum(bs, in_len) != 0) return -1;
249 memcpy(bs->accum + bs->accum_len, in, in_len);
250 bs->accum_len += in_len;
251 *out_len = 0;
252 return 0;
253 }
254
255 /* On flush, compress accumulated input plus the final chunk. */
256 size_t total_in_len = bs->accum_len + in_len;
257 const uint8_t *total_in = NULL;
258 uint8_t *merged = NULL;
259 if (bs->accum_len > 0) {
260 merged = (uint8_t *)malloc(total_in_len);
261 if (!merged) return -1;
262 if (bs->accum_len > 0) memcpy(merged, bs->accum, bs->accum_len);
263 if (in_len > 0) memcpy(merged + bs->accum_len, in, in_len);
264 total_in = merged;
265 } else {
266 total_in = (const uint8_t *)in;
267 }
268
269 size_t encoded_size = BrotliEncoderMaxCompressedSize(total_in_len);
270 if (encoded_size == 0) {
271 free(merged);
272 return -1;
273 }
274
275 uint8_t *encoded = (uint8_t *)malloc(encoded_size);
276 if (!encoded) {
277 free(merged);
278 return -1;
279 }
280
281 if (!BrotliEncoderCompress(BROTLI_DEFAULT_QUALITY, BROTLI_DEFAULT_WINDOW, BROTLI_DEFAULT_MODE,
282 total_in_len, total_in, &encoded_size, encoded)) {
283 free(encoded);
284 free(merged);
285 return -1;
286 }
287
288 free(merged);
289 bs->accum_len = 0;
290
291 /* Replace any previous output buffer with the new one. */
292 free(bs->out);
293 bs->out = (char *)encoded;
294 bs->out_len = encoded_size;
295 bs->out_pos = 0;
296
297 size_t to_copy = encoded_size;
298 if (to_copy > *out_len) to_copy = *out_len;
299 if (to_copy > 0 && out) memcpy(out, encoded, to_copy);
300 bs->out_pos = to_copy;
301 *out_len = to_copy;
302 return 0;
303}

426static int zstd_compress(void *state, const char *in, size_t in_len, char *out, size_t *out_len,
427 int flush) {
428 cwist_zstd_state_t *zs = (cwist_zstd_state_t *)state;
429 if (!zs || !out || !out_len) return -1;
430
431 if (!flush) {
432 if (in_len == 0) {
433 *out_len = 0;
434 return 0;
435 }
436 if (zstd_ensure_accum(zs, in_len) != 0) return -1;
437 memcpy(zs->accum + zs->accum_len, in, in_len);
438 zs->accum_len += in_len;
439 *out_len = 0;
440 return 0;
441 }
442
443 size_t total_in_len = zs->accum_len + in_len;
444 char *total_in = (char *)malloc(total_in_len);
445 if (!total_in) return -1;
446 if (zs->accum_len > 0) memcpy(total_in, zs->accum, zs->accum_len);
447 if (in_len > 0) memcpy(total_in + zs->accum_len, in, in_len);
448
449 size_t out_cap = ZSTD_compressBound(total_in_len);
450 char *out_buf = (char *)malloc(out_cap);
451 if (!out_buf) {
452 free(total_in);
453 return -1;
454 }
455
456 ZSTD_inBuffer ib = {total_in, total_in_len, 0};
457 ZSTD_outBuffer ob = {out_buf, out_cap, 0};
458
459 size_t rc = ZSTD_compressStream2(zs->cstream, &ob, &ib, ZSTD_e_end);
460 free(total_in);
461 if (ZSTD_isError(rc)) {
462 free(out_buf);
463 return -1;
464 }
465
466 zs->accum_len = 0;
467 free(zs->out);
468 zs->out = out_buf;
469 zs->out_len = ob.pos;
470 zs->out_pos = 0;
471
472 size_t to_copy = (ob.pos < *out_len) ? ob.pos : *out_len;
473 memcpy(out, out_buf, to_copy);
474 zs->out_pos = to_copy;
475 *out_len = to_copy;
476 return 0;
477}

535static int str_contains_ci(const char *haystack, const char *needle) {
536 if (!haystack || !needle) return 0;
537 size_t hlen = strlen(haystack);
538 size_t nlen = strlen(needle);
539 if (nlen == 0) return 1;
540 if (hlen < nlen) return 0;
541 for (size_t i = 0; i <= hlen - nlen; ++i) {
542 size_t j = 0;
543 for (; j < nlen; ++j) {
544 char a = haystack[i + j];
545 char b = needle[j];
546 if (a >= 'A' && a <= 'Z') a = a - 'A' + 'a';
547 if (b >= 'A' && b <= 'Z') b = b - 'A' + 'a';
548 if (a != b) break;
549 }
550 if (j == nlen) return 1;
551 }
552 return 0;
553}

612static void cwist_mw_compress_handler(cwist_http_request *req, cwist_http_response *res,
613 cwist_handler_func next) {
614 const char *accept = cwist_http_header_get(req->headers, "Accept-Encoding");
615 const cwist_compress_backend *backend = select_backend(accept);
616
617 next(req, res);
618
619 /* A deferred response belongs to its cwist_async completion, which may
620 * be writing it on another thread right now (cwist_async_respond(), the
621 * timeout job); leave it alone. */
622 if (res->deferred) return;
623 if (!backend) return;
624 if (!res->body || res->body->size == 0) return;
625 if (res->is_ptr_body || res->use_file_stream) return;
626 if (res->status_code < 200 || res->status_code >= 300) return;
627 if (res->body->size < g_compress_min_size) return;
628
629 void *state = NULL;
630 if (backend->init(&state) != 0) return;
631
632 size_t out_cap = res->body->size + 256;
633 char *out_buf = (char *)cwist_malloc(out_cap);
634 if (!out_buf) {
635 backend->cleanup(state);
636 return;
637 }
638
639 size_t total_out = 0;
640 int ok = 1;
641
642 /* Stream the entire body through the compressor */
643 size_t in_offset = 0;
644 while (in_offset < res->body->size && ok) {
645 size_t chunk_in = res->body->size - in_offset;
646 size_t chunk_out = out_cap - total_out;
647 if (chunk_out == 0) {
648 /* Grow output buffer */
649 size_t new_cap = out_cap * 2;
650 char *new_buf = (char *)cwist_realloc(out_buf, new_cap);
651 if (!new_buf) {
652 ok = 0;
653 break;
654 }
655 out_buf = new_buf;
656 out_cap = new_cap;
657 chunk_out = out_cap - total_out;
658 }
659 int flush = (in_offset + chunk_in >= res->body->size) ? 1 : 0;
660 size_t written = chunk_out;
661 if (backend->compress(state, res->body->data + in_offset, chunk_in, out_buf + total_out,
662 &written, flush) != 0) {
663 ok = 0;
664 break;
665 }
666 total_out += written;
667 in_offset += chunk_in;
668 }
669
670 if (ok) {
671 /* Drain trailing bytes, growing the buffer until the backend is
672 * exhausted (brotli/zstd buffer the whole stream internally, so a
673 * single fixed-size tail can truncate the output). */
674 for (;;) {
675 size_t tail = out_cap - total_out;
676 if (tail == 0) {
677 size_t new_cap = out_cap * 2;
678 char *new_buf = (char *)cwist_realloc(out_buf, new_cap);
679 if (!new_buf) {
680 ok = 0;
681 break;
682 }
683 out_buf = new_buf;
684 out_cap = new_cap;
685 continue;
686 }
687 size_t wrote = tail;
688 if (backend->finish(state, out_buf + total_out, &wrote) != 0) {
689 ok = 0;
690 break;
691 }
692 total_out += wrote;
693 if (wrote < tail) break; /* drained */
694 }
695 }
696
697 backend->cleanup(state);
698
699 if (ok && total_out > 0 && total_out < res->body->size) {
700 cwist_sstring_assign_len(res->body, out_buf, total_out);
701 cwist_http_header_add(&res->headers, "Content-Encoding", backend->encoding_name);
702 remove_header(&res->headers, "Content-Length");
703 }
704 cwist_free(out_buf);
705}
