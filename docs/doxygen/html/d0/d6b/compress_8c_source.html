94static int zlib_compress(void *state, const char *in, size_t in_len, char *out, size_t *out_len,
95 int flush) {
96 z_stream *zs = (z_stream *)state;
97 zs->next_in = (Bytef *)in;
98 zs->avail_in = (uInt)in_len;
99 zs->next_out = (Bytef *)out;
100 zs->avail_out = (uInt)(*out_len);
101 int rc = deflate(zs, flush ? Z_FINISH : Z_NO_FLUSH);
102 /* All-or-nothing: the caller advances the input by the full chunk, so a
103 * partially consumed chunk would silently drop the tail of the body.
104 * Report an error instead; the middleware then serves the body
105 * uncompressed. */
106 if (flush) {
107 if (rc != Z_STREAM_END) return -1;
108 } else {
109 if (rc != Z_OK || zs->avail_in != 0) return -1;
110 }
111 *out_len = (size_t)(*out_len - zs->avail_out);
112 return 0;
113}

123static int zlib_finish(void *state, char *out, size_t *out_len) {
124 z_stream *zs = (z_stream *)state;
125 zs->next_in = NULL;
126 zs->avail_in = 0;
127 zs->next_out = (Bytef *)out;
128 zs->avail_out = (uInt)(*out_len);
129 int rc = deflate(zs, Z_FINISH);
130 /* Z_BUF_ERROR here means the trailer did not fit; accepting it would
131 * emit a truncated gzip stream. */
132 if (rc != Z_STREAM_END) return -1;
133 *out_len = (size_t)(*out_len - zs->avail_out);
134 return 0;
135}

212static int brotli_ensure_accum(cwist_brotli_state_t *bs, size_t need) {
213 if (!bs) return -1;
214 size_t required = bs->accum_len + need;
215 if (required > bs->accum_cap) {
216 size_t new_cap = bs->accum_cap ? bs->accum_cap * 2 : 4096;
217 while (new_cap < required) new_cap *= 2;
218 char *tmp = (char *)realloc(bs->accum, new_cap);

243static int brotli_compress(void *state, const char *in, size_t in_len, char *out, size_t *out_len,
244 int flush) {
245 cwist_brotli_state_t *bs = (cwist_brotli_state_t *)state;
246 if (!bs || !out || !out_len) return -1;
247
248 if (!flush) {
249 if (in_len == 0) {
250 *out_len = 0;
251 return 0;
252 }
253 if (brotli_ensure_accum(bs, in_len) != 0) return -1;
254 memcpy(bs->accum + bs->accum_len, in, in_len);
255 bs->accum_len += in_len;
256 *out_len = 0;
257 return 0;
258 }
259
260 /* On flush, compress accumulated input plus the final chunk. */
261 size_t total_in_len = bs->accum_len + in_len;
262 const uint8_t *total_in = NULL;
263 uint8_t *merged = NULL;
264 if (bs->accum_len > 0) {
265 merged = (uint8_t *)malloc(total_in_len);
266 if (!merged) return -1;
267 if (bs->accum_len > 0) memcpy(merged, bs->accum, bs->accum_len);
268 if (in_len > 0) memcpy(merged + bs->accum_len, in, in_len);
269 total_in = merged;
270 } else {
271 total_in = (const uint8_t *)in;
272 }
273
274 size_t encoded_size = BrotliEncoderMaxCompressedSize(total_in_len);
275 if (encoded_size == 0) {
276 free(merged);
277 return -1;
278 }
279
280 uint8_t *encoded = (uint8_t *)malloc(encoded_size);
281 if (!encoded) {
282 free(merged);
283 return -1;
284 }
285
286 if (!BrotliEncoderCompress(BROTLI_DEFAULT_QUALITY, BROTLI_DEFAULT_WINDOW, BROTLI_DEFAULT_MODE,
287 total_in_len, total_in, &encoded_size, encoded)) {
288 free(encoded);
289 free(merged);
290 return -1;
291 }
292
293 free(merged);
294 bs->accum_len = 0;
295
296 /* Replace any previous output buffer with the new one. */
297 free(bs->out);
298 bs->out = (char *)encoded;
299 bs->out_len = encoded_size;
300 bs->out_pos = 0;
301
302 size_t to_copy = encoded_size;
303 if (to_copy > *out_len) to_copy = *out_len;
304 if (to_copy > 0 && out) memcpy(out, encoded, to_copy);
305 bs->out_pos = to_copy;
306 *out_len = to_copy;
307 return 0;
308}

431static int zstd_compress(void *state, const char *in, size_t in_len, char *out, size_t *out_len,
432 int flush) {
433 cwist_zstd_state_t *zs = (cwist_zstd_state_t *)state;
434 if (!zs || !out || !out_len) return -1;
435
436 if (!flush) {
437 if (in_len == 0) {
438 *out_len = 0;
439 return 0;
440 }
441 if (zstd_ensure_accum(zs, in_len) != 0) return -1;
442 memcpy(zs->accum + zs->accum_len, in, in_len);
443 zs->accum_len += in_len;
444 *out_len = 0;
445 return 0;
446 }
447
448 size_t total_in_len = zs->accum_len + in_len;
449 char *total_in = (char *)malloc(total_in_len);
450 if (!total_in) return -1;
451 if (zs->accum_len > 0) memcpy(total_in, zs->accum, zs->accum_len);
452 if (in_len > 0) memcpy(total_in + zs->accum_len, in, in_len);
453
454 size_t out_cap = ZSTD_compressBound(total_in_len);
455 char *out_buf = (char *)malloc(out_cap);
456 if (!out_buf) {
457 free(total_in);
458 return -1;
459 }
460
461 ZSTD_inBuffer ib = {total_in, total_in_len, 0};
462 ZSTD_outBuffer ob = {out_buf, out_cap, 0};
463
464 size_t rc = ZSTD_compressStream2(zs->cstream, &ob, &ib, ZSTD_e_end);
465 free(total_in);
466 if (ZSTD_isError(rc)) {
467 free(out_buf);
468 return -1;
469 }
470
471 zs->accum_len = 0;
472 free(zs->out);
473 zs->out = out_buf;
474 zs->out_len = ob.pos;
475 zs->out_pos = 0;
476
477 size_t to_copy = (ob.pos < *out_len) ? ob.pos : *out_len;
478 memcpy(out, out_buf, to_copy);
479 zs->out_pos = to_copy;
480 *out_len = to_copy;
481 return 0;
482}

533static int coding_equal(const char *a, const char *b, size_t len) {
534 for (size_t i = 0; i < len; ++i) {
535 char ca = a[i], cb = b[i];
536 if (ca >= 'A' && ca <= 'Z') ca = (char)(ca - 'A' + 'a');
537 if (cb >= 'A' && cb <= 'Z') cb = (char)(cb - 'A' + 'a');
538 if (ca != cb) return 0;
539 }
540 return 1;
541}

543static double parse_encoding_quality(const char *params, const char *end) {
544 const char *p = params;
545 while (p < end) {
546 while (p < end && (*p == ';' || *p == ' ' || *p == '\t')) p++;
547 const char *param_end = p;
548 while (param_end < end && *param_end != ';') param_end++;
549 const char *eq = memchr(p, '=', (size_t)(param_end - p));
550 if (eq && eq - p == 1 && (p[0] == 'q' || p[0] == 'Q')) {
551 const char *value = eq + 1;
552 while (value < param_end && (*value == ' ' || *value == '\t')) value++;
553 while (param_end > value && (param_end[-1] == ' ' || param_end[-1] == '\t'))
554 param_end--;
555 char buf[16];
556 size_t len = (size_t)(param_end - value);
557 if (len == 0 || len >= sizeof(buf)) return 0.0;
558 memcpy(buf, value, len);
559 buf[len] = '\0';
560 char *parsed = NULL;
561 double quality = strtod(buf, &parsed);
562 return parsed != buf && parsed && *parsed == '\0' && quality >= 0.0 && quality <= 1.0
563 ? quality
564 : 0.0;
565 }
566 p = param_end;
567 }
568 return 1.0;
569}

571static double encoding_quality(const char *accept, const char *coding) {
572 size_t coding_len = strlen(coding);
573 double wildcard_quality = -1.0;
574 const char *p = accept;
575 while (*p) {
576 const char *end = strchr(p, ',');
577 if (!end) end = p + strlen(p);
578 while (p < end && (*p == ' ' || *p == '\t')) p++;
579 const char *token_end = p;
580 while (token_end < end && *token_end != ';' && *token_end != ' ' && *token_end != '\t')
581 token_end++;
582 size_t token_len = (size_t)(token_end - p);
583 if (token_len == coding_len && coding_equal(p, coding, coding_len))
584 return parse_encoding_quality(token_end, end);
585 if (token_len == 1 && *p == '*' && wildcard_quality < 0.0)
586 wildcard_quality = parse_encoding_quality(token_end, end);
587 p = *end ? end + 1 : end;
588 }
589 return wildcard_quality > 0.0 ? wildcard_quality : 0.0;
590}

594static const cwist_compress_backend *select_backend(const char *accept_encoding) {
595 if (!accept_encoding) return NULL;
596 const cwist_compress_backend *best = NULL;

634static bool vary_has_accept_encoding(const cwist_http_header_node *headers) {
635 for (const cwist_http_header_node *node = headers; node; node = node->next) {
636 if (!node->key || !node->value || strcasecmp(node->key->data, "Vary") != 0) continue;
637 const char *p = node->value->data;
638 while (*p) {
639 while (*p == ' ' || *p == '\t' || *p == ',') p++;
640 const char *end = p;
641 while (*end && *end != ',') end++;
642 const char *trim = end;
643 while (trim > p && (trim[-1] == ' ' || trim[-1] == '\t')) trim--;
644 if ((trim - p == 1 && *p == '*') ||
645 (trim - p == 15 && strncasecmp(p, "Accept-Encoding", 15) == 0))
646 return true;
647 p = *end ? end + 1 : end;
648 }
649 }
650 return false;
651}

673static void cwist_mw_compress_handler_core(cwist_http_request *req, cwist_http_response *res,
674 cwist_handler_func next, size_t min_body_size) {
675 const char *accept = cwist_http_header_get(req->headers, "Accept-Encoding");
676 const cwist_compress_backend *backend = select_backend(accept);
677
678 next(req, res);
679
680 /* A deferred response belongs to its cwist_async completion, which may
681 * be writing it on another thread right now (cwist_async_respond(), the
682 * timeout job); leave it alone. */
683 if (cwist_async_res_deferred(res)) return;
684 if (res->body && res->body->size >= min_body_size && res->body->size > 0 && !res->is_ptr_body &&
685 !res->use_file_stream && res->status_code >= 200 && res->status_code < 300)
686 add_vary_accept_encoding(res);
687 if (!backend) return;
688 if (!res->body || res->body->size == 0) return;
689 if (res->is_ptr_body || res->use_file_stream) return;
690 if (res->status_code < 200 || res->status_code >= 300) return;
691 if (res->body->size < min_body_size) return;
692
693 void *state = NULL;
694 if (backend->init(&state) != 0) return;
695
696 size_t out_cap = res->body->size + 256;
697 char *out_buf = (char *)cwist_malloc(out_cap);
698 if (!out_buf) {
699 backend->cleanup(state);
700 return;
701 }
702
703 size_t total_out = 0;
704 int ok = 1;
705
706 /* Stream the entire body through the compressor */
707 size_t in_offset = 0;
708 while (in_offset < res->body->size && ok) {
709 size_t chunk_in = res->body->size - in_offset;
710 size_t chunk_out = out_cap - total_out;
711 if (chunk_out == 0) {
712 /* Grow output buffer */
713 size_t new_cap = out_cap * 2;
714 char *new_buf = (char *)cwist_realloc(out_buf, new_cap);
715 if (!new_buf) {
716 ok = 0;
717 break;
718 }
719 out_buf = new_buf;
720 out_cap = new_cap;
721 chunk_out = out_cap - total_out;
722 }
723 int flush = (in_offset + chunk_in >= res->body->size) ? 1 : 0;
724 size_t written = chunk_out;
725 if (backend->compress(state, res->body->data + in_offset, chunk_in, out_buf + total_out,
726 &written, flush) != 0) {
727 ok = 0;
728 break;
729 }
730 total_out += written;
731 in_offset += chunk_in;
732 }
733
734 if (ok) {
735 /* Drain trailing bytes, growing the buffer until the backend is
736 * exhausted (brotli/zstd buffer the whole stream internally, so a
737 * single fixed-size tail can truncate the output). */
738 for (;;) {
739 size_t tail = out_cap - total_out;
740 if (tail == 0) {
741 size_t new_cap = out_cap * 2;
742 char *new_buf = (char *)cwist_realloc(out_buf, new_cap);
743 if (!new_buf) {
744 ok = 0;
745 break;
746 }
747 out_buf = new_buf;
748 out_cap = new_cap;
749 continue;
750 }
751 size_t wrote = tail;
752 if (backend->finish(state, out_buf + total_out, &wrote) != 0) {
753 ok = 0;
754 break;
755 }
756 total_out += wrote;
757 if (wrote < tail) break; /* drained */
758 }
759 }
760
761 backend->cleanup(state);
762
763 if (ok && total_out > 0 && total_out < res->body->size) {
764 cwist_sstring_assign_len(res->body, out_buf, total_out);
765 cwist_http_header_add(&res->headers, "Content-Encoding", backend->encoding_name);
766 remove_header(&res->headers, "Content-Length");
767 }
768 cwist_free(out_buf);
769}

793cwist_middleware_func_ex cwist_mw_compress_ex(size_t min_body_size, void **user_ctx,
794 cwist_middleware_ctx_destroy_func *destroy) {
795 if (!user_ctx || !destroy) return NULL;
