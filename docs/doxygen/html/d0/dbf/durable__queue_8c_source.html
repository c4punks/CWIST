249static char *jq_blob(const char *type, uint32_t attempts, const void *payload, size_t payload_len,
250 size_t *out_len) {
251 uint32_t type_len = type ? (uint32_t)strlen(type) : 0;
252 size_t len = 8 + type_len + payload_len;
253 char *blob = cwist_alloc(len ? len : 1);
254 if (!blob) return NULL;
255 blob[0] = (char)(type_len >> 24);
256 blob[1] = (char)(type_len >> 16);
257 blob[2] = (char)(type_len >> 8);
258 blob[3] = (char)type_len;
259 blob[4] = (char)(attempts >> 24);
260 blob[5] = (char)(attempts >> 16);
261 blob[6] = (char)(attempts >> 8);
262 blob[7] = (char)attempts;
263 if (type_len) memcpy(blob + 8, type, type_len);
264 if (payload_len) memcpy(blob + 8 + type_len, payload, payload_len);
265 *out_len = len;
266 return blob;
267}

483static cwist_error_t jq_redis_claim(cwist_job_queue_t *q, uint64_t timeout_ms, cwist_job_t **job) {
484 uint64_t deadline = jq_now_ms() + timeout_ms;
485 bool polled = false;
486 for (;;) {
487 jq_redis_reap_delay(q);
488 cwist_job_t *found = jq_redis_autoclaim(q);
489 if (found) {
490 *job = found;
491 return jq_err(0);
492 }
493 uint64_t now = jq_now_ms();
494 if (polled && now >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
495 polled = true;
496 uint64_t block = deadline > now ? deadline - now : 1;
497 if (block > CWIST_JQ_MAX_POLL_SLICE_MS) block = CWIST_JQ_MAX_POLL_SLICE_MS;
498 found = jq_redis_readgroup(q, block);
499 if (found) {
500 *job = found;
501 return jq_err(0);
502 }
503 if (jq_now_ms() >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
504 }
505}

666cwist_job_queue_t *cwist_job_queue_create_nats(cwist_nats_t *nats,
667 const cwist_job_queue_opts_t *opts) {
668 if (!nats) return NULL;
669 natsConnection *nc = cwist_nats_native(nats);
670 if (!nc) return NULL;
671 cwist_job_queue_t *q = cwist_alloc(sizeof(*q));
672 if (!q) return NULL;
673 memset(q, 0, sizeof(*q));
674 if (!jq_opts_init(q, opts)) goto fail;
675 q->backend = 1;
676 q->dead_count = jq_nats_dead_count;
677 jq_nats_t *ns = &q->u.nats;
678 ns->conn = nc;
679 char *base = jq_sanitize(q->name);
680 char *consumer = jq_sanitize(q->consumer);
681 if (!base || !consumer) {
682 cwist_free(base);
683 cwist_free(consumer);
684 goto fail;
685 }
686 size_t n = strlen(base);
687 ns->subject = cwist_alloc(n + 6);
688 ns->dead_subject = cwist_alloc(n + 6);
689 ns->dead_stream = cwist_alloc(n + 6);
690 if (!ns->subject || !ns->dead_subject || !ns->dead_stream) {
691 cwist_free(base);
692 cwist_free(consumer);
693 goto fail;
694 }
695 snprintf(ns->subject, n + 6, "%s.jobs", base);
696 snprintf(ns->dead_subject, n + 6, "%s.dead", base);
697 snprintf(ns->dead_stream, n + 5, "%s_DEAD", base);
698
699 if (natsConnection_JetStream(&ns->js, nc, NULL) != NATS_OK) {
700 cwist_free(base);
701 cwist_free(consumer);
702 goto fail;
703 }
704 cwist_error_t serr = jq_nats_ensure_stream(ns, base, ns->subject);
705 if (cwist_error_is_ok(&serr))
706 serr = jq_nats_ensure_stream(ns, ns->dead_stream, ns->dead_subject);
707 if (!cwist_error_is_ok(&serr)) {
708 cwist_free(base);
709 cwist_free(consumer);
710 goto fail;
711 }
712
713 /* Durable pull consumer; tolerate "already exists" (same config from a
714 * previous run). */
715 jsErrCode jerr = 0;
716 jsConsumerInfo *ci = NULL;
717 jsConsumerConfig cc;
718 memset(&cc, 0, sizeof(cc));
719 cc.Durable = consumer;
720 cc.AckPolicy = js_AckExplicit;
721 cc.AckWait = (int64_t)q->visibility_timeout_ms * 1000000;
722 cc.MaxDeliver = (int64_t)q->max_retries + 1;
723 cc.DeliverPolicy = js_DeliverAll;
724 cc.FilterSubject = ns->subject;
725 if (js_AddConsumer(&ci, ns->js, base, &cc, NULL, &jerr) == NATS_OK) {
726 jsConsumerInfo_Destroy(ci);
727 } else if (js_GetConsumerInfo(&ci, ns->js, base, consumer, NULL, &jerr) != NATS_OK) {
728 cwist_free(base);
729 cwist_free(consumer);
730 goto fail;
731 } else {
732 jsConsumerInfo_Destroy(ci);
733 }
734 if (js_PullSubscribe(&ns->sub, ns->js, ns->subject, consumer, NULL, NULL, &jerr) != NATS_OK ||
735 natsConnection_Flush(nc) != NATS_OK) {
736 cwist_free(base);
737 cwist_free(consumer);
738 goto fail;
739 }
740 cwist_free(base);
741 cwist_free(consumer);
742 return q;
743
744fail:
745 if (q) {
746 if (q->u.nats.sub) natsSubscription_Destroy(q->u.nats.sub);
747 if (q->u.nats.js) jsCtx_Destroy(q->u.nats.js);
748 cwist_free(q->u.nats.subject);
749 cwist_free(q->u.nats.dead_subject);
750 cwist_free(q->u.nats.dead_stream);
751 cwist_free(q->name);
752 cwist_free(q->consumer);
753 cwist_free(q);
754 }
755 return NULL;

802static cwist_error_t jq_nats_claim(cwist_job_queue_t *q, uint64_t timeout_ms, cwist_job_t **job) {
803 jq_nats_t *ns = &q->u.nats;
804 uint64_t deadline = jq_now_ms() + timeout_ms;
805 for (;;) {
806 uint64_t now = jq_now_ms();
807 int64_t slice = (int64_t)(now >= deadline ? 1 : deadline - now);
808 if (slice > (int64_t)CWIST_JQ_MAX_POLL_SLICE_MS)
809 slice = (int64_t)CWIST_JQ_MAX_POLL_SLICE_MS;
810 natsMsgList list;
811 memset(&list, 0, sizeof(list));
812 natsStatus s = natsSubscription_Fetch(&list, ns->sub, 1, slice, NULL);
813 if (s != NATS_OK) {
814 if (s == NATS_TIMEOUT) {
815 if (jq_now_ms() >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
816 continue;
817 }
818 return jq_err(CWIST_FAILURE);
819 }
820 for (int i = 0; i < list.Count; i++) {
821 natsMsg *msg = list.Msgs[i];
822 list.Msgs[i] = NULL; /* keep the list from double-destroying it */
823 jsMsgMetaData *meta = NULL;
824 if (natsMsg_GetMetaData(&meta, msg) != NATS_OK) {
825 natsMsg_Destroy(msg);
826 continue;
827 }
828 uint64_t delivered = meta->NumDelivered;
829 if (delivered > (uint64_t)q->max_retries) {
830 /* Max deliveries exhausted: park and terminate. */
831 jq_nats_dead_letter(ns, msg);
832 natsMsg_Destroy(msg);
833 jsMsgMetaData_Destroy(meta);
834 continue;
835 }
836 cwist_job_t *j = cwist_alloc(sizeof(*j));
837 if (!j) {
838 natsMsg_Destroy(msg);
839 jsMsgMetaData_Destroy(meta);
840 natsMsgList_Destroy(&list);
841 return jq_err(CWIST_ERROR_NOMEM);
842 }
843 memset(j, 0, sizeof(*j));
844 j->q = q;
845 j->backend = msg; /* ownership moves to the job */
846 char id_buf[32];
847 snprintf(id_buf, sizeof(id_buf), "%llu", (unsigned long long)meta->Sequence.Stream);
848 j->id = cwist_strdup(id_buf);
849 int dlen = natsMsg_GetDataLength(msg);
850 j->payload_len = dlen > 0 ? (size_t)dlen : 0;
851 if (j->payload_len) {
852 j->payload = cwist_alloc(j->payload_len);
853 if (j->payload) memcpy(j->payload, natsMsg_GetData(msg), j->payload_len);
854 }
855 const char *type = NULL;
856 if (natsMsgHeader_Get(msg, "Cwist-Type", &type) == NATS_OK && type && *type)
857 j->type = cwist_strdup(type);
858 j->attempts = (uint32_t)delivered;
859 jsMsgMetaData_Destroy(meta);
860 if (!j->id || (j->payload_len && !j->payload)) {
861 cwist_job_free(j);
862 natsMsgList_Destroy(&list);
863 return jq_err(CWIST_ERROR_NOMEM);
864 }
865 natsMsgList_Destroy(&list); /* frees the array, not the claimed msg */
866 *job = j;
867 return jq_err(0);
868 }
869 natsMsgList_Destroy(&list);
870 if (jq_now_ms() >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
871 }
872}
