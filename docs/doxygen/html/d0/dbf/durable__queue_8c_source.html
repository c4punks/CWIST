46/* Atomically schedules a nacked job: XACK, then either park it in the
47 * delay zset or dead-letter it. Returns 1 (delayed), 2 (dead-lettered),
48 * 0 (entry no longer pending). */
49static const char *JQ_NACK_SCRIPT = "local ok = redis.call('XACK', KEYS[1], KEYS[2], ARGV[1])\n"
50 "if ok == 0 then return 0 end\n"
51 "redis.call('HDEL', KEYS[5], ARGV[1])\n"
52 "if tonumber(ARGV[2]) > tonumber(ARGV[3]) then\n"
53 " redis.call('RPUSH', KEYS[3], ARGV[4])\n"
54 " return 2\n"
55 "end\n"
56 "redis.call('ZADD', KEYS[4], ARGV[5], ARGV[4])\n"
57 "return 1\n";

261static char *jq_blob(const char *type, uint32_t attempts, const void *payload, size_t payload_len,
262 size_t *out_len) {
263 uint32_t type_len = type ? (uint32_t)strlen(type) : 0;
264 size_t len = 8 + type_len + payload_len;
265 char *blob = cwist_alloc(len ? len : 1);
266 if (!blob) return NULL;
267 blob[0] = (char)(type_len >> 24);
268 blob[1] = (char)(type_len >> 16);
269 blob[2] = (char)(type_len >> 8);
270 blob[3] = (char)type_len;
271 blob[4] = (char)(attempts >> 24);
272 blob[5] = (char)(attempts >> 16);
273 blob[6] = (char)(attempts >> 8);
274 blob[7] = (char)attempts;
275 if (type_len) memcpy(blob + 8, type, type_len);
276 if (payload_len) memcpy(blob + 8 + type_len, payload, payload_len);
277 *out_len = len;
278 return blob;
279}

313static cwist_job_t *jq_job_from_entry(cwist_job_queue_t *q, const cwist_redis_reply_t *entry) {
314 if (!entry || entry->type != '*' || entry->elements != 2 || !entry->element[0] ||
315 !entry->element[0]->str)
316 return NULL;
317 const cwist_redis_reply_t *id = entry->element[0];
318 const cwist_redis_reply_t *fields = entry->element[1];
319 const cwist_redis_reply_t *type = jq_entry_field(fields, "type");
320 const cwist_redis_reply_t *payload = jq_entry_field(fields, "payload");
321
322 cwist_job_t *job = cwist_alloc(sizeof(*job));
323 if (!job) return NULL;
324 memset(job, 0, sizeof(*job));
325 job->q = q;
326 job->id = cwist_strndup(id->str, id->len);
327 if (type && type->str && type->len) job->type = cwist_strndup(type->str, type->len);
328 job->payload_len = payload && payload->str ? payload->len : 0;
329 if (job->payload_len) {
330 job->payload = cwist_alloc(job->payload_len);
331 if (job->payload) memcpy(job->payload, payload->str, job->payload_len);
332 }
333 if (!job->id || (job->payload_len && !job->payload)) {
334 cwist_job_free(job);
335 return NULL;

495static cwist_error_t jq_redis_claim(cwist_job_queue_t *q, uint64_t timeout_ms, cwist_job_t **job) {
496 uint64_t deadline = jq_now_ms() + timeout_ms;
497 bool polled = false;
498 for (;;) {
499 jq_redis_reap_delay(q);
500 cwist_job_t *found = jq_redis_autoclaim(q);
501 if (found) {
502 *job = found;
503 return jq_err(0);
504 }
505 uint64_t now = jq_now_ms();
506 if (polled && now >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
507 polled = true;
508 uint64_t block = deadline > now ? deadline - now : 1;
509 if (block > CWIST_JQ_MAX_POLL_SLICE_MS) block = CWIST_JQ_MAX_POLL_SLICE_MS;
510 found = jq_redis_readgroup(q, block);
511 if (found) {
512 *job = found;
513 return jq_err(0);
514 }
515 if (jq_now_ms() >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
516 }
517}

679cwist_job_queue_t *cwist_job_queue_create_nats(cwist_nats_t *nats,
680 const cwist_job_queue_opts_t *opts) {
681 if (!nats) return NULL;
682 natsConnection *nc = cwist_nats_native(nats);
683 if (!nc) return NULL;
684 cwist_job_queue_t *q = cwist_alloc(sizeof(*q));
685 if (!q) return NULL;
686 memset(q, 0, sizeof(*q));
687 if (!jq_opts_init(q, opts)) goto fail;
688 q->backend = 1;
689 q->dead_count = jq_nats_dead_count;
690 jq_nats_t *ns = &q->u.nats;
691 ns->conn = nc;
692 char *base = jq_sanitize(q->name);
693 char *consumer = jq_sanitize(q->consumer);
694 if (!base || !consumer) {
695 cwist_free(base);
696 cwist_free(consumer);
697 goto fail;
698 }
699 size_t n = strlen(base);
700 ns->subject = cwist_alloc(n + 6);
701 ns->dead_subject = cwist_alloc(n + 6);
702 ns->dead_stream = cwist_alloc(n + 6);
703 if (!ns->subject || !ns->dead_subject || !ns->dead_stream) {
704 cwist_free(base);
705 cwist_free(consumer);
706 goto fail;
707 }
708 snprintf(ns->subject, n + 6, "%s.jobs", base);
709 snprintf(ns->dead_subject, n + 6, "%s.dead", base);
710 snprintf(ns->dead_stream, n + 5, "%s_DEAD", base);
711
712 if (natsConnection_JetStream(&ns->js, nc, NULL) != NATS_OK) {
713 cwist_free(base);
714 cwist_free(consumer);
715 goto fail;
716 }
717 cwist_error_t serr = jq_nats_ensure_stream(ns, base, ns->subject);
718 if (cwist_error_is_ok(&serr))
719 serr = jq_nats_ensure_stream(ns, ns->dead_stream, ns->dead_subject);
720 if (!cwist_error_is_ok(&serr)) {
721 cwist_free(base);
722 cwist_free(consumer);
723 goto fail;
724 }
725
726 /* Durable pull consumer; tolerate "already exists" (same config from a
727 * previous run). */
728 jsErrCode jerr = 0;
729 jsConsumerInfo *ci = NULL;
730 jsConsumerConfig cc;
731 memset(&cc, 0, sizeof(cc));
732 cc.Durable = consumer;
733 cc.AckPolicy = js_AckExplicit;
734 cc.AckWait = (int64_t)q->visibility_timeout_ms * 1000000;
735 cc.MaxDeliver = (int64_t)q->max_retries + 1;
736 cc.DeliverPolicy = js_DeliverAll;
737 cc.FilterSubject = ns->subject;
738 if (js_AddConsumer(&ci, ns->js, base, &cc, NULL, &jerr) == NATS_OK) {
739 jsConsumerInfo_Destroy(ci);
740 } else if (js_GetConsumerInfo(&ci, ns->js, base, consumer, NULL, &jerr) != NATS_OK) {
741 cwist_free(base);
742 cwist_free(consumer);
743 goto fail;
744 } else {
745 jsConsumerInfo_Destroy(ci);
746 }
747 if (js_PullSubscribe(&ns->sub, ns->js, ns->subject, consumer, NULL, NULL, &jerr) != NATS_OK ||
748 natsConnection_Flush(nc) != NATS_OK) {
749 cwist_free(base);
750 cwist_free(consumer);
751 goto fail;
752 }
753 cwist_free(base);
754 cwist_free(consumer);
755 return q;
756
757fail:
758 if (q) {
759 if (q->u.nats.sub) natsSubscription_Destroy(q->u.nats.sub);
760 if (q->u.nats.js) jsCtx_Destroy(q->u.nats.js);
761 cwist_free(q->u.nats.subject);
762 cwist_free(q->u.nats.dead_subject);
763 cwist_free(q->u.nats.dead_stream);
764 cwist_free(q->name);
765 cwist_free(q->consumer);
766 cwist_free(q);
767 }
768 return NULL;
769}

775static cwist_error_t jq_nats_enqueue(cwist_job_queue_t *q, const void *payload, size_t payload_len,
776 const char *type) {
777 jq_nats_t *ns = &q->u.nats;
778 natsMsg *msg = NULL;
779 if (natsMsg_Create(&msg, ns->subject, NULL, (const char *)payload, (int)payload_len) != NATS_OK)
780 return jq_err(CWIST_FAILURE);
781 if (type && natsMsgHeader_Set(msg, "Cwist-Type", type) != NATS_OK) {
782 natsMsg_Destroy(msg);
783 return jq_err(CWIST_FAILURE);
784 }
785 jsErrCode jerr = 0;
786 jsPubAck *pa = NULL;
787 natsStatus s = js_PublishMsg(&pa, ns->js, msg, NULL, &jerr);
788 if (pa) jsPubAck_Destroy(pa);
789 natsMsg_Destroy(msg);
790 return s == NATS_OK ? jq_err(0) : jq_err(CWIST_FAILURE);
791}

794static void jq_nats_dead_letter(jq_nats_t *ns, natsMsg *msg) {
795 natsMsg *dm = NULL;
796 if (natsMsg_Create(&dm, ns->dead_subject, NULL, natsMsg_GetData(msg),
797 natsMsg_GetDataLength(msg)) == NATS_OK) {
798 const char *type = NULL;
799 if (natsMsgHeader_Get(msg, "Cwist-Type", &type) == NATS_OK && type)
800 natsMsgHeader_Set(dm, "Cwist-Type", type);
801 jsErrCode jerr = 0;
802 jsPubAck *pa = NULL;
803 if (js_PublishMsg(&pa, ns->js, dm, NULL, &jerr) == NATS_OK && pa) jsPubAck_Destroy(pa);
804 natsMsg_Destroy(dm);
805 }
806 natsMsg_Term(msg, NULL);

815static cwist_error_t jq_nats_claim(cwist_job_queue_t *q, uint64_t timeout_ms, cwist_job_t **job) {
816 jq_nats_t *ns = &q->u.nats;
817 uint64_t deadline = jq_now_ms() + timeout_ms;
818 for (;;) {
819 uint64_t now = jq_now_ms();
820 int64_t slice = (int64_t)(now >= deadline ? 1 : deadline - now);
821 if (slice > (int64_t)CWIST_JQ_MAX_POLL_SLICE_MS)
822 slice = (int64_t)CWIST_JQ_MAX_POLL_SLICE_MS;
823 natsMsgList list;
824 memset(&list, 0, sizeof(list));
825 natsStatus s = natsSubscription_Fetch(&list, ns->sub, 1, slice, NULL);
826 if (s != NATS_OK) {
827 if (s == NATS_TIMEOUT) {
828 if (jq_now_ms() >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
829 continue;
830 }
831 return jq_err(CWIST_FAILURE);
832 }
833 for (int i = 0; i < list.Count; i++) {
834 natsMsg *msg = list.Msgs[i];
835 list.Msgs[i] = NULL; /* keep the list from double-destroying it */
836 jsMsgMetaData *meta = NULL;
837 if (natsMsg_GetMetaData(&meta, msg) != NATS_OK) {
838 natsMsg_Destroy(msg);
839 continue;
840 }
841 uint64_t delivered = meta->NumDelivered;
842 if (delivered > (uint64_t)q->max_retries) {
843 /* Max deliveries exhausted: park and terminate. */
844 jq_nats_dead_letter(ns, msg);
845 natsMsg_Destroy(msg);
846 jsMsgMetaData_Destroy(meta);
847 continue;
848 }
849 cwist_job_t *j = cwist_alloc(sizeof(*j));
850 if (!j) {
851 natsMsg_Destroy(msg);
852 jsMsgMetaData_Destroy(meta);
853 natsMsgList_Destroy(&list);
854 return jq_err(CWIST_ERROR_NOMEM);
855 }
856 memset(j, 0, sizeof(*j));
857 j->q = q;
858 j->backend = msg; /* ownership moves to the job */
859 char id_buf[32];
860 snprintf(id_buf, sizeof(id_buf), "%llu", (unsigned long long)meta->Sequence.Stream);
861 j->id = cwist_strdup(id_buf);
862 int dlen = natsMsg_GetDataLength(msg);
863 j->payload_len = dlen > 0 ? (size_t)dlen : 0;
864 if (j->payload_len) {
865 j->payload = cwist_alloc(j->payload_len);
866 if (j->payload) memcpy(j->payload, natsMsg_GetData(msg), j->payload_len);
867 }
868 const char *type = NULL;
869 if (natsMsgHeader_Get(msg, "Cwist-Type", &type) == NATS_OK && type && *type)
870 j->type = cwist_strdup(type);
871 j->attempts = (uint32_t)delivered;
872 jsMsgMetaData_Destroy(meta);
873 if (!j->id || (j->payload_len && !j->payload)) {
874 cwist_job_free(j);
875 natsMsgList_Destroy(&list);
876 return jq_err(CWIST_ERROR_NOMEM);
877 }
878 natsMsgList_Destroy(&list); /* frees the array, not the claimed msg */
879 *job = j;
880 return jq_err(0);
881 }
882 natsMsgList_Destroy(&list);
883 if (jq_now_ms() >= deadline) return jq_err(CWIST_ERROR_TIMEOUT);
884 }
885}
