63bool cwist_seq_chunk_parse(const uint8_t *data, size_t len, cwist_seq_chunk_t *out) {
64 if (!data || !out || len < CWIST_SEQ_HEADER_SIZE) return false;
65
66 out->seq = seq_read_u16(data + 0);
67 out->total = seq_read_u16(data + 2);
68 out->payload_len = seq_read_u16(data + 4);
69 out->chunk_size = seq_read_u16(data + 6);
70 out->payload = data + CWIST_SEQ_HEADER_SIZE;
71
72 /* Reject missing or malformed sequence pairs. */
73 if (out->seq == 0 || out->total == 0 || out->total > CWIST_SEQ_MAX_CHUNKS ||
74 out->seq > out->total)
75 return false;
76 if (out->payload_len == 0 || out->chunk_size == 0) return false;
77 if (out->payload_len > out->chunk_size) return false;
78 if (out->seq != out->total && out->payload_len != out->chunk_size) return false;
79 if (len - CWIST_SEQ_HEADER_SIZE < out->payload_len) return false;
80
81 return true;
82}

125bool cwist_seq_split(const uint8_t *data, size_t len, uint16_t chunk_payload_size,
126 cwist_seq_message_t *out) {
127 if (!data || len == 0 || chunk_payload_size == 0 || !out) return false;
128 memset(out, 0, sizeof(*out));
129
130 uint32_t total32 = (uint32_t)((len + chunk_payload_size - 1) / chunk_payload_size);
131 if (total32 == 0 || total32 > UINT16_MAX || total32 > CWIST_SEQ_MAX_CHUNKS ||
132 len > CWIST_SEQ_MAX_REASSEMBLED_SIZE)
133 return false;
134 uint16_t total = (uint16_t)total32;
135
136 out->chunks = (uint8_t **)cwist_alloc_array(total, sizeof(uint8_t *));
137 out->chunk_lens = (size_t *)cwist_alloc_array(total, sizeof(size_t));
138 if (!out->chunks || !out->chunk_lens) {
139 cwist_free(out->chunks);
140 cwist_free(out->chunk_lens);
141 memset(out, 0, sizeof(*out));
142 return false;
143 }
144
145 for (uint16_t i = 0; i < total; i++) {
146 size_t offset = (size_t)i * chunk_payload_size;
147 size_t plen = len - offset;
148 if (plen > chunk_payload_size) plen = chunk_payload_size;
149 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + plen;
150
151 uint8_t *chunk = (uint8_t *)cwist_alloc(chunk_len);
152 if (!chunk) {
153 for (uint16_t j = 0; j < i; j++) cwist_free(out->chunks[j]);
154 cwist_free(out->chunks);
155 cwist_free(out->chunk_lens);
156 memset(out, 0, sizeof(*out));
157 return false;
158 }
159
160 cwist_seq_chunk_build_header(chunk, i + 1, total, (uint16_t)plen, chunk_payload_size);
161 memcpy(chunk + CWIST_SEQ_HEADER_SIZE, data + offset, plen);
162 out->chunks[i] = chunk;
163 out->chunk_lens[i] = chunk_len;
164 }
165
166 out->count = total;
167 return true;
168}

268bool cwist_seq_assembler_feed(cwist_seq_assembler_t *a, const cwist_seq_chunk_t *chunk) {
269 if (!a || !chunk) return false;
270
271 /* Validate sequence pair. */
272 if (chunk->seq == 0 || chunk->total == 0 || chunk->total > CWIST_SEQ_MAX_CHUNKS ||
273 chunk->seq > chunk->total)
274 return false;
275 if (chunk->payload_len == 0 || chunk->chunk_size == 0) return false;
276 if (chunk->payload_len > chunk->chunk_size) return false;
277 if (chunk->seq != chunk->total && chunk->payload_len != chunk->chunk_size) return false;
278 if (!chunk->payload) return false;
279
280 if (!a->have_state) {
281 a->total = chunk->total;
282 a->chunk_size = chunk->chunk_size;
283 /* The first packet may be the short final packet. Allocate for the
284 * full layout and determine total_len only after seq == total is
285 * actually received; deriving it from arrival order was the source of
286 * false protocol errors on reordered HTTP/2/3-adjacent transports. */
287 if ((size_t)chunk->total > SIZE_MAX / (size_t)chunk->chunk_size) return false;
288 a->data_cap = (size_t)chunk->total * chunk->chunk_size;
289 size_t ceiling = a->max_data_len ? a->max_data_len : CWIST_SEQ_MAX_REASSEMBLED_SIZE;
290 if (a->data_cap > ceiling) {
291 a->data_cap = 0;
292 return false;
293 }
294 a->data = (uint8_t *)cwist_alloc(a->data_cap);
295 a->received = (bool *)cwist_alloc_array(chunk->total, sizeof(bool));
296 a->payload_lens = (uint16_t *)cwist_alloc_array(chunk->total, sizeof(uint16_t));
297 if (!a->data || !a->received || !a->payload_lens) {
298 cwist_free(a->data);
299 cwist_free(a->received);
300 cwist_free(a->payload_lens);
301 memset(a, 0, sizeof(*a));
302 return false;
303 }
304 a->have_state = true;
305 } else {
306 if (chunk->total != a->total || chunk->chunk_size != a->chunk_size) return false;
307 }
308
309 size_t index = (size_t)chunk->seq - 1;
310 size_t offset = index * a->chunk_size;
311 if (a->received[index]) {
312 /* Safe duplicates are idempotent. A different duplicate is data
313 * corruption, not a retry, and must never overwrite good bytes. */
314 bool identical = a->payload_lens[index] == chunk->payload_len &&
315 memcmp(a->data + offset, chunk->payload, chunk->payload_len) == 0;
316 if (!identical) a->contaminated = true;
317 return identical;
318 }
319
320 if (offset > a->data_cap || chunk->payload_len > a->data_cap - offset) return false;
321
322 memcpy(a->data + offset, chunk->payload, chunk->payload_len);
323 a->received[index] = true;
324 a->payload_lens[index] = chunk->payload_len;
325 if (chunk->seq == chunk->total) {
326 a->total_len = offset + chunk->payload_len;
327 }
328 a->received_count++;
329 return true;
330}

362size_t cwist_seq_assembler_recovery_targets(const cwist_seq_assembler_t *a, uint16_t *out,
363 size_t out_cap) {
