31static uint64_t cwist_db_integer_root(uint64_t value) {
32 if (value == 0) return 0;
33 uint64_t x = value;
34 uint64_t y = (x + 1) >> 1;
35 while (y < x) {
36 x = y;
37 y = (x + value / x) >> 1;
38 }
39 return x;
40}

58static bool cwist_db_match_keyword(const char *sql, size_t pos, const char *kw) {
59 size_t i = 0;
60 while (kw[i]) {
61 char current = sql[pos + i];
62 if (!current) return false;
63 if ((char)toupper((unsigned char)current) != kw[i]) {
64 return false;
65 }
66 ++i;
67 }
68 if (pos > 0) {
69 char prev = sql[pos - 1];
70 if (cwist_db_is_alpha(prev)) return false;
71 }
72 char next = sql[pos + i];
73 if (cwist_db_is_alpha(next)) return false;
74 return true;

82static cwist_db_plan_hint cwist_db_analyze_sql(const char *sql) {
83 cwist_db_plan_hint hint = {0, 0, 0};
84 if (!sql) return hint;
85 size_t len = strlen(sql);
86 uint64_t accum = 0xcbf29ce484222325ULL;
87 uint32_t depth = 0;
88 for (size_t i = 0; i < len; ++i) {
89 char upper = (char)toupper((unsigned char)sql[i]);
90 accum ^= (uint64_t)upper;
91 accum *= 0x100000001b3ULL;
92 if (upper == '(')
93 depth++;
94 else if (upper == ')' && depth > 0)
95 depth--;
96
97 if (upper == 'J' && cwist_db_match_keyword(sql, i, "JOIN")) {
98 hint.join_count++;
99 } else if (upper == 'W' && cwist_db_match_keyword(sql, i, "WHERE")) {
100 hint.predicate_count++;
101 } else if (upper == 'A' && cwist_db_match_keyword(sql, i, "AND")) {
102 hint.predicate_count++;
103 } else if (upper == 'O' && cwist_db_match_keyword(sql, i, "OR")) {
104 hint.predicate_count++;
105 }
106 }
107 hint.symbolic_weight = accum ^ ((uint64_t)depth << 48);
108 return hint;
109}

116static void cwist_db_apply_hint(sqlite3 *conn, const cwist_db_plan_hint *hint) {
117 if (!conn || !hint) return;
118 uint64_t complexity = hint->symbolic_weight + ((uint64_t)hint->join_count * 131ULL) +
119 ((uint64_t)hint->predicate_count * 53ULL) + 1ULL;
120 uint64_t root = cwist_db_integer_root(complexity);
121 int busy_ms = 50 + (int)(root % 350);
122 sqlite3_busy_timeout(conn, busy_ms);
123 if (hint->join_count >= 2) {
124 sqlite3_exec(conn, "PRAGMA temp_store = MEMORY;", NULL, NULL, NULL);
125 }
126 if (hint->predicate_count >= 3) {
127 sqlite3_exec(conn, "PRAGMA cache_spill = 0;", NULL, NULL, NULL);
128 }
129}

413/*
414 * Build an INSERT statement from a cJSON object.
415 * Returns a heap-allocated (sqlite3_malloc) SQL string, or NULL on failure.
416 * The caller must free it with sqlite3_free().
417 *
418 * String values are escaped with %Q; numbers, booleans and nulls are
419 * rendered directly. Nested objects/arrays are serialised to a JSON
420 * string so they can be stored as TEXT.
421 */
428static char *build_insert_sql(const char *table, const cJSON *obj) {
429 /* Collect column/value fragments. */
430 int count = cJSON_GetArraySize((cJSON *)obj);
431 if (count <= 0) return NULL;
432
433 /* Worst-case sizes – we'll sqlite3_mprintf each fragment individually. */
434 char *cols_buf = (char *)sqlite3_malloc(count * 64 + 16);
435 char *vals_buf = (char *)sqlite3_malloc(count * 256 + 16);
436 if (!cols_buf || !vals_buf) {
437 sqlite3_free(cols_buf);
438 sqlite3_free(vals_buf);
439 return NULL;
440 }
441 cols_buf[0] = '\0';
442 vals_buf[0] = '\0';
443
444 const cJSON *child = obj->child;
445 bool first = true;
446 while (child) {
447 /* Column name – double-quoted for safety */
448 char col_frag[128];
449 snprintf(col_frag, sizeof(col_frag), first ? "\"%s\"" : ",\"%s\"", child->string);
450 size_t cols_cap = (size_t)(count * 64 + 16);
451 size_t cols_used = strlen(cols_buf);
452 if (cols_used < cols_cap - 1) strncat(cols_buf, col_frag, cols_cap - 1 - cols_used);
453
454 /* Value */
455 char *val_frag = NULL;
456 if (cJSON_IsString(child)) {
457 val_frag = sqlite3_mprintf(first ? "%Q" : ",%Q", child->valuestring);
458 } else if (cJSON_IsNumber(child)) {
459 val_frag = sqlite3_mprintf(first ? "%g" : ",%g", child->valuedouble);
460 } else if (cJSON_IsTrue(child)) {
461 val_frag = sqlite3_mprintf(first ? "1" : ",1");
462 } else if (cJSON_IsFalse(child)) {
463 val_frag = sqlite3_mprintf(first ? "0" : ",0");
464 } else if (cJSON_IsNull(child)) {
465 val_frag = sqlite3_mprintf(first ? "NULL" : ",NULL");
466 } else {
467 /* Object / Array → store as JSON text */
468 char *serialised = cJSON_PrintUnformatted(child);
469 if (serialised) {
470 val_frag = sqlite3_mprintf(first ? "%Q" : ",%Q", serialised);
471 free(serialised);
472 }
473 }
474
475 if (!val_frag) {
476 sqlite3_free(cols_buf);
477 sqlite3_free(vals_buf);
478 return NULL;
479 }
480 size_t vals_cap = (size_t)(count * 256 + 16);
481 size_t vals_used = strlen(vals_buf);
482 if (vals_used < vals_cap - 1) strncat(vals_buf, val_frag, vals_cap - 1 - vals_used);
483 sqlite3_free(val_frag);
484
485 first = false;
486 child = child->next;
487 }
488
489 char *sql = sqlite3_mprintf("INSERT INTO \"%s\" (%s) VALUES (%s);", table, cols_buf, vals_buf);
490 sqlite3_free(cols_buf);
491 sqlite3_free(vals_buf);
492 return sql;
493}

504cwist_error_t cwist_db_insert_healed(cwist_db *db, const char *table, const char *json_str,
505 const cwist_schema_t *schema,
506 const cwist_heal_config_t *heal_cfg) {
507 cwist_error_t err = make_error(CWIST_ERR_INT16);
508 err.error.err_i16 = -1;
509
510 if (!db || !table || !json_str) return err;
511
512 /* Build effective heal config (merge schema if not already set). */
513 cwist_heal_config_t effective_cfg;
514 if (heal_cfg) {
515 effective_cfg = *heal_cfg;
516 } else {
517 memset(&effective_cfg, 0, sizeof(effective_cfg));
518 effective_cfg.threshold = 0.8;
519 }
520 if (schema && !effective_cfg.schema) effective_cfg.schema = schema;
521
522 /* Step 1: heal the JSON. */
523 cwist_heal_result_t healed = cwist_json_heal(json_str, &effective_cfg);
524 if (!healed.json) {
525 fprintf(stderr, "[CWIST-DB] insert_healed: healing failed for table '%s'\n", table);
526 cwist_heal_result_free(&healed);
527 return err;
528 }
529
530 /* Step 2: strict Zod validation if a schema is provided. */
531 if (schema) {
532 cJSON *validated = NULL;
533 cwist_zod_result_t zod = cwist_zod_parse(healed.json, schema, &validated);
534 if (!zod.valid) {
535 fprintf(stderr, "[CWIST-DB] insert_healed: Zod validation failed for table '%s':\n",
536 table);
537 cwist_zod_print_errors(&zod);
538 cwist_heal_result_free(&healed);
539 return err;
540 }
541 if (validated) cJSON_Delete(validated);
542 }
543
544 /* Step 3: parse the healed JSON and build INSERT SQL. */
545 cJSON *obj = cJSON_Parse(healed.json);
546 cwist_heal_result_free(&healed);
547 if (!obj || !cJSON_IsObject(obj)) {
548 if (obj) cJSON_Delete(obj);
549 return err;
550 }
551
552 char *sql = build_insert_sql(table, obj);
553 cJSON_Delete(obj);
554 if (!sql) return err;
555
556 /* Step 4: execute. */
557 err = cwist_db_exec(db, sql);
558 sqlite3_free(sql);
559 return err;
560}

574cwist_error_t cwist_db_query_strict(cwist_db *db, const char *sql, cJSON **result,
575 const cwist_schema_t *schema) {

577 if (!schema) return cwist_db_query(db, sql, result);
578
579 cJSON *all = NULL;
580 cwist_error_t err = cwist_db_query(db, sql, &all);
581 if (err.errtype == CWIST_ERR_JSON || !all) return err;
582
583 cJSON *conforming = cJSON_CreateArray();
584 if (!conforming) {
585 cJSON_Delete(all);
586 err.error.err_i16 = -1;
587 return err;
588 }
589
590 cJSON *row = NULL;
591 cJSON_ArrayForEach(row, all) {
592 cwist_zod_result_t zod = cwist_zod_validate(row, schema);
593 if (zod.valid) {
594 cJSON *clone = cJSON_Duplicate(row, 1);
595 if (clone) cJSON_AddItemToArray(conforming, clone);
596 }
597 }
598
599 cJSON_Delete(all);
600 *result = conforming;
601
602 err = make_error(CWIST_ERR_INT16);
603 err.error.err_i16 = 0;
604 return err;
