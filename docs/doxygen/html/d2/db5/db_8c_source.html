240static int query_callback(void *data, int argc, char **argv, char **azColName) {
241 query_context *ctx = (query_context *)data;
242 cJSON *row = cJSON_CreateObject();
243
244 for (int i = 0; i < argc; i++) {
245 // SQLite returns everything as char* by default in exec callback
246 // For simplicity, we treat everything as string or try to guess types if needed.
247 // But for a generic wrapper, string is safest unless we parse schema.
248 if (argv[i]) {

250 } else {
251 cJSON_AddNullToObject(row, azColName[i]);
252 }
253 }
254
255 cJSON_AddItemToArray(ctx->rows, row);

304/*
305 * Build an INSERT statement from a cJSON object.
306 * Returns a heap-allocated (sqlite3_malloc) SQL string, or NULL on failure.
307 * The caller must free it with sqlite3_free().
308 *
309 * String values are escaped with %Q; numbers, booleans and nulls are
310 * rendered directly. Nested objects/arrays are serialised to a JSON
311 * string so they can be stored as TEXT.
312 */
319static char *build_insert_sql(const char *table, const cJSON *obj) {
320 /* Collect column/value fragments. */
321 int count = cJSON_GetArraySize((cJSON *)obj);
322 if (count <= 0) return NULL;
323
324 /* Initial capacities; each fragment grows its buffer to the actual length. */
325 char *cols_buf = (char *)sqlite3_malloc(count * 64 + 16);
326 char *vals_buf = (char *)sqlite3_malloc(count * 256 + 16);
327 if (!cols_buf || !vals_buf) {
328 sqlite3_free(cols_buf);
329 sqlite3_free(vals_buf);
330 return NULL;
331 }
332 cols_buf[0] = '\0';
333 vals_buf[0] = '\0';
334
335 const cJSON *child = obj->child;
336 bool first = true;
337 while (child) {
338 /* Column name – double-quoted for safety */
339 char *col_frag = sqlite3_mprintf(first ? "\"%w\"" : ",\"%w\"", child->string);
340 if (!col_frag) {
341 sqlite3_free(cols_buf);
342 sqlite3_free(vals_buf);
343 return NULL;
344 }
345 size_t cols_cap = (size_t)(count * 64 + 16);
346 size_t cols_used = strlen(cols_buf);
347 if (cols_used < cols_cap - 1) strncat(cols_buf, col_frag, cols_cap - 1 - cols_used);
348 sqlite3_free(col_frag);
349
350 /* Value */
351 char *val_frag = NULL;
352 if (cJSON_IsString(child)) {
353 val_frag = sqlite3_mprintf(first ? "%Q" : ",%Q", child->valuestring);
354 } else if (cJSON_IsNumber(child)) {
355 char num[64];
356 cwist_number_format_double(num, sizeof(num), child->valuedouble);
357 val_frag = sqlite3_mprintf(first ? "%s" : ",%s", num);
358 } else if (cJSON_IsTrue(child)) {
359 val_frag = sqlite3_mprintf(first ? "1" : ",1");
360 } else if (cJSON_IsFalse(child)) {
361 val_frag = sqlite3_mprintf(first ? "0" : ",0");
362 } else if (cJSON_IsNull(child)) {
363 val_frag = sqlite3_mprintf(first ? "NULL" : ",NULL");
364 } else {
365 /* Object / Array → store as JSON text */
366 char *serialised = cJSON_PrintUnformatted(child);
367 if (serialised) {
368 val_frag = sqlite3_mprintf(first ? "%Q" : ",%Q", serialised);
369 free(serialised);
370 }
371 }
372
373 if (!val_frag) {
374 sqlite3_free(cols_buf);
375 sqlite3_free(vals_buf);
376 return NULL;
377 }
378 char *new_vals = sqlite3_mprintf("%s%s", vals_buf, val_frag);
379 if (!new_vals) {
380 sqlite3_free(val_frag);
381 sqlite3_free(cols_buf);
382 sqlite3_free(vals_buf);
383 return NULL;
384 }
385 sqlite3_free(vals_buf);
386 vals_buf = new_vals;
387 sqlite3_free(val_frag);
388
389 first = false;
390 child = child->next;
391 }
392
393 char *sql = sqlite3_mprintf("INSERT INTO \"%w\" (%s) VALUES (%s);", table, cols_buf, vals_buf);
394 sqlite3_free(cols_buf);
395 sqlite3_free(vals_buf);
396 return sql;
397}

408cwist_error_t cwist_db_insert_healed(cwist_db *db, const char *table, const char *json_str,
409 const cwist_schema_t *schema,
410 const cwist_heal_config_t *heal_cfg) {
411 cwist_error_t err = make_error(CWIST_ERR_INT16);
412 err.error.err_i16 = -1;
413
414 if (!db || !table || !json_str) return err;
415
416 /* Build effective heal config (merge schema if not already set). */
417 cwist_heal_config_t effective_cfg;
418 if (heal_cfg) {
419 effective_cfg = *heal_cfg;
420 } else {
421 memset(&effective_cfg, 0, sizeof(effective_cfg));
422 effective_cfg.threshold = 0.8;
423 }
424 if (schema && !effective_cfg.schema) effective_cfg.schema = schema;
425
426 /* Step 1: heal the JSON. */
427 cwist_heal_result_t healed = cwist_json_heal(json_str, &effective_cfg);
428 if (!healed.json) {
429 fprintf(stderr, "[CWIST-DB] insert_healed: healing failed for table '%s'\n", table);
430 cwist_heal_result_free(&healed);
431 return err;
432 }
433
434 /* Step 2: strict Zod validation if a schema is provided. */
435 if (schema) {
436 cJSON *validated = NULL;
437 cwist_zod_result_t zod = cwist_zod_parse(healed.json, schema, &validated);
438 if (!zod.valid) {
439 fprintf(stderr, "[CWIST-DB] insert_healed: Zod validation failed for table '%s':\n",
440 table);
441 cwist_zod_print_errors(&zod);
442 cwist_heal_result_free(&healed);
443 return err;
444 }
445 if (validated) cJSON_Delete(validated);
446 }
447
448 /* Step 3: parse the healed JSON and build INSERT SQL. */
449 cJSON *obj = cJSON_Parse(healed.json);
450 cwist_heal_result_free(&healed);
451 if (!obj || !cJSON_IsObject(obj)) {
452 if (obj) cJSON_Delete(obj);
453 return err;
454 }
455
456 char *sql = build_insert_sql(table, obj);
457 cJSON_Delete(obj);
458 if (!sql) return err;
459
460 /* Step 4: execute. */
461 err = cwist_db_exec(db, sql);
462 sqlite3_free(sql);
463 return err;
464}

478cwist_error_t cwist_db_query_strict(cwist_db *db, const char *sql, cJSON **result,
479 const cwist_schema_t *schema) {
480 if (!result) {
481 cwist_error_t err = make_error(CWIST_ERR_INT16);
482 err.error.err_i16 = -1;
483 return err;
484 }
485 *result = NULL;
486 /* When no schema is given, behave identically to cwist_db_query(). */
487 if (!schema) return cwist_db_query(db, sql, result);
488
489 cJSON *all = NULL;
490 cwist_error_t err = cwist_db_query(db, sql, &all);
491 if (err.errtype == CWIST_ERR_JSON || !all) return err;
492
493 cJSON *conforming = cJSON_CreateArray();
494 if (!conforming) {
495 cJSON_Delete(all);
496 err.error.err_i16 = -1;
497 return err;
498 }
499
500 cJSON *row = NULL;
501 cJSON_ArrayForEach(row, all) {
502 cwist_zod_result_t zod = cwist_zod_validate(row, schema);
503 if (zod.valid) {
504 cJSON *clone = cJSON_Duplicate(row, 1);
505 if (clone) cJSON_AddItemToArray(conforming, clone);
506 }
507 }
508
509 cJSON_Delete(all);
510 *result = conforming;
511
512 err = make_error(CWIST_ERR_INT16);
513 err.error.err_i16 = 0;
514 return err;
515}
