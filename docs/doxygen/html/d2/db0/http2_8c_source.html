246/* --- Async defer completion queue ---
247 *
248 * Plan B for deferred responses: the worker thread that completes an async
249 * job only enqueues the finished (stream_id, req, res) node and pokes the
250 * wake fd; the connection thread drains the queue and does HPACK encoding +
251 * frame writes itself, preserving single-threaded HPACK dynamic table and
252 * flow-control integrity. refs: one held by the connection, one per
253 * in-flight cwist_async handle (acquired in cwist_async_defer). */
254typedef struct h2_async_node {
255 uint32_t stream_id;
256 cwist_http_request *req;
257 cwist_http_response *send; /* response to transmit (final_res) */
258 cwist_http_response *res; /* handler's response (request arena) */
259 bool send_owned; /* send came from respond_with */
260 struct h2_async_node *next;
261} h2_async_node;

294static cwist_h2_async_queue *cwist_h2_async_queue_create(void) {
295 cwist_h2_async_queue *q = (cwist_h2_async_queue *)cwist_alloc(sizeof(*q));
296 if (!q) return NULL;
297 memset(q, 0, sizeof(*q));
298 q->wake_rd = -1;
299 q->wake_wr = -1;
300 pthread_mutex_init(&q->mu, NULL);
301 atomic_init(&q->refs, 1);
302#ifdef __linux__
303 q->wake_rd = eventfd(0, EFD_NONBLOCK | EFD_CLOEXEC);
304 q->wake_wr = q->wake_rd;
305#else
306 int pfds[2];
307 if (pipe(pfds) == 0) {
308 for (int i = 0; i < 2; i++) {
309 int fl = fcntl(pfds[i], F_GETFL, 0);
310 if (fl >= 0) fcntl(pfds[i], F_SETFL, fl | O_NONBLOCK);
311 }
312 q->wake_rd = pfds[0];
313 q->wake_wr = pfds[1];
314 }
315#endif
316 /* Explicit queue refs can outlive the connection thread's TLS GC. */
317 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(q);
318 return q;
319}

338void cwist_h2_async_queue_release(cwist_h2_async_queue *q) {
339 if (!q) return;
340 if (atomic_fetch_sub_explicit(&q->refs, 1, memory_order_acq_rel) != 1) return;
341 if (q->wake_rd >= 0) close(q->wake_rd);
342 if (q->wake_wr >= 0 && q->wake_wr != q->wake_rd) close(q->wake_wr);
343 pthread_mutex_destroy(&q->mu);
344 cwist_free(q);
345}

347/* Worker-thread entry point: enqueue the finished exchange and wake the
348 * connection thread. After connection teardown (closed) the node is
349 * discarded without sending. */
350int cwist_h2_async_queue_enqueue(cwist_h2_async_queue *q, uint32_t stream_id,
351 cwist_http_request *req, cwist_http_response *send,
352 cwist_http_response *res, bool send_owned) {
353 if (!q) return -1;
354 h2_async_node *n = (h2_async_node *)cwist_alloc(sizeof(*n));
355 if (!n) {
356 /* Enqueue consumes the exchange even when staging fails. Its
357 * producer already relinquished ownership before calling us. */
358 if (send_owned && send && send != res) cwist_http_response_destroy(send);
359 cwist_http_response_destroy(res);
360 cwist_http_request_destroy(req);
361 return -1;
362 }
363 n->stream_id = stream_id;
364 n->req = req;
365 n->send = send;
366 n->res = res;
367 n->send_owned = send_owned;
368 n->next = NULL;
369 pthread_mutex_lock(&q->mu);
370 if (q->closed) {
371 pthread_mutex_unlock(&q->mu);
372 h2_async_node_discard(n);
373 return -1;
374 }
375 /* The connection may drain immediately after the mutex is released;
376 * producer TLS sweeping must no longer own this published node. */
377 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(n);
378 if (q->tail)
379 q->tail->next = n;
380 else
381 q->head = n;
382 q->tail = n;
383 /* Wake while holding mu: after close()+release the fds may be recycled,
384 * and closed is flipped under the same mutex before fds are closed. */
385 if (q->wake_wr >= 0) {
386 uint64_t one = 1;
387 ssize_t w = write(q->wake_wr, &one, sizeof(one));
388 (void)w; /* EAGAIN: counter already non-zero, fd stays readable */
389 }
390 pthread_mutex_unlock(&q->mu);
391 return 0;
392}

576static void h2_conn_destroy(h2_conn *hc) {
577 h2_stream *s = hc->streams;
578 while (s) {
579 h2_stream *next = s->next;
580 /* Drain before on_close: a handler thread parked in a send holds its
581 * session write mutex, which on_close needs to detach. */
582 h2_stream_drain_waiters(hc, s);
583 if (s->hook_ctx && hc->hooks && hc->hooks->on_close)
584 hc->hooks->on_close(hc->hook_ctx, s->hook_ctx);
585 if (s->req) cwist_http_request_destroy(s->req);
586 cwist_seq_assembler_destroy(s->body_assembler);
587 cwist_free(s);
588 s = next;
589 }
590 if (hc->hooks && hc->hooks->on_conn_close) hc->hooks->on_conn_close(hc->hook_ctx);
591 cwist_free(hc->cont_buf);
592 cwist_free(hc->out_buf);
593 h2_hpack_entry *he = hc->hpack_head;
594 while (he) {
595 h2_hpack_entry *next = he->next;
596 cwist_free(he->name);
597 cwist_free(he->value);
598 cwist_free(he);
599 he = next;
600 }
601 hc->hpack_head = NULL;
602 hc->hpack_size = 0;
603 h2_deferred_frame *df = hc->deferred_head;
604 while (df) {
605 h2_deferred_frame *next = df->next;
606 cwist_free(df->payload);
607 cwist_free(df);
608 df = next;
609 }
610 if (hc->out_mu_init) pthread_mutex_destroy(&hc->out_mu);
611 if (hc->fc_mu_init) {
612 pthread_cond_destroy(&hc->fc_cond);
613 pthread_mutex_destroy(&hc->fc_mu);
614 }
615 if (hc->async_q) {
616 /* Pending completions are discarded; completions arriving after the
617 * close hit queue->closed and discard themselves. The handle-held
618 * references keep the queue alive until the last worker releases. */
619 cwist_h2_async_queue_close(hc->async_q);
620 cwist_h2_async_queue_release(hc->async_q);
621 hc->async_q = NULL;
622 }

633static bool h2_conn_consume_rst_budget(h2_conn *hc) {
634 uint64_t now = h2_now_ms();
635 uint64_t elapsed = (now > hc->rst_last_refill_ms) ? (now - hc->rst_last_refill_ms) : 0;
636 if (elapsed >= 1000) {
637 hc->rst_budget = h2_max_rst_burst();
638 hc->rst_last_refill_ms = now;
639 } else if (elapsed > 0) {
640 uint32_t add = (uint32_t)((elapsed * h2_max_rst_rate()) / 1000);
641 if (add > 0) {
642 hc->rst_budget += add;
643 if (hc->rst_budget > h2_max_rst_burst()) {
644 hc->rst_budget = h2_max_rst_burst();
645 }
646 hc->rst_last_refill_ms = now;
647 }
648 }
649
650 if (hc->rst_budget == 0) {
651 return false;
652 }
653 hc->rst_budget--;
654 return true;
655}

667 if (elapsed >= 1000) {
668 hc->out_rst_budget = h2_max_rst_burst();
669 hc->out_rst_last_refill_ms = now;

674 if (hc->out_rst_budget > h2_max_rst_burst()) hc->out_rst_budget = h2_max_rst_burst();
675 hc->out_rst_last_refill_ms = now;
676 }
677 }
678 if (hc->out_rst_budget == 0) return false;
679 hc->out_rst_budget--;
680 return true;

853static int h2_read(cwist_https_connection *conn, void *buf, int len) {
854 if (conn->ssl) return SSL_read(conn->ssl, buf, len);
855 /* Cleartext connections may carry bytes the accept-path sniffer already
856 * consumed (e.g. the h2c preface and the frames pipelined with it);
857 * drain that prebuffer before touching the socket. */
858 if (conn->read_buf && conn->buf_len > 0) {
859 size_t n = conn->buf_len < (size_t)len ? conn->buf_len : (size_t)len;
860 memcpy(buf, conn->read_buf, n);
861 memmove(conn->read_buf, conn->read_buf + n, conn->buf_len - n);
862 conn->buf_len -= n;
863 return (int)n;
864 }
865 return read(conn->fd, buf, len);
866}

893/* Poll the socket for the direction OpenSSL is waiting on, or for plain
894 * socket writability. Returns 0 when ready, -1 on timeout/error. */
895static int h2_wait_socket(cwist_https_connection *conn, int ssl_error, int timeout_ms) {
896 struct pollfd pfd = {.fd = conn->fd, .events = 0};
897 if (conn->ssl) {
898 if (ssl_error == SSL_ERROR_WANT_READ) {
899 pfd.events = POLLIN;
900 } else if (ssl_error == SSL_ERROR_WANT_WRITE) {
901 pfd.events = POLLOUT;
902 } else {
903 return -1;
904 }
905 } else {
906 pfd.events = POLLOUT;
907 }
908 int ret = poll(&pfd, 1, timeout_ms);
909 if (ret <= 0) return -1;
910 if (pfd.revents & (POLLERR | POLLHUP | POLLNVAL)) return -1;
911 return 0;
912}

914/* Poll until frame bytes can be read, an async defer completion lands on the
915 * connection queue, or the connection idle deadline expires. Skips the poll
916 * when decrypted bytes are already buffered.
917 * @p deadline_ms optionally bounds the wait to an absolute monotonic
918 * timestamp (used for the post-GOAWAY grace window); 0 disables it.
919 * Returns 0 when readable, 1 when the async queue woke the wait, -1 on idle
920 * timeout, deadline expiry, or socket error. */
921static int h2_out_flush(h2_conn *hc);
922
923static int h2_wait_readable(h2_conn *hc, uint64_t deadline_ms) {
924 /* Never block with unflushed frames in the batch buffer. */
925 if (h2_out_flush(hc) != 0) return -1;
926 int afd = cwist_h2_async_queue_fd(hc->async_q);
927 while (!h2_has_buffered_input(hc->conn)) {
928 uint64_t idle_ms = (uint64_t)h2_idle_timeout_ms();
929 uint64_t now = h2_now_ms();
930 if (now - hc->last_activity >= idle_ms) return -1;
931 uint64_t idle_left = idle_ms - (now - hc->last_activity);
932 if (deadline_ms) {
933 if (now >= deadline_ms) return -1;
934 uint64_t deadline_left = deadline_ms - now;
935 if (deadline_left < idle_left) idle_left = deadline_left;
936 }
937 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
938 if (idle_left < (uint64_t)wait_ms) wait_ms = (int)idle_left;
939 struct pollfd pfd[2];
940 nfds_t nfd = 1;
941 pfd[0].fd = hc->conn->fd;
942 pfd[0].events = POLLIN;
943 pfd[0].revents = 0;
944 if (afd >= 0) {
945 pfd[1].fd = afd;
946 pfd[1].events = POLLIN;
947 pfd[1].revents = 0;
948 nfd = 2;
949 }
950 int pret = poll(pfd, nfd, wait_ms);
951 if (pret < 0) return -1;
952 if (pret > 0) {
953 if (nfd == 2 && (pfd[1].revents & POLLIN)) return 1;
954 if (pfd[0].revents & POLLIN) {
955 if (afd >= 0 && hc->async_q) {
956 /* If async queue has pending completions, drain them first */
957 pthread_mutex_lock(&hc->async_q->mu);
958 bool has_items = (hc->async_q->head != NULL);
959 pthread_mutex_unlock(&hc->async_q->mu);
960 if (has_items) return 1;
961 }
962 return 0;
963 }
964 return 0; /* error/hup on the socket: let the read path report it */
965 }
966 }
967 return 0;
968}

970/* Read exactly len bytes, tolerating SSL_ERROR_WANT_READ/WANT_WRITE (and
971 * plain EAGAIN) on the non-blocking sockets the pool hands us. Bounded by
972 * the connection idle deadline so a stalled peer cannot pin a worker.
973 * Returns 0 on success, -1 on EOF, timeout, or socket error. */
974static int h2_read_full(h2_conn *hc, void *buf, size_t len) {
975 unsigned char *p = (unsigned char *)buf;
976 size_t got = 0;
977 while (got < len) {
978 int n = h2_read(hc->conn, p + got, (int)(len - got));
979 if (n > 0) {
980 got += (size_t)n;
981 hc->last_activity = h2_now_ms();
982 continue;
983 }
984 if (n < 0) {
985 int retryable = 0;
986 int ssl_err = 0;
987 if (hc->conn->ssl) {
988 ssl_err = SSL_get_error(hc->conn->ssl, n);
989 retryable = (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE);
990 } else {
991 retryable = (errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR);
992 }
993 if (retryable) {
994 uint64_t idle_ms = (uint64_t)h2_idle_timeout_ms();
995 uint64_t now = h2_now_ms();
996 if (now - hc->last_activity >= idle_ms) return -1;
997 uint64_t left = idle_ms - (now - hc->last_activity);
998 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
999 if (left < (uint64_t)wait_ms) wait_ms = (int)left;
1000 if (hc->conn->ssl) {
1001 if (h2_wait_socket(hc->conn, ssl_err, wait_ms) != 0) return -1;
1002 } else {
1003 struct pollfd pfd = {.fd = hc->conn->fd, .events = POLLIN};
1004 if (poll(&pfd, 1, wait_ms) <= 0) return -1;
1005 }
1006 continue;
1007 }
1008 }
1009 return -1; /* EOF (0) or a real read error */
1010 }
1011 return 0;
1012}

1014static int h2_write_all(cwist_https_connection *conn, const void *buf, size_t len) {
1015 const unsigned char *p = (const unsigned char *)buf;
1016 int retries = 0;
1017 /* 100ms polls; keep the total wait under the connection idle deadline. */
1018 int max_retries = h2_idle_timeout_ms() / 100;
1019 if (max_retries > 1000) max_retries = 1000;
1020 if (max_retries < 1) max_retries = 1;
1021 while (len > 0) {
1022 int n = h2_write(conn, p, (int)len);
1023 if (n <= 0) {
1024 if (n < 0) {
1025 if (conn->ssl) {
1026 int err = SSL_get_error(conn->ssl, n);
1027 if (err == SSL_ERROR_WANT_WRITE || err == SSL_ERROR_WANT_READ) {
1028 if (++retries > max_retries) return -1; /* ~overall timeout guard */
1029 if (h2_wait_socket(conn, err, 100) != 0) return -1;
1030 continue;
1031 }
1032 } else {
1033 if (errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR) {
1034 if (++retries > max_retries) return -1;
1035 if (h2_wait_socket(conn, 0, 100) != 0) return -1;
1036 continue;
1037 }
1038 }
1039 }
1040 return -1;
1041 }
1042 retries = 0;
1043 p += n;
1044 len -= (size_t)n;
1045 }
1046 return 0;
1047}

1062/* --- Userspace frame batching ---
1063 * Frames accumulate in hc->out_buf and flush when the batch reaches the
1064 * threshold (CWIST_H2_BATCH_BYTES, default 64 KiB) or before any blocking
1065 * wait. One write/SSL_write per batch instead of two per frame. */
1066#define CWIST_H2_BATCH_DEFAULT_BYTES (64 * 1024)

1068static size_t h2_batch_threshold(void) {
1069 static size_t threshold = 0;
1070 if (threshold == 0) {
1071 const char *env = getenv("CWIST_H2_BATCH_BYTES");
1072 long v = 0;
1073 if (env && *env) {
1074 char *end = NULL;
1075 v = strtol(env, &end, 10);
1076 if (end == env || *end != '\0') v = 0;
1077 }
1078 threshold = (v >= 4096) ? (size_t)v : (size_t)CWIST_H2_BATCH_DEFAULT_BYTES;
1079 }
1080 return threshold;

1091static int h2_out_append_raw(h2_conn *hc, const void *buf, size_t len) {
1092 if (len == 0) return 0;
1093 if (hc->out_len + len > hc->out_cap) {
1094 size_t cap = hc->out_cap ? hc->out_cap : 16384;
1095 while (cap < hc->out_len + len) cap *= 2;
1096 unsigned char *nb = (unsigned char *)cwist_realloc(hc->out_buf, cap);
1097 if (!nb) return -1;
1098 hc->out_buf = nb;
1099 hc->out_cap = cap;
1100 }
1101 memcpy(hc->out_buf + hc->out_len, buf, len);
1102 hc->out_len += len;
1103 if (hc->out_len >= h2_batch_threshold()) return h2_out_flush_raw(hc);
1104 return 0;
1105}

1189 payload[2] = (unsigned char)((error_code >> 8) & 0xff);

1227/* How many bytes may be sent right now: peer window credit, throttled by
1228 * RTT pacing once the first PING sample has calibrated it. */
1229static uint32_t h2_send_allowance(h2_conn *hc, h2_stream *s, uint32_t requested) {
1230 uint32_t allowed = requested;
1231 if (allowed > hc->fc.send_window) allowed = hc->fc.send_window;
1232 if (s && allowed > s->fc.send_window) allowed = s->fc.send_window;
1233 if (s && hc->fc.rtt_initialized && allowed > 0) {
1234 size_t paced =
1235 cwist_http2_flow_control_pacing_allowance(&hc->fc, &s->fc, allowed, h2_now_us());
1236 if (paced < allowed) allowed = (uint32_t)paced;
1237 }

1239}

1241static void h2_commit_send(h2_conn *hc, h2_stream *s, uint32_t bytes) {
1242 if (s && hc->fc.rtt_initialized) {
1243 /* Allowance was already vetted by h2_send_allowance, so this always
1244 * succeeds and debits windows + pacing tokens together. */
1245 cwist_http2_flow_control_reserve_send(&hc->fc, &s->fc, bytes, h2_now_us());
1246 return;
1247 }
1248 hc->fc.send_window -= bytes;
1249 if (s) s->fc.send_window -= bytes;
1250}

1252/* Park a hook-taken handler thread until send credit may be available again,
1253 * the stream is aborted/detached, or no frame has arrived for a full idle
1254 * deadline. The dispatcher thread owns socket reads and applies WINDOW_UPDATE
1255 * credit, then signals fc_cond; the caller must NOT hold out_mu (the
1256 * dispatcher may need it to flush its own frames while we wait).
1257 * Returns 0 when credit may be available, -1 on abort, teardown, or stall
1258 * timeout. */
1259static int h2_fc_wait_credit(h2_conn *hc, h2_stream *s) {
1260 if (!hc->fc_mu_init) return -1;
1261 pthread_mutex_lock(&hc->fc_mu);
1262 s->fc_waiters++;
1263 int rc = -1;
1264 for (;;) {
1265 if (s->send_aborted || s->fc_detached) break;
1266 if (h2_send_allowance(hc, s, 1) > 0) {
1267 rc = 0;
1268 break;
1269 }
1270 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) break;
1271 struct timespec ts;
1272 clock_gettime(CLOCK_REALTIME, &ts);
1273 ts.tv_nsec += 100000000L; /* 100ms recheck: covers pacing-token refill */
1274 if (ts.tv_nsec >= 1000000000L) {
1275 ts.tv_sec++;
1276 ts.tv_nsec -= 1000000000L;
1277 }
1278 pthread_cond_timedwait(&hc->fc_cond, &hc->fc_mu, &ts);
1279 }
1280 s->fc_waiters--;
1281 /* Wake h2_stream_drain_waiters if it is waiting for us to leave. */
1282 pthread_cond_broadcast(&hc->fc_cond);
1283 pthread_mutex_unlock(&hc->fc_mu);
1284 return rc;
1285}

1287/* Send a PING carrying a monotonic timestamp; the matching ACK yields an
1288 * RTT sample that feeds the adaptive window/pacing tuning. */
1289static int h2_send_ping(h2_conn *hc) {
1290 uint64_t now = h2_now_us();
1291 unsigned char p[8];
1292 for (int i = 0; i < 8; i++) p[i] = (unsigned char)(now >> (56 - 8 * i));
1293 if (h2_write_frame(hc, CWIST_HTTP2_FRAME_PING, 0, 0, p, 8) != 0) return -1;
1294 memcpy(hc->ping_payload, p, 8);
1295 hc->ping_sent_us = now;
1296 hc->ping_outstanding = true;
1297 return 0;
1298}

1301
1302int h2_decode_integer(const unsigned char *buf, size_t len, size_t *pos, uint8_t prefix_bits,
1303 uint32_t *value) {
1304 if (*pos >= len || prefix_bits == 0 || prefix_bits > 8) return -1;
1305 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
1306 uint32_t n = buf[*pos] & mask;
1307 (*pos)++;
1308 if (n < mask) {
1309 *value = n;
1310 return 0;
1311 }
1312
1313 uint32_t m = 0;
1314 while (*pos < len) {
1315 uint8_t b = buf[(*pos)++];
1316 if (m > 28) return -1;
1317 n += (uint32_t)(b & 0x7f) << m;
1318 if ((b & 0x80) == 0) {
1319 *value = n;
1320 return 0;
1321 }
1322 m += 7;
1323 }
1324 return -1;
1325}

1477char *h2_huffman_decode(const unsigned char *src, size_t src_len, size_t *out_len) {
1478 pthread_once(&h2_huffman_once, h2_huffman_init);
1479 if (!h2_huffman_root) return NULL;
1480 size_t cap = src_len * 2 + 1;
1481 if (cap < 16) cap = 16;
1482 char *out = (char *)cwist_alloc(cap);
1483 if (!out) return NULL;
1484 size_t out_pos = 0;
1485
1486 uint32_t bit_buf = 0;
1487 int bit_cnt = 0;
1488 size_t src_idx = 0;
1489 h2_huffman_node *node = h2_huffman_root;
1490
1491 while (src_idx < src_len || bit_cnt > 0) {
1492 /* Keep the bit buffer filled with up to 24 bits. */
1493 while (bit_cnt <= 24 && src_idx < src_len) {
1494 bit_buf = (bit_buf << 8) | src[src_idx++];
1495 bit_cnt += 8;
1496 }
1497
1498 /* Speed-up: If we are at the root, try decoding 8 bits using the LUT. */
1499 if (node == h2_huffman_root && bit_cnt >= 8) {
1500 uint8_t index = (uint8_t)(bit_buf >> (bit_cnt - 8));
1501 h2_huffman_lut_entry entry = h2_huffman_root_lut[index];
1502 if (entry.symbol != -1) {
1503 if (entry.symbol == 256) {
1504 cwist_free(out);
1505 return NULL;
1506 }
1507 if (out_pos + 1 >= cap) {
1508 cap *= 2;
1509 char *tmp = (char *)cwist_realloc(out, cap);
1510 if (!tmp) {
1511 cwist_free(out);
1512 return NULL;
1513 }
1514 out = tmp;
1515 }
1516 out[out_pos++] = (char)entry.symbol;
1517 bit_cnt -= entry.bits;
1518 continue;
1519 }
1520 }
1521
1522 /* Fallback: bit-by-bit traversal. */
1523 if (bit_cnt > 0) {
1524 int bit = (bit_buf >> (bit_cnt - 1)) & 1;
1525 bit_cnt--;
1526 node = node->child[bit];
1527 if (!node) {
1528 cwist_free(out);
1529 return NULL;
1530 }
1531 if (node->is_terminal) {
1532 if (node->symbol == 256) {
1533 cwist_free(out);
1534 return NULL;
1535 }
1536 if (out_pos + 1 >= cap) {
1537 cap *= 2;
1538 char *tmp = (char *)cwist_realloc(out, cap);
1539 if (!tmp) {
1540 cwist_free(out);
1541 return NULL;
1542 }
1543 out = tmp;
1544 }
1545 out[out_pos++] = (char)node->symbol;
1546 node = h2_huffman_root;
1547 }
1548 } else {
1549 break;
1550 }
1551 }
1552
1553 if (node != h2_huffman_root) {
1554 h2_huffman_node *check = node;
1555 while (check && !check->is_terminal) {
1556 check = check->child[1];
1557 }
1558 if (!check || check->symbol != 256) {
1559 cwist_free(out);
1560 return NULL;
1561 }
1562 }
1563
1564 out[out_pos] = '\0';
1565 if (out_len) *out_len = out_pos;
1566 return out;
1567}

1613/* Evict the oldest entries (list tail) until the table fits within limit. */
1614static void h2_hpack_evict_to(h2_conn *hc, uint32_t limit) {
1615 while (hc->hpack_size > limit && hc->hpack_head) {
1616 h2_hpack_entry **pp = &hc->hpack_head;
1617 while ((*pp)->next) pp = &(*pp)->next;
1618 h2_hpack_entry *old = *pp;
1619 *pp = NULL;
1620 hc->hpack_size -= (uint32_t)old->size;
1621 cwist_free(old->name);
1622 cwist_free(old->value);
1623 cwist_free(old);
1624 }
1625}

1627/* Resolve a full HPACK index: 1..61 static, 62.. dynamic (62 = newest). */
1628static const h2_hpack_entry *h2_hpack_dynamic_get(const h2_conn *hc, uint32_t index) {
1629 size_t static_count = sizeof(cwist_http2_static_table) / sizeof(cwist_http2_static_table[0]);
1630 if (index < static_count) return NULL;
1631 uint32_t pos = index - (uint32_t)static_count + 1; /* 1-based into dynamic table */
1632 const h2_hpack_entry *e = hc->hpack_head;
1633 while (e && --pos) e = e->next;
1634 return e;
1635}

1637/* Insert a name/value pair at the head of the dynamic table, evicting as
1638 * needed. An entry larger than the capacity empties the table and is not
1639 * added (RFC 7541 section 4.4). */
1640static int h2_hpack_insert(h2_conn *hc, const char *name, const char *value) {
1641 size_t entry_size = strlen(name) + strlen(value) + 32;
1642 if (entry_size > hc->hpack_capacity) {
1643 h2_hpack_evict_to(hc, 0);
1644 return 0;
1645 }
1646 h2_hpack_entry *e = (h2_hpack_entry *)cwist_alloc(sizeof(*e));
1647 if (!e) return -1;
1648 e->name = cwist_strdup(name);
1649 e->value = cwist_strdup(value);
1650 if (!e->name || !e->value) {
1651 cwist_free(e->name);
1652 cwist_free(e->value);
1653 cwist_free(e);
1654 return -1;
1655 }
1656 e->size = entry_size;
1657 h2_hpack_evict_to(hc, hc->hpack_capacity - (uint32_t)entry_size);
1658 e->next = hc->hpack_head;
1659 hc->hpack_head = e;
1660 hc->hpack_size += (uint32_t)entry_size;
1661 return 0;
1662}

1712 st->header_count++;
1713 if (st->header_count > CWIST_HTTP2_MAX_HEADERS_PER_REQUEST) {
1714 CWIST_LOG_WARN("[h2] request exceeds maximum header count %u",
1715 (unsigned)CWIST_HTTP2_MAX_HEADERS_PER_REQUEST);
1716 return -1;
1717 }
1718
1719 if (name[0] == ':') {
1720 /* Pseudo-headers must precede all regular headers. */
1721 if (st->seen_regular) return -1;
1722 bool *flag = NULL;
1723 if (strcmp(name, ":method") == 0)
1724 flag = &st->seen_method;
1725 else if (strcmp(name, ":path") == 0)
1726 flag = &st->seen_path;
1727 else if (strcmp(name, ":scheme") == 0)
1728 flag = &st->seen_scheme;
1729 else if (strcmp(name, ":authority") == 0)
1730 flag = &st->seen_authority;
1731 if (flag) {
1732 if (*flag) return -1; /* duplicate pseudo-header */
1733 *flag = true;
1734 }
1735 if (strcmp(name, ":method") == 0) {
1736 req->method = cwist_http_string_to_method(value);
1737 } else if (strcmp(name, ":path") == 0) {
1738 h2_parse_path(req, value);
1739 } else if (strcmp(name, ":authority") == 0) {
1740 cwist_http_header_add(&req->headers, "host", value);
1741 }
1742 /* :scheme and unknown pseudo-headers are accepted but not mapped. */
1743 return 0;
1744 }
1745
1746 st->seen_regular = true;
1747 /* HTTP/2 field names are lowercase by mandate (RFC 9113 section 8.2.1);
1748 * normalize lenient peers so lookups (e.g. gRPC metadata) are uniform. */
1749 char lower_buf[256];
1750 if (strcmp(name, "host") != 0) {
1751 size_t name_len = strlen(name);
1752 int needs_lower = 0;
1753 for (size_t i = 0; i < name_len; ++i) {
1754 if (name[i] >= 'A' && name[i] <= 'Z') {
1755 needs_lower = 1;
1756 break;
1757 }
1758 }
1759 if (needs_lower && name_len < sizeof(lower_buf)) {
1760 for (size_t i = 0; i < name_len; ++i)
1761 lower_buf[i] = (char)tolower((unsigned char)name[i]);
1762 lower_buf[name_len] = '\0';
1763 name = lower_buf;
1764 }
1765 }
1766 if (strcmp(name, "host") == 0) {
1767 cwist_http_header_add(&req->headers, "host", value);
1768 } else {
1769 cwist_http_header_add(&req->headers, name, value);
1770 }
1771 return 0;
1772}

1774/* Decode result codes: OK, stream-level error (RST_STREAM/PROTOCOL_ERROR),
1775 * or compression error (connection-level, GOAWAY). */
1776#define H2_DECODE_OK 0
1777#define H2_DECODE_STREAM_ERROR 1
1778#define H2_DECODE_COMPRESSION_ERROR 2
1779
1780static int h2_decode_header_block(h2_conn *hc, cwist_http_request *req,
1781 const unsigned char *payload, size_t len, bool is_request) {
1782 size_t pos = 0;
1783 h2_header_state st;
1784 memset(&st, 0, sizeof(st));
1785 /* Pseudo-headers are forbidden outside the request header block
1786 * (e.g. trailers); seed the state so any ':' name errors out. */
1787 st.seen_regular = !is_request;
1788 /* A semantically invalid field (duplicate/misplaced pseudo-header,
1789 * missing :method/:path, ...) is a stream error, not a connection
1790 * error: the request is answered with RST_STREAM, but the HPACK
1791 * dynamic table is a connection-wide compression context shared with
1792 * the peer's encoder. Bailing out mid-block on a stream error would
1793 * skip any incremental-indexing insertions still to come and desync
1794 * the decoder's table from the encoder's, corrupting *unrelated*
1795 * requests later on the same connection. So keep decoding the full
1796 * block to keep the table in sync, and only report the stream error
1797 * once the whole block has been consumed. */
1798 bool stream_error = false;
1799
1800 while (pos < len) {
1801 uint8_t b = payload[pos];
1802 uint32_t name_index = 0;
1803 char *name = NULL;
1804 char *value = NULL;
1805
1806 if (b & 0x80) {
1807 /* Indexed header field: static table or dynamic table. */
1808 uint32_t index = 0;
1809 if (h2_decode_integer(payload, len, &pos, 7, &index) != 0)
1810 return H2_DECODE_COMPRESSION_ERROR;
1811 const cwist_http2_static_header *entry = h2_static_header(index);
1812 if (entry && entry->name) {
1813 if (h2_apply_header(req, entry->name, entry->value, &st) != 0) stream_error = true;
1814 continue;
1815 }
1816 const h2_hpack_entry *dyn = h2_hpack_dynamic_get(hc, index);
1817 if (!dyn) {
1818 CWIST_LOG_WARN("[h2] header field index=%u beyond static+dynamic tables", index);
1819 return H2_DECODE_COMPRESSION_ERROR;
1820 }
1821 if (h2_apply_header(req, dyn->name, dyn->value, &st) != 0) stream_error = true;
1822 continue;
1823 }
1824
1825 if ((b & 0xE0) == 0x20) {
1826 /* Dynamic table size update. */
1827 uint32_t new_size = 0;
1828 if (h2_decode_integer(payload, len, &pos, 5, &new_size) != 0)
1829 return H2_DECODE_COMPRESSION_ERROR;
1830 if (h2_hpack_set_capacity(hc, new_size) != 0) {
1831 CWIST_LOG_WARN("[h2] dynamic table size update %u exceeds advertised capacity",
1832 new_size);
1833 return H2_DECODE_COMPRESSION_ERROR;
1834 }
1835 continue;
1836 }
1837
1838 bool incremental_indexing = false;
1839 if ((b & 0x40) == 0x40) {
1840 incremental_indexing = true;
1841 if (h2_decode_integer(payload, len, &pos, 6, &name_index) != 0)
1842 return H2_DECODE_COMPRESSION_ERROR;
1843 } else if ((b & 0xf0) == 0x00) {
1844 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0)
1845 return H2_DECODE_COMPRESSION_ERROR;
1846 } else if ((b & 0xf0) == 0x10) {
1847 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0)
1848 return H2_DECODE_COMPRESSION_ERROR;
1849 } else {
1850 return H2_DECODE_COMPRESSION_ERROR;
1851 }
1852
1853 if (name_index > 0) {
1854 const cwist_http2_static_header *entry = h2_static_header(name_index);
1855 if (entry && entry->name) {
1856 name = cwist_strdup(entry->name);
1857 } else {
1858 const h2_hpack_entry *dyn = h2_hpack_dynamic_get(hc, name_index);
1859 if (dyn) name = cwist_strdup(dyn->name);
1860 }
1861 if (!name) {
1862 CWIST_LOG_WARN("[h2] literal field name index=%u beyond static+dynamic tables",
1863 name_index);
1864 return H2_DECODE_COMPRESSION_ERROR;
1865 }
1866 } else {
1867 name = h2_decode_string(payload, len, &pos);
1868 }
1869 value = h2_decode_string(payload, len, &pos);
1870 if (!name || !value) {
1871 cwist_free(name);
1872 cwist_free(value);
1873 return H2_DECODE_COMPRESSION_ERROR;
1874 }
1875 int rc = h2_apply_header(req, name, value, &st);
1876 /* Insert into the dynamic table regardless of rc: incremental
1877 * indexing is a compression-context effect, independent of
1878 * whether this particular field made the request semantically
1879 * invalid (see comment above stream_error). */
1880 if (incremental_indexing) {
1881 if (h2_hpack_insert(hc, name, value) != 0) rc = -2;
1882 }
1883 cwist_free(name);
1884 cwist_free(value);
1885 if (rc == -2) return H2_DECODE_COMPRESSION_ERROR;
1886 if (rc != 0) stream_error = true;
1887 }
1888
1889 if (stream_error) return H2_DECODE_STREAM_ERROR;
1890
1891 /* RFC 7540 section 8.1.2.3: request header blocks must carry :method and :path. */
1892 if (is_request && (!st.seen_method || !st.seen_path)) {
1893 CWIST_LOG_WARN("[h2] request header block missing :method or :path");
1894 return H2_DECODE_STREAM_ERROR;
1895 }
1896 return H2_DECODE_OK;
1897}

1910size_t h2_encode_integer(unsigned char *dst, size_t dst_cap, uint32_t value, uint8_t prefix_bits) {
1911 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
1912 unsigned char first = dst[0] & ~mask;
1913 if (value < mask) {
1914 dst[0] = first | (uint8_t)value;
1915 return 1;
1916 }
1917 dst[0] = first | mask;
1918 value -= mask;
1919 size_t i = 1;
1920 while (value >= 128) {
1921 if (i >= dst_cap) return 0;
1922 dst[i++] = (unsigned char)((value & 0x7F) | 0x80);
1923 value >>= 7;
1924 }
1925 if (i >= dst_cap) return 0;
1926 dst[i++] = (unsigned char)value;
1927 return i;
1928}

1938size_t h2_encode_string(unsigned char *dst, size_t dst_cap, const char *str) {
1939 size_t len = strlen(str);
1940 dst[0] = 0x00;

1942 if (n == 0 || n + len > dst_cap) return 0;
1943 memcpy(dst + n, str, len);
1944 return n + len;

1985static size_t h2_encode_response_headers(cwist_http_response *res, unsigned char *dst,
1986 size_t dst_cap, bool grpc_mode) {
1987 size_t pos = 0;
1988 const bool bodyless = h2_status_forbids_body(res->status_code);
1989
1990 switch (res->status_code) {
1991 case 200:
1992 if (pos >= dst_cap) return 0;
1993 dst[pos++] = 0x88;
1994 break;
1995 case 204:
1996 if (pos >= dst_cap) return 0;
1997 dst[pos++] = 0x89;
1998 break;
1999 case 206:
2000 if (pos >= dst_cap) return 0;
2001 dst[pos++] = 0x8a;
2002 break;
2003 case 304:
2004 if (pos >= dst_cap) return 0;
2005 dst[pos++] = 0x8b;
2006 break;
2007 case 400:
2008 if (pos >= dst_cap) return 0;
2009 dst[pos++] = 0x8c;
2010 break;
2011 case 404:
2012 if (pos >= dst_cap) return 0;
2013 dst[pos++] = 0x8d;
2014 break;
2015 case 500:
2016 if (pos >= dst_cap) return 0;
2017 dst[pos++] = 0x8e;
2018 break;
2019 default: {
2020 char status_str[16];
2021 snprintf(status_str, sizeof(status_str), "%d", res->status_code);
2022 if (pos + 1 > dst_cap) return 0;
2023 dst[pos] = 0x00;
2024 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
2025 if (n == 0) return 0;
2026 pos += n;
2027 n = h2_encode_string(dst + pos, dst_cap - pos, ":status");
2028 if (n == 0) return 0;
2029 pos += n;
2030 n = h2_encode_string(dst + pos, dst_cap - pos, status_str);
2031 if (n == 0) return 0;
2032 pos += n;
2033 break;
2034 }
2035 }
2036
2037 size_t body_len = 0;
2038 if (res->use_file_stream)
2039 body_len = res->file_stream_len;
2040 else if (res->is_ptr_body)
2041 body_len = res->ptr_body_len;
2042 else if (res->body)
2043 body_len = res->body->size;
2044
2045 if (!grpc_mode && !bodyless) {
2046 char cl_str[32];
2047 snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
2048 int name_idx = h2_static_table_find_name("content-length");
2049 if (name_idx > 0) {
2050 if (pos + 1 > dst_cap) return 0;
2051 dst[pos] = 0x00;
2052 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, (uint32_t)name_idx, 4);
2053 if (n == 0) return 0;
2054 pos += n;
2055 } else {
2056 if (pos + 1 > dst_cap) return 0;
2057 dst[pos] = 0x00;
2058 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
2059 if (n == 0) return 0;
2060 pos += n;
2061 n = h2_encode_string(dst + pos, dst_cap - pos, "content-length");
2062 if (n == 0) return 0;
2063 pos += n;
2064 }
2065 size_t n = h2_encode_string(dst + pos, dst_cap - pos, cl_str);
2066 if (n == 0) return 0;
2067 pos += n;
2068 }
2069
2070 cwist_http_header_node *curr = res->headers;
2071 while (curr) {
2072 if (!curr->key || !curr->key->data || !curr->value || !curr->value->data) {
2073 curr = curr->next;
2074 continue;
2075 }
2076 /* RFC 9113: skip content-length (already encoded from body_len) */
2077 if (strcasecmp(curr->key->data, "content-length") == 0) {
2078 curr = curr->next;
2079 continue;
2080 }
2081 /* RFC 9113 section 8.2.2: connection-specific fields are malformed in
2082 * HTTP/2; te is only allowed with the value "trailers". */
2083 if (strcasecmp(curr->key->data, "connection") == 0 ||
2084 strcasecmp(curr->key->data, "keep-alive") == 0 ||
2085 strcasecmp(curr->key->data, "proxy-connection") == 0 ||
2086 strcasecmp(curr->key->data, "transfer-encoding") == 0 ||
2087 strcasecmp(curr->key->data, "upgrade") == 0 ||
2088 (strcasecmp(curr->key->data, "te") == 0 &&
2089 strcasecmp(curr->value->data, "trailers") != 0)) {
2090 curr = curr->next;
2091 continue;
2092 }
2093 if (bodyless && strcasecmp(curr->key->data, "content-length") == 0) {
2094 curr = curr->next;
2095 continue;
2096 }
2097 /* gRPC status travels in the trailer HEADERS frame, not here. */
2098 if (grpc_mode && (strcasecmp(curr->key->data, "grpc-status") == 0 ||

2101 curr = curr->next;
2102 continue;
2103 }
2104 /* NUL/CR/LF in a field value makes the response malformed; drop the
2105 * offending field instead of poisoning the whole response. */
2106 if (strpbrk(curr->value->data, "\r\n") || strlen(curr->value->data) != curr->value->size) {
2107 curr = curr->next;
2108 continue;
2109 }
2110
2111 char lower_name[256];
2112 size_t name_len = strlen(curr->key->data);
2113 if (name_len >= sizeof(lower_name)) name_len = sizeof(lower_name) - 1;
2114 for (size_t i = 0; i < name_len; ++i) {
2115 lower_name[i] = (char)tolower((unsigned char)curr->key->data[i]);
2116 }
2117 lower_name[name_len] = '\0';
2118
2119 int name_idx = h2_static_table_find_name(lower_name);
2120 if (name_idx > 0) {
2121 if (pos + 1 > dst_cap) return 0;
2122 dst[pos] = 0x00;
2123 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, (uint32_t)name_idx, 4);
2124 if (n == 0) return 0;
2125 pos += n;
2126 } else {
2127 if (pos + 1 > dst_cap) return 0;
2128 dst[pos] = 0x00;
2129 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
2130 if (n == 0) return 0;
2131 pos += n;
2132 n = h2_encode_string(dst + pos, dst_cap - pos, lower_name);
2133 if (n == 0) return 0;
2134 pos += n;
2135 }
2136 size_t n = h2_encode_string(dst + pos, dst_cap - pos, curr->value->data);
2137 if (n == 0) return 0;
2138 pos += n;
2139
2140 curr = curr->next;
2141 }
2142
2143 return pos;
2144}

2148/* Send an encoded header block as HEADERS, splitting into CONTINUATION
2149 * frames when it exceeds the peer's advertised max frame size. base_flags
2150 * (e.g. END_STREAM) applies to the HEADERS frame only. */
2151static int h2_send_header_block(h2_conn *hc, cwist_https_connection *conn, uint32_t stream_id,
2152 const unsigned char *block, size_t block_len, uint32_t max_frame,
2153 uint8_t base_flags) {
2154 size_t sent = 0;
2155 uint8_t type = CWIST_HTTP2_FRAME_HEADERS;
2156 uint8_t flags = base_flags;
2157 while (true) {
2158 size_t chunk = block_len - sent;
2159 if (chunk > max_frame) chunk = max_frame;
2160 uint8_t frame_flags = flags;
2161 if (sent + chunk == block_len) frame_flags |= CWIST_HTTP2_FLAG_END_HEADERS;
2162 if (h2_write_frame_any(hc, conn, type, frame_flags, stream_id, block + sent,
2163 (uint32_t)chunk) != 0) {
2164 return -1;
2165 }
2166 sent += chunk;
2167 if (sent == block_len) return 0;
2168 type = CWIST_HTTP2_FRAME_CONTINUATION;
2169 flags = 0; /* END_STREAM is illegal on CONTINUATION */
2170 }
2171}

2173/* Encode response headers into a heap buffer that doubles until the block
2174 * fits (up to 1 MiB). The old fixed 8 KiB stack buffer made any response
2175 * with a large header set (big cookies, JWTs) abort the whole connection,
2176 * which browsers surface as ERR_INVALID_RESPONSE. */
2177static unsigned char *h2_encode_response_block(cwist_http_response *res, size_t *out_len,
2178 bool grpc_mode) {
2179 size_t cap = 16384;
2180 unsigned char *buf = NULL;
2181 while (cap <= (1u << 20)) {
2182 unsigned char *nb = (unsigned char *)cwist_realloc(buf, cap);
2183 if (!nb) {
2184 cwist_free(buf);
2185 return NULL;
2186 }
2187 buf = nb;
2188 size_t n = h2_encode_response_headers(res, buf, cap, grpc_mode);
2189 if (n > 0) {
2190 *out_len = n;
2191 return buf;
2192 }
2193 cap *= 2;
2194 }
2195 cwist_free(buf);
2196 return NULL;
2197}

2316static int h2_read_all(h2_conn *hc, void *buf, int len) {
2317 /* h2_read_full tolerates WANT_READ on the non-blocking TLS sockets;
2318 * 0/EOF and real errors stay fatal. */
2319 if (h2_read_full(hc, buf, (size_t)len) != 0) return -1;
2320 return len;
2321}

2323/* Returns 0 normally, -1 when the peer vanished mid-frame (EOF/error).
2324 *
2325 * Two traps are handled here:
2326 * 1. poll(fd, 0) alone misses WINDOW_UPDATEs in the TLS or h2c replay
2327 * buffer. These frames can remain after all socket bytes were read.
2328 * Without the buffered-input check, the window wait loop misses them and the
2329 * connection is torn down mid-body - the "some chunks arrive, then
2330 * stuck" symptom.
2331 * 2. Frames this loop is not responsible for (HEADERS/SETTINGS of other
2332 * streams, etc.) used to be read and discarded. They are now queued so
2333 * the main frame loop can dispatch them. */
2334static int h2_process_incoming_frames_nonblocking(h2_conn *hc, h2_stream *s) {
2335 /* Never wait on the peer with unflushed frames in the batch buffer. */
2336 if (h2_out_flush(hc) != 0) return -1;
2337 struct pollfd pfd;
2338 pfd.fd = hc->conn->fd;
2339 pfd.events = POLLIN;
2340
2341 int pr = poll(&pfd, 1, 0);
2342 if (pr > 0 && (pfd.revents & (POLLHUP | POLLERR)) && !(pfd.revents & POLLIN)) {
2343 if (!h2_has_buffered_input(hc->conn)) return -1;
2344 }
2345
2346 while (h2_has_buffered_input(hc->conn) || (poll(&pfd, 1, 0) > 0 && (pfd.revents & POLLIN))) {
2347 unsigned char hdr[9];
2348 int n = h2_read_all(hc, hdr, 9);
2349 if (n != 9) return -1;
2350
2351 uint32_t len = ((uint32_t)hdr[0] << 16) | ((uint32_t)hdr[1] << 8) | hdr[2];
2352 uint8_t type = hdr[3];
2353 uint8_t flags = hdr[4];
2354 uint32_t stream_id = (((uint32_t)hdr[5] & 0x7f) << 24) | ((uint32_t)hdr[6] << 16) |
2355 ((uint32_t)hdr[7] << 8) | hdr[8];
2356
2357 unsigned char *payload = NULL;
2358 if (len > 0) {
2359 payload = (unsigned char *)cwist_alloc(len);
2360 if (!payload) return -1;
2361 int r = h2_read_all(hc, payload, len);
2362 if (r != (int)len) {
2363 cwist_free(payload);
2364 return -1;
2365 }
2366 }
2367
2368 hc->last_activity = h2_now_ms();
2369
2370 if (type == 0x08) { // WINDOW_UPDATE
2371 if (len == 4) {
2372 int32_t increment = (((int32_t)payload[0] & 0x7f) << 24) |
2373 ((int32_t)payload[1] << 16) | ((int32_t)payload[2] << 8) |
2374 (int32_t)payload[3];
2375 if (increment > 0) {
2376 if (stream_id == 0) {
2377 cwist_http2_flow_control_add_send_window(&hc->fc, (uint32_t)increment);
2378 } else if (s && stream_id == s->stream_id) {
2379 cwist_http2_stream_flow_control_add_send_window(&s->fc,
2380 (uint32_t)increment);
2381 } else {
2382 h2_stream *other = h2_stream_find(hc, stream_id);
2383 if (other)
2384 cwist_http2_stream_flow_control_add_send_window(&other->fc,
2385 (uint32_t)increment);
2386 }
2387 h2_fc_credit_signal(hc);
2388 }
2389 }
2390 } else if (type == 0x06) { // PING
2391 if ((flags & 0x01) == 0 && len == 8) { // ACK flag is 0x01
2392 if (!h2_conn_consume_ping_budget(hc)) {
2393 cwist_free(payload);
2394 return -1;
2395 }
2396 h2_write_frame(hc, 0x06, 0x01, 0, payload, 8);
2397 }
2398 } else if (type == 0x03) { // RST_STREAM
2399 if (!h2_conn_consume_rst_budget(hc)) {
2400 cwist_free(payload);
2401 return -1;
2402 }
2403 if (s && stream_id == s->stream_id) {
2404 s->send_aborted = true;
2405 }
2406 } else {
2407 /* Not ours to consume: hand the intact frame to the main loop. */
2408 h2_deferred_frame *df = (h2_deferred_frame *)cwist_alloc(sizeof(*df));
2409 if (!df) {
2410 cwist_free(payload);
2411 return -1;
2412 }
2413 memcpy(df->hdr, hdr, 9);
2414 df->payload = payload;
2415 df->next = NULL;
2416 if (hc->deferred_tail)
2417 hc->deferred_tail->next = df;
2418 else
2419 hc->deferred_head = df;
2420 hc->deferred_tail = df;
2421 continue;
2422 }
2423 cwist_free(payload);
2424 }
2425 return 0;
2426}

2431static uint16_t h2_seq_chunk_payload(h2_conn *hc, h2_stream *s, uint32_t max_frame_size) {
2432 uint16_t max_payload = CWIST_HTTP2_SEQ_CHUNK_PAYLOAD(max_frame_size);
2433 uint32_t eff_window = h2_send_allowance(hc, s, max_payload + CWIST_SEQ_HEADER_SIZE);
2434 if (eff_window <= CWIST_SEQ_HEADER_SIZE) return 0;
2435 uint32_t allowed_payload = eff_window - CWIST_SEQ_HEADER_SIZE;
2436 return (allowed_payload < (uint32_t)max_payload) ? (uint16_t)allowed_payload : max_payload;
2437}

2442static int h2_send_seq_file_body(h2_conn *hc, h2_stream *s, uint32_t stream_id, int fd,
2443 off_t offset, size_t body_len, uint32_t max_frame_size) {
2444 uint16_t chunk_payload = h2_seq_chunk_payload(hc, s, max_frame_size);
2445 while (chunk_payload == 0 && body_len > 0) {
2446 /* Window exhausted below one sequenced chunk: the HEADERS block is
2447 * already on the wire, so failing here tears down the connection
2448 * and the client sees a 0-byte body. Wait for WINDOW_UPDATE. */
2449 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
2450 if (s && s->send_aborted) return -1;
2451 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
2452 struct timespec ts = {0, 2000000};
2453 nanosleep(&ts, NULL);
2454 chunk_payload = h2_seq_chunk_payload(hc, s, max_frame_size);
2455 }
2456 if (body_len == 0) return -1;
2457
2458 uint32_t total_chunks = (uint32_t)((body_len + chunk_payload - 1) / chunk_payload);
2459 size_t remaining = body_len;
2460 off_t cur_offset = offset;
2461
2462 for (uint32_t i = 0; i < total_chunks && remaining > 0; i++) {
2463 while (hc->fc.send_window == 0 || (s && s->fc.send_window == 0)) {
2464 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
2465 if (s && s->send_aborted) return -1;
2466 if (hc->fc.send_window == 0 || (s && s->fc.send_window == 0)) {
2467 /* Give up if no frame arrived within the idle deadline. */
2468 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
2469 struct timespec ts = {0, 2000000};
2470 nanosleep(&ts, NULL);
2471 }
2472 }
2473
2474 size_t plen = remaining > chunk_payload ? chunk_payload : remaining;
2475 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + plen;
2476 if (h2_send_allowance(hc, s, (uint32_t)chunk_len) < chunk_len) {
2477 /* Window too small for a full sequenced chunk; wait for update. */
2478 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
2479 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
2480 struct timespec ts = {0, 2000000};
2481 nanosleep(&ts, NULL);
2482 i--;
2483 continue;
2484 }
2485
2486 cwist_scratch_t chunk_s CWIST_SCRATCH_DEFER = {0};
2487 unsigned char *chunk = (unsigned char *)cwist_scratch_alloc(&chunk_s, chunk_len);
2488 if (!chunk) return -1;
2489 ssize_t r = pread(fd, chunk + CWIST_SEQ_HEADER_SIZE, plen, cur_offset);
2490 if (r <= 0 || (size_t)r != plen) return -1;
2491 cwist_seq_chunk_build_header(chunk, i + 1, (uint16_t)total_chunks, (uint16_t)plen,
2492 chunk_payload);
2493 uint8_t flags = (i + 1 == total_chunks) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
2494 if (h2_write_frame(hc, CWIST_HTTP2_FRAME_DATA, flags, stream_id, chunk,
2495 (uint32_t)chunk_len) != 0) {
2496 return -1;
2497 }
2498 if (s) s->send_xor ^= h2_xor_bytes(chunk, chunk_len);
2499 h2_commit_send(hc, s, (uint32_t)chunk_len);
2500 cur_offset += (off_t)plen;
2501 remaining -= plen;
2502 }
2503 return 0;
2504}

2509static int h2_send_seq_memory_body(h2_conn *hc, h2_stream *s, uint32_t stream_id,
2510 const uint8_t *body_data, size_t body_len,
2511 uint32_t max_frame_size) {
2512 uint16_t chunk_payload = h2_seq_chunk_payload(hc, s, max_frame_size);
2513 while (chunk_payload == 0) {

2515 * already on the wire, so failing here tears down the connection
2516 * and the client sees a 0-byte body. Wait for WINDOW_UPDATE. */

2518 if (s && s->send_aborted) return -1;
2519 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;

2521 nanosleep(&ts, NULL);
2522 chunk_payload = h2_seq_chunk_payload(hc, s, max_frame_size);
2523 }
2524
2525 cwist_seq_message_t msg;
2526 if (!cwist_seq_split(body_data, body_len, chunk_payload, &msg)) return -1;
2527
2528 for (size_t i = 0; i < msg.count; i++) {

2530 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) goto fail;
2531 if (s && s->send_aborted) goto fail;
2532 if (hc->fc.send_window == 0 || (s && s->fc.send_window == 0)) {
2533 /* Give up if no frame arrived within the idle deadline. */
2534 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) goto fail;
2535 struct timespec ts = {0, 2000000};
2536 nanosleep(&ts, NULL);
2537 }
2538 }
2539
2540 size_t chunk_len = msg.chunk_lens[i];
2541 if (h2_send_allowance(hc, s, (uint32_t)chunk_len) < chunk_len) {
2542 /* Window too small for a full sequenced chunk; wait for update. */
2543 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) goto fail;
2544 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) goto fail;
2545 struct timespec ts = {0, 2000000};
2546 nanosleep(&ts, NULL);
2547 i--;
2548 continue;
2549 }
2550
2551 uint8_t flags = (i + 1 == msg.count) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
2552 if (h2_write_frame(hc, CWIST_HTTP2_FRAME_DATA, flags, stream_id, msg.chunks[i],
2553 (uint32_t)chunk_len) != 0) {
2554 goto fail;
2555 }
2556 if (s) s->send_xor ^= h2_xor_bytes(msg.chunks[i], chunk_len);
2557 h2_commit_send(hc, s, (uint32_t)chunk_len);
2558 }
2559
2560 cwist_seq_message_free(&msg);
2561 return 0;
2562
2563fail:
2564 cwist_seq_message_free(&msg);
2565 return -1;
2566}

2592static int h2_stream_write_frame_locked(h2_conn *hc, uint8_t type, uint8_t flags,
2593 uint32_t stream_id, const unsigned char *payload,

2602/* Caller holds out_mu. Encode :status plus literal header pairs into a
2603 * heap block (doubling until it fits, capped at 1 MiB). */
2604static unsigned char *h2_encode_stream_block_locked(int status, const cwist_http2_header *pairs,
2605 size_t count, size_t *out_len) {
2606 size_t cap = 1024;
2607 unsigned char *buf = NULL;
2608 while (cap <= (1u << 20)) {
2609 unsigned char *nb = (unsigned char *)cwist_realloc(buf, cap);
2610 if (!nb) {
2611 cwist_free(buf);
2612 return NULL;
2613 }
2614 buf = nb;
2615 size_t pos = 0;
2616 if (status == 200) {
2617 buf[pos++] = 0x88;
2618 } else {
2619 char status_str[16];
2620 snprintf(status_str, sizeof(status_str), "%d", status);
2621 buf[pos] = 0x00;
2622 size_t n = h2_encode_integer(buf + pos, cap - pos, 0, 4);
2623 if (n == 0) goto grow;
2624 pos += n;
2625 n = h2_encode_string(buf + pos, cap - pos, ":status");
2626 if (n == 0) goto grow;
2627 pos += n;
2628 n = h2_encode_string(buf + pos, cap - pos, status_str);
2629 if (n == 0) goto grow;
2630 pos += n;
2631 }
2632 {
2633 size_t n = h2_encode_header_pairs(buf + pos, cap - pos, pairs, count);
2634 if (n == 0 && count > 0) goto grow;
2635 pos += n;
2636 }
2637 *out_len = pos;
2638 return buf;
2639 grow:
2640 cap *= 2;
2641 }
2642 cwist_free(buf);
2643 return NULL;
2644}

2680int cwist_http2_stream_send_data(cwist_h2_stream *stream, const unsigned char *data, size_t len) {
2681 if (!stream || !stream->hc) return -1;
2682 h2_conn *hc = stream->hc;
2683 uint32_t max_frame = hc->peer_max_frame_size;
2684 if (max_frame == 0) max_frame = CWIST_HTTP2_MAX_FRAME_SIZE;
2685
2686 size_t sent = 0;
2687 int rc = 0;
2688 while (sent < len) {
2689 uint32_t chunk = (uint32_t)((len - sent) > max_frame ? max_frame : (len - sent));
2690 if (h2_send_allowance(hc, stream, chunk) == 0) {
2691 /* No peer credit: the handler thread must not read the socket
2692 * (the dispatcher owns reads), so flush whatever is batched and
2693 * wait for the dispatcher to apply WINDOW_UPDATE credit. */
2694 if (h2_out_flush(hc) != 0) {
2695 rc = -1;
2696 break;
2697 }
2698 if (h2_fc_wait_credit(hc, stream) != 0) {
2699 rc = -1;
2700 break;
2701 }
2702 }
2703 h2_out_lock(hc);
2704 /* Recheck under out_mu: another handler thread may have consumed the
2705 * credit between our wait and the lock. */
2706 uint32_t allowed = h2_send_allowance(hc, stream, chunk);
2707 if (allowed == 0) {
2708 h2_out_unlock(hc);
2709 continue;
2710 }
2711 if (h2_stream_write_frame_locked(hc, CWIST_HTTP2_FRAME_DATA, 0, stream->stream_id,
2712 data + sent, allowed) != 0) {
2713 h2_out_unlock(hc);
2714 rc = -1;
2715 break;
2716 }
2717 h2_commit_send(hc, stream, allowed);
2718 sent += allowed;
2719 if (sent >= len) rc = h2_out_flush_raw(hc);
2720 h2_out_unlock(hc);
2721 if (rc != 0) break;
2722 }
2723 return rc;
2724}

2726int cwist_http2_stream_send_trailers(cwist_h2_stream *stream, const cwist_http2_header *trailers,
2727 size_t trailer_count) {
2728 if (!stream || !stream->hc) return -1;
2729 h2_conn *hc = stream->hc;
2730 unsigned char stack_block[1024];
2731 size_t block_len =
2732 h2_encode_header_pairs(stack_block, sizeof(stack_block), trailers, trailer_count);
2733 unsigned char *block = stack_block;
2734 if (block_len == 0 && trailer_count > 0) {
2735 /* Oversized trailer set: grow on the heap. */
2736 size_t cap = 4096;
2737 block = NULL;
2738 while (cap <= (1u << 20)) {
2739 unsigned char *nb = (unsigned char *)cwist_realloc(block, cap);
2740 if (!nb) {
2741 cwist_free(block);
2742 return -1;
2743 }
2744 block = nb;
2745 block_len = h2_encode_header_pairs(block, cap, trailers, trailer_count);
2746 if (block_len > 0) break;
2747 cap *= 2;
2748 }
2749 if (block_len == 0) {
2750 cwist_free(block);
2751 return -1;
2752 }
2753 }
2754 h2_out_lock(hc);
2755 int rc = h2_stream_write_frame_locked(
2756 hc, CWIST_HTTP2_FRAME_HEADERS, CWIST_HTTP2_FLAG_END_HEADERS | CWIST_HTTP2_FLAG_END_STREAM,
2757 stream->stream_id, block, (uint32_t)block_len);
2758 if (rc == 0) rc = h2_out_flush_raw(hc);
2759 h2_out_unlock(hc);
2760 if (block != stack_block) cwist_free(block);
2761 return rc;
2762}

2879static int h2_send_response_hc(h2_conn *hc, uint32_t stream_id, cwist_http_response *res) {
2880 h2_stream *s = h2_stream_find(hc, stream_id);
2881
2882 /* gRPC responses carry grpc-status in a trailer HEADERS frame, so the
2883 * main header block and the DATA frames never set END_STREAM. */
2884 bool grpc_mode = h2_response_is_grpc(res);
2885
2886 /* Error responses go out Trailers-Only (PROTOCOL-HTTP2, gRFC A6): one
2887 * HEADERS frame with END_STREAM, so the call stays retryable. */
2888 if (grpc_mode) {
2889 const char *gs = h2_grpc_header_value(res, "grpc-status");
2890 if (gs && strcmp(gs, "0") != 0) {
2891 uint32_t tf = hc->peer_max_frame_size;
2892 if (tf == 0) tf = CWIST_HTTP2_MAX_FRAME_SIZE;
2893 return h2_send_grpc_trailers_only(hc, stream_id, res, tf);
2894 }
2895 }
2896
2897 size_t block_len = 0;
2898 unsigned char *block = h2_encode_response_block(res, &block_len, grpc_mode);
2899 if (!block) return -1;
2900
2901 size_t body_len = 0;
2902 const unsigned char *body_data = NULL;
2903 if (grpc_mode) {
2904 /* gRPC bodies are always the in-memory framed message buffer. */
2905 if (res->body) {
2906 body_len = res->body->size;
2907 body_data = (const unsigned char *)res->body->data;
2908 }
2909 } else if (res->use_file_stream) {

2911 } else if (res->is_ptr_body) {

2920 * content-length header (already encoded from the real body length for
2921 * HEAD; suppressed by the encoder for bodyless statuses) stays, but no
2922 * DATA frames may follow. */
2923 bool no_content = h2_status_forbids_body(res->status_code) ||
2924 (s && s->req && s->req->method == CWIST_HTTP_HEAD);
2925
2926 uint8_t header_flags = 0;
2927 if (body_len == 0 || no_content) {
2928 body_len = 0;
2929 if (!grpc_mode) header_flags |= CWIST_HTTP2_FLAG_END_STREAM;
2930 }

2933 if (max_frame == 0) max_frame = CWIST_HTTP2_MAX_FRAME_SIZE;

2943 }

2951 if (s && s->send_aborted) return -1;

2955 struct timespec ts = {0, 2000000};
2956 nanosleep(&ts, NULL);
2957 }

2999 struct timespec ts = {0, 2000000};

3005 uint32_t allowed = h2_send_allowance(hc, s, chunk);
3006 if (allowed == 0) {
3007 /* Pacing throttled us below one byte; flush and wait. */
3008 if (h2_out_flush(hc) != 0) return -1;

3010 nanosleep(&ts, NULL);

3041 }
3042 }

3084 if (rst_rc == -2) {
3085 *goaway_sent = true;
3086 return NULL;
3087 }
3088 *refused = true;
3089 return NULL;
3090 }

3095 h2_stream_remove(hc, stream_id);
3096 return NULL;
3097 }
3098 cwist_sstring_assign(s->req->version, "HTTP/2");
3099 s->req->stream_id = stream_id;
3100 s->req->private_data = hc->conn;
3101 s->req->https_conn = hc->conn;
3102 s->req->client_fd = hc->conn ? hc->conn->fd : -1;
3103 s->req->h2_queue = hc->async_q; /* async defer completions route here */
3104 return s;
3105}

3151static int h2_begin_headers(h2_conn *hc, uint32_t stream_id, const unsigned char *payload,
3152 size_t len, bool end_headers, bool end_stream) {
3153 if (hc->expecting_continuation) {
3154 return -1; /* PROTOCOL_ERROR: HEADERS while expecting CONTINUATION */
3155 }
3156
3157 size_t block_offset = 0;
3158 size_t block_len = len;
3159
3160 bool is_new = (h2_stream_find(hc, stream_id) == NULL);
3161
3162 if (end_headers) {
3163 h2_stream *s = h2_stream_find(hc, stream_id);
3164 if (!s) {
3165 bool refused = false, goaway_sent = false;
3166 s = h2_request_stream_create(hc, stream_id, &refused, &goaway_sent);
3167 if (!s) return goaway_sent ? -2 : (refused ? 1 : -1);
3168 }
3169 if (payload && block_len > 0) {
3170 return h2_decode_stream_headers(hc, s, payload + block_offset, block_len, is_new);
3171 }
3172 /* An empty request header block still owes :method and :path. */

3179 }

3183 (unsigned)CWIST_HTTP2_MAX_HEADER_BLOCK_SIZE);
3184 return -2;
3185 }
3186
3187 /* Refused streams must still have their CONTINUATION sequence consumed
3188 * (and their header block ignored) so the connection framing stays in
3189 * sync. */
3190 if (is_new && hc->active_streams >= CWIST_HTTP2_MAX_CONCURRENT_STREAMS) {
3191 CWIST_LOG_WARN("[h2] refusing stream %u: concurrent stream limit %u reached", stream_id,

3193 int rst_rc = h2_send_rst_stream(hc, stream_id, H2_ERR_REFUSED_STREAM);

3197 hc->cont_end_stream = end_stream;

3201 return 1;
3202 }

3204 /* Need continuation */
3205 hc->expecting_continuation = true;
3206 hc->cont_stream_id = stream_id;
3207 hc->cont_end_stream = end_stream;
3208 hc->cont_discard = false;
3209 hc->cont_frame_count = 0;

3212 hc->cont_buf = (unsigned char *)cwist_alloc(hc->cont_cap);
3213 if (!hc->cont_buf) {

3216 }

3219 }

3240static int h2_handle_continuation(h2_conn *hc, uint32_t stream_id, const unsigned char *payload,
3241 size_t len, bool end_headers) {
3242 if (!hc->expecting_continuation || hc->cont_stream_id != stream_id) {
3243 return -1; /* PROTOCOL_ERROR */
3244 }
3245
3246 hc->cont_frame_count++;
3247 if (hc->cont_frame_count > CWIST_HTTP2_MAX_CONTINUATIONS) {
3248 CWIST_LOG_WARN("[h2] excessive CONTINUATION frames (%u) received (CVE-2024-27983)",
3249 hc->cont_frame_count);
3250 return -2;
3251 }
3252
3253 if (hc->cont_len + len > CWIST_HTTP2_MAX_HEADER_BLOCK_SIZE) {
3254 CWIST_LOG_WARN("[h2] CONTINUATION accumulated header block size %zu exceeds maximum %u",
3255 hc->cont_len + len, (unsigned)CWIST_HTTP2_MAX_HEADER_BLOCK_SIZE);
3256 return -2;
3257 }
3258
3259 /* Header block of a stream we already refused: consume the frames so
3260 * framing stays in sync, but do not decode or buffer them. */
3261 if (hc->cont_discard) {
3262 if (end_headers) {
3263 hc->expecting_continuation = false;
3264 hc->cont_discard = false;
3265 hc->cont_stream_id = 0;
3266 hc->cont_frame_count = 0;
3267 }
3268 return 0;
3269 }
3270
3271 if (len > 0) {
3272 if (hc->cont_len + len > hc->cont_cap) {
3273 size_t new_cap = hc->cont_cap * 2;
3274 while (new_cap < hc->cont_len + len) new_cap *= 2;
3275 unsigned char *tmp = (unsigned char *)cwist_realloc(hc->cont_buf, new_cap);
3276 if (!tmp) return -1;
3277 hc->cont_buf = tmp;
3278 hc->cont_cap = new_cap;
3279 }
3280 memcpy(hc->cont_buf + hc->cont_len, payload, len);
3281 hc->cont_len += len;
3282 }
3283

3286 bool is_new = (s == NULL);

3289 bool refused = false, goaway_sent = false;
3290 s = h2_request_stream_create(hc, hc->cont_stream_id, &refused, &goaway_sent);
3291 if (!s) rc = goaway_sent ? -2 : (refused ? 1 : -1);
3292 }
3293 if (s && hc->cont_buf && hc->cont_len > 0) {
3294 rc = h2_decode_stream_headers(hc, s, hc->cont_buf, hc->cont_len, is_new);
3295 } else if (s && is_new) {

3298 h2_stream_remove(hc, s->stream_id);
3299 rc = (rst_rc == -2) ? -2 : 1;
3300 }
3301 hc->expecting_continuation = false;
3302 hc->cont_len = 0;
3303 hc->cont_stream_id = 0;
3304 hc->cont_frame_count = 0;
3305 return rc;
3306 }
3307 return 0;
3308}

3311
3324static int h2_handle_settings(h2_conn *hc, const unsigned char *payload, size_t len, bool ack) {
3325 if (ack) {
3326 return 0;
3327 }
3328
3329 if (len % 6 != 0) {
3330 return -1; /* FRAME_SIZE_ERROR */
3331 }
3332
3333 for (size_t i = 0; i + 6 <= len; i += 6) {
3334 uint16_t id = ((uint16_t)payload[i] << 8) | payload[i + 1];
3335 uint32_t value = ((uint32_t)payload[i + 2] << 24) | ((uint32_t)payload[i + 3] << 16) |
3336 ((uint32_t)payload[i + 4] << 8) | (uint32_t)payload[i + 5];
3337 switch (id) {
3338 case 0x1: /* SETTINGS_HEADER_TABLE_SIZE */ break;
3339 case 0x2: /* SETTINGS_ENABLE_PUSH */
3340 if (value > 1) return -1; /* PROTOCOL_ERROR */
3341 break;
3342 case 0x3: /* SETTINGS_MAX_CONCURRENT_STREAMS */
3343 hc->peer_max_concurrent_streams = value;
3344 break;
3345 case 0x4: /* SETTINGS_INITIAL_WINDOW_SIZE */
3346 if (value > 2147483647U) return -1; /* FLOW_CONTROL_ERROR */
3347 {
3348 int32_t delta = (int32_t)value - (int32_t)hc->peer_initial_window_size;
3349 hc->peer_initial_window_size = value;
3350 h2_stream *s = hc->streams;
3351 while (s) {
3352 if (delta >= 0) {
3353 uint32_t inc = (uint32_t)delta;
3354 s->fc.send_window = inc > CWIST_HTTP2_MAX_WINDOW - s->fc.send_window
3355 ? CWIST_HTTP2_MAX_WINDOW
3356 : s->fc.send_window + inc;
3357 } else {
3358 uint32_t dec = (uint32_t)(-delta);
3359 /* RFC 7540 allows the adjusted window to go
3360 * negative; clamping to 0 is the safe subset. */
3361 s->fc.send_window =
3362 dec > s->fc.send_window ? 0 : s->fc.send_window - dec;
3363 }
3364 s = s->next;
3365 }
3366 h2_fc_credit_signal(hc);
3367 }
3368 break;
3369 case 0x5: /* SETTINGS_MAX_FRAME_SIZE */
3370 if (value < 16384 || value > 16777215) return -1; /* PROTOCOL_ERROR */
3371 hc->peer_max_frame_size = value;
3372 break;
3373 case 0x6: /* SETTINGS_MAX_HEADER_LIST_SIZE */ break;
3374 }
3375 }
3376 return 0;
3377}

3474/* Drain completed deferred responses on the connection thread. For each
3475 * node: when the stream is still open the request is re-attached (HEAD
3476 * checks in the send path, and h2_stream_remove destroys it), the response
3477 * is encoded and written here — never on the worker thread — and the stream
3478 * is retired. Completions for streams the peer RST'd (already removed from
3479 * the table) are dropped without touching the socket. Returns -1 when a
3480 * send failed and the connection must be torn down. */
3481static int h2_async_drain(h2_conn *hc) {
3482 cwist_h2_async_queue *q = hc->async_q;
3483 if (!q) return 0;
3484 int afd = cwist_h2_async_queue_fd(q);
3485 if (afd >= 0) {
3486 uint64_t junk;
3487 while (read(afd, &junk, sizeof(junk)) == (ssize_t)sizeof(junk)) {
3488 }
3489 }
3490 pthread_mutex_lock(&q->mu);
3491 h2_async_node *list = q->head;
3492 q->head = q->tail = NULL;
3493 pthread_mutex_unlock(&q->mu);
3494
3495 while (list) {
3496 h2_async_node *n = list;
3497 list = n->next;
3498 h2_stream *s = h2_stream_find(hc, n->stream_id);
3499 if (!s) {
3500 h2_async_node_discard(n);
3501 continue;
3502 }
3503 s->req = NULL; /* Ownership of req/res was transferred to cwist_async */
3504 h2_inject_alt_svc(hc->conn, n->send);
3505 int rc = h2_send_response_hc(hc, n->stream_id, n->send);
3506 if (n->send_owned && n->send != n->res) cwist_http_response_destroy(n->send);
3507 cwist_http_response_destroy(n->res);
3508 cwist_http_request_destroy(n->req);
3509 h2_stream_remove(hc, n->stream_id);
3510 cwist_free(n);
3511 if (rc != 0) {
3512 while (list) {
3513 h2_async_node *next = list->next;
3514 h2_async_node_discard(list);
3515 list = next;
3516 }
3517 return -1;
3518 }
3519 }
3520 return 0;
3521}

4150 }

4161 }

4163 case CWIST_HTTP2_FRAME_GOAWAY: {
4164 connected = false;
4165 break;
4166 }
4167
4168 case CWIST_HTTP2_FRAME_WINDOW_UPDATE: {
4169 if (len != 4) {
4170 h2_send_goaway(hc, hc->last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
4171 connected = false;
4172 break;
4173 }

4175 h2_send_goaway(hc, hc->last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
4176 connected = false;
4177 }
4178 break;
4179 }
4180
4181 default:
4182 /* Unknown frames MUST be ignored (RFC 7540 Section 5.5) */
4183 break;
4184 }
4185 }
4186
4187 if (connected && !atomic_load(&g_cwist_running) && !sent_goaway) {
4188 h2_send_goaway(hc, hc->last_processed_stream_id, H2_ERR_NO_ERROR);
4189 sent_goaway = true;
4190 }
4191
4192 h2_out_flush(hc); /* best effort: last GOAWAY/batched frames */
4193 conn->proto_state = NULL;
4194 conn->proto_state_free = NULL;
4195 h2_session_free(st);
4196
4197 result = make_error(CWIST_ERR_INT16);
4198 result.error.err_i16 = 0;
4199 return result;
4200}
