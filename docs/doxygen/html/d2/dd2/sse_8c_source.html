61static int append_field(cwist_sstring *out, const char *name, const char *value) {
62 const char *line = value ? value : "";
63 do {
64 const char *end = strchr(line, '\n');
65 size_t len = end ? (size_t)(end - line) : strlen(line);
66 if (len && line[len - 1] == '\r') len--;
67 if (!sse_append(out, name, strlen(name)) || !sse_append(out, ":", 1) ||
68 (len && !sse_append(out, line, len)) || !sse_append(out, "\n", 1))
69 return -1;
70 line = end ? end + 1 : NULL;
71 } while (line);
72 return 0;

89static cwist_sstring *format_event(const char *event, const char *id, int retry_ms,
90 const char *data, int is_comment) {
91 cwist_sstring *frame = cwist_sstring_create();
92 if (!frame) return NULL;
93 int failed = is_comment ? append_field(frame, "", data) : 0;
94 if (!is_comment && id) failed |= append_field(frame, "id", id);
95 if (!is_comment && event) failed |= append_field(frame, "event", event);
96 if (!is_comment && retry_ms >= 0) {
97 char retry[32];
98 snprintf(retry, sizeof(retry), "%d", retry_ms);
99 failed |= append_field(frame, "retry", retry);
100 }
101 if (!is_comment) failed |= append_field(frame, "data", data);
102 if (failed || !sse_append(frame, "\n", 1)) {
103 cwist_sstring_destroy(frame);
104 return NULL;
105 }
106 return frame;
107}

183static int send_all(int fd, const char *data, size_t len) {
184 while (len) {
185 ssize_t n = send(fd, data, len, MSG_NOSIGNAL);
186 if (n < 0 && errno == EINTR) continue;
187 if (n <= 0) return -1;
188 data += n;
189 len -= (size_t)n;
190 }
191 return 0;
192}
