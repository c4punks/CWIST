62static int append_field(cwist_sstring *out, const char *name, const char *value) {
63 const char *line = value ? value : "";
64 do {
65 size_t len = strcspn(line, "\r\n");
66 if (!sse_append(out, name, strlen(name)) || !sse_append(out, ": ", 2) ||
67 (len && !sse_append(out, line, len)) || !sse_append(out, "\n", 1))
68 return -1;

94static cwist_sstring *format_event(const char *event, const char *id, int retry_ms,
95 const char *data, int is_comment) {
96 cwist_sstring *frame = cwist_sstring_create();
97 if (!frame) return NULL;
98 int failed = is_comment ? append_field(frame, "", data) : 0;
99 if (!is_comment && id) failed |= append_field(frame, "id", id);
100 if (!is_comment && event) failed |= append_field(frame, "event", event);
101 if (!is_comment && retry_ms >= 0) {
102 char retry[32];
103 snprintf(retry, sizeof(retry), "%d", retry_ms);
104 failed |= append_field(frame, "retry", retry);
105 }
106 if (!is_comment) failed |= append_field(frame, "data", data);
107 if (failed || !sse_append(frame, "\n", 1)) {
108 cwist_sstring_destroy(frame);
109 return NULL;
110 }
111 return frame;
112}

188static int sse_send_all(int fd, const char *data, size_t len) {
189 while (len) {
190 ssize_t n = send(fd, data, len, MSG_NOSIGNAL);
191 if (n < 0 && errno == EINTR) continue;
192 if (n <= 0) return -1;
193 data += n;
194 len -= (size_t)n;
195 }
196 return 0;
197}

199/* True when @p req arrived on a cleartext HTTP/1.x byte stream, the only
200 * transport a live stream can write to directly. On HTTPS client_fd is the
201 * TLS socket and on HTTP/2 or HTTP/3 it carries framed streams, so raw
202 * writes would corrupt the connection (#956). */
203static bool sse_req_is_plain_h1(const cwist_http_request *req) {
204 if (req->https_conn || req->h2_queue || req->stream_id != 0) return false;
205 if (req->app && req->app->use_ssl) return false;
206 if (req->version && req->version->data &&
207 (strncmp(req->version->data, "HTTP/2", 6) == 0 ||
208 strncmp(req->version->data, "HTTP/3", 6) == 0))
209 return false;
210 return true;
211}

231cwist_sse_stream_t *cwist_sse_stream_open(cwist_http_request *req) {
232 static const char headers[] =
233 "HTTP/1.1 200 OK\r\nContent-Type: text/event-stream; charset=utf-8\r\nCache-Control: no-cache\r\nX-Accel-Buffering: no\r\nConnection: keep-alive\r\n\r\n";
234 if (!req || req->client_fd < 0 || req->upgraded || !sse_req_is_plain_h1(req)) return NULL;
235 /* Allocate first: once the head is on the wire the request must be
236 * marked upgraded, or the app would write a second response after it. */
237 cwist_sse_stream_t *stream = cwist_alloc(sizeof(*stream));
238 if (!stream) return NULL;
239 if (sse_send_all(req->client_fd, headers, sizeof(headers) - 1)) {
240 cwist_free(stream);
241 return NULL;
242 }
243 stream->fd = req->client_fd;
244 atomic_init(&stream->closed, 0);
245 pthread_mutex_init(&stream->mutex, NULL);
246 req->upgraded = true;
247 return stream;
248}

263cwist_error_t cwist_sse_stream_send(cwist_sse_stream_t *stream, const char *event, const char *id,
264 int retry_ms, const char *data) {
265 if (!stream || retry_ms < -1) return sse_error(-1);
266 cwist_sstring *frame = format_event(event, id, retry_ms, data, 0);
267 if (!frame) return sse_error(-1);
268 pthread_mutex_lock(&stream->mutex);
269 int rc = atomic_load(&stream->closed) ? -1 : sse_send_all(stream->fd, frame->data, frame->size);
270 if (rc) atomic_store(&stream->closed, 1);
271 pthread_mutex_unlock(&stream->mutex);
272 cwist_sstring_destroy(frame);
