83/* --- Thread Pool for HTTPS --- */
84/* The task queue is a ring that starts small and doubles when a burst fills
85 * it, up to HTTPS_TASK_QUEUE_MAX tasks; past that, submitters block until a
86 * worker frees a slot (the old fixed-size behaviour). It used to be a static

88 * made resident in every worker process before the first connection. The
89 * ring never shrinks: a process keeps the capacity of its largest burst. */

144static inline https_pool_task_t *resize_queue(https_pool_task_t *queue, size_t new_len) {
145 https_pool_task_t *old = g_https_pool.queue;
146 size_t count = g_https_pool.count;
147 if (count > 0) {
148 /* Two contiguous copies instead of a per-slot modulo walk: this is
149 * the only part of a resize that still runs under the mutex. */
150 size_t first = g_https_pool.queue_len - g_https_pool.head;
151 if (first > count) first = count;
152 memcpy(queue, old + g_https_pool.head, first * sizeof(*queue));
153 memcpy(queue + first, old, (count - first) * sizeof(*queue));
154 }
155 g_https_pool.queue = queue;
156 g_https_pool.queue_len = new_len;
157 g_https_pool.head = 0;
158 g_https_pool.tail = count;
159 return old;
160}

174static bool https_pool_reserve_locked(void) {
175 while (g_https_pool.count >= g_https_pool.queue_len) {
176 if (g_https_pool.resizing || g_https_pool.shutdown) return false;
177 if (g_https_pool.queue_len >= HTTPS_TASK_QUEUE_MAX) return false;
178 size_t new_len = g_https_pool.queue_len ? g_https_pool.queue_len * 2 : HTTPS_TASK_QUEUE_INITIAL;
179 if (new_len > HTTPS_TASK_QUEUE_MAX) new_len = HTTPS_TASK_QUEUE_MAX;
180
181 g_https_pool.resizing = true;
182 pthread_mutex_unlock(&g_https_pool.mutex);
183 /* cwist_malloc, not cwist_alloc: every pool thread touches the ring,
184 * so it must not sit on one thread's full-GC sweep list. */
185 https_pool_task_t *queue = cwist_malloc(new_len * sizeof(*queue));
186 pthread_mutex_lock(&g_https_pool.mutex);
187 if (!queue || g_https_pool.shutdown) {
188 /* On shutdown https_pool_destroy() owns the ring; drop ours. */
189 cwist_free(queue);
190 g_https_pool.resizing = false;
191 pthread_cond_broadcast(&g_https_pool.cond_not_full);
192 return false;
193 }
194
195 /* Only this submitter grows the ring and nothing shrinks it, so
196 * queue_len is unchanged and new_len still exceeds count. */
197 https_pool_task_t *old = resize_queue(queue, new_len);
198 pthread_cond_broadcast(&g_https_pool.cond_not_full); /* slots are free now */
199 pthread_mutex_unlock(&g_https_pool.mutex);
200 cwist_free(old);
201 pthread_mutex_lock(&g_https_pool.mutex);
202 /* resizing stays set until here so https_pool_destroy() cannot tear
203 * down the mutex while this thread is still outside it. */
204 g_https_pool.resizing = false;
205 pthread_cond_broadcast(&g_https_pool.cond_not_full);
206 /* Others may have filled the new slots meanwhile: loop rechecks. */
207 }
208 return true;
209}

408/* --- Idle connection park set ---------------------------------------------
409 * Why this exists: a pool thread serving a TLS connection used to wait in
410 * poll() for that connection's next request (30 s for HTTP/1.1 keep-alive,
411 * up to the HTTP/2 idle timeout). An idle client therefore held a whole
412 * pool thread, and with one or two pool threads per worker only a few dozen
413 * TLS connections could be served at once; the rest queued behind idle
414 * ones. A pool thread with nothing to read now parks the connection here
415 * and returns. One epoll thread per process watches every parked fd and
416 * resubmits a connection to the pool when bytes arrive, so an idle
417 * connection costs memory, not a thread.
418 *
419 * Each class keeps a FIFO list; every entry of a class gets the same idle
420 * budget at park time, so the list is (nearly) ordered by deadline and
421 * expiry only looks at the head. HTTP/1.1 entries are closed on expiry,
422 * exactly as the blocking header read closed them. HTTP/2 entries are
423 * resubmitted with park_expired set so the session sends GOAWAY and keeps
424 * its grace window, as before.
425 *
426 * Parking is disabled under full GC: its allocations are owned per thread,
427 * and a parked HTTP/2 session would move between threads with them.
428 * ------------------------------------------------------------------------- */

484static void *https_park_thread(void *arg) {
485 (void)arg;
486 struct epoll_event events[1024];
487 while (1) {
488 /* epoll_fd is (re)assigned under the lock by https_park_start_locked(),
489 * so read it there together with running. */
490 pthread_mutex_lock(&g_park.lock);
491 bool running = g_park.running;
492 int epfd = g_park.epoll_fd;
493 pthread_mutex_unlock(&g_park.lock);
494 if (!running) break;
495
496 int n = epoll_wait(epfd, events, 1024, 200);
497 for (int i = 0; i < n; i++) {
498 https_parked_t *p = (https_parked_t *)events[i].data.ptr;
499 pthread_mutex_lock(&g_park.lock);
500 bool mine = p->linked;
501 if (mine) park_unlink_locked(p);
502 pthread_mutex_unlock(&g_park.lock);
503 if (!mine) continue;
504 epoll_ctl(epfd, EPOLL_CTL_DEL, p->conn->fd, NULL);
505 /* Readable or hung up: the pool thread's read reports either. */
506 park_resubmit(p->conn, false);
507 cwist_free(p);
508 }
509
510 uint64_t now = cwist_https_now_ms();
511 for (int cls = 0; cls < 2; cls++) {
512 https_parked_t *expired = NULL;
513 pthread_mutex_lock(&g_park.lock);
514 while (g_park.head[cls] && g_park.head[cls]->deadline_ms <= now) {
515 https_parked_t *p = g_park.head[cls];
516 park_unlink_locked(p);
517 p->next = expired;
518 expired = p;
519 }
520 pthread_mutex_unlock(&g_park.lock);
521 while (expired) {
522 https_parked_t *p = expired;
523 expired = p->next;
524 epoll_ctl(epfd, EPOLL_CTL_DEL, p->conn->fd, NULL);
525 if (cls == CWIST_HTTPS_PARK_HTTP1) {
526 https_park_teardown(p->conn);
527 } else {
528 park_resubmit(p->conn, true);
529 }
530 cwist_free(p);
531 }
532 }
533 }
534 return NULL;
535}

545static bool https_park_start_locked(void) {
546 pid_t pid = getpid();
547 if (g_park.running && g_park.owner == pid) return true;
548 if (g_park.closed && g_park.owner == pid) return false; /* pool is shutting down */
549 /* A forked worker inherits the parent's lists and epoll fd but not its
550 * thread; start its own set (the parent's parked connections are the
551 * parent's to serve). */
552 g_park.head[0] = g_park.head[1] = g_park.tail[0] = g_park.tail[1] = NULL;
553 g_park.running = false;
554 g_park.epoll_fd = epoll_create1(EPOLL_CLOEXEC);
555 if (g_park.epoll_fd < 0) return false;
556 g_park.running = true;
557 g_park.owner = pid;
558 if (pthread_create(&g_park.thread, NULL, https_park_thread, NULL) != 0) {
559 g_park.running = false;
560 close(g_park.epoll_fd);
561 g_park.epoll_fd = -1;
562 return false;
563 }
564 return true;
565}

597bool cwist_https_park(cwist_https_connection *conn, uint64_t idle_ms, cwist_https_park_class cls) {
598 if (!conn || conn->fd < 0 || !conn->pool_handler || !https_park_enabled()) return false;
599 if (!atomic_load(&g_cwist_running)) return false;
600 /* An idle connection with nothing buffered does not need its 16 KiB
601 * read buffer while parked; cwist_https_receive_request() allocates a
602 * fresh one when the connection is served again. Measured on the C100k
603 * gate this was ~4 KB of resident memory per idle connection. Done
604 * before the fd is armed below, while no other thread can see conn. */
605 if (conn->read_buf && conn->buf_len == 0) {
606 cwist_free(conn->read_buf);
607 conn->read_buf = NULL;
608 }
609 https_parked_t *p = (https_parked_t *)cwist_alloc(sizeof(*p));
610 if (!p) return false;
611 memset(p, 0, sizeof(*p));
612 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(p);
613 p->conn = conn;
614 p->cls = cls == CWIST_HTTPS_PARK_HTTP2 ? CWIST_HTTPS_PARK_HTTP2 : CWIST_HTTPS_PARK_HTTP1;
615 p->deadline_ms = cwist_https_now_ms() + idle_ms;
616
617 pthread_mutex_lock(&g_park.lock);
618 if (!https_park_start_locked()) {
619 pthread_mutex_unlock(&g_park.lock);
620 cwist_free(p);
621 return false;
622 }
623 p->prev = g_park.tail[p->cls];
624 if (p->prev) p->prev->next = p;
625 else g_park.head[p->cls] = p;
626 g_park.tail[p->cls] = p;
627 p->linked = true;
628 conn->park_expired = false;
629 /* Register while holding the lock: once the fd is armed the park thread
630 * may resubmit the connection, and it takes this lock first. */
631 struct epoll_event ev = {.events = EPOLLIN | EPOLLRDHUP | EPOLLONESHOT, .data.ptr = p};
632 if (epoll_ctl(g_park.epoll_fd, EPOLL_CTL_ADD, conn->fd, &ev) != 0) {
633 park_unlink_locked(p);
634 pthread_mutex_unlock(&g_park.lock);
635 cwist_free(p);
636 return false;
637 }
638 t_https_parked = true;
639 pthread_mutex_unlock(&g_park.lock);
640 return true;
641}

652 bool running = g_park.running && g_park.owner == getpid();
653 g_park.running = false;
654 if (running) g_park.closed = true;

657 pthread_join(g_park.thread, NULL);
658 for (int cls = 0; cls < 2; cls++) {

660 while (p) {

690/* Re-arm immediate ACKs on a connected socket.
691 *
692 * Why: TCP_QUICKACK is not persistent - the kernel re-enters delayed-ACK
693 * (pingpong) mode during the TLS handshake's request/response segment
694 * pattern. A client that sends its TLS Finished and then holds the first
695 * HTTP request behind Nagle (no TCP_NODELAY: stock ab, many OpenSSL-based
696 * clients) waits for the ACK of the Finished; the server's delayed ACK
697 * holds it for ~40 ms, adding a fixed ~40-50 ms to every connection and
698 * capping churn throughput at ~20 handshakes/s/thread. Re-arming around
699 * the handshake keeps the ACK of the client's final flight immediate, so
700 * the request arrives right behind the handshake instead of one delayed-ACK
701 * window later. Best-effort: non-Linux or failure is fine, this only
702 * tunes ACK timing. */
703static void cwist_tcp_quickack(int fd) {
704#if defined(__linux__) && defined(TCP_QUICKACK)
705 int one = 1;
706 (void)setsockopt(fd, IPPROTO_TCP, TCP_QUICKACK, &one, sizeof(one));
707#endif
708}

743/* --- TLS handshake shepherd ------------------------------------------------
744 * Why this exists: the pool parks one worker per connection for the whole
745 * connection lifetime. A synchronous in-worker handshake made that fatal
746 * under churn — a client that delays its ClientHello by a few seconds
747 * (routine for h2load at thousands of connections per process) parked a
748 * worker for the full 30 s poll, the pool collapsed to ~tens of
749 * handshakes/s, the accept loop fell behind, the kernel accept queue
750 * overflowed, and overflowing handshakes were dropped silently, leaving
751 * clients ESTABLISHED with no server-side socket ("phantom" connections).
752 *
753 * The shepherd owns every in-progress handshake: cwist_https_dispatch()
754 * tries SSL_accept once on a non-blocking socket; incomplete handshakes are
755 * parked in a private epoll set and retried by a shepherd thread, which also
756 * reaps connections whose handshake stalls for CWIST_HTTPS_HANDSHAKE_TIMEOUT_MS
757 * (any completed event round refreshes the budget — the deadline bounds
758 * stalls, not queueing, so healthy clients under connect bursts are never
759 * swept for waiting). Only fully established sessions enter the worker pool.
760 *
761 * The shepherd is sharded (CWIST_HTTPS_HS_MAX_SHARDS threads, hashed by fd):
762 * a single thread caps per-process handshake crypto at one core, and under
763 * C100k-scale connect bursts the backlog outlived the handshake deadline,
764 * resetting clients mid-connect ("errored" in h2load). Shards carry
765 * independent epoll sets, locks and pending lists, so throughput scales with
766 * cores exactly like the pre-shepherd pool path did.
767 *
768 * The shepherd is epoll-based and therefore Linux-only. On other platforms
769 * cwist_https_dispatch() falls back to the legacy blocking pool path. */
770

773typedef struct https_hs_pending {
774 int fd;
775 SSL *ssl;
776 cwist_https_context *ctx;
777 void (*handler)(cwist_https_connection *, void *);
778 void *user_ctx;
779 uint64_t deadline_ms;
780 struct https_hs_pending *prev, *next;
781 bool linked;
782} https_hs_pending_t;

784typedef struct https_hs_shard {
785 int epoll_fd;
786 int wakeup_rd;
787 int wakeup_wr;
788 pthread_t thread;
789 /* Read lock-free by the shepherd loop and written by start/stop on other
790 * threads, so it must stay atomic. */
791 _Atomic bool running;
792 pthread_mutex_t lock;
793 /* Pending handshakes in deadline order: every entry is (re)queued at the
794 * tail with now + CWIST_HTTPS_HANDSHAKE_TIMEOUT_MS, so the sweep only
795 * looks at the head. */
796 https_hs_pending_t *head, *tail;
797 _Atomic uint32_t pending;
798} https_hs_shard_t;
799
800/* One shepherd thread per shard; the count is fixed at start (see below). */
801#define CWIST_HTTPS_HS_MAX_SHARDS 16
802
803static https_hs_shard_t g_hs_shards[CWIST_HTTPS_HS_MAX_SHARDS];
804static long g_hs_shard_count = 0;
805static pthread_mutex_t g_hs_start_lock = PTHREAD_MUTEX_INITIALIZER;
806
807/* Both called with sh->lock held; O(1), so the shepherd's per-event and
808 * sweep costs no longer grow with the number of pending handshakes. */
809static void https_hs_link_locked(https_hs_shard_t *sh, https_hs_pending_t *p) {
810 p->next = NULL;
811 p->prev = sh->tail;
812 if (sh->tail)
813 sh->tail->next = p;
814 else
815 sh->head = p;
816 sh->tail = p;
817 p->linked = true;
818 atomic_fetch_add_explicit(&sh->pending, 1, memory_order_release);
819}
820
821static void https_hs_unlink_locked(https_hs_shard_t *sh, https_hs_pending_t *p) {
822 if (!p->linked) return;
823 if (p->prev)
824 p->prev->next = p->next;
825 else
826 sh->head = p->next;
827 if (p->next)
828 p->next->prev = p->prev;
829 else
830 sh->tail = p->prev;
831 p->prev = p->next = NULL;
832 p->linked = false;
833 atomic_fetch_sub_explicit(&sh->pending, 1, memory_order_release);
834}

836static void https_hs_abort(https_hs_shard_t *sh, https_hs_pending_t *p) {
837 if (sh->epoll_fd >= 0) epoll_ctl(sh->epoll_fd, EPOLL_CTL_DEL, p->fd, NULL);
838 SSL_free(p->ssl);
839 close(p->fd);
840 cwist_free(p);
841}

843/* Move a completed handshake into the request worker pool. */
844static void https_hs_complete(https_hs_shard_t *sh, https_hs_pending_t *p) {
845 if (sh->epoll_fd >= 0) epoll_ctl(sh->epoll_fd, EPOLL_CTL_DEL, p->fd, NULL);
846 /* The request phase (cwist_https_receive_request & friends) predates
847 * non-blocking sockets; restore blocking mode before handing over. */
848 int fl = fcntl(p->fd, F_GETFL, 0);
849 if (fl >= 0) fcntl(p->fd, F_SETFL, fl & ~O_NONBLOCK);
850 https_pool_submit_ready(p->fd, p->ssl, p->ctx, p->handler, p->user_ctx);
851 cwist_free(p);
852}

854static void *https_hs_shepherd(void *arg) {
855 https_hs_shard_t *sh = (https_hs_shard_t *)arg;
856 struct epoll_event events[512];
857 while (sh->running) {
858 int n = epoll_wait(sh->epoll_fd, events, 512, 500);
859 if (n < 0) {
860 if (errno == EINTR) continue;
861 break;
862 }
863 uint64_t now = cwist_https_now_ms();
864 for (int i = 0; i < n; i++) {
865 if (events[i].data.ptr == NULL) {
866 /* wakeup pipe: drain so writers never block */
867 char drain[256];
868 while (read(sh->wakeup_rd, drain, sizeof(drain)) > 0) { /* discard */
869 }
870 continue;
871 }
872 https_hs_pending_t *p = events[i].data.ptr;
873
874 /* Detach from the list first: every event path below either
875 * re-arms (re-add) or finishes with p freed. */
876 pthread_mutex_lock(&sh->lock);
877 bool found = p->linked;
878 https_hs_unlink_locked(sh, p);
879 pthread_mutex_unlock(&sh->lock);
880 if (!found) continue; /* already reaped by the sweeper */
881
882 int rc = SSL_accept(p->ssl);
883 cwist_tcp_quickack(p->fd);
884 if (rc > 0) {
885 https_hs_complete(sh, p);
886 continue;
887 }
888 int ssl_err = SSL_get_error(p->ssl, rc);
889 if ((ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) &&
890 now < p->deadline_ms) {
891 /* The deadline bounds *stall*, not queueing: every completed
892 * event round means the client is alive and the handshake is
893 * progressing, so refresh the budget. Under C100k-scale
894 * bursts a fixed from-accept deadline was consumed by queue
895 * delay alone and healthy handshakes were swept en masse. */
896 p->deadline_ms = now + CWIST_HTTPS_HANDSHAKE_TIMEOUT_MS;
897 struct epoll_event ev = {
898 .events = (ssl_err == SSL_ERROR_WANT_WRITE ? EPOLLOUT : EPOLLIN) | EPOLLRDHUP,
899 .data.ptr = p,
900 };
901 pthread_mutex_lock(&sh->lock);
902 https_hs_link_locked(sh, p);
903 pthread_mutex_unlock(&sh->lock);
904 if (epoll_ctl(sh->epoll_fd, EPOLL_CTL_MOD, p->fd, &ev) != 0 &&
905 epoll_ctl(sh->epoll_fd, EPOLL_CTL_ADD, p->fd, &ev) != 0) {
906 pthread_mutex_lock(&sh->lock);
907 https_hs_unlink_locked(sh, p);
908 pthread_mutex_unlock(&sh->lock);
909 https_hs_abort(sh, p);
910 }
911 continue;
912 }
913 https_hs_abort(sh, p); /* hard failure or deadline exceeded */
914 }
915
916 /* Sweep expired handshakes that never became readable; the list is
917 * in deadline order, so stop at the first live one. */
918 uint64_t sweep_now = cwist_https_now_ms();
919 pthread_mutex_lock(&sh->lock);
920 while (sh->head && sweep_now >= sh->head->deadline_ms) {
921 https_hs_pending_t *expired = sh->head;
922 https_hs_unlink_locked(sh, expired);
923 pthread_mutex_unlock(&sh->lock);
924 https_hs_abort(sh, expired);
925 pthread_mutex_lock(&sh->lock);
926 }
927 pthread_mutex_unlock(&sh->lock);
928 }
929 return NULL;
930}

932static void https_hs_shard_stop(https_hs_shard_t *sh) {
933 pthread_mutex_lock(&sh->lock);
934 if (!sh->running) {
935 pthread_mutex_unlock(&sh->lock);
936 return;
937 }
938 sh->running = false;
939 if (sh->wakeup_wr >= 0) {
940 char b = 1;
941 if (write(sh->wakeup_wr, &b, 1) < 0) { /* shutdown path */
942 }
943 }
944 pthread_mutex_unlock(&sh->lock);
945 pthread_join(sh->thread, NULL);
946
947 pthread_mutex_lock(&sh->lock);
948 https_hs_pending_t *cur = sh->head;
949 sh->head = sh->tail = NULL;
950 pthread_mutex_unlock(&sh->lock);
951 while (cur) {
952 https_hs_pending_t *next = cur->next;
953 SSL_free(cur->ssl);
954 close(cur->fd);
955 cwist_free(cur);
956 cur = next;
957 }
958 if (sh->epoll_fd >= 0) close(sh->epoll_fd);
959 if (sh->wakeup_rd >= 0) close(sh->wakeup_rd);
960 if (sh->wakeup_wr >= 0) close(sh->wakeup_wr);
961 sh->epoll_fd = -1;
962 sh->wakeup_rd = sh->wakeup_wr = -1;
963}

965int https_hs_shepherd_start(void) {
966 pthread_mutex_lock(&g_hs_start_lock);
967 if (g_hs_shard_count > 0) {
968 pthread_mutex_unlock(&g_hs_start_lock);
969 return 0;
970 }
971 /* Spread handshakes across as many shepherd threads as this process has
972 * request workers (bounded), matching the parallelism of the legacy
973 * blocking pool path without parking request workers on handshakes. */
974 long want = get_optimal_thread_count();
975 if (want < 4) want = 4;
976 if (want > CWIST_HTTPS_HS_MAX_SHARDS) want = CWIST_HTTPS_HS_MAX_SHARDS;
977 /* CWIST_HTTPS_HS_SHARDS overrides the computed count for connect-burst
978 * tuning; values outside [1, CWIST_HTTPS_HS_MAX_SHARDS] are ignored. */
979 const char *env = getenv("CWIST_HTTPS_HS_SHARDS");
980 if (env && *env) {
981 char *end = NULL;
982 long parsed = strtol(env, &end, 10);
983 if (end != env && *end == '\0' && parsed >= 1 && parsed <= CWIST_HTTPS_HS_MAX_SHARDS)
984 want = parsed;
985 }
986
987 long started = 0;
988 for (long i = 0; i < want; i++) {
989 https_hs_shard_t *sh = &g_hs_shards[i];
990 memset(sh, 0, sizeof(*sh));
991 sh->epoll_fd = -1;
992 sh->wakeup_rd = sh->wakeup_wr = -1;
993 pthread_mutex_init(&sh->lock, NULL);
994 atomic_init(&sh->pending, 0);
995
996 sh->epoll_fd = epoll_create1(0);
997 if (sh->epoll_fd < 0) break;
998 int pfd[2];
999 if (pipe(pfd) != 0) {
1000 close(sh->epoll_fd);
1001 sh->epoll_fd = -1;
1002 break;
1003 }
1004 sh->wakeup_rd = pfd[0];
1005 sh->wakeup_wr = pfd[1];
1006 fcntl(sh->wakeup_rd, F_SETFL, O_NONBLOCK);
1007 fcntl(sh->wakeup_wr, F_SETFL, O_NONBLOCK);
1008 struct epoll_event ev = {.events = EPOLLIN, .data.ptr = NULL};
1009 epoll_ctl(sh->epoll_fd, EPOLL_CTL_ADD, sh->wakeup_rd, &ev);
1010 sh->running = true;
1011 if (pthread_create(&sh->thread, NULL, https_hs_shepherd, sh) != 0) {
1012 sh->running = false;
1013 close(sh->wakeup_rd);
1014 close(sh->wakeup_wr);
1015 close(sh->epoll_fd);
1016 sh->epoll_fd = -1;
1017 sh->wakeup_rd = sh->wakeup_wr = -1;
1018 break;
1019 }
1020 started++;
1021 }
1022 if (started == 0) {
1023 pthread_mutex_unlock(&g_hs_start_lock);
1024 return -1;
1025 }
1026 g_hs_shard_count = started;
1027 pthread_mutex_unlock(&g_hs_start_lock);
1028 return 0;
1029}

1032 pthread_mutex_lock(&g_hs_start_lock);
1033 long count = g_hs_shard_count;
1034 g_hs_shard_count = 0;
1035 pthread_mutex_unlock(&g_hs_start_lock);
1036 for (long i = 0; i < count; i++) {
1037 https_hs_shard_stop(&g_hs_shards[i]);
1038 }

1056void cwist_https_dispatch(int client_fd, cwist_https_context *ctx,
1057 void (*handler)(cwist_https_connection *, void *), void *user_ctx) {
1058 if (!ctx || client_fd < 0) {
1059 if (client_fd >= 0) close(client_fd);
1060 return;
1061 }
1062 if (https_hs_shepherd_start() != 0) {
1063 /* No shepherd, no dispatch: fall back to the blocking pool path. */
1064 https_pool_submit(client_fd, ctx, handler, user_ctx);
1065 return;
1066 }
1067
1068 int fl = fcntl(client_fd, F_GETFL, 0);
1069 if (fl >= 0) fcntl(client_fd, F_SETFL, fl | O_NONBLOCK);
1070
1071 SSL *ssl = cwist_https_ssl_new(ctx);
1072 if (!ssl) {
1073 close(client_fd);
1074 return;
1075 }
1076 SSL_set_fd(ssl, client_fd);
1077
1078 https_hs_pending_t *p = cwist_alloc(sizeof(*p));
1079 if (!p) {
1080 SSL_free(ssl);
1081 close(client_fd);
1082 return;
1083 }
1084 p->fd = client_fd;
1085 p->ssl = ssl;
1086 p->ctx = ctx;
1087 p->handler = handler;
1088 p->user_ctx = user_ctx;
1089 p->deadline_ms = cwist_https_now_ms() + CWIST_HTTPS_HANDSHAKE_TIMEOUT_MS;
1090
1091 /* P2C (Power of Two Choices) Shard Selection:
1092 * Samples two shepherd shards and assigns to the one with lower pending count. */
1093 unsigned shard_idx = 0;
1094 if (g_hs_shard_count > 1) {
1095 _Atomic uint32_t pending_arr[CWIST_HTTPS_HS_MAX_SHARDS];
1096 for (long i = 0; i < g_hs_shard_count; i++) {
1097 atomic_init(&pending_arr[i],
1098 atomic_load_explicit(&g_hs_shards[i].pending, memory_order_relaxed));
1099 }
1100 shard_idx =
1101 (unsigned)cwist_sched_p2c_select_worker(pending_arr, (uint32_t)g_hs_shard_count);
1102 }
1103
1104 https_hs_shard_t *sh = &g_hs_shards[shard_idx];
1105 struct epoll_event ev = {
1106 .events = EPOLLIN | EPOLLOUT | EPOLLRDHUP | EPOLLET,
1107 .data.ptr = p,
1108 };
1109
1110 pthread_mutex_lock(&sh->lock);
1111 https_hs_link_locked(sh, p);
1112 pthread_mutex_unlock(&sh->lock);
1113
1114 if (epoll_ctl(sh->epoll_fd, EPOLL_CTL_ADD, client_fd, &ev) != 0) {
1115 pthread_mutex_lock(&sh->lock);
1116 https_hs_unlink_locked(sh, p);
1117 pthread_mutex_unlock(&sh->lock);
1118 SSL_free(ssl);
1119 close(client_fd);
1120 cwist_free(p);
1121 return;
1122 }
1123 /* Wake the shepherd so pending handshakes are processed promptly. */
1124 char b = 1;
1125 if (write(sh->wakeup_wr, &b, 1) < 0) { /* non-fatal */
1126 }
1127}

1258/* --- Shared session ticket key -------------------------------------------
1259 * cwist_app_listen() forks one worker per CPU core. BoringSSL/OpenSSL
1260 * generate their internal session-ticket keys lazily on first use, i.e.
1261 * AFTER the fork, so every worker would mint its own key and tickets issued
1262 * by one worker could never be resumed on another (SO_REUSEPORT scatters
1263 * connections across workers). Instead we generate one random key at
1264 * context creation (in the parent, pre-fork) and install an explicit
1265 * ticket-key callback; every forked worker inherits identical key material
1266 * and tickets resume regardless of which worker accepts the connection.
1267 *
1268 * Security trade-off — process-lifetime STEK (tracked in issue #98):
1269 *
1270 * The key lives for the process lifetime (no rotation), the same design
1271 * nginx uses with a static `ssl_session_ticket_key` file. That is a
1272 * deliberate choice: rotation in a pre-fork model requires either a shared
1273 * mmap region with locking (cross-process visibility) or a control socket
1274 * from a parent/daemon process — significant complexity for a server that
1275 * is typically restarted nightly anyway.
1276 *
1277 * Actual exposure with the current codebase:
1278 *
1279 * 1. 0-RTT / early-data replay does NOT apply here. CWIST never calls
1280 * SSL_CTX_set_max_early_data() or equivalent, so early data is
1281 * disabled at the library level and TLS 1.3 0-RTT is never offered.
1282 *
1283 * 2. What remains is the blast-radius argument: if the STEK leaks (memory
1284 * disclosure, core dump, cold-boot) an attacker can decrypt any
1285 * session ticket issued by this process for the rest of its uptime.
1286 * Rotation would cap the valid window to the rotation interval.
1287 *
1288 * Mitigations short of full rotation:
1289 * - Enable perfect forward secrecy by preferring ECDHE cipher suites
1290 * (already the default here via BoringSSL's suite ordering) so that
1291 * bulk traffic keys are independent of the STEK even if the STEK leaks.
1292 * - Restart the server on a short cadence (e.g. daily) if the uptime
1293 * window is a concern for your threat model; a new process generates a
1294 * fresh STEK via RAND_bytes() in cwist_tls_setup_shared_ticket_key().
1295 * - Disable ticket resumption entirely (SSL_CTX_set_options with
1296 * SSL_OP_NO_TICKET) to opt into full handshakes on every connection.
1297 *
1298 * If in-process periodic rotation is ever added, the right approach is:
1299 * a) Generate a new key on a timer (e.g. every hour).
1300 * b) Keep the *previous* key for one rotation interval so in-flight
1301 * tickets from the overlap window can still decrypt.
1302 * c) Use an atomic pointer swap or reader-writer lock to avoid a
1303 * race between the rotate timer and cwist_tls_ticket_key_cb().
1304 *
1305 * Certificate hot reload (cwist_https_reload_context()) does NOT rotate this
1306 * key: the reloaded SSL_CTX carries the same STEK so session resumption
1307 * survives the swap. See docs/decisions/0002-tls-hot-reload-stek.md.
1308 * ------------------------------------------------------------------------- */
1309typedef struct cwist_tls_ticket_key {
1310 unsigned char name[16]; /* key_name sent in the ticket */
1311 unsigned char aes_key[32]; /* AES-256-CBC encryption key */
1312 unsigned char hmac_key[32]; /* HMAC-SHA-256 MAC key */
1313} cwist_tls_ticket_key;

1315static int g_ticket_key_ex_data_idx = -1;
1316static pthread_once_t g_ticket_key_ex_data_once = PTHREAD_ONCE_INIT;
1317
1318static void cwist_tls_ticket_key_ex_data_init(void) {
1319 g_ticket_key_ex_data_idx = SSL_CTX_get_ex_new_index(0, NULL, NULL, NULL, NULL);
1320}

1322static int cwist_tls_ticket_key_cb(SSL *ssl, uint8_t *key_name, uint8_t *iv, EVP_CIPHER_CTX *ectx,
1323 HMAC_CTX *hctx, int encrypt) {
1324 SSL_CTX *ssl_ctx = ssl ? SSL_get_SSL_CTX(ssl) : NULL;
1325 const cwist_tls_ticket_key *key = NULL;
1326 if (ssl_ctx && g_ticket_key_ex_data_idx >= 0) {
1327 key = (const cwist_tls_ticket_key *)SSL_CTX_get_ex_data(ssl_ctx, g_ticket_key_ex_data_idx);
1328 }
1329 if (!key) return 0;
1330
1331 if (encrypt) {
1332 memcpy(key_name, key->name, sizeof(key->name));
1333 if (RAND_bytes(iv, 16) != 1) return 0;
1334 if (EVP_EncryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
1335 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1)
1336 return 0;
1337 return 1;
1338 }
1339
1340 if (memcmp(key_name, key->name, sizeof(key->name)) != 0) {
1341 return 0; /* Unknown key: decline the ticket, do a full handshake. */
1342 }
1343 if (EVP_DecryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
1344 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1) return 0;
1345 return 1;
1346}

1353static void cwist_tls_setup_shared_ticket_key(SSL_CTX *ssl_ctx, void **out_key) {
1354 *out_key = NULL;
1355 pthread_once(&g_ticket_key_ex_data_once, cwist_tls_ticket_key_ex_data_init);
1356 if (g_ticket_key_ex_data_idx < 0) return;
1357
1358 cwist_tls_ticket_key *key = (cwist_tls_ticket_key *)cwist_alloc(sizeof(*key));
1359 if (!key) return;
1360 if (RAND_bytes((uint8_t *)key, sizeof(*key)) != 1) {
1361 cwist_free(key);
1362 return; /* Fall back to the library-internal (per-worker) keys. */
1363 }
1364 SSL_CTX_set_ex_data(ssl_ctx, g_ticket_key_ex_data_idx, key);
1365 SSL_CTX_set_tlsext_ticket_key_cb(ssl_ctx, cwist_tls_ticket_key_cb);
1366 *out_key = key;
1367}

1388 const char *key_path,
1389 const cwist_https_options *options,
1390 cwist_app *app) {

1393 bool enable_http2 = options && (options->enable_http2 || enable_http3);

1395 // Initialize OpenSSL
1396 SSL_load_error_strings();

1402 return make_ssl_error("Unable to create SSL context");

1408 }

1410

1414 }
1415

1417 err = cwist_https_apply_http2_tls_profile(ssl_ctx);
1418 if (!cwist_error_is_ok(&err)) {
1419 SSL_CTX_free(ssl_ctx);
1420 return err;
1421 }
1422 }
1423

1425 if (SSL_CTX_use_certificate_chain_file(ssl_ctx, cert_path) <= 0) {
1426 SSL_CTX_free(ssl_ctx);
1427 return make_ssl_error("Unable to load certificate");
1428 }
1429
1430 if (cwist_tls_autoload_intermediates(ssl_ctx) < 0) {
1431 SSL_CTX_free(ssl_ctx);

1433 }
1434
1435 if (SSL_CTX_use_PrivateKey_file(ssl_ctx, key_path, SSL_FILETYPE_PEM) <= 0) {
1436 SSL_CTX_free(ssl_ctx);
1437 return make_ssl_error("Unable to load private key");
1438 }
1439
1440 // Verify key matches cert
1441 if (!SSL_CTX_check_private_key(ssl_ctx)) {
1442 SSL_CTX_free(ssl_ctx);
1443 return make_ssl_error("Private key does not match certificate");
1444 }
1445
1446 *out = ssl_ctx;
1447 err.error.err_i16 = 0;
1448 return err;
1449}

1508cwist_error_t cwist_https_reload_context(cwist_https_context *ctx, const char *cert_path,
1509 const char *key_path, cwist_app *app) {
1510 cwist_error_t err = make_error(CWIST_ERR_INT16);
1511 if (!ctx || !cert_path || !key_path) {
1512 err.error.err_i16 = -1;
1513 return err;
1514 }
1515
1516 cwist_https_options options = {.enable_http2 = ctx->http2_enabled,
1517 .enable_http3 = ctx->http3_enabled};
1518 SSL_CTX *new_ctx = NULL;
1519 err = cwist_https_build_ssl_ctx(&new_ctx, cert_path, key_path, &options, app);
1520 if (!cwist_error_is_ok(&err)) {
1521 return err;
1522 }
1523
1524 /* Carry the shared STEK over: the wrapper owns it and outlives every
1525 * SSL_CTX it is installed on (it is freed only by
1526 * cwist_https_destroy_context(), after the pool is drained). The once
1527 * gate runs here because the reloading thread may not be the one that
1528 * created the context. */
1529 pthread_once(&g_ticket_key_ex_data_once, cwist_tls_ticket_key_ex_data_init);
1530 if (ctx->ticket_key && g_ticket_key_ex_data_idx >= 0) {
1531 SSL_CTX_set_ex_data(new_ctx, g_ticket_key_ex_data_idx, ctx->ticket_key);
1532 SSL_CTX_set_tlsext_ticket_key_cb(new_ctx, cwist_tls_ticket_key_cb);
1533 } else if (!ctx->ticket_key) {
1534 /* No shared key was ever armed: do it now so the reloaded context
1535 * matches a fresh one. */
1536 cwist_tls_setup_shared_ticket_key(new_ctx, &ctx->ticket_key);
1537 }
1538
1539 SSL_CTX_set_alpn_select_cb(new_ctx, cwist_https_alpn_select_cb, (void *)ctx);
1540
1541 pthread_rwlock_wrlock(&ctx->ctx_lock);
1542 SSL_CTX *old_ctx = ctx->ctx;
1543 ctx->ctx = new_ctx;
1544 pthread_rwlock_unlock(&ctx->ctx_lock);
1545
1546 /* Drops only the listener's reference; in-flight handshakes and
1547 * established connections hold their own. */
1548 if (old_ctx) SSL_CTX_free(old_ctx);
1549
1550 err.error.err_i16 = 0;
1551 return err;
1552}

1633/* Record a freshly established TLS session: version/cipher buckets and
1634 * resumption. Called from https_wrap_established only, which both the
1635 * in-worker handshake path (cwist_https_accept) and the shepherd's
1636 * pre-handshaked dispatch path funnel through. */
1637static void https_tls_record_handshake(SSL *ssl) {
1638 atomic_fetch_add_explicit(&g_tls_handshakes_total, 1, memory_order_relaxed);
1639 if (SSL_session_reused(ssl))
1640 atomic_fetch_add_explicit(&g_tls_handshakes_resumed, 1, memory_order_relaxed);
1641
1642 switch (SSL_version(ssl)) {
1643 case TLS1_2_VERSION:
1644 atomic_fetch_add_explicit(&g_tls_handshakes_tls12, 1, memory_order_relaxed);
1645 break;
1646 case TLS1_3_VERSION:
1647 atomic_fetch_add_explicit(&g_tls_handshakes_tls13, 1, memory_order_relaxed);
1648 break;
1649 default: break;
1650 }
1651
1652 const char *cipher = SSL_get_cipher_name(ssl);
1653 if (cipher) {
1654 if (strcmp(cipher, "TLS_AES_128_GCM_SHA256") == 0)
1655 atomic_fetch_add_explicit(&g_tls_ciphers_aes128_gcm, 1, memory_order_relaxed);
1656 else if (strcmp(cipher, "TLS_AES_256_GCM_SHA384") == 0)
1657 atomic_fetch_add_explicit(&g_tls_ciphers_aes256_gcm, 1, memory_order_relaxed);
1658 else if (strcmp(cipher, "TLS_CHACHA20_POLY1305_SHA256") == 0)
1659 atomic_fetch_add_explicit(&g_tls_ciphers_chacha20, 1, memory_order_relaxed);
1660 else
1661 atomic_fetch_add_explicit(&g_tls_ciphers_other, 1, memory_order_relaxed);
1662 }
1663}

1665/* Build the connection wrapper around an already-established TLS session.
1666 * Shared by the blocking cwist_https_accept path and the shepherd's
1667 * non-blocking dispatch path. */
1668static cwist_error_t https_wrap_established(cwist_https_context *ctx, int client_fd, SSL *ssl,
1669 cwist_https_connection **conn) {
1670 cwist_error_t err = make_error(CWIST_ERR_INT16);
1671
1672 *conn = (cwist_https_connection *)cwist_alloc(sizeof(cwist_https_connection));
1673 if (!*conn) {
1674 err.error.err_i16 = -1;
1675 return err;
1676 }
1677
1678 memset(*conn, 0, sizeof(**conn));
1679 (*conn)->fd = client_fd;
1680 (*conn)->ssl = ssl;
1681 (*conn)->read_buf = cwist_alloc(CWIST_HTTP_READ_BUFFER_SIZE);
1682 if (!(*conn)->read_buf) {
1683 cwist_free(*conn);
1684 *conn = NULL;
1685 err.error.err_i16 = -1;
1686 return err;
1687 }
1688 (*conn)->buf_len = 0;
1689 (*conn)->read_buf[0] = '\0';
1690 (*conn)->negotiated_http2 = false;
1691 (*conn)->negotiated_protocol = CWIST_HTTPS_PROTOCOL_HTTP11;
1692 (*conn)->http3_enabled = ctx->http3_enabled;
1693 (*conn)->deferred = false;
1694
1695 if (cwist_full_gc_enabled()) {
1696 /* *conn and its read_buf were just handed to the connection
1697 * registry's close_fn (below) as their sole eventual releaser.
1698 * They must NOT also sit on this thread's generic cwist_alloc
1699 * pending-sweep list: that list's own thread-exit destructor runs
1700 * independently of (and in unspecified order relative to) the
1701 * connection registry's destructor, so if it fired first it would
1702 * cwist_ebr_free() *conn out from under the registry's still-
1703 * pending entry -- the registry's sweep would then read a freed
1704 * conn->ssl. Disowning here hands sole ownership to the registry,
1705 * exactly the cross-list handoff cwist_gc_scope_disown() exists
1706 * for (see io_queue.c's job-handoff use of the same call). */
1707 cwist_gc_scope_disown(*conn);
1708 cwist_gc_scope_disown((*conn)->read_buf);
1709 /* Full-GC's safety net: if this connection is never explicitly
1710 * closed (worker crashes mid-request, an error path forgets), the
1711 * owning thread's exit -- or, failing that, process exit -- still
1712 * closes it instead of leaking the fd/TLS session. Normal explicit
1713 * close (cwist_https_close_connection) untracks first, so this is
1714 * a no-op on the common path. Registering here (rather than only
1715 * in cwist_https_accept) covers the shepherd's pre-handshaked
1716 * dispatch path too, which calls this function directly. */
1717 cwist_conn_registry_track(*conn, https_close_conn_cb);
1718 }
1719
1720 const unsigned char *alpn = NULL;
1721 unsigned int alpn_len = 0;
1722 SSL_get0_alpn_selected(ssl, &alpn, &alpn_len);
1723 if (alpn && alpn_len == 2 && memcmp(alpn, "h2", 2) == 0) {
1724 (*conn)->negotiated_http2 = true;
1725 (*conn)->negotiated_protocol = CWIST_HTTPS_PROTOCOL_HTTP2;
1726 }
1727 /* h3 is QUIC-only and never negotiated over TCP TLS */
1728
1729 /* The handshake's final client flight was just ACKed; stay in quickack
1730 * mode so the request that follows (held back by a Nagle-bound client)
1731 * is ACKed immediately and the connection does not stall one
1732 * delayed-ACK window before the request phase even starts. */
1733 cwist_tcp_quickack(client_fd);
1734
1735 /* Both handshake paths funnel through here, so this is the single
1736 * choke point for TLS observability (issue #306). */
1737 https_tls_record_handshake(ssl);
1738 atomic_fetch_add_explicit(&g_tls_connections_active, 1, memory_order_relaxed);
1739
1740 err.error.err_i16 = 0;
1741 return err;
1742}

1753 https_connection_teardown((cwist_https_connection *)handle);
1754}

1756cwist_error_t cwist_https_accept(cwist_https_context *ctx, int client_fd,
1757 cwist_https_connection **conn) {
1758 cwist_error_t err = make_error(CWIST_ERR_INT16);
1759
1760 if (!ctx || client_fd < 0) {
1761 err.error.err_i16 = -1;
1762 return err;
1763 }
1764
1765 SSL *ssl = cwist_https_ssl_new(ctx);
1766 if (!ssl) {
1767 return make_ssl_error("Failed to create SSL structure");
1768 }
1769
1770 SSL_set_fd(ssl, client_fd);
1771 cwist_tcp_quickack(client_fd);
1772
1773 /* Bound the whole handshake so a client dribbling bytes cannot pin a
1774 * pool worker forever. */
1775 uint64_t handshake_deadline = cwist_https_now_ms() + CWIST_HTTPS_HANDSHAKE_TIMEOUT_MS;
1776 int rc;
1777 while ((rc = SSL_accept(ssl)) <= 0) {
1778 /* Keep immediate ACKs armed: the kernel drops out of quickack mode
1779 * as the handshake segments flow, and the delayed ACK of the
1780 * client's Finished would hold a Nagle-bound request by ~40 ms. */
1781 cwist_tcp_quickack(client_fd);
1782 int ssl_err = SSL_get_error(ssl, rc);
1783 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
1784 uint64_t now = cwist_https_now_ms();
1785 if (now >= handshake_deadline) {
1786 cwist_error_t err_obj = make_ssl_error("SSL handshake timed out");
1787 SSL_free(ssl);
1788 return err_obj;
1789 }
1790 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
1791 uint64_t remaining = handshake_deadline - now;
1792 if (remaining < (uint64_t)wait_ms) wait_ms = (int)remaining;
1793 if (cwist_ssl_wait(client_fd, ssl_err, wait_ms) != 0) {
1794 cwist_error_t err_obj = make_ssl_error("SSL handshake timed out or socket error");
1795 SSL_free(ssl);
1796 return err_obj;
1797 }
1798 continue;
1799 }
1800 cwist_error_t err_obj = make_ssl_error("SSL handshake failed");
1801 SSL_free(ssl);
1802 return err_obj;
1803 }

1837static void https_connection_teardown(cwist_https_connection *conn) {
1838 if (conn) {
1839 /* Balances the increment at wrap time; teardown is the single exit
1840 * point for every established connection (public close and the
1841 * full-GC registry sweep alike). */

1843 if (conn->proto_state && conn->proto_state_free) {

1846 }

1849 SSL_free(conn->ssl);

1853 * kernel answer with RST instead of a graceful FIN. TLS clients
1854 * routinely send their close_notify (or a pipelined next request)
1855 * that the request path never consumed, so under connection churn

1857 * kernel flushes queued response bytes on the RST (the "read

1865 /* discard */
1866 }
1867 (void)n;
1868 close(conn->fd);
1869 }
1870 cwist_free(conn->read_buf);
1871 cwist_free(conn);
1872 }

1887cwist_http_request *cwist_https_receive_request(cwist_https_connection *conn) {
1888 if (!conn || !conn->ssl) return NULL;
1889 if (!conn->read_buf) {
1890 /* Released when the connection was parked (cwist_https_park()). */
1891 conn->read_buf = cwist_alloc(CWIST_HTTP_READ_BUFFER_SIZE);
1892 if (!conn->read_buf) return NULL;
1893 /* Same ownership as at wrap time: the conn's teardown frees it, so
1894 * it must not sit on this thread's full-GC sweep list. */
1895 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(conn->read_buf);
1896 conn->buf_len = 0;
1897 conn->read_buf[0] = '\0';
1898 }
1899
1900 size_t total_received = conn->buf_len;
1901 char *header_end = NULL;
1902
1903 /* Total deadline for assembling the request headers. */
1904 uint64_t headers_deadline = cwist_https_now_ms() + CWIST_HTTP_HEADERS_TIMEOUT_MS;
1905
1906 /* Rescanning from offset 0 on every read is O(n^2) over dribbled
1907 * headers; resume 3 bytes back so a terminator split across two reads
1908 * is still found. */
1909 size_t scan_from = 0;
1910 bool deadline_grace_used = false;
1911
1912 while (1) {
1913 header_end = strstr(conn->read_buf + scan_from, "\r\n\r\n");
1914 if (header_end) break;
1915 scan_from = total_received > 3 ? total_received - 3 : 0;
1916
1917 if (total_received >= CWIST_HTTP_READ_BUFFER_SIZE - 1) {
1918 return NULL;
1919 }
1920
1921 if (SSL_pending(conn->ssl) == 0) {
1922 uint64_t now = cwist_https_now_ms();
1923 if (now >= headers_deadline) {
1924 /* A request may have landed just as the idle deadline
1925 * expired; dropping it here makes strict clients surface a
1926 * network protocol error where lenient ones silently retry.
1927 * Give the wire one non-blocking chance before closing.
1928 * Bounded to a single retry so a dribbling peer cannot pin
1929 * the worker past the deadline. */
1930 bool readable = false;
1931 if (!deadline_grace_used) {
1932 deadline_grace_used = true;
1933 struct pollfd gfd = {.fd = conn->fd, .events = POLLIN};
1934 readable = (poll(&gfd, 1, 0) > 0 && (gfd.revents & POLLIN));
1935 }
1936 if (!readable) {
1937 return NULL;
1938 }
1939 } else {
1940 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
1941 uint64_t remaining = headers_deadline - now;
1942 if (remaining < (uint64_t)wait_ms) wait_ms = (int)remaining;
1943
1944 struct pollfd pfd = {.fd = conn->fd, .events = POLLIN};
1945 int pret = poll(&pfd, 1, wait_ms);
1946 if (pret <= 0) {
1947 return NULL;
1948 }
1949 }
1950 }
1951
1952 int bytes = SSL_read(conn->ssl, conn->read_buf + total_received,
1953 (int)(CWIST_HTTP_READ_BUFFER_SIZE - 1 - total_received));
1954 if (bytes <= 0) {
1955 int ssl_err = SSL_get_error(conn->ssl, bytes);
1956 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
1957 if (cwist_ssl_wait(conn->fd, ssl_err, CWIST_HTTP_TIMEOUT_MS) != 0) return NULL;
1958 continue;
1959 }
1960 return NULL;
1961 }
1962
1963 total_received += (size_t)bytes;
1964 conn->read_buf[total_received] = '\0';
1965 }
1966
1967 cwist_http_request *req = cwist_http_parse_request(conn->read_buf);
1968 if (!req) return NULL;
1969
1970 req->client_fd = conn->fd;
1971 req->https_conn = conn;
1972
1973 size_t header_len = (header_end + 4) - conn->read_buf;
1974 size_t body_received = total_received - header_len;
1975
1976 if (req->content_length > 0) {
1977 if (req->content_length > CWIST_HTTP_MAX_BODY_SIZE) {
1978 cwist_http_request_destroy(req);
1979 return NULL;
1980 }
1981
1982 char *body = cwist_alloc(req->content_length + 1);
1983 if (!body) {
1984 cwist_http_request_destroy(req);
1985 return NULL;
1986 }
1987
1988 size_t to_copy = body_received < req->content_length ? body_received : req->content_length;
1989 memcpy(body, header_end + 4, to_copy);
1990 size_t current_body_len = to_copy;
1991
1992 /* No total cap (slow 1 GiB uploads must keep working); abort only
1993 * when no bytes arrive for a cumulative idle span. Any successful
1994 * read resets the idle clock. */
1995 uint64_t body_idle_start = cwist_https_now_ms();
1996
1997 while (current_body_len < req->content_length) {
1998 if (SSL_pending(conn->ssl) == 0) {
1999 uint64_t now = cwist_https_now_ms();
2000 if (now - body_idle_start >= CWIST_HTTP_BODY_IDLE_TIMEOUT_MS) {
2001 cwist_free(body);
2002 cwist_http_request_destroy(req);
2003 return NULL;
2004 }
2005 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
2006 uint64_t idle_left = CWIST_HTTP_BODY_IDLE_TIMEOUT_MS - (now - body_idle_start);
2007 if (idle_left < (uint64_t)wait_ms) wait_ms = (int)idle_left;
2008
2009 struct pollfd pfd = {.fd = conn->fd, .events = POLLIN};
2010 int pret = poll(&pfd, 1, wait_ms);
2011 if (pret <= 0) {
2012 cwist_free(body);
2013 cwist_http_request_destroy(req);
2014 return NULL;
2015 }
2016 }
2017
2018 int bytes = SSL_read(conn->ssl, body + current_body_len,
2019 (int)(req->content_length - current_body_len));
2020 if (bytes <= 0) {
2021 int ssl_err = SSL_get_error(conn->ssl, bytes);
2022 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
2023 if (cwist_ssl_wait(conn->fd, ssl_err, CWIST_HTTP_TIMEOUT_MS) != 0) {
2024 cwist_free(body);
2025 cwist_http_request_destroy(req);
2026 return NULL;
2027 }
2028 continue;
2029 }
2030 cwist_free(body);
2031 cwist_http_request_destroy(req);
2032 return NULL;
2033 }
2034 current_body_len += (size_t)bytes;
2035 body_idle_start = cwist_https_now_ms();
2036 }
2037 body[req->content_length] = '\0';
2038 /* Adopt the filled buffer: one allocation, zero copies. */
2039 cwist_sstring_adopt_len(req->body, body, req->content_length);
2040
2041 if (body_received > req->content_length) {
2042 size_t leftover_len = body_received - req->content_length;
2043 memmove(conn->read_buf, header_end + 4 + req->content_length, leftover_len);
2044 conn->buf_len = leftover_len;
2045 } else {
2046 conn->buf_len = 0;
2047 }
2048 } else {
2049 if (body_received > 0) {
2050 memmove(conn->read_buf, header_end + 4, body_received);
2051 conn->buf_len = body_received;
2052 } else {
2053 conn->buf_len = 0;
2054 }
2055 }
2056 conn->read_buf[conn->buf_len] = '\0';
2057
2058 return req;
2059}
