1/* Internal RX-uring receive path (issue #179).
2 *
3 * The C1M async HTTP server normally waits for readability with a one-shot
4 * POLL_ADD and then drains the fd with recv(). When the reactor runs a real
5 * io_uring ring, the receive wait can instead be an IORING_OP_RECV whose
6 * completion IS the readiness signal: one SQE replaces the POLL+recv pair
7 * and rides the existing deferred-SQE batch, so no extra io_uring_enter is
8 * paid beyond what the run loop already spends.
9 *
10 * SQE namespace: the run loop dispatches on cqe->user_data, historically an
11 * ev_ctx pointer. RECV SQEs tag the low bit of user_data and carry the
12 * connection pointer in the remaining bits:
13 *
14 * user_data = (uint64_t)conn | 1
15 *
16 * ev_ctx pointers come from cwist_alloc (malloc-backed, so at least
17 * max_align_t aligned) and can never carry the tag bit; reactor.c asserts
18 * this when arming.
19 *
20 * This header is private to the build (src/sys/io); it is not installed.
21 * Everything compiles to no-ops on non-Linux so callers need no ifdefs
22 * beyond cheap inline checks. */

55/* Arm one IORING_OP_RECV on the reactor ring. The SQE references
56 * [buf, buf+len) and must stay valid until the completion arrives, so the
57 * caller must not move or consume that buffer while the SQE is in flight.
58 * At most one RECV per connection at a time. When the latency probe is
59 * enabled, *armed_ns (if non-NULL) receives the arm timestamp so the caller
60 * can compute queue delay. Run-thread calls during dispatch defer into the
61 * existing SQE batch; other threads submit immediately so parked workers
62 * wake. Returns false when unsupported or on submission failure; the caller
63 * then falls back to the legacy POLL path. */
64bool cwist_reactor_recv_arm(cwist_reactor_t *reactor, int fd, void *buf, unsigned len, void *conn,
65 uint64_t *armed_ns);

67/* Record one latency-probe sample (queue=true for queue_delay, false for

70 * wrapper in cwist_reactor_run. */
