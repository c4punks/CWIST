76static int proxy_transport_listener(int transport, unsigned short port) {
77 char service[8];
78 snprintf(service, sizeof(service), "%hu", port);
79 struct addrinfo hints = {
80 .ai_family = AF_UNSPEC,
81 .ai_socktype = transport == CWIST_PROXY_TRANSPORT_TCP ? SOCK_STREAM : SOCK_DGRAM,
82 .ai_flags = AI_PASSIVE,
83 };
84 struct addrinfo *addrs = NULL;
85 if (getaddrinfo(NULL, service, &hints, &addrs) != 0) return -1;
86 int fd = -1;
87 for (struct addrinfo *ai = addrs; ai; ai = ai->ai_next) {
88 fd = socket(ai->ai_family, ai->ai_socktype, ai->ai_protocol);
89 if (fd < 0) continue;
90 int one = 1;
91 (void)setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &one, sizeof(one));
92 if (ai->ai_family == AF_INET6) {
93 int zero = 0;
94 (void)setsockopt(fd, IPPROTO_IPV6, IPV6_V6ONLY, &zero, sizeof(zero));
95 }
96 proxy_transport_set_nonblock(fd);
97 if (bind(fd, ai->ai_addr, ai->ai_addrlen) == 0 &&
98 (transport != CWIST_PROXY_TRANSPORT_TCP || listen(fd, 128) == 0)) {
99 break;
100 }
101 close(fd);
102 fd = -1;
103 }
104 freeaddrinfo(addrs);
105 return fd;
106}

108static int proxy_transport_upstream(const char *target_host, unsigned short target_port,
109 int transport) {
110 char service[8];
111 snprintf(service, sizeof(service), "%hu", target_port);
112 struct addrinfo hints = {
113 .ai_family = AF_UNSPEC,
114 .ai_socktype = transport == CWIST_PROXY_TRANSPORT_TCP ? SOCK_STREAM : SOCK_DGRAM,
115 };
116 struct addrinfo *addrs = NULL;

145static bool proxy_transport_send_all(int fd, const unsigned char *buf, size_t len) {
146 size_t off = 0;
147 uint64_t deadline = proxy_transport_now_ms() + PROXY_TRANSPORT_IDLE_TIMEOUT_MS;
148 while (off < len && proxy_transport_active()) {
149 ssize_t n = send(fd, buf + off, len - off, MSG_NOSIGNAL);
150 if (n > 0) {
151 off += (size_t)n;
152 continue;
153 }
154 if (n < 0 && errno == EINTR) continue;
155 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
156 uint64_t now = proxy_transport_now_ms();
157 if (now >= deadline) return false;
158 uint64_t remaining = deadline - now;
159 int timeout =
160 remaining > PROXY_TRANSPORT_POLL_MS ? PROXY_TRANSPORT_POLL_MS : (int)remaining;
161 struct pollfd pfd = {.fd = fd, .events = POLLOUT};
162 if (poll(&pfd, 1, timeout) < 0 && errno != EINTR) return false;
163 continue;
164 }
165 return false;
166 }
167 return off == len;
168}

173 char *target_host = conn->target_host;
174 unsigned short target_port = conn->target_port;

184 bool upstream_open = true;
185 unsigned char buf[16384];

201 } else if (n == 0) {

205 break;
206 }

212 } else if (n == 0) {
213 upstream_open = false;

215 } else if (errno != EINTR && errno != EAGAIN && errno != EWOULDBLOCK) {

217 }
218 }
219 }

225static void *proxy_tcp_listener(void *arg) {
226 proxy_transport_rule *rule = arg;
227 while (proxy_transport_active()) {
228 struct pollfd pfd = {.fd = rule->listen_fd, .events = POLLIN};

230 if (rc < 0) {
231 if (errno == EINTR) continue;
232 break;
233 }
234 if (rc == 0) continue;
235 int client_fd = accept(rule->listen_fd, NULL, NULL);
236 if (client_fd < 0) {

238 continue;
239 break;
240 }

242 if (!conn) {

245 }

426 }

428 rule->next = g_transport_rules;

430 g_transport_rule_count++;

443int cwist_proxy_transport_start(void) {
444 pthread_mutex_lock(&g_transport_lock);
445 if (g_transport_rule_count == 0 || atomic_load(&g_transport_running)) {
446 pthread_mutex_unlock(&g_transport_lock);
447 return g_transport_rule_count == 0 ? 0 : -1;
448 }

452 atomic_store(&g_transport_running, 1);
453 proxy_transport_rule *failed = NULL;
454 for (proxy_transport_rule *rule = g_transport_rules; rule; rule = rule->next) {
455 rule->listen_fd = proxy_transport_listener(rule->transport, rule->origin_port);
456 if (rule->listen_fd < 0) {

468 pthread_mutex_unlock(&g_transport_lock);
