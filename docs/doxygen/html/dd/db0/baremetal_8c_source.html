154static inline uint32_t atomic_irq_save(void) {
155 uint32_t primask;
156 __asm volatile("mrs %0, primask\n\tcpsid i" : "=r"(primask)::"memory");
157 return primask;
158}
159static inline void atomic_irq_restore(uint32_t primask) {
160 __asm volatile("msr primask, %0" ::"r"(primask) : "memory");
161}

165uint64_t __atomic_load_8(const volatile void *mem, int order) {
166 (void)order;
167 uint32_t s = atomic_irq_save();
168 uint64_t v = *(const volatile uint64_t *)mem;
169 atomic_irq_restore(s);
170 return v;
171}

186bool __atomic_compare_exchange_8(volatile void *mem, void *expected, uint64_t desired, bool weak,
187 int success, int failure) {
188 (void)weak;
189 (void)success;
190 (void)failure;
191 uint32_t s = atomic_irq_save();
192 uint64_t cur = *(volatile uint64_t *)mem;
193 bool ok = cur == *(uint64_t *)expected;
194 if (ok)
195 *(volatile uint64_t *)mem = desired;
196 else
197 *(uint64_t *)expected = cur;
198 atomic_irq_restore(s);
199 return ok;
200}
