80bool cwist_io_queue_submit(cwist_io_queue *q, cwist_job_func func, void *arg) {
81 if (!q || !func) return false;
82
83 /* Count the job before checking running, so a runner that sees
84 * running == false and pending_jobs == 0 knows no submit is in flight. */
85 atomic_fetch_add_explicit(&q->pending_jobs, 1, memory_order_acq_rel);
86 if (!atomic_load_explicit(&q->running, memory_order_acquire)) {
87 atomic_fetch_sub_explicit(&q->pending_jobs, 1, memory_order_acq_rel);
88 return false;
89 }
90
91 struct job_wrapper *job = cwist_alloc(sizeof(struct job_wrapper));
92 if (!job) {
93 atomic_fetch_sub_explicit(&q->pending_jobs, 1, memory_order_acq_rel);
94 return false;
95 }
96 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(job);
97 job->func = func;
98 job->arg = arg;
99
100 struct kevent kev;
101 // We use EVFILT_USER to trigger a custom event.
102 // Ident = pointer to job (unique ID)
103 EV_SET(&kev, (uintptr_t)job, EVFILT_USER, EV_ADD | EV_ENABLE | EV_ONESHOT, NOTE_TRIGGER, 0,
104 job);
105
106 if (kevent(q->kq_fd, &kev, 1, NULL, 0, NULL) < 0) {
107 perror("kevent submit");
108 cwist_free(job);
109 atomic_fetch_sub_explicit(&q->pending_jobs, 1, memory_order_acq_rel);
110 return false;
111 }
112 return true;
113}

119void cwist_io_queue_run(cwist_io_queue *q) {
120 if (!q) return;
121 struct kevent events[32];
122 while (atomic_load_explicit(&q->running, memory_order_acquire) ||
123 atomic_load_explicit(&q->pending_jobs, memory_order_acquire) > 0) {
124 int n = kevent(q->kq_fd, NULL, 0, events, 32, NULL);
125 if (n < 0) {
126 if (errno == EINTR) continue;
127 perror("kevent wait");
128 break;
129 }
130
131 int ran = 0;
132 for (int i = 0; i < n; i++) {
133 if (events[i].filter != EVFILT_USER || events[i].ident == KQ_WAKE_IDENT) continue;
134 struct job_wrapper *job = (struct job_wrapper *)events[i].udata;
135 if (!job) continue;
136 atomic_fetch_sub_explicit(&q->pending_jobs, 1, memory_order_acq_rel);
137 job->func(job->arg);
138 cwist_free(job);
139 ran++;
140
141 // Full-GC mode: catch whatever this job forgot to cwist_free()
142 // right away instead of letting it pile up for the lifetime of
143 // this (long-lived, pooled) worker thread, as io_queue.c does.
144 if (cwist_full_gc_enabled()) {
145 cwist_gc_scope_flush();
146 }
147 }
148
149 /* Stopping, but another runner still holds the last jobs: the wake
150 * event keeps kevent() returning at once, so back off instead of
151 * spinning until that runner takes them. */
152 if (ran == 0 && n > 0 && !atomic_load_explicit(&q->running, memory_order_acquire)) {
153 struct timespec ts = {0, 1000000}; /* 1ms */
154 nanosleep(&ts, NULL);
155 }
156 }
157}

185void cwist_io_queue_destroy(cwist_io_queue *q) {
186 if (!q) return;
187 atomic_store_explicit(&q->running, false, memory_order_release);
188
189 /* Drop the level-triggered wake first, or the drain below never ends. */
190 struct kevent kev;
191 EV_SET(&kev, KQ_WAKE_IDENT, EVFILT_USER, EV_DELETE, 0, 0, NULL);
192 (void)kevent(q->kq_fd, &kev, 1, NULL, 0, NULL);
193
194 struct kevent events[32];
195 const struct timespec zero = {0, 0};
196 int n;
197 while ((n = kevent(q->kq_fd, NULL, 0, events, 32, &zero)) > 0) {
198 for (int i = 0; i < n; i++) {
199 if (events[i].filter == EVFILT_USER && events[i].udata) {
200 cwist_free((void *)events[i].udata);
201 }
202 }
203 }
