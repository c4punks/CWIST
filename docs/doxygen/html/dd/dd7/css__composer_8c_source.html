25cwist_color_hsl cwist_color_rgb_to_hsl(cwist_color_rgb rgb) {
26 cwist_color_hsl hsl = {0.0f, 0.0f, 0.0f};
27 float r = rgb.r / 255.0f;
28 float g = rgb.g / 255.0f;
29 float b = rgb.b / 255.0f;
30
31 float max = MAX3(r, g, b);
32 float min = MIN3(r, g, b);
33 hsl.l = (max + min) / 2.0f;
34
35 if (max == min) {
36 hsl.h = 0.0f;
37 hsl.s = 0.0f;
38 } else {
39 float d = max - min;
40 hsl.s = hsl.l > 0.5f ? d / (2.0f - max - min) : d / (max + min);
41
42 if (max == r) {
43 hsl.h = (g - b) / d + (g < b ? 6.0f : 0.0f);
44 } else if (max == g) {
45 hsl.h = (b - r) / d + 2.0f;
46 } else {
47 hsl.h = (r - g) / d + 4.0f;
48 }
49 hsl.h /= 6.0f;
50 }
51 hsl.h *= 360.0f;
52 return hsl;
53}

62static float hue2rgb(float p, float q, float t) {
63 if (t < 0.0f) t += 1.0f;
64 if (t > 1.0f) t -= 1.0f;
65 if (t < 1.0f / 6.0f) return p + (q - p) * 6.0f * t;
66 if (t < 1.0f / 2.0f) return q;
67 if (t < 2.0f / 3.0f) return p + (q - p) * (2.0f / 3.0f - t) * 6.0f;
68 return p;
69}

77cwist_color_rgb cwist_color_hsl_to_rgb(cwist_color_hsl hsl) {
78 cwist_color_rgb rgb = {0, 0, 0};
79 if (!cwist_number_double_is_finite(hsl.h) || !cwist_number_double_is_finite(hsl.s) ||
80 !cwist_number_double_is_finite(hsl.l))
81 return rgb;
82 /* Wrap the hue into [0, 360): hue2rgb() only folds one turn. */
83 float h = fmodf(hsl.h, 360.0f) / 360.0f;
84 if (h < 0.0f) h += 1.0f;
85 float s = hsl.s;
86 float l = hsl.l;
87
88 /* Clamp first: converting a float outside 0..255 to unsigned char is
89 * undefined. Gray is rounded like the chromatic channels below. */
90 if (s < 0.0f) s = 0.0f;
91 if (s > 1.0f) s = 1.0f;
92 if (l < 0.0f) l = 0.0f;
93 if (l > 1.0f) l = 1.0f;
94 if (s == 0.0f) {
95 rgb.r = rgb.g = rgb.b = (unsigned char)roundf(l * 255.0f);
96 } else {
97 float q = l < 0.5f ? l * (1.0f + s) : l + s - l * s;
98 float p = 2.0f * l - q;
99 rgb.r = (unsigned char)(roundf(hue2rgb(p, q, h + 1.0f / 3.0f) * 255.0f));
100 rgb.g = (unsigned char)(roundf(hue2rgb(p, q, h) * 255.0f));
101 rgb.b = (unsigned char)(roundf(hue2rgb(p, q, h - 1.0f / 3.0f) * 255.0f));
102 }
103 return rgb;
104}

125cwist_color_rgb cwist_color_hex_to_rgb(const char *hex) {
126 cwist_color_rgb rgb = {0, 0, 0};
127 if (!hex) return rgb;
128 if (hex[0] == '#') hex++;
129 size_t len = strlen(hex);
130 if (len == 6) {
131 int r1 = hex_char_to_val(hex[0]);
132 int r2 = hex_char_to_val(hex[1]);
133 int g1 = hex_char_to_val(hex[2]);
134 int g2 = hex_char_to_val(hex[3]);
135 int b1 = hex_char_to_val(hex[4]);
136 int b2 = hex_char_to_val(hex[5]);
137 if (r1 >= 0 && r2 >= 0 && g1 >= 0 && g2 >= 0 && b1 >= 0 && b2 >= 0) {
138 rgb.r = (unsigned char)((r1 << 4) | r2);
139 rgb.g = (unsigned char)((g1 << 4) | g2);
140 rgb.b = (unsigned char)((b1 << 4) | b2);
141 }
142 } else if (len == 3) {
143 int r = hex_char_to_val(hex[0]);
144 int g = hex_char_to_val(hex[1]);
145 int b = hex_char_to_val(hex[2]);
146 if (r >= 0 && g >= 0 && b >= 0) {
147 rgb.r = (unsigned char)((r << 4) | r);
148 rgb.g = (unsigned char)((g << 4) | g);
149 rgb.b = (unsigned char)((b << 4) | b);
150 }
151 }
152 return rgb;
153}

367static struct cwist_css_scope_entry *scope_get_or_add(cwist_css_scope *scope,
368 const char *base_class) {
369 struct cwist_css_scope_entry *entry = scope_find(scope, base_class);
370 if (entry) return entry;
371
372 if (scope->count == scope->capacity) {
373 size_t new_capacity = scope->capacity ? scope->capacity * 2 : 8;
374 struct cwist_css_scope_entry *grown = (struct cwist_css_scope_entry *)cwist_realloc(
375 scope->entries, new_capacity * sizeof(*grown));
376 if (!grown) return NULL;
377 scope->entries = grown;
378 scope->capacity = new_capacity;
379 }
380
381 char *base_copy = cwist_strdup(base_class);
382 cwist_sstring *scoped = cwist_sstring_create();
383 if (!base_copy || !scoped || !append_ok(scoped, base_class) || !append_ok(scoped, "-") ||
384 !append_ok(scoped, scope->suffix)) {
385 cwist_free(base_copy);
386 cwist_sstring_destroy(scoped);
387 return NULL;
388 }
389
390 entry = &scope->entries[scope->count++];
391 entry->base_class = base_copy;
392 entry->scoped_class = scoped;
393 entry->declarations = NULL;
394 entry->used = false;
395 return entry;

411 * produced by different worker processes and must agree on the suffix. */

419 scope->entries = NULL;

447static bool css_declarations_are_safe(const char *d) {
448 if (strpbrk(d, "{}<")) return false;
449 char quote = 0;
450 for (const char *p = d; *p; p++) {
451 if (*p == '\\') {
452 /* An escape needs a following character; an escaped newline is
453 * only valid inside a string, so reject it outside one. */
454 if (p[1] == '\0' || (!quote && (p[1] == '\n' || p[1] == '\r' || p[1] == '\f')))
455 return false;
456 p++;
457 } else if (quote) {
458 if (*p == quote)
459 quote = 0;
460 else if (*p == '\n' || *p == '\r' || *p == '\f')
461 return false; /* bad-string token */
462 } else if (*p == '"' || *p == '\'') {
463 quote = *p;
464 } else if (p[0] == '/' && p[1] == '*') {
465 const char *end = strstr(p + 2, "*/");
466 if (!end) return false;
467 p = end + 1;
468 }
469 }
470 return quote == 0;
471}

586cwist_sstring *cwist_css_minify(const char *css) {
587 if (!css) return NULL;
588 size_t n = strlen(css);
589 /* Every step copies or drops input bytes, so the output never grows. */
590 char *out = (char *)cwist_alloc(n + 1);
591 if (!out) return NULL;
592
593 size_t o = 0;
594 size_t verbatim_end = 0; /* out[0..verbatim_end) ends in copied, non-structural bytes */
595 bool pending_space = false;
596 size_t i = 0;
597 while (i < n) {
598 char c = css[i];
599
600 if (c == '/' && css[i + 1] == '*') {
601 const char *close = strstr(css + i + 2, "*/");
602 size_t stop = close ? (size_t)(close - css) + 2 : n;
603 if (css[i + 2] == '!') {
604 if (pending_space && o > 0) out[o++] = ' ';
605 pending_space = false;
606 memcpy(out + o, css + i, stop - i);
607 o += stop - i;
608 verbatim_end = o;
609 } else if (o > 0 && stop < n && css_is_word(out[o - 1]) && css_is_word(css[stop])) {
610 /* Removing the comment must not glue two tokens together. */
611 pending_space = true;
612 }
613 i = stop;
614 continue;
615 }
616 if (css_is_space(c)) {
617 pending_space = true;
618 i++;
619 continue;
620 }
621
622 if (pending_space) {
623 bool after_structural = o > 0 && o != verbatim_end && strchr("{};,:>(", out[o - 1]);
624 bool before_structural = strchr("{};,>)", c) != NULL;
625 if (o > 0 && !after_structural && !before_structural) out[o++] = ' ';
626 pending_space = false;
627 }
628
629 if (c == '"' || c == '\'') {
630 size_t j = i + 1;
631 while (j < n && css[j] != c) {
632 j += (css[j] == '\\' && j + 1 < n) ? 2 : 1;
633 }
634 size_t stop = j < n ? j + 1 : n;
635 memcpy(out + o, css + i, stop - i);
636 o += stop - i;
637 verbatim_end = o;
638 i = stop;
639 continue;
640 }
641 if (c == '\\') {
642 size_t stop = i + 1 < n ? i + 2 : n;
643 memcpy(out + o, css + i, stop - i);
644 o += stop - i;
645 verbatim_end = o;
646 i = stop;
647 continue;
648 }
649 if (css_starts_url(css + i) && (o == 0 || !css_is_word(out[o - 1]))) {
650 size_t j = i + 4;
651 while (j < n && css_is_space(css[j])) j++;
652 if (j < n && css[j] != '"' && css[j] != '\'') {
653 /* Unquoted url(): copy the whole argument as written. */
654 const char *close = strchr(css + j, ')');
655 size_t stop = close ? (size_t)(close - css) + 1 : n;
656 memcpy(out + o, css + i, stop - i);
657 o += stop - i;
658 verbatim_end = o;
659 i = stop;
660 continue;
661 }
662 }
663 if (c == '}' && o > 0 && o != verbatim_end && out[o - 1] == ';') o--;
664
665 out[o++] = c;
666 i++;
667 }
668 out[o] = '\0';
669
670 cwist_sstring *result = cwist_sstring_create();
671 bool ok = result != NULL;
672 if (ok) {
673 cwist_error_t err = cwist_sstring_assign_len(result, out, o);
674 ok = cwist_error_is_ok(&err);
675 cwist_error_dispose(&err);
676 }
677 cwist_free(out);
678 if (!ok) {
679 if (result) cwist_sstring_destroy(result);
680 return NULL;
681 }
682 return result;
683}
