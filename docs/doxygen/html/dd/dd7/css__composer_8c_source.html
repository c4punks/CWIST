24cwist_color_hsl cwist_color_rgb_to_hsl(cwist_color_rgb rgb) {
25 cwist_color_hsl hsl = {0.0f, 0.0f, 0.0f};
26 float r = rgb.r / 255.0f;
27 float g = rgb.g / 255.0f;
28 float b = rgb.b / 255.0f;
29
30 float max = MAX3(r, g, b);
31 float min = MIN3(r, g, b);
32 hsl.l = (max + min) / 2.0f;
33
34 if (max == min) {
35 hsl.h = 0.0f;
36 hsl.s = 0.0f;
37 } else {
38 float d = max - min;
39 hsl.s = hsl.l > 0.5f ? d / (2.0f - max - min) : d / (max + min);
40
41 if (max == r) {
42 hsl.h = (g - b) / d + (g < b ? 6.0f : 0.0f);
43 } else if (max == g) {
44 hsl.h = (b - r) / d + 2.0f;
45 } else {
46 hsl.h = (r - g) / d + 4.0f;
47 }
48 hsl.h /= 6.0f;
49 }
50 hsl.h *= 360.0f;
51 return hsl;
52}

61static float hue2rgb(float p, float q, float t) {
62 if (t < 0.0f) t += 1.0f;
63 if (t > 1.0f) t -= 1.0f;
64 if (t < 1.0f / 6.0f) return p + (q - p) * 6.0f * t;
65 if (t < 1.0f / 2.0f) return q;
66 if (t < 2.0f / 3.0f) return p + (q - p) * (2.0f / 3.0f - t) * 6.0f;
67 return p;
68}

76cwist_color_rgb cwist_color_hsl_to_rgb(cwist_color_hsl hsl) {
77 cwist_color_rgb rgb = {0, 0, 0};
78 float h = hsl.h / 360.0f;
79 float s = hsl.s;
80 float l = hsl.l;
81
82 if (s == 0.0f) {
83 rgb.r = rgb.g = rgb.b = (unsigned char)(l * 255.0f);
84 } else {
85 float q = l < 0.5f ? l * (1.0f + s) : l + s - l * s;
86 float p = 2.0f * l - q;
87 rgb.r = (unsigned char)(roundf(hue2rgb(p, q, h + 1.0f / 3.0f) * 255.0f));
88 rgb.g = (unsigned char)(roundf(hue2rgb(p, q, h) * 255.0f));
89 rgb.b = (unsigned char)(roundf(hue2rgb(p, q, h - 1.0f / 3.0f) * 255.0f));
90 }
91 return rgb;
92}

113cwist_color_rgb cwist_color_hex_to_rgb(const char *hex) {
114 cwist_color_rgb rgb = {0, 0, 0};
115 if (!hex) return rgb;
116 if (hex[0] == '#') hex++;
117 size_t len = strlen(hex);
118 if (len == 6) {
119 int r1 = hex_char_to_val(hex[0]);
120 int r2 = hex_char_to_val(hex[1]);
121 int g1 = hex_char_to_val(hex[2]);
122 int g2 = hex_char_to_val(hex[3]);
123 int b1 = hex_char_to_val(hex[4]);
124 int b2 = hex_char_to_val(hex[5]);
125 if (r1 >= 0 && r2 >= 0 && g1 >= 0 && g2 >= 0 && b1 >= 0 && b2 >= 0) {
126 rgb.r = (unsigned char)((r1 << 4) | r2);
127 rgb.g = (unsigned char)((g1 << 4) | g2);
128 rgb.b = (unsigned char)((b1 << 4) | b2);
129 }
130 } else if (len == 3) {
131 int r = hex_char_to_val(hex[0]);
132 int g = hex_char_to_val(hex[1]);
133 int b = hex_char_to_val(hex[2]);
134 if (r >= 0 && g >= 0 && b >= 0) {
135 rgb.r = (unsigned char)((r << 4) | r);
136 rgb.g = (unsigned char)((g << 4) | g);
137 rgb.b = (unsigned char)((b << 4) | b);
138 }
139 }
140 return rgb;
141}

355static struct cwist_css_scope_entry *scope_get_or_add(cwist_css_scope *scope,
356 const char *base_class) {
357 struct cwist_css_scope_entry *entry = scope_find(scope, base_class);
358 if (entry) return entry;
359
360 if (scope->count == scope->capacity) {
361 size_t new_capacity = scope->capacity ? scope->capacity * 2 : 8;
362 struct cwist_css_scope_entry *grown = (struct cwist_css_scope_entry *)cwist_realloc(
363 scope->entries, new_capacity * sizeof(*grown));
364 if (!grown) return NULL;
365 scope->entries = grown;
366 scope->capacity = new_capacity;
367 }
368
369 char *base_copy = cwist_strdup(base_class);
370 cwist_sstring *scoped = cwist_sstring_create();
371 if (!base_copy || !scoped || !append_ok(scoped, base_class) || !append_ok(scoped, "-") ||
372 !append_ok(scoped, scope->suffix)) {
373 cwist_free(base_copy);
374 cwist_sstring_destroy(scoped);
375 return NULL;
376 }
377
378 entry = &scope->entries[scope->count++];
379 entry->base_class = base_copy;
380 entry->scoped_class = scoped;
381 entry->declarations = NULL;
382 entry->used = false;
383 return entry;

394void cwist_css_scope_init(cwist_css_scope *scope, const char *component_name) {
395 if (!scope) return;
396 const char *name = component_name ? component_name : "";
397
398 /* FNV-1a, 32-bit. Deliberately unseeded: markup and stylesheet may be
399 * produced by different worker processes and must agree on the suffix. */
400 uint32_t hash = 2166136261u;
401 for (const unsigned char *p = (const unsigned char *)name; *p; p++) {
402 hash ^= *p;
403 hash *= 16777619u;
404 }
405
406 snprintf(scope->suffix, sizeof(scope->suffix), "%08x", (unsigned int)hash);
407 scope->entries = NULL;
408 scope->count = 0;
409 scope->capacity = 0;

539cwist_sstring *cwist_css_minify(const char *css) {
540 if (!css) return NULL;
541 size_t n = strlen(css);
542 /* Every step copies or drops input bytes, so the output never grows. */
543 char *out = (char *)cwist_alloc(n + 1);
544 if (!out) return NULL;
545
546 size_t o = 0;
547 size_t verbatim_end = 0; /* out[0..verbatim_end) ends in copied, non-structural bytes */
548 bool pending_space = false;
549 size_t i = 0;
550 while (i < n) {
551 char c = css[i];
552
553 if (c == '/' && css[i + 1] == '*') {
554 const char *close = strstr(css + i + 2, "*/");
555 size_t stop = close ? (size_t)(close - css) + 2 : n;
556 if (css[i + 2] == '!') {
557 if (pending_space && o > 0) out[o++] = ' ';
558 pending_space = false;
559 memcpy(out + o, css + i, stop - i);
560 o += stop - i;
561 verbatim_end = o;
562 } else if (o > 0 && stop < n && css_is_word(out[o - 1]) && css_is_word(css[stop])) {
563 /* Removing the comment must not glue two tokens together. */
564 pending_space = true;
565 }
566 i = stop;
567 continue;
568 }
569 if (css_is_space(c)) {
570 pending_space = true;
571 i++;
572 continue;
573 }
574
575 if (pending_space) {
576 bool after_structural = o > 0 && o != verbatim_end && strchr("{};,:>(", out[o - 1]);
577 bool before_structural = strchr("{};,>)", c) != NULL;
578 if (o > 0 && !after_structural && !before_structural) out[o++] = ' ';
579 pending_space = false;
580 }
581
582 if (c == '"' || c == '\'') {
583 size_t j = i + 1;
584 while (j < n && css[j] != c) {
585 j += (css[j] == '\\' && j + 1 < n) ? 2 : 1;
586 }
587 size_t stop = j < n ? j + 1 : n;
588 memcpy(out + o, css + i, stop - i);
589 o += stop - i;
590 verbatim_end = o;
591 i = stop;
592 continue;
593 }
594 if (c == '\\') {
595 size_t stop = i + 1 < n ? i + 2 : n;
596 memcpy(out + o, css + i, stop - i);
597 o += stop - i;
598 verbatim_end = o;
599 i = stop;
600 continue;
601 }
602 if (css_starts_url(css + i) && (o == 0 || !css_is_word(out[o - 1]))) {
603 size_t j = i + 4;
604 while (j < n && css_is_space(css[j])) j++;
605 if (j < n && css[j] != '"' && css[j] != '\'') {
606 /* Unquoted url(): copy the whole argument as written. */
607 const char *close = strchr(css + j, ')');
608 size_t stop = close ? (size_t)(close - css) + 1 : n;
609 memcpy(out + o, css + i, stop - i);
610 o += stop - i;
611 verbatim_end = o;
612 i = stop;
613 continue;
614 }
615 }
616 if (c == '}' && o > 0 && o != verbatim_end && out[o - 1] == ';') o--;
617
618 out[o++] = c;
619 i++;
620}
621out[o] = '\0';
622
623cwist_sstring *result = cwist_sstring_create();
624bool ok = result != NULL;
625if (ok) {
626 cwist_error_t err = cwist_sstring_assign_len(result, out, o);
627 ok = cwist_error_is_ok(&err);
628 cwist_error_dispose(&err);
629}
630cwist_free(out);
631if (!ok) {
632 if (result) cwist_sstring_destroy(result);
633 return NULL;
634}
635return result;
636}
