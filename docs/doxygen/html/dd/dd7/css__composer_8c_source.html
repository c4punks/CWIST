24cwist_color_hsl cwist_color_rgb_to_hsl(cwist_color_rgb rgb) {
25 cwist_color_hsl hsl = {0.0f, 0.0f, 0.0f};
26 float r = rgb.r / 255.0f;
27 float g = rgb.g / 255.0f;
28 float b = rgb.b / 255.0f;
29
30 float max = MAX3(r, g, b);
31 float min = MIN3(r, g, b);
32 hsl.l = (max + min) / 2.0f;
33
34 if (max == min) {
35 hsl.h = 0.0f;
36 hsl.s = 0.0f;
37 } else {
38 float d = max - min;
39 hsl.s = hsl.l > 0.5f ? d / (2.0f - max - min) : d / (max + min);
40
41 if (max == r) {
42 hsl.h = (g - b) / d + (g < b ? 6.0f : 0.0f);
43 } else if (max == g) {
44 hsl.h = (b - r) / d + 2.0f;
45 } else {
46 hsl.h = (r - g) / d + 4.0f;
47 }
48 hsl.h /= 6.0f;
49 }
50 hsl.h *= 360.0f;
51 return hsl;
52}

61static float hue2rgb(float p, float q, float t) {
62 if (t < 0.0f) t += 1.0f;
63 if (t > 1.0f) t -= 1.0f;
64 if (t < 1.0f / 6.0f) return p + (q - p) * 6.0f * t;
65 if (t < 1.0f / 2.0f) return q;
66 if (t < 2.0f / 3.0f) return p + (q - p) * (2.0f / 3.0f - t) * 6.0f;
67 return p;
68}

76cwist_color_rgb cwist_color_hsl_to_rgb(cwist_color_hsl hsl) {
77 cwist_color_rgb rgb = {0, 0, 0};
78 float h = hsl.h / 360.0f;
79 float s = hsl.s;
80 float l = hsl.l;
81
82 if (s == 0.0f) {
83 rgb.r = rgb.g = rgb.b = (unsigned char)(l * 255.0f);
84 } else {
85 float q = l < 0.5f ? l * (1.0f + s) : l + s - l * s;
86 float p = 2.0f * l - q;
87 rgb.r = (unsigned char)(roundf(hue2rgb(p, q, h + 1.0f / 3.0f) * 255.0f));
88 rgb.g = (unsigned char)(roundf(hue2rgb(p, q, h) * 255.0f));
89 rgb.b = (unsigned char)(roundf(hue2rgb(p, q, h - 1.0f / 3.0f) * 255.0f));
90 }
91 return rgb;
92}

99static inline int hex_char_to_val(char c) {
100 if (c >= '0' && c <= '9') return c - '0';
101 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
102 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
103 return -1;

113cwist_color_rgb cwist_color_hex_to_rgb(const char *hex) {
114 cwist_color_rgb rgb = {0, 0, 0};
115 if (!hex) return rgb;
116 if (hex[0] == '#') hex++;
117 size_t len = strlen(hex);
118 if (len == 6) {
119 int r1 = hex_char_to_val(hex[0]);
120 int r2 = hex_char_to_val(hex[1]);
121 int g1 = hex_char_to_val(hex[2]);
122 int g2 = hex_char_to_val(hex[3]);
123 int b1 = hex_char_to_val(hex[4]);
124 int b2 = hex_char_to_val(hex[5]);
125 if (r1 >= 0 && r2 >= 0 && g1 >= 0 && g2 >= 0 && b1 >= 0 && b2 >= 0) {
126 rgb.r = (unsigned char)((r1 << 4) | r2);
127 rgb.g = (unsigned char)((g1 << 4) | g2);
128 rgb.b = (unsigned char)((b1 << 4) | b2);
129 }
130 } else if (len == 3) {
131 int r = hex_char_to_val(hex[0]);
132 int g = hex_char_to_val(hex[1]);
133 int b = hex_char_to_val(hex[2]);
134 if (r >= 0 && g >= 0 && b >= 0) {
135 rgb.r = (unsigned char)((r << 4) | r);
136 rgb.g = (unsigned char)((g << 4) | g);
137 rgb.b = (unsigned char)((b << 4) | b);
138 }
139 }
140 return rgb;
141}

103static void get_shifted_hex(cwist_color_rgb base, float light_shift, char out[8]) {
104 cwist_color_hsl hsl = cwist_color_rgb_to_hsl(base);
105 hsl.l += light_shift;
