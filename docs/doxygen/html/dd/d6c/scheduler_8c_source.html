56static uint64_t now_ms(void) {
57 struct timespec ts;
58 if (clock_gettime(CLOCK_MONOTONIC, &ts) != 0) return 0;
59 return (uint64_t)ts.tv_sec * 1000ULL + (uint64_t)ts.tv_nsec / 1000000ULL;
60}

81static void cond_deadline_after(uint64_t wait_ms, struct timespec *ts) {
82 /* The caller re-checks the heap after every wake, so a far deadline can
83 * be capped; this keeps tv_sec from overflowing. */
84 if (wait_ms > 86400000ULL) wait_ms = 86400000ULL;
85 struct timespec now = {0, 0};
86 clock_gettime(SCHED_COND_CLOCK, &now);
87 uint64_t sec = wait_ms / 1000ULL;
88 long nsec = now.tv_nsec + (long)((wait_ms % 1000ULL) * 1000000ULL);
89 if (nsec >= 1000000000L) {
90 nsec -= 1000000000L;
91 sec++;
92 }
93 ts->tv_sec = now.tv_sec + (time_t)sec;
94 ts->tv_nsec = nsec;
95}

106static bool cond_init_monotonic(pthread_cond_t *cond) {
107 pthread_condattr_t attr;
108 if (pthread_condattr_init(&attr) != 0) return false;
109 bool ok = true;
110#if defined(__linux__) || defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__)
111 ok = pthread_condattr_setclock(&attr, SCHED_COND_CLOCK) == 0;
112#endif
113 if (ok) ok = pthread_cond_init(cond, &attr) == 0;
114 pthread_condattr_destroy(&attr);
115 return ok;
116}

131static int heap_grow(cwist_scheduler_t *s) {
132 if (atomic_load_explicit(&s->heap_count, memory_order_relaxed) < s->heap_capacity) return 0;
133 size_t new_cap = s->heap_capacity ? s->heap_capacity : CWIST_SCHEDULER_HEAP_INIT;
134 if (s->heap_capacity) {
135 if (s->heap_capacity > SIZE_MAX / 2) return -1;
136 new_cap = s->heap_capacity * 2;
137 }
138 if (new_cap > SIZE_MAX / sizeof(scheduled_job_t)) return -1;
139 scheduled_job_t *new_heap = cwist_realloc(s->heap, new_cap * sizeof(scheduled_job_t));
140 if (!new_heap) return -1;
141 /* s->heap is owned by the scheduler for its whole lifetime, not by
142 * whichever caller thread happens to trigger a resize -- cwist_realloc()
143 * tracks it on that thread's full-GC pending-sweep list (matching
144 * cwist_alloc()'s tracking of a fresh NULL-in allocation). Left tracked
145 * there, that thread exiting would have the sweep free this array out
146 * from under a scheduler that's still using it. Disown it immediately;
147 * cwist_scheduler_destroy() is what actually releases it. */
148 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(new_heap);
149 s->heap = new_heap;
150 s->heap_capacity = new_cap;
151 return 0;
152}

172static void heap_push(cwist_scheduler_t *s, scheduled_job_t job) {
173 if (heap_grow(s) != 0) return;
174 size_t i = atomic_load_explicit(&s->heap_count, memory_order_relaxed);
175 atomic_store_explicit(&s->heap_count, i + 1, memory_order_release);
176 s->heap[i] = job;
177 while (i > 0) {
178 size_t parent = (i - 1) / 2;
179 if (s->heap[parent].deadline_ms <= s->heap[i].deadline_ms) break;
180 heap_swap(&s->heap[parent], &s->heap[i]);
181 i = parent;
182 }
183}

192static scheduled_job_t heap_pop(cwist_scheduler_t *s) {
193 scheduled_job_t root = s->heap[0];
194 size_t count = atomic_load_explicit(&s->heap_count, memory_order_relaxed) - 1;
195 atomic_store_explicit(&s->heap_count, count, memory_order_release);
196 s->heap[0] = s->heap[count];
197 size_t i = 0;
198 while (1) {
199 size_t left = i * 2 + 1;
200 size_t right = left + 1;
201 size_t smallest = i;
202 if (left < count && s->heap[left].deadline_ms < s->heap[smallest].deadline_ms) {
203 smallest = left;
204 }
205 if (right < count && s->heap[right].deadline_ms < s->heap[smallest].deadline_ms) {
206 smallest = right;
207 }
208 if (smallest == i) break;
209 heap_swap(&s->heap[i], &s->heap[smallest]);
210 i = smallest;
211 }
212 return root;
213}

238static void *scheduler_timer_thread(void *arg) {
239 cwist_scheduler_t *s = (cwist_scheduler_t *)arg;
240
241 pthread_mutex_lock(&s->delayed_mtx);
242 while (s->running) {
243 if (atomic_load_explicit(&s->heap_count, memory_order_relaxed) == 0) {
244 pthread_cond_wait(&s->delayed_cond, &s->delayed_mtx);
245 continue;
246 }
247
248 uint64_t current_ms = now_ms();
249 if (s->heap[0].deadline_ms <= current_ms) {
250 scheduled_job_t job = s->heap[0];
251 if (cwist_io_queue_submit(s->queue, job.func, job.arg)) {
252 (void)heap_pop(s);
253 continue;
254 }
255 struct timespec retry;
256 cond_deadline_after(1, &retry);
257 pthread_cond_timedwait(&s->delayed_cond, &s->delayed_mtx, &retry);
258 continue;
259 }
260
261 uint64_t wait_ms = s->heap[0].deadline_ms - current_ms;
262 struct timespec deadline;
263 cond_deadline_after(wait_ms, &deadline);
264 pthread_cond_timedwait(&s->delayed_cond, &s->delayed_mtx, &deadline);
265 }
266 pthread_mutex_unlock(&s->delayed_mtx);
267 return NULL;
268}

301cwist_scheduler_t *cwist_scheduler_create(size_t worker_count, size_t queue_capacity) {
302 if (worker_count == 0) worker_count = 1;
303
304 cwist_scheduler_t *s = (cwist_scheduler_t *)cwist_alloc(sizeof(*s));
305 if (!s) return NULL;
306 /* Explicit destroy owns the scheduler, not its creator's TLS scope.
307 * Detach on the allocating thread before publishing to workers. */
308 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(s);
309 memset(s, 0, sizeof(*s));
310
311 s->queue = cwist_io_queue_create(queue_capacity);
312 if (!s->queue) {
313 cwist_free(s);
314 return NULL;
315 }
316
317 s->workers = (pthread_t *)cwist_alloc_array(worker_count, sizeof(pthread_t));
318 if (s->workers && cwist_full_gc_enabled()) cwist_gc_scope_disown(s->workers);
319 if (!s->workers) {
320 cwist_io_queue_destroy(s->queue);
321 cwist_free(s);
322 return NULL;
323 }
324 s->worker_count = worker_count;
325
326 if (pthread_mutex_init(&s->delayed_mtx, NULL) != 0) {
327 cwist_free(s->workers);
328 cwist_io_queue_destroy(s->queue);
329 cwist_free(s);
330 return NULL;
331 }
332 if (!cond_init_monotonic(&s->delayed_cond)) {
333 pthread_mutex_destroy(&s->delayed_mtx);
334 cwist_free(s->workers);
335 cwist_io_queue_destroy(s->queue);
336 cwist_free(s);
337 return NULL;
338 }
339 s->running = true;
340
341 for (size_t i = 0; i < worker_count; i++) {
342 if (pthread_create(&s->workers[i], NULL, scheduler_worker_thread, s) != 0) {
343 scheduler_cleanup_threads(s, i);
344 pthread_mutex_destroy(&s->delayed_mtx);
345 pthread_cond_destroy(&s->delayed_cond);
346 cwist_free(s->workers);
347 cwist_io_queue_destroy(s->queue);
348 cwist_free(s);
349 return NULL;
350 }
351 }
352
353 if (pthread_create(&s->timer_thread, NULL, scheduler_timer_thread, s) != 0) {
354 scheduler_cleanup_threads(s, worker_count);
355 pthread_mutex_destroy(&s->delayed_mtx);
356 pthread_cond_destroy(&s->delayed_cond);
357 cwist_free(s->workers);
358 cwist_io_queue_destroy(s->queue);
359 cwist_free(s);
360 return NULL;
361 }
362
363 return s;
364}

375void cwist_scheduler_destroy(cwist_scheduler_t *s) {
376 if (!s) return;
377
378 pthread_mutex_lock(&s->delayed_mtx);
379 s->running = false;
380 pthread_cond_broadcast(&s->delayed_cond);
381 pthread_mutex_unlock(&s->delayed_mtx);
382
383 cwist_io_queue_stop(s->queue);
384
385 for (size_t i = 0; i < s->worker_count; i++) {
386 pthread_join(s->workers[i], NULL);
387 }
388 pthread_join(s->timer_thread, NULL);
389
390 cwist_io_queue_destroy(s->queue);
391 cwist_free(s->heap);
392 cwist_free(s->workers);
393 pthread_mutex_destroy(&s->delayed_mtx);
394 pthread_cond_destroy(&s->delayed_cond);
395 cwist_free(s);

431bool cwist_scheduler_schedule(cwist_scheduler_t *s, cwist_job_func func, void *arg,
432 uint64_t delay_ms) {
433 if (!s || !func) return false;
434 uint64_t current_ms = now_ms();
435 if (delay_ms > UINT64_MAX - current_ms) return false;
436 scheduled_job_t job = {
437 .deadline_ms = current_ms + delay_ms,
438 .func = func,
439 .arg = arg,
440 };
441
442 pthread_mutex_lock(&s->delayed_mtx);
443 if (!s->running) {
444 pthread_mutex_unlock(&s->delayed_mtx);
445 return false;
446 }
447 if (delay_ms == 0) {
448 bool ok = cwist_io_queue_submit(s->queue, func, arg);
449 pthread_mutex_unlock(&s->delayed_mtx);
450 return ok;
451 }
452 size_t before = atomic_load_explicit(&s->heap_count, memory_order_relaxed);
453 heap_push(s, job);
454 if (atomic_load_explicit(&s->heap_count, memory_order_relaxed) != before + 1) {
455 pthread_mutex_unlock(&s->delayed_mtx);
456 return false;
457 }
458 pthread_cond_signal(&s->delayed_cond);
459 pthread_mutex_unlock(&s->delayed_mtx);
460 return true;
461}

489 if (!s) return 0;
