102static int heap_grow(cwist_scheduler_t *s) {
103 if (s->heap_count < s->heap_capacity) return 0;
104 size_t new_cap = s->heap_capacity ? s->heap_capacity * 2 : CWIST_SCHEDULER_HEAP_INIT;
105 scheduled_job_t *new_heap = cwist_realloc(s->heap, new_cap * sizeof(scheduled_job_t));
106 if (!new_heap) return -1;
107 /* s->heap is owned by the scheduler for its whole lifetime, not by
108 * whichever caller thread happens to trigger a resize -- cwist_realloc()
109 * tracks it on that thread's full-GC pending-sweep list (matching
110 * cwist_alloc()'s tracking of a fresh NULL-in allocation). Left tracked
111 * there, that thread exiting would have the sweep free this array out
112 * from under a scheduler that's still using it. Disown it immediately;
113 * cwist_scheduler_destroy() is what actually releases it. */
114 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(new_heap);
115 s->heap = new_heap;
116 s->heap_capacity = new_cap;
117 return 0;
118}

138static void heap_push(cwist_scheduler_t *s, scheduled_job_t job) {
139 if (heap_grow(s) != 0) return;
140 size_t i = s->heap_count++;
141 s->heap[i] = job;
142 while (i > 0) {
143 size_t parent = (i - 1) / 2;
144 if (s->heap[parent].deadline_ms <= s->heap[i].deadline_ms) break;
145 heap_swap(&s->heap[parent], &s->heap[i]);
146 i = parent;
147 }
148}

157static scheduled_job_t heap_pop(cwist_scheduler_t *s) {
158 scheduled_job_t root = s->heap[0];
159 s->heap[0] = s->heap[--s->heap_count];
160 size_t i = 0;
161 while (1) {
162 size_t left = i * 2 + 1;
163 size_t right = left + 1;
164 size_t smallest = i;
165 if (left < s->heap_count && s->heap[left].deadline_ms < s->heap[smallest].deadline_ms) {
166 smallest = left;
167 }
168 if (right < s->heap_count && s->heap[right].deadline_ms < s->heap[smallest].deadline_ms) {
169 smallest = right;
170 }
171 if (smallest == i) break;
172 heap_swap(&s->heap[i], &s->heap[smallest]);
173 i = smallest;
174 }
175 return root;
176}

200static void *scheduler_timer_thread(void *arg) {
201 cwist_scheduler_t *s = (cwist_scheduler_t *)arg;
202
203 pthread_mutex_lock(&s->delayed_mtx);
204 while (s->running) {
205 if (s->heap_count == 0) {
206 pthread_cond_wait(&s->delayed_cond, &s->delayed_mtx);
207 continue;
208 }
209
210 uint64_t current_ms = now_ms();
211 if (s->heap[0].deadline_ms <= current_ms) {
212 scheduled_job_t job = heap_pop(s);
213 pthread_mutex_unlock(&s->delayed_mtx);
214 cwist_io_queue_submit(s->queue, job.func, job.arg);
215 pthread_mutex_lock(&s->delayed_mtx);
216 continue;
217 }
218
219 uint64_t wait_ms = s->heap[0].deadline_ms - current_ms;
220 struct timespec deadline;
221 ms_to_timespec(current_ms + wait_ms, &deadline);
222 pthread_cond_timedwait(&s->delayed_cond, &s->delayed_mtx, &deadline);
223 }
224 pthread_mutex_unlock(&s->delayed_mtx);
225 return NULL;
226}

259cwist_scheduler_t *cwist_scheduler_create(size_t worker_count, size_t queue_capacity) {
260 if (worker_count == 0) worker_count = 1;
261
262 cwist_scheduler_t *s = (cwist_scheduler_t *)cwist_alloc(sizeof(*s));
263 if (!s) return NULL;
264 /* Explicit destroy owns the scheduler, not its creator's TLS scope.
265 * Detach on the allocating thread before publishing to workers. */
266 if (cwist_full_gc_enabled()) cwist_gc_scope_disown(s);
267 memset(s, 0, sizeof(*s));
268
269 s->queue = cwist_io_queue_create(queue_capacity);
270 if (!s->queue) {
271 cwist_free(s);
272 return NULL;
273 }
274
275 s->workers = (pthread_t *)cwist_alloc_array(worker_count, sizeof(pthread_t));
276 if (!s->workers) {
277 cwist_io_queue_destroy(s->queue);
278 cwist_free(s);
279 return NULL;
280 }
281 s->worker_count = worker_count;
282
283 pthread_mutex_init(&s->delayed_mtx, NULL);
284 cond_init_monotonic(&s->delayed_cond);
285 s->running = true;
286
287 for (size_t i = 0; i < worker_count; i++) {
288 if (pthread_create(&s->workers[i], NULL, scheduler_worker_thread, s) != 0) {
289 scheduler_cleanup_threads(s, i);
290 pthread_mutex_destroy(&s->delayed_mtx);
291 pthread_cond_destroy(&s->delayed_cond);
292 cwist_free(s->workers);
293 cwist_io_queue_destroy(s->queue);
294 cwist_free(s);
295 return NULL;
296 }
297 }
298
299 if (pthread_create(&s->timer_thread, NULL, scheduler_timer_thread, s) != 0) {
300 scheduler_cleanup_threads(s, worker_count);
301 pthread_mutex_destroy(&s->delayed_mtx);
302 pthread_cond_destroy(&s->delayed_cond);
303 cwist_free(s->workers);
304 cwist_io_queue_destroy(s->queue);
305 cwist_free(s);
306 return NULL;
307 }
308
309 return s;
310}

321void cwist_scheduler_destroy(cwist_scheduler_t *s) {
322 if (!s) return;
323
324 s->running = false;
325 pthread_mutex_lock(&s->delayed_mtx);
326 pthread_cond_broadcast(&s->delayed_cond);
327 pthread_mutex_unlock(&s->delayed_mtx);
328
329 cwist_io_queue_stop(s->queue);
330
331 for (size_t i = 0; i < s->worker_count; i++) {
332 pthread_join(s->workers[i], NULL);
333 }
334 pthread_join(s->timer_thread, NULL);

337 cwist_free(s->heap);
338 cwist_free(s->workers);
339 pthread_mutex_destroy(&s->delayed_mtx);
340 pthread_cond_destroy(&s->delayed_cond);
341 cwist_free(s);

374bool cwist_scheduler_schedule(cwist_scheduler_t *s, cwist_job_func func, void *arg,
375 uint64_t delay_ms) {
376 if (!s || !func || !s->running) return false;
377 if (delay_ms == 0) {
378 return cwist_io_queue_submit(s->queue, func, arg);
379 }
380
381 scheduled_job_t job;
382 job.deadline_ms = now_ms() + delay_ms;
383 job.func = func;
384 job.arg = arg;
385
386 pthread_mutex_lock(&s->delayed_mtx);
387 size_t before = s->heap_count;
388 heap_push(s, job);
389 if (s->heap_count != before + 1) {
390 pthread_mutex_unlock(&s->delayed_mtx);
391 return false;
392 }
393 pthread_cond_signal(&s->delayed_cond);
394 pthread_mutex_unlock(&s->delayed_mtx);
395 return true;
396}
