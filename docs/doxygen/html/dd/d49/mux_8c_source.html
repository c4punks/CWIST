29static cwist_mux_signature cwist_mux_signature_from_path(cwist_http_method_t method,
30 const char *path) {
31 uint64_t hi = 14695981039346656037ULL;
32 uint64_t lo = 0xcbf29ce484222325ULL;
33
34 const uint64_t prime_hi = 1099511628211ULL; /* FNV Prime 64 */
35 const uint64_t prime_lo = 0x9e3779b185ebca87ULL; /* SplitMix Prime */
36
37 /* 1. Method hashing (single pass) */
38 const uint8_t *m_ptr = (const uint8_t *)&method;
39 for (size_t i = 0; i < sizeof(method); ++i) {
40 hi = (hi ^ m_ptr[i]) * prime_hi;
41 lo = (lo ^ m_ptr[i]) * prime_lo;
42 }
43
44 /* 2. Path normalization and single-pass hashing */
45 if (!path) {
46 path = "/";
47 }
48
49 size_t seg_count = 0;
50 const char *cursor = path;
51
52 while (*cursor) {
53 while (*cursor == '/') {
54 cursor++;
55 }
56 if (!*cursor) {
57 break;
58 }
59
60 const char *start = cursor;
61 while (*cursor && *cursor != '/') {
62 cursor++;
63 }
64
65 /* Inject normalized segment delimiter '/' */
66 hi = (hi ^ '/') * prime_hi;
67 lo = (lo ^ '/') * prime_lo;
68
69 /* Hash segment bytes */
70 for (const char *p = start; p < cursor; ++p) {
71 uint8_t b = (uint8_t)*p;
72 hi = (hi ^ b) * prime_hi;
73 lo = (lo ^ b) * prime_lo;
74 }
75 seg_count++;
76 }
77
78 /* Handle root path ('/') or empty path */
79 if (seg_count == 0) {
80 hi = (hi ^ '/') * prime_hi;
81 lo = (lo ^ '/') * prime_lo;
82 }
83
84 return (cwist_mux_signature){.hi = hi, .lo = lo};
85}

93static size_t cwist_mux_bucket_index(const cwist_mux_router *router,
94 const cwist_mux_signature *sig) {
95 uint64_t hash = sig->hi ^ (sig->lo + 0x9e3779b97f4a7c15ULL);
96
97 /* SplitMix64-style finalizer to maximize avalanche effect. */
98 hash ^= hash >> 30;
99 hash *= 0xbf58476d1ce4e5b9ULL;
100 hash ^= hash >> 27;
101 hash *= 0x94d049bb133111ebULL;
102 hash ^= hash >> 31;
103
104 return (size_t)(hash % router->bucket_count);
105}

210static bool match_parametric_route(const char *route_tmpl, const char *req_path,
211 cwist_query_map **out_params) {
212 const char *t = route_tmpl;
213 const char *p = req_path;
214 cwist_query_map *params = NULL;
215
216 while (*t && *p) {
217 if (*t == ':') {
218 t++;
219 const char *t_next = strchr(t, '/');
220 size_t t_len = t_next ? (size_t)(t_next - t) : strlen(t);
221 char param_name[256] = {0};
222 if (t_len < sizeof(param_name)) memcpy(param_name, t, t_len);
223
224 const char *p_next = strchr(p, '/');
225 size_t p_len = p_next ? (size_t)(p_next - p) : strlen(p);
226 char param_value[256] = {0};
227 if (p_len < sizeof(param_value)) memcpy(param_value, p, p_len);
228
229 if (!params) params = cwist_query_map_create();
230 cwist_query_map_set(params, param_name, param_value);
231
232 t += t_len;
233 p += p_len;
234 } else if (*t == *p) {
235 t++;
236 p++;
237 } else {
238 if (params) cwist_query_map_destroy(params);
239 return false;
240 }
241 }
242
243 // Ignore trailing slashes
244 if (*t == '/' && *(t + 1) == '\0') t++;
245 if (*p == '/' && *(p + 1) == '\0') p++;
246
247 if (*t == '\0' && *p == '\0') {
248 *out_params = params;
249 return true;
250 }
251
252 if (params) cwist_query_map_destroy(params);
253 return false;
254}

434bool cwist_mux_serve(cwist_mux_router *router, cwist_http_request *req, cwist_http_response *res) {
435 if (!router || !req || !res) return false;
436
437 const char *path = (req->path && req->path->data) ? req->path->data : "/";
438 cwist_mux_signature signature = cwist_mux_signature_from_path(req->method, path);
439 size_t idx = cwist_mux_bucket_index(router, &signature);
440 cwist_mux_route *curr = router->buckets[idx];
441
442 // 1. Try exact match (O(1))
443 while (curr) {
444 if (curr->method == req->method && curr->signature_hi == signature.hi &&
445 curr->signature_lo == signature.lo && curr->path && curr->path->data &&
446 strcmp(curr->path->data, path) == 0) {
447 if (curr->middleware) {
448 run_middleware_chain(curr->middleware, req, res, curr->handler);
449 } else {
450 curr->handler(req, res);
451 }
452 return true;
453 }
454 curr = curr->bucket_next;
455 }
456
457 // 2. Try parametric matching (linear search)
458 curr = router->param_routes;
459 while (curr) {
460 if (curr->method == req->method && curr->path && curr->path->data) {
461 cwist_query_map *extracted_params = NULL;
462 if (match_parametric_route(curr->path->data, path, &extracted_params)) {
463 if (extracted_params) {
464 if (req->path_params) {
465 cwist_query_map_destroy(req->path_params);
466 }
467 req->path_params = extracted_params;
468 }
469 if (curr->middleware) {
470 run_middleware_chain(curr->middleware, req, res, curr->handler);
471 } else {
472 curr->handler(req, res);
473 }
474 return true;
475 }
476 }
477 curr = curr->param_next;
478 }
479
480 // 3. Try wildcard matching (linear search)
481 curr = router->wildcard_routes;
482 while (curr) {
483 if (curr->method == req->method && curr->path && curr->path->data) {
484 if (match_wildcard_route(curr->path->data, path)) {
485 if (curr->middleware) {
486 run_middleware_chain(curr->middleware, req, res, curr->handler);
487 } else {
488 curr->handler(req, res);
489 }
490 return true;
491 }
492 }
493 curr = curr->param_next;
494 }
495
496 return false;
