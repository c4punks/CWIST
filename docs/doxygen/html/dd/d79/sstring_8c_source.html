47static char *cwist_sstring_reserve(cwist_sstring *str, size_t needed, size_t preserve_len) {
48 if (!str->borrows_buffer && str->data && str->capacity >= needed) return str->data;
49
50 size_t new_cap = str->capacity ? str->capacity : 16;
51 while (new_cap < needed) {
52 if (new_cap > (SIZE_MAX - 1) / 2) {
53 new_cap = needed;
54 break;
55 }
56 new_cap *= 2;
57 }
58
59 char *result;
60 if (str->borrows_buffer) {
61 result = (char *)cwist_alloc(new_cap + 1);
62 if (result) {
63 if (str->data && preserve_len > 0) memcpy(result, str->data, preserve_len);
64 str->base = result;
65 str->capacity = new_cap;
66 }
67 return result;
68 }
69
70 char *orig = str->base ? str->base : str->data;
71 const size_t lead = (str->base && str->data) ? (size_t)(str->data - str->base) : 0;
72 char *grown = (char *)cwist_realloc(orig, lead + new_cap + 1);
73 if (!grown) return NULL;
74 str->base = grown;
75 str->capacity = new_cap;
76 return grown + lead;
77}

581cwist_error_t cwist_sstring_append_escaped(cwist_sstring *str, const char *data) {
582 if (!str) {
583 cwist_error_t err = make_error(CWIST_ERR_INT8);
584 err.error.err_i8 = ERR_SSTRING_NULL_STRING;
585 return err;
586 }
587 if (!data) {
588 // Appending nothing is success
589 cwist_error_t err = make_error(CWIST_ERR_INT8);
590 err.error.err_i8 = ERR_SSTRING_OKAY;
591 return err;
592 }
593
594 size_t current_len = str->data ? strlen(str->data) : 0;
595 size_t input_len = strlen(data);
596 size_t new_size = current_len + (input_len * 6) + 1;
597
598 if (str->is_fixed) {
599 if (new_size > str->size) {
600 cwist_error_t err = make_error(CWIST_ERR_JSON);
601 err.error.err_json = cJSON_CreateObject();
602 if (err.error.err_json) {
603 cJSON_AddStringToObject(err.error.err_json, "err",
604 "Cannot append: would exceed fixed size");
605 }
606 return err;
607 }
608 } else {
609 char *new_data = cwist_sstring_reserve(str, new_size, current_len);
610 if (!new_data) {
611 cwist_error_t err = make_error(CWIST_ERR_JSON);
612 err.error.err_json = cJSON_CreateObject();
613 if (err.error.err_json) {
614 cJSON_AddStringToObject(err.error.err_json, "err", "Cannot append: memory full");
615 }
616 return err;
617 }
618 str->data = new_data;
619 str->borrows_buffer = false;
620 str->size = new_size;
621 }
622
623 char *ptr = str->data + current_len;
624 for (size_t i = 0; i < input_len; i++) {
625 switch (data[i]) {
626 case '<':
627 memcpy(ptr, "&lt;", 4);
628 ptr += 4;
629 break;
630 case '>':
631 memcpy(ptr, "&gt;", 4);
632 ptr += 4;
633 break;
634 case '&':
635 memcpy(ptr, "&amp;", 5);
636 ptr += 5;
637 break;
638 case '"':
639 memcpy(ptr, "&quot;", 6);
640 ptr += 6;
641 break;
642 case '\'':
643 memcpy(ptr, "&#39;", 5);
644 ptr += 5;
645 break;
646 default: *ptr++ = data[i]; break;
647 }
648 }
649
650 *ptr = '\0';
651 str->size = (size_t)(ptr - str->data);
652
653 cwist_error_t err = make_error(CWIST_ERR_INT8);
654 err.error.err_i8 = ERR_SSTRING_OKAY;
655 return err;
656}
