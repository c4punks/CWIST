49static char *cwist_sstring_reserve(cwist_sstring *str, size_t needed, size_t preserve_len) {
50 if (!str->borrows_buffer && str->data && str->capacity >= needed) return str->data;
51
52 size_t new_cap = str->capacity ? str->capacity : 16;
53 while (new_cap < needed) {
54 if (new_cap > (SIZE_MAX - 1) / 2) {
55 new_cap = needed;
56 break;
57 }
58 new_cap *= 2;
59 }
60
61 char *result;
62 if (str->borrows_buffer) {
63 result = (char *)cwist_alloc(new_cap + 1);
64 if (result) {
65 if (str->data && preserve_len > 0) memcpy(result, str->data, preserve_len);
66 str->base = result;
67 str->capacity = new_cap;
68 }
69 return result;
70 }
71
72 char *orig = str->base ? str->base : str->data;
73 const size_t lead = (str->base && str->data) ? (size_t)(str->data - str->base) : 0;
74 char *grown = (char *)cwist_realloc(orig, lead + new_cap + 1);
75 if (!grown) return NULL;
76 str->base = grown;
77 str->capacity = new_cap;
78 return grown + lead;
79}

675 }

677 size_t current_len = sstring_content_len(str);

684 err.error.err_json = cJSON_CreateObject();

698 }

701 str->data = new_data;
702 str->borrows_buffer = false;
703 str->size = new_size;
704 }
705
706 char *ptr = str->data + current_len;
707 for (size_t i = 0; i < input_len; i++) {
708 switch (data[i]) {
709 case '<':
710 memcpy(ptr, "&lt;", 4);
711 ptr += 4;
712 break;
713 case '>':
714 memcpy(ptr, "&gt;", 4);
715 ptr += 4;
716 break;
717 case '&':
718 memcpy(ptr, "&amp;", 5);
719 ptr += 5;
720 break;
721 case '"':
722 memcpy(ptr, "&quot;", 6);
723 ptr += 6;
724 break;
725 case '\'':
726 memcpy(ptr, "&#39;", 5);
727 ptr += 5;
728 break;
729 default: *ptr++ = data[i]; break;
730 }
731 }
732
733 *ptr = '\0';
734 str->size = (size_t)(ptr - str->data);
735
736 cwist_error_t err = make_error(CWIST_ERR_INT8);
737 err.error.err_i8 = ERR_SSTRING_OKAY;

910cwist_sstring *cwist_sstring_substr(cwist_sstring *str, int start, int length) {
911 if (!str || !str->data || start < 0 || length < 0) return NULL;
912
913 size_t current_len = strlen(str->data);
914 if ((size_t)start >= current_len) return NULL;
915
916 // Adjust length if it goes beyond end
917 if ((size_t)length > current_len - (size_t)start) {
918 length = (int)(current_len - (size_t)start);
919 }
920
921 cwist_sstring *sub = cwist_sstring_create();
922 if (!sub) return NULL;
923
924 sub->data = (char *)cwist_alloc(length + 1);
925 if (!sub->data) {
926 cwist_sstring_destroy(sub);
927 return NULL;
928 }
929 sub->base = sub->data;
930 sub->capacity = length;
931 sub->size = length;
932
933 memcpy(sub->data, str->data + start, length);
934 sub->data[length] = '\0';
935
936 return sub;
