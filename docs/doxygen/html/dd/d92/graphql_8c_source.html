81static int gql_hex_digit(char c) {
82 if (c >= '0' && c <= '9') return c - '0';
83 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
84 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
85 return -1;

88static int gql_hex_quad(const char *p, unsigned long *out) {
89 unsigned long value = 0;
90 for (int i = 0; i < 4; i++) {
91 int digit = gql_hex_digit(p[i]);
92 if (digit < 0) return -1;
93 value = value * 16 + (unsigned long)digit;
94 }
95 *out = value;

99static void gql_utf8_append(char *out, size_t *pos, unsigned long cp) {
100 if (cp <= 0x7f) {
101 out[(*pos)++] = (char)cp;
102 } else if (cp <= 0x7ff) {
103 out[(*pos)++] = (char)(0xc0 | (cp >> 6));
104 out[(*pos)++] = (char)(0x80 | (cp & 0x3f));
105 } else if (cp <= 0xffff) {
106 out[(*pos)++] = (char)(0xe0 | (cp >> 12));
107 out[(*pos)++] = (char)(0x80 | ((cp >> 6) & 0x3f));
108 out[(*pos)++] = (char)(0x80 | (cp & 0x3f));
109 } else {
110 out[(*pos)++] = (char)(0xf0 | (cp >> 18));
111 out[(*pos)++] = (char)(0x80 | ((cp >> 12) & 0x3f));
112 out[(*pos)++] = (char)(0x80 | ((cp >> 6) & 0x3f));
113 out[(*pos)++] = (char)(0x80 | (cp & 0x3f));
114 }

117static char *gql_decode_string(const char *src, size_t len) {
118 char *out = cwist_alloc(len + 1);
119 if (!out) return NULL;
120 size_t pos = 0;
121 for (size_t i = 0; i < len; i++) {
122 unsigned char c = (unsigned char)src[i];
123 if (c < 0x20) {
124 cwist_free(out);
125 return NULL;
126 }
127 if (c != '\\') {
128 out[pos++] = (char)c;
129 continue;
130 }
131 if (++i >= len) {
132 cwist_free(out);
133 return NULL;
134 }
135 char escape = src[i];
136 const char *simple = NULL;
137 switch (escape) {
138 case '"': simple = "\""; break;
139 case '\\': simple = "\\"; break;
140 case '/': simple = "/"; break;
141 case 'b': simple = "\b"; break;
142 case 'f': simple = "\f"; break;
143 case 'n': simple = "\n"; break;
144 case 'r': simple = "\r"; break;
145 case 't': simple = "\t"; break;
146 case 'u': {
147 if (i + 4 >= len || gql_hex_quad(src + i + 1, &(unsigned long){0}) != 0) {
148 cwist_free(out);
149 return NULL;
150 }
151 unsigned long cp = 0;
152 (void)gql_hex_quad(src + i + 1, &cp);
153 i += 4;
154 if (cp >= 0xd800 && cp <= 0xdbff) {
155 unsigned long low = 0;
156 if (i + 6 >= len || src[i + 1] != '\\' || src[i + 2] != 'u' ||
157 gql_hex_quad(src + i + 3, &low) != 0 || low < 0xdc00 || low > 0xdfff) {
158 cwist_free(out);
159 return NULL;
160 }
161 cp = 0x10000 + ((cp - 0xd800) << 10) + (low - 0xdc00);
162 i += 6;
163 } else if (cp >= 0xdc00 && cp <= 0xdfff) {
164 cwist_free(out);
165 return NULL;
166 }
167 gql_utf8_append(out, &pos, cp);
168 continue;
169 }
170 default: cwist_free(out); return NULL;
171 }
172 out[pos++] = *simple;
173 }
174 out[pos] = '\0';
175 return out;
176}

301static const char *skip_ws_comments(const char *p) {

305 } else if (*p == '#') {

327 cJSON *errors) {
328 *out_args = NULL;
329 p = skip_ws_comments(p);
330 if (*p != '(') return p;
331 p++;
332
333 cJSON *args = cJSON_CreateObject();
334 if (!args) {
335 gql_add_error(errors, "Internal memory error parsing arguments");
336 return p;
337 }
338
339 bool closed = false;
340 while (*p) {
341 p = skip_ws_comments(p);
342 if (*p == ')') {
343 closed = true;
344 p++;
345 break;
346 }
347 if (*p == '\0') break;
348
349 const char *name_start = p;
350 if (!gql_name_char(*p, true)) {
351 gql_add_error(errors, "Invalid argument name");
352 cJSON_Delete(args);
353 return p;
354 }

356 char *arg_name = gql_name_dup(name_start, (size_t)(p - name_start));
357 if (!arg_name) {
358 gql_add_error(errors, "Internal memory error parsing argument name");
359 cJSON_Delete(args);
360 return p;
361 }
362
363 p = skip_ws_comments(p);
364 if (*p != ':') {
365 gql_add_error(errors, "Expected ':' after argument name");
366 cwist_free(arg_name);
367 cJSON_Delete(args);
368 return p;
369 }
370 p++;
371 p = skip_ws_comments(p);
372
373 cJSON *val = NULL;
374 if (*p == '"') {
375 p++;
376 const char *str_start = p;
377 while (*p && *p != '"') {
378 if (*p == '\\' && *(p + 1))
379 p += 2;
380 else
381 p++;
382 }
383 if (*p != '"') {
384 gql_add_error(errors, "Unterminated string in argument");
385 cwist_free(arg_name);
386 cJSON_Delete(args);
387 return p;
388 }
389 char *decoded = gql_decode_string(str_start, (size_t)(p - str_start));
390 if (decoded) {
391 val = cJSON_CreateString(decoded);
392 cwist_free(decoded);
393 } else {
394 gql_add_error(errors, "Invalid string escape sequence");
395 }
396 p++;
397 } else if (*p == '$') {
398 p++;
399 const char *var_start = p;
400 if (!gql_name_char(*p, true)) {
401 gql_add_error(errors, "Invalid variable name");
402 cwist_free(arg_name);
403 cJSON_Delete(args);
404 return p;
405 }
406 while (*p && gql_name_char(*p, false)) p++;
407 char *var_name = gql_name_dup(var_start, (size_t)(p - var_start));
408 if (!var_name) {
409 gql_add_error(errors, "Internal memory error parsing variable name");
410 cwist_free(arg_name);
411 cJSON_Delete(args);
412 return p;
413 }
414 if (variables && cJSON_IsObject(variables)) {
415 cJSON *v = cJSON_GetObjectItemCaseSensitive(variables, var_name);
416 if (v) val = cJSON_Duplicate(v, true);
417 }
418 if (!val) val = cJSON_CreateNull();
419 cwist_free(var_name);
420 } else if (isdigit((unsigned char)*p) || *p == '-') {
421 char *endptr = NULL;
422 double num = strtod(p, &endptr);
423 val = cJSON_CreateNumber(num);
424 p = endptr;
425 } else if (strncmp(p, "true", 4) == 0 && !gql_name_char(p[4], false)) {
426 val = cJSON_CreateBool(true);
427 p += 4;
428 } else if (strncmp(p, "false", 5) == 0 && !gql_name_char(p[5], false)) {
429 val = cJSON_CreateBool(false);
430 p += 5;
431 } else if (strncmp(p, "null", 4) == 0 && !gql_name_char(p[4], false)) {
432 val = cJSON_CreateNull();
433 p += 4;
434 } else {
435 gql_add_error(errors, "Unsupported argument value format");
436 cwist_free(arg_name);
437 cJSON_Delete(args);
438 return p;
439 }
440
441 if (!val) {
442 cwist_free(arg_name);
443 cJSON_Delete(args);
444 return p;
445 }

447 cwist_free(arg_name);
448 }

473static const char *parse_selection_set(const char *p, graphql_field_t *schema_fields,
474 const cJSON *variables, cJSON *parent_data, cJSON *errors,
475 unsigned int depth, bool validate_only) {
476 if (depth > CWIST_GRAPHQL_MAX_DEPTH) {
477 gql_add_error(errors, "Query selection depth exceeds maximum allowed");
478 return p;
479 }
480
481 p = skip_ws_comments(p);
482 if (*p != '{') return p;
483 p++;
484
485 size_t field_count = 0;
486 bool closed = false;
487 while (*p) {
488 p = skip_ws_comments(p);
489 if (*p == '}') {
490 closed = true;
491 p++;
492 break;
493 }
494 if (*p == '\0') break;
495 if (++field_count > CWIST_GRAPHQL_MAX_FIELDS) {
496 gql_add_error(errors, "Field count limit exceeded");
497 break;
498 }
499
500 const char *name_start = p;
501 if (!gql_name_char(*p, true)) {
502 gql_add_error(errors, "Invalid field name syntax");
503 break;
504 }
505 while (*p && gql_name_char(*p, false)) p++;
506 size_t name_len = (size_t)(p - name_start);
507 char *token1 = gql_name_dup(name_start, name_len);
508 char *field_name = token1 ? gql_name_dup(name_start, name_len) : NULL;
509 char *output_key = token1 ? gql_name_dup(name_start, name_len) : NULL;
510 if (!token1 || !field_name || !output_key) {
511 gql_add_error(errors, "Internal memory error parsing field name");
512 cwist_free(token1);
513 cwist_free(field_name);
514 cwist_free(output_key);
515 break;
516 }
517
518 p = skip_ws_comments(p);
519 if (*p == ':') {
520 p++;
521 p = skip_ws_comments(p);
522 const char *field_start = p;
523 if (!gql_name_char(*p, true)) {
524 gql_add_error(errors, "Invalid field name after alias");
525 cwist_free(token1);
526 cwist_free(field_name);
527 cwist_free(output_key);
528 break;
529 }
530 while (*p && gql_name_char(*p, false)) p++;
531 char *alias_name = gql_name_dup(field_start, (size_t)(p - field_start));
532 if (!alias_name) {
533 gql_add_error(errors, "Internal memory error parsing aliased field");
534 cwist_free(token1);
535 cwist_free(field_name);
536 cwist_free(output_key);
537 break;
538 }
539 cwist_free(field_name);
540 field_name = alias_name;
541 }
542
543 cJSON *args = NULL;
544 p = parse_arguments(p, variables, &args, errors);
545
546 cJSON *val = NULL;
547 graphql_field_t *f = gql_lookup(schema_fields, field_name);
548 if (validate_only) {
549 if (!f) {
550 size_t err_len = strlen(field_name) + 32;
551 char *err_buf = cwist_alloc(err_len);
552 if (err_buf) {
553 snprintf(err_buf, err_len, "Cannot query field '%s' on schema", field_name);
554 gql_add_error(errors, err_buf);
555 cwist_free(err_buf);
556 }
557 }
558 val = cJSON_CreateObject();
559 } else if (!f) {
560 size_t err_len = strlen(field_name) + 32;
561 char *err_buf = cwist_alloc(err_len);
562 if (err_buf) {
563 snprintf(err_buf, err_len, "Cannot query field '%s' on schema", field_name);
564 gql_add_error(errors, err_buf);
565 cwist_free(err_buf);
566 }
567 val = cJSON_CreateNull();
568 } else {
569 val = f->fn(args, variables, f->ctx);
570 if (!val) val = cJSON_CreateNull();
571 }
572 if (args) cJSON_Delete(args);
573
574 p = skip_ws_comments(p);
575 if (*p == '{') {
576 if (cJSON_IsObject(val)) {
577 p = parse_selection_set(p, schema_fields, variables, val, errors, depth + 1,
578 validate_only);
579 } else {
580 int brace_count = 0;
581 while (*p) {
582 if (*p == '{')
583 brace_count++;
584 else if (*p == '}') {
585 brace_count--;
586 if (brace_count == 0) {
587 p++;
588 break;
589 }
590 }
591 p++;
592 }
593 }
594 }
595
596 if (val) cJSON_AddItemToObject(parent_data, output_key, val);
597 cwist_free(token1);
598 cwist_free(field_name);
599 cwist_free(output_key);
600 }
601
602 if (!closed) gql_add_error(errors, "Unterminated selection set");
603 return p;
604}

618cwist_error_t cwist_graphql_execute(cwist_graphql_schema_t *schema, const char *request_json,
619 cwist_sstring **out_json) {
620 if (!schema || !request_json || !out_json) return gql_error(-1);
621 *out_json = NULL;
622
623 cJSON *request = cJSON_Parse(request_json);
624 cJSON *root = cJSON_CreateObject();
625 cJSON *data = cJSON_CreateObject();
626 cJSON *errors = cJSON_CreateArray();
627
628 if (!request || !root || !data || !errors) {
629 cJSON_Delete(request);
630 cJSON_Delete(root);
631 cJSON_Delete(data);
632 cJSON_Delete(errors);
633 return gql_error(-1);
634 }
635 cJSON_AddItemToObject(root, "data", data);
636 cJSON_AddItemToObject(root, "errors", errors);
637
638 cJSON *query = cJSON_GetObjectItemCaseSensitive(request, "query");
639 cJSON *variables = cJSON_GetObjectItemCaseSensitive(request, "variables");
640
641 if (!cJSON_IsString(query) || !query->valuestring) {
642 gql_add_error(errors, "GraphQL request must contain a string 'query'");
643 } else if (strlen(query->valuestring) > CWIST_GRAPHQL_MAX_QUERY_SIZE) {
644 gql_add_error(errors, "Query string exceeds maximum allowed size");
645 } else {
646 const char *p = skip_ws_comments(query->valuestring);
647 gql_op_type_t op_type = GQL_OP_QUERY;
648
649 if (strncmp(p, "mutation", 8) == 0 && !gql_name_char(p[8], false)) {
650 op_type = GQL_OP_MUTATION;
651 p += 8;
652 p = skip_ws_comments(p);
653 /* Skip optional mutation name / variable defs until '{' */
654 while (*p && *p != '{') p++;
655 } else if (strncmp(p, "query", 5) == 0 && !gql_name_char(p[5], false)) {
656 op_type = GQL_OP_QUERY;
657 p += 5;
658 p = skip_ws_comments(p);
659 while (*p && *p != '{') p++;
660 }
661
662 if (*p != '{') {
663 gql_add_error(errors, "Expected '{' to begin selection set");
664 } else {
665 graphql_field_t *target_schema =
666 (op_type == GQL_OP_MUTATION) ? schema->mutations : schema->queries;
667 cJSON *validated = cJSON_CreateObject();
668 if (!validated) {
669 gql_add_error(errors, "Internal memory error validating query");
670 } else {
671 const char *end =
672 parse_selection_set(p, target_schema, variables, validated, errors, 1, true);
673 end = skip_ws_comments(end);
674 if (*end) gql_add_error(errors, "Unexpected trailing content after operation");
675 if (cJSON_GetArraySize(errors) == 0)
676 (void)parse_selection_set(p, target_schema, variables, data, errors, 1, false);
677 cJSON_Delete(validated);
678 }
679 }
680 }
681
682 if (cJSON_GetArraySize(errors) == 0) {
683 cJSON_DeleteItemFromObjectCaseSensitive(root, "errors");
684 }
685
686 char *printed = cJSON_PrintUnformatted(root);
687 cJSON_Delete(request);
688 cJSON_Delete(root);
689
690 if (!printed) return gql_error(-1);
691
692 *out_json = cwist_sstring_create();
693 if (!*out_json) {
694 free(printed);
695 return gql_error(-1);
696 }
697 cwist_sstring_assign(*out_json, printed);
698 free(printed);
699
700 return gql_error(0);
701}

719bool cwist_graphql_parse_subscription(const char *query, const cJSON *variables, char *field_out,
720 size_t field_cap, cJSON **args_out) {
721 if (!query || !field_out || field_cap == 0 || !args_out) return false;
722 *args_out = NULL;
723 field_out[0] = '\0';
724
725 const char *p = skip_ws_comments(query);
726 if (strncmp(p, "subscription", 12) != 0 || gql_name_char(p[12], false)) return false;
727 p += 12;
728 p = skip_ws_comments(p);
729 /* Skip optional operation name / variable definitions until '{' (same
730 * treatment as the query/mutation keyword in cwist_graphql_execute). */
731 while (*p && *p != '{') p++;
732 if (*p != '{') return false;
733 p++;
734 p = skip_ws_comments(p);
735 if (*p == '}' || !gql_name_char(*p, true)) return false;
736
737 const char *name_start = p;
738 while (*p && gql_name_char(*p, false)) p++;
739 size_t name_len = (size_t)(p - name_start);
740 char *token = gql_name_dup(name_start, name_len);
741 if (!token) return false;
742
743 p = skip_ws_comments(p);
744 if (*p == ':') {
745 p++;
746 p = skip_ws_comments(p);
747 name_start = p;
748 if (!gql_name_char(*p, true)) {
749 cwist_free(token);
750 return false;
751 }
752 while (*p && gql_name_char(*p, false)) p++;
753 char *alias = gql_name_dup(name_start, (size_t)(p - name_start));
754 if (!alias) {
755 cwist_free(token);
756 return false;
757 }
758 cwist_free(token);
759 token = alias;
760 }
761
762 size_t out_len = strlen(token);
763 if (out_len >= field_cap) {
764 cwist_free(token);
765 return false;
766 }
767 memcpy(field_out, token, out_len + 1);
768 cwist_free(token);
769
770 cJSON *errors = cJSON_CreateArray();
771 if (!errors) return false;
772 parse_arguments(p, variables, args_out, errors);
773 cJSON_Delete(errors);
774 /* *args_out stays NULL when the field carries no arguments. */
775 return true;
776}

794

804 cwist_sstring_destroy(json);
