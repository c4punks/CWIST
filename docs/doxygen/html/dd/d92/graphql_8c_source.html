124static bool add_field(graphql_field_t **head, const char *field, cwist_graphql_resolver_fn resolver,
125 void *ctx) {
126 if (!head || !gql_name_valid(field) || !resolver) return false;
127 for (graphql_field_t *f = *head; f; f = f->next) {

221static const char *parse_arguments(const char *p, const cJSON *variables, cJSON **out_args,
222 cJSON *errors) {
223 *out_args = NULL;
224 p = skip_ws_comments(p);
225 if (*p != '(') return p;
226 p++; /* skip '(' */
227
228 cJSON *args = cJSON_CreateObject();
229 if (!args) {
230 gql_add_error(errors, "Internal memory error parsing arguments");
231 return p;
232 }
233
234 while (*p) {
235 p = skip_ws_comments(p);
236 if (*p == ')' || *p == '\0') {
237 if (*p == ')') p++;
238 break;
239 }
240
241 /* Read argument name */
242 const char *name_start = p;
243 if (!gql_name_char(*p, true)) {
244 gql_add_error(errors, "Invalid argument name");
245 cJSON_Delete(args);
246 return p;
247 }
248 while (*p && gql_name_char(*p, false)) p++;
249 size_t name_len = (size_t)(p - name_start);
250 char arg_name[128];
251 if (name_len >= sizeof(arg_name)) name_len = sizeof(arg_name) - 1;
252 memcpy(arg_name, name_start, name_len);
253 arg_name[name_len] = '\0';
254
255 p = skip_ws_comments(p);
256 if (*p != ':') {
257 gql_add_error(errors, "Expected ':' after argument name");
258 cJSON_Delete(args);
259 return p;
260 }
261 p++; /* skip ':' */
262 p = skip_ws_comments(p);
263
264 /* Parse argument value */
265 cJSON *val = NULL;
266 if (*p == '"') {
267 p++;
268 const char *str_start = p;
269 while (*p && *p != '"') {
270 if (*p == '\\' && *(p + 1))
271 p += 2;
272 else
273 p++;
274 }
275 if (*p != '"') {
276 gql_add_error(errors, "Unterminated string in argument");
277 cJSON_Delete(args);
278 return p;
279 }
280 size_t str_len = (size_t)(p - str_start);
281 cwist_scratch_t str_val_s CWIST_SCRATCH_DEFER = {0};
282 char *str_val = cwist_scratch_alloc(&str_val_s, str_len + 1);
283 if (str_val) {
284 memcpy(str_val, str_start, str_len);
285 str_val[str_len] = '\0';
286 val = cJSON_CreateString(str_val);
287 }
288 p++; /* skip closing '"' */
289 } else if (*p == '$') { /* Variable reference */
290 p++;
291 const char *var_start = p;
292 while (*p && gql_name_char(*p, false)) p++;
293 size_t var_len = (size_t)(p - var_start);
294 char var_name[128];
295 if (var_len >= sizeof(var_name)) var_len = sizeof(var_name) - 1;
296 memcpy(var_name, var_start, var_len);
297 var_name[var_len] = '\0';
298
299 if (variables && cJSON_IsObject(variables)) {
300 cJSON *v = cJSON_GetObjectItemCaseSensitive(variables, var_name);
301 if (v) val = cJSON_Duplicate(v, true);
302 }
303 if (!val) val = cJSON_CreateNull();
304 } else if (isdigit((unsigned char)*p) || *p == '-') {
305 char *endptr = NULL;
306 double num = strtod(p, &endptr);
307 val = cJSON_CreateNumber(num);
308 p = endptr;
309 } else if (strncmp(p, "true", 4) == 0 && !gql_name_char(p[4], false)) {
310 val = cJSON_CreateBool(true);
311 p += 4;
312 } else if (strncmp(p, "false", 5) == 0 && !gql_name_char(p[5], false)) {
313 val = cJSON_CreateBool(false);
314 p += 5;
315 } else if (strncmp(p, "null", 4) == 0 && !gql_name_char(p[4], false)) {
316 val = cJSON_CreateNull();
317 p += 4;
318 } else {
319 gql_add_error(errors, "Unsupported argument value format");
320 cJSON_Delete(args);
321 return p;
322 }
323
324 if (val) {
325 cJSON_AddItemToObject(args, arg_name, val);
326 }
327 }
328
329 *out_args = args;
330 return p;
331}

347static const char *parse_selection_set(const char *p, graphql_field_t *schema_fields,
348 const cJSON *variables, cJSON *parent_data, cJSON *errors,
349 unsigned int depth) {
350 if (depth > CWIST_GRAPHQL_MAX_DEPTH) {
351 gql_add_error(errors, "Query selection depth exceeds maximum allowed");
352 return p;
353 }
354
355 p = skip_ws_comments(p);
356 if (*p != '{') return p;
357 p++; /* skip '{' */
358
359 size_t field_count = 0;
360 while (*p) {
361 p = skip_ws_comments(p);
362 if (*p == '}' || *p == '\0') {
363 if (*p == '}') p++;
364 break;
365 }
366
367 if (++field_count > CWIST_GRAPHQL_MAX_FIELDS) {
368 gql_add_error(errors, "Field count limit exceeded");
369 break;
370 }
371
372 /* Read Alias or Field Name */
373 const char *name_start = p;
374 if (!gql_name_char(*p, true)) {
375 gql_add_error(errors, "Invalid field name syntax");
376 break;
377 }
378 while (*p && gql_name_char(*p, false)) p++;
379 size_t name_len = (size_t)(p - name_start);
380 char token1[128];
381 if (name_len >= sizeof(token1)) name_len = sizeof(token1) - 1;
382 memcpy(token1, name_start, name_len);
383 token1[name_len] = '\0';
384
385 char field_name[128];
386 char output_key[128];
387 strcpy(output_key, token1);
388 strcpy(field_name, token1);
389
390 p = skip_ws_comments(p);
391 if (*p == ':') { /* Alias: output_key: field_name */
392 p++;
393 p = skip_ws_comments(p);
394 const char *field_start = p;
395 if (!gql_name_char(*p, true)) {
396 gql_add_error(errors, "Invalid field name after alias");
397 break;
398 }
399 while (*p && gql_name_char(*p, false)) p++;
400 size_t f_len = (size_t)(p - field_start);
401 if (f_len >= sizeof(field_name)) f_len = sizeof(field_name) - 1;
402 memcpy(field_name, field_start, f_len);
403 field_name[f_len] = '\0';
404 }
405
406 /* Parse Field Arguments if present */
407 cJSON *args = NULL;
408 p = parse_arguments(p, variables, &args, errors);
409
410 /* Resolve Field Value */
411 cJSON *val = NULL;
412 graphql_field_t *f = gql_lookup(schema_fields, field_name);
413 if (!f) {
414 char err_buf[256];
415 snprintf(err_buf, sizeof(err_buf), "Cannot query field '%s' on schema", field_name);
416 gql_add_error(errors, err_buf);
417 val = cJSON_CreateNull();
418 } else {
419 val = f->fn(args, variables, f->ctx);
420 if (!val) {
421 val = cJSON_CreateNull();
422 }
423 }
424 if (args) cJSON_Delete(args);
425
426 /* Parse nested selection set if present and value is object */
427 p = skip_ws_comments(p);
428 if (*p == '{') {
429 if (cJSON_IsObject(val)) {
430 p = parse_selection_set(p, schema_fields, variables, val, errors, depth + 1);
431 } else {
432 /* Skip sub-selection if non-object or null */
433 int brace_count = 0;
434 while (*p) {
435 if (*p == '{')
436 brace_count++;
437 else if (*p == '}') {
438 brace_count--;
439 if (brace_count == 0) {
440 p++;
441 break;
442 }
443 }
444 p++;
445 }
446 }
447 }
448
449 cJSON_AddItemToObject(parent_data, output_key, val);
450 }
451
452 return p;
453}

467cwist_error_t cwist_graphql_execute(cwist_graphql_schema_t *schema, const char *request_json,
468 cwist_sstring **out_json) {
469 if (!schema || !request_json || !out_json) return gql_error(-1);
470 *out_json = NULL;
471
472 cJSON *request = cJSON_Parse(request_json);
473 cJSON *root = cJSON_CreateObject();
474 cJSON *data = cJSON_CreateObject();
475 cJSON *errors = cJSON_CreateArray();
476
477 if (!request || !root || !data || !errors) {
478 cJSON_Delete(request);
479 cJSON_Delete(root);
480 cJSON_Delete(data);
481 cJSON_Delete(errors);
482 return gql_error(-1);
483 }
484 cJSON_AddItemToObject(root, "data", data);
485 cJSON_AddItemToObject(root, "errors", errors);
486
487 cJSON *query = cJSON_GetObjectItemCaseSensitive(request, "query");
488 cJSON *variables = cJSON_GetObjectItemCaseSensitive(request, "variables");
489
490 if (!cJSON_IsString(query) || !query->valuestring) {
491 gql_add_error(errors, "GraphQL request must contain a string 'query'");
492 } else if (strlen(query->valuestring) > CWIST_GRAPHQL_MAX_QUERY_SIZE) {
493 gql_add_error(errors, "Query string exceeds maximum allowed size");
494 } else {
495 const char *p = skip_ws_comments(query->valuestring);
496 gql_op_type_t op_type = GQL_OP_QUERY;
497
498 if (strncmp(p, "mutation", 8) == 0 && !gql_name_char(p[8], false)) {
499 op_type = GQL_OP_MUTATION;
500 p += 8;
501 p = skip_ws_comments(p);
502 /* Skip optional mutation name / variable defs until '{' */
503 while (*p && *p != '{') p++;
504 } else if (strncmp(p, "query", 5) == 0 && !gql_name_char(p[5], false)) {
505 op_type = GQL_OP_QUERY;
506 p += 5;
507 p = skip_ws_comments(p);
508 while (*p && *p != '{') p++;
509 }
510
511 if (*p != '{') {
512 gql_add_error(errors, "Expected '{' to begin selection set");
513 } else {
514 graphql_field_t *target_schema =
515 (op_type == GQL_OP_MUTATION) ? schema->mutations : schema->queries;
516 parse_selection_set(p, target_schema, variables, data, errors, 1);
517 }
518 }
519
520 if (cJSON_GetArraySize(errors) == 0) {
521 cJSON_DeleteItemFromObjectCaseSensitive(root, "errors");
522 }
523
524 char *printed = cJSON_PrintUnformatted(root);
525 cJSON_Delete(request);
526 cJSON_Delete(root);
527
528 if (!printed) return gql_error(-1);
529
530 *out_json = cwist_sstring_create();
531 if (!*out_json) {
532 free(printed);
533 return gql_error(-1);
534 }
535 cwist_sstring_assign(*out_json, printed);
536 free(printed);
537
538 return gql_error(0);
539}

557bool cwist_graphql_parse_subscription(const char *query, const cJSON *variables, char *field_out,
558 size_t field_cap, cJSON **args_out) {
559 if (!query || !field_out || field_cap == 0 || !args_out) return false;
560 *args_out = NULL;
561 field_out[0] = '\0';
562
563 const char *p = skip_ws_comments(query);
564 if (strncmp(p, "subscription", 12) != 0 || gql_name_char(p[12], false)) return false;
565 p += 12;
566 p = skip_ws_comments(p);
567 /* Skip optional operation name / variable definitions until '{' (same
568 * treatment as the query/mutation keyword in cwist_graphql_execute). */
569 while (*p && *p != '{') p++;
570 if (*p != '{') return false;
571 p++;
572 p = skip_ws_comments(p);
573 if (*p == '}' || !gql_name_char(*p, true)) return false;
574
575 char token[128];
576 const char *name_start = p;
577 while (*p && gql_name_char(*p, false)) p++;
578 size_t name_len = (size_t)(p - name_start);
579 if (name_len >= sizeof(token)) name_len = sizeof(token) - 1;
580 memcpy(token, name_start, name_len);
581 token[name_len] = '\0';
582
583 p = skip_ws_comments(p);
584 if (*p == ':') { /* Alias: the resolvable name follows the alias. */
585 p++;
586 p = skip_ws_comments(p);
587 name_start = p;
588 if (!gql_name_char(*p, true)) return false;
589 while (*p && gql_name_char(*p, false)) p++;
590 name_len = (size_t)(p - name_start);
591 if (name_len >= sizeof(token)) name_len = sizeof(token) - 1;
592 memcpy(token, name_start, name_len);
593 token[name_len] = '\0';
594 }
595
596 size_t out_len = strlen(token);
597 if (out_len >= field_cap) return false;
598 memcpy(field_out, token, out_len + 1);
599
600 cJSON *errors = cJSON_CreateArray();
601 if (!errors) return false;
602 parse_arguments(p, variables, args_out, errors);
603 cJSON_Delete(errors);
604 /* *args_out stays NULL when the field carries no arguments. */
605 return true;
606}
