24/* Arena maps carve from the request arena, which is a fixed 8 KiB
25 * generation: a query of a few KiB used to exhaust it, and every later
26 * key/value was dropped silently. Allocations now fall back to the heap,
27 * and every release path frees whatever the arena does not own. */
28static void *query_alloc(void *arena, size_t size) {
29 if (arena) {
30 void *p = cwist_arena_alloc((cwist_arena_t *)arena, size);
31 if (p) return p;
32 }
33 return cwist_alloc(size);
34}

36static void query_release(void *arena, void *ptr) {
37 if (!ptr) return;
38 if (arena && cwist_arena_owns((cwist_arena_t *)arena, ptr)) return;
39 cwist_free(ptr);
40}

117static void query_map_grow(cwist_query_map *map) {
118 if (map->size && map->count + 1 <= map->size - map->size / 4) return;
119 size_t new_size = map->size ? map->size * 2 : CWIST_QUERY_MAP_DEFAULT_SIZE;
120 if (new_size <= map->size || new_size > SIZE_MAX / sizeof(cwist_query_bucket *)) return;
121 cwist_query_bucket **buckets =
122 (cwist_query_bucket **)query_alloc(map->arena, new_size * sizeof(cwist_query_bucket *));
123 if (!buckets) return;
124 memset(buckets, 0, new_size * sizeof(cwist_query_bucket *));
125 for (size_t i = 0; i < map->size; i++) {
126 cwist_query_bucket *curr = map->buckets[i];
127 while (curr) {
128 cwist_query_bucket *next = curr->next;
129 size_t index = siphash24(curr->key, strlen(curr->key), map->seed) % new_size;
130 curr->next = buckets[index];
131 buckets[index] = curr;
132 curr = next;
133 }
134 }
135 query_release(map->arena, map->buckets);
136 map->buckets = buckets;
137 map->size = new_size;
138}

146void cwist_query_map_set(cwist_query_map *map, const char *key, const char *value) {
147 if (!map || !key || !value) return;
148 if (!map->size) {
149 query_map_grow(map);
150 if (!map->size) return;
151 }
152
153 uint64_t hash = siphash24(key, strlen(key), map->seed);
154 size_t index = hash % map->size;
155
156 cwist_query_bucket *curr = map->buckets[index];
157 while (curr) {
158 if (strcmp(curr->key, key) == 0) {
159 // Update existing; keep the old value if the copy fails
160 char *copy = query_strdup(map->arena, value);
161 if (!copy) return;
162 query_release(map->arena, curr->value);
163 curr->value = copy;
164 return;
165 }
166 curr = curr->next;
167 }
168
169 // Insert new
170 if (map->count + 1 > map->size - map->size / 4) {
171 query_map_grow(map);
172 index = hash % map->size;
173 }
174 cwist_query_bucket *node = (cwist_query_bucket *)query_alloc(map->arena, sizeof(cwist_query_bucket));
175 if (!node) return;
176 node->key = query_strdup(map->arena, key);
177 node->value = query_strdup(map->arena, value);
178 if (!node->key || !node->value) {
179 /* A NULL key in a bucket would crash the next strcmp(). */
180 query_release(map->arena, node->key);
181 query_release(map->arena, node->value);
182 query_release(map->arena, node);
183 return;
184 }
185 node->next = map->buckets[index];
186 map->buckets[index] = node;
187 map->count++;
188}

279static char *url_decode(void *arena, const char *src) {
280 if (!src) return NULL;
281
282 size_t len = strlen(src);
283 char *out = (char *)query_alloc(arena, len + 1);
284 if (!out) return NULL;
285
286 size_t j = 0;
287 size_t i = 0;
288 while (i < len) {
289 /* Copy the plain run up to the next '%' or '+' in one shot. */
290 size_t plain = strcspn(src + i, "%+");
291 if (plain > 0) {
292 memcpy(out + j, src + i, plain);
293 j += plain;
294 i += plain;
295 if (i >= len) break;
296 }
297 if (src[i] == '+') {
298 out[j++] = 0x20; /* SPACE (U+0020) */
299 i++;
300 } else if (i + 2 < len && isxdigit((unsigned char)src[i + 1]) &&
301 isxdigit((unsigned char)src[i + 2])) {
302 out[j++] = (char)((url_decode_hex[(unsigned char)src[i + 1]] << 4) |
303 url_decode_hex[(unsigned char)src[i + 2]]);
304 i += 3;
305 } else {
306 out[j++] = src[i];
307 i++;
308 }
309 }
310 out[j] = '\0';
311 return out;
312}

319void cwist_query_map_parse(cwist_query_map *map, const char *raw_query) {
320 if (!map || !raw_query || strlen(raw_query) == 0) return;
321
322 char *buffer = query_strdup(map->arena, raw_query);
323 if (!buffer) return;
324
325 /* Single pass over buffer splitting on '&' and '='; strtok_r's empty
326 * token collapsing is reproduced by skipping zero-length segments. */
327 char *seg = buffer;
328 char *end = buffer + strlen(buffer);
329 while (seg < end) {
330 char *amp = memchr(seg, '&', (size_t)(end - seg));
331 char *seg_end = amp ? amp : end;
332 if (amp) *amp = '\0';
333
334 char *token = seg;
335 seg = seg_end + (amp ? 1 : 0);
336
337 if (seg_end == token) continue; /* strtok_r skips empty tokens */
338
339 char *eq = memchr(token, '=', (size_t)(seg_end - token));
340 if (!eq) continue;
341
342 *eq = '\0';
343 const char *key_raw = token;
344 const char *value_raw = eq + 1;
345
346 char *key_dec = url_decode(map->arena, key_raw);
347 char *value_dec = url_decode(map->arena, value_raw);
348
349 if (key_dec && cwist_query_map_get(map, key_dec) == NULL) {
350 cwist_query_map_set(map, key_dec, value_dec ? value_dec : "");
351 }
352
353 query_release(map->arena, key_dec);
354 query_release(map->arena, value_dec);
355 }
356
357 query_release(map->arena, buffer);
358}
