147void quic_flow_control_update_rtt(quic_conn_fc_t *fc, uint64_t rtt_sample_us) {
148 uint64_t difference;
149
150 if (!fc || rtt_sample_us == 0) return;
151
152 if (!fc->rtt_initialized) {
153 fc->srtt_us = rtt_sample_us;
154 fc->rttvar_us = rtt_sample_us / 2;
155 fc->min_rtt_us = rtt_sample_us;
156 fc->rtt_initialized = true;
157 } else {
158 difference =
159 fc->srtt_us > rtt_sample_us ? fc->srtt_us - rtt_sample_us : rtt_sample_us - fc->srtt_us;
160 fc->rttvar_us = (3 * fc->rttvar_us + difference) / 4;
161 fc->srtt_us = (7 * fc->srtt_us + rtt_sample_us) / 8;
162 if (rtt_sample_us < fc->min_rtt_us) fc->min_rtt_us = rtt_sample_us;
163 }
164
165 quic_flow_control_evaluate_quality(fc);
166}

268int quic_flow_control_maybe_send_max_data(quic_conn_fc_t *fc, uint8_t *buf, size_t buf_len,
269 size_t *written, bool force) {
270 uint64_t maximum;
271 size_t value_size;
272
273 if (written) *written = 0;
