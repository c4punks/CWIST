149void quic_flow_control_update_rtt(quic_conn_fc_t *fc, uint64_t rtt_sample_us) {
150 uint64_t difference;
151
152 if (!fc || rtt_sample_us == 0) return;
153
154 if (!fc->rtt_initialized) {
155 fc->srtt_us = rtt_sample_us;
156 fc->rttvar_us = rtt_sample_us / 2;
157 fc->min_rtt_us = rtt_sample_us;
158 fc->rtt_initialized = true;
159 } else {
160 difference =
161 fc->srtt_us > rtt_sample_us ? fc->srtt_us - rtt_sample_us : rtt_sample_us - fc->srtt_us;
162 fc->rttvar_us = (3 * fc->rttvar_us + difference) / 4;
163 fc->srtt_us = (7 * fc->srtt_us + rtt_sample_us) / 8;
164 if (rtt_sample_us < fc->min_rtt_us) fc->min_rtt_us = rtt_sample_us;
165 }
166
167 quic_flow_control_evaluate_quality(fc);
168}

274 size_t value_size;
