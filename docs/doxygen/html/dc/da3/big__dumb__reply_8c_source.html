51static void bdr_key_init(void) {
52 int fd = open("/dev/urandom", O_RDONLY);
53 if (fd < 0) return;
54 size_t off = 0;
55 while (off < sizeof(bdr_key)) {
56 ssize_t n = read(fd, bdr_key + off, sizeof(bdr_key) - off);
57 if (n > 0) {
58 off += (size_t)n;
59 } else if (n < 0 && errno == EINTR) {
60 continue;
61 } else {
62 break;
63 }
64 }
65 close(fd);
66#if defined(__wasi__) || defined(__EMSCRIPTEN__) || defined(__APPLE__)
67 if (off < sizeof(bdr_key) && getentropy(bdr_key + off, sizeof(bdr_key) - off) == 0)
68 off = sizeof(bdr_key);
69#endif
70#ifdef CWIST_BAREMETAL
71 if (off < sizeof(bdr_key) && cwist_port_entropy(bdr_key + off, sizeof(bdr_key) - off) == 0)
72 off = sizeof(bdr_key);
73#endif
74 bdr_key_ok = off == sizeof(bdr_key);
75}

87static void bdr_blob_free_cb(void *ptr) {
88 bdr_blob_t *blob = (bdr_blob_t *)ptr;
89 if (!blob) return;
90 /* Read the fields before releasing: for inline blobs mem == blob, so the
91 * free below clobbers the very bytes the branch would inspect (glibc
92 * writes its tcache links into freed chunks). */
93 void *mem = blob->mem;
94 void (*free_fn)(void *) = blob->free_fn;
95 free_fn(mem);
96 if (mem != ptr) cwist_free(blob);
97}

153static void bdr_bytes_sub(cwist_bdr_t *bdr, size_t len) {
154 size_t cur = atomic_load_explicit(&bdr->current_bytes, memory_order_relaxed);
155 while (cur > 0) {
156 size_t next = cur >= len ? cur - len : 0;
157 if (atomic_compare_exchange_weak_explicit(&bdr->current_bytes, &cur, next,
158 memory_order_relaxed, memory_order_relaxed))
159 return;
160 }

231static bool bdr_entry_should_decay(const cwist_bdr_t *bdr, const bdr_entry_t *entry, time_t now) {
232 if (!bdr || !entry) return false;
233 int64_t created = atomic_load_explicit(&entry->created_at, memory_order_relaxed);
234 if (bdr->max_entry_age_sec > 0 && created > 0) {
235 if (now - (time_t)created > bdr->max_entry_age_sec) return true;
236 }
237 if (atomic_load_explicit(&entry->is_stable, memory_order_relaxed) && bdr->revalidate_hits > 0 &&
238 atomic_load_explicit(&entry->hits, memory_order_relaxed) >= bdr->revalidate_hits) {
239 return true;
240 }
241 return false;
242}

254static bdr_entry_t *bdr_find(cwist_bdr_t *bdr, uint64_t req_h) {
255 size_t idx = req_h % bdr->bucket_count;
256 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
257 while (curr) {
258 if (curr->request_hash == req_h &&
259 !atomic_load_explicit(&curr->retired, memory_order_acquire)) {
260 return curr;
261 }
262 curr = atomic_load_explicit(&curr->next, memory_order_acquire);
263 }
264 return NULL;
265}

279static bdr_entry_t *bdr_find_or_insert(cwist_bdr_t *bdr, uint64_t req_h) {
280 bdr_entry_t *found = bdr_find(bdr, req_h);
281 if (found) return found;
282
283 bdr_entry_t *entry = cwist_alloc(sizeof(bdr_entry_t));
284 if (!entry) return NULL;
285 entry->request_hash = req_h;
286 atomic_init(&entry->response_hash, 0);
287 atomic_init(&entry->is_stable, false);
288 atomic_init(&entry->retired, false);
289 atomic_init(&entry->blob, NULL);
290 atomic_init(&entry->hits, 0);
291 atomic_init(&entry->created_at, (int64_t)time(NULL));
292 entry->revalidate = NULL;
293 entry->revalidate_arg = NULL;
294 entry->retire_next = NULL;
295
296 size_t idx = req_h % bdr->bucket_count;
297 bdr_entry_t *head = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
298 do {
299 /* A concurrent insert of the same key wins; ours becomes garbage. */
300 bdr_entry_t *race = bdr_find(bdr, req_h);
301 if (race) {
302 cwist_free(entry);
303 return race;
304 }
305 atomic_store_explicit(&entry->next, head, memory_order_relaxed);
306 } while (!atomic_compare_exchange_weak_explicit(&bdr->buckets[idx], &head, entry,
307 memory_order_release, memory_order_acquire));
308 return entry;
309}

325static void bdr_retire_entry(cwist_bdr_t *bdr, size_t idx, bdr_entry_t *prev, bdr_entry_t *entry) {
326 atomic_store_explicit(&entry->retired, true, memory_order_release);
327 if (prev) {
328 atomic_store_explicit(&prev->next, atomic_load_explicit(&entry->next, memory_order_acquire),
329 memory_order_release);
330 } else {
331 atomic_store_explicit(&bdr->buckets[idx],
332 atomic_load_explicit(&entry->next, memory_order_acquire),
333 memory_order_release);
334 }
335 bdr_entry_publish(bdr, entry, NULL);
336 entry->retire_next = bdr->retired_entries;
337 bdr->retired_entries = entry;
338}

348static void bdr_sweep(cwist_bdr_t *bdr, size_t steps) {
349 if (!bdr || bdr->bucket_count == 0 || steps == 0) return;
350 time_t now = time(NULL);
351 for (size_t i = 0; i < steps; ++i) {
352 size_t idx = bdr->gc_cursor % bdr->bucket_count;
353 bdr->gc_cursor = (bdr->gc_cursor + 1) % bdr->bucket_count;
354 bdr_entry_t *prev = NULL;
355 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
356 while (curr) {
357 bdr_entry_t *next = atomic_load_explicit(&curr->next, memory_order_acquire);
358 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
359 bdr_entry_should_decay(bdr, curr, now)) {
360 bdr_retire_entry(bdr, idx, prev, curr);
361 } else {
362 prev = curr;
363 }
364 curr = next;
365 }
366 }

379static bool bdr_trim_oldest(cwist_bdr_t *bdr, time_t now) {
380 size_t victim_idx = SIZE_MAX;
381 bdr_entry_t *victim = NULL;
382 bdr_entry_t *victim_prev = NULL;
383 int64_t oldest = (int64_t)now;
384
385 for (size_t i = 0; i < bdr->bucket_count; ++i) {
386 bdr_entry_t *prev = NULL;
387 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[i], memory_order_acquire);
388 while (curr) {
389 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
390 atomic_load_explicit(&curr->blob, memory_order_relaxed)) {
391 int64_t created = atomic_load_explicit(&curr->created_at, memory_order_relaxed);
392 if (!victim || created < oldest) {
393 victim = curr;
394 victim_prev = prev;
395 victim_idx = i;
396 oldest = created;
397 }
398 }
399 prev = curr;
400 curr = atomic_load_explicit(&curr->next, memory_order_acquire);
401 }
402 }
403
404 if (!victim) return false;
405 bdr_retire_entry(bdr, victim_idx, victim_prev, victim);
406 return true;
407}

420static void bdr_check_ram(cwist_bdr_t *bdr) {
421#if defined(__EMSCRIPTEN__) || !defined(CWIST_DB)
422 /* No SQLite disk fallback in the WASM core or when the DB tier is
423 * compiled out (CWIST_DB=0); the in-RAM cache stands alone. */
424 (void)bdr;
425 return;
426#else
427 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_relaxed)) return;
428
429 if (cwist_is_ram_critical(CWIST_MIB(64))) {
430 printf("[BDR] Low RAM. Switching to Disk Cache.\n");
431 if (sqlite3_open("cwist_bdr_fallback.db", &bdr->disk_db) == SQLITE_OK) {
432 char *err = NULL;
433 sqlite3_exec(bdr->disk_db,
434 "CREATE TABLE IF NOT EXISTS bdr (hash INTEGER PRIMARY KEY, blob BLOB);",
435 NULL, NULL, &err);
436 if (err) sqlite3_free(err);
437
438 sqlite3_exec(bdr->disk_db, "BEGIN TRANSACTION;", NULL, NULL, NULL);
439 for (size_t i = 0; i < bdr->bucket_count; i++) {
440 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[i], memory_order_acquire);
441 while (curr) {
442 bdr_entry_t *next = atomic_load_explicit(&curr->next, memory_order_acquire);
443 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
444 atomic_load_explicit(&curr->is_stable, memory_order_relaxed)) {
445 bdr_blob_t *blob = atomic_load_explicit(&curr->blob, memory_order_acquire);
446 if (blob) {
447 sqlite3_stmt *stmt;
448 sqlite3_prepare_v2(bdr->disk_db,
449 "INSERT INTO bdr (hash, blob) VALUES (?, ?);", -1,
450 &stmt, NULL);
451 sqlite3_bind_int64(stmt, 1, curr->request_hash);
452 sqlite3_bind_blob(stmt, 2, blob->data, (int)blob->len, SQLITE_STATIC);
453 sqlite3_step(stmt);
454 sqlite3_finalize(stmt);
455 }
456 }
457 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed)) {
458 atomic_store_explicit(&curr->retired, true, memory_order_release);
459 bdr_entry_publish(bdr, curr, NULL);
460 curr->retire_next = bdr->retired_entries;
461 bdr->retired_entries = curr;
462 }
463 curr = next;
464 }
465 atomic_store_explicit(&bdr->buckets[i], NULL, memory_order_release);
466 }
467 sqlite3_exec(bdr->disk_db, "COMMIT;", NULL, NULL, NULL);
468 atomic_store_explicit(&bdr->is_disk_mode, true, memory_order_release);
469 }
470 }
471#endif
472}

484static void bdr_janitor_tick(cwist_bdr_t *bdr) {
485 uint64_t n = atomic_fetch_add_explicit(&bdr->put_count, 1, memory_order_relaxed) + 1;
486 if (n % BDR_JANITOR_PERIOD != 0) return;
487 if (pthread_mutex_trylock(&bdr->lock) != 0) return; /* another tick holds it */
488 bdr_check_ram(bdr);
489 bdr_sweep(bdr, BDR_GC_SWEEP);
490 if (bdr->max_bytes > 0) {
491 time_t now = time(NULL);
492 while (atomic_load_explicit(&bdr->current_bytes, memory_order_relaxed) > bdr->max_bytes) {
493 if (!bdr_trim_oldest(bdr, now)) break;
494 }
495 }
496 pthread_mutex_unlock(&bdr->lock);
497}

509cwist_bdr_t *cwist_bdr_create(void) {
510 pthread_once(&bdr_key_once, bdr_key_init);
511 if (!bdr_key_ok) return NULL;
512 cwist_bdr_t *bdr = cwist_alloc(sizeof(cwist_bdr_t));
513 if (!bdr) return NULL;
514 if (pthread_mutex_init(&bdr->lock, NULL) != 0) {
515 cwist_free(bdr);
516 return NULL;
517 }
518 bdr->bucket_count = BDR_BUCKETS;
519 bdr->buckets = cwist_alloc_array(BDR_BUCKETS, sizeof(bdr_entry_t *));
520 if (!bdr->buckets) {
521 pthread_mutex_destroy(&bdr->lock);
522 cwist_free(bdr);
523 return NULL;
524 }
525 bdr->latency_threshold_ms = 10;
526 atomic_init(&bdr->current_bytes, 0);
527 bdr->max_bytes = BDR_DEFAULT_MAX_BYTES;
528 bdr->max_entry_age_sec = BDR_DEFAULT_ENTRY_TTL;
529 bdr->revalidate_hits = BDR_DEFAULT_REVALIDATE_HITS;
530 bdr->gc_cursor = 0;
531 atomic_init(&bdr->put_count, 0);
532 bdr->retired_entries = NULL;
533 bdr->disk_db = NULL;
534 atomic_init(&bdr->is_disk_mode, false);
535 return bdr;
536}

547void cwist_bdr_destroy(cwist_bdr_t *bdr) {
548 if (!bdr) return;
549 pthread_mutex_lock(&bdr->lock);
550 for (size_t i = 0; i < bdr->bucket_count; i++) {
551 bdr_entry_t *curr = bdr->buckets[i];
552 while (curr) {
553 bdr_entry_t *next = curr->next;
554 if (curr == NULL) break;
555 if (curr->blob != NULL)
556 bdr_blob_free_cb(atomic_load_explicit(&curr->blob, memory_order_relaxed));
557 cwist_free(curr);
558 curr = next;
559 }
560 }
561 while (bdr->retired_entries) {
562 bdr_entry_t *next = bdr->retired_entries->retire_next;
563 bdr_blob_free_cb(atomic_load_explicit(&bdr->retired_entries->blob, memory_order_relaxed));
564 cwist_free(bdr->retired_entries);
565 bdr->retired_entries = next;
566 }
567 cwist_free(bdr->buckets);
568#if !defined(__EMSCRIPTEN__) && defined(CWIST_DB)
569 if (bdr->disk_db) {
570 sqlite3_close(bdr->disk_db);
571 remove("cwist_bdr_fallback.db");
572 }
573#endif
574 pthread_mutex_unlock(&bdr->lock);
575 pthread_mutex_destroy(&bdr->lock);
576 cwist_free(bdr);
577}

595static const void *bdr_serve_hit(cwist_bdr_t *bdr, uint64_t req_h, bdr_entry_t *hint,
596 size_t *out_len, bdr_blob_t **out_pin) {
597 ttak_epoch_enter();
598 bdr_entry_t *entry = hint;
599 if (entry && (atomic_load_explicit(&entry->retired, memory_order_acquire) ||
600 entry->request_hash != req_h)) {
601 entry = NULL;
602 }
603 if (!entry) entry = bdr_find(bdr, req_h);
604 if (!entry) {
605 ttak_epoch_exit();
606 return NULL;
607 }
608 atomic_fetch_add_explicit(&entry->hits, 1, memory_order_relaxed);
609
610 /* Hit-time revalidation: a true report swaps in the callback's buffer
611 * with one pointer exchange; the predecessor retires across an epoch. */
612 cwist_bdr_revalidate_fn hook = entry->revalidate;
613 if (hook) {
614 void *fresh = NULL;
615 size_t fresh_len = 0;
616 void (*fresh_free)(void *) = NULL;
617 if (hook(entry->revalidate_arg, &fresh, &fresh_len, &fresh_free) && fresh &&
618 fresh_len > 0) {
619 bdr_blob_t *nb = bdr_blob_wrap(fresh, fresh_len, fresh_free);
620 if (nb) {
621 bdr_entry_publish(bdr, entry, nb);
622 } else if (fresh_free) {
623 fresh_free(fresh);
624 }
625 }
626 }
627
628 bdr_blob_t *blob = atomic_load_explicit(&entry->blob, memory_order_acquire);
629 if (!blob || !atomic_load_explicit(&entry->is_stable, memory_order_acquire)) {
630 ttak_epoch_exit();
631 return NULL;
632 }
633 if (out_len) *out_len = blob->len;
634 *out_pin = blob;
635 return blob->data;
636}

677const void *cwist_bdr_get_pinned_cursor(cwist_bdr_t *bdr, const char *method, const char *path,
678 size_t path_len, size_t *out_len, bdr_blob_t **out_pin,
679 cwist_bdr_cursor_t *cursor) {
680 if (!bdr || !method || !path || !out_pin || !cursor) return NULL;
681 if (strcmp(method, "GET") != 0) return NULL;
682 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) return NULL;
683
684 uint64_t req_h;
685 bdr_entry_t *hint = NULL;
686 if (cursor->entry && cursor->path_len == path_len && path_len < CWIST_BDR_CURSOR_PATH_MAX &&
687 cursor->req_hash != 0 && memcmp(cursor->path, path, path_len) == 0) {
688 hint = cursor->entry;
689 req_h = cursor->req_hash;
690 } else {
691 req_h = bdr_hash_n(method, path, path_len);
692 }
693
694 const void *data = bdr_serve_hit(bdr, req_h, hint, out_len, out_pin);
695 if (data && path_len < CWIST_BDR_CURSOR_PATH_MAX) {
696 bdr_entry_t *served = hint;
697 if (!served || atomic_load_explicit(&served->retired, memory_order_relaxed) ||
698 served->request_hash != req_h) {
699 /* The hint was stale (or absent): recover the served entry from
700 * the bucket chain so the cursor points at the live node. */
701 served = bdr_find(bdr, req_h);
702 }
703 if (served) {
704 cursor->entry = served;
705 cursor->req_hash = req_h;
706 cursor->path_len = path_len;
707 memcpy(cursor->path, path, path_len);
708 } else {
709 cursor->entry = NULL;
710 }
711 } else if (!data) {
712 cursor->entry = NULL;
713 }
714 return data;
715}

833 bdr_entry_t *entry = bdr_find_or_insert(bdr, req_h);
834 if (!entry) return;
835
836 if (atomic_load_explicit(&entry->is_stable, memory_order_acquire)) {
837 /* Content changed under a stable entry: demote to candidate. */
838 if (atomic_load_explicit(&entry->response_hash, memory_order_relaxed) != res_h) {
839 bdr_entry_publish(bdr, entry, NULL);
840 atomic_store_explicit(&entry->response_hash, res_h, memory_order_release);
841 atomic_store_explicit(&entry->is_stable, false, memory_order_release);
842 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
843 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL), memory_order_relaxed);
844 }
845 } else {
846 uint64_t expect = res_h;
847 if (atomic_compare_exchange_strong_explicit(&entry->response_hash, &expect, res_h,
848 memory_order_acq_rel, memory_order_relaxed)) {
849 /* Candidate reproduced the same bytes: stabilize. A concurrent
850 * stabilizer publishes an identical blob; the loser's duplicate
851 * comes back from the exchange and is retired unused. */
852 bdr_blob_t *blob = bdr_blob_learn(data, len);
853 if (blob) {
854 bdr_entry_publish(bdr, entry, blob);
855 atomic_store_explicit(&entry->is_stable, true, memory_order_release);
856 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
857 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL),
858 memory_order_relaxed);
859 }
860 } else {
861 /* Different bytes: the entry now tracks the new candidate hash. */
862 atomic_store_explicit(&entry->response_hash, res_h, memory_order_release);
863 }
864 }
865
866 bdr_janitor_tick(bdr);
867}
