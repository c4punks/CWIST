51static void bdr_blob_free_cb(void *ptr) {
52 bdr_blob_t *blob = (bdr_blob_t *)ptr;
53 if (!blob) return;
54 /* Read the fields before releasing: for inline blobs mem == blob, so the
55 * free below clobbers the very bytes the branch would inspect (glibc
56 * writes its tcache links into freed chunks). */
57 void *mem = blob->mem;
58 void (*free_fn)(void *) = blob->free_fn;
59 free_fn(mem);
60 if (mem != ptr) cwist_free(blob);
61}

117static void bdr_bytes_sub(cwist_bdr_t *bdr, size_t len) {
118 size_t cur = atomic_load_explicit(&bdr->current_bytes, memory_order_relaxed);

195static bool bdr_entry_should_decay(const cwist_bdr_t *bdr, const bdr_entry_t *entry, time_t now) {
196 if (!bdr || !entry) return false;
197 int64_t created = atomic_load_explicit(&entry->created_at, memory_order_relaxed);
198 if (bdr->max_entry_age_sec > 0 && created > 0) {
199 if (now - (time_t)created > bdr->max_entry_age_sec) return true;
200 }
201 if (atomic_load_explicit(&entry->is_stable, memory_order_relaxed) && bdr->revalidate_hits > 0 &&
202 atomic_load_explicit(&entry->hits, memory_order_relaxed) >= bdr->revalidate_hits) {
203 return true;
204 }
205 return false;
206}

218static bdr_entry_t *bdr_find(cwist_bdr_t *bdr, uint64_t req_h) {
219 size_t idx = req_h % bdr->bucket_count;
220 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
221 while (curr) {
222 if (curr->request_hash == req_h &&
223 !atomic_load_explicit(&curr->retired, memory_order_acquire)) {
224 return curr;
225 }
226 curr = atomic_load_explicit(&curr->next, memory_order_acquire);
227 }
228 return NULL;

243static bdr_entry_t *bdr_find_or_insert(cwist_bdr_t *bdr, uint64_t req_h) {
244 bdr_entry_t *found = bdr_find(bdr, req_h);
245 if (found) return found;
246
247 bdr_entry_t *entry = cwist_alloc(sizeof(bdr_entry_t));
248 if (!entry) return NULL;
249 entry->request_hash = req_h;
250 atomic_init(&entry->response_hash, 0);
251 atomic_init(&entry->is_stable, false);
252 atomic_init(&entry->retired, false);
253 atomic_init(&entry->blob, NULL);
254 atomic_init(&entry->hits, 0);
255 atomic_init(&entry->created_at, (int64_t)time(NULL));
256 entry->revalidate = NULL;
257 entry->revalidate_arg = NULL;
258 entry->retire_next = NULL;
259
260 size_t idx = req_h % bdr->bucket_count;
261 bdr_entry_t *head = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
262 do {
263 /* A concurrent insert of the same key wins; ours becomes garbage. */
264 bdr_entry_t *race = bdr_find(bdr, req_h);
265 if (race) {
266 cwist_free(entry);
267 return race;
268 }
269 atomic_store_explicit(&entry->next, head, memory_order_relaxed);
270 } while (!atomic_compare_exchange_weak_explicit(&bdr->buckets[idx], &head, entry,
271 memory_order_release, memory_order_acquire));
272 return entry;
273}

312static void bdr_sweep(cwist_bdr_t *bdr, size_t steps) {
313 if (!bdr || bdr->bucket_count == 0 || steps == 0) return;
314 time_t now = time(NULL);
315 for (size_t i = 0; i < steps; ++i) {
316 size_t idx = bdr->gc_cursor % bdr->bucket_count;
317 bdr->gc_cursor = (bdr->gc_cursor + 1) % bdr->bucket_count;
318 bdr_entry_t *prev = NULL;
319 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
320 while (curr) {
321 bdr_entry_t *next = atomic_load_explicit(&curr->next, memory_order_acquire);
322 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
323 bdr_entry_should_decay(bdr, curr, now)) {
324 bdr_retire_entry(bdr, idx, prev, curr);
325 } else {
326 prev = curr;
327 }
328 curr = next;
329 }
330 }

343static bool bdr_trim_oldest(cwist_bdr_t *bdr, time_t now) {
344 size_t victim_idx = SIZE_MAX;
345 bdr_entry_t *victim = NULL;
346 bdr_entry_t *victim_prev = NULL;
347 int64_t oldest = (int64_t)now;
348
349 for (size_t i = 0; i < bdr->bucket_count; ++i) {
350 bdr_entry_t *prev = NULL;
351 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[i], memory_order_acquire);
352 while (curr) {
353 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
354 atomic_load_explicit(&curr->blob, memory_order_relaxed)) {
355 int64_t created = atomic_load_explicit(&curr->created_at, memory_order_relaxed);
356 if (!victim || created < oldest) {
357 victim = curr;
358 victim_prev = prev;
359 victim_idx = i;
360 oldest = created;
361 }
362 }
363 prev = curr;
364 curr = atomic_load_explicit(&curr->next, memory_order_acquire);
365 }
366 }
367
368 if (!victim) return false;
369 bdr_retire_entry(bdr, victim_idx, victim_prev, victim);
370 return true;
371}

384static void bdr_check_ram(cwist_bdr_t *bdr) {
385#ifdef __EMSCRIPTEN__
386 /* No SQLite disk fallback in the WASM core; the in-RAM cache stands alone. */
387 (void)bdr;
388 return;
389#else
390 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_relaxed)) return;
391
392 if (cwist_is_ram_critical(CWIST_MIB(64))) {
393 printf("[BDR] Low RAM. Switching to Disk Cache.\n");
394 if (sqlite3_open("cwist_bdr_fallback.db", &bdr->disk_db) == SQLITE_OK) {
395 char *err = NULL;
396 sqlite3_exec(bdr->disk_db,

398 NULL, NULL, &err);
399 if (err) sqlite3_free(err);
400
401 sqlite3_exec(bdr->disk_db, "BEGIN TRANSACTION;", NULL, NULL, NULL);
402 for (size_t i = 0; i < bdr->bucket_count; i++) {
403 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[i], memory_order_acquire);
404 while (curr) {
405 bdr_entry_t *next = atomic_load_explicit(&curr->next, memory_order_acquire);
406 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
407 atomic_load_explicit(&curr->is_stable, memory_order_relaxed)) {
408 bdr_blob_t *blob = atomic_load_explicit(&curr->blob, memory_order_acquire);
409 if (blob) {
410 sqlite3_stmt *stmt;
411 sqlite3_prepare_v2(bdr->disk_db,
412 "INSERT INTO bdr (hash, blob) VALUES (?, ?);", -1,
413 &stmt, NULL);
414 sqlite3_bind_int64(stmt, 1, curr->request_hash);
415 sqlite3_bind_blob(stmt, 2, blob->data, (int)blob->len, SQLITE_STATIC);
416 sqlite3_step(stmt);
417 sqlite3_finalize(stmt);
418 }
419 }
420 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed)) {
421 atomic_store_explicit(&curr->retired, true, memory_order_release);
422 bdr_entry_publish(bdr, curr, NULL);
423 curr->retire_next = bdr->retired_entries;
424 bdr->retired_entries = curr;
425 }
426 curr = next;
427 }
428 atomic_store_explicit(&bdr->buckets[i], NULL, memory_order_release);
429 }
430 sqlite3_exec(bdr->disk_db, "COMMIT;", NULL, NULL, NULL);
431 atomic_store_explicit(&bdr->is_disk_mode, true, memory_order_release);
432 }
433 }
434#endif

447static void bdr_janitor_tick(cwist_bdr_t *bdr) {
448 uint64_t n = atomic_fetch_add_explicit(&bdr->put_count, 1, memory_order_relaxed) + 1;
449 if (n % BDR_JANITOR_PERIOD != 0) return;
450 if (pthread_mutex_trylock(&bdr->lock) != 0) return; /* another tick holds it */
451 bdr_check_ram(bdr);
452 bdr_sweep(bdr, BDR_GC_SWEEP);
453 if (bdr->max_bytes > 0) {
454 time_t now = time(NULL);
455 while (atomic_load_explicit(&bdr->current_bytes, memory_order_relaxed) > bdr->max_bytes) {
456 if (!bdr_trim_oldest(bdr, now)) break;
457 }
458 }
459 pthread_mutex_unlock(&bdr->lock);
460}

472cwist_bdr_t *cwist_bdr_create(void) {
473 cwist_bdr_t *bdr = cwist_alloc(sizeof(cwist_bdr_t));
474 if (!bdr) return NULL;
475 if (pthread_mutex_init(&bdr->lock, NULL) != 0) {
476 cwist_free(bdr);
477 return NULL;
478 }
479 bdr->bucket_count = BDR_BUCKETS;
480 bdr->buckets = cwist_alloc_array(BDR_BUCKETS, sizeof(bdr_entry_t *));
481 if (!bdr->buckets) {
482 pthread_mutex_destroy(&bdr->lock);
483 cwist_free(bdr);
484 return NULL;
485 }
486 bdr->latency_threshold_ms = 10;
487 atomic_init(&bdr->current_bytes, 0);
488 bdr->max_bytes = BDR_DEFAULT_MAX_BYTES;
489 bdr->max_entry_age_sec = BDR_DEFAULT_ENTRY_TTL;
490 bdr->revalidate_hits = BDR_DEFAULT_REVALIDATE_HITS;
491 bdr->gc_cursor = 0;
492 atomic_init(&bdr->put_count, 0);
493 bdr->retired_entries = NULL;
494 bdr->disk_db = NULL;
495 atomic_init(&bdr->is_disk_mode, false);
496 return bdr;

508void cwist_bdr_destroy(cwist_bdr_t *bdr) {
509 if (!bdr) return;
510 pthread_mutex_lock(&bdr->lock);
511 for (size_t i = 0; i < bdr->bucket_count; i++) {
512 bdr_entry_t *curr = bdr->buckets[i];
513 while (curr) {
514 bdr_entry_t *next = curr->next;
515 if (curr == NULL) break;
516 if (curr->blob != NULL)
517 bdr_blob_free_cb(atomic_load_explicit(&curr->blob, memory_order_relaxed));
518 cwist_free(curr);
519 curr = next;
520 }
521 }
522 while (bdr->retired_entries) {
523 bdr_entry_t *next = bdr->retired_entries->retire_next;
524 bdr_blob_free_cb(atomic_load_explicit(&bdr->retired_entries->blob, memory_order_relaxed));
525 cwist_free(bdr->retired_entries);
526 bdr->retired_entries = next;
527 }
528 cwist_free(bdr->buckets);
529#ifndef __EMSCRIPTEN__
530 if (bdr->disk_db) {
531 sqlite3_close(bdr->disk_db);
532 remove("cwist_bdr_fallback.db");
533 }
534#endif
535 pthread_mutex_unlock(&bdr->lock);
536 pthread_mutex_destroy(&bdr->lock);
537 cwist_free(bdr);
538}

556static const void *bdr_serve_hit(cwist_bdr_t *bdr, uint64_t req_h, bdr_entry_t *hint,
557 size_t *out_len, bdr_blob_t **out_pin) {
558 ttak_epoch_enter();
559 bdr_entry_t *entry = hint;
560 if (entry &&
561 (atomic_load_explicit(&entry->retired, memory_order_acquire) ||
562 entry->request_hash != req_h)) {
563 entry = NULL;
564 }
565 if (!entry) entry = bdr_find(bdr, req_h);
566 if (!entry) {
567 ttak_epoch_exit();
568 return NULL;
569 }
570 atomic_fetch_add_explicit(&entry->hits, 1, memory_order_relaxed);
571
572 /* Hit-time revalidation: a true report swaps in the callback's buffer
573 * with one pointer exchange; the predecessor retires across an epoch. */
574 cwist_bdr_revalidate_fn hook = entry->revalidate;
575 if (hook) {
576 void *fresh = NULL;
577 size_t fresh_len = 0;
578 void (*fresh_free)(void *) = NULL;
579 if (hook(entry->revalidate_arg, &fresh, &fresh_len, &fresh_free) && fresh &&
580 fresh_len > 0) {
581 bdr_blob_t *nb = bdr_blob_wrap(fresh, fresh_len, fresh_free);
582 if (nb) {
583 bdr_entry_publish(bdr, entry, nb);
584 } else if (fresh_free) {
585 fresh_free(fresh);
586 }
587 }
588 }
589
590 bdr_blob_t *blob = atomic_load_explicit(&entry->blob, memory_order_acquire);
591 if (!blob || !atomic_load_explicit(&entry->is_stable, memory_order_acquire)) {
592 ttak_epoch_exit();
593 return NULL;
594 }
595 if (out_len) *out_len = blob->len;
596 *out_pin = blob;
597 return blob->data;
598}

639const void *cwist_bdr_get_pinned_cursor(cwist_bdr_t *bdr, const char *method, const char *path,
640 size_t path_len, size_t *out_len, bdr_blob_t **out_pin,
641 cwist_bdr_cursor_t *cursor) {
642 if (!bdr || !method || !path || !out_pin || !cursor) return NULL;
643 if (strcmp(method, "GET") != 0) return NULL;
644 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) return NULL;
645
646 uint64_t req_h;
647 bdr_entry_t *hint = NULL;
648 if (cursor->entry && cursor->path_len == path_len &&
649 path_len < CWIST_BDR_CURSOR_PATH_MAX && cursor->req_hash != 0 &&
650 memcmp(cursor->path, path, path_len) == 0) {
651 hint = cursor->entry;

658 if (data && path_len < CWIST_BDR_CURSOR_PATH_MAX) {
659 bdr_entry_t *served = hint;

661 served->request_hash != req_h) {
662 /* The hint was stale (or absent): recover the served entry from
663 * the bucket chain so the cursor points at the live node. */
664 served = bdr_find(bdr, req_h);
665 }
666 if (served) {
667 cursor->entry = served;
668 cursor->req_hash = req_h;
669 cursor->path_len = path_len;
670 memcpy(cursor->path, path, path_len);
671 } else {
672 cursor->entry = NULL;
673 }
674 } else if (!data) {
675 cursor->entry = NULL;
676 }
677 return data;

779void cwist_bdr_put(cwist_bdr_t *bdr, const char *method, const char *path, const void *data,
780 size_t len) {
781 if (!bdr || !method || !path || !data || len == 0) return;
782 if (strcmp(method, "GET") != 0) return;
783
784 uint64_t req_h = bdr_hash(method, path);
785 uint64_t res_h = bdr_hash_data(data, len);
786
787 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) {
788 cwist_bdr_put_disk(bdr, req_h, data, len);
789 return;
790 }
791
792 bdr_entry_t *entry = bdr_find_or_insert(bdr, req_h);
793 if (!entry) return;
794
795 if (atomic_load_explicit(&entry->is_stable, memory_order_acquire)) {
796 /* Content changed under a stable entry: demote to candidate. */
797 if (atomic_load_explicit(&entry->response_hash, memory_order_relaxed) != res_h) {
798 bdr_entry_publish(bdr, entry, NULL);
799 atomic_store_explicit(&entry->response_hash, res_h, memory_order_release);
800 atomic_store_explicit(&entry->is_stable, false, memory_order_release);
801 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
802 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL), memory_order_relaxed);
803 }
804 } else {
805 uint64_t expect = res_h;
806 if (atomic_compare_exchange_strong_explicit(&entry->response_hash, &expect, res_h,
807 memory_order_acq_rel, memory_order_relaxed)) {
808 /* Candidate reproduced the same bytes: stabilize. A concurrent
809 * stabilizer publishes an identical blob; the loser's duplicate
810 * comes back from the exchange and is retired unused. */
811 bdr_blob_t *blob = bdr_blob_learn(data, len);
812 if (blob) {
813 bdr_entry_publish(bdr, entry, blob);
814 atomic_store_explicit(&entry->is_stable, true, memory_order_release);
815 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
816 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL),
817 memory_order_relaxed);
818 }
819 } else {
820 /* Different bytes: the entry now tracks the new candidate hash. */
821 atomic_store_explicit(&entry->response_hash, res_h, memory_order_release);
822 }

841void cwist_bdr_put_fixed(cwist_bdr_t *bdr, const char *method, const char *path, const void *data,
842 size_t len) {
843 if (!bdr || !method || !path || !data || len == 0) return;
844 if (strcmp(method, "GET") != 0) return;
845
846 uint64_t req_h = bdr_hash(method, path);
847
848 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) {
849 cwist_bdr_put_disk(bdr, req_h, data, len);
850 return;
851 }
852

854 if (!entry) return;

857 if (!blob) return;
858 atomic_store_explicit(&entry->response_hash, bdr_hash_data(data, len), memory_order_release);
859 bdr_entry_publish(bdr, entry, blob);
860 atomic_store_explicit(&entry->is_stable, true, memory_order_release);
861 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
862 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL), memory_order_relaxed);
863
864 bdr_janitor_tick(bdr);
865}
