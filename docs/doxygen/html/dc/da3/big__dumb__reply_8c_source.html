53static void bdr_blob_free_cb(void *ptr) {
54 bdr_blob_t *blob = (bdr_blob_t *)ptr;
55 if (!blob) return;
56 /* Read the fields before releasing: for inline blobs mem == blob, so the
57 * free below clobbers the very bytes the branch would inspect (glibc
58 * writes its tcache links into freed chunks). */
59 void *mem = blob->mem;
60 void (*free_fn)(void *) = blob->free_fn;
61 free_fn(mem);
62 if (mem != ptr) cwist_free(blob);
63}

70 if (blob) ttak_epoch_retire(blob, bdr_blob_free_cb);

119static void bdr_bytes_sub(cwist_bdr_t *bdr, size_t len) {
120 size_t cur = atomic_load_explicit(&bdr->current_bytes, memory_order_relaxed);
121 while (cur > 0) {
122 size_t next = cur >= len ? cur - len : 0;
123 if (atomic_compare_exchange_weak_explicit(&bdr->current_bytes, &cur, next,
124 memory_order_relaxed, memory_order_relaxed))
125 return;
126 }

197static bool bdr_entry_should_decay(const cwist_bdr_t *bdr, const bdr_entry_t *entry, time_t now) {
198 if (!bdr || !entry) return false;
199 int64_t created = atomic_load_explicit(&entry->created_at, memory_order_relaxed);
200 if (bdr->max_entry_age_sec > 0 && created > 0) {
201 if (now - (time_t)created > bdr->max_entry_age_sec) return true;
202 }
203 if (atomic_load_explicit(&entry->is_stable, memory_order_relaxed) && bdr->revalidate_hits > 0 &&
204 atomic_load_explicit(&entry->hits, memory_order_relaxed) >= bdr->revalidate_hits) {
205 return true;
206 }
207 return false;
208}

220static bdr_entry_t *bdr_find(cwist_bdr_t *bdr, uint64_t req_h) {
221 size_t idx = req_h % bdr->bucket_count;
222 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
223 while (curr) {
224 if (curr->request_hash == req_h &&
225 !atomic_load_explicit(&curr->retired, memory_order_acquire)) {
226 return curr;
227 }
228 curr = atomic_load_explicit(&curr->next, memory_order_acquire);
229 }
230 return NULL;
231}

245static bdr_entry_t *bdr_find_or_insert(cwist_bdr_t *bdr, uint64_t req_h) {
246 bdr_entry_t *found = bdr_find(bdr, req_h);
247 if (found) return found;
248
249 bdr_entry_t *entry = cwist_alloc(sizeof(bdr_entry_t));
250 if (!entry) return NULL;
251 entry->request_hash = req_h;
252 atomic_init(&entry->response_hash, 0);
253 atomic_init(&entry->is_stable, false);
254 atomic_init(&entry->retired, false);
255 atomic_init(&entry->blob, NULL);
256 atomic_init(&entry->hits, 0);
257 atomic_init(&entry->created_at, (int64_t)time(NULL));
258 entry->revalidate = NULL;
259 entry->revalidate_arg = NULL;
260 entry->retire_next = NULL;
261
262 size_t idx = req_h % bdr->bucket_count;
263 bdr_entry_t *head = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
264 do {
265 /* A concurrent insert of the same key wins; ours becomes garbage. */
266 bdr_entry_t *race = bdr_find(bdr, req_h);
267 if (race) {
268 cwist_free(entry);
269 return race;
270 }
271 atomic_store_explicit(&entry->next, head, memory_order_relaxed);
272 } while (!atomic_compare_exchange_weak_explicit(&bdr->buckets[idx], &head, entry,
273 memory_order_release, memory_order_acquire));
274 return entry;
275}

291static void bdr_retire_entry(cwist_bdr_t *bdr, size_t idx, bdr_entry_t *prev, bdr_entry_t *entry) {
292 atomic_store_explicit(&entry->retired, true, memory_order_release);
293 if (prev) {
294 atomic_store_explicit(&prev->next, atomic_load_explicit(&entry->next, memory_order_acquire),
295 memory_order_release);
296 } else {
297 atomic_store_explicit(&bdr->buckets[idx],
298 atomic_load_explicit(&entry->next, memory_order_acquire),
299 memory_order_release);
300 }
301 bdr_entry_publish(bdr, entry, NULL);
302 entry->retire_next = bdr->retired_entries;
303 bdr->retired_entries = entry;
304}

314static void bdr_sweep(cwist_bdr_t *bdr, size_t steps) {
315 if (!bdr || bdr->bucket_count == 0 || steps == 0) return;
316 time_t now = time(NULL);
317 for (size_t i = 0; i < steps; ++i) {
318 size_t idx = bdr->gc_cursor % bdr->bucket_count;
319 bdr->gc_cursor = (bdr->gc_cursor + 1) % bdr->bucket_count;
320 bdr_entry_t *prev = NULL;
321 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[idx], memory_order_acquire);
322 while (curr) {
323 bdr_entry_t *next = atomic_load_explicit(&curr->next, memory_order_acquire);
324 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
325 bdr_entry_should_decay(bdr, curr, now)) {
326 bdr_retire_entry(bdr, idx, prev, curr);
327 } else {
328 prev = curr;
329 }
330 curr = next;
331 }
332 }
333}

345static bool bdr_trim_oldest(cwist_bdr_t *bdr, time_t now) {
346 size_t victim_idx = SIZE_MAX;
347 bdr_entry_t *victim = NULL;
348 bdr_entry_t *victim_prev = NULL;
349 int64_t oldest = (int64_t)now;
350
351 for (size_t i = 0; i < bdr->bucket_count; ++i) {
352 bdr_entry_t *prev = NULL;
353 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[i], memory_order_acquire);
354 while (curr) {
355 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
356 atomic_load_explicit(&curr->blob, memory_order_relaxed)) {
357 int64_t created = atomic_load_explicit(&curr->created_at, memory_order_relaxed);
358 if (!victim || created < oldest) {
359 victim = curr;
360 victim_prev = prev;
361 victim_idx = i;
362 oldest = created;
363 }
364 }
365 prev = curr;
366 curr = atomic_load_explicit(&curr->next, memory_order_acquire);
367 }
368 }
369
370 if (!victim) return false;
371 bdr_retire_entry(bdr, victim_idx, victim_prev, victim);
372 return true;
373}

386static void bdr_check_ram(cwist_bdr_t *bdr) {
387#if defined(__EMSCRIPTEN__) || !defined(CWIST_DB)
388 /* No SQLite disk fallback in the WASM core or when the DB tier is
389 * compiled out (CWIST_DB=0); the in-RAM cache stands alone. */
390 (void)bdr;
391 return;
392#else
393 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_relaxed)) return;
394
395 if (cwist_is_ram_critical(CWIST_MIB(64))) {
396 printf("[BDR] Low RAM. Switching to Disk Cache.\n");
397 if (sqlite3_open("cwist_bdr_fallback.db", &bdr->disk_db) == SQLITE_OK) {
398 char *err = NULL;
399 sqlite3_exec(bdr->disk_db,
400 "CREATE TABLE IF NOT EXISTS bdr (hash INTEGER PRIMARY KEY, blob BLOB);",
401 NULL, NULL, &err);
402 if (err) sqlite3_free(err);
403
404 sqlite3_exec(bdr->disk_db, "BEGIN TRANSACTION;", NULL, NULL, NULL);
405 for (size_t i = 0; i < bdr->bucket_count; i++) {
406 bdr_entry_t *curr = atomic_load_explicit(&bdr->buckets[i], memory_order_acquire);
407 while (curr) {
408 bdr_entry_t *next = atomic_load_explicit(&curr->next, memory_order_acquire);
409 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed) &&
410 atomic_load_explicit(&curr->is_stable, memory_order_relaxed)) {
411 bdr_blob_t *blob = atomic_load_explicit(&curr->blob, memory_order_acquire);
412 if (blob) {
413 sqlite3_stmt *stmt;
414 sqlite3_prepare_v2(bdr->disk_db,
415 "INSERT INTO bdr (hash, blob) VALUES (?, ?);", -1,
416 &stmt, NULL);
417 sqlite3_bind_int64(stmt, 1, curr->request_hash);
418 sqlite3_bind_blob(stmt, 2, blob->data, (int)blob->len, SQLITE_STATIC);
419 sqlite3_step(stmt);
420 sqlite3_finalize(stmt);
421 }
422 }
423 if (!atomic_load_explicit(&curr->retired, memory_order_relaxed)) {
424 atomic_store_explicit(&curr->retired, true, memory_order_release);
425 bdr_entry_publish(bdr, curr, NULL);
426 curr->retire_next = bdr->retired_entries;
427 bdr->retired_entries = curr;
428 }
429 curr = next;
430 }
431 atomic_store_explicit(&bdr->buckets[i], NULL, memory_order_release);
432 }
433 sqlite3_exec(bdr->disk_db, "COMMIT;", NULL, NULL, NULL);
434 atomic_store_explicit(&bdr->is_disk_mode, true, memory_order_release);
435 }
436 }
437#endif
438}

450static void bdr_janitor_tick(cwist_bdr_t *bdr) {
451 uint64_t n = atomic_fetch_add_explicit(&bdr->put_count, 1, memory_order_relaxed) + 1;
452 if (n % BDR_JANITOR_PERIOD != 0) return;
453 if (pthread_mutex_trylock(&bdr->lock) != 0) return; /* another tick holds it */
454 bdr_check_ram(bdr);
455 bdr_sweep(bdr, BDR_GC_SWEEP);
456 if (bdr->max_bytes > 0) {
457 time_t now = time(NULL);
458 while (atomic_load_explicit(&bdr->current_bytes, memory_order_relaxed) > bdr->max_bytes) {
459 if (!bdr_trim_oldest(bdr, now)) break;
460 }
461 }
462 pthread_mutex_unlock(&bdr->lock);
463}

475cwist_bdr_t *cwist_bdr_create(void) {
476 cwist_bdr_t *bdr = cwist_alloc(sizeof(cwist_bdr_t));
477 if (!bdr) return NULL;
478 if (pthread_mutex_init(&bdr->lock, NULL) != 0) {
479 cwist_free(bdr);
480 return NULL;
481 }
482 bdr->bucket_count = BDR_BUCKETS;
483 bdr->buckets = cwist_alloc_array(BDR_BUCKETS, sizeof(bdr_entry_t *));
484 if (!bdr->buckets) {
485 pthread_mutex_destroy(&bdr->lock);
486 cwist_free(bdr);
487 return NULL;
488 }
489 bdr->latency_threshold_ms = 10;
490 atomic_init(&bdr->current_bytes, 0);
491 bdr->max_bytes = BDR_DEFAULT_MAX_BYTES;
492 bdr->max_entry_age_sec = BDR_DEFAULT_ENTRY_TTL;
493 bdr->revalidate_hits = BDR_DEFAULT_REVALIDATE_HITS;
494 bdr->gc_cursor = 0;
495 atomic_init(&bdr->put_count, 0);
496 bdr->retired_entries = NULL;
497 bdr->disk_db = NULL;
498 atomic_init(&bdr->is_disk_mode, false);
499 return bdr;
500}

511void cwist_bdr_destroy(cwist_bdr_t *bdr) {
512 if (!bdr) return;
513 pthread_mutex_lock(&bdr->lock);
514 for (size_t i = 0; i < bdr->bucket_count; i++) {
515 bdr_entry_t *curr = bdr->buckets[i];
516 while (curr) {
517 bdr_entry_t *next = curr->next;
518 if (curr == NULL) break;
519 if (curr->blob != NULL)
520 bdr_blob_free_cb(atomic_load_explicit(&curr->blob, memory_order_relaxed));
521 cwist_free(curr);
522 curr = next;
523 }
524 }
525 while (bdr->retired_entries) {
526 bdr_entry_t *next = bdr->retired_entries->retire_next;
527 bdr_blob_free_cb(atomic_load_explicit(&bdr->retired_entries->blob, memory_order_relaxed));
528 cwist_free(bdr->retired_entries);
529 bdr->retired_entries = next;
530 }
531 cwist_free(bdr->buckets);
532#if !defined(__EMSCRIPTEN__) && defined(CWIST_DB)
533 if (bdr->disk_db) {
534 sqlite3_close(bdr->disk_db);
535 remove("cwist_bdr_fallback.db");
536 }
537#endif
538 pthread_mutex_unlock(&bdr->lock);
539 pthread_mutex_destroy(&bdr->lock);
540 cwist_free(bdr);
541}

559static const void *bdr_serve_hit(cwist_bdr_t *bdr, uint64_t req_h, bdr_entry_t *hint,
560 size_t *out_len, bdr_blob_t **out_pin) {
561 ttak_epoch_enter();
562 bdr_entry_t *entry = hint;
563 if (entry &&
564 (atomic_load_explicit(&entry->retired, memory_order_acquire) ||
565 entry->request_hash != req_h)) {
566 entry = NULL;
567 }
568 if (!entry) entry = bdr_find(bdr, req_h);
569 if (!entry) {
570 ttak_epoch_exit();
571 return NULL;
572 }
573 atomic_fetch_add_explicit(&entry->hits, 1, memory_order_relaxed);
574
575 /* Hit-time revalidation: a true report swaps in the callback's buffer
576 * with one pointer exchange; the predecessor retires across an epoch. */
577 cwist_bdr_revalidate_fn hook = entry->revalidate;
578 if (hook) {
579 void *fresh = NULL;
580 size_t fresh_len = 0;
581 void (*fresh_free)(void *) = NULL;
582 if (hook(entry->revalidate_arg, &fresh, &fresh_len, &fresh_free) && fresh &&
583 fresh_len > 0) {
584 bdr_blob_t *nb = bdr_blob_wrap(fresh, fresh_len, fresh_free);
585 if (nb) {
586 bdr_entry_publish(bdr, entry, nb);
587 } else if (fresh_free) {
588 fresh_free(fresh);
589 }
590 }
591 }
592
593 bdr_blob_t *blob = atomic_load_explicit(&entry->blob, memory_order_acquire);
594 if (!blob || !atomic_load_explicit(&entry->is_stable, memory_order_acquire)) {
595 ttak_epoch_exit();
596 return NULL;
597 }
598 if (out_len) *out_len = blob->len;
599 *out_pin = blob;
600 return blob->data;
601}

642const void *cwist_bdr_get_pinned_cursor(cwist_bdr_t *bdr, const char *method, const char *path,
643 size_t path_len, size_t *out_len, bdr_blob_t **out_pin,
644 cwist_bdr_cursor_t *cursor) {
645 if (!bdr || !method || !path || !out_pin || !cursor) return NULL;
646 if (strcmp(method, "GET") != 0) return NULL;
647 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) return NULL;
648
649 uint64_t req_h;
650 bdr_entry_t *hint = NULL;
651 if (cursor->entry && cursor->path_len == path_len &&
652 path_len < CWIST_BDR_CURSOR_PATH_MAX && cursor->req_hash != 0 &&
653 memcmp(cursor->path, path, path_len) == 0) {
654 hint = cursor->entry;
655 req_h = cursor->req_hash;
656 } else {
657 req_h = bdr_hash_n(method, path, path_len);
658 }
659
660 const void *data = bdr_serve_hit(bdr, req_h, hint, out_len, out_pin);
661 if (data && path_len < CWIST_BDR_CURSOR_PATH_MAX) {
662 bdr_entry_t *served = hint;
663 if (!served || atomic_load_explicit(&served->retired, memory_order_relaxed) ||
664 served->request_hash != req_h) {
665 /* The hint was stale (or absent): recover the served entry from
666 * the bucket chain so the cursor points at the live node. */
667 served = bdr_find(bdr, req_h);
668 }
669 if (served) {
670 cursor->entry = served;
671 cursor->req_hash = req_h;
672 cursor->path_len = path_len;
673 memcpy(cursor->path, path, path_len);
674 } else {
675 cursor->entry = NULL;
676 }
677 } else if (!data) {
678 cursor->entry = NULL;
679 }
680 return data;
681}

784void cwist_bdr_put(cwist_bdr_t *bdr, const char *method, const char *path, const void *data,
785 size_t len) {
786 if (!bdr || !method || !path || !data || len == 0) return;
787 if (strcmp(method, "GET") != 0) return;
788
789 uint64_t req_h = bdr_hash(method, path);
790 uint64_t res_h = bdr_hash_data(data, len);
791
792#ifdef CWIST_DB
793 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) {
794 cwist_bdr_put_disk(bdr, req_h, data, len);
795 return;
796 }
797#endif
798
799 bdr_entry_t *entry = bdr_find_or_insert(bdr, req_h);
800 if (!entry) return;
801
802 if (atomic_load_explicit(&entry->is_stable, memory_order_acquire)) {
803 /* Content changed under a stable entry: demote to candidate. */
804 if (atomic_load_explicit(&entry->response_hash, memory_order_relaxed) != res_h) {
805 bdr_entry_publish(bdr, entry, NULL);
806 atomic_store_explicit(&entry->response_hash, res_h, memory_order_release);
807 atomic_store_explicit(&entry->is_stable, false, memory_order_release);
808 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
809 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL), memory_order_relaxed);
810 }
811 } else {
812 uint64_t expect = res_h;
813 if (atomic_compare_exchange_strong_explicit(&entry->response_hash, &expect, res_h,
814 memory_order_acq_rel, memory_order_relaxed)) {
815 /* Candidate reproduced the same bytes: stabilize. A concurrent
816 * stabilizer publishes an identical blob; the loser's duplicate
817 * comes back from the exchange and is retired unused. */
818 bdr_blob_t *blob = bdr_blob_learn(data, len);
819 if (blob) {
820 bdr_entry_publish(bdr, entry, blob);
821 atomic_store_explicit(&entry->is_stable, true, memory_order_release);
822 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);

848void cwist_bdr_put_fixed(cwist_bdr_t *bdr, const char *method, const char *path, const void *data,
849 size_t len) {
850 if (!bdr || !method || !path || !data || len == 0) return;
851 if (strcmp(method, "GET") != 0) return;
852
853 uint64_t req_h = bdr_hash(method, path);
854
855#ifdef CWIST_DB
856 if (atomic_load_explicit(&bdr->is_disk_mode, memory_order_acquire)) {
857 cwist_bdr_put_disk(bdr, req_h, data, len);
858 return;
859 }
860#endif
861
862 bdr_entry_t *entry = bdr_find_or_insert(bdr, req_h);
863 if (!entry) return;
864
865 bdr_blob_t *blob = bdr_blob_learn(data, len);
866 if (!blob) return;
867 atomic_store_explicit(&entry->response_hash, bdr_hash_data(data, len), memory_order_release);
868 bdr_entry_publish(bdr, entry, blob);
869 atomic_store_explicit(&entry->is_stable, true, memory_order_release);
870 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
871 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL), memory_order_relaxed);
872
873 bdr_janitor_tick(bdr);
874}

894void cwist_bdr_put_revalidatable(cwist_bdr_t *bdr, const char *method, const char *path,
895 const void *data, size_t len, cwist_bdr_revalidate_fn fn,
896 void *arg) {
897 if (!bdr || !method || !path || !data || len == 0 || !fn) return;
898 if (strcmp(method, "GET") != 0) return;
899
900 uint64_t req_h = bdr_hash(method, path);
901
902#ifdef CWIST_DB

904 cwist_bdr_put_disk(bdr, req_h, data, len);
905 return;
906 }
907#endif
908
909 bdr_entry_t *entry = bdr_find_or_insert(bdr, req_h);
910 if (!entry) return;
911
912 /* Publish the hook before the blob so a reader never observes a
913 * revalidatable blob without its hook. */
914 entry->revalidate_arg = arg;
915 atomic_store_explicit((_Atomic(cwist_bdr_revalidate_fn) *)&entry->revalidate, fn,
916 memory_order_release);
917

919 if (!blob) return;
920 atomic_store_explicit(&entry->response_hash, bdr_hash_data(data, len), memory_order_release);
921 bdr_entry_publish(bdr, entry, blob);
922 atomic_store_explicit(&entry->is_stable, true, memory_order_release);
923 atomic_store_explicit(&entry->hits, 0, memory_order_relaxed);
924 atomic_store_explicit(&entry->created_at, (int64_t)time(NULL), memory_order_relaxed);
925
926 bdr_janitor_tick(bdr);
