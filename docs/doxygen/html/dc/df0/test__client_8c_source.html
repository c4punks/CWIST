65static bool cookie_path_matches(const char *cookie_path, const char *req_path) {
66 if (!cookie_path) return true;

74static char *cookie_default_path(const char *req_path) {

94 const cwist_test_client_kv *adhoc, size_t adhoc_count) {
95 if (!client || !req) return;

97 const char *req_path = req->path && req->path->data ? req->path->data : "/";

99 size_t jar_count = 0;
100 for (cwist_test_client_cookie *c = client->cookies; c; c = c->next) {
101 if (JAR_SENDS(c)) jar_count++;
102 }
103 size_t valid_adhoc = 0;
104 for (size_t i = 0; i < adhoc_count; i++) {
105 if (adhoc && adhoc[i].key && adhoc[i].value) valid_adhoc++;
106 }
107 if (jar_count == 0 && valid_adhoc == 0) return;
108
109 size_t buf_len = 0;
110 for (cwist_test_client_cookie *c = client->cookies; c; c = c->next) {
111 if (JAR_SENDS(c)) buf_len += strlen(c->name) + 1 + strlen(c->value) + 2;
112 }
113 for (size_t i = 0; i < adhoc_count; i++) {
114 if (adhoc && adhoc[i].key && adhoc[i].value)
115 buf_len += strlen(adhoc[i].key) + 1 + strlen(adhoc[i].value) + 2;
116 }
117
118 char *cookie_header CWIST_DEFER_FREE = (char *)cwist_alloc(buf_len + 1);
119 if (!cookie_header) return;
120 cookie_header[0] = '\0';
121
122 size_t pos = 0;
123 for (cwist_test_client_cookie *c = client->cookies; c; c = c->next) {
124 if (!JAR_SENDS(c)) continue;
125 if (pos > 0) cookie_header[pos++] = ';';
126 if (pos > 0) cookie_header[pos++] = ' ';
127 pos += (size_t)snprintf(cookie_header + pos, buf_len + 1 - pos, "%s=%s", c->name, c->value);
128 }
129 for (size_t i = 0; i < adhoc_count; i++) {
130 if (!adhoc || !adhoc[i].key || !adhoc[i].value) continue;
131 if (pos > 0) cookie_header[pos++] = ';';
132 if (pos > 0) cookie_header[pos++] = ' ';
133 pos += (size_t)snprintf(cookie_header + pos, buf_len + 1 - pos, "%s=%s", adhoc[i].key,

177static void cookie_jar_update(cwist_test_client *client, cwist_http_response *res,
178 const char *req_path) {
179 if (!client || !res) return;
180 for (cwist_http_header_node *h = res->headers; h; h = h->next) {
181 if (!h->key || !h->value || strcasecmp(h->key->data, "Set-Cookie") != 0) continue;
182 char *buf = clone_str(h->value->data);
183 if (!buf) continue;
184 char *saveptr = NULL;
185 char *name_val = strtok_r(buf, ";", &saveptr);
186 if (!name_val) {
187 cwist_free(buf);
188 continue;
189 }
190 name_val = trim(name_val);
191 char *eq = strchr(name_val, '=');
192 if (!eq) {
193 cwist_free(buf);
194 continue;
195 }
196 *eq = '\0';
197 char *name = clone_str(trim(name_val));
198 char *value = clone_str(trim(eq + 1));
199 if (!name || !value) {
200 cwist_free(name);
201 cwist_free(value);
202 cwist_free(buf);
203 continue;
204 }
205 char *path = NULL;
206 bool has_max_age = false, expired = false, expires_past = false;
207 char *rest = strtok_r(NULL, ";", &saveptr);
208 while (rest) {
209 rest = trim(rest);
210 if (strncasecmp(rest, "Path=", 5) == 0 && !path) {
211 path = clone_str(trim(rest + 5));
212 } else if (strncasecmp(rest, "Max-Age=", 8) == 0) {
213 /* RFC 6265 section 5.2.2: digits with an optional leading
214 * '-'; anything else is ignored. */
215 const char *v = trim(rest + 8);
216 const char *d = *v == '-' ? v + 1 : v;
217 if (*d && strspn(d, "0123456789") == strlen(d)) {
218 has_max_age = true;
219 expired = *v == '-' || strspn(d, "0") == strlen(d);
220 }
221 } else if (strncasecmp(rest, "Expires=", 8) == 0) {
222 time_t when = cwist_http_parse_date(trim(rest + 8));
223 expires_past = when != (time_t)-1 && when <= time(NULL);
224 }
225 rest = strtok_r(NULL, ";", &saveptr);
226 }
227 /* No (or a relative) Path: the request's default-path (RFC 6265
228 * section 5.2.4). */
229 if (!path || path[0] != '/') {
230 cwist_free(path);
231 path = cookie_default_path(req_path);
232 if (!path) {
233 cwist_free(name);
234 cwist_free(value);
235 cwist_free(buf);
236 continue;
237 }
238 }
239 /* Max-Age wins over Expires (RFC 6265 section 5.3, step 3). */
240 if (has_max_age ? expired : expires_past) {
241 cookie_jar_remove(client, name, path);
242 cwist_free(name);
243 cwist_free(value);
244 cwist_free(path);
245 cwist_free(buf);
246 continue;
247 }
248 cwist_test_client_cookie *c = find_cookie_at(client->cookies, name, path);
249 if (c) {
250 cwist_free(c->value);
251 c->value = value;
252 cwist_free(path);
253 cwist_free(name);
254 } else {
255 c = (cwist_test_client_cookie *)cwist_alloc(sizeof(*c));
256 if (c) {
257 c->name = name;
258 c->value = value;
259 c->path = path;
260 c->next = client->cookies;
261 client->cookies = c;
262 } else {
263 cwist_free(name);
264 cwist_free(value);
265 cwist_free(path);
266 }
267 }
268 cwist_free(buf);
269 }
270}
