189static void cookie_jar_update(cwist_test_client *client, cwist_http_response *res,
190 const char *req_path) {
191 if (!client || !res) return;
192 for (cwist_http_header_node *h = res->headers; h; h = h->next) {
193 if (!h->key || !h->value || strcasecmp(h->key->data, "Set-Cookie") != 0) continue;
194 char *buf = clone_str(h->value->data);
195 if (!buf) continue;
196 char *saveptr = NULL;
197 char *name_val = strtok_r(buf, ";", &saveptr);
198 if (!name_val) {
199 cwist_free(buf);
200 continue;
201 }
202 name_val = trim(name_val);
203 char *eq = strchr(name_val, '=');
204 if (!eq) {
205 cwist_free(buf);
206 continue;
207 }
208 *eq = '\0';
209 char *name = clone_str(trim(name_val));
210 char *value = clone_str(trim(eq + 1));
211 if (!name || !value) {
212 cwist_free(name);
213 cwist_free(value);
214 cwist_free(buf);
215 continue;
216 }
217 char *path = NULL;
218 bool has_max_age = false, expired = false, expires_past = false;
219 char *rest = strtok_r(NULL, ";", &saveptr);
220 while (rest) {
221 rest = trim(rest);
222 if (strncasecmp(rest, "Path=", 5) == 0 && !path) {
223 path = clone_str(trim(rest + 5));
224 } else if (strncasecmp(rest, "Max-Age=", 8) == 0) {
225 /* RFC 6265 section 5.2.2: digits with an optional leading
226 * '-'; anything else is ignored. */
227 const char *v = trim(rest + 8);
228 const char *d = *v == '-' ? v + 1 : v;
229 if (*d && strspn(d, "0123456789") == strlen(d)) {
230 has_max_age = true;
231 expired = *v == '-' || strspn(d, "0") == strlen(d);
232 }
233 } else if (strncasecmp(rest, "Expires=", 8) == 0) {
234 time_t when = cwist_http_parse_date(trim(rest + 8));
235 expires_past = when != (time_t)-1 && when <= time(NULL);
236 }
237 rest = strtok_r(NULL, ";", &saveptr);
238 }
239 /* No (or a relative) Path: the request's default-path (RFC 6265
240 * section 5.2.4). */
241 if (!path || path[0] != '/') {
242 cwist_free(path);
243 path = cookie_default_path(req_path);
244 if (!path) {
245 cwist_free(name);
246 cwist_free(value);
247 cwist_free(buf);
248 continue;
249 }
250 }
251 /* Max-Age wins over Expires (RFC 6265 section 5.3, step 3). */
252 if (has_max_age ? expired : expires_past) {
253 cookie_jar_remove(client, name, path);
254 cwist_free(name);
255 cwist_free(value);
256 cwist_free(path);
257 cwist_free(buf);
258 continue;
259 }
260 cwist_test_client_cookie *c = find_cookie_at(client->cookies, name, path);
261 if (c) {
262 cwist_free(c->value);
263 c->value = value;
264 cwist_free(path);
265 cwist_free(name);
266 } else {
267 c = (cwist_test_client_cookie *)cwist_alloc(sizeof(*c));
268 if (c) {
269 c->name = name;
270 c->value = value;
271 c->path = path;
272 c->next = client->cookies;
273 client->cookies = c;
274 } else {
275 cwist_free(name);
276 cwist_free(value);
277 cwist_free(path);
278 }
279 }
280 cwist_free(buf);
281 }
282}
