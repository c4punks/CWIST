40static bool peer_ip(int fd, char ip[INET6_ADDRSTRLEN]) {
41 struct sockaddr_storage peer = {0};
42 socklen_t size = sizeof(peer);
43 if (getpeername(fd, (struct sockaddr *)&peer, &size) != 0) return false;
44 if (peer.ss_family == AF_INET && size >= sizeof(struct sockaddr_in)) {
45 return inet_ntop(AF_INET, &((struct sockaddr_in *)&peer)->sin_addr, ip, INET6_ADDRSTRLEN) !=
46 NULL;
47 }
48 if (peer.ss_family == AF_INET6 && size >= sizeof(struct sockaddr_in6)) {
49 struct in6_addr *addr = &((struct sockaddr_in6 *)&peer)->sin6_addr;
50 if (IN6_IS_ADDR_V4MAPPED(addr)) {
51 return inet_ntop(AF_INET, (const unsigned char *)addr + 12, ip, INET6_ADDRSTRLEN) !=
52 NULL;
53 }
54 return inet_ntop(AF_INET6, addr, ip, INET6_ADDRSTRLEN) != NULL;
55 }
56 return false;
57}

77static void redis_rate_handler(size_t slot, cwist_http_request *req, cwist_http_response *res,
78 cwist_handler_func next) {
79 const redis_rate_cfg *cfg = &rate_cfgs[slot];
80 char ip[INET6_ADDRSTRLEN];
81 if (!peer_ip(req->client_fd, ip)) {
82 backend_failure(cfg, req, res, next, -ENOTCONN);
83 return;
84 }
85 char key[REDIS_RATE_PREFIX_MAX + 1 + INET6_ADDRSTRLEN];
86 snprintf(key, sizeof(key), "%s%s", cfg->prefix, ip);
87 const void *args[] = {"EVAL", rate_script, "1", key};
88 size_t lengths[] = {4, sizeof(rate_script) - 1, 1, strlen(key)};
89 cwist_redis_reply_t *reply = NULL;
90 cwist_error_t err = cwist_redis_pool_command_argv_reply_timeout(cfg->pool, 4, args, lengths,
91 cfg->timeout_ms, &reply);
92 if (!cwist_error_is_ok(&err) || !reply || reply->type != ':' || reply->integer <= 0) {
93 int code = cwist_error_is_ok(&err) ? -EPROTO : err.error.err_i16;
94 cwist_error_dispose(&err);
95 cwist_redis_reply_free(reply);
96 backend_failure(cfg, req, res, next, code);
97 return;
98 }
99 bool allowed = reply->integer <= cfg->rpm;
100 cwist_redis_reply_free(reply);
101 if (allowed) {
102 next(req, res);
103 return;
104 }
105 res->status_code = 429;
106 cwist_sstring_assign(res->body, "Too Many Requests");
107 cwist_http_header_add(&res->headers, "Retry-After", "60");
108}

129cwist_middleware_func cwist_mw_rate_limit_ip_redis(const cwist_rate_limit_redis_cfg_t *cfg) {
130 if (!cfg || !cfg->pool || cfg->requests_per_minute <= 0 || cfg->timeout_ms > 60000) return NULL;
131 const char *prefix = cfg->key_prefix ? cfg->key_prefix : "cwist:rl:";
132 if (strnlen(prefix, REDIS_RATE_PREFIX_MAX + 1) > REDIS_RATE_PREFIX_MAX) return NULL;
133 unsigned timeout = cfg->timeout_ms ? cfg->timeout_ms : 50;
134 pthread_mutex_lock(&rate_cfg_mutex);
135 for (size_t i = 0; i < rate_cfg_count; ++i) {
136 const redis_rate_cfg *old = &rate_cfgs[i];
137 if (old->pool == cfg->pool && old->rpm == cfg->requests_per_minute &&
138 old->timeout_ms == timeout && old->fail_closed == cfg->fail_closed &&
139 strcmp(old->prefix, prefix) == 0) {
140 pthread_mutex_unlock(&rate_cfg_mutex);
141 return wrappers[i];
142 }
143 }
144 if (rate_cfg_count == REDIS_RATE_SLOTS) {
145 pthread_mutex_unlock(&rate_cfg_mutex);
146 return NULL;
147 }
148 size_t slot = rate_cfg_count++;
149 rate_cfgs[slot] = (redis_rate_cfg){.pool = cfg->pool,
150 .rpm = cfg->requests_per_minute,
151 .timeout_ms = timeout,
152 .fail_closed = cfg->fail_closed};
153 memcpy(rate_cfgs[slot].prefix, prefix, strlen(prefix) + 1);
154 pthread_mutex_unlock(&rate_cfg_mutex);
155 return wrappers[slot];
156}
